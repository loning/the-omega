\begin{proposition}[Puiseux fine expansion at zero-temperature endpoint: three-term closed form for sub-growth branch]\label{prop:real-input-40-arity-charge-subbranch-puiseux}
Adopting the notation of Theorem \ref{thm:real-input-40-arity-charge-det-closed}, besides the maximal positive real root branch \(\lambda(q)=\rho(M(q))\) (whose zero-temperature expansion is in Proposition \ref{prop:real-input-40-arity-charge-zero-temp-expansion}), the algebraic equation \(P_7(\Lambda,q)=0\) admits another positive real growth branch \(\lambda_{\mathrm{sub}}(q)\).
As \(q\to+\infty\), this branch has the Puiseux asymptotic expansion
\[
\boxed{\;
\lambda_{\mathrm{sub}}(q)=\sqrt{2q}+\frac{\sqrt2}{4}\,q^{-1/2}-\frac78\,q^{-1}+O(q^{-3/2})\; }.
\]
\end{proposition}
\begin{proof}
By Theorem \ref{thm:real-input-40-arity-charge-det-closed}, \(\lambda_{\mathrm{sub}}(q)\) is a positive real algebraic branch of \(P_7(\Lambda,q)=0\).
Setting \(q=t^{-2}\) (\(t\downarrow 0\)) and making the Puiseux ansatz
\[
\lambda_{\mathrm{sub}}(q)=\frac{\sqrt2}{t}+b_1t+b_2t^2+O(t^3).
\]
Substituting into \(P_7(\lambda_{\mathrm{sub}}(t^{-2}),t^{-2})=0\) and successively comparing coefficients in powers of \(t\), we obtain
\(
b_1=\frac{\sqrt2}{4}
\)
and
\(
b_2=-\frac78
\),
thus yielding the shown expansion. This completes the proof.
\end{proof}

\begin{proposition}[Analytic jump slope at zero-charge endpoint: closed algebraic expression for \texorpdfstring{$\lambda'(0)$}{lambda'(0)}]\label{prop:real-input-40-arity-charge-derivative-q0}
Adopting the notation of Theorem \ref{thm:real-input-40-arity-charge-det-closed}, let \(\kappa>1\) be the unique real root of \(\kappa^4-\kappa^3-1=0\) (same as Proposition \ref{prop:real-input-40-arity-charge-zero-zeta}). Then \(\lambda(q)\) is analytic in a neighborhood of \(q=0\) and satisfies
\[
\lambda(0)=\kappa,\qquad
\lambda(q)=\kappa+\lambda'(0)\,q+O(q^2),
\]
where the jump slope has an explicit closed form
\[
\boxed{\;
\lambda'(0)=\frac{2\kappa^3+\kappa^2+8\kappa-2}{\kappa(4\kappa-3)}\; }.
\]
\end{proposition}
\begin{proof}
From the explicit form of \(P_7\) in Theorem \ref{thm:real-input-40-arity-charge-det-closed}, we directly factorize
\[
P_7(\Lambda,0)=\Lambda^2(\Lambda-1)(\Lambda^4-\Lambda^3-1),
\]
thus \(P_7(\kappa,0)=0\) and \(\kappa\) is a simple root. By the implicit function theorem, there exists an analytic branch \(\lambda(q)\) in a neighborhood of \(q=0\) satisfying \(\lambda(0)=\kappa\) and \(P_7(\lambda(q),q)=0\).
Taking the implicit derivative with respect to \(q\) and evaluating at \(q=0\) yields
\[
\lambda'(0)=-\frac{\partial_q P_7(\kappa,0)}{\partial_\Lambda P_7(\kappa,0)}.
\]
Substituting the explicit coefficients of \(P_7\) and simplifying using \(\kappa^4-\kappa^3-1=0\) yields the shown rational closed form. This completes the proof.
\end{proof}
