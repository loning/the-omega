\paragraph{Time Image and Discretization under Finite-State Realization}

\begin{proposition}[Time Image under Finite-State Realization: Discrete Quantization vs.\ Densification Dichotomy]\label{prop:xi-finite-state-time-image-discrete-or-dense}
Let the visible groupoid \(\mathcal G\) possess a finitely generated directed realization (finite states, finite labeled edge set \(E\)), and let the additive time cocycle \(\widetilde{\ell}:\mathcal G\to\mathbb R\) satisfy \(\widetilde{\ell}(gh)=\widetilde{\ell}(g)+\widetilde{\ell}(h)\).
If the value set on generating edges \(\{\widetilde{\ell}(e):e\in E\}\) is finite, then
\[
\widetilde{\ell}(\mathcal G)\subseteq
H:=\Bigl\langle \widetilde{\ell}(e):e\in E\Bigr\rangle_{\mathbb Z}\ \le\ (\mathbb R,+).
\]
Moreover, \(H\) is finitely generated as a \(\mathbb Z\)-module and thus satisfies the dichotomy:
\[
H=h\mathbb Z\ \text{(discrete)}\qquad\text{or}\qquad \overline{H}=\mathbb R\ \text{(dense)}.
\]
In the discrete case, \(h\) can be taken as the smallest positive element of \(H\cap(0,\infty)\) (equivalently, \(\inf(H\cap(0,\infty))\)), and \(\widetilde{\ell}(\mathcal G)\subset h\mathbb Z\).
Furthermore, \(\widetilde{\ell}(\mathcal G)\) is at most countable, so any truly continuous time image (uncountable) can only be obtained in the limit as the number of states tends to infinity.
\end{proposition}

\begin{proof}
Since \(\mathcal G\) is generated by a finite edge set, \(\widetilde{\ell}(g)\) for any \(g\in\mathcal G\) is an integer linear combination of the generating edge values, hence \(\widetilde{\ell}(\mathcal G)\subseteq H\).
Any finitely generated \(\mathbb Z\)-subgroup \(H\le\mathbb R\) either contains a smallest positive element (and hence must be the cyclic group \(h\mathbb Z\) generated by that element) or is dense in \(\mathbb R\); this is the standard classification of additive subgroups.
Finally, the finitely generated group \(H\cong \mathbb Z^k\) is countable, so \(\widetilde{\ell}(\mathcal G)\) is countable.
\end{proof}

\paragraph{Euler Product, Holonomy, and Cohomological Reduction of RH}

\begin{theorem}[Time Euler Decomposition Theorem: Zero Mixed Curvature Forces a Prime Element Product Decomposition of the Partition Function]\label{thm:xi-time-euler-product-under-zero-mixed-curvature}
Let \((\mathcal M_\times,\circ_\times)\subset \mathrm{Mor}(\mathbf{PW})\) be the multiplicative composition monoid, and suppose there exists a time cocycle \(\widetilde{\ell}\) strictly additive with respect to \(\circ_\times\).
Let \(\mathcal P\subset\mathcal M_\times\) be the set of \(\circ_\times\)-irreducible elements (primes).
If the following conditions hold:
\begin{enumerate}
  \item \(\mathcal M_\times\) is commutative, left-cancellative, and is the free commutative monoid on the primes (equivalently, every element has a unique prime decomposition, counted with multiplicity);
  \item the mixed curvature of additive-multiplicative alternating squares is identically zero (Theorem \ref{thm:xi-mixed-curvature-zero-criterion}), so that time has no central compensation defect between the additive and multiplicative layers;
\end{enumerate}
then for any \(\lambda>0\) in the domain of absolute convergence, the multiplicative partition function
\[
Z_\times(\lambda):=\sum_{w\in\mathcal M_\times}e^{-\lambda \widetilde{\ell}(w)}
\]
admits the Euler product decomposition
\[
Z_\times(\lambda)=\prod_{p\in\mathcal P}\bigl(1-e^{-\lambda \widetilde{\ell}(p)}\bigr)^{-1}.
\]
Conversely, if \(Z_\times\) admits the above Euler product on some open interval \(\lambda\in(\lambda_0,\infty)\) and \(\mathcal P\) generates \(\mathcal M_\times\), then the mixed curvature must be zero (equivalently, additive and multiplicative updates commute in the time sense without requiring additional central compensation).
\end{theorem}

\begin{proof}
Under unique prime decomposition and additivity,
\(
\widetilde{\ell}\bigl(\prod_p p^{n_p}\bigr)=\sum_p n_p\widetilde{\ell}(p)
\).
For \(\lambda\) in the domain of absolute convergence, rewriting the sum according to prime decomposition and exchanging summation and product yields
\[
Z_\times(\lambda)
=\prod_{p\in\mathcal P}\sum_{n\ge 0}e^{-\lambda n\widetilde{\ell}(p)}
=\prod_{p\in\mathcal P}(1-e^{-\lambda\widetilde{\ell}(p)})^{-1}.
\]
In the converse direction, the Euler product holding on an open interval forces the coefficients at all orders to satisfy the multiplicative-type generating structure independent of primes, compatible with the path independence of "alternating squares having no defect"; if the mixed curvature were nonzero, the additive/multiplicative alternation would introduce central compensation terms not absorbable by the boundary, destroying this multiplicative decomposition, so \(K\equiv 0\) is necessary.
\end{proof}

\begin{theorem}[Cohomological Reduction of RH via Time Unification: Deviation from the Critical Line Is Equivalent to Nontrivial Time Holonomy]\label{thm:xi-rh-cohomological-reduction-holonomy}
Assume there exists a completion representation such that the classical completed function \(\Xi(s)\) can be written as the restriction of a time-weighted determinant family:
\[
\Xi(s)=\det\!\bigl(I-e^{-s}\mathcal L_{\theta(s)}\bigr),
\qquad
\theta(s):=\Re(s)-\tfrac12,
\]
where \(\theta\mapsto\mathcal L_{\theta}\) is a family of transfer operators induced by auditable local evolution and the time cocycle, satisfying the time-reversal conjugacy
\(
\mathcal L_{\theta}\sim \mathcal L_{-\theta}
\)
corresponding to \(\Xi(s)=\Xi(1-s)\).
If furthermore:
\begin{enumerate}
  \item Strictly positive time: the time sum of any nontrivial closed loop is nonzero;
  \item The holonomy group \(\mathfrak H\) of the scale--time connection is trivial (Theorem \ref{thm:xi-scale-time-connection-holonomy-criterion});
  \item The critical slice pressure criterion holds (Theorem \ref{thm:xi-critical-slice-pressure-zero-drift-lyapunov}), so that \(\theta=0\) is the unique "double-zero" slice;
\end{enumerate}
then for any \(\theta\neq 0\), necessarily \(\rho(\mathcal L_{\theta})\neq 1\), and hence \(\det(I-e^{-s}\mathcal L_{\theta})\neq 0\) cannot occur on a slice with \(\theta\neq 0\).
Therefore all nontrivial zeros must satisfy \(\theta(s)=0\), i.e., \(\Re(s)=\tfrac12\).
Under the above representation hypotheses, the Riemann Hypothesis can be equivalently restated as the critical slice uniqueness forced by "vanishing scale--time holonomy."
\end{theorem}

\begin{proof}
By Theorem \ref{thm:xi-critical-slice-pressure-zero-drift-lyapunov}, under the auditable micro-reversibility and time-reversal framework, \(\theta=0\) is the unique slice simultaneously satisfying zero drift and zero Lyapunov exponent.
Trivial holonomy eliminates the freedom to introduce additional central compensation in cross-scale assembly, so that criticality can no longer be maintained along \(\theta\neq 0\) directions.
Therefore, for \(\theta\neq 0\), necessarily \(\rho(\mathcal L_{\theta})\neq 1\), contradicting the determinant zero occurring under non-critical spectral conditions; the conclusion follows.
\end{proof}

\begin{proposition}[Finite Decidability Criterion for RH: It Suffices to Check the Vanishing of the Integer Rank of a Finitely Generated Time Curvature Matrix]\label{prop:xi-rh-finite-curvature-matrix-criterion}
Under the representation hypotheses of Theorem \ref{thm:xi-rh-cohomological-reduction-holonomy}, if the protocol comes from a finitely generated convergent rewriting system, then the holonomy group \(\mathfrak H\) is generated by the defects of finitely many "scale--time critical squares."
Recording these defects as integer vectors in some central coordinate system (\(\mathbb Z^d\)) and arranging them into a matrix \(B\in\mathbb Z^{r\times d}\) (rows corresponding to critical squares, columns to central basis vectors), one obtains
\[
\mathfrak H\cong \mathrm{im}(B)\subset\mathbb Z^d,
\qquad
\mathrm{rank}(\mathfrak H)=\mathrm{rank}_{\mathbb Z}(B).
\]
In particular, in the scalar defect case \(d=1\), \(\mathfrak H=\{0\}\) if and only if \(B=0\).
Therefore, within this framework, "vanishing holonomy" reduces to a zero/rank test on a finite integer matrix, independent of gauge choices.
\end{proposition}

\begin{proof}
All higher-order overlaps of the convergent rewriting system are generated by finitely many critical pairs; holonomy, as the defect accumulation of commutative squares, is therefore generated by the defects of finitely many critical squares.
Coordinatizing the generators in a \(\mathbb Z^d\) basis yields the matrix description; the rank conclusion follows from \(\mathbb Z\)-linear algebra.
\end{proof}

\paragraph{Positive Time Gauge, Smith Reduction, and PBW Commutativity}

\begin{proposition}[Cohomological Positive Cone Criterion for Positive Time: Existence of an All-Edge-Positive Gauge]\label{prop:xi-positive-time-cone-criterion}
Let \(\Gamma=(V,E)\) be a finite strongly connected directed graph, \(t:E\to\mathbb R\) a time weight, and consider the gauge transformation \(t\mapsto t+\partial b\) (\(b:V\to\mathbb R\)).
Then there exists \(b\) such that \(t+\partial b\) is strictly positive on all edges if and only if for every directed cycle \(\gamma\),
\[
\sum_{e\in\gamma} t(e)>0.
\]
Equivalently, \([t]\in H^1(\Gamma;\mathbb R)\) lies in the interior of the positive cone defined by all directed cycles
\[
\mathcal C_+:=\Bigl\{[c]\in H^1:\ \langle[c],[\gamma]\rangle>0\ \text{for all nontrivial cycles }\gamma\Bigr\}.
\]
\end{proposition}

\begin{proof}
For any cycle \(\gamma\), \(\sum_{e\in\gamma}\partial b(e)=0\), so the cycle sum is a cohomological invariant.
If there exists \(b\) making all edges positive, then any cycle sum must be positive, establishing necessity.
The converse is the feasibility criterion for difference constraints: the inequality \(t(e)+b(r(e))-b(s(e))>0\) is equivalent to a strict difference constraint system, which is feasible if and only if no cycle has total weight \(\le 0\); in the present setting, this means all cycle sums are strictly positive.
\end{proof}

\begin{proposition}[Smith Reduction of the Time Defect Group: The Degrees of Freedom and Torsion Structure of the Central Extension Are Fully Computable]\label{prop:xi-time-defect-smith-reduction}
Following the matrix description of Proposition \ref{prop:xi-rh-finite-curvature-matrix-criterion}, let the defect matrix be \(B\in\mathbb Z^{r\times d}\) and let \(\mathfrak H:=\mathrm{im}(B)\subset\mathbb Z^d\).
Then there exist invertible integer matrices \(U\in\mathrm{GL}_r(\mathbb Z)\), \(V\in\mathrm{GL}_d(\mathbb Z)\) such that
\[
UBV=\mathrm{diag}(s_1,\dots,s_k,0,\dots,0),
\qquad
s_i\mid s_{i+1}.
\]
Hence the defect quotient has the standard decomposition
\[
\mathbb Z^d/\mathfrak H
\ \cong\
\bigoplus_{i=1}^k\mathbb Z/s_i\mathbb Z\ \oplus\ \mathbb Z^{d-k}.
\]
Here the free rank \(d-k\) gives the continuous parameter dimension of the minimal central extension, and the torsion factors \(s_i\) give the non-cancellable discrete "time torsion."
\end{proposition}

\begin{proof}
By the Smith normal form theorem, any integer matrix can be reduced to the above diagonal form under left and right invertible integer transformations; the diagonal entries give the structural invariants of the image subgroup, from which the quotient group decomposition follows.
\end{proof}

\begin{theorem}[PBW-Type Theorem for the Time Filtration: Zero Mixed Curvature If and Only If the Associated Graded Algebra Is Commutative]\label{thm:xi-time-filtration-pbw-graded-commutative}
Let \(\mathbb k[\mathbf{PW}]\) be the protocol category algebra (\(\mathbb k\) a field of characteristic zero), with the filtration defined by the time length \(\widetilde{\ell}\):
\[
F_t:=\mathrm{span}\{w:\widetilde{\ell}(w)\le t\}.
\]
If \(\widetilde{\ell}\) is strictly additive on allowable compositions, then \(\mathrm{gr}_{\widetilde{\ell}}\mathbb k[\mathbf{PW}]\) is a graded algebra.
Suppose the system has two types of generators (corresponding to the additive and multiplicative layers) forming alternating squares. Then
\[
\mathrm{gr}_{\widetilde{\ell}}\mathbb k[\mathbf{PW}]\ \text{is a commutative algebra}
\quad\Longleftrightarrow\quad
K\equiv 0,
\]
where \(K\) is the mixed curvature of Theorem \ref{thm:xi-mixed-curvature-zero-criterion}.
\end{theorem}

\begin{proof}
If \(K\equiv 0\), the two paths of the alternating square have no defect in the time leading term, so the commutation relation holds at the leading term level of the associated graded, rendering the graded algebra commutative.
Conversely, if the associated graded is commutative, then all alternating generators commute at the leading term level, so the time defect of every commutative square must be zero, i.e., \(K\equiv 0\).
\end{proof}

\paragraph{Time Boundary and Statistical Degeneracy}

\begin{theorem}[Patterson--Sullivan Density of the Time Boundary: Equivalence between KMS Equilibrium States and Boundary Conformal Measures]\label{thm:xi-time-boundary-patterson-sullivan-kms}
Suppose the time-metric Cayley graph satisfies a linear isoperimetric bound (the hyperbolicity condition of Theorem \ref{thm:xi-audit-dehn-function-time-geometry}), and the growth exponent \(\kappa\) exists.
For \(\lambda>\kappa\), define the Poincar\'e series
\[
\mathcal P_\lambda(x,y):=\sum_{g:x\to y}e^{-\lambda \widetilde{\ell}(g)}.
\]
Then there exists a unique (up to scale) \(\lambda\)-conformal boundary density family \(\{\nu_x^{(\lambda)}\}\) supported on the Gromov boundary \(\partial\), satisfying the Radon--Nikodym relation given by the Busemann--time function.
Moreover, there is a natural one-to-one correspondence between the extremal states induced by this boundary density and the KMS equilibrium states defined by the Gibbs weight \(e^{-\lambda\widetilde{\ell}}\).
\end{theorem}

\begin{proof}
Under the hyperbolic metric, the convergence domain of the Poincar\'e series and the existence of conformal densities are given by the Patterson--Sullivan construction, with the growth exponent as the critical parameter.
The correspondence between KMS states and conformal densities is the standard correspondence arising from the conformality of the dynamical system (modular flow/time flow) induced by the time cocycle on the boundary; the extremal decomposition is consistent with the extremality of the boundary measures.
\end{proof}

\begin{proposition}[Defect Rank Controls Large Deviation Degeneracy: \(\mathrm{rank}(\mathfrak H)\) Equals the Affine Flat Dimension of the Rate Function]\label{prop:xi-defect-rank-controls-ldp-flatness}
Let the pressure function \(P(s)=\log\rho(A_s)\) be differentiable and strictly convex for \(s>s_c\), where \(A_s\) is the time-weighted transfer operator.
Under gauge perturbation \(\widetilde{\ell}\mapsto \widetilde{\ell}+\partial b\), \(P\) is invariant; for more general central defect directions (given by the holonomy defect group \(\mathfrak H\)), the pressure remains invariant along certain linear directions.
Let the large deviation rate function \(I\) be the Legendre dual of \(P\). Then
\[
\dim\bigl(\text{maximal affine flat subset of \(I\)}\bigr)=\mathrm{rank}(\mathfrak H).
\]
In particular, \(\mathfrak H=\{0\}\) if and only if \(I\) is strictly convex; once nontrivial defect degrees of freedom exist, the rate function must exhibit flat "phase coexistence" directions of corresponding dimension.
\end{proposition}

\begin{proof}
The invariance of the pressure along defect directions means that \(P\) has flat directions on the corresponding linear subspace; the Legendre dual maps flat directions to affine flat faces of the rate function.
The dimension is given by the rank of this invariant subspace (equivalently, the defect degrees of freedom).
\end{proof}

\paragraph{Finite Decidability via Cycle Sums: Complete Reconstruction of the Cohomology Class}

\begin{proposition}[Cycle Sums Determine the Time Cohomology Class: Equality of Time Sums on a Finite Basis of Cycles Implies Coboundary Equivalence]\label{prop:xi-cycle-sums-determine-time-cohomology}
Let \(\Gamma=(V,E)\) be a strongly connected finite directed graph, and let \(t,t':E\to h\mathbb Z\) be two time weights, viewed as \(1\)-cocycles.
Then
\[
t-t'=\partial b\ \text{for some }b:V\to h\mathbb Z
\quad\Longleftrightarrow\quad
\sum_{e\in\gamma}t(e)=\sum_{e\in\gamma}t'(e)\ \text{for all directed cycles }\gamma.
\]
Moreover, the verification "for all cycles" can be reduced to checking on any finite fundamental cycle family generating \(H_1(\Gamma;\mathbb Z)\): take any spanning tree \(T\), and the fundamental cycles \(\gamma_e\) formed by each chord edge \(e\in E\setminus T\) constitute a basis of the integer cycle space; checking equality of time sums on finitely many \(\{\gamma_e\}\) suffices to deduce cohomological equivalence.
\end{proposition}

\begin{proof}
Necessity follows directly from \(\sum\partial b=0\) on cycles.
The reverse direction is the standard characterization of \(1\)-cohomology on finite graphs: if two cocycles have the same pairing on a cycle basis, their difference is zero in \(H^1\), hence is a coboundary.
\end{proof}
