\section{Modular scale exchange, Morita equivalence, and Fourier exchange (notes)}
\label{app:morita_fourier_exchange}

\AuditTag This appendix records standard symmetry/equivalence structures that strengthen the ``equivalence semantics'' viewpoint (Section~\ref{app:equivalence_semantics}).
It introduces no new axioms beyond tick and CAP and is not used as a premise in theorem-level folding proofs.

\subsection{Modular inversion as a scale-exchange template}
\label{subsec:morita_modular_inversion}

Consider the upper half-plane $\mathbb{H}=\{\tau=x+\iu y:\ y>0\}$ and the modular generators
\[
T:\tau\mapsto \tau+1,\qquad
S:\tau\mapsto -\frac{1}{\tau}.
\]
Writing $\tau=x+\iu y$, one has
\[
S(\tau)=-\frac{x-\iu y}{x^2+y^2},
\qquad
\ImPart(S(\tau))=\frac{y}{x^2+y^2}.
\]
\InterfaceTag Thus, away from large $|x|$, the transformation $S$ exchanges large and small imaginary parts (height) at the level of scale magnitude.
On the modular curve $X(1)$, the cusp orbit is unique, so $0$ and $\infty$ represent the same cusp in the quotient; the involution $S$ provides a canonical endpoint exchange on the quotient.

\subsection{Morita equivalence of rotation algebras and the \texorpdfstring{$\mathrm{SL}_2(\ZZ)$}{SL2(Z)} action}
\label{subsec:morita_rotation_algebra}

Let $A_\alpha$ be the (irrational) rotation algebra generated by unitaries $U,V$ with
\[
UV=\e^{2\pi\iu\alpha}VU,\qquad \alpha\in\RR\setminus\QQ.
\]
There is a canonical $\mathrm{SL}_2(\ZZ)$ action on the slope parameter
\[
\alpha'=\frac{a\alpha+b}{c\alpha+d},
\qquad
\gamma=\begin{pmatrix}a&b\\c&d\end{pmatrix}\in \mathrm{SL}_2(\ZZ).
\]

\begin{theorem}[Morita equivalence classification for noncommutative tori (standard)]
\label{thm:morita_classification_rotation}
\MathTag For irrational parameters $\alpha,\beta\in\RR\setminus\QQ$, the rotation algebras $A_\alpha$ and $A_\beta$ are (strongly) Morita equivalent if and only if there exists $\gamma=\begin{pmatrix}a&b\\c&d\end{pmatrix}\in \mathrm{SL}_2(\ZZ)$ such that
\[
\beta=\frac{a\alpha+b}{c\alpha+d}.
\]
\end{theorem}
\begin{remark}
See \cite{Rieffel1981,Rieffel1988,Connes1994} for proofs and further structure (projective modules and $K$-theory invariants).
\end{remark}

\AuditTag In the scan algebra language of this paper, the parameter $\alpha$ controls the commutation phase between shift and multiplication (Definition~\ref{def:weyl_pair}).
Morita equivalence provides a hard mathematical mechanism for treating different $\alpha$ as ``the same geometry'' once the equivalence class is declared.

\subsection{Fourier exchange: swapping scan shift and phase multiplication}
\label{subsec:morita_fourier_exchange}

We record a concrete scan--readout exchange for the canonical Weyl-pair model.

\begin{proposition}[Fourier exchange for the covariant Weyl pair]
\label{prop:fourier_exchange_weyl_pair}
\MathTag Let $\mathcal{H}=L^2(\RR/\ZZ)$ and define, for fixed $\alpha\in\RR$,
\[
(U\psi)(x)=\psi(x+\alpha),\qquad (V\psi)(x)=\e^{2\pi\iu x}\psi(x).
\]
Let $\mathcal{F}:\mathcal{H}\to \ell^2(\ZZ)$ be the Fourier transform
\[
(\mathcal{F}\psi)(k)=\int_0^1 \psi(x)\,\e^{-2\pi\iu k x}\,\dd x.
\]
Then on $\ell^2(\ZZ)$ one has
\[
\mathcal{F}U\mathcal{F}^{-1}:\widehat\psi(k)\mapsto \e^{2\pi\iu k\alpha}\widehat\psi(k),
\qquad
\mathcal{F}V\mathcal{F}^{-1}:\widehat\psi(k)\mapsto \widehat\psi(k-1).
\]
In particular, translation (scan shift) becomes phase multiplication in Fourier space, while phase multiplication becomes an index shift, realizing a concrete scan--readout exchange.
\end{proposition}
\begin{proof}
For the first identity,
\[
(\mathcal{F}U\psi)(k)=\int_0^1 \psi(x+\alpha)\e^{-2\pi\iu k x}\,\dd x
=\e^{2\pi\iu k\alpha}\int_0^1 \psi(u)\e^{-2\pi\iu k u}\,\dd u
=\e^{2\pi\iu k\alpha}(\mathcal{F}\psi)(k),
\]
using the substitution $u=x+\alpha$ and periodicity on $\RR/\ZZ$.
For the second identity,
\[
(\mathcal{F}V\psi)(k)=\int_0^1 \e^{2\pi\iu x}\psi(x)\e^{-2\pi\iu k x}\,\dd x
=\int_0^1 \psi(x)\e^{-2\pi\iu (k-1)x}\,\dd x
=(\mathcal{F}\psi)(k-1).
\]
\end{proof}
