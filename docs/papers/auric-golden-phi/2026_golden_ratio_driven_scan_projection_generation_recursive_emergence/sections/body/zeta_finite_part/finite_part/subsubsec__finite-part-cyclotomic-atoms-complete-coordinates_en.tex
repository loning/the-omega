\subsubsection{Cyclotomic Atoms as Complete Coordinates of Non-Perron Spectrum: Divisor Lattice Inversion and Homotopy Additivity}\label{subsubsec:finite-part-cyclotomic-atoms-complete-coordinates}
In \S\ref{subsec:finite-part-cyclic-lift-cyclotomic-microscope}, the constant-layer anomaly \(\mathcal{E}_q(M)\) was decomposed into cyclotomic atoms \(\mathcal{C}_r(M)\) and shown to be invertible on the divisor lattice (Proposition \ref{prop:finite-part-cyclic-lift-cyclotomic-divisor-mobius}).
This subsection supplements two structural consequences: first, the atom sequence itself is a complete coordinate of the non-Perron normalized spectrum in an information-theoretic sense; second, the atom coordinates possess natural additivity and rigidity under spectral multiset unions and continuous deformations.

\begin{theorem}[Completeness of cyclotomic atom sequence]\label{thm:finite-part-cyclotomic-atoms-complete-coordinate}
Let \(M\in\ZZ_{\ge0}^{d\times d}\) be primitive, \(\rho=\rho(M)\), and denote normalized non-Perron spectral points \(\lambda_j=\mu_j/\rho\ (j=2,\dots,d)\).
Let \(\mathcal{C}_r(M)\ (r\ge2)\) be the cyclotomic atoms defined in Proposition \ref{prop:finite-part-cyclic-lift-cyclotomic-divisor-mobius}.
Then the atom sequence \(\bigl(\mathcal{C}_r(M)\bigr)_{r\ge2}\) uniquely determines the reduced spectral polynomial
\[
F_M(t)=\prod_{j=2}^{d}(1-\lambda_j t),
\]
thus uniquely determining the normalized non-Perron spectral multiset \(\{\lambda_j\}_{j=2}^{d}\) (with algebraic multiplicity).
More specifically: if another primitive matrix \(N\) satisfies
\(
\mathcal{C}_r(M)=\mathcal{C}_r(N)
\)
for all \(r\ge2\), then \(F_M\equiv F_N\), and thus their normalized non-Perron spectra coincide.
\end{theorem}
\begin{proof}
By Proposition \ref{prop:finite-part-cyclic-lift-cyclotomic-divisor-mobius}, \(\{\mathcal{C}_r\}_{r\ge2}\) uniquely determines \(\{\mathcal{E}_q\}_{q\ge1}\) (and conversely).
Moreover, by \(\mathcal{E}_q(M)=-\sum_{j=2}^{d}\log\!\bigl(\frac{1-\lambda_j^{q}}{1-\lambda_j}\bigr)\) and \(|\lambda_j|<1\),
\[
\lim_{q\to\infty}\mathcal{E}_q(M)=\sum_{j=2}^{d}\log(1-\lambda_j)=\log\det\nolimits'\!\left(I-\frac{M}{\rho}\right)=-\log C(M),
\]
so \(\{\mathcal{C}_r\}\) also uniquely determines \(C(M)\).
Thus, by the relation \(\Psi_q(M)=\mathcal{E}_q(M)+\log C(M)\) in Definition \ref{def:finite-part-cyclic-lift-psi}, all \(\{\Psi_q(M)\}_{q\ge1}\) are obtained.
Furthermore, by Proposition \ref{prop:finite-part-cyclic-lift-mobius-inversion}, the power sum sequence \(\{u_n(M)\}_{n\ge1}\) can be exactly recovered for each \(n\), where \(u_n(M)=\sum_{j=2}^{d}\lambda_j^{n}\).
Finally, Theorem \ref{thm:finite-part-cyclic-lift-spectrum-identifiability} (Prony--Hankel identifiability) shows that \(\{u_n(M)\}\) uniquely determines \(\{\lambda_j\}\) (with multiplicity), thus uniquely determining \(F_M\). \qed
\end{proof}

\begin{definition}[Spectral--information atom coordinates and their additivity]\label{def:finite-part-spectral-information-atoms}
Denote
\[
\mathbf{C}(M):=\bigl(\mathcal{C}_r(M)\bigr)_{r\ge2}
\]
as the \textbf{spectral--information atom coordinates} of \(M\).
Since \(\mathcal{C}_r(M)\) is a pointwise sum over normalized non-Perron spectral points (by the defining formula in Proposition \ref{prop:finite-part-cyclic-lift-cyclotomic-divisor-mobius}), it satisfies termwise additivity under spectral multiset unions: if two (formal) spectral multisets are united, the corresponding \(\mathbf{C}\) coordinates add termwise.
\end{definition}

\begin{theorem}[Homotopy rigidity and concatenation additivity]\label{thm:finite-part-cyclotomic-atoms-homotopy-additive}
Let \(M(t)\) be a family (parameterized by \(t\in[0,1]\)) of primitive matrices such that the normalized non-Perron spectral points remain strictly inside the unit disk throughout the deformation (so the analytic branch of \(\log\Phi_r(\lambda_j(t))\) can be chosen continuously along \(t\)).
Then:
\begin{enumerate}
  \item \(\mathbf{C}(M(t))\) is termwise continuous in coordinates;
  \item \textbf{(Endpoint rigidity)} If \(\mathbf{C}(M(0))=\mathbf{C}(M(1))\), then the normalized non-Perron spectral multisets of \(M(0)\) and \(M(1)\) coincide;
  \item \textbf{(Concatenation additivity)} Under spectral multiset union (formal direct sum), \(\mathbf{C}\) adds termwise.
\end{enumerate}
\end{theorem}
\begin{proof}
Termwise continuity follows from the analyticity of \(\Phi_r\) and the chosen logarithm branch in the unit disk.
Endpoint rigidity is a direct corollary of Theorem \ref{thm:finite-part-cyclotomic-atoms-complete-coordinate}.
Concatenation additivity follows immediately from the pointwise summation structure in Definition \ref{def:finite-part-spectral-information-atoms}. \qed
\end{proof}
