\section{Chinese-bridge refinements}\label{app:cn-bridge}

\subsection{Puiseux-slope cubic and explicit quadratic involution}
\label{sec:cn-puiseux-u}

\begin{proposition}[Explicit Puiseux slope parameterization]\label{prop:puiseux-u-bridge}
For a nontrivial finite branch value \(y_\star\) (\(c(y_\star)=0\)), let \(\lambda_\star\) be the corresponding double root of \(\Pi(\lambda,y_\star)=0\) and let
\[
u:=\kappa_\star^2
\]
from Theorem~\ref{thm:puiseux-amplitude}, so that
\[
\lambda(y)=\lambda_\star\pm\kappa_\star\sqrt{y-y_\star}+O(y-y_\star).
\]
Then the slope parameter \(u\) is given by
\begin{equation}\label{eq:cn-kappa}
u=\frac{2}{3}\frac{3\lambda_\star^{2}-1}{\lambda_\star^{2}-1}.
\end{equation}
Hence \(\lambda_\star\in\QQ(u)\) and, equivalently,
\begin{equation}\label{eq:cn-lambda-u}
\lambda_\star=\frac{3(2u-1)}{4(3u-2)}.
\end{equation}
Eliminating \(\lambda_\star\) between \eqref{eq:cn-lambda-u} and the ramification relation \(16\lambda_\star^3-9\lambda_\star^2+1=0\) yields
\[
g(u):=324u^{3}-540u^{2}+333u-74=0.
\]
Using \eqref{eq:double-root-cubic} to eliminate \(\lambda_\star\) gives
\begin{equation}\label{eq:cn-y-u}
y_\star=-\frac{1091}{256}+\frac{675}{64}u-\frac{567}{64}u^2,
\qquad
256y_\star^3+411y_\star^2+165y_\star+32
=\frac{3^4\cdot 7^3}{2^{14}}\,g(u)\,g(\iota(u)),
\end{equation}
where \(\iota(u):=\frac{25}{21}-u\). Equivalently, \(y_\star\) is invariant under the deck involution \(u\mapsto\iota(u)\), and the branch cubic pulls back as a quadratic product in the \(u\)-line:
\[
c\!\bigl(y(u)\bigr)=\frac{3^4\cdot 7^3}{2^{14}}\,g(u)\,g(\iota(u)).
\]
\end{proposition}

\begin{proof}
Equation~\eqref{eq:cn-kappa} is Theorem~\ref{thm:puiseux-amplitude} for nontrivial branch points after rewriting \(\partial_y\Pi\), \(\partial_{\lambda\lambda}\Pi\) in the coordinates \(y=\lambda^2+\xi\). Solving \eqref{eq:cn-kappa} for \(\lambda_\star\) gives \eqref{eq:cn-lambda-u}.
Substituting \eqref{eq:cn-lambda-u} into \(16\lambda_\star^3-9\lambda_\star^2+1=0\) gives the cubic \(g(u)=0\).
The formula \eqref{eq:cn-y-u} is obtained by substituting \(\lambda_\star\) into the linear closure \eqref{eq:double-root-cubic}; the quadratic pullback identity is then a direct resultant computation in \(u\), using the explicit expression \(c(y)=256y^3+411y^2+165y+32\).
\end{proof}

\begin{corollary}[Quadratic reconstruction of branch data]\label{cor:cn-branch-quadratic-reconstruction}
Retain the notation of Proposition~\ref{prop:puiseux-u-bridge}. Let
\[
\lambda(y)=\lambda_\star\pm\kappa_\star\sqrt{y-y_\star}+\beta_\star(y-y_\star)+O\!\left((y-y_\star)^{3/2}\right)
\]
be the local branch expansion at a nontrivial branch value \(y_\star\) (\(c(y_\star)=0\)).
In \(K:=\QQ(u)=\QQ(\lambda_\star)=\QQ(y_\star)\) the following identities hold:
\begin{equation}\label{eq:cn-branch-reconstruction}
16\lambda_\star+31=108u(1-u),\qquad
256y_\star+1091=108u(25-21u),\qquad
72\beta_\star+22=9u(10u-1).
\end{equation}
Equivalently,
\[
\lambda_\star=-\frac{31}{16}+\frac{27}{4}u-\frac{27}{4}u^2,\qquad
y_\star=-\frac{1091}{256}+\frac{675}{64}u-\frac{567}{64}u^2,\qquad
\beta_\star=-\frac{11}{36}-\frac{1}{8}u+\frac{5}{4}u^2.
\]
\end{corollary}

\begin{proof}
From \eqref{eq:cn-lambda-u}, \(\lambda_\star\in\QQ(u)\). The remaining identities follow by substituting into the expanded local parametrization and eliminating \(\lambda_\star\) in \(\QQ[u]/(324u^3-540u^2+333u-74)\).
\end{proof}

\begin{corollary}[Integral generators from branch coefficients]\label{cor:cn-branch-integrality}
With \(\beta_\star\) as above,
\[
419904\,\beta_\star^3+93312\,\beta_\star^2+72279\,\beta_\star-1112=0.
\]
Equivalently, for
\[
T:=162u,\qquad S:=52488\,\beta_\star,
\]
\(T\) and \(S\) are algebraic integers satisfying
\[
T^3-270T^2+26973T-971028=0,\qquad
S^3+11664S^2+474222519S-382943630016=0.
\]
\end{corollary}

\begin{proof}
Equation~\eqref{eq:cn-branch-reconstruction} gives \(\beta_\star\in\QQ(u)\). Eliminating \(u\) from \eqref{eq:cn-branch-reconstruction} and the cubic
\[
324u^3-540u^2+333u-74=0
\]
produces the cubic in \(\beta_\star\). The monic integer forms follow from clearing denominators in the definitions of \(T\) and \(S\).
\end{proof}

\subsection{Finite-field spectral point counting}
\label{sec:cn-finite-field}

\begin{proposition}[Finite-field point count for the spectral curve]\label{prop:cn-finite-field}
Let \(p\) be an odd prime \(p\neq 37\), and define
\[
C_p(\mathbb F_p):=\{(\lambda,y)\in\mathbb F_p^2:\Pi(\lambda,y)\equiv 0\},
\qquad
E_{0,p}(\mathbb F_p):=\{(x,v)\in\mathbb F_p^2:v^2+v=x^3-x\}\cup\{O\}.
\]
Then
\[
\#C_p(\mathbb F_p)=\#E_{0,p}(\mathbb F_p)-1.
\]
Equivalently, if \(a_p:=p+1-\#E_{0,p}(\mathbb F_p)\), then
\[
\#C_p(\mathbb F_p)=p-a_p,\qquad
a_p=-\sum_{x\in\mathbb F_p}\left(\frac{4x^3-4x+1}{p}\right),
\]
where \(\left(\frac{\cdot}{p}\right)\) is the Legendre symbol with \(\left(\frac{0}{p}\right)=0\).
\end{proposition}

\begin{proof}
For odd \(p\neq37\), the change of variables
\[
(x,v)=(\lambda,y-\lambda^2),\qquad (\lambda,y)=(x,v+x^2),
\]
identifies affine points of \(\Pi(\lambda,y)\equiv 0\) over \(\mathbb F_p\) with affine points of \(E_0: v^2+v=x^3-x\). The only missing point on \(E_0\) is \(O\), so \(\#C_p=\#E_{0,p}-1\).

For \(p\neq 2,37\), counting points on \(E_{0,p}\) by \(x\) gives
\[
\#E_{0,p}(\mathbb F_p)=1+\sum_{x\in\mathbb F_p}\left(1+\left(\frac{4x^3-4x+1}{p}\right)\right)
=p+1+\sum_{x\in\mathbb F_p}\left(\frac{4x^3-4x+1}{p}\right).
\]
Hence \(a_p:=p+1-\#E_{0,p}(\mathbb F_p)=-\sum_{x\in\mathbb F_p}\left(\frac{4x^3-4x+1}{p}\right)\), and the claim follows.
\end{proof}
