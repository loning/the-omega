\begin{theorem}[Defect-induced exclusion radius (horizon audit no-entry law)]\label{thm:dephys-defect-repulsion-radius}
Let \(F_{\zeta}\) be the disk completion function from Appendix \ref{app:unit-circle-phase-gate}, and \(D_{\zeta}(\varrho)\) the Jensen defect (Definition \ref{def:app-jensen-defect}).
For any \(0<\varrho<1\), define the exclusion radius
\[
r_{\ast}(\varrho):=\varrho\exp\!\bigl(-D_{\zeta}(\varrho)\bigr)\in(0,\varrho].
\]
Then \(F_{\zeta}\) has no zeros in the subdisk \(|w|<r_{\ast}(\varrho)\).
More quantitatively, if \(D_{\zeta}(\varrho)\le \varepsilon\), then \(r_{\ast}(\varrho)\ge \varrho e^{-\varepsilon}\).
Moreover, if there exists \(\varrho_n\uparrow 1\) such that \(r_{\ast}(\varrho_n)\uparrow 1\), then \(F_{\zeta}\) has no zeros in \(|w|<1\), and thus RH holds.
\end{theorem}

\begin{proof}
The first two assertions are given directly by Proposition \ref{prop:app-jensen-zero-repulsion-subdisk}.
The final RH certificate follows from Corollary \ref{cor:app-jensen-repulsion-radius-to-one}.\qed
\end{proof}

\begin{proposition}[Logarithmic radius derivative law: defect derivative equals in-disk zero count]\label{prop:dephys-jensen-defect-log-derivative}
Let \(N_{F_{\zeta}}(\varrho)\) be the zero count (with multiplicity) of \(F_{\zeta}\) in \(|w|<\varrho\):
\[
N_{F_{\zeta}}(\varrho)
:=\#\bigl\{a:\ F_{\zeta}(a)=0,\ |a|<\varrho\bigr\}.
\]
Then for almost every \(\varrho\in(0,1)\),
\[
\frac{\dd}{\dd(\log\varrho)}D_{\zeta}(\varrho)=N_{F_{\zeta}}(\varrho),
\qquad
D_{\zeta}'(\varrho)=\frac{1}{\varrho}\,N_{F_{\zeta}}(\varrho).
\]
Furthermore, \(D_{\zeta}\) is continuous and nondecreasing on \((0,1)\), and if \(D_{\zeta}(\varrho_0)=0\) for some \(\varrho_0\in(0,1)\), then \(D_{\zeta}(\varrho)=0\) for all \(0<\varrho\le \varrho_0\).
\end{proposition}

\begin{proof}
See Appendix Proposition \ref{prop:app-jensen-defect-log-derivative}.
\end{proof}

\begin{corollary}[Exclusion radius zero-count driving law]\label{cor:dephys-repulsion-radius-zero-count-driving}
In the notation of Proposition \ref{prop:dephys-jensen-defect-log-derivative}, let \(r_{\ast}(\varrho)=\varrho e^{-D_{\zeta}(\varrho)}\) as in Theorem \ref{thm:dephys-defect-repulsion-radius}.
Then for almost every \(\varrho\in(0,1)\),
\[
\frac{\dd}{\dd(\log\varrho)}\log r_{\ast}(\varrho)
=1-\frac{\dd}{\dd(\log\varrho)}D_{\zeta}(\varrho)
=1-N_{F_{\zeta}}(\varrho).
\]
Therefore, once an in-disk zero appears (i.e., for some \(\varrho\) we have \(N_{F_{\zeta}}(\varrho)\ge 1\)), the growth rate of \(\log r_{\ast}\) at all subsequent radius layers is no longer positive;
in particular, if there exists an in-disk zero \(a\) satisfying \(|a|<1\), then for all \(\varrho\in(|a|,1)\) we have \(N_{F_{\zeta}}(\varrho)\ge 1\), and thus \(r_{\ast}(\varrho)\) cannot grow strictly to \(1\) as \(\varrho\uparrow 1\).
\end{corollary}

\begin{proof}
From \(r_{\ast}(\varrho)=\varrho e^{-D_{\zeta}(\varrho)}\) we get \(\log r_{\ast}=\log\varrho-D_{\zeta}\); differentiating with respect to \(\log\varrho\) and substituting Proposition \ref{prop:dephys-jensen-defect-log-derivative} yields the result.
The last assertion follows directly from the monotonicity of the zero-count function \(N_{F_{\zeta}}(\varrho)\ge 1\) (when \(\varrho>|a|\)) and the above derivative formula.\qed
\end{proof}

\begin{theorem}[Time fiber two-ended touching and endpoint compression]\label{thm:dephys-timefiber-two-ended}

t(\varrho,\theta)
=\frac{2\varrho\sin\theta}{1+\varrho^2+2\varrho\cos\theta}
\;+\;
\mathrm{i}\,\frac{1-\varrho^2}{1+\varrho^2+2\varrho\cos\theta}.

Furthermore, the preimage of the equal-radius layer \(|w|=\varrho\) in the upper half-plane is an explicit circle (Proposition \ref{prop:app-cayley-preimage-modulus-circle}), whose intersections with the imaginary axis are

\end{theorem}

\begin{corollary}[Poisson--Busemann identity: time fiber coordinate is endpoint Poisson weight]\label{cor:dephys-poisson-busemann-timefiber}
For any \(w\in\mathbb{D}\), let \(t=\mathcal{C}^{-1}(w)=x+\mathrm{i}y\in\mathbb{H}\).
Then the strict identity holds
\[
\boxed{\;y=\mathcal{B}_{-1}(w)=\frac{1-|w|^2}{|1+w|^2}\;}
\]
where \(\mathcal{B}_{-1}\) is the exponential Busemann coordinate based at endpoint \(-1\) (Proposition \ref{prop:app-busemann-poisson-minus1}).
Therefore, under the endpoint compactification calibration, the "time fiber" is not an additionally introduced continuous coordinate axis, but the harmonic measure density (Poisson kernel) at endpoint \(-1\) itself.
\end{corollary}

\begin{proof}
Let \(t=x+\mathrm{i}y\in\mathbb{H}\) and \(w=\mathcal{C}(t)=\frac{1+\mathrm{i}t}{1-\mathrm{i}t}\).
Then
\[
1+w
=1+\frac{1+\mathrm{i}t}{1-\mathrm{i}t}
=\frac{2}{1-\mathrm{i}t},
\qquad
|1+w|^2=\frac{4}{|1-\mathrm{i}t|^2}=\frac{4}{x^2+(1+y)^2}.
\]
And from
\(
|w|^2=\frac{|1+\mathrm{i}t|^2}{|1-\mathrm{i}t|^2}
\)
we obtain
\[
1-|w|^2
=\frac{|1-\mathrm{i}t|^2-|1+\mathrm{i}t|^2}{|1-\mathrm{i}t|^2}
=\frac{4y}{x^2+(1+y)^2}.
\]
Dividing the two equations yields \(\frac{1-|w|^2}{|1+w|^2}=y\), hence \(\mathcal{B}_{-1}(w)=y\).
In particular, the constant \(4\) appearing in both \(1-|w|^2\) and \(|1+w|^2\) is merely the square of the conformal factor \(2\) under Cayley normalization, and strictly cancels in this invariant ratio.\qed
\end{proof}

\begin{remark}[Cayley scale and the normalizability of constant \(4\): invariants depend only on ratios and logarithmic differences]\label{rem:dephys-cayley-scale-gauge-four}
The "\(4\)" in the above derivation is a coordinate normalization constant, not a new dynamical invariant.
To clarify this, consider the scale-parametrized Cayley family with \(\lambda>0\)
\[
\mathcal{C}_{\lambda}(t):=\frac{\lambda+\mathrm{i}t}{\lambda-\mathrm{i}t}\qquad(t\in\mathbb{H}),
\]
which preserves the same endpoint reference (\(t=\infty\) still maps to \(-1\)) but changes the linear scale.
Direct calculation yields
\[
|1+\mathcal{C}_{\lambda}(x+\mathrm{i}y)|^2
=\frac{4\lambda^2}{x^2+(\lambda+y)^2},
\qquad
1-|\mathcal{C}_{\lambda}(x+\mathrm{i}y)|^2
=\frac{4\lambda y}{x^2+(\lambda+y)^2}.
\]
Therefore the exponential Busemann/Poisson ratio still strictly cancels the conformal square constant, yielding
\[
\mathcal{B}_{-1}\!\bigl(\mathcal{C}_{\lambda}(x+\mathrm{i}y)\bigr)=\frac{y}{\lambda}.
\]
Taking logarithm gives \(b_{-1}:=\log\mathcal{B}_{-1}=\log y-\log\lambda\), indicating that scale transformation only changes the additive zero of \(b_{-1}\);
therefore any observable geometric quantity can only appear in combinations invariant under rescaling (such as ratios of \(\mathcal{B}_{-1}\), or logarithmic differences of \(b_{-1}\)), and cannot depend on the isolated constant "\(4\)".
In ledger semantics, \(b_{-1}\) can also be viewed as a kind of "entropy potential/logarithmic potential," where the normalization constant only determines the additive zero without changing differential observations; its formal role is homologous to fixed normalization constants in geometric entropy formulas (e.g., the \(1/4\) in black hole entropy).
\end{remark}

\begin{remark}[Conformal factor \(2^2\) in conformal compactification and "area law" formal analogy]\label{rem:dephys-conformal-factor-2square-area-law}
The conformal square constant in Corollary \ref{cor:dephys-poisson-busemann-timefiber} is not a one-dimensional peculiarity.
In higher-dimensional compactification/projection coordinates, the conformal factor \(\Omega(x)=\frac{2}{1+|x|^2}\) and its square (or more general powers) also appear, so the pullback density of spherical metrics and area elements carries "\(4\)"-type constants.

This further indicates that the "\(4\)" appearing in this paper's endpoint horizon interface should be understood as the universal square constant of the compactification conformal factor, not an invariant unique to specific arithmetic objects.
\end{remark}

\begin{proposition}[Curvature normalization law: coefficient \(4\) in disk hyperbolic metric equivalent to constant curvature \(-1\) unique scale]\label{prop:dephys-hyperbolic-curvature-normalization-four}
The upper half-plane model takes the hyperbolic metric
\[
\dd s_{\mathbb{H}}^{2}:=\frac{\dd x^{2}+\dd y^{2}}{y^{2}}
\]
whose Gaussian curvature is \(-1\).
Any Möbius transformation \(w=w(t)\) conformally mapping \(\mathbb{H}\) to the unit disk \(\mathbb{D}\) induces a hyperbolic metric on the disk that can be written as
\[
\boxed{\;
\dd s_{\mathbb{D}}^{2}=\frac{4\,|\dd w|^{2}}{(1-|w|^{2})^{2}}\;}
\]
and this coefficient \(4\) is uniquely locked by "curvature normalization" to \(-1\):
if we write
\(
\dd s_{\mathbb{D}}^{2}=\frac{\alpha\,|\dd w|^{2}}{(1-|w|^{2})^{2}}
\),
then the curvature becomes
\[
\boxed{\;K_{\alpha}=-\frac{4}{\alpha}.\;}
\]
\end{proposition}

\begin{proof}
For a conformal metric \(\dd s^{2}=\rho(w)^{2}(\dd x^{2}+\dd y^{2})=\rho(w)^{2}|\dd w|^{2}\), the Gaussian curvature satisfies
\(
K=-\rho^{-2}\Delta(\log\rho)
\)
(where \(\Delta\) is the Euclidean Laplacian).
Take \(\rho_{\alpha}(w)=\frac{\sqrt{\alpha}}{1-|w|^{2}}\). Then
\(
\log\rho_{\alpha}=\frac12\log\alpha-\log(1-|w|^{2})
\); the constant term has zero Laplacian, and direct calculation gives
\(
\Delta\bigl(-\log(1-|w|^{2})\bigr)=\frac{4}{(1-|w|^{2})^{2}}
\).
Hence
\[
K_{\alpha}
=-\frac{(1-|w|^{2})^{2}}{\alpha}\cdot \frac{4}{(1-|w|^{2})^{2}}
=-\frac{4}{\alpha}.
\]
Therefore \(K_{\alpha}=-1\) if and only if \(\alpha=4\).
Since any Möbius equivalence \(w=w(t)\) only changes coordinates without changing the "unique conformally complete metric of curvature \(-1\)" normalization conclusion, the disk model must fall on the scale \(\alpha=4\).\qed
\end{proof}

\paragraph{Endpoint attracting semigroup and "time" definition: unique Koenigs depth}
Time fiber in this paper is not an externally added coordinate, but a one-dimensional depth ledger forcibly arising from the dilation semigroup after endpoint compactification.
The following theorem makes this precise as rigidity and uniqueness of endpoint linearization: any invertible audit compatible with scale dilation can have continuous cumulative product only along this one-dimensional Koenigs depth.

\begin{theorem}[Rigidity of endpoint linearization: uniqueness of Koenigs depth]\label{thm:dephys-endpoint-koenigs-rigidity}
Let \(\Phi=\{\phi_m\}_{m>0}\subset \mathrm{Aut}(\mathbb{D})\) be a family of Möbius automorphisms satisfying \(\phi_{mn}=\phi_m\circ\phi_n\) and \(\phi_m(\pm 1)=\pm 1\) for all \(m>0\).
Suppose there exists a coordinate \(h\) analytic in some neighborhood of \(u=-1\) satisfying \(h(-1)=0\), \(h'(-1)\neq 0\) and
\begin{equation}\label{eq:dephys-koenigs-functional-equation}
h\bigl(\phi_m(u)\bigr)=\frac{1}{m}\,h(u)\qquad(\forall\,m>0).
\end{equation}
Then in that neighborhood there exists a constant \(c\neq 0\) such that \(h(u)=c\,\mathfrak{w}(u)\), where \(\mathfrak{w}(u)=\frac{1+u}{1-u}\) is the endpoint half-plane coordinate (equation \eqref{eq:scale-linearization-halfplane-coordinate}); and for all \(m>0\) and all \(u\in\mathbb{D}\), \(\phi_m=\psi_m\), where \(\psi_m\) is given by equation \eqref{eq:scale-psi-m}.
Therefore the endpoint depth (Busemann/Koenigs depth)
\[
Y(u):=-\log\bigl|\mathfrak{w}(u)\bigr|
\]
satisfies the strict translation law
\[
Y\!\bigl(\psi_m(u)\bigr)=Y(u)+\log m
\]
(i.e., equation \eqref{eq:endpoint-busemann-addition-law}).
\end{theorem}

\begin{proof}
Differentiating \eqref{eq:dephys-koenigs-functional-equation} at \(u=-1\) and using \(h'(-1)\neq 0\) yields \(\phi_m'(-1)=1/m\).
On the other hand, the endpoint coordinate \(\mathfrak{w}\) conjugates \(\mathbb{D}\) to the right half-plane \(\mathbb{H}_{\mathrm{R}}\) (equation \eqref{eq:scale-linearization-halfplane-coordinate}), and conjugates \(\phi_m\) to an automorphism of \(\mathbb{H}_{\mathrm{R}}\) fixing \(0\) and \(\infty\); hence there exists \(a_m>0\) such that
\(

\), and by the chain rule we get \(a_m=\phi_m'(-1)=1/m\).
Thus \(\mathfrak{w}(\phi_m(u))=\mathfrak{w}(u)/m\), consistent with Proposition \ref{prop:scale-linearization-identity-multipliers}' \(\mathfrak{w}(\psi_m(u))=\mathfrak{w}(u)/m\); since \(\mathfrak{w}\) is injective, \(\phi_m=\psi_m\).
Now let \(g(u):=\frac{h(u)}{\mathfrak{w}(u)}\) (analytic in a neighborhood of \(u=-1\)); by \eqref{eq:dephys-koenigs-functional-equation} and \(\mathfrak{w}(\psi_m(u))=\mathfrak{w}(u)/m\) we have \(g(\psi_m(u))=g(u)\);
for any \(m>1\), iteration \(\psi_m^{\circ n}(u)\to-1\) (endpoint attraction), so by continuity \(g(u)=g(-1)\) is constant in the neighborhood, hence \(h=c\,\mathfrak{w}\).
The final translation law is equation \eqref{eq:endpoint-busemann-addition-law}.\qed
\end{proof}

\begin{theorem}[Minimal invertibilization of Dirichlet imaginary folding: non-eliminability of winding certificate]\label{thm:dephys-dirichlet-winding-minimal}

\]
Therefore for any set \(S\), if there exists a map
\[
(\sigma,t)\longmapsto \bigl(|z|,\mathrm{Arg}(z),\eta(\sigma,t)\bigr)\in (0,1)\times(-\pi,\pi]\times S

(equation \eqref{eq:dirichlet-arg-consistency}), thus giving an auditable discrete externalization; merging with the non-periodic phase axis \(u_{\mathrm C}(t)\) (equation \eqref{eq:dirichlet-cayley-phase-axis}) yields the tri-fiber injection coordinate \(\mathcal{E}_{\lambda}\) (Definition \ref{def:dirichlet-trifiber-injection}, Proposition \ref{prop:dirichlet-trifiber-injection-audit}).
\end{theorem}

\begin{proof}
Fix any \(\sigma>0\) and \(t_0\in\RR\). For each \(k\in\ZZ\), the point \((\sigma,t_0+kT_{\lambda})\) has the same image under \((|z|,\mathrm{Arg}(z))\).

The second half is given directly by Definition \eqref{eq:dirichlet-winding-certificate} and Proposition \ref{prop:dirichlet-trifiber-injection-audit}.\qed
\end{proof}

\paragraph{\texorpdfstring{$p$}{p}-typical difference tower and stable residue spectrum: compiling "scale tower" into auditable discrete hair}

\begin{theorem}[\texorpdfstring{$p$}{p}-typical invertibilization: Dwork congruence \texorpdfstring{$\Leftrightarrow$}{<->} primitive difference tower]\label{thm:dephys-ptypical-primitive-tower}
Let \(a_n(u)\in\ZZ[u]\) be the trace ghost sequence, fix a prime \(p\) and consider the \(p\)-power subsequence \(\{a_{p^k}(u)\}_{k\ge 0}\).

\begin{enumerate}
  \item For all \(k\ge 1\), the Dwork-type congruence holds
  \[
  a_{p^k}(u)\equiv a_{p^{k-1}}(u^p)\pmod{p^k}\qquad\text{in}\ \ZZ[u]
  \]
  (Corollary \ref{cor:witt-dwork-congruence});
  \item There exists (and is uniquely determined by formula) primitive difference coordinate
  \[
  p_{p^k}(u):=\frac{a_{p^k}(u)-a_{p^{k-1}}(u^p)}{p^k}\in\ZZ[u]\qquad(k\ge 1)
  \]
  (Proposition \ref{prop:witt-ppower-primitive-coordinate}), satisfying level-by-level decomposition
  \(a_{p^k}(u)=a_{p^{k-1}}(u^p)+p^k p_{p^k}(u)\);
  \item \(\{a_{p^k}(u)\}\) as a \(p\)-power subsequence of Big-Witt ghost satisfies \(p\)-adic compatibility under Adams/Frobenius operation \(u\mapsto u^p\) (Remark \ref{rem:witt-ptypical-ghost}).
\end{enumerate}
Therefore, \(\{p_{p^k}(u)\}_{k\ge 1}\) provides a completely internal, level-by-level reproducible "scale tower difference fingerprint": dilation information does not leak as a new continuous axis, but is externalized layer by layer at \(p^k\)-precision as primitive coordinates.
\end{theorem}

\begin{proof}
\((1)\Leftrightarrow(2)\) is obtained by direct equivalent rewriting from Corollary \ref{cor:witt-dwork-congruence} and Proposition \ref{prop:witt-ppower-primitive-coordinate}; \((3)\) is an interpretive restatement of the same fact under ghost/Witt calibration (Remark \ref{rem:witt-ptypical-ghost}).\qed
\end{proof}

\begin{proof}
