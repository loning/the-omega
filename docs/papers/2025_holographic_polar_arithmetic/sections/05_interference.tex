\section{Geometric interference and the quantum gap}
\label{sec:interference}

\subsection{Vector synthesis}
In $\CC$ the interaction of two holographic integers is expressed as Euclidean vector composition. For $a,b\in M$ define the synthesized vector
$$
v(a,b):=\mathcal{Z}(a)+\mathcal{Z}(b)\in\CC.
$$
In polar form, $\mathcal{Z}(a)$ and $\mathcal{Z}(b)$ are vectors; synthesis depends on their relative phase.

\begin{itemize}
    \item If $\theta_\times(a)\approx\theta_\times(b)$, magnitudes reinforce.
    \item If $\theta_\times(a)\approx\theta_\times(b)+\pi$, partial cancellation occurs.
\end{itemize}

The central point is strategic: in a multiplicative ontology we do not ask addition to be complete. Instead, we ask how and why it fails, and what kind of readout must replace it.

\subsection{Lattice mismatch: why linear addition is not preserved}
The embedding $\mathcal{Z}$ is designed to be multiplicative (Theorem~\ref{thm:Z_mult}). A naive expectation is that it should also respect linear addition, but this target is impossible except in a degenerate trivial case.

\begin{theorem}[No-go for additive completeness]
\label{thm:no_go_additive}
Let $\Phi:\NN_{>0}\to\CC$ satisfy
$$
\Phi(mn)=\Phi(m)\Phi(n)
\quad\text{and}\quad
\Phi(a+b)=\Phi(a)+\Phi(b)
$$
for all $m,n,a,b\in\NN_{>0}$. Then either $\Phi\equiv 0$ or $\Phi(n)=n$ for all $n\in\NN_{>0}$.
\end{theorem}

\begin{proof}
Let $z:=\Phi(1)$. Multiplicativity gives $z=\Phi(1)=\Phi(1\cdot 1)=\Phi(1)^2=z^2$, hence $z\in\{0,1\}$.
If $z=0$, then for any $n\in\NN_{>0}$ one has $\Phi(n)=\Phi(n\cdot 1)=\Phi(n)\Phi(1)=0$, so $\Phi\equiv 0$.
If $z=1$, then by additivity,
$$
\Phi(n)=\Phi(\underbrace{1+\cdots+1}_{n\ \text{terms}})=n\Phi(1)=n.
$$
\end{proof}

\begin{corollary}[Non-additivity is necessary]
\label{cor:nonadditive}
Let $\mathcal{Z}:M\to\CC^*$ be multiplicative and satisfy $\mathcal{Z}(1)=1$.
If $\mathcal{Z}$ is not the identity map, then it cannot satisfy $\mathcal{Z}(a+b)=\mathcal{Z}(a)+\mathcal{Z}(b)$ for all $a,b\in M$.
\end{corollary}

\begin{proof}
If $\mathcal{Z}$ were additive on $M=\NN_{>0}$, then it would satisfy the hypotheses of Theorem~\ref{thm:no_go_additive}. Since $\mathcal{Z}$ takes values in $\CC^*$ it is not identically zero, hence $\mathcal{Z}(n)=n$ for all $n\in M$, contradicting the assumption that $\mathcal{Z}$ is not the identity.
\end{proof}

This is the \emph{lattice mismatch}: vector synthesis takes place in the continuous plane $\CC$, while integer readout is constrained to the discrete point set $\mathcal{Z}(M)\subset\CC$. A quantitative obstruction is the distance-to-image function
$$
g(v):=\inf_{n\in M}\bigl|v-\mathcal{Z}(n)\bigr|.
$$
When $v=v(a,b)$, the value $g(v)$ measures the residual error incurred when one attempts to interpret synthesis as an exact integer operation. Under the minimal radial choice $\rho(n)=n$, Proposition~\ref{prop:projection_well_defined}(1) implies the infimum is a minimum, hence $g(v)=0$ if and only if $v\in\mathcal{Z}(M)$, and otherwise $g(v)>0$.

\subsection{Quantum projection: lossy additive readout}
Since $v(a,b)$ is not constrained to lie in $\mathcal{Z}(M)$, any macroscopic ``additive'' bookkeeping must be implemented as a \emph{projection} back to $M$. A minimal deterministic choice is a nearest-point readout.

\begin{definition}[Nearest-point readout]
\label{def:nearest_point}
For $v\in\CC$ define
$$
\pi(v):=\operatorname*{argmin}_{n\in M}\bigl|v-\mathcal{Z}(n)\bigr|,
$$
with a fixed tie-breaking rule (e.g. choose the smallest $n$ among minimizers).
\end{definition}

This produces a lossy ``sum readout'' $c=\pi(v(a,b))$. The residual
$$
\delta(a,b):=v(a,b)-\mathcal{Z}\!\bigl(\pi(v(a,b))\bigr)
$$
is the \emph{quantum gap}: the part of synthesis that cannot be represented inside the integer image. In physical terms, $\delta$ encodes the fluctuations that remain when a continuous interference pattern is coerced into a discrete lattice readout.

A probabilistic refinement replaces the hard nearest-point rule by a soft projection kernel.

\begin{definition}[Projection kernel]
\label{def:projection_kernel}
Fix a scale $\sigma>0$. For $v\in\CC$ define a probability distribution on $M$ by
$$
\pi_\sigma(c\mid v)
:=
\frac{\exp\!\bigl(-|v-\mathcal{Z}(c)|^2/\sigma^2\bigr)}
{\sum_{n\in M}\exp\!\bigl(-|v-\mathcal{Z}(n)|^2/\sigma^2\bigr)}.
$$
\end{definition}

\begin{proposition}[Well-definedness of the projection readout]
\label{prop:projection_well_defined}
Assume the minimal radial choice $\rho(n)=n$. Then for every $v\in\CC$:
\begin{enumerate}
    \item The minimizer set
    $$
    \operatorname*{Argmin}_{n\in M}\,|v-\mathcal{Z}(n)|
    $$
    is nonempty and finite, hence the nearest-point rule $\pi(v)$ is well-defined.
    \item For every $\sigma>0$ the normalizing sum
    $$
    Z_\sigma(v):=\sum_{n\in M}\exp\!\bigl(-|v-\mathcal{Z}(n)|^2/\sigma^2\bigr)
    $$
    converges and satisfies $0<Z_\sigma(v)<\infty$, hence $\pi_\sigma(\cdot\mid v)$ is a well-defined probability distribution on $M$.
\end{enumerate}
\end{proposition}

\begin{proof}
Under $\rho(n)=n$ we have $|\mathcal{Z}(n)|=n$, so $|\mathcal{Z}(n)|\to\infty$ as $n\to\infty$.
\begin{enumerate}
    \item Fix $v\in\CC$ and set $d_0:=|v-\mathcal{Z}(1)|$. For any $n\in M$,
    $$
    |v-\mathcal{Z}(n)|\ge \bigl||\mathcal{Z}(n)|-|v|\bigr|=|n-|v||.
    $$
    If $n>|v|+d_0$ then $|n-|v||>d_0$, hence $|v-\mathcal{Z}(n)|>d_0=|v-\mathcal{Z}(1)|$. Therefore every minimizer lies in the finite set $\{1,2,\dots,\lceil |v|+d_0\rceil\}$, so the minimum is attained and the minimizer set is nonempty and finite.
    \item For all $n\in M$ we have
    $$
    |v-\mathcal{Z}(n)|\ge \bigl||\mathcal{Z}(n)|-|v|\bigr|=|n-|v||,
    $$
    hence
    $$
    \exp\!\bigl(-|v-\mathcal{Z}(n)|^2/\sigma^2\bigr)\le \exp\!\bigl(-|n-|v||^2/\sigma^2\bigr).
    $$
    The comparison series $\sum_{n\ge 1}\exp(-|n-|v||^2/\sigma^2)$ converges, so $Z_\sigma(v)<\infty$. Positivity is immediate since each summand is positive.
\end{enumerate}
\end{proof}

\begin{remark}[Beyond the minimal radial choice]
Proposition~\ref{prop:projection_well_defined} uses the minimal choice $\rho(n)=n$.
For general radial characters, Section~\ref{sec:quantitative} gives sufficient conditions (radial properness and growth) ensuring discreteness of $\mathcal{Z}(M)$ and well-definedness of both the nearest-point and soft projection readouts.
\end{remark}

\begin{proposition}[Sharp projection limit]
\label{prop:sharp_projection_limit}
Assume the minimal radial choice $\rho(n)=n$. Fix $v\in\CC$ and let
$$
d_{\min}(v):=\min_{n\in M}|v-\mathcal{Z}(n)|.
$$
Define the minimizer set $D(v):=\{n\in M:\ |v-\mathcal{Z}(n)|=d_{\min}(v)\}$.
Then as $\sigma\downarrow 0$ the kernel $\pi_\sigma(\cdot\mid v)$ converges to the uniform distribution on $D(v)$.
In particular, if $D(v)$ is a singleton, then $\pi_\sigma(\cdot\mid v)$ converges to a point mass at that minimizer.
\end{proposition}

\begin{proof}
Let $d=d_{\min}(v)$. For $c\in D(v)$ one has $\exp(-|v-\mathcal{Z}(c)|^2/\sigma^2)=\exp(-d^2/\sigma^2)$.
For $m\notin D(v)$ set $\Delta_m:=|v-\mathcal{Z}(m)|^2-d^2>0$. Then
$$
\frac{\exp(-|v-\mathcal{Z}(m)|^2/\sigma^2)}{\exp(-d^2/\sigma^2)}
=\exp(-\Delta_m/\sigma^2)\xrightarrow[\sigma\downarrow 0]{}0.
$$
Since $D(v)$ is finite by Proposition~\ref{prop:projection_well_defined}(1), dividing numerator and denominator by $\exp(-d^2/\sigma^2)$ yields
$$
\pi_\sigma(c\mid v)
=
\frac{1}{|D(v)|+\sum_{m\notin D(v)}\exp(-\Delta_m/\sigma^2)}\xrightarrow[\sigma\downarrow 0]{}\frac{1}{|D(v)|}
$$
for $c\in D(v)$, and $\pi_\sigma(m\mid v)\to 0$ for $m\notin D(v)$.
\end{proof}

Given $a,b\in M$, set $v=v(a,b)$ and draw $C\sim \pi_\sigma(\,\cdot\,\mid v)$. The random gap
$$
\delta:=v-\mathcal{Z}(C)
$$
realizes a probability cloud around the synthesized vector. After readout, the integer outcome $C$ is canonically encoded in Zeckendorf/Ostrowski coordinates (Section~\ref{sec:zeckendorf}), making explicit that ``addition'' in the observable layer is a statistical projection onto a discrete coding lattice rather than a closed microscopic law.

\subsection{Cohomology of the quantum gap}
\label{subsec:gap_cohomology}
In the preceding construction the quantum gap $\delta$ appears as an arithmetic--geometric residual attached to a projection readout.
From a structural viewpoint, a natural question is why such residuals do not simply cancel to zero at macroscopic scales, or else accumulate into an uncontrolled ultraviolet divergence when one sums over many microscopic modes.
One way to separate ``pure gauge'' residuals from topologically protected ones is to treat the quantum gap as a discrete $1$-form on an interaction complex and classify it by cohomology.

\subsubsection{An interaction complex and cochains}
\begin{definition}[HPA interaction complex]
\label{def:hpa_interaction_complex}
Let $\mathcal{K}$ be a (finite) oriented simplicial complex; see e.g.~\cite{Hatcher2002}.
Its vertex set $V$ represents a chosen family of internal states (e.g.\ prime-lift states and composite states), its oriented edges $E$ represent chosen elementary HPA interaction/readout events, and its oriented $2$-simplices represent minimal closed interaction loops.
The specific choice of $\mathcal{K}$ is an application-layer modeling input, analogous in spirit to the choice of projection kernel and scale.
\end{definition}

\begin{definition}[Discrete $k$-forms as cochains]
\label{def:cochains_discrete_forms}
Let $A$ be an abelian group (e.g.\ $A=(\CC,+)$ in the complex embedding, or $A=(\OO,+)$ when gaps are lifted to octonionic internal coordinates).
Define
$
C^k(\mathcal{K};A)
$
to be the group of $A$-valued simplicial $k$-cochains on $\mathcal{K}$, i.e.\ functions on oriented $k$-simplices which are alternating with respect to orientation.
Elements of $C^0$ are discrete scalar fields and elements of $C^1$ are discrete $1$-forms.
See e.g.~\cite{Hatcher2002,DesbrunHiraniLeokMarsden2005DEC}.
\end{definition}

\subsubsection{Discrete exterior derivative and gauge}
\begin{definition}[Coboundary operator (discrete exterior derivative)]
\label{def:discrete_exterior_derivative}
Define $d^0:C^0(\mathcal{K};A)\to C^1(\mathcal{K};A)$ by
$$
(d^0 f)(u,v):=f(v)-f(u)
$$
on an oriented edge $(u,v)$.
Define $d^1:C^1(\mathcal{K};A)\to C^2(\mathcal{K};A)$ by
$$
(d^1\omega)(u,v,w):=\omega(v,w)-\omega(u,w)+\omega(u,v)
$$
on an oriented face $(u,v,w)$.
Write $d$ for the graded operator $d^k$.
See e.g.~\cite{Hatcher2002,DesbrunLeokMarsden2005Poincare}.
\end{definition}

\begin{definition}[Exact and closed gap fields]
\label{def:exact_closed_gap}
A $1$-cochain $\delta\in C^1(\mathcal{K};A)$ is \emph{exact} if $\delta=d^0 f$ for some $f\in C^0(\mathcal{K};A)$, and \emph{closed} if $d^1\delta=0$.
\end{definition}

\begin{remark}[Quantum gap as a $1$-cochain]
Fix a model choice of $\mathcal{K}$.
To each oriented interaction/readout edge $e=(u,v)\in E$, associate a synthesized vector $v_e$ (in the embedding space) and an output lattice state $c_e$ (deterministic $c_e=\pi(v_e)$ or sampled $c_e\sim\pi_\sigma(\cdot\mid v_e)$).
Then the quantum gap field can be recorded as the edge cochain
$
\delta(e):=v_e-\mathcal{Z}(c_e)\in A.
$
\end{remark}

\begin{remark}[Gauge interpretation]
If $\delta=d^0 f$ is exact, then $\delta$ can be eliminated by a redefinition of local reference data $f$ at the vertices.
In this sense, exact components represent a choice of local phase/origin convention rather than a topological obstruction.
\end{remark}

\subsubsection{First cohomology and topological obstruction}
\begin{definition}[First cohomology of the quantum gap]
\label{def:h1_gap}
Define
$$
H^1(\mathcal{K};A):=\ker(d^1)/\mathrm{im}(d^0).
$$
The cohomology class $[\delta]\in H^1(\mathcal{K};A)$ is invariant under gauge shifts $\delta\mapsto \delta+d^0 f$.
\end{definition}

\begin{proposition}[Loop holonomy depends only on the cohomology class]
\label{prop:gap_holonomy_class}
Let $\gamma$ be an oriented $1$-cycle in $\mathcal{K}$ and let $\delta\in C^1(\mathcal{K};A)$ be closed.
Then the cycle sum $\langle \delta,\gamma\rangle$ depends only on the class $[\delta]\in H^1(\mathcal{K};A)$.
In particular, if $[\delta]=0$ then $\langle \delta,\gamma\rangle=0$ for every $1$-cycle $\gamma$.
\end{proposition}

\begin{remark}[A toy cycle model for nontrivial $H^1$]
\label{rem:gap_h1_toy_cycle}
Let $\mathcal{K}$ be the $1$-dimensional simplicial complex given by an oriented $N$-cycle with vertices $v_0,\dots,v_{N-1}$ and edges $e_k=(v_k,v_{k+1})$ (indices mod $N$).
Since $\mathcal{K}$ has no $2$-simplices, every $1$-cochain $\delta\in C^1(\mathcal{K};A)$ is closed.
For any $a\in A$ define the constant $1$-cochain $\delta_a(e_k):=a$.
For the fundamental cycle $\gamma:=\sum_{k=0}^{N-1}e_k$ one has $\langle \delta_a,\gamma\rangle=Na$.
If $\delta_a$ were exact, $\delta_a=d^0 f$, then the cycle sum telescopes and equals $0$, hence $\delta_a$ is non-exact whenever $a\ne 0$.
In particular, when $A$ is a real vector space (e.g.\ $A=\CC$ or $A=\OO$ under addition), one has $H^1(\mathcal{K};A)\cong A$, and the loop holonomy provides an explicit gauge-invariant obstruction to global cancellation.
\end{remark}

\begin{assumption}[Matter as nontrivial gap cohomology (interpretation layer)]
\label{assump:matter_gap_cohomology}
Assume the interpretation layer of a concrete HPA-based model identifies long-lived physical excitations with topologically protected gap modes, i.e.\ with closed but non-exact components of the gap field.
Then $[\delta]\neq 0$ provides a gauge-invariant obstruction to global cancellation and can be treated as a stability label for ``real'' excitations, while exact components represent renormalizable/gauge fluctuations.
\end{assumption}

\begin{remark}[Vanishing $H^1$ implies no topological gap modes]
If $H^1(\mathcal{K};A)=0$, then every closed gap field is exact, hence gauge.
Thus any model in which matter is identified with nontrivial $[\delta]$ requires an interaction complex $\mathcal{K}$ with nontrivial first cohomology (non-contractible loops).
In practice this is expected to arise from the combinatorics of allowable readout paths and, in octonionic lifts, from the presence of multiple inequivalent composition histories.
\end{remark}

\begin{remark}[Associator-to-cohomology link (algebra $\to$ topology)]
In an octonionic lift, non-associativity provides a canonical mechanism for generating inequivalent composition histories: for three internal factors one can compare $(xy)z$ and $x(yz)$, and their discrepancy is the associator $\mathcal{A}(x,y,z)$ (Definition~\ref{def:associator}).
When such path dependence is fed through a readout/projection map, it naturally produces loop-level residuals in the gap cochain $\delta$, hence nontrivial cohomology classes $[\delta]\in H^1(\mathcal{K};A)$.
Combined with the mass--associator ansatz (Assumption~\ref{assump:mass_associator}), this motivates the slogan: \emph{mass is the topological shadow of algebraic non-associativity}.
\end{remark}

\subsubsection{Stability and ultraviolet finiteness (interpretive consequences)}
\begin{corollary}[Topological stability heuristic]
\label{cor:gap_topological_stability}
If an excitation is labeled by a nontrivial class $[\delta]\in H^1(\mathcal{K};A)$, then it cannot be removed by any local gauge redefinition $\delta\mapsto\delta+d^0 f$.
In particular, any decay channel that would annihilate the excitation must correspond to a change of the effective interaction complex or to a transition in which the relevant class becomes trivial in a larger complex.
\end{corollary}

\begin{remark}[Vacuum-energy bookkeeping and finite parts]
The cohomology viewpoint suggests a separation of bookkeeping:
exact/gauge components of $\delta$ can be treated as renormalizable fluctuations (to be removed or regularized by the finite-part prescriptions developed in Appendix~\ref{sec:cyclic_calculus}),
while nontrivial cohomology classes provide discrete topological sectors which do not cancel by local rephasings.
Making this quantitative requires specifying a model-dependent weighting of sectors and a controlled summation scheme.
\end{remark}

Any statement about divergent sums or ``finite residuals'' must be made using the orbit-trace and finite-part machinery introduced in Appendix~\ref{sec:cyclic_calculus}.
