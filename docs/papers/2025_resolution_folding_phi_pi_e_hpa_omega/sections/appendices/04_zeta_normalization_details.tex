
\section{Zeta identities and normalization details}
\label{app:zeta_details}

\subsection{Factorization of \texorpdfstring{$1-z-z^2$}{1-z-z2}}
Let $\varphi=(1+\sqrt{5})/2$.
Then
\[
1-z-z^2=(1-\varphi z)\bigl(1+\varphi^{-1}z\bigr).
\]
Indeed,
\[
(1-\varphi z)(1+\varphi^{-1}z)=1+(\varphi^{-1}-\varphi)z-z^2
=1-z-z^2,
\]
since $\varphi-\varphi^{-1}=1$.
Thus the zeta function in Proposition~\ref{prop:zeta_det} has poles at $z=\varphi^{-1}$ and $z=-\varphi$.

\subsection{Unit-disk normalization}
With the normalization $z=r/\varphi$ one has
\[
\widetilde{\zeta}(r)=\zeta\Big(\frac{r}{\varphi}\Big)
=\frac{1}{(1-r)(1+\varphi^{-2}r)}.
\]
Hence $\widetilde{\zeta}$ is holomorphic on $|r|<1$ and its principal pole sits at $r=1$.

\subsection{Comparison with the full shift}
The full two-symbol shift has adjacency matrix
\[
B=\begin{pmatrix}1&1\\1&1\end{pmatrix},
\qquad \rho(B)=2,
\qquad \det(I-zB)=1-2z.
\]
Thus its zeta function is $(1-2z)^{-1}$ with principal pole at $z=1/2$.
Under the same normalization $z=r/\varphi$, this pole becomes $r=\varphi/2<1$, i.e.\ it moves inside the unit disk.
This cleanly illustrates the pole-barrier language: relative to the $\varphi$ scaling, allowing the forbidden transition produces an interior pole.
