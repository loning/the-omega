\section{Problem 1: Translation of the Finite-Volume $\Phi^4_3$ Measure}

\begin{proposition}
Let $\mu$ be the finite-volume $\Phi^4_3$ measure on $\mathcal D'(\mathbb T^3)$. For every nonzero smooth function $\psi\in C^\infty(\mathbb T^3)$, with translation map
\[
T_\psi(u)=u+\psi,
\]
the measures $\mu$ and $T_{\psi\#}\mu$ are mutually singular. In particular, they are not equivalent.
\end{proposition}

\begin{proof}
For measurable sets $A\subset \mathcal D'(\mathbb T^3)$, define
\[
(T_{\psi\#}\mu)(A):=\mu\bigl(T_\psi^{-1}(A)\bigr).
\]
By Hairer (2022, Theorem 1.1), there exist constants $a,b\in\mathbb R$ such that for every smooth test function $\varphi$,
\[
A_\varphi
:=\left\{u:\lim_{n\to\infty}e^{-3n/4}\left\langle u_n^3-3ae^{e^n}u_n-9be^n u,\varphi\right\rangle=0\right\}
\]
satisfies $\mu(A_\varphi)=1$, and for every nonzero smooth shift $\widehat\psi$ there exists smooth $\varphi$ with
\[
\mu(A_\varphi+\widehat\psi)=0.
\]
Set $\widehat\psi=-\psi\neq 0$. Choose the corresponding $\varphi$. Then
\[
\mu(A_\varphi-\psi)=\mu(A_\varphi+\widehat\psi)=0.
\]
Since $T_\psi^{-1}(A)=A-\psi$ for translations,
\[
(T_{\psi\#}\mu)(A_\varphi)=\mu\bigl(T_\psi^{-1}(A_\varphi)\bigr)=\mu(A_\varphi-\psi)=0.
\]
At the same time, $\mu(A_\varphi)=1$. Hence there exists measurable $A_\varphi$ such that
\[
\mu(A_\varphi)=1,
\qquad
(T_{\psi\#}\mu)(A_\varphi)=0,
\]
which is exactly $\mu\perp T_{\psi\#}\mu$.
Therefore $\mu$ and $T_{\psi\#}\mu$ are not equivalent.
\end{proof}

\begin{remark}
The proof is a direct measure-theoretic consequence of the witness-event statement in Hairer (2022, Theorem 1.1). No additional quasi-invariance argument is needed.
\end{remark}
