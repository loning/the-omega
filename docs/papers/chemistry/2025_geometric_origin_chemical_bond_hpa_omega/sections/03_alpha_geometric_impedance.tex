\section{The geometric origin of \texorpdfstring{$\alpha_{\mathrm{em}}$}{alpha\_em}: three-channel impedance}
\label{sec:alpha_geo}

Chemistry requires an absolute electromagnetic energy scale. In standard physics this scale is set by the dimensionless electromagnetic invariant $\alpha_{\mathrm{em}}$ through atomic units. In the HPA--$\Omega$ constant-geometry program, $\alpha_{\mathrm{em}}$ is a \emph{target} dimensionless invariant of the readout protocol, not an external knob; the minimal closed model produces a theorem-level geometric prediction $\alpha_{\mathrm{geo}}$ for its value. We record the minimal closed-theory chain that yields the theorem-level value
\begin{equation}
\alpha_{\mathrm{geo}}^{-1}=4\pi^3+\pi^2+\pi,
\end{equation}
following the constants-geometry manuscript \cite{Ma2025PCG}.

\subsection{Serial composition and logarithmic readout cost}
\begin{axiom}[Serial composition and geometric impedance]\label{ax:serial_impedance_chem}
If a readout protocol must satisfy constraints in a fixed hierarchy of channels, assign each channel $j$ a multiplicative weight $w_j\in(0,1]$ and define its readout cost by $V_j:=-\log w_j$. The total weight is
\begin{equation}
w_{\mathrm{tot}}=\prod_j w_j,
\end{equation}
and the associated geometric impedance is the additive log-cost
\begin{equation}
\alpha_{\mathrm{geo}}^{-1}:=-\log w_{\mathrm{tot}}=\sum_j V_j.
\end{equation}
\end{axiom}
\begin{remark}
The axiom enforces a precise meaning of ``impedance'': the ontological composition is multiplicative; additivity is an induced feature after the logarithmic readout projection.
This logarithmic additivity is the standard ``product-to-sum'' mechanism underlying information measures and cost aggregation; see, e.g., \cite{Shannon1948,CoverThomas2006,Aczel1966}.
\end{remark}

\subsection{Three strata and canonical phase volumes}
\begin{axiom}[Geometric cost as canonical volume]\label{ax:impedance_volume_chem}
For each channel $j$, the process cost is identified with a canonical geometric invariant of its compact phase space $\mathcal{M}_j$. In the minimal model,
\begin{equation}
V_j=\mathrm{Vol}(\mathcal{M}_j),
\end{equation}
where $\mathrm{Vol}$ is the volume induced by the standard bi-invariant metric on compact Lie groups and quotient metrics.
\end{axiom}

\begin{axiom}[Electromagnetic three-stratum phase spaces]\label{ax:em_phase_spaces_chem}
The electromagnetic readout channel is represented by three topologically distinct strata:
\begin{equation}
\mathcal{M}_{\mathrm{bulk}} \cong U(1)\times SU(2),\qquad
\mathcal{M}_{\mathrm{boundary}} \cong SO(3),\qquad
\mathcal{M}_{\mathrm{line}} \cong \mathbb{R}P^1.
\end{equation}
\end{axiom}
\begin{remark}
The projective quotients encode the intrinsic ray identification of readout. In particular, $SU(2)\to SO(3)=SU(2)/\{\pm 1\}$ reflects the spinorial double cover, and $\mathbb{R}P^1=U(1)/\{\pm 1\}$ reflects phase-ray identification \cite{Nakahara2003,Hall2015}.
\end{remark}

\subsection{Theorem-level value from group volumes}
\begin{theorem}[Three-channel geometric impedance]\label{thm:alpha_three_channel_chem}
Under Axioms~\ref{ax:serial_impedance_chem}--\ref{ax:em_phase_spaces_chem}, the minimal-model geometric value of the inverse fine-structure constant is
\begin{equation}
\alpha_{\mathrm{geo}}^{-1}=4\pi^3+\pi^2+\pi\approx 137.0363037759.
\end{equation}
\end{theorem}
\begin{proof}
By Axiom~\ref{ax:serial_impedance_chem}, $\alpha_{\mathrm{geo}}^{-1}=V_{\mathrm{bulk}}+V_{\mathrm{boundary}}+V_{\mathrm{line}}$. By Axiom~\ref{ax:impedance_volume_chem}, $V_j=\mathrm{Vol}(\mathcal{M}_j)$, and by Axiom~\ref{ax:em_phase_spaces_chem} this reduces to canonical volumes. Standard values \cite{Nakahara2003,Hall2015} are
\begin{equation}
\mathrm{Vol}(U(1))=2\pi,\quad \mathrm{Vol}(SU(2))=2\pi^2,\quad
\mathrm{Vol}(SO(3))=\pi^2,\quad \mathrm{Vol}(\mathbb{R}P^1)=\pi.
\end{equation}
Therefore
\begin{equation}
V_{\mathrm{bulk}}=\mathrm{Vol}(U(1)\times SU(2))=4\pi^3,\qquad
V_{\mathrm{boundary}}=\pi^2,\qquad
V_{\mathrm{line}}=\pi,
\end{equation}
and the sum gives $4\pi^3+\pi^2+\pi$.
\end{proof}

\subsection{Interface matching to CODATA}
Let $\alpha_{\mathrm{exp}}$ denote the CODATA recommended fine-structure constant \cite{CODATA2022RMP}. Define the inverse-impedance gap
\begin{equation}
\Delta\alpha^{-1}:=\alpha_{\mathrm{geo}}^{-1}-\alpha_{\mathrm{exp}}^{-1}.
\end{equation}
In the multiplicative readout variable $w=\exp(-\alpha^{-1})$, the mismatch is absorbed by a single matching input
\begin{equation}
s_\alpha:=\frac{w_{\mathrm{exp}}}{w_{\mathrm{geo}}}=\exp(\Delta\alpha^{-1}),
\end{equation}
which encodes the interface between the minimal geometric model and operational low-energy coupling \cite{Ma2025PCG}. In this paper we keep the interface explicit rather than treating it as a tuning parameter: $s_\alpha$ is recorded and can be propagated into any chemistry-facing prediction requiring high precision.

\begin{remark}[Why the weight variable \texorpdfstring{$w=\exp(-\alpha^{-1})$}{w=exp(-alpha^{-1})}]
The variable $w=\exp(-\alpha^{-1})$ is the natural multiplicative counterpart of the additive impedance $\alpha^{-1}$ under the logarithmic readout map. It is introduced only to make the interface matching multiplicative (a single factor $s_\alpha$), avoiding hidden additive retuning across different applications.
\end{remark}

\paragraph{Quantitative size.}
Using the CODATA 2022 central inverse coupling quoted in \cite{Ma2025PCG,CODATA2022RMP}, one has
\[
\Delta\alpha^{-1}\approx 3.046\times 10^{-4},
\qquad
\frac{\Delta\alpha^{-1}}{\alpha_{\mathrm{exp}}^{-1}}\approx 2.22\times 10^{-6},
\qquad
s_\alpha\approx 1.0003046.
\]
Thus the interface mismatch is at the few-ppm level in inverse impedance and at the few-ppm level in any derived chemical scale that depends smoothly on $\alpha_{\mathrm{em}}$.

\end{proposition}
\begin{proof}
This is a finite exhaustive enumeration recorded in \cite{Ma2025PCG}.
\end{proof}
