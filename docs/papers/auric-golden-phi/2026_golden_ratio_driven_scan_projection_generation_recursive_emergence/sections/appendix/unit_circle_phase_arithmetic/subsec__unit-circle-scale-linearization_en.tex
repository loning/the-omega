\subsection{Endpoint Linearization of the Dilation Semigroup: Half-Plane Coordinates, Precision Potential, and Mellin Diagonalization}\label{subsec:unit-circle-scale-linearization}
This subsection completely linearizes the nonlinear action of the ``scale dilation \(x\mapsto mx\)'' on unit circle/disk phase variables and thereby produces a diagonalizing coordinate system whose carrier is the endpoint precision potential.
This coordinate system aligns with the Dirichlet--Witt dilation operator caliber of Appendix~\ref{app:real-input-40-dirichlet}: the former writes dilations as M\"obius automorphisms of the unit disk, while the latter writes them as \(s\mapsto ms\) in the Dirichlet variable.

\paragraph{Completely linearizing coordinate for the dilation semigroup}
Let \(u\in\mathbb{D}\) be the unit disk variable (boundary \(u\in\partial\mathbb{D}\) corresponds to the unitary slice) and define the Cayley half-plane coordinate (right half-plane model)
\begin{equation}\label{eq:scale-linearization-halfplane-coordinate}
\boxed{\;
\mathfrak{w}(u):=\frac{1+u}{1-u}\in\mathbb{H}_{\mathrm{R}}:=\{\,\mathfrak{w}\in\CC:\Re(\mathfrak{w})>0\,\}\; }.
\end{equation}
It sends the endpoint \(u=-1\) to \(\mathfrak{w}=0\) and sends \(u=1\) to infinity; the inverse is
\begin{equation}\label{eq:scale-linearization-halfplane-inverse}
\boxed{\;u(\mathfrak{w})=\frac{\mathfrak{w}-1}{\mathfrak{w}+1}\; }.
\end{equation}
From \eqref{eq:unit-circle-phase-inverse} one reads off (at \(u\neq-1\)) the relation between the real-axis coordinate and the half-plane coordinate:
\begin{equation}\label{eq:scale-linearization-w-vs-x}
\boxed{\;
x(u)=\mathrm{i}\,\frac{1-u}{1+u},
\qquad
\mathfrak{w}(u)=\frac{\mathrm{i}}{x(u)}\; }.
\end{equation}
Hence ``infinite height \(x\to\infty\)'' in the \(\mathfrak{w}\)-coordinate is equivalent to the endpoint \(\mathfrak{w}\to 0\).

\medskip
For any scale parameter \(m>0\), let \(\psi_m\) be the M\"obius automorphism induced on the \(u\)-domain by the dilation \(x\mapsto mx\), giving the explicit formula
\begin{equation}\label{eq:scale-psi-m}
\boxed{\;
\psi_m(u):=\frac{(m+1)u+(1-m)}{(1-m)u+(m+1)}\; }.
\end{equation}
This family preserves \(\{\pm 1\}\) as fixed points and satisfies a strict linearization conjugacy identity.

\begin{proposition}[Strict linearization identity and endpoint multipliers]\label{prop:scale-linearization-identity-multipliers}
For any \(m>0\) and any \(u\neq 1\),
\begin{equation}\label{eq:scale-linearization-identity}
\boxed{\;\mathfrak{w}\!\bigl(\psi_m(u)\bigr)=\frac{1}{m}\,\mathfrak{w}(u)\; }.
\end{equation}
Moreover, the multipliers at the fixed points are
\begin{equation}\label{eq:scale-psi-m-multipliers}
\boxed{\;\psi_m'(1)=m,\qquad \psi_m'(-1)=\frac{1}{m}\; }.
\end{equation}
In particular, when \(m>1\), \(u=-1\) is the attracting endpoint of the scale chain, and the linearization contraction rate in its neighborhood is strictly \(m^{-1}\).
\end{proposition}
\begin{proof}
From \eqref{eq:scale-psi-m} one directly computes
\[
1+\psi_m(u)=\frac{2(1+u)}{(1-m)u+(m+1)},
\qquad
1-\psi_m(u)=\frac{2m(1-u)}{(1-m)u+(m+1)}.
\]
Dividing the two gives
\(
\frac{1+\psi_m(u)}{1-\psi_m(u)}=\frac{1}{m}\frac{1+u}{1-u}
\), which is \eqref{eq:scale-linearization-identity}.
Writing \eqref{eq:scale-psi-m} in M\"obius form \(\psi_m(u)=(au+b)/(cu+d)\) with
\(
a=d=m+1,\ b=c=1-m
\), one has \(ad-bc=(m+1)^2-(1-m)^2=4m\), hence
\[
\psi_m'(u)=\frac{ad-bc}{(cu+d)^2}=\frac{4m}{\bigl((1-m)u+(m+1)\bigr)^2}.
\]
Substituting \(u=\pm 1\) immediately yields \eqref{eq:scale-psi-m-multipliers}. \qed
\end{proof}

\paragraph{Additive law for the endpoint precision potential and hyperbolic geometric time}
Define the endpoint precision potential (Busemann-type scale coordinate)
\begin{equation}\label{eq:endpoint-busemann-potential}
\boxed{\;y(u):=-\log|\mathfrak{w}(u)|\; }.
\end{equation}
From \eqref{eq:scale-linearization-identity} one immediately obtains the strict scale addition law
\begin{equation}\label{eq:endpoint-busemann-addition-law}
\boxed{\;y\!\bigl(\psi_m(u)\bigr)=y(u)+\log m\; }.
\end{equation}
Thus the scale dilation is converted, on the endpoint precision potential, into purely additive time evolution; the accumulated complexity of the infinite scale chain is collected into the one-dimensional additive ledger \(\sum\log m\).

\medskip
The right half-plane model carries the Poincar{\'e} hyperbolic metric
\begin{equation}\label{eq:poincare-right-halfplane-metric}
\boxed{\;\dd s^{2}=\frac{|\dd \mathfrak{w}|^{2}}{\Re(\mathfrak{w})^{2}}\; }.
\end{equation}
For any constant \(a>0\), the rescaling \(\mathfrak{w}\mapsto a\mathfrak{w}\) preserves \eqref{eq:poincare-right-halfplane-metric} and is hence a hyperbolic isometry.
In particular, the ``time parameter'' of \(\mathfrak{w}\mapsto \mathfrak{w}/m\) is precisely \(\log m\), in complete agreement with the additive time of \eqref{eq:endpoint-busemann-addition-law}.

\paragraph{Mellin spectral diagonalization of the phase--Witt transfer operator}
In the completely linearized coordinate \(\mathfrak{w}\), introduce the discrete scale transfer operator
\begin{equation}\label{eq:scale-transfer-operator}
\boxed{\;
(\mathcal{T}G)(\mathfrak{w})
:=\sum_{m\ge 1}\frac{1}{m}\,G\!\left(\frac{\mathfrak{w}}{m}\right)\; }.
\end{equation}
For the power function \(G_{\alpha}(\mathfrak{w})=\mathfrak{w}^{\alpha}\) (\(\Re(\alpha)>0\)) one has the pointwise eigenrelation
\begin{equation}\label{eq:scale-transfer-eigen-monomial}
\boxed{\;
\mathcal{T}(\mathfrak{w}^{\alpha})=\zeta(1+\alpha)\,\mathfrak{w}^{\alpha}\; }.
\end{equation}
Furthermore, let \(\mathcal{M}\) denote the Mellin transform (with multiplicative Haar measure)
\begin{equation}\label{eq:scale-mellin-transform}
\boxed{\;
(\mathcal{M}G)(s):=\int_{0}^{\infty} G(\mathfrak{w})\,\mathfrak{w}^{\,1-s}\,\frac{\dd \mathfrak{w}}{\mathfrak{w}}\; }.
\end{equation}
Under the condition \(\Re(s)>1\) and admissibility of interchange of summation and integration, one has the strict spectral diagonalization identity
\begin{equation}\label{eq:scale-mellin-diagonalization}
\boxed{\;\mathcal{M}(\mathcal{T}G)(s)=\zeta(s)\,\mathcal{M}G(s)\; }.
\end{equation}
\begin{proof}
Under the admissibility condition,
\[
\mathcal{M}(\mathcal{T}G)(s)
=\sum_{m\ge 1}\frac{1}{m}\int_{0}^{\infty}G(\mathfrak{w}/m)\,\mathfrak{w}^{\,1-s}\,\frac{\dd\mathfrak{w}}{\mathfrak{w}}.
\]
Making the substitution \(\mathfrak{w}=m\xi\) gives \(\dd\mathfrak{w}/\mathfrak{w}=\dd\xi/\xi\) and \(\mathfrak{w}^{1-s}=m^{1-s}\xi^{1-s}\), hence
\[
\int_{0}^{\infty}G(\mathfrak{w}/m)\,\mathfrak{w}^{\,1-s}\,\frac{\dd\mathfrak{w}}{\mathfrak{w}}
=m^{1-s}\int_{0}^{\infty}G(\xi)\,\xi^{\,1-s}\,\frac{\dd\xi}{\xi}
=m^{1-s}\,\mathcal{M}G(s).
\]
Substituting back and summing yields
\[
\mathcal{M}(\mathcal{T}G)(s)
=\Bigl(\sum_{m\ge 1}m^{-s}\Bigr)\mathcal{M}G(s)
=\zeta(s)\,\mathcal{M}G(s),
\]
where \(\Re(s)>1\) ensures absolute convergence of \(\sum_{m\ge 1}m^{-s}\). \qed
\end{proof}
This conclusion provides a structural equivalence: the Dirichlet series of \(\zeta\) in this caliber becomes the Mellin-domain multiplier of the scale chain operator \(\mathcal{T}\); its analytic properties may be viewed as the ``symbol structure'' of the transfer operator in the frequency domain.

\paragraph{Operator decomposition of the Euler product: primes as scale generators}
Along the endpoint axis with \(\mathfrak{w}>0\), letting the additive scale coordinate be \(y:=-\log \mathfrak{w}\), the map \(\mathfrak{w}\mapsto \mathfrak{w}/m\) is equivalent to \(y\mapsto y+\log m\).
For the shift operator
\begin{equation}\label{eq:scale-shift-operator}
\boxed{\;(U_{a}f)(y):=f(y+\log a)\; }
\end{equation}
define
\begin{equation}\label{eq:scale-K-operator}
\boxed{\;(\mathcal{K}f)(y):=\sum_{m\ge 1}\frac{1}{m}\,f(y+\log m)=\sum_{m\ge 1}\frac{1}{m}\,(U_m f)(y)\; }.
\end{equation}
Under the commutative semigroup structure \(U_aU_b=U_{ab}\), the Euler product yields the following operator factorization (valid in function classes/operator topologies ensuring convergence; it may also be viewed as a formal identity):
\begin{equation}\label{eq:scale-euler-operator-product}
\boxed{\;\mathcal{K}=\prod_{p}\left(I-\frac{1}{p}U_{p}\right)^{-1}\; }.
\end{equation}
The uniqueness of the expansion coefficients comes from the fundamental theorem of arithmetic: for each \(m=\prod p^{\nu_p(m)}\), the product \(\prod_p (p^{-1}U_p)^{\nu_p(m)}\) contributes exactly the term \(\frac{1}{m}U_m\).

\paragraph{Character-twisted dilation operator and the phase--depth spectral theorem}
On the depth line \(y=-\log|\mathfrak{w}|\in\RR\), write the shift representation as \((T_{\tau}g)(y):=g(y+\tau)\).
For any Dirichlet character \(\chi\) (a character modulo \(q\), or more generally a completely multiplicative coefficient with \(|\chi|\le 1\)) and \(\sigma>0\), define the \textbf{character--dilation} operator
\begin{equation}\label{eq:scale-K-sigma-chi}
\boxed{\;
\mathcal{K}_{\sigma,\chi}:=\sum_{m\ge 1}\frac{\chi(m)}{m^{1+\sigma}}\,T_{\log m},
\qquad\text{acting on }L^2(\RR,\dd y)\; }.
\end{equation}

\begin{theorem}[Phase--depth spectral theorem: frequency-domain multiplier as Dirichlet \(L\)-function]\label{thm:phase-depth-spectrum-L}
Let \(\mathcal{F}\) denote the Fourier transform in the \(y\)-variable: \((\mathcal{F}g)(\xi)=\int_{\RR}g(y)e^{-\mathrm{i}\xi y}\dd y\).
Then for all \(g\in L^2(\RR)\),
\begin{equation}\label{eq:phase-depth-fourier-diagonalization}
\boxed{\;
\mathcal{F}(\mathcal{K}_{\sigma,\chi}g)(\xi)
=L(1+\sigma+\mathrm{i}\xi,\chi)\cdot(\mathcal{F}g)(\xi)\; },
\end{equation}
where \(L(s,\chi)=\sum_{m\ge 1}\chi(m)m^{-s}\) is the Dirichlet \(L\)-series.
In particular, the spectral obstruction (non-invertibility points) of \(\mathcal{K}_{\sigma,\chi}\) on \(L^2\) is strictly equivalent to the zeros of \(L(1+\sigma+\mathrm{i}\xi,\chi)=0\).
\end{theorem}
\begin{proof}
From the definition of \(\mathcal{K}_{\sigma,\chi}\) and the Fourier representation of the shift operator \(T_{\log m}g\mapsto e^{\mathrm{i}\xi\log m}\widehat g(\xi)\),
\[
\mathcal{F}(\mathcal{K}_{\sigma,\chi}g)(\xi)
=\sum_{m\ge 1}\frac{\chi(m)}{m^{1+\sigma}}\,e^{\mathrm{i}\xi\log m}\,\widehat g(\xi)
=\Bigl(\sum_{m\ge 1}\frac{\chi(m)}{m^{1+\sigma+\mathrm{i}\xi}}\Bigr)\,\widehat g(\xi)
=L(1+\sigma+\mathrm{i}\xi,\chi)\,(\mathcal{F}g)(\xi).
\]
Spectral obstruction is equivalent to the multiplier vanishing, giving the stated conclusion. \qed
\end{proof}
This conclusion provides a closed restatement of the ``infinite dilation chain'': dilations are no longer viewed as complex structures at infinity but rather as translational convolution on the depth line; all of their arithmetic content converges completely to the \(L\)-function multiplier in the frequency domain. This is consistent with the strict commutative structure of the Dirichlet--Witt dilation bridge package of Appendix~\ref{app:real-input-40-dirichlet}---\((Wf)(s)=\sum_{m\ge 1}\frac{1}{m}f(ms)\), \(\log\mathcal{Z}(s)=(W\mathcal{P})(s)\), and M\"obius inversion \(\mathcal{P}(s)=\sum_{m\ge 1}\frac{\mu(m)}{m}\log\mathcal{Z}(ms)\)---with character twisting introduced here for the dilation kernel.

\paragraph{Endpointization of ``infinity'' (coordinate level)}
Under the phase compactification gate of this appendix, the critical-line parameter (the height axis) can be strictly endpointized at the coordinate level: \(x\to\infty\) is equivalent to \(\mathfrak{w}(u)\to 0\) (equation~\eqref{eq:scale-linearization-w-vs-x}).
Proposition~\ref{prop:scale-linearization-identity-multipliers}--\eqref{eq:endpoint-busemann-addition-law} show that the nonlinear dynamics of the scale chain are completely diagonalized as pure multiplicative rescaling in the endpoint coordinate \(\mathfrak{w}\), degenerating to additive time evolution on the precision potential \(y=-\log|\mathfrak{w}|\).
Furthermore, the scale superposition operator \(\mathcal{T}\) is completely diagonalized in the Mellin frequency domain by \(\zeta(s)\) (equation~\eqref{eq:scale-mellin-diagonalization}), and the prime structure is realized at the operator level as the irreducible generator decomposition \eqref{eq:scale-euler-operator-product}.
Thus the three types of non-compactness---``infinite height/primes/Euler product''---can be uniformly organized within the same endpoint dynamical system: geometrically as the precision load at the endpoint \(\mathfrak{w}=0\), and arithmetically as the frequency-domain multiplier of the scale transfer operator and the prime generator decomposition.
