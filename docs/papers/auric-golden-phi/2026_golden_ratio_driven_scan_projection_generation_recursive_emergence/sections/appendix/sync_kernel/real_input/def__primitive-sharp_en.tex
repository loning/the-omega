\begin{definition}[Primitive sharp: Perron-stripped primitive]\label{def:primitive-sharp}
Following the notation of Proposition~\ref{prop:finite-rh-sqrt-resonance-general}, define
\[
p_n^\sharp
:=\frac{1}{n}\sum_{d\mid n}\mu(d)\,\bigl(a_{n/d}-\lambda^{n/d}\bigr),
\qquad
\mathsf{E}_n^\sharp
:=\frac{n}{\lambda^{n/2}}\,p_n^\sharp.
\]
This is equivalent to stripping the Perron mode \(\lambda^k\) from \((a_k)\) term by term in ghost coordinates and then applying the Witt/M\"obius inversion; hence \(p_n^\sharp\) is a \textbf{purely non-Perron spectrum driven} primitive observable.
\end{definition}

\begin{proposition}[Pure phase limit after Perron stripping]\label{prop:primitive-sharp-phase-limit}
Under the hypotheses of Proposition~\ref{prop:finite-rh-sqrt-resonance-general}, if the critical modulus spectrum \(\mathcal{C}\) is semisimple on \(|\mu|=\sqrt{\lambda}\), then as \(n\to\infty\),
\[
p_n^\sharp
=\frac{\lambda^{n/2}}{n}\sum_{\mu\in\mathcal{C}}\omega_\mu^{\,n}
+O\!\left(\frac{\Lambda^n}{n}\right),
\qquad
\mathsf{E}_n^\sharp
=\sum_{\mu\in\mathcal{C}}\omega_\mu^{\,n}
+O\!\left(\Bigl(\frac{\Lambda}{\sqrt{\lambda}}\Bigr)^n\right),
\]
where \(\omega_\mu=\mu/\sqrt{\lambda}\) and \(\Lambda=\max\{R,\lambda^{1/3}\}\) is as in Proposition~\ref{prop:finite-rh-sqrt-resonance-general}.
\end{proposition}
\begin{proof}
By the spectral decomposition in the proof of Proposition~\ref{prop:finite-rh-sqrt-resonance-general},
for any \(k\ge 1\) one has
\(
a_k-\lambda^k=\lambda^{k/2}\sum_{\mu\in\mathcal{C}}\omega_\mu^{\,k}+O(R^k)
\).
Substituting into the definition yields
\[
np_n^\sharp=\sum_{d\mid n}\mu(d)\,\bigl(a_{n/d}-\lambda^{n/d}\bigr)
=(a_n-\lambda^n)+\sum_{\substack{d\mid n\\ d\ge 2}}\mu(d)\,\bigl(a_{n/d}-\lambda^{n/d}\bigr).
\]
The first term equals \(\lambda^{n/2}\sum_{\mu\in\mathcal{C}}\omega_\mu^{\,n}+O(R^n)\).
For \(d\ge 2\), setting \(k=n/d\le n/2\), one has
\(
|a_k-\lambda^k|
\ll \lambda^{k/2}+R^k
\ll \lambda^{n/4}+R^{n/2}
\ll \Lambda^n
\)
(where \(\Lambda=\max\{R,\lambda^{1/3}\}\) and \(\Lambda<\sqrt{\lambda}\)).
Thus
\(
np_n^\sharp
=\lambda^{n/2}\sum_{\mu\in\mathcal{C}}\omega_\mu^{\,n}+O(\Lambda^n)
\);
dividing by \(n\) and renormalizing yields the conclusion. \qed
\end{proof}

\begin{remark}[Multi-frequency quasi-random limit: the limit distribution of \texorpdfstring{$\mathsf{E}_n^\sharp$}{E_n^sharp}]\label{rem:primitive-sharp-limit-law}
Under the hypotheses of Proposition~\ref{prop:primitive-sharp-phase-limit}, suppose the critical modulus spectrum (counted with algebraic multiplicity) is
\[
\mu_j=\sqrt{\lambda}\,e^{i\alpha_j}\qquad(j=1,\dots,J),
\]
and assume \(\alpha_1,\dots,\alpha_J\) are rationally independent over \(2\pi\) (equivalently, \(\{\alpha_j/2\pi\}\) are linearly independent over \(\QQ\)).
Let
\(
S_n:=\sum_{j=1}^{J}e^{in\alpha_j}
\).
Then \((n\alpha_1,\dots,n\alpha_J)\bmod 2\pi\) is equidistributed on \(\mathbb{T}^J\) (Kronecker/Weyl), so for any continuous function \(f:\CC\to\CC\),
\[
\lim_{N\to\infty}\frac{1}{N}\sum_{n=1}^{N} f(S_n)
=\int_{\mathbb{T}^J} f\!\left(\sum_{j=1}^{J}e^{i\theta_j}\right)\,d\theta,
\]
where \(d\theta\) is the Haar probability measure on \(\mathbb{T}^J\).
Since Proposition~\ref{prop:primitive-sharp-phase-limit} gives \(\mathsf{E}_n^\sharp=S_n+o(1)\), the sequence \(\mathsf{E}_n^\sharp\) has the same limit distribution: it is the pushforward of the Haar measure under the map \((\theta_1,\dots,\theta_J)\mapsto\sum_{j=1}^{J}e^{i\theta_j}\).
\end{remark}

\begin{remark}[``Super-strengthened'' leading sub-term for even lengths]\label{rem:finite-rh-even-strengthen}
At even lengths, the $\lambda^{n/2}$ term is exactly cancelled by $d=1$ and $d=2$. A case-by-case analysis yields a stronger leading sub-term:
\[
p_n=\frac{\lambda^n}{n}-\frac{\lambda^{n/3}}{n}+O\!\left(\frac{\lambda^{n/4}}{n}\right),
\qquad n\ \text{even and}\ 3\mid n,
\]
\[
p_n=\frac{\lambda^n}{n}+(-1)^{n/2+1}\frac{\lambda^{n/4}}{n}
O\!\left(\frac{\lambda^{n/5}}{n}\right),
\qquad n\ \text{even and}\ 3\nmid n.
\]
In the first identity, $-\lambda^{n/3}$ comes from the $d=3$ term ($\mu(3)=-1$); in the second, $\lambda^{n/4}$ comes from the sub-leading oscillation in the $d=2$ term
$(-\varphi)^{n/2}=(-1)^{n/2}\lambda^{n/4}$; and $\mu(4)=0$ prevents $d=4$ from contributing at the leading sub-level.
\end{remark}

\begin{remark}[The sub-leading term is jointly locked by ``spectral information $\times$ factor structure'']\label{rem:finite-rh-factor-controls-secondary}
On this kernel, the only nontrivial eigenvalue satisfying $|\mu|=\sqrt{\lambda}$ is $-\sqrt{\lambda}$, so the ``square-root-level error'' and the ``even-length cancellation'' on the primitive side are both jointly locked by the critical modulus spectral phase and the $d=1,2$ terms of the M\"obius inversion (Proposition~\ref{prop:finite-rh-sqrt-resonance-general} and its special case Corollary~\ref{cor:finite-rh-explicit-formula}).
Once this level is cancelled, the new leading sub-term no longer depends solely on the spectrum but also explicitly on the factor structure of $n$: for any odd prime factor $p\mid n$, the $d=p$ term in the M\"obius inversion contributes a correction of magnitude $\lambda^{n/p}/n$ (with sign determined by $\mu(p)=-1$).
\par
More compactly, define
\[
p_{\min}(n):=\min\{p:\ p\ \text{is an odd prime and}\ p\mid n\},
\qquad(\text{set }p_{\min}(n)=+\infty\text{ if no odd prime factor exists}).
\]
Then for even $n$, after the exact cancellation of the $\lambda^{n/2}$-level by the $d=1,2$ terms, the exponential scale of the residual sub-leading term is determined by
\[
\max\Bigl(\frac14,\ \frac{1}{p_{\min}(n)}\Bigr):
\]
when $3\mid n$, the $d=3$ term gives $\lambda^{n/3}/n$; when $3\nmid n$, the sub-leading oscillation of the $d=2$ term gives $\lambda^{n/4}/n$; and larger odd prime factors contribute only lower-order $\lambda^{n/p}$ corrections.
\end{remark}

\begin{corollary}[Sharpness of the RH-type error (in both length and $x$ caliber)]\label{cor:finite-rh-prime-orbit-osc}
Let
\[
\Pi(N):=\sum_{n\le N}p_n,\qquad x:=\lambda^N.
\]
By Corollary~\ref{cor:finite-rh-explicit-formula},
\[
\Pi(N)=\frac{\lambda}{\lambda-1}\cdot\frac{\lambda^N}{N}+O\!\left(\frac{\lambda^{N/2}}{N}\right),
\]
and because of the actual existence of $(-\varphi)$, the square-root level in the above cannot be improved to $\lambda^{(1/2-\varepsilon)N}$. Equivalently,
\[
\pi(x)=\frac{\lambda\log\lambda}{\lambda-1}\cdot\frac{x}{\log x}
\;+\Omega_{\pm}\!\left(\frac{x^{1/2}}{\log x}\right),
\]
with oscillation frequency given by $\Im(s)=\frac{(2k+1)\pi}{\log\lambda}$ (period $2\pi/\log\lambda$).
\end{corollary}

\begin{proof}
From the closed form
\(
a_n=\varphi^{2n}+(-\varphi)^n+2+3(-1)^n+\varphi^{-2n}+\varphi^{-n}
\)
one obtains
\(
a_n=\lambda^n+(-1)^n\lambda^{n/2}+O(1)
\).

By M\"obius inversion $p_n=\frac1n\sum_{d\mid n}\mu(d)a_{n/d}$:
if $n$ is odd, then all $d\ge 2$ satisfy $d\ge 3$, so $a_{n/d}=O(\lambda^{n/3})$. Thus
\[
np_n=a_n+\sum_{\substack{d\mid n\\ d\ge 3}}\mu(d)a_{n/d}
=\lambda^n-\lambda^{n/2}+O(\lambda^{n/3}),
\]
giving the first identity.

If $n$ is even, write out the $d=1,2$ terms:
\[
np_n=a_n-\;a_{n/2}+\sum_{\substack{d\mid n\\ d\ge 3}}\mu(d)a_{n/d}.
\]
Here $a_n=\lambda^n+\lambda^{n/2}+O(1)$ (since $(-1)^n=1$), while $a_{n/2}=\lambda^{n/2}+O(\lambda^{n/4})$, so
\[
a_n-a_{n/2}=\lambda^n+O(\lambda^{n/4}),
\]
and the remainder from $d\ge 3$ gives $O(\lambda^{n/3})$, yielding the second identity. Here the cancellation of $\lambda^{n/2}$ is precisely the exact cancellation between the square-root term of $a_n$ ($d=1$) and the leading term of $a_{n/2}$ ($d=2$).
\end{proof}
