\paragraph{Rationality of the multivariate generating function for linear constraint counting: explicit cancellation with characteristic polynomial denominator}

\begin{equation}\label{eq:gm-genfunc-resolvent}
\boxed{\

=u_{Q,b}^{\mathsf T}\,(I-zB_{Q,b})^{-1}v_{Q,b}\ } ,

\begin{proof}

yielding \eqref{eq:gm-genfunc-resolvent}.
The pole containment follows from the determinant denominator of the matrix resolvent; the exponential-polynomial representation comes from the Jordan decomposition and the Cayley--Hamilton theorem. \qed
\end{proof}

\paragraph{Mellin analytification of the matrix subdivision equation: meromorphic continuation on \texorpdfstring{$\varphi^{-s}$}{phi^{-s}} and pole spectral fingerprint}
\begin{theorem}[Mellin analytification of the matrix subdivision equation: spectral fingerprint on \texorpdfstring{$\varphi^{-s}$}{phi^{-s}}]\label{thm:gm-mellin-meromorphic-fingerprint}

\begin{equation}\label{eq:gm-matrix-refinement-abstract}
\vec\mu=\sum_{j\in\mathcal{J}}T_j\cdot(\vec\mu\circ S_j^{-1}),
\qquad
S_j(x)=\varphi^{-a_j}x+t_j,
\end{equation}
where \(a_j\in\NN\) are fixed, \(t_j\in[0,1]\), and \(T_j\) are non-negative matrices (of finite dimension).
Let
\[
M(s):=\int_0^1 x^{s}\,d\mu(x),\qquad \Re(s)>-1,
\]
and let \(z:=\varphi^{-s}\) (taking the principal branch in the \(z\)-plane).
Then there exists a finite-dimensional linear system
\begin{equation}\label{eq:gm-mellin-linear-system}
\boxed{\ (I-P(z))\,\mathbf{M}(s)=\mathbf{H}(s)\ },
\qquad
P(z):=\sum_{\substack{j\in\mathcal{J}\\ t_j=0}} z^{a_j}\,T_j,
\end{equation}
where \(\mathbf{M}(s)\) is a vector consisting of finitely many ``state Mellin moments'' (including the component of \(M(s)\) in the Perron direction), and \(\mathbf{H}(s)\) is analytic in the half-plane \(\Re(s)>-1\) and extends as an entire function in the \(s\)-plane.
Therefore, \(\mathbf{M}(s)\) (and hence \(M(s)\)) is meromorphic as a function of \(z\), and its pole set is contained in
\[
\det(I-P(z))=0;
\]
the order of each pole is determined by the Jordan structure of \(P(z)\) at that point.
This pole data constitutes the ``Mellin spectral fingerprint of the limit distribution'': endpoint-type (\(x\to 0^+\)) power-law/logarithmic corrections and moment growth can be read off from the finite matrix spectrum in an offline caliber.
\end{theorem}

\begin{proof}[Key ideas]
Take the Mellin transform (or componentwise \(\int x^s\)) of \eqref{eq:gm-matrix-refinement-abstract}:
when \(t_j=0\), \(\int (S_j x)^s\,d\mu=\varphi^{-a_js}\int x^s\,d\mu\) gives a strict \(z^{a_j}\) coefficient;
when \(t_j>0\), \(x\mapsto (t_j+\varphi^{-a_j}x)^s\) is analytic in \(\Re(s)>-1\) and uniformly controlled for \(x\in[0,1]\), contributing an analytic function term that is absorbed into \(\mathbf{H}(s)\).
Moving all \(t_j=0\) branch terms to the left-hand side yields \eqref{eq:gm-mellin-linear-system}.
Meromorphicity and the pole set follow from the matrix resolvent \((I-P(z))^{-1}\); the pole order is determined by the Jordan structure. \qed
\end{proof}

\begin{corollary}[Mellin pole golden comb cluster: finitely many arithmetic vertical lines]\label{cor:gm-mellin-pole-golden-comb}

\end{corollary}

\begin{proof}
By Theorem \ref{thm:gm-mellin-meromorphic-fingerprint}, \(\mathbf{M}(s)\) as a function of \(z=\varphi^{-s}\) can only have poles at \(D(z)=0\).
For a given \(z_0\neq 0\), solving \(\varphi^{-s}=z_0\) is equivalent to
\(
-s\log\varphi=\log z_0-2\pi i k
\)
(multivaluedness of the complex logarithm), yielding the stated arithmetic vertical tower.
\end{proof}

\begin{corollary}[Linear upper bound on Mellin pole count: explicit constant controlled by \texorpdfstring{$\deg D$}{deg D}]\label{cor:gm-mellin-pole-count-linear}

\end{proof}

\paragraph{Jensen profile and pole skeleton: convex encoding and residue visualization via radial average of the determinant}
\begin{proposition}[Jensen profile of the matrix Mellin fingerprint: \texorpdfstring{$\det(I-P(z))$}{det(I-P(z))} radial average is completely solvable]\label{prop:gm-mellin-jensen-profile-det}
Under the setting of Theorem \ref{thm:gm-mellin-meromorphic-fingerprint}, let
\[
D(z):=\det(I-P(z))\in\CC[z]
\]
be the polynomial, and write its zeros (with multiplicity) as \(\{z_j\}_{j=1}^{J}\subset\CC\), with leading coefficient \(c\neq 0\) (i.e., \(D(z)=c\prod_{j=1}^{J}(z-z_j)\)).
Define the Jensen profile (vertical line average) as
\[
\mathcal{M}(\sigma)
:=\frac{\log\varphi}{2\pi}\int_{0}^{2\pi/\log\varphi}
\log\Bigl|D\bigl(\varphi^{-(\sigma+\mathrm{i}t)}\bigr)\Bigr|\,\dd t,
\qquad \sigma\in\RR.
\]
Then there is a complete closed form
\[
\boxed{\;
\mathcal{M}(\sigma)=\log|c|+\sum_{j=1}^{J}\log\max\!\bigl\{\varphi^{-\sigma},|z_j|\bigr\}\; }.
\]
In particular, \(\mathcal{M}\) is a convex piecewise affine function with breakpoint set exactly
\(
\{\,-\log|z_j|/\log\varphi\,\}_{j=1}^{J}
\)
(counted with multiplicity).
Therefore the breakpoint spectrum of \(\mathcal{M}\) on the \(\sigma\)-axis precisely encodes the (real-part set of) Mellin pole vertical towers induced by
\(\det(I-P(\varphi^{-s}))\).
\end{proposition}
\begin{proof}
Let \(r:=\varphi^{-\sigma}\).
As \(t\) varies over \([0,2\pi/\log\varphi]\),
\(
z=\varphi^{-(\sigma+\mathrm{i}t)}=r\,e^{-\mathrm{i}t\log\varphi}
\)
traces the circle \(|z|=r\) once.
Thus \(\mathcal{M}(\sigma)\) equals \(\frac{1}{2\pi}\int_{0}^{2\pi}\log|D(re^{\mathrm{i}\theta})|\,\dd\theta\).
Applying Jensen's formula for polynomials (equivalently, computing the circular average of \(\log|z-z_j|\) term by term) yields
\[
\frac{1}{2\pi}\int_{0}^{2\pi}\log|D(re^{\mathrm{i}\theta})|\,\dd\theta
=\log|c|+\sum_{j=1}^{J}\log\max\{r,|z_j|\},
\]
giving the stated closed form.
Convexity and breakpoint locations follow from the piecewise linear structure of \(\log\max\{r,|z_j|\}=\max\{-\sigma\log\varphi,\log|z_j|\}\). \qed
\end{proof}

\begin{proposition}[Pole vertical tower skeleton and residue closed form: explicit partial fraction visualization via adjugate matrix for \texorpdfstring{$\mathbf{M}(s)$}{M(s)}]\label{prop:gm-mellin-pole-skeleton-residue-adjugate}
Under the setting of Theorem \ref{thm:gm-mellin-meromorphic-fingerprint}, still write \(D(z)=\det(I-P(z))\).
If all zeros of \(D\) in \(\CC^\times\) are simple roots \(z_1,\dots,z_J\) (counted with multiplicity), then for each \(j\), any choice of logarithm branch yields some \(s_{j,0}\) satisfying \(\varphi^{-s_{j,0}}=z_j\), with all solutions given by
\[
s_{j,k}:=s_{j,0}+\frac{2\pi\mathrm{i}}{\log\varphi}\,k\qquad(k\in\ZZ),
\]
and \(\mathbf{M}(s)\) can only have poles at \(\{s_{j,k}\}_{j,k}\) (except possibly removable singularities).
In the non-cancellation case, \(\mathbf{M}(s)\) has a simple pole at each \(s_{j,k}\), with residue closed form
\[
\boxed{\;
\operatorname*{Res}_{s=s_{j,k}}\mathbf{M}(s)
=-\frac{1}{\log\varphi}\cdot
\frac{\operatorname{adj}(I-P(z_j))}{z_j\,D'(z_j)}\,
\mathbf{H}(s_{j,k})\; }.
\]
Equivalently, writing \(\partial_{\log z}D(z):=zD'(z)\) for the logarithmic derivative,
\[
\operatorname*{Res}_{s=s_{j,k}}\mathbf{M}(s)
=-\frac{1}{\log\varphi}\cdot
\frac{\operatorname{adj}(I-P(z_j))}{\partial_{\log z}D(z_j)}\,
\mathbf{H}(s_{j,k}).
\]
\end{proposition>
\begin{proof}
By \eqref{eq:gm-mellin-linear-system},
\[
\mathbf{M}(s)=(I-P(z))^{-1}\mathbf{H}(s)=\frac{\operatorname{adj}(I-P(z))}{D(z)}\,\mathbf{H}(s),
\qquad z=\varphi^{-s}.
\]
When \(D(z_j)=0\) is a simple root, \(D(z)=D'(z_j)(z-z_j)+O((z-z_j)^2)\), and \(\operatorname{adj}(I-P(z))\) is analytic at \(z_j\).
Hence in the \(z\)-variable,
\(
(I-P(z))^{-1}
\)
has a simple pole at \(z=z_j\) with principal part coefficient \(\operatorname{adj}(I-P(z_j))/D'(z_j)\).
\par
Substituting \(z=\varphi^{-s}\) and expanding near \(s_{j,k}\):
\(
\varphi^{-s}-z_j=-(\log\varphi)z_j\,(s-s_{j,k})+o(s-s_{j,k})
\).
This yields the residue coefficient in the \(s\)-variable as \(-1/(\log\varphi\,z_j)\), which when multiplied by the \(z\)-variable principal part coefficient gives the stated closed form.
If at some \(s_{j,k}\) we have exactly \(\operatorname{adj}(I-P(z_j))\mathbf{H}(s_{j,k})=0\), then that pole is removable; otherwise it is a simple pole. \qed
\end{proof}

\)
and take \(W=\{t\in[0,T]:|D(t)|\ge V\}\) to be the \(1\)-separated large value point set.
Then at the scaling \(T=N^{6/5}\) and threshold \(V\asymp N^{3/4}\), there exists \(\epsilon>0\) (determined only by the spectral gap data and the automaton) such that
\[
|W|\ \ll\ N^{12/5-\epsilon}\,V^{-4}.
\]
\end{corollary}

\begin{proof}[Key ideas]
Combining the third trace/higher-order moment certificate chain (\S\ref{subsubsec:gm-third-trace-dual-certificates} and \S\ref{subsubsec:gm-affine-inverse-gram-arithmetic-dichotomy}) with the non-asymptotic moment saving of Theorem \ref{thm:gm-automatic-dirichlet-moment-bound}, one can break through the bottleneck term caused by the energy coupling channel at the critical interval, and read off the stated power saving at the \(T=N^{6/5}\) scaling. \qed
\end{proof}

\paragraph{Compilability of multiplicative energy and sum--product exponent law: double energy complementarity}
\begin{theorem}[Compilability of multiplicative energy for Zeckendorf--sofic: Perron fingerprint of product equation counting (conditional version)]\label{thm:gm-multiplicative-energy-compilable}
Let \(A_m=\{\mathrm{val}(\omega):\omega\in\mathcal{L}_m\}\) come from a primitive sofic--Zeckendorf system.
If there exists a ``multiplicative normalization transducer'' (implementing multiplication carry/normalization in finite states, coupled with the four-fold synchronization product of \(\mathcal{L}\)), then there exist a finite non-negative integer matrix \(B_{\times}\) and vectors \(u,v\ge 0\) such that
\[
E_{\times}(A_m):=\#\{a_1a_2=a_3a_4:\ a_i\in A_m\}
=u^{\mathsf T}B_{\times}^{\,m}v,
\]
so that \(\lim_{m\to\infty}m^{-1}\log E_\times(A_m)=\log\rho(B_\times)\) exists and satisfies a finite-order linear recurrence closure.
\end{theorem}

\begin{theorem}[Spectral form of additive and multiplicative double energy complementarity: strict interior domain constraint under tensor transducer]\label{thm:gm-add-mult-energy-complementarity}
Under the conditions of Theorem \ref{thm:gm-multiplicative-energy-compilable}, write
\[
|A_m|\asymp \alpha^m,\qquad
E_+(A_m)\asymp \beta_+^m,\qquad
E_\times(A_m)\asymp \beta_\times^m.
\]
Then there exists a constant \(\kappa>0\) depending only on \(\mathcal{L}\) such that
\[
\boxed{\ \log\beta_+ + \log\beta_\times\ \le\ 6\log\alpha-\kappa\ }.
\]
Therefore the same Zeckendorf--sofic set cannot, in the exponential sense, simultaneously be close to ``additively highly structured'' and ``multiplicatively highly structured''; this complementarity is strictly enforced by the finite-state tensor spectral radius.
\end{theorem}

\begin{corollary}[Sum--product exponent law for Fibonacci-base: exponential expansion from double energy complementarity]\label{cor:gm-sumproduct-exponent-law}
Let \(\Sigma_m=A_m+A_m\) and \(\Pi_m=A_m\cdot A_m\).
Under the hypothesis of Theorem \ref{thm:gm-add-mult-energy-complementarity}, there exists \(\delta>0\) such that for sufficiently large \(m\),
\[
\max\{|\Sigma_m|,|\Pi_m|\}\ \ge\ |A_m|^{1+\delta}
\]
(expansion in the exponential sense), and \(\delta\) can be explicitly bounded from below by \(\kappa\) and \(\alpha\):
\(
\delta\ge \kappa/(2\log\alpha)
\).
\end{corollary}

\paragraph{Double large sieve input and cyclotomic factors of graph \texorpdfstring{$\zeta$}{zeta}: algebraic discrimination of modular obstructions}
\begin{proposition}[Unified form of double large sieve input: simultaneous spectral gapping of additive and multiplicative characters]\label{prop:gm-double-large-sieve-input}
Let \((b_n)\) be generated by the same finite-state system, and suppose both its additive character twist and multiplicative Dirichlet character twist transfer operators satisfy uniform spectral gaps (equivalent to excluding the finite obstruction family).
Then there exist \(\delta_1,\delta_2>0\) such that for any fixed \(\theta>0\), any \(Q\le N^\theta\), and any non-trivial Dirichlet character \(\chi\ (\bmod q)\) (\(q\le Q\)) and additive character \(e(an/q)\) (same \(q\le Q\)),
\[
\sum_{n\sim N} b_n\chi(n)\ \ll\ N^{1-\delta_1},
\qquad
\sum_{n\sim N} b_n e(an/q)\ \ll\ N^{1-\delta_2},
\]
with implied constants depending only on the finite-state spectral data.
\end{proposition}

\begin{theorem}[Cyclotomic factor classification of the graph \texorpdfstring{$\zeta$}{zeta} function: \texorpdfstring{$\zeta$}{zeta}--spectral equivalence characterization of modular obstructions]\label{thm:gm-graph-zeta-cyclotomic-classification}
Let \(G\) be a primitive directed graph with edge weights \(w\) inducing mod \(q\) character-twisted adjacency operators \(A(\chi)\).
Define the twisted graph \(\zeta\) function (dynamical \(\zeta\)):
\[
Z_G(u,\chi):=\prod_{[\gamma]}\bigl(1-u^{\ell(\gamma)}\chi(w(\gamma))\bigr)^{-1},
\]
where the product runs over prime cycle homotopy classes \([\gamma]\).
Then there exists an explicit polynomial \(P_\chi(u)\) such that
\[
Z_G(u,\chi)=\frac{P_\chi(u)}{\det(I-uA(\chi))}.
\]
Moreover, a non-trivial modular obstruction exists if and only if \(Z_G(u,\chi)\) contains a cyclotomic factor for some \(\chi\ne 1\); equivalently,
\[
\rho(A(\chi))=\rho(A(1))
\]
holds for some non-trivial \(\chi\).
Therefore the entire source of ``finite modular obstructions'' can be equivalently reduced to an enumerable cyclotomic factor structure.
\end{theorem}

\begin{proof}[Key ideas]
For SFT/sofic dynamical \(\zeta\), there is a standard determinant representation \(Z(u)=\exp(\sum_{n\ge 1}\mathrm{tr}(A^n)u^n/n)=1/\det(I-uA)\) (see \cite[Chapter~6]{LindMarcus1995}); the twisted case replaces \(A\) by \(A(\chi)\) in the denominator.
The numerator \(P_\chi\) absorbs corrections from finite prefixes/transient states (in the primitive case it can be taken as a constant).
Cyclotomic factors correspond to roots on the unit circle, hence to iso-radius between twisted and untwisted spectral radii; their equivalence with the modular coboundary/obstruction is given by the structural characterization of Proposition \ref{prop:gm-cycle-gcd-mod-obstruction} and Theorem \ref{thm:gm-uniform-twist-gap-from-gcd}. \qed
\end{proof}
