\section{Auxiliary Derivations and Standard Results}

\subsection{Poisson Source Implies a 1/r Phase Potential}
\label{app:phase_poisson}
Let $\Phi$ be a static scalar potential on $\RR^3$ solving the Poisson equation in the sense of distributions,
\begin{equation}
  \Delta \Phi = 4\pi \mathcal{Q}\,\delta_0 .
\end{equation}
Then $\Phi$ is harmonic on $\RR^3\setminus\{0\}$ and, assuming spherical symmetry, satisfies for $r>0$ the ODE
\begin{equation}
  0=\Delta \Phi(r)=\frac{1}{r^2}\frac{d}{dr}\left(r^2\Phi'(r)\right).
\end{equation}
Thus $r^2\Phi'(r)=C$ and $\Phi(r)= -C/r + \mathrm{const}$. The constant is fixed by the flux normalization:
integrating $\Delta\Phi=4\pi\mathcal{Q}\delta_0$ over a ball $B_R$ and applying the divergence theorem gives
\begin{equation}
  4\pi\mathcal{Q}=\int_{B_R}\Delta\Phi\,dV=\int_{\partial B_R}\nabla\Phi\cdot d\mathbf{A}=4\pi R^2\Phi'(R),
\end{equation}
so $R^2\Phi'(R)=\mathcal{Q}$ and therefore
\begin{equation}
  \Phi(r)=-\frac{\mathcal{Q}}{r}+\mathrm{const}.
\end{equation}
This is the standard three-dimensional $1/r$ kernel, with the overall sign fixed by the chosen Poisson convention.

\subsection{Dirichlet Principle for the Static Phase Potential}
\label{app:dirichlet_poisson}
Let $\rho_\Phi$ be a prescribed (compactly supported) source density on $\RR^3$ and consider the functional
\begin{equation}
  \mathcal{F}[\Phi]
  :=\int_{\RR^3}\left(\frac{1}{8\pi}|\nabla \Phi|^2+\rho_\Phi\,\Phi\right)d^3x,
\end{equation}
over sufficiently regular test functions with $\Phi(\mathbf{x})\to 0$ as $|\mathbf{x}|\to\infty$.
Varying $\Phi\mapsto \Phi+\epsilon\,\eta$ with compactly supported $\eta$ and integrating by parts gives
\begin{equation}
  \frac{d}{d\epsilon}\mathcal{F}[\Phi+\epsilon\eta]\Big|_{\epsilon=0}
  =\int_{\RR^3}\left(-\frac{1}{4\pi}\Delta\Phi+\rho_\Phi\right)\eta\,d^3x,
\end{equation}
so stationarity for all $\eta$ implies the Poisson equation
\begin{equation}
  \Delta\Phi=4\pi\rho_\Phi.
\end{equation}
This variational characterization is standard in potential theory; see, e.g., \cite{Jackson1998} for the electrostatic case (note that sign conventions for Poisson's equation may differ).

\subsection{Poisson Equation Implies Gauss Law}
\label{app:poisson_gauss}
Assume $\Phi$ is sufficiently regular on a bounded region $V\subset\RR^3$ with smooth boundary $\partial V$ and satisfies
\begin{equation}
  \Delta \Phi = 4\pi \rho_\Phi
\end{equation}
in the distributional sense on $V$. Integrating and applying the divergence theorem gives
\begin{equation}
  \oint_{\partial V}\nabla\Phi\cdot d\mathbf{A}
  =\int_V \Delta\Phi\,d^3x
  =4\pi\int_V\rho_\Phi\,d^3x.
\end{equation}
Thus, defining the total phase charge in $V$ by $\mathcal{Q}_\Phi(V):=\int_V\rho_\Phi\,d^3x$, one obtains the Gauss-law form
\begin{equation}
  \oint_{\partial V}\nabla\Phi\cdot d\mathbf{A}=4\pi \mathcal{Q}_\Phi(V),
\end{equation}
which in particular yields $\oint\nabla\Phi\cdot d\mathbf{A}=4\pi \mathcal{Q}$ for a point source $\rho_\Phi=\mathcal{Q}\delta_0$.
See \cite{Jackson1998} for the standard potential-theoretic formulation.

\subsection{Irrational Rotation Coding and Sturmian Sequences}
\label{app:sturmian}
Fix $\alpha\in(0,1)\setminus\QQ$ and $\beta\in[0,1)$. Consider the Kronecker orbit $x_n=\{n\alpha+\beta\}\in[0,1)$ and the two-interval partition $[0,1-\alpha)\cup[1-\alpha,1)$. Its itinerary is the binary sequence
\begin{equation}
  s_n := \lfloor (n+1)\alpha+\beta\rfloor-\lfloor n\alpha+\beta\rfloor\in\{0,1\}.
\end{equation}
This $s_n$ is a (lower) \emph{mechanical word}. A classical theorem in symbolic dynamics states that for irrational $\alpha$, mechanical words are precisely the (aperiodic) \emph{Sturmian} sequences, characterized equivalently by minimal subword complexity $p(m)=m+1$ among all aperiodic bi-infinite binary sequences \cite{MorseHedlund1940, Lothaire2002}.

For coarse-grained observables, Weyl equidistribution implies the orbit $\{n\alpha+\beta\}$ is uniformly distributed mod $1$ \cite{Weyl1916}, so single-step statistics can appear thermal even when the underlying sequence is deterministic. The information capacity in Omega Dynamics is attributed to the long-range correlations of the itinerary (which are not visible to marginal statistics).

\subsection{Modular Flow Is Unitary}
\label{app:modular_unitary}
In Tomita--Takesaki theory, given a von Neumann algebra $\mathcal{A}$ and cyclic separating vector $|\Omega\rangle$, the modular operator $\Delta_\Omega$ is positive self-adjoint and $\Delta_\Omega^{it}$ is unitary for every $t\in\RR$, implementing the modular automorphism group $\sigma_t(A)=\Delta_\Omega^{it}A\Delta_\Omega^{-it}$ \cite{Takesaki2002}.

\subsection{Newtonian Limit of the Einstein Equation}
\label{app:newtonian_limit}
Assume a weak, static field in which the metric takes the standard Newtonian form
\begin{equation}
  ds^2 = -(1+2\phi)\,dt^2 + (1-2\phi)\,\delta_{ij}\,dx^i dx^j,
  \qquad |\phi|\ll 1,
\end{equation}
with $\phi=\phi(\mathbf{x})$ and negligible pressures, so that $T_{00}\approx \rho_m$ and other components are subleading.
Then one has, to leading order in $\phi$,
\begin{equation}
  G_{00} = 2\,\Delta \phi + O(\phi\,\partial^2\phi,(\partial\phi)^2),
\end{equation}
so the field equation $G_{\mu\nu}=\kappa_G T_{\mu\nu}$ reduces to
\begin{equation}
  \Delta \phi = \frac{\kappa_G}{2}\,\rho_m,
\end{equation}
which fixes $\kappa_G=8\pi$ by matching to $\Delta\phi = 4\pi\rho_m$ in geometric units \cite{Wald1984}.
