\paragraph{Horizon as "Quotient Data Structure": External Observable Algebra, Equivalence Class Fibers, and Structured Statement of Capacity/Holography}
Theorem \ref{thm:dephys-pw-nohair} excludes additional continuous registers parallel to phase; under this caliber, the "horizon" can be understood as a quotient structure induced by the external observable algebra: external readout only depends on an accessible operator algebra (or visible \(\sigma\)-algebra), while numerous internal degrees of freedom are compressed into the same external equivalence class fiber.
The following conclusions align the four interpretive semantics of "no-hair—area law—holography—zero-knowledge"; they serve no role as derivation premises.

\begin{definition}[External Observable Algebra and External Equivalence Relation]\label{def:dephys-ext-observable-algebra-equivalence}
Let \(\mathcal{H}_{\mathrm{bulk}}\) be the total Hilbert space containing "internal-external" degrees of freedom, and let \(\mathcal{A}_{\mathrm{ext}}\subseteq \mathcal{B}(\mathcal{H}_{\mathrm{bulk}})\) be the \(\ast\)-algebra (external measurable algebra) accessible to external observers.
Denote the density operator state space by
\[
\mathcal{D}(\mathcal{H}_{\mathrm{bulk}})
:=\{\rho\succeq 0:\ \operatorname{Tr}\rho=1\}.
\]
For \(\rho,\tilde\rho\in\mathcal{D}(\mathcal{H}_{\mathrm{bulk}})\), define the external equivalence relation
\begin{equation}\label{eq:dephys-ext-equivalence}
\boxed{\;
\rho\sim_{\mathrm{ext}}\tilde\rho
\quad\Longleftrightarrow\quad
\operatorname{Tr}(\rho A)=\operatorname{Tr}(\tilde\rho A)\ \ (\forall\,A\in\mathcal{A}_{\mathrm{ext}}).\;}
\end{equation}
Denote the state space on \(\mathcal{A}_{\mathrm{ext}}\) by \(\mathrm{St}(\mathcal{A}_{\mathrm{ext}})\):
\[
\mathrm{St}(\mathcal{A}_{\mathrm{ext}})
:=\{\omega:\mathcal{A}_{\mathrm{ext}}\to\CC:\ \omega\ge 0,\ \omega(I)=1\}.
\]
\end{definition}

\begin{proposition}[Horizon as "Quotient Data Structure": Equivalence Class Fibers Induced by External Observable Algebra]\label{prop:dephys-horizon-quotient-data-structure}
Under the setting of Definition \ref{def:dephys-ext-observable-algebra-equivalence}, there exists a natural quotient map (restriction map)
\begin{equation}\label{eq:dephys-ext-quotient-map}
\boxed{\;
\pi_{\mathrm{ext}}:\ \mathcal{D}(\mathcal{H}_{\mathrm{bulk}})\twoheadrightarrow \mathrm{St}(\mathcal{A}_{\mathrm{ext}}),
\qquad
(\pi_{\mathrm{ext}}(\rho))(A):=\operatorname{Tr}(\rho A)\; }.
\end{equation}
Moreover, for any \(\rho,\tilde\rho\in\mathcal{D}(\mathcal{H}_{\mathrm{bulk}})\),
\[
\boxed{\;
\rho\sim_{\mathrm{ext}}\tilde\rho
\quad\Longleftrightarrow\quad
\pi_{\mathrm{ext}}(\rho)=\pi_{\mathrm{ext}}(\tilde\rho).\;}
\]
Therefore, for any \(\sigma\in \mathrm{St}(\mathcal{A}_{\mathrm{ext}})\), its fiber \(\pi_{\mathrm{ext}}^{-1}(\sigma)\) can be understood as "the set of internal configurations that are externally indistinguishable" (external equivalence class).
\end{proposition}

\begin{proof}
By the definition in \eqref{eq:dephys-ext-quotient-map}, \(\pi_{\mathrm{ext}}(\rho)\) is precisely the linear functional induced by \(\rho\) on \(\mathcal{A}_{\mathrm{ext}}\); and \(\pi_{\mathrm{ext}}(\rho)=\pi_{\mathrm{ext}}(\tilde\rho)\) is equivalent to \(\operatorname{Tr}(\rho A)=\operatorname{Tr}(\tilde\rho A)\) for all \(A\in\mathcal{A}_{\mathrm{ext}}\), which is exactly \eqref{eq:dephys-ext-equivalence}.
This completes the proof.
\end{proof}

\begin{remark}[Isomorphism with Folding Pushforward: Probability Pushforward Under Commutative External Algebra]\label{rem:dephys-ext-quotient-as-pushforward}
When \(\mathcal{A}_{\mathrm{ext}}\) is commutative and can be identified as the function algebra \(L^\infty(X)\) on some finite set \(X\), \(\mathrm{St}(\mathcal{A}_{\mathrm{ext}})\) is equivalent to the probability simplex \(\Delta(X)\).
At this point, \(\pi_{\mathrm{ext}}\) degenerates to a pushforward operator "microscopic distribution \(\mapsto\) macroscopic distribution", which is completely isomorphic to the readout operator \(P_m=(\Fold_m)_*\) in Section \ref{subsec:pom-information-ledger}: external equivalence class fibers correspond to microscopic configurations under the same macroscopic readout, and the ledger three-term decomposition yields the strict identity "visible entropy + fiber capacity \(-\) residual information" (Theorem \ref{thm:pom-kl-ledger}).
\end{remark}
