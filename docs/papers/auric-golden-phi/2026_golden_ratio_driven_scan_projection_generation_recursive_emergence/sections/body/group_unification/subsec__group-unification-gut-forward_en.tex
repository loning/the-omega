\begin{definition}[Cyclotomic common slow-mode layer]\label{def:cyclotomic-common-slowmode-layer}

\[
\widehat{\Delta}_{\varrho}(w_m,s_m)=0.
\]
Denote the set of \(m\) satisfying this property by \(\mathcal{U}(\mathcal{R}_{\mathrm{obs}})\).
\end{definition}

\begin{theorem}[Cyclotomic specialization rigidity: infinitely many common slow-mode layers \texorpdfstring{$\Rightarrow$}{=>} existence of a unification sector]\label{thm:cyclotomic-specialization-rigidity}
If \(\mathcal{U}(\mathcal{R}_{\mathrm{obs}})\) is an infinite set, then there exists a nonconstant polynomial
\[
H(s,w)\in \QQ[s,w]\setminus\QQ^\times
\]
such that \(H\mid \widehat{\Delta}_{\varrho}\) (divisibility in \(\QQ[s,w]\)) for all \(\varrho\in\mathcal{R}_{\mathrm{obs}}\).
In other words,
\[
\gcd_{\QQ[s,w]}\bigl(\widehat{\Delta}_{\varrho}:\varrho\in\mathcal{R}_{\mathrm{obs}}\bigr)\ \neq\ 1,
\]
and we call this greatest common divisor (in monic representative) the \textbf{unification sector}.
\end{theorem}
\begin{proof}
First take any two channels \(\varrho_1,\varrho_2\in\mathcal{R}_{\mathrm{obs}}\) and set
\[
E(s):=\operatorname{Res}_{w}\bigl(\widehat{\Delta}_{\varrho_1}(w,s),\widehat{\Delta}_{\varrho_2}(w,s)\bigr)\in\ZZ[s],
\]
where \(\operatorname{Res}_{w}\) denotes the resultant eliminating the variable \(w\).
If \(\widehat{\Delta}_{\varrho_1},\widehat{\Delta}_{\varrho_2}\) are coprime in \(\QQ[s,w]\), then \(E(s)\not\equiv 0\);
moreover, for any complex number \(s_0\), \(E(s_0)=0\) if and only if at \(s=s_0\) the two share a common \(w\)-root.

Iterating the gcd over the remaining channels in \(\mathcal{R}_{\mathrm{obs}}\) yields the common factor \(H\). \qed
\end{proof}

\begin{theorem}[Gcd stability of specialization: except for finitely many exceptions the common factor does not grow]\label{thm:cyclotomic-gcd-stability}

\end{theorem}
\begin{proof}
If for infinitely many \(m\) the gcd
\(
\gcd_{\mathbb{K}_m[w]}(Q_{\varrho}(w,s_m):\varrho\in\mathcal{R}_{\mathrm{obs}})
\)
is nontrivial, then there are infinitely many layers \(m\) at which \(\{Q_{\varrho}(w,s_m)\}\) share some common \(w\)-root.

\gcd_{\QQ[s,w]}(Q_{\varrho}:\varrho\in\mathcal{R}_{\mathrm{obs}})\neq 1
\), contradicting the construction.
Therefore extra gcd can only appear at finitely many \(m\); for the remaining \(m\), the common factor is \(H(s_m,w)\) (up to a unit in \(\mathbb{K}_m[w]\)). \qed
\end{proof}
