\begin{corollary}[Spectral Product Rigidity of Integer Gram Normalization: \texorpdfstring{$\prod_{p=1}^k4\sin^2(\phi_p)=1$}{prod 4 sin^2(phi)=1}]\label{cor:pom-Kk-det-sine-product-normalization}
Let \(K_k\in\ZZ^{k\times k}\) be defined by \(K_k(i,j)=\min(i,j)\), and let
\[
\phi_p:=\frac{(2p-1)\pi}{4k+2}\qquad(p=1,\dots,k).
\]
Then \(\det K_k=1\).
Thus from the spectral closed form \(\lambda_p(k)=\frac{1}{4\sin^2(\phi_p)}\) in Lemma \ref{lem:pom-Kk-eigenvalues}, we immediately obtain
\[
\boxed{\;\prod_{p=1}^{k}4\sin^2(\phi_p)=1.\;}
\]
\end{corollary}
\begin{proof}
Let \(B_k\in\ZZ^{k\times k}\) be the unit lower-triangular all-\(1\) matrix: \((B_k)_{ij}=1\) when \(i\ge j\), otherwise \(0\).
Then for any \(1\le i,j\le k\),
\[
(B_kB_k^{\mathsf T})_{ij}=\sum_{\ell=1}^{k}\mathbf{1}_{\{\ell\le i\}}\mathbf{1}_{\{\ell\le j\}}=\min(i,j),
\]
hence \(K_k=B_kB_k^{\mathsf T}\), thus \(\det K_k=(\det B_k)^2=1\).
On the other hand, the determinant of a real symmetric matrix equals the product of its eigenvalues, so
\(
\det K_k=\prod_{p=1}^{k}\lambda_p(k)
=\prod_{p=1}^{k}\frac{1}{4\sin^2(\phi_p)}
\);
combining with \(\det K_k=1\) yields the shown identity. QED.
\end{proof}
