\section{Problem 7: Uniform Lattices with 2-Torsion and $\mathbb Q$-Acyclic Covers}

\begin{theorem}\label{thm:q7}
Let $G$ be a connected real semi-simple Lie group with maximal compact subgroup $K$, and let $\Gamma\subset G$ be a uniform lattice all of whose torsion elements have order a power of~$2$. Set $d=\dim(G/K)$.
If $d\ge 5$, there exists a closed topological $d$-manifold $M$ with
\[
\pi_1(M)\cong\Gamma,\qquad \widetilde H_*(\widetilde M;\mathbb Q)=0.
\]
In particular, the answer to Problem~7 is \textbf{yes}: it is possible for $\Gamma$ to be the fundamental group of a compact manifold without boundary whose universal cover is $\mathbb Q$-acyclic.
\end{theorem}

\medskip
The proof proceeds in five steps.

\subsection*{Step 1: $\Gamma$ is a $d$-dimensional $\mathbb Q$-Poincar\'e duality group}

\begin{lemma}\label{lem:q7-qpd}
Under the hypotheses of Theorem~\ref{thm:q7},
\[
H^n(\Gamma;\mathbb Q[\Gamma])=
\begin{cases}\mathbb Q,&n=d,\\0,&n\ne d.\end{cases}
\]
Hence $\Gamma$ is a $d$-dimensional $\mathbb Q$-Poincar\'e duality group.
\end{lemma}

\begin{proof}
By Selberg's lemma, choose a torsion-free normal subgroup $\Gamma'\trianglelefteq\Gamma$ of finite index $m=[\Gamma:\Gamma']=2^k$, and set $F=\Gamma/\Gamma'$ (a finite $2$-group).

Since $\Gamma'$ acts freely and cocompactly on the contractible manifold $X=G/K$:
\[
H^q(\Gamma';\mathbb Q[\Gamma'])
\cong H^q_c(X;\mathbb Q)
=\begin{cases}\mathbb Q,&q=d,\\0,&\text{otherwise.}\end{cases}
\]

Consider the Lyndon--Hochschild--Serre spectral sequence for $1\to\Gamma'\to\Gamma\to F\to 1$:
\[
E_2^{p,q}=H^p\bigl(F;\,H^q(\Gamma';\mathbb Q[\Gamma])\bigr)\;\Longrightarrow\;H^{p+q}(\Gamma;\mathbb Q[\Gamma]).
\]
As a left $\mathbb Q[\Gamma']$-module, $\mathbb Q[\Gamma]\cong\mathbb Q[\Gamma']^{\oplus m}$ (decompose along coset representatives). Therefore
\[
H^q(\Gamma';\mathbb Q[\Gamma])\cong H^q(\Gamma';\mathbb Q[\Gamma'])^{\oplus m},
\]
which is $\mathbb Q^m$ for $q=d$ and $0$ otherwise.

The $F$-module structure on $\mathbb Q^m=\mathbb Q[\Gamma/\Gamma']=\mathbb Q[F]$ is the regular representation. Since $\mathbb Q[F]$ is free of rank~$1$ over~$\mathbb Q[F]$:
\[
H^p(F;\mathbb Q[F])=\begin{cases}\mathbb Q,&p=0,\\0,&p>0.\end{cases}
\]
The spectral sequence therefore has $E_2^{p,q}=\mathbb Q$ at $(0,d)$ and $0$ everywhere else, giving $H^n(\Gamma;\mathbb Q[\Gamma])=\mathbb Q$ if $n=d$ and $0$ otherwise.
\end{proof}

\subsection*{Step 2: The orbifold $\mathcal O=X/\Gamma$ is a $\mathbb Q$-homology manifold}

\begin{lemma}\label{lem:q7-qhm}
The compact orbifold $\mathcal O=X/\Gamma$ is a $d$-dimensional $\mathbb Q$-homology manifold: for every $x\in\mathcal O$,
\[
H_i(\mathcal O,\mathcal O\setminus\{x\};\mathbb Q)\cong
\begin{cases}\mathbb Q,&i=d,\\0,&i\ne d.\end{cases}
\]
\end{lemma}

\begin{proof}
At a point $x\in\mathcal O$ with isotropy group $\Gamma_x$ (a finite $2$-group), the local model is $\mathbb R^d/\Gamma_x$ with $\Gamma_x$ acting linearly by the isotropy representation.

The transfer homomorphism $\mathrm{tr}\colon H_i(\mathbb R^d/\Gamma_x,(\mathbb R^d\setminus\{0\})/\Gamma_x;\mathbb Q)\to H_i(\mathbb R^d,\mathbb R^d\setminus\{0\};\mathbb Q)$
composed with the projection gives multiplication by $|\Gamma_x|=2^j$. Since $2^j$ is invertible in~$\mathbb Q$, both the projection and transfer are $\mathbb Q$-isomorphisms. Hence the local homology at~$x$ with $\mathbb Q$-coefficients equals $H_i(\mathbb R^d,\mathbb R^d\setminus\{0\};\mathbb Q)$, which is $\mathbb Q$ for $i=d$ and $0$ otherwise.
\end{proof}

\subsection*{Step 3: $\pi_1^{\mathrm{top}}(\mathcal O)=\Gamma$}

\begin{lemma}\label{lem:q7-pi1}
The topological fundamental group of $\mathcal O$ is isomorphic to $\Gamma$, and the universal cover of~$\mathcal O$ (as a topological space) is~$X$.
\end{lemma}

\begin{proof}
Since $G$ is connected, $K$ is connected (Iwasawa decomposition), so every element of $K$ acts on $T_{eK}X\cong\mathfrak p$ with determinant~$+1$ (the adjoint action $K\to\mathrm{GL}(\mathfrak p)$ factors through~$\mathrm{SO}(\mathfrak p)$). In particular, every non-trivial finite-order element $\gamma\in\Gamma$ acts on~$X$ as an orientation-preserving isometry whose fixed-point set $X^\gamma$ is a totally geodesic submanifold.

The eigenvalues of $\gamma$ on $T_xX\cong\mathfrak p$ are roots of unity. Since $\gamma$ has $2$-power order and $\det=+1$, the eigenvalue $-1$ occurs with even multiplicity $\ge 2$. Therefore $\mathrm{codim}_X(X^\gamma)\ge 2$.

Denote the singular set $S=\bigcup_{1\ne\gamma\in\Gamma_{\mathrm{tor}}}X^\gamma/\Gamma$ (a closed subset of $\mathcal O$ of codimension $\ge 2$). On the complement $\mathcal O\setminus S$, the covering $X\setminus\widetilde S\to\mathcal O\setminus S$ is a genuine (free) covering with deck group~$\Gamma$.

Since $\mathcal O$ is a $d$-dimensional ANR with $d\ge 3$ and $S$ has codimension~$\ge 2$, removing $S$ does not change $\pi_1$:
\[
\pi_1^{\mathrm{top}}(\mathcal O)\cong\pi_1(\mathcal O\setminus S).
\]
The covering $X\setminus\widetilde S\to\mathcal O\setminus S$ with deck group~$\Gamma$ and simply connected total space ($X\setminus\widetilde S$ is simply connected since $X$ is contractible and $\widetilde S$ has codimension~$\ge 2$) implies $\pi_1(\mathcal O\setminus S)\cong\Gamma$.
\end{proof}

\subsection*{Step 4: Rational surgery produces a manifold}

\begin{proposition}\label{prop:q7-surgery}
For $d\ge 5$, there exists a closed topological $d$-manifold $M$ and a degree-$1$ map $f\colon M\to\mathcal O$ such that $f$ is a $\mathbb Q$-homology equivalence and $f_*\colon\pi_1(M)\xrightarrow{\;\sim\;}\pi_1(\mathcal O)=\Gamma$ is an isomorphism.
\end{proposition}

\begin{proof}[Proof (sketch using external inputs)]
Since $\mathcal O$ is a compact $d$-dimensional $\mathbb Q$-homology manifold and ANR (Lemma~\ref{lem:q7-qhm}), it admits a stable topological microbundle (hence a topological reduction of its Spivak normal fibration). This yields a degree-$1$ normal map $(f_0,b_0)\colon M_0\to\mathcal O$ from some closed $d$-manifold~$M_0$.

The obstruction to performing surgery on $(f_0,b_0)$ to obtain a $\mathbb Q$-homology equivalence lies in $L_d(\mathbb Z[\Gamma])\otimes\mathbb Q\cong L_d(\mathbb Q[\Gamma])$.

\textbf{External input 1 (Farrell--Jones).}
The isomorphism conjecture in $L$-theory is proved for lattices in virtually connected Lie groups (Farrell--Jones, 1993; Bartels--L\"uck--Reich for the full conjecture). This gives an isomorphism
\[
A\colon H_d(B\Gamma;\mathbf L^{\langle-\infty\rangle}(\mathbb Z))\otimes\mathbb Q\;\xrightarrow{\;\sim\;}\;L_d(\mathbb Z[\Gamma])\otimes\mathbb Q.
\]

\textbf{External input 2 (Sullivan's rational surgery).}
The set of $\mathbb Q$-normal invariants is $[\mathcal O,G/\mathrm{Top}]\otimes\mathbb Q\cong\bigoplus_k H^{4k}(\mathcal O;\mathbb Q)$ (Sullivan). The $\mathbb Q$-surgery obstruction map
\[
\bar\theta\colon\bigoplus_k H^{4k}(\mathcal O;\mathbb Q)\;\to\;L_d(\mathbb Q[\Gamma])
\]
factors through the Poincar\'e-duality isomorphism of the $\mathbb Q$-Poincar\'e complex $\mathcal O$ and the assembly map~$A$, and is therefore surjective.

Surjectivity of $\bar\theta$ implies: the initial surgery obstruction $\sigma(f_0,b_0)\otimes 1\in L_d(\mathbb Q[\Gamma])$ lies in the image of $\bar\theta$. We may therefore choose a modified normal invariant $\eta\in[\mathcal O,G/\mathrm{Top}]\otimes\mathbb Q$ with $\bar\theta(\eta)=-\sigma(f_0,b_0)\otimes 1$. Adjusting $(f_0,b_0)$ by $\eta$ and performing surgery below the middle dimension (Wall, $d\ge 5$) yields a closed manifold $M$ and a $\mathbb Q$-homology equivalence $f\colon M\to\mathcal O$ with $f_*\colon\pi_1(M)\xrightarrow{\sim}\Gamma$.
\end{proof}

\subsection*{Step 5: $\widetilde M$ is $\mathbb Q$-acyclic}

\begin{proof}[Proof of Theorem~\ref{thm:q7}]
Let $M$ and $f$ be as in Proposition~\ref{prop:q7-surgery}. Since $f_*\colon\pi_1(M)\to\pi_1(\mathcal O)=\Gamma$ is an isomorphism, $f$ lifts to a proper map $\widetilde f\colon\widetilde M\to X$ on universal covers.

The Cartan--Leray spectral sequences for $\widetilde M\to M$ and $X\to\mathcal O$ read
\[
E_2^{p,q}(M)=H_p(\Gamma;H_q(\widetilde M;\mathbb Q))\Longrightarrow H_{p+q}(M;\mathbb Q),
\]
\[
E_2^{p,q}(\mathcal O)=H_p(\Gamma;H_q(X;\mathbb Q))\Longrightarrow H_{p+q}(\mathcal O;\mathbb Q).
\]
Since $X$ is contractible, the second spectral sequence has $E_2^{p,0}=H_p(\Gamma;\mathbb Q)$ and $E_2^{p,q}=0$ for $q>0$; it collapses at $E_2$ and gives $H_n(\mathcal O;\mathbb Q)=H_n(\Gamma;\mathbb Q)$.

The map $f$ induces a morphism of spectral sequences. Since $f_*\colon H_*(M;\mathbb Q)\to H_*(\mathcal O;\mathbb Q)$ is an isomorphism and the target spectral sequence is concentrated on the $q=0$ line, a standard comparison argument shows $H_q(\widetilde M;\mathbb Q)=0$ for $q>0$. (If not, the lowest nonzero $H_q(\widetilde M;\mathbb Q)$ for $q>0$ would contribute a nonzero $E_2^{0,q}(M)=H_0(\Gamma;H_q(\widetilde M;\mathbb Q))$ that persists to $E_\infty$ and produces a filtration quotient with no counterpart in $H_*({\mathcal O};\mathbb Q)$, contradicting the isomorphism on abutments.)

Therefore $\widetilde M$ is $\mathbb Q$-acyclic: $\widetilde H_*(\widetilde M;\mathbb Q)=0$.
\end{proof}

\begin{remark}[Low-dimensional cases]
The hypothesis $d\ge 5$ is used only for Wall's surgery below the middle dimension. For $d\le 4$ (arising from rank-$1$ groups $\mathrm{SL}_2(\mathbb R)$, $\mathrm{SL}_2(\mathbb C)$, $\mathrm{SU}(2,1)$, $\mathrm{SO}(4,1)$), the $\mathbb Q$-surgery machinery requires additional low-dimensional arguments. However, these cases are exceptional and do not affect the general answer.
\end{remark}

\begin{remark}[External inputs]
The proof relies on three external results:
\begin{enumerate}
\item The Farrell--Jones isomorphism conjecture in $L$-theory for lattices in semi-simple Lie groups.
\item Sullivan's rational computation of the surgery classifying space $G/\mathrm{Top}$.
\item Wall's surgery exact sequence for topological manifolds in dimensions $\ge 5$.
\end{enumerate}
All three are standard and well-established in the literature.
\end{remark}
