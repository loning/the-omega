\paragraph{Cross-interface terminal conclusions: Heine--Stieltjes reduction, short-range potential tail vanishing, Schwarzian rigidity, and closed-form structure of anchored Cauchy interpolation}
This paragraph summarizes a set of conclusions that can be repeatedly invoked at anchored rational kernel, Stieltjes-type second-order equations, and Möbius covariant interfaces.
Here, the Schrödinger reduction and vanishing condition at infinity provide a strict algebraic characterization of ``short-range potential;''
Schwarzian rigidity translates the second-order pole principal coefficient into ramification index;
while Cauchy--Lagrange interpolation and the explicit inverse of the Cauchy matrix provide completely closed forms for full-rank anchor linear decoding.

\begin{conclusion}[Heine--Stieltjes equation Schr\"odinger reduction and local exponent rigidity]\label{con:xi-terminal-heine-stieltjes-schrodinger-reduction-local-exponent-rigidity}
Let \(\kappa\ge 1\), and take distinct points
\[
p_k=\gamma_k-\mathrm{i}a_k\in\CC_{-}\qquad(a_k>0,\ 1\le k\le \kappa),
\]
and set
\[
A(z):=\prod_{k=1}^{\kappa}(z-p_k)(z-\overline{p_k})\in\RR[z],
\qquad
\deg A=2\kappa.
\]
Let \(y\in\RR[z]\) be a polynomial of degree \(\kappa\), and suppose there exists \(B\in\RR[z]\) such that
\[
A(z)y''(z)-A'(z)y'(z)=B(z)y(z).
\]
Take any local analytic branch \(s(z):=\sqrt{A(z)}\), and define
\[
\phi(z):=\frac{y(z)}{s(z)}.
\]
Then \(\phi\) satisfies the Schr\"odinger-type equation
\[
\phi''(z)+U(z)\phi(z)=0,
\qquad
U(z):=\frac{1}{2}\frac{A''(z)}{A(z)}-\frac{3}{4}\Bigl(\frac{A'(z)}{A(z)}\Bigr)^2-\frac{B(z)}{A(z)}.
\]
Moreover, \(U\) is a rational function, \(\RR\)-symmetric (\(\overline{U(\overline z)}=U(z)\)), has a second-order pole at each \(p_k,\overline{p_k}\), and the principal coefficient rigidity is
\[
U(z)=-\frac{3}{4}\frac{1}{(z-p_k)^2}+O\Bigl(\frac{1}{z-p_k}\Bigr)\quad(z\to p_k),
\qquad
U(z)=-\frac{3}{4}\frac{1}{(z-\overline{p_k})^2}+O\Bigl(\frac{1}{z-\overline{p_k}}\Bigr)\quad(z\to\overline{p_k}).
\]
Hence, the Frobenius index set of \(\phi\) at \(p_k\) is \(\{-\tfrac12,\tfrac32\}\) (similarly for \(\overline{p_k}\)), and thus the local indices of \(y=s\phi\) at these points are \(\{0,2\}\).
\end{conclusion}
\begin{proof}
From \(y=s\phi\) we get
\[
y'=s'\phi+s\phi',\qquad
y''=s''\phi+2s'\phi'+s\phi''.
\]
Substituting into \(Ay''-A'y'\), and simplifying using \(s'/s=\tfrac12 A'/A\), yields
\[
Ay''-A'y'
=A^{3/2}\phi''+A^{1/2}\Bigl(\frac12A''-\frac34\frac{(A')^2}{A}\Bigr)\phi.
\]
Dividing the equation \(Ay''-A'y'=By\) first by \(A^{1/2}\), then by \(A\), we obtain \(\phi''+U\phi=0\) with the indicated \(U\).
\par
For the local principal part: write \(A(z)=(z-p_k)g_k(z)\) with \(g_k(p_k)\neq 0\), then
\[
\frac{A'}{A}=\frac{1}{z-p_k}+\frac{g_k'}{g_k}.
\]
So \((A'/A)^2\) has second-order principal part \((z-p_k)^{-2}\) at \(p_k\), while
\(
A''/A=(A'/A)'+(A'/A)^2
\)
has its second-order term cancel, so \(A''/A\) has no second-order pole at \(p_k\); also since \(B/A\) only has a first-order pole, the second-order principal coefficient of \(U\) is \(-\tfrac34\).
From the indicial equation \(r(r-1)-\tfrac34=0\) we get \(r\in\{-\tfrac12,\tfrac32\}\).
Multiplying by \(s\sim (z-p_k)^{1/2}\) shows the local indices of \(y=s\phi\) are \(\{0,2\}\).
For \(\overline{p_k}\), the argument is similar.
\end{proof}

\begin{conclusion}[Necessary and sufficient condition for complete vanishing of the ``inverse-square tail'' at infinity: strict equivalent characterization of short-range potential]\label{con:xi-terminal-heine-stieltjes-short-range-tail-vanish}
Under the setting of Conclusion \ref{con:xi-terminal-heine-stieltjes-schrodinger-reduction-local-exponent-rigidity}, suppose
\[
B(z)=b_0 z^{2\kappa-2}+O(z^{2\kappa-3})\qquad(z\to\infty).
\]
Then
\[
U(z)=\frac{-\kappa(\kappa+1)-b_0}{z^{2}}+O\Bigl(\frac{1}{z^{3}}\Bigr)\qquad(z\to\infty).
\]
Therefore, the following two conditions are equivalent:
\begin{enumerate}
  \item \(U(z)=O(z^{-3})\) (short-range potential);
  \item \(b_0=-\kappa(\kappa+1)\).
\end{enumerate}
\end{conclusion}
\begin{proof}
Write
\(
A(z)=z^{2\kappa}+A_1 z^{2\kappa-1}+O(z^{2\kappa-2})
\), then
\[
\frac{A'}{A}=\frac{2\kappa}{z}+O\Bigl(\frac{1}{z^{2}}\Bigr),\qquad
\frac{A''}{A}=\frac{2\kappa(2\kappa-1)}{z^{2}}+O\Bigl(\frac{1}{z^{3}}\Bigr),\qquad
\frac{B}{A}=\frac{b_0}{z^{2}}+O\Bigl(\frac{1}{z^{3}}\Bigr).
\]
Substituting into
\(
U=\tfrac12 A''/A-\tfrac34(A'/A)^2-B/A
\)
yields
\[
\frac12\frac{A''}{A}-\frac34\Bigl(\frac{A'}{A}\Bigr)^2
=\frac{\kappa(2\kappa-1)}{z^{2}}-\frac{3\kappa^{2}}{z^{2}}+O(z^{-3})
=-\frac{\kappa(\kappa+1)}{z^{2}}+O(z^{-3}),
\]
subtracting \(B/A\) yields the indicated expansion.
Short-range potential means the leading coefficient is zero, equivalent to \(b_0=-\kappa(\kappa+1)\).
\end{proof}

\begin{conclusion}[Zero-energy scattering characterization of polynomial solutions: degree condition equivalent to Jost normalization]\label{con:xi-terminal-heine-stieltjes-zero-energy-scattering-jost-normalization}
Still under the setting of Conclusion \ref{con:xi-terminal-heine-stieltjes-schrodinger-reduction-local-exponent-rigidity}, and assume the short-range potential condition \(U(z)=O(z^{-3})\) holds.
Then in any fixed sector \(|\arg z|<\pi-\varepsilon\), there exists a set of fundamental solutions \(\phi_{0},\phi_{1}\) uniquely determined at infinity satisfying
\[
\phi_{0}(z)=1+O\Bigl(\frac{1}{z}\Bigr),
\qquad
\phi_{1}(z)=z+O(1)
\qquad(z\to\infty),
\]
and any solution can be uniquely represented as \(\phi=c_{0}\phi_{0}+c_{1}\phi_{1}\).
Under this normalization, let the infinity branch be \(\sqrt A=z^{\kappa}(1+O(1/z))\), then
\[
y(z)=\sqrt{A(z)}\,\phi(z)
\]
is a polynomial of degree exactly \(\kappa\) if and only if \(c_{1}=0\); equivalently, \(\phi\) satisfies the Jost boundary condition \(\phi(z)=1+O(z^{-1})\).
\end{conclusion}
\begin{proof}
From \(U(z)=O(z^{-3})\), the equation \(\phi''+U\phi=0\) is an integrable perturbation of the free equation \(\phi''=0\) at infinity.
Inside the sector, one can construct solutions with specified asymptotics using Volterra-type integral equations and obtain uniqueness: for \(\phi_{0}\) as an example, take
\[
\phi_{0}(z)=1+\int_{z}^{\infty}(t-z)U(t)\phi_{0}(t)\,\dd t,
\]
where the integral is along a fixed ray; \(U(t)=O(t^{-3})\) ensures convergence and yields a contraction mapping, so there exists a unique solution satisfying \(\phi_{0}(z)=1+O(1/z)\).
Similarly, we get \(\phi_{1}(z)=z+O(1)\).
Basic theory of linear ODEs yields unique linear decomposition of general solutions.
\par
From \(\sqrt A=z^{\kappa}(1+O(1/z))\) we get
\[
\sqrt A\,\phi_{0}=z^{\kappa}\Bigl(1+O\Bigl(\frac{1}{z}\Bigr)\Bigr),
\qquad
\sqrt A\,\phi_{1}=z^{\kappa+1}\bigl(1+o(1)\bigr),
\]
so \(\sqrt A(c_{0}\phi_{0}+c_{1}\phi_{1})\) has highest degree \(\kappa\) if and only if \(c_{1}=0\).
And \(c_{1}=0\) is equivalent to \(\phi(z)=1+O(z^{-1})\).
\end{proof}

\begin{conclusion}[Pole expansion of short-range potential and ``renormalization sum rules'': implementing tail vanishing at infinity via single-pole coefficient constraint]\label{con:xi-terminal-heine-stieltjes-pole-expansion-renormalization-sum-rules}
Under the setting of Conclusion \ref{con:xi-terminal-heine-stieltjes-schrodinger-reduction-local-exponent-rigidity}, let
\[
r_k:=\operatorname*{Res}_{z=p_k}\frac{B(z)}{A(z)}=\frac{B(p_k)}{A'(p_k)}\in\CC,
\]
as well as
\[
s_k:=\left(\frac{A'}{A}-\frac{1}{z-p_k}\right)\Big|_{z=p_k}
=\frac{1}{p_k-\overline{p_k}}+\sum_{j\ne k}\left(\frac{1}{p_k-p_j}+\frac{1}{p_k-\overline{p_j}}\right).
\]
Then the Schr\"odinger potential \(U\) has a complete partial fraction expansion
\[
U(z)=
-\frac34\sum_{k=1}^{\kappa}\left(\frac{1}{(z-p_k)^2}+\frac{1}{(z-\overline{p_k})^2}\right)
+\sum_{k=1}^{\kappa}\left(\frac{c_k}{z-p_k}+\frac{\overline{c_k}}{z-\overline{p_k}}\right),
\qquad
c_k:=-r_k-\frac12 s_k.
\]
Moreover, the short-range potential condition \(U(z)=O(z^{-3})\) is equivalent to the following two real linear constraints (``renormalization sum rules''):
\[
\sum_{k=1}^{\kappa}\Re(c_k)=0,
\qquad
\sum_{k=1}^{\kappa}\Re(p_k c_k)=\frac{3\kappa}{4}.
\]
\end{conclusion}
\begin{proof}
By definition
\(
U=\frac12\frac{A''}{A}-\frac34(\frac{A'}{A})^2-\frac{B}{A}
\).
At \(p_k\) write
\[
\frac{A'}{A}=\frac{1}{z-p_k}+s_k+O(z-p_k),
\]
then
\[
\Bigl(\frac{A'}{A}\Bigr)^2=\frac{1}{(z-p_k)^2}+\frac{2s_k}{z-p_k}+O(1),
\qquad
\frac{A''}{A}=\Bigl(\frac{A'}{A}\Bigr)'+\Bigl(\frac{A'}{A}\Bigr)^2=\frac{2s_k}{z-p_k}+O(1),
\]
and
\(
\frac{B}{A}=\frac{r_k}{z-p_k}+O(1)
\).
Therefore, the second-order principal coefficient of \(U\) at \(p_k\) is \(-\tfrac34\), and the first-order residue is
\(
s_k-\tfrac32 s_k-r_k=-\tfrac12 s_k-r_k=c_k
\).
For \(\overline{p_k}\), similarly we get \(\overline{c_k}\), and summing the principal parts at all points yields the partial fraction expansion.
\par
Expand the expansion formula in Laurent series at infinity: for \(z\to\infty\),
\[
\frac{1}{z-p}=\frac{1}{z}+\frac{p}{z^{2}}+O(z^{-3}),
\qquad
\frac{1}{(z-p)^2}=\frac{1}{z^{2}}+\frac{2p}{z^{3}}+O(z^{-4}).
\]
Thus
\[
U(z)=\frac{2\sum_k \Re(c_k)}{z}
+\frac{2\sum_k\Re(p_k c_k)-\frac34\cdot 2\kappa}{z^{2}}
+O(z^{-3}).
\]
So \(U(z)=O(z^{-3})\) if and only if the \(z^{-1}\) and \(z^{-2}\) coefficients are both zero, i.e.,
\(
\sum\Re(c_k)=0
\)
and
\(
\sum\Re(p_k c_k)=\tfrac{3\kappa}{4}
\).
\end{proof}

\begin{conclusion}[\texorpdfstring{$\mathrm{PSL}(2,\RR)$}{PSL(2,R)} covariance: potential \(U\) is a real quadratic differential]\label{con:xi-terminal-schrodinger-psl2r-quadratic-differential-covariance}
Let \(m(z)=\dfrac{az+b}{cz+d}\) (with \(a,b,c,d\in\RR,\ ad-bc>0\)) be an orientation-preserving Möbius transformation.
Suppose \(\phi\) satisfies \(\phi''(z)+U(z)\phi(z)=0\).
Let \(\zeta=m(z)\), and define
\[
\widetilde\phi(\zeta):=\sqrt{\frac{\dd z}{\dd\zeta}}\;\phi(z),\qquad z=m^{-1}(\zeta).
\]
Then \(\widetilde\phi\) satisfies the same-type equation
\[
\widetilde\phi''(\zeta)+\widetilde U(\zeta)\widetilde\phi(\zeta)=0,
\qquad
\widetilde U(\zeta)=\Bigl(\frac{\dd z}{\dd\zeta}\Bigr)^{2}U(z),
\]
i.e., \(U(z)\,\dd z^{2}\) transforms covariantly as a quadratic differential under \(\mathrm{PSL}(2,\RR)\) (the Schwarzian of a Möbius transformation is zero, so there is no additional term).
\end{conclusion}
\begin{proof}
By the chain rule and \(\widetilde\phi=\sqrt{z_\zeta}\phi\) (denote \(z_\zeta=\dd z/\dd\zeta\)), we obtain the general transformation law
\[
\widetilde U(\zeta)=z_\zeta^{2}U(z)+\frac12\{z,\zeta\},
\]
where \(\{z,\zeta\}\) is the Schwarzian derivative.
For a Möbius transformation \(z=m^{-1}(\zeta)\) we have \(\{z,\zeta\}=0\), so \(\widetilde U=z_\zeta^{2}U\).
\end{proof}

\begin{conclusion}[Cauchy--Lagrange rational interpolation and Cauchy matrix explicit inverse under full-rank anchor]\label{con:xi-terminal-cauchy-lagrange-rational-interpolation-cauchy-inverse}
Take distinct real points \(a_1,\dots,a_\kappa\in\RR\), and set
\[
\Pi(z):=\prod_{k=1}^{\kappa}(z-p_k),\qquad
y(z):=\prod_{j=1}^{\kappa}(z-a_j).
\]
Consider the \(\kappa\)-dimensional rational function space
\[
\mathcal R:=\left\{f(z)=\sum_{k=1}^{\kappa}\frac{c_k}{z-p_k}\;:\;c_k\in\CC\right\}.
\]
Let the Cauchy matrix \(C\in\CC^{\kappa\times\kappa}\) be \(C_{jk}=\frac{1}{a_j-p_k}\).
Then \(C\) is invertible, and for any data \(f_j\in\CC\) there exists a unique \(f\in\mathcal R\) satisfying \(f(a_j)=f_j\), with explicit interpolation formula
\[
f(z)=\sum_{j=1}^{\kappa} f_j\,L_j(z),
\qquad
L_j(z):=\frac{y(z)}{(z-a_j)y'(a_j)}\cdot\frac{\Pi(a_j)}{\Pi(z)}.
\]
Furthermore, \(C^{-1}\) has a complete closed form: for \(1\le k,j\le\kappa\),
\[
(C^{-1})_{k j}
=\frac{y(p_k)\,\Pi(a_j)}{(p_k-a_j)\,\Pi'(p_k)\,y'(a_j)}.
\]
\end{conclusion}
\begin{proof}
First prove the interpolation basis function: for \(i\ne j\), since \(y(a_i)=0\) and \(y(z)/(z-a_j)\) still contains the factor \((z-a_i)\) at \(z=a_i\), we have \(L_j(a_i)=0\);
while
\[
\lim_{z\to a_j}\frac{y(z)}{(z-a_j)y'(a_j)}=1,
\qquad
\frac{\Pi(a_j)}{\Pi(a_j)}=1,
\]
so \(L_j(a_j)=1\).
Therefore \(f(z)=\sum f_j L_j(z)\) satisfies \(f(a_i)=f_i\).
Each \(L_j\) is a proper fraction with denominator \(\Pi(z)\), so can be partial-fraction decomposed and belongs to \(\mathcal R\); uniqueness follows from \(\dim\mathcal R=\kappa\) and the full-rank property of the evaluation map.
\par
Now prove the \(C^{-1}\) closed form: write \(L_j(z)=\sum_{k} \alpha_{kj}/(z-p_k)\).
Then for each \(i\),
\[
L_j(a_i)=\sum_k\frac{\alpha_{kj}}{a_i-p_k},
\]
in matrix notation \(C\alpha_{(:,j)}=e_j\), so \(\alpha=C^{-1}\).
On the other hand, from
\[
L_j(z)=\frac{\Pi(a_j)}{y'(a_j)}\cdot\frac{y(z)}{(z-a_j)\Pi(z)}
\]
taking the residue at \(z=p_k\) yields
\[
\alpha_{k j}
=\operatorname*{Res}_{z=p_k}L_j(z)
=\frac{\Pi(a_j)}{y'(a_j)}\cdot\frac{y(p_k)}{(p_k-a_j)\Pi'(p_k)}.
\]
So \((C^{-1})_{kj}=\alpha_{kj}\) as indicated.
\end{proof}

\begin{conclusion}[Schwarzian rigidity and double branching: \texorpdfstring{$-\tfrac34(z-p)^{-2}$}{-(3/4)(z-p)^(-2)} equivalent to ramification index \texorpdfstring{$2$}{2}]\label{con:xi-terminal-schwarzian-rigidity-branch-index-2}
Let \(U\) be a meromorphic function in a punctured neighborhood of \(p\in\CC\).
The following two conditions are equivalent:
\begin{enumerate}
  \item There exists a local meromorphic function \(w\) such that its Schwarzian derivative satisfies \(\{w,z\}=2U(z)\), and
  \[
  U(z)=-\frac34\frac{1}{(z-p)^2}+O\Bigl(\frac{1}{z-p}\Bigr)\qquad(z\to p).
  \]
  \item There exists a local coordinate \(\zeta=z-p\) and an invertible analytic function \(h\) such that
  \[
  w(z)=h\bigl(\zeta^{2}\bigr),
  \]
  i.e., \(w\) has ramification index exactly \(2\) at \(p\); equivalently, its local inverse map has square-root-type branching.
\end{enumerate}
Therefore, for the Schr\"odinger potential \(U\) of Conclusion \ref{con:xi-terminal-heine-stieltjes-schrodinger-reduction-local-exponent-rigidity}, taking any two linearly independent solutions \(\phi_1,\phi_2\), their ratio \(w=\phi_1/\phi_2\) satisfies \(\{w,z\}=2U(z)\), and has ramification index \(2\) at each \(p_k,\overline{p_k}\) (zero/pole of order \(2\)), and this index is rigidly locked by the second-order principal coefficient \(-\tfrac34\).
\end{conclusion}
\begin{proof}
Classical fact shows: the ratio \(w=\phi_1/\phi_2\) of two linearly independent solutions of \(\phi''+U\phi=0\) satisfies \(\{w,z\}=2U(z)\).
Moreover, if \(w\) has local leading term \(w\sim \zeta^{\alpha}\) at \(\zeta=0\) (with \(\alpha\neq 0\)), then the second-order principal part of the Schwarzian is
\[
\{w,\zeta\}=\frac{1-\alpha^{2}}{2}\,\zeta^{-2}+O(\zeta^{-1}).
\]
\par
\((2)\Rightarrow(1)\): Take \(w(\zeta)=\zeta^{2}\), then \(\{w,\zeta\}=-\tfrac{3}{2}\zeta^{-2}\).
By the chain rule and \(\{w,z\}=(\dd\zeta/\dd z)^2\{w,\zeta\}+\{\zeta,z\}\), the second-order principal coefficient is invariant under analytic coordinate changes;
moreover, post-composing with an invertible analytic function \(h\) only introduces a \((w')^{2}\{h,w\}\)-type term at \(\{h,w\}\), and when \(w\) has ramification index \(2\) at \(\zeta=0\), \(w'(\zeta)=O(\zeta)\), so this term does not affect the \(\zeta^{-2}\) principal coefficient.
Therefore we can obtain \(2U=\{w,z\}=-\tfrac{3}{2}(z-p)^{-2}+O((z-p)^{-1})\), i.e., the second-order principal coefficient of \(U\) is \(-\tfrac34\).
\par
\((1)\Rightarrow(2)\): If \(\{w,z\}=-\tfrac{3}{2}(z-p)^{-2}+O((z-p)^{-1})\), in local coordinate \(\zeta=z-p\), taking Puiseux leading term \(w(\zeta)\sim \zeta^{\alpha}\) and substituting into the above, comparing \(\zeta^{-2}\) coefficients yields \((1-\alpha^{2})/2=-3/2\), so \(\alpha^{2}=4\), hence \(\alpha=\pm 2\).
Therefore \(w\) has double branching at \(p\): if \(\alpha=2\) then \(w=h(\zeta^{2})\); if \(\alpha=-2\) then \(1/w\) satisfies the previous case and can be reduced to the \(\alpha=2\) case (equivalent to \(w\) having a second-order pole at \(p\), corresponding to ramification index \(2\) mapping to infinity).
The last assertion follows directly from the second-order principal coefficient rigidity given by Conclusion \ref{con:xi-terminal-heine-stieltjes-schrodinger-reduction-local-exponent-rigidity}.
\end{proof}
