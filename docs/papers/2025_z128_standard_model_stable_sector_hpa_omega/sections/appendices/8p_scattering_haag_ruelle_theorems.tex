\subsection{Haag--Ruelle scattering in a gapped sector (T4-1)}
\label{app:scattering_haag_ruelle_theorems}

\AuditTag This appendix records the strong-closure target for scattering: in a declared gapped/stable one-particle sector, construct asymptotic states and define the scattering operator as a theorem under explicit hypotheses.
The failure modes are normalized by the canonical S1--S3 records (Appendix~\ref{app:minimal_failure_point_templates}).

\subsubsection{Condition bundle: gapped one-particle sector}
\begin{assumption}[Gapped one-particle sector (template)]
\label{ass:gapped_one_particle_sector_template}
Assume a vacuum representation with:
\begin{itemize}
\item a spectrum condition (Appendix~\ref{app:covariance_spectrum_from_protocol_dynamics}),
\item a stable isolated one-particle mass shell with a mass gap to the multi-particle continuum,
\item locality/microcausality in the relevant sense.
\end{itemize}
\end{assumption}

\paragraph{Auditable evidence contract.}
\AuditTag For an explicit protocol-facing evidence contract linking a finite spectral-gap witness to the ``gapped sector'' prerequisite, see Appendix~\ref{app:gapped_sector_evidence_contract}.

\subsubsection{Haag--Ruelle theorem (template)}
\begin{theorem}[Haag--Ruelle scattering (template)]
\label{thm:haag_ruelle_template}
\MathTag Under Assumption~\ref{ass:gapped_one_particle_sector_template}, asymptotic in/out states exist for sufficiently regular wave packets, and the scattering operator $S$ is a well-defined unitary map between the asymptotic Hilbert spaces.
\end{theorem}

\paragraph{Scope boundary.}
\AuditTag Theorem~\ref{thm:haag_ruelle_template} is recorded as the theorem-facing form of scattering in a gapped sector.
Asymptotic completeness is a separate (often open) property and is treated via the canonical S3 record.
