\section{Reproducibility Protocol and Falsifiability Indicators}

\begin{enumerate}
  \item Six-dimensional strip comparison: reproduce the projection differences between rational $(1/1)$ and irrational $(\tau/1)$ strips \cite{PRB46_6091}.
  \item Sturmian verification: estimate $p(n)$ from binary sequences and verify $p(n)=n+1$ \cite{CambridgeSturmianWordsLothaire}.
  \item Zeckendorf stabilization: verify $\mathrm{Fold}_6$ output and fiber spectrum on the complete enumeration baseline of $m=6$ \cite{PlanetMathZeckendorfCombinatorial}.
  \item SMB fitting: estimate cylinder set measure decay rate and align with entropy rate \cite{GatechSMB}.
  \item Observer parameter variation: change $u,\alpha,\mathcal{P}$ to compare local patch differences and statistical invariant consistency (geometric part see \cite{IUCRAcceptanceDomain}).
\end{enumerate}
