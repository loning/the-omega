\subsection{Running couplings as resolution flow in the \texorpdfstring{$r$}{r} coordinate (self-contained interface)}
\label{app:running_couplings_resolution_flow}

This appendix makes the RG dictionary in the Fibonacci resolution coordinate $r$ self-contained.
The main text already records the basic chain rule (Proposition~\ref{prop:rg_dictionary_r}); here we add the standard one-loop templates and the semantics of threshold matching and scheme shifts, in a form aligned with the tick\,+\,CAP stance (ratio-first, then matching).

\subsubsection{Scale map and chain rule}
\label{subsec:rg_chain_rule}

\paragraph{Resolution map.}
Work with the Fibonacci map
\[
\mu(r)=\mu_0\,\varphi^{\,r},
\]
so that $\log\mu=\log\mu_0+r\log\varphi$ and therefore $\dd/\dd r=(\log\varphi)\,\dd/\dd\log\mu$ (Proposition~\ref{prop:rg_dictionary_r}).

\paragraph{RG in \texorpdfstring{$r$}{r}.}
If a coupling $g(\mu)$ satisfies an RG equation
\[
\frac{\dd g}{\dd\log\mu}=\beta(g),
\]
then in the $r$ coordinate,
\begin{equation}
\label{eq:rg_in_r}
\frac{\dd g}{\dd r}=(\log\varphi)\,\beta(g).
\end{equation}

\subsubsection{One-loop QED running (leading log)}
\label{subsec:qed_one_loop}

At one loop (mass-independent scheme, away from thresholds), QED running gives
\begin{equation}
\label{eq:qed_one_loop}
\alpha^{-1}(\mu)\approx \alpha^{-1}(\mu_0)-\frac{b}{2\pi}\log\!\left(\frac{\mu}{\mu_0}\right),
\qquad
b=\frac{2}{3}\sum_f N_c^{(f)}Q_f^2,
\end{equation}
so in the $r$ coordinate one obtains a linear flow
\begin{equation}
\label{eq:qed_one_loop_r}
\alpha^{-1}(r)\approx \alpha^{-1}(0)-\frac{b\log\varphi}{2\pi}\,r.
\end{equation}

\subsubsection{One-loop QCD running and dimensional transmutation}
\label{subsec:qcd_one_loop}

For QCD, the beta function has the opposite sign; at one loop,
\begin{equation}
\label{eq:qcd_one_loop}
\frac{\dd\alpha_s}{\dd\log\mu}=-\frac{b_0}{2\pi}\alpha_s^2,
\qquad
b_0=11-\frac{2}{3}n_f,
\end{equation}
so
\begin{equation}
\label{eq:qcd_solution}
\alpha_s(\mu)=\frac{2\pi}{b_0\log(\mu/\Lambda)},
\end{equation}
where $\Lambda$ is a scheme-dependent transmutation scale.
In the $r$ coordinate, a constant rescaling of $\Lambda$ is an additive shift of $r$, so ``scheme dependence'' becomes an explicit equivalence-class shift rather than an uncontrolled knob.

\begin{lemma}[Scheme rescaling as an additive shift in the resolution coordinate]\label{lem:lambda_rescaling_shift_r}
\MathTag Fix $\varphi>1$ and define the resolution coordinate by $r=\log_\varphi(\mu/\mu_0)$ so that $\mu=\mu_0\varphi^r$.
Define also the $\Lambda$-relative coordinate
\[
r_\Lambda(\mu):=\log_\varphi\!\left(\frac{\mu}{\Lambda}\right).
\]
If $\Lambda' = c\,\Lambda$ for some constant $c>0$, then for all $\mu>0$,
\[
r_{\Lambda'}(\mu)=r_\Lambda(\mu)-\log_\varphi c.
\]
\end{lemma}
\begin{proof}
By definition,
\[
r_{\Lambda'}(\mu)=\log_\varphi\!\left(\frac{\mu}{c\Lambda}\right)
=\log_\varphi\!\left(\frac{\mu}{\Lambda}\right)-\log_\varphi c
=r_\Lambda(\mu)-\log_\varphi c.
\]
\end{proof}

\subsubsection{Threshold matching as discrete uplifts}
\label{subsec:threshold_matching}

\InterfaceTag Effective-field-theory decoupling implies that charged fields contribute to running only above their thresholds; across a threshold one matches schemes by a finite jump/offset.
In the protocol language, thresholds are resolution-uplift points: as $m_{\mathrm{eff}}(\mu)$ changes discretely, effective degrees of freedom appear/disappear and induce discrete changes in $\beta$-coefficients.
In the joint protocol-state convention, this is a statement about the $(m,n)$ component of $(m,n,K)$ (Appendix~\ref{subsec:tick_cap_joint_protocol_state}); kernel choice $K$ affects kernel-dependent weighted readouts, while the one-loop representation-counting coefficients are fixed by the closed labeling data.
This is the semantic link between RG matching and the discrete staircase used elsewhere in this paper (Section~\ref{subsec:p3_resolution_jumps}).

\subsubsection{Standard Model $\beta$-functions from representation counting (protocol-derived)}
\label{subsec:sm_beta_functions_dictionary}

\InterfaceTag The one-loop Standard Model $\beta$-function coefficients are derived from representation counting on the closed labeling (Appendix~\ref{app:yukawa_beta_protocol_closure}, Proposition~\ref{prop:beta_from_labeling}).
The inputs are: (i) the gauge group closed via Proposition~\ref{prop:channel_to_gauge}, (ii) the 18 chiral fermion multiplets closed via the labeling map (Theorem~\ref{thm:labeling_unique}), and (iii) one Higgs doublet as the minimal scalar uplift interface.
The resulting coefficients $(b_1,b_2,b_3)=(41/6,-19/6,-7)$ are outputs of the tick+CAP framework, not matching-layer inputs.
For the general formulas and higher-loop extensions, see the classic references \cite{MachacekVaughn1983,MachacekVaughn1984,MachacekVaughn1985}; an independent SM-specialized compilation is given in \cite{LuoXiao2003}.

\paragraph{One-loop gauge running in the SM (reference).}
In a conventional normalization with $Q=T_3+Y$ and covariant derivative $D_\mu=\partial_\mu-\iu g_3 G_\mu^A T^A-\iu g_2 W_\mu^a\tau^a-\iu g_1 Y B_\mu$ (as in \eqref{eq:protocol_eft_covariant_derivative}), the one-loop gauge $\beta$-functions take the form
\begin{equation}
\frac{\dd g_i}{\dd\log\mu}=\frac{b_i}{16\pi^2}g_i^3,\qquad i\in\{1,2,3\},
\end{equation}
with $(b_1,b_2,b_3)=(41/6,-19/6,-7)$ for three generations and one Higgs doublet in the SM.
We use these only as a reference dictionary; all scheme/threshold details are treated as matching-layer conventions in the audit sense of Part~I (Contract).

\paragraph{EFT completion status.}
\InterfaceTag When one embeds the closed gauge and chiral labeling interface into a renormalizable Standard Model EFT (Appendix~\ref{app:protocol_eft_embedding}), the scale dependence of the gauge couplings $(g_1,g_2,g_3)$, Yukawa matrices $Y_f$, and Higgs potential couplings $(\mu^2,\lambda)$ is governed by the standard perturbative renormalization-group equations.
The one-loop $\beta$-coefficients for gauge couplings follow from representation counting on the closed labeling (Appendix~\ref{app:yukawa_beta_protocol_closure}); the RG equations then serve as a dictionary translating scheme/scale dependence into a shift along the resolution coordinate $r$.
