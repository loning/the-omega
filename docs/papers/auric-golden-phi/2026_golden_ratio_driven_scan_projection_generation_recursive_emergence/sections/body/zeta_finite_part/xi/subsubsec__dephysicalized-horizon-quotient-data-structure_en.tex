\begin{definition}[External Observable Algebra and External Equivalence Relation]\label{def:dephys-ext-observable-algebra-equivalence}
Let \(\mathcal{H}_{\mathrm{bulk}}\) be the total Hilbert space containing "internal-external" degrees of freedom, and let
\(\mathcal{A}_{\mathrm{ext}}\subseteq \mathcal{B}(\mathcal{H}_{\mathrm{bulk}})\)
be the \(\ast\)-algebra (external measurable algebra) accessible to external observers.
Denote the density operator state space by
\[
\mathcal{D}(\mathcal{H}_{\mathrm{bulk}})
:=\{\rho\succeq 0:\ \operatorname{Tr}\rho=1\}.
\]
For \(\rho,\tilde\rho\in\mathcal{D}(\mathcal{H}_{\mathrm{bulk}})\), define the external equivalence relation
\begin{equation}\label{eq:dephys-ext-equivalence}
\boxed{\;
\rho\sim_{\mathrm{ext}}\tilde\rho
\quad\Longleftrightarrow\quad
\operatorname{Tr}(\rho A)=\operatorname{Tr}(\tilde\rho A)\ \ (\forall\,A\in\mathcal{A}_{\mathrm{ext}}).\;}
\end{equation}
Denote the state space on \(\mathcal{A}_{\mathrm{ext}}\) by \(\mathrm{St}(\mathcal{A}_{\mathrm{ext}})\):
\[
\mathrm{St}(\mathcal{A}_{\mathrm{ext}})
:=\{\omega:\mathcal{A}_{\mathrm{ext}}\to\CC:\ \omega\ge 0,\ \omega(I)=1\}.
\]
\end{definition}

\begin{proposition}[Horizon as "Quotient Data Structure": Equivalence Class Fibers Induced by External Observable Algebra]\label{prop:dephys-horizon-quotient-data-structure}
Under the setting of Definition \ref{def:dephys-ext-observable-algebra-equivalence}, there exists a natural quotient map (restriction map)
\begin{equation}\label{eq:dephys-ext-quotient-map}
\boxed{\;
\pi_{\mathrm{ext}}:\ \mathcal{D}(\mathcal{H}_{\mathrm{bulk}})\twoheadrightarrow \mathrm{St}(\mathcal{A}_{\mathrm{ext}}),
\qquad
(\pi_{\mathrm{ext}}(\rho))(A):=\operatorname{Tr}(\rho A)\; }.
\end{equation}
Moreover, for any \(\rho,\tilde\rho\in\mathcal{D}(\mathcal{H}_{\mathrm{bulk}})\),
\[
\boxed{\;
\rho\sim_{\mathrm{ext}}\tilde\rho
\quad\Longleftrightarrow\quad
\pi_{\mathrm{ext}}(\rho)=\pi_{\mathrm{ext}}(\tilde\rho).\;}
\]
Therefore, for any \(\sigma\in \mathrm{St}(\mathcal{A}_{\mathrm{ext}})\), its fiber
\(
\pi_{\mathrm{ext}}^{-1}(\sigma)
\)
can be understood as "the set of internal configurations that are externally indistinguishable" (external equivalence class).
\end{proposition}

\begin{proof}
By the definition in \eqref{eq:dephys-ext-quotient-map}, \(\pi_{\mathrm{ext}}(\rho)\) is precisely the linear functional induced by \(\rho\) on \(\mathcal{A}_{\mathrm{ext}}\);
and \(\pi_{\mathrm{ext}}(\rho)=\pi_{\mathrm{ext}}(\tilde\rho)\) is equivalent to
\(\operatorname{Tr}(\rho A)=\operatorname{Tr}(\tilde\rho A)\) for all \(A\in\mathcal{A}_{\mathrm{ext}}\), which is exactly \eqref{eq:dephys-ext-equivalence}.
This completes the proof.
\end{proof}

\paragraph{Horizon Sufficiency and Exponential Capacity: Recoverability Criterion Under Conditional Expectation Caliber}
To lift the above "quotient data structure" from set-theoretic level equivalence class characterization to auditable operator algebra caliber,
the following adopts the setting of \(\sigma\)-finite von Neumann algebra embedding and faithful normal conditional expectation,
and measures "external compression" information loss by Araki--Umegaki relative entropy (\cite{Araki1976RelativeEntropyVonNeumann}).
This caliber makes "external verification = zero-knowledge predicate" strictly equivalent to recoverability in the sense of sufficient statistics (Petz sufficiency theorem; \cite{Petz1986SufficientSubalgebrasRelativeEntropy,Petz2003MonotonicityRelativeEntropyRevisited}),
and locks the "fiber capacity" worst-case logarithmic scale to the logarithm of finite index (\cite{PimsnerPopa1986EntropyIndexSubfactors,Hiai1991MinimumIndexSubfactorsEntropyII}).

\begin{definition}[Externalization Channel and Fixed-Point State Space]\label{def:dephys-conditional-expectation-externalization-map}
Suppose \(\mathcal{A}_{\mathrm{ext}}\subseteq \mathcal{A}_{\mathrm{bulk}}\) is a \(\sigma\)-finite von Neumann algebra embedding,
and let \(E:\mathcal{A}_{\mathrm{bulk}}\to\mathcal{A}_{\mathrm{ext}}\) be a faithful normal conditional expectation (thus \(E\) is a unit-preserving, completely positive and idempotent normal map, and is the identity on \(\mathcal{A}_{\mathrm{ext}}\)).
Denote by \(\mathcal{S}(\mathcal{A})\) all normal states on \(\mathcal{A}\).
\par
Define the externalization operator (idempotent projection) on the state space
\begin{equation}\label{eq:dephys-ext-externalization-Pi}
\boxed{\;
\Pi:\mathcal{S}(\mathcal{A}_{\mathrm{bulk}})\to \mathcal{S}(\mathcal{A}_{\mathrm{bulk}}),
\qquad
\Pi(\rho):=\rho\circ E.\;}
\end{equation}
and define its fixed-point set
\[
\mathrm{Fix}(\Pi):=\{\rho\in\mathcal{S}(\mathcal{A}_{\mathrm{bulk}}):\ \Pi(\rho)=\rho\}.
\]
Since \(E\) is the identity on \(\mathcal{A}_{\mathrm{ext}}\), the restriction map \(\rho\mapsto \rho|_{\mathcal{A}_{\mathrm{ext}}}\) is naturally identified with \(\mathcal{S}(\mathcal{A}_{\mathrm{ext}})\) on \(\mathrm{Fix}(\Pi)\).
\end{definition}

\begin{proposition}[Horizon Sufficiency Theorem: "Zero-Knowledge-Type" Necessary and Sufficient Condition for External Verification]\label{prop:dephys-horizon-sufficiency-zero-knowledge}
Under the setting of Definition \ref{def:dephys-conditional-expectation-externalization-map},
let predicate \(V:\mathcal{S}(\mathcal{A}_{\mathrm{bulk}})\to\{0,1\}\) satisfy external invariance:
\[
\rho|_{\mathcal{A}_{\mathrm{ext}}}=\rho'|_{\mathcal{A}_{\mathrm{ext}}}
\ \Longrightarrow\
V(\rho)=V(\rho').
\]
Then there exists a unique predicate \(v:\mathrm{Fix}(\Pi)\to\{0,1\}\) such that
\begin{equation}\label{eq:dephys-zero-knowledge-factorization-by-Pi}
\boxed{\;
V=v\circ\Pi.\;}
\end{equation}
In particular, any external "verifiable statement" can only depend on the externalized state \(\Pi(\rho)\) (equivalently depend on \(\rho|_{\mathcal{A}_{\mathrm{ext}}}\)), and cannot distinguish microscopic states within the same external equivalence class fiber.
\end{proposition}

\begin{proof}
Note that for any \(\rho\in\mathcal{S}(\mathcal{A}_{\mathrm{bulk}})\) and any \(A\in\mathcal{A}_{\mathrm{ext}}\), we have
\(
\Pi(\rho)(A)=\rho(E(A))=\rho(A)
\), so \(\Pi(\rho)|_{\mathcal{A}_{\mathrm{ext}}}=\rho|_{\mathcal{A}_{\mathrm{ext}}}\).
Therefore external invariance is equivalent to \(V\) being constant on fibers of \(\Pi\).
Let \(v(\sigma):=V(\rho)\), where \(\sigma=\Pi(\rho)\in\mathrm{Fix}(\Pi)\);
this definition is well-defined by constancy, and clearly satisfies \(V=v\circ\Pi\).
Uniqueness comes from \(\Pi\) being the identity on its image: if \(V=v_1\circ\Pi=v_2\circ\Pi\), then for any \(\sigma\in\mathrm{Fix}(\Pi)\) take \(\rho=\sigma\) to get \(v_1(\sigma)=v_2(\sigma)\). This completes the proof.
\end{proof}

\begin{theorem}[Horizon Sufficiency: Relative Entropy Fidelity \(\Longleftrightarrow\) Recoverability]\label{thm:dephys-petz-sufficiency-equivalences}
Under the setting of Definition \ref{def:dephys-conditional-expectation-externalization-map},
let \(\mathcal{F}\subset\mathcal{S}(\mathcal{A}_{\mathrm{bulk}})\) be a family of normal states.
The following are equivalent:
\begin{enumerate}
  \item[(i)] \(E\) is sufficient for \(\mathcal{F}\): externalization \(\rho\mapsto \rho|_{\mathcal{A}_{\mathrm{ext}}}\) is a lossless characterization on \(\mathcal{F}\) (i.e., there exists an external state family \(\mathcal{F}_{\mathrm{ext}}\subset\mathcal{S}(\mathcal{A}_{\mathrm{ext}})\) and a bijective correspondence, such that statistical discrimination in \(\mathcal{F}\) remains recoverable after externalization);
  \item[(ii)] For any \(\rho,\sigma\in\mathcal{F}\), Araki--Umegaki relative entropy satisfies the fidelity equality
  \begin{equation}\label{eq:dephys-relative-entropy-sufficiency-equality}
  \boxed{\;
  D(\rho\|\sigma)=D(\rho\circ E\ \|\ \sigma\circ E)\;}
  \end{equation}
  (where \(D\) is relative entropy on von Neumann algebras; \cite{Araki1976RelativeEntropyVonNeumann});
  \item[(iii)] There exists a completely positive unital map (recovery operator) \(R:\mathcal{A}_{\mathrm{ext}}\to\mathcal{A}_{\mathrm{bulk}}\), such that for all \(\rho\in\mathcal{F}\)
  \begin{equation}\label{eq:dephys-recovery-map-identity-on-family}
  \boxed{\;
  \rho=\rho\circ R\circ E.\;}
  \end{equation}
\end{enumerate}
This equivalence makes "horizon as verification interface" rigorous as a recoverability criterion in the sense of sufficient statistics (Petz sufficiency theorem).
\end{theorem}

\begin{proof}[Proof outline]
By the data processing inequality for relative entropy (\cite{Petz2003MonotonicityRelativeEntropyRevisited}), for any \(\rho,\sigma\) we always have
\(
D(\rho\|\sigma)\ge D(\rho\circ E\|\sigma\circ E)
\).
Petz theory characterizes the equality condition, and gives the necessary and sufficient condition that equality holds if and only if there exists a recovery operator such that \(\rho=\rho\circ R\circ E\) (\cite{Petz1986SufficientSubalgebrasRelativeEntropy}).
Therefore \((ii)\Longleftrightarrow(iii)\) holds; and "sufficiency" in the statistical sense is synonymous with "relative entropy full fidelity/recoverability" under this setting, so it is equivalent to (i). This completes the proof.
\end{proof}

\begin{corollary}[Sufficient Statistic \(\Rightarrow\) Universal Zero-Knowledge Simulation: Channel Factorization Induced by Recovery]\label{cor:dephys-universal-zk-simulation-factorization}
Under the setting of Theorem \ref{thm:dephys-petz-sufficiency-equivalences},
if there exists a recovery operator \(R:\mathcal{A}_{\mathrm{ext}}\to\mathcal{A}_{\mathrm{bulk}}\) such that
\(\rho=\rho\circ R\circ E\) holds for all \(\rho\in\mathcal{F}\),
then for any output von Neumann algebra \(\mathcal{C}\) and any normal quantum channel

\begin{equation}\label{eq:dephys-universal-zk-factorization-on-family}
\boxed{\;
\Gamma_*(\rho)=\widetilde\Gamma_*(\rho\circ E)\qquad(\forall\,\rho\in\mathcal{F}).\;}
\end{equation}
And an explicit simulator construction can be taken:
\[
\boxed{\;

\]
Therefore, "externally verifiable output on \(\mathcal{F}\) is completely determined by external image \(\rho\circ E\)" is not only a set-theoretic level quotient space predicate closure,

\end{corollary}

\begin{proof}
For any \(\rho\in\mathcal{F}\), by assumption \(\rho=\rho\circ R\circ E\) we get
\[
\widetilde\Gamma_*(\rho\circ E)
=\Gamma_*\bigl((\rho\circ E)\circ R\bigr)
=\Gamma_*(\rho),
\]
which gives \eqref{eq:dephys-universal-zk-factorization-on-family}.
By normality and affinity of \(\sigma\mapsto \sigma\circ R\) and \(\Gamma_*\), \(\widetilde\Gamma_*\) remains a normal quantum channel. This completes the proof.
\end{proof}

\begin{corollary}[Full State Family Recoverability \(\Rightarrow\) Trivial Horizon: \(\mathrm{Ind}(E)=1\)]\label{cor:dephys-full-state-recovery-forces-trivial-index1}

\end{corollary}

\begin{proof}
By \eqref{eq:dephys-left-inverse-recovery-map}, \(E\) is injective.

By Definition \ref{def:dephys-pimsner-popa-index}, \(\mathrm{Ind}(\mathrm{id})=1\). This completes the proof.

\begin{definition}[Pimsner--Popa Finite Index]\label{def:dephys-pimsner-popa-index}

\boxed{\;
E(x^\ast x)\ \ge\ \Lambda^{-1}\,x^\ast x.\;}
\end{equation}
Define its index as
\[
\boxed{\;
\mathrm{Ind}(E):=\inf\{\Lambda:\ \eqref{eq:dephys-pimsner-popa-inequality}\ \text{holds}\}\ \in\ (1,\infty].\;}
\]
\end{definition}

E_{21}:\mathcal{A}_2\to\mathcal{A}_1
\), \(
E_{10}:\mathcal{A}_1\to\mathcal{A}_0
\), and let \(E_{20}:=E_{10}\circ E_{21}\).
If all three have finite index, then
\[
\boxed{\;
\mathrm{Ind}(E_{20})=\mathrm{Ind}(E_{21})\,\mathrm{Ind}(E_{10}),\qquad
\log\mathrm{Ind}(E_{20})=\log\mathrm{Ind}(E_{21})+\log\mathrm{Ind}(E_{10}).\;}
\]
And for any \(\rho,\sigma\in\mathcal{S}(\mathcal{A}_2)\) we have the chain identity for relative entropy loss
\[
\boxed{\;
D(\rho\|\sigma)-D(\rho\circ E_{20}\ \|\ \sigma\circ E_{20})
=\Delta_{21}(\rho,\sigma)+\Delta_{10}(\rho\circ E_{21},\sigma\circ E_{21}),\;}
\]
where \(\Delta_{ab}(\cdot,\cdot):=D(\cdot\|\cdot)-D(\cdot\circ E_{ab}\ \|\ \cdot\circ E_{ab})\).
\end{corollary}

\begin{proof}
Index multiplicativity is a standard property of finite index conditional expectations (see \cite{PimsnerPopa1986EntropyIndexSubfactors,Hiai1991MinimumIndexSubfactorsEntropyII}).
The chain identity is obtained directly by substituting \(E_{20}=E_{10}\circ E_{21}\) and expanding term by term with successive cancellation. This completes the proof.
\end{proof}

\begin{proposition}[Commuting Square and Mutual Information Defect: Decidable Compatibility of Two Externalization Processes]\label{prop:dephys-commuting-square-mutual-information-defect}
Suppose \(\mathcal{A}_{\mathrm{ext}}^{(1)},\mathcal{A}_{\mathrm{ext}}^{(2)}\subseteq\mathcal{A}_{\mathrm{bulk}}\) are two external subalgebras,
and let \(E_1,E_2\) be the corresponding faithful normal conditional expectations.
If the commuting square compatibility is satisfied
\[
E_1E_2=E_2E_1=:E_{12},
\]
then for any \(\rho,\sigma\in\mathcal{S}(\mathcal{A}_{\mathrm{bulk}})\) we have the non-negative defect identity
\begin{equation}\label{eq:dephys-mutual-information-defect}
\boxed{\;
\Delta_{1}(\rho,\sigma)+\Delta_{2}(\rho,\sigma)-\Delta_{12}(\rho,\sigma)\ \ge\ 0,\;}
\end{equation}
where \(\Delta_i(\rho,\sigma):=D(\rho\|\sigma)-D(\rho\circ E_i\ \|\ \sigma\circ E_i)\).
Equality holds if and only if the two externalization processes do not introduce additional unrecoverable loss on \((\rho,\sigma)\) (equivalently: relative entropy attains equality for both two-step externalizations simultaneously; see \cite{Petz2003MonotonicityRelativeEntropyRevisited} for characterization of equality conditions).
\end{proposition}

\begin{proof}[Proof outline]
This inequality is an equivalent statement of monotonicity and strong subadditivity structure of relative entropy under commuting-square constraints;
\(E_{12}\) can be viewed as the common refinement of two-step coarse-graining, and Petz's equality condition can be used to translate "no additional loss" into recoverability conditions. This completes the proof.
\end{proof}

\begin{theorem}[Möbius--Markov Uniqueness: Poisson Semigroup is the Unique Covariant Flow for Horizon Coarse-Graining]\label{thm:dephys-mobius-markov-unique-poisson}
Suppose \((T_t)_{t\ge 0}\) is a strongly continuous Markov semigroup acting on integrable densities on \(\RR\), satisfying:
\begin{enumerate}
  \item[(i)] Covariance with boundary affine Möbius automorphisms of \(\mathbb{H}\) (\(x\mapsto ax+b\), \(a>0\));
  \item[(ii)] Preserves Lebesgue measure and satisfies the strong maximum principle;
  \item[(iii)] The generator in frequency domain is a radially isotropic homogeneous first-order operator: there exists constant \(c>0\) such that its Fourier symbol is \(c|\xi|\).
\end{enumerate}
Then \(T_t\) agrees with the Poisson semigroup: for any \(f\in L^1(\RR)\)
\begin{equation}\label{eq:dephys-poisson-semigroup-fourier}
\boxed{\;
\widehat{T_t f}(\xi)=e^{-t|\xi|}\,\widehat f(\xi)\qquad(\xi\in\RR),\;}
\end{equation}
thus \(T_t f=P_t*f\), where \(P_t(x)=\frac{1}{\pi}\frac{t}{x^2+t^2}\) is the Poisson kernel.
Therefore under the constraint of "horizon as the unique continuous interface", coarse-graining dynamics under covariance and Markov requirements is unique.
\end{theorem}

\begin{proof}
By strongly continuous Markov semigroup and the homogeneous symbol assumption in (iii), \(T_t\) in Fourier representation must be a multiplier semigroup
\(
\widehat{T_t f}(\xi)=e^{-t c|\xi|}\widehat f(\xi)
\).
By normalization of (i)–(ii) (preserving measure/probability kernel mass) we can take \(c=1\), thus obtaining \eqref{eq:dephys-poisson-semigroup-fourier}.
Taking inverse Fourier transform of \(e^{-t|\xi|}\) gives the Poisson kernel \(P_t\), so \(T_t f=P_t*f\). This completes the proof.
\end{proof}

\begin{corollary}[Stieltjes Inversion of Scale Entropy Curve: Depth Spectrum Can Be Uniquely Recovered on Single-Variable Orbit]\label{cor:dephys-stieltjes-inversion-scale-entropy}
Under finite point defect caliber, the scale entropy curve
\[
S_{\mathrm{def}}(t):=\sum_j m_j\,\frac{\delta_j}{t+1+\delta_j}\qquad(t>0)
\]
is a Stieltjes function, and its complex analytic continuation has simple poles at \(t=-1-\delta_j\) with residue \(m_j\delta_j\);
therefore \(S_{\mathrm{def}}(t)\) on any set containing infinitely many points in \(t>0\) uniquely determines the multiset \(\{(\delta_j,m_j)\}\) (merging atoms of the same depth, ignoring permutation).
Moreover, its derivative chain gives a completely positive moment chain
\[
\boxed{\;
\frac{(-1)^n}{n!}\,S_{\mathrm{def}}^{(n)}(0)
=\sum_j m_j\,\delta_j(1+\delta_j)^{-n-1}\qquad(n\ge 0).\;}
\]
This conclusion is consistent with the complete monotonicity/pole identification of Proposition \ref{prop:xi-defect-entropy-scale-fingerprint-stieltjes}, and lifts it to a unique recovery statement in the sense of Stieltjes inversion.
\end{corollary}

\begin{corollary}[Distinguishability Lower Bound for Coarse-Graining: \texorpdfstring{$L^1$}{L1} Convergence Under Poisson Flow Controlled by Second-Order Central Moment]\label{cor:dephys-poisson-tv-bound-variance}
Suppose normalized profile density \(h:=H/\|H\|_{L^1(\RR)}\) is a finite Cauchy--Poisson mixture, and let \(h_t:=P_t*h\) (\(P_t\) is the Poisson kernel).
Denote the center of mass of \(h\) by \(\bar\gamma\), and give the second-order central moment with depth weight as
\(
\mathrm{Var}(\gamma)
\).
Then there exists an absolute constant \(C>0\) such that as \(t\to\infty\)
\[
\boxed{\;
\|h_t-\mathrm{Cauchy}(\bar\gamma,t)\|_{L^1(\RR)}
\ \le\ C\,\frac{\mathrm{Var}(\gamma)}{t^2}\;}
\]
where \(\mathrm{Cauchy}(\bar\gamma,t)(x)=\frac{1}{\pi}\frac{t}{(x-\bar\gamma)^2+t^2}\).
This inequality is compatible with the consistent \(O(t^{-2})\) asymptotics of Proposition \ref{prop:xi-poisson-coarsegraining-nohair}, and lifts the regime conclusion of "large-scale information collapse to two-parameter family" to a quantitative lower bound in total variation sense.
\end{corollary}

\begin{proof}[Proof outline]
By Taylor expansion in Proposition \ref{prop:xi-poisson-coarsegraining-nohair}, the difference between \(h_t\) and the single Cauchy kernel determined by \((\|H\|_1,\bar\gamma)\) is uniformly \(O(t^{-2})\) over \(x\), with constant controlled by the second-order central moment;
integrating over \(x\in\RR\) and using \(\int_{\RR}\frac{t}{(x^2+t^2)}\,dx=\pi\) and the \(t^{-3}\) uniform bound of the second derivative gives the \(L^1\) estimate. This completes the proof.
\end{proof}

\begin{theorem}[Idempotent Monad Characterization of Horizon Data Structure: Recycling Closure is Information Projection]\label{thm:dephys-idempotent-monad-i-projection}
Under the setting of Definition \ref{def:dephys-conditional-expectation-externalization-map}, \(\Pi(\rho)=\rho\circ E\) satisfies \(\Pi^2=\Pi\), thus defining an idempotent projection on the state space;
its fixed-point set \(\mathrm{Fix}(\Pi)\) is precisely all "externally visible states".
And for any \(\rho\in\mathcal{S}(\mathcal{A}_{\mathrm{bulk}})\) and any \(\sigma\in\mathrm{Fix}(\Pi)\), relative entropy satisfies the strict Pythagorean decomposition
\begin{equation}\label{eq:dephys-relative-entropy-pythagoras}
\boxed{\;
D(\rho\|\sigma)=D(\rho\|\Pi(\rho)) + D(\Pi(\rho)\|\sigma).\;}
\end{equation}
Therefore \(\Pi(\rho)\) is the unique information projection (\(I\)-projection) of \(\rho\) onto the set \(\mathrm{Fix}(\Pi)\):
invisible degrees of freedom are precisely encapsulated as the logarithmic potential \(D(\rho\|\Pi(\rho))\), while all externally verifiable predicates only depend on quotient class data \(\Pi(\rho)\).
\end{theorem}

\begin{proof}[Proof outline]
From \(E^2=E\) we get \(\Pi^2=\Pi\), and \(\sigma\in\mathrm{Fix}(\Pi)\) is equivalent to \(\sigma=\sigma\circ E\).
The Pythagorean decomposition \eqref{eq:dephys-relative-entropy-pythagoras} can be viewed as the orthogonal decomposition of relative entropy on "information projection" induced by conditional expectation:
its essence is equivalent to the chain formula of relative entropy and the "sufficient statistics/recoverability" structure of \(E\) (see \cite{Petz2003MonotonicityRelativeEntropyRevisited} for equality conditions and quantum Markov structure discussions).
In the finite-dimensional (density matrix) case, this decomposition can also be directly verified by orthogonal decomposition of log-likelihood; the general von Neumann algebra case is given by the modular operator representation of Araki relative entropy. This completes the proof.
\end{proof}

\begin{proposition}[Unique Entropy of Idempotent Recycling Monad: Axiomatic Characterization of \texorpdfstring{$D(\rho\|\Pi(\rho))$}{D(rho||Pi(rho))}]\label{prop:dephys-idempotent-monad-entropy-axiomatic-uniqueness}
Let \(\Pi\) be the externalization idempotent projection defined in \eqref{eq:dephys-ext-externalization-Pi}.
Suppose recycling cost functional \(C:\mathcal{S}(\mathcal{A}_{\mathrm{bulk}})\to[0,\infty)\) satisfies the following axiom family:
\begin{enumerate}
  \item \textbf{Monad invariance:} \(C(\rho)=0\iff \rho\in\mathrm{Fix}(\Pi)\);
  \item \textbf{Additivity:} for tensor product externalization \(\Pi_{12}:=\Pi_1\otimes\Pi_2\) we have \(C(\rho_1\otimes\rho_2)=C(\rho_1)+C(\rho_2)\);
  \item \textbf{Monotonicity:} for any quantum channel \(\mathcal{N}\) commuting with \(\Pi\) we have \(C(\mathcal{N}\rho)\le C(\rho)\);
  \item \textbf{\(I\)-projection decomposition:} there exists constant \(c>0\) such that for any \(\sigma\in\mathrm{Fix}(\Pi)\)
  \[
  \boxed{\;
  D(\rho\|\sigma)=c^{-1}C(\rho)+D(\Pi(\rho)\|\sigma)\;}
  \]
  (isomorphic to \eqref{eq:dephys-relative-entropy-pythagoras}).
\end{enumerate}
Then \(C\) must be the unique scaling of relative entropy projection potential:
\[
\boxed{\;
C(\rho)=c\,D\bigl(\rho\ \|\ \Pi(\rho)\bigr)\;}
\]
where the proportionality constant \(c\) only reflects logarithm base/unit choice.
\end{proposition}
\begin{proof}[Proof outline]
By taking \(\sigma=\Pi(\rho)\in\mathrm{Fix}(\Pi)\) in the fourth condition we get
\(
D(\rho\|\Pi(\rho))=c^{-1}C(\rho)
\), i.e., \(C(\rho)=cD(\rho\|\Pi(\rho))\).
Compatibility of other axioms with the structure of Theorem \ref{thm:dephys-idempotent-monad-i-projection} (monotonicity, additivity) is given by standard properties of relative entropy. This completes the proof.
\end{proof}

\begin{corollary}[Rényi Flatness: Worst-Case Loss Under Finite Folding Horizon Independent of Order \texorpdfstring{$\alpha$}{alpha}]\label{cor:dephys-renyi-flatness-fold-commutative}
Suppose \(\Omega\twoheadrightarrow X\) is a finite commutative folding, and let \(E:\CC^{\Omega}\to\CC^{X}\) be the conditional expectation of uniform averaging over fibers.
Denote the maximum fiber multiplicity as \(M:=\max_{x\in X}d(x)\), then \(\mathrm{Ind}(E)=M\) (commutative caliber of Proposition/Corollary \ref{prop:op-algebra-index-extremal-entropy-loss-maxfiber}).
For any \(\alpha\in[1,\infty]\), let \(D_\alpha\) be the sandwiched Rényi relative entropy (in the commutative case), then the flat extremal law holds
\[
\boxed{\;
\sup_{\rho}D_\alpha(\rho\mid\rho\circ E)=\log M=\log\mathrm{Ind}(E),\qquad(\forall\,\alpha\in[1,\infty]).\;}
\]
And the supremum is achieved by "point mass on maximum fiber": take \(a\in\Omega\) such that \(d(\Fold(a))=M\), then for \(\rho=\delta_a\)
\(
D_\alpha(\rho\mid\rho\circ E)=\log M
\)
holds simultaneously for all \(\alpha\in[1,\infty]\).
\end{corollary}

\begin{proof}
By Appendix Corollary \ref{cor:op-algebra-renyi-loss-spectrum-capped-by-index}, for any \(\rho\) and any \(\alpha\) we have
\(
D_\alpha(\rho\mid\rho\circ E)\le \log\mathrm{Ind}(E)=\log M
\).
On the other hand, for \(a\) belonging to a maximum fiber, when \(\rho=\delta_a\), \(\rho\circ E\) is the uniform state on that fiber;
in the commutative case, directly substituting into the Rényi divergence definition gives
\(
D_\alpha(\delta_a\mid \mathrm{Unif}_{F^\star})=\log M
\)
(independent of \(\alpha\)), thus the upper bound is attained and gives the supremum. This completes the proof.
\end{proof}
