
\section{算法伪代码}

\subsection{折叠与时间推进（元胞规则）}

\paragraph{输入：} $x\in\{0,1\}^n$。

\paragraph{Fold/Head（$\Head(x)$）：}
\begin{verbatim}
H[1] <- x[1]
for i = 2..n:
  H[i] <- x[i] * (1 - x[i-1])
\end{verbatim}

\paragraph{Time/Tail（$\Tail(x)$）：}
\begin{verbatim}
T[1] <- 0
for i = 2..n:
  T[i] <- x[i] * x[i-1]
\end{verbatim}

\subsection{Slack 分解与逆映射}

\paragraph{输入：} $z\in \Zn$。
\begin{verbatim}
Find positions i1 < ... < ik where z[ij] = 1
for j = 1..k-1:
  s[j] <- i[j+1] - i[j] - 1
s[k] <- n - i[k] + 1

Choose t[j] in {0,1,...,s[j]-1}, set ell[j] <- t[j] + 1
Initialize x = all zeros
for j = 1..k:
  for r = 0..ell[j]-1:
    x[i[j] + r] <- 1
return x
\end{verbatim}

\subsection{直接构造宏观加权图 $Q_n$}

\paragraph{输入：} $n$。
\begin{verbatim}
Enumerate all z in Z_n
for each z:
  compute head positions i1<...<ik and slack vector s[1..k]
  enumerate all subsets J of {j : s[j] > 1}:
    define z_J by putting 1 at positions i[j]+1 for j in J
    weight w <- product over j in J of (s[j] - 1)
    add directed edge z -> z_J with weight w
\end{verbatim}
