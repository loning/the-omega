\paragraph{Busemann coordinates and horocycle foliation with basepoint \texorpdfstring{$w=-1$}{w=-1}}\label{par:app-horocycle-busemann}
Under the Cayley compactification \eqref{eq:app-cayley-disk}, the endpoint \(w=-1\) is the common image of \(t\to\pm\infty\).
Around this endpoint, the ``off-critical-line shift'' can be strictly expressed as a hyperbolic-geometric coordinate with basepoint \(-1\), thereby yielding a horocycle foliation aligned with the Jensen defect resolution threshold.

\begin{proposition}[Endpoint Busemann/Poisson coordinate (exponentiated Busemann function)]\label{prop:app-busemann-poisson-minus1}
For \(w\in\mathbb{D}\) define
\[
\mathcal{B}_{-1}(w):=\Im\bigl(\mathcal{C}^{-1}(w)\bigr).
\]
Then \(\mathcal{B}_{-1}(w)\) has the completely explicit closed form
\begin{equation}\label{eq:app-busemann-poisson-minus1-closed}
\boxed{\;
\mathcal{B}_{-1}(w)
=\frac{1-|w|^{2}}{|1+w|^{2}}
\; }.
\end{equation}
Moreover, \(b_{-1}(w):=\log\mathcal{B}_{-1}(w)\) (up to a constant) can be taken as the Busemann function in the disk model with basepoint at the boundary point \(-1\); accordingly \(\mathcal{B}_{-1}\) is its exponentiated coordinate (equivalently, the corresponding Poisson kernel).
\end{proposition}

\begin{proof}
Let \(t=\mathcal{C}^{-1}(w)=x+\mathrm{i}y\in\mathbb{H}\).
From \eqref{eq:app-cayley-disk} one obtains
\(
1+w=1+\frac{1+\mathrm{i}t}{1-\mathrm{i}t}=\frac{2}{1-\mathrm{i}t}
\), whence
\[
|1+w|^{2}=\frac{4}{|1-\mathrm{i}t|^{2}}
=\frac{4}{x^{2}+(1+y)^{2}}.
\]
On the other hand,
\(
|w|^{2}=\frac{|1+\mathrm{i}t|^{2}}{|1-\mathrm{i}t|^{2}}
=\frac{x^{2}+(1-y)^{2}}{x^{2}+(1+y)^{2}}
\), so
\[
1-|w|^{2}
=\frac{(1+y)^{2}-(1-y)^{2}}{x^{2}+(1+y)^{2}}
=\frac{4y}{x^{2}+(1+y)^{2}}.
\]
Dividing the two expressions yields
\(
\frac{1-|w|^{2}}{|1+w|^{2}}=y=\Im(t)=\mathcal{B}_{-1}(w)
\), so \eqref{eq:app-busemann-poisson-minus1-closed} holds.
The statement about the Busemann function is a standard fact of hyperbolic geometry: the Busemann function in the disk model with basepoint \(\xi\in\partial\mathbb{D}\) can be taken as \(\log\frac{1-|w|^{2}}{|w-\xi|^{2}}\) (up to a constant), so at \(\xi=-1\) one obtains \(b_{-1}(w)=\log\mathcal{B}_{-1}(w)\). \qed
\end{proof}

\begin{theorem}[Horocycle foliation: level sets of \texorpdfstring{$\mathcal{B}_{-1}(w)=\delta$}{Bminus1=delta} are internally tangent circles]\label{thm:app-horocycle-foliation-minus1}
For any \(\delta>0\) define
\[
\mathcal{H}_{\delta}:=\Bigl\{w\in\mathbb{D}:\ \mathcal{B}_{-1}(w)=\delta\Bigr\}
=\Bigl\{w\in\mathbb{D}:\ \frac{1-|w|^{2}}{|1+w|^{2}}=\delta\Bigr\}.
\]
Then \(\mathcal{H}_{\delta}\) is a Euclidean circle internally tangent to the unit circle \(\partial\mathbb{D}\) at the point \(-1\) (a horocycle in hyperbolic geometry), with the explicit circle equation
\begin{equation}\label{eq:app-horocycle-circle-euclid}
\boxed{\;
\mathcal{H}_{\delta}
=\left\{w\in\mathbb{D}:\ \left|w+\frac{\delta}{1+\delta}\right|=\frac{1}{1+\delta}\right\}.
\;}
\end{equation}
In the upper half-plane coordinates, \(\mathcal{C}^{-1}(\mathcal{H}_{\delta})=\{t\in\mathbb{H}:\Im(t)=\delta\}\) is a horizontal line; thus \(\delta\) can be understood as the Busemann coordinate layer with basepoint \(w=-1\).
\end{theorem}

\begin{proof}
By Proposition~\ref{prop:app-busemann-poisson-minus1}, \(\mathcal{C}^{-1}(\mathcal{H}_{\delta})\) equals the horizontal line \(\Im(t)=\delta\), which in the upper half-plane model is the horocycle with basepoint \(\infty\).
Pushing it back to the disk via \(\mathcal{C}\) yields the horocycle with basepoint \(-1\).
\par
To obtain the explicit Euclidean circle equation, set \(w=x+\mathrm{i}y\).
From \(\frac{1-|w|^{2}}{|1+w|^{2}}=\delta\) one obtains
\[
1-(x^{2}+y^{2})=\delta\bigl((1+x)^{2}+y^{2}\bigr).
\]
Rearranging and completing the square gives
\[
\left(x+\frac{\delta}{1+\delta}\right)^{2}+y^{2}=\frac{1}{(1+\delta)^{2}},
\]
which is \eqref{eq:app-horocycle-circle-euclid}.
The center is \(-\frac{\delta}{1+\delta}\in(-1,0)\) with radius \(\frac{1}{1+\delta}\), and its leftmost point is \(-1\), so the circle is internally tangent to \(\partial\mathbb{D}\) at \(-1\). \qed
\end{proof}

\begin{corollary}[Horocycle geometric realization and second-order suppression of off-critical-line shift]\label{cor:app-offcritical-horocycle-busemann}
Under the setup of Proposition~\ref{prop:app-offcritical-radius-compression}, if \(\rho=\beta+\mathrm{i}\gamma\) is an off-critical-line zero (\(\beta<\tfrac12\)), let \(\delta=\tfrac12-\beta>0\) and \(w_{\rho}=\mathcal{C}(\gamma+\mathrm{i}\delta)\).
Then
\[
\boxed{\;\delta=\mathcal{B}_{-1}(w_{\rho})=\frac{1-|w_{\rho}|^{2}}{|1+w_{\rho}|^{2}}\;},
\qquad\text{and hence}\qquad
w_{\rho}\in\mathcal{H}_{\delta}.
\]
Moreover, the following completely explicit identities hold:
\[
\boxed{\;
|1+w_{\rho}|^{2}=\frac{4}{\gamma^{2}+(1+\delta)^{2}},
\qquad
1-|w_{\rho}|^{2}=\frac{4\delta}{\gamma^{2}+(1+\delta)^{2}}
=\delta\,|1+w_{\rho}|^{2}\; }.
\]
Therefore, in the natural scale of the endpoint neighborhood \(|1+w|\ll 1\), the ``off-critical-line shift \(\delta\)'' enters at \emph{second order in the endpoint chordal distance}: the radial gap \(1-|w|^{2}\) is strictly written as \(\delta|1+w|^{2}\), consistent with the ``second-order radial channel'' formulation of Remark~\ref{rem:app-offcritical-second-order-radial-suppression}.
\end{corollary}

\begin{proof}
By Proposition~\ref{prop:app-offcritical-radius-compression}, \(t_{\rho}=\gamma+\mathrm{i}\delta\) and \(w_{\rho}=\mathcal{C}(t_{\rho})\).
Therefore \(\delta=\Im(t_{\rho})=\Im(\mathcal{C}^{-1}(w_{\rho}))=\mathcal{B}_{-1}(w_{\rho})\), and by Theorem~\ref{thm:app-horocycle-foliation-minus1} one has \(w_{\rho}\in\mathcal{H}_{\delta}\).
On the other hand, from \(1+w_{\rho}=2/(1-\mathrm{i}t_{\rho})\) one obtains
\[
|1+w_{\rho}|^{2}=\frac{4}{|1-\mathrm{i}t_{\rho}|^{2}}
=\frac{4}{\gamma^{2}+(1+\delta)^{2}}.
\]
Combining with \eqref{eq:app-offcritical-radius-closed} gives
\(
1-|w_{\rho}|^{2}=\frac{4\delta}{\gamma^{2}+(1+\delta)^{2}}
\), and hence \(1-|w_{\rho}|^{2}=\delta|1+w_{\rho}|^{2}\). \qed
\end{proof}

\begin{theorem}[Comoving horizon localization: depth formula for the shifted Cayley transform]\label{thm:app-comoving-horizon-localization}
For any \(x_{0}\in\RR\) define the shifted Cayley compactification
\[
\mathcal{C}_{x_{0}}(t):=\mathcal{C}(t-x_{0}).
\]
Under the setup of Proposition~\ref{prop:app-offcritical-radius-compression}, if \(\rho=\beta+\mathrm{i}\gamma\) is an off-critical-line zero (\(\beta<\tfrac12\)), let \(\delta=\tfrac12-\beta>0\), and define
\[
w_{\rho,x_{0}}:=\mathcal{C}_{x_{0}}(\gamma+\mathrm{i}\delta)=\mathcal{C}\bigl((\gamma-x_{0})+\mathrm{i}\delta\bigr)\in\mathbb{D},
\qquad
h_{x_{0}}(\rho):=1-|w_{\rho,x_{0}}|^{2}.
\]
Then one has the closed forms
\[
\boxed{\;
\mathcal{B}_{-1}(w_{\rho,x_{0}})=\delta,\qquad
|1+w_{\rho,x_{0}}|^{2}=\frac{4}{(\gamma-x_{0})^{2}+(1+\delta)^{2}},\qquad
h_{x_{0}}(\rho)=\frac{4\delta}{(\gamma-x_{0})^{2}+(1+\delta)^{2}}
=\delta\,|1+w_{\rho,x_{0}}|^{2}\; }.
\]
In particular, taking \(x_{0}=\gamma\) gives
\[
w_{\rho,\gamma}=\mathcal{C}(\mathrm{i}\delta)=\frac{1-\delta}{1+\delta}\in(0,1),
\qquad
h_{\gamma}(\rho)=\frac{4\delta}{(1+\delta)^{2}}\sim 4\delta\qquad(\delta\downarrow 0),
\]
so that in the localized coordinates with comoving parameter \(x_{0}=\gamma\), the off-critical-line shift no longer suffers second-order compression as the height \(|\gamma|\) increases.
\end{theorem}

\begin{proof}
By definition \(w_{\rho,x_{0}}=\mathcal{C}(t)\) where \(t=(\gamma-x_{0})+\mathrm{i}\delta\), so
\(
\mathcal{B}_{-1}(w_{\rho,x_{0}})=\Im(\mathcal{C}^{-1}(w_{\rho,x_{0}}))=\Im(t)=\delta
\).
The remaining identities are entirely analogous to the proof of Corollary~\ref{cor:app-offcritical-horocycle-busemann}: from \(1+w_{\rho,x_{0}}=2/(1-\mathrm{i}t)\) one obtains
\[
|1+w_{\rho,x_{0}}|^{2}=\frac{4}{|1-\mathrm{i}t|^{2}}=\frac{4}{(\gamma-x_{0})^{2}+(1+\delta)^{2}},
\]
and from \(|\mathcal{C}(t)|^{2}=\frac{|1+\mathrm{i}t|^{2}}{|1-\mathrm{i}t|^{2}}=\frac{(\gamma-x_{0})^{2}+(1-\delta)^{2}}{(\gamma-x_{0})^{2}+(1+\delta)^{2}}\) one obtains
\(
h_{x_{0}}(\rho)=1-|w_{\rho,x_{0}}|^{2}=\frac{4\delta}{(\gamma-x_{0})^{2}+(1+\delta)^{2}}
\), and immediately \(h_{x_{0}}(\rho)=\delta\,|1+w_{\rho,x_{0}}|^{2}\).
Taking \(x_{0}=\gamma\) gives \(t=\mathrm{i}\delta\), so \(w_{\rho,\gamma}=\mathcal{C}(\mathrm{i}\delta)=\frac{1-\delta}{1+\delta}\), and substituting into the closed form yields \(h_{\gamma}(\rho)=\frac{4\delta}{(1+\delta)^{2}}\). \qed
\end{proof}

\begin{corollary}[Endpoint resolution lower bound (horocycle formulation)]\label{cor:app-horocycle-visible-resolution}
Suppose there exists an off-critical-line zero \(\rho=\beta+\mathrm{i}\gamma\) (\(\beta<\tfrac12\)), and let \(w_{\rho}\) be as in Corollary~\ref{cor:app-offcritical-horocycle-busemann}.
If one wishes to force a positive defect contribution from this zero at the radius audit point \(\varrho\in(0,1)\) (i.e., to guarantee \(\varrho>|w_{\rho}|\), so that \(D_{\zeta}(\varrho)\ge \log(\varrho/|w_{\rho}|)>0\), Corollary~\ref{cor:app-jensen-visible-threshold-offcritical}), then the following boundary-layer resolution threshold must be satisfied:
\[
1-\varrho<1-|w_{\rho}|
\le 1-|w_{\rho}|^{2}
=\delta\,|1+w_{\rho}|^{2}
=\frac{4\delta}{\gamma^{2}+(1+\delta)^{2}}.
\]
In particular, when \(|\gamma|\to\infty\) with \(\delta\) bounded, this necessary condition becomes
\[
1-\varrho\ \lesssim\ \frac{\delta}{\gamma^{2}},
\]
i.e., the radius audit required to exclude off-critical-line zeros in high-height intervals must enter the endpoint boundary-layer resolution of \(O(\delta/\gamma^{2})\).
\end{corollary}

\begin{proof}
If \(\varrho\le |w_{\rho}|\), then the disk \(|w|<\varrho\) does not contain the zero \(w_{\rho}\), and the Jensen summation formula \eqref{eq:app-jensen-sum} at \(\varrho\) does not include that zero's term, hence one cannot force a positive defect lower bound attributable to that zero.
Therefore one must have \(\varrho>|w_{\rho}|\), equivalently \(1-\varrho<1-|w_{\rho}|\).
The inequality \(1-|w_{\rho}|\le 1-|w_{\rho}|^{2}\) holds for all \(|w_{\rho}|<1\); the final two steps follow from the identities given in Corollary~\ref{cor:app-offcritical-horocycle-busemann}.
The high-height asymptotics follow from Corollary~\ref{cor:app-offcritical-radius-high-height} or directly from taking \(|\gamma|\to\infty\) in \(\frac{4\delta}{\gamma^{2}+(1+\delta)^{2}}\). \qed
\end{proof}
