\section{The Holographic Origin of Gravity: Phase Pressure}
\label{sec:phase_pressure}

\subsection{Quantum Gap and Projection Discrepancy}
In Paper I \cite{Ma2025HPA}, we established that the superposition of two polar states $\Psi_A + \Psi_B$ is not perfectly closed under unitary $\Theta$-scanning. The residual term, or ``quantum gap'', is not random noise but a deterministic consequence of projecting a high-dimensional helix onto a lower-dimensional grid. Rigorously, this is quantified by the \textbf{Discrepancy} $D_N$ of the sequence of projection phases $\{\vartheta_n\}$.

\begin{definition}[Phase readout sequence (local model)]
\label{def:phase_readout}
Fix a coarse-grained readout cell (spatial point) $\mathbf{x}$. A \emph{holographic readout} is a map that assigns to the scan orbit $|\Psi_n\rangle=\Theta^n|\Psi_0\rangle$ a phase $\vartheta_n(\mathbf{x})\in[0,2\pi)$.
We write the normalized phases $x_n(\mathbf{x}) := \vartheta_n(\mathbf{x})/(2\pi)\pmod{1}\in[0,1)$ and suppress the explicit $\mathbf{x}$-dependence when no confusion can arise.
\end{definition}

\begin{definition}[Star discrepancy of a phase orbit]
Let $x_n:=\vartheta_n/(2\pi)\pmod{1}\in[0,1)$ be the normalized projection phases. The (one-dimensional) star discrepancy of the first $N$ points is
\begin{equation}
  D_N^* := \sup_{0\le a\le 1}\left|\frac{1}{N}\#\{1\le n\le N: x_n<a\}-a\right|.
\end{equation}
\end{definition}

Weyl's equidistribution theorem implies $D_N^*\to 0$ for Kronecker orbits $x_n=\{n\alpha+\beta\}$ with $\alpha\notin\mathbb{Q}$ \cite{Weyl1916, KuipersNiederreiter1974}. Moreover, for \emph{badly approximable} $\alpha$ (in particular for the golden slope singled out in Paper~I), one has the quantitative bound
\begin{equation}
  D_N^* = O\!\left(\frac{\log N}{N}\right)
\end{equation}
with an implied constant depending on $\alpha$ \cite{KuipersNiederreiter1974}.

The key point for dynamics is that although the \emph{per-step} mismatch vanishes ($D_N^*\to 0$), the \emph{accumulated} mismatch
\begin{equation}
  \mathcal{E}_N := N D_N^*
\end{equation}
can grow slowly. In particular, for badly approximable slopes (including the golden case), quantitative discrepancy estimates imply
\begin{equation}
  \mathcal{E}_N=O(\log N),
\end{equation}
with constants controlled by the continued-fraction data of $\alpha$; see, e.g., \cite{Denjoy1932, Koksma1935, KuipersNiederreiter1974}.

\begin{remark}[Mismatch as a maximal counting deviation]
The quantity $\mathcal{E}_N$ can be rewritten as
\begin{equation}
  \mathcal{E}_N
  =\sup_{0\le a\le 1}\left|\#\{1\le n\le N:x_n<a\}-Na\right|,
\end{equation}
so it measures the \emph{largest} deviation (in absolute count) of empirical phase statistics from uniformity over all anchored intervals $[0,a)$.
\end{remark}

\begin{assumption}[Continuum phase potential sourced by mismatch]
\label{assump:mismatch_potential}
In the macroscopic continuum limit, the holographic readout assigns to each point (or coarse-grained cell) a scalar \emph{mismatch density} $\sigma(\mathbf{x})$ obtained by coarse-graining the microscopic discrepancy accumulation (e.g.\ from $\mathcal{E}_N(\mathbf{x})$ at large $N$).
This mismatch density sources a static scalar \textbf{phase potential} $\Phi(\mathbf{x})$ through a local Poisson law
\begin{equation}
  \Delta \Phi = 4\pi \rho_\Phi,\qquad \rho_\Phi := \kappa_\Phi\,\sigma,
\end{equation}
where the coupling $\kappa_\Phi$ is fixed by matching to the Newtonian limit. The associated \textbf{Phase Pressure} (force per unit test mass in the Newtonian sector) is conservative and given by
\begin{equation}
  \mathcal{P}_\Phi := -\nabla \Phi.
\end{equation}
Equivalently, $\Phi$ is characterized as a stationary point of a local quadratic Dirichlet functional (Appendix~\ref{app:dirichlet_poisson}).
\end{assumption}

\begin{remark}[Coarse-graining and renormalization of mismatch]
A fully microscopic definition of $\sigma(\mathbf{x})$ requires specifying how the readout chooses a time window $N$ and a spatial coarse-graining scale. One natural formalization is to introduce a family of coarse-grained fields $\sigma_\ell(\mathbf{x})$ obtained from $\mathcal{E}_{N(\ell)}(\mathbf{x})$ at a resolution scale $\ell$ (with $N(\ell)\to\infty$ as $\ell\to 0$) and assume $\sigma_\ell\to\sigma$ in a suitable weak sense. The present paper uses only the existence of the continuum field $\sigma$ and its Poisson coupling.
\end{remark}

Unlike Entropic Gravity \cite{Verlinde2011}, which relies on thermodynamic statistics, Phase Pressure is modeled here as a deterministic response of the readout potential $\Phi$ to projection mismatch.

\subsection{Mass as Topological Phase Flux}
We model mass as a defect sourcing the phase potential $\Phi$.

\begin{axiom}[Phase Gauss law for isolated defects]
\label{ax:phase_gauss}
In the static sector, an isolated defect of charge $\mathcal{Q}_{top}$ produces a phase potential satisfying the Gauss/Poisson law
\begin{equation}
  \oint_{\partial V} \nabla \Phi \cdot d\mathbf{A} = 4\pi \mathcal{Q}_{top}.
\end{equation}
Equivalently, in distributional form $\Delta \Phi = 4\pi \rho_\Phi$ with $\rho_\Phi=\mathcal{Q}_{top}\,\delta_0$ (Appendix~\ref{app:poisson_gauss}). The spherically symmetric solution is $\Phi(r)=-\mathcal{Q}_{top}/r$ up to an additive constant (Appendix~\ref{app:phase_poisson}).
\end{axiom}

To match Newtonian gravity in the weak-field limit, we identify the Newtonian potential as $\phi_N:=\Phi$ and fix $\mathcal{Q}_{top}=M$ by comparing $g_{tt}\approx -(1+2\phi_N)$ to the standard Schwarzschild asymptotics $g_{tt}\approx -(1-2M/r)$ \cite{Schwarzschild1916}.

\begin{proposition}[Newtonian acceleration from phase potential]
Interpreting Phase Pressure as force per unit test mass, the induced acceleration is $\mathbf{a}=-\nabla\phi_N=-\nabla\Phi$. For $\phi_N=\Phi=-M/r$ one obtains the Newtonian field
\begin{equation}
  \mathbf{a}(r)= -\frac{M}{r^2}\,\hat{\mathbf{r}} .
\end{equation}
\end{proposition}

\begin{remark}
If the phase is realized globally as a $U(1)$ connection (so that $\Phi$ is only locally a single-valued potential), then the defect charge can be identified with a Chern number and is quantized, exactly as in the Dirac monopole construction \cite{Dirac1931, WuYang1975}. In Omega Dynamics we adopt this quantization as a structural input for defect charges:
\begin{equation}
  \mathcal{Q}_{top}=n\,q_0,\qquad n\in\ZZ,
\end{equation}
with a fundamental unit $q_0$ fixed by the microscopic theory (in geometric units one may take $q_0$ at the Planck scale). Macroscopic masses correspond to large $n$; the present paper uses only the topological conservation/additivity implied by this structure.
\end{remark}

A massive object is thus a \textbf{Phase Monopole} in the static sector. It introduces a persistent lag in the $\Theta$-scan readout, and the induced potential $\phi_N=-M/r$ reproduces the usual weak-field metric deformation
\begin{equation}
  g_{tt}=-(1+2\phi_N)+O(\phi_N^2),\qquad
  g_{rr}=(1-2\phi_N)+O(\phi_N^2),
\end{equation}
so that $g_{tt}\approx -(1-2M/r)$ and $g_{rr}\approx 1+2M/r$ in Schwarzschild areal radius \cite{Wald1984}.

\subsection{Covariant Completion: From Phase Pressure to the Einstein Equation}
\label{sec:covariant_completion}
The previous subsections fix the \emph{Newtonian} sector: the static phase potential $\Phi$ solves a Poisson equation sourced by a defect charge, yielding the $1/r$ potential and therefore the $1/r^2$ acceleration law. To obtain a relativistic field equation, we now state the covariant closure assumptions and invoke standard uniqueness results.

\begin{assumption}[Local covariant continuum limit]
\label{assump:covariant_continuum}
At macroscopic scales, the coarse-grained Omega dynamics admits a continuum description by a Lorentzian metric $g_{\mu\nu}$ and an effective stress tensor $T_{\mu\nu}$ encoding phase/mismatch density, such that:
\begin{enumerate}
  \item the dynamics is local and diffeomorphism invariant,
  \item the metric field equation is second order in derivatives of $g_{\mu\nu}$,
  \item asymptotically isolated defects admit an asymptotically flat limit.
\end{enumerate}
\end{assumption}

\begin{theorem}[Uniqueness of the gravitational field equation in 4D]
Under the above assumptions, the most general local, symmetric, divergence-free rank-2 tensor built from $g_{\mu\nu}$ and at most its second derivatives is $G_{\mu\nu}+\Lambda g_{\mu\nu}$, where $\Lambda$ is a constant \cite{Lovelock1971}. Hence the macroscopic field equation must take the form
\begin{equation}
  G_{\mu\nu} + \Lambda g_{\mu\nu} = \kappa_G\, T_{\mu\nu},
\end{equation}
for some coupling constant $\kappa_G$.
\end{theorem}

\begin{proposition}[Fixing $\kappa_G$ by the Newtonian limit]
In the weak-field, slow-motion regime, write
\begin{equation}
  g_{tt}=-(1+2\phi_N)+O(\phi_N^2),\qquad \phi_N\ll 1,
\end{equation}
and take $T_{tt}\approx \rho_m$ for rest-mass density $\rho_m$. Then the $tt$-component reduces to the Poisson equation
\begin{equation}
  \Delta \phi_N = \frac{\kappa_G}{2}\,\rho_m ,
\end{equation}
see Appendix~\ref{app:newtonian_limit}. Matching to $\Delta \phi_N = 4\pi \rho_m$ fixes $\kappa_G=8\pi$ in geometric units \cite{Wald1984}.
\end{proposition}

\begin{corollary}[Schwarzschild exterior from Phase Pressure]
Assume an isolated defect whose macroscopic exterior is vacuum and spherically symmetric. Phase Pressure fixes $\phi_N=-M/r$ and therefore the asymptotic mass parameter. Outside the defect support, $T_{\mu\nu}=0$ and the field equation reduces to $G_{\mu\nu}=0$ (taking $\Lambda=0$ for asymptotically flat solutions). By Birkhoff's theorem, the exterior geometry is Schwarzschild with mass $M$ \cite{MisnerThorneWheeler1973, Wald1984}.
\end{corollary}

\begin{remark}[Thermodynamic reading]
The same conclusion is compatible with the ``Einstein equation as an equation of state'' viewpoint: assuming local horizon thermodynamics (Unruh temperature and area entropy) one can derive the Einstein equation from a Clausius relation \cite{Jacobson1995}. In Omega Dynamics, the phase/mismatch sector supplies the relevant coarse-grained heat/entropy flux.
\end{remark}
