\subsubsection{Cohomological Non-Negativization of Arity-Charge, Density Upper Bound, and Zero-Charge Subsystem}\label{app:real-input-40-arity-charge}
\paragraph{Overview}
This subsection studies the arity-charge observable $\chi$ on the real input essential $20$-state core of Proposition \ref{prop:real-input-40-essential-20}.
The main conclusions are: (\textup{i}) $\chi$ can be cohomologically gauged to an edge-wise $0/1$ indicator, so that $\chi(\gamma)\ge 0$ on closed orbits and the primitive generating function $P_n(q)$ contains no Laurent negative powers;
(\textup{ii}) every primitive orbit satisfies the density upper bound $\chi(\gamma)\le \lfloor|\gamma|/2\rfloor$, and this bound is sharp;
(\textup{iii}) the zero-charge stratum ($\chi(\gamma)=0$) induces a new SFT subsystem whose $\zeta$ and entropy admit low-degree algebraic closed forms;
(\textup{iv}) the $\chi$-weighted pressure at $\theta\to+\infty$ has the exact asymptotic line $P_\chi(\theta)=\frac12\theta+\frac12\log 3+o(1)$, so that the endpoint slope is explicitly $1/2$ with intercept $\frac12\log 3$.

\paragraph{Definition (arity-charge stratification and generating functions)}
In the graph representation of the real input kernel, each allowed edge corresponds to one step of legal input $(x,y)\in\{0,1\}^2$ (subject to a one-step memory constraint) and carries edge labels $(d,e)$:
where $d=x+y\in\{0,1,2\}$ and $e\in\{0,1\}$ is the synchronization kernel output bit.
Define the one-step arity-charge
\[
\chi(s,d):=e-\mathbf{1}_{\{d=2\}}\in\{-1,0,1\},
\]
and assign weight $q^{\chi}$ to each edge of the real input kernel, yielding the weighted matrix $M(q)$.
Let
\[
c_n(q):=\operatorname{tr}(M(q)^n),
\qquad
\zeta(z,q)=\exp\!\left(\sum_{n\ge 1}\frac{c_n(q)}{n}z^n\right)
=\frac{1}{\det(I-zM(q))}.
\]
Define the arity generating polynomial of primitive orbits
\[
P_n(q):=\sum_{\substack{\gamma\ \mathrm{primitive}\|\gamma|=n}} q^{\chi(\gamma)},
\qquad
\chi(\gamma):=\sum_{e\in\gamma}\chi(e),
\]
whose coefficients give stratified counts:
\[
[q^k]P_n(q)=\#\{\gamma:\ |\gamma|=n,\ \chi(\gamma)=k\}.
\]
The weighted M\"obius inversion then yields
\[
c_n(q)=\sum_{d\mid n} d\,P_d\!\left(q^{n/d}\right),
\qquad
P_n(q)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,c_{n/d}\!\left(q^{d}\right).
\]

\begin{theorem}[$0/1$ coboundary normal form of arity-charge (on the essential core)]\label{thm:real-input-40-arity-charge-coboundary}
On the essential $20$-state core of Proposition \ref{prop:real-input-40-essential-20}, there exists a potential function
\[
V:\Sigma_{\mathrm{ess}}\to\{0,1\},
\]
such that for every allowed edge $e:u\to v$ on the core,
\[
g(e):=\chi(e)+V(v)-V(u)\in\{0,1\}.
\]
Moreover, one may take $V=1$ if and only if
\[
(s,p_x,p_y)\in\{(1\overline{1}2,1,1),(101,1,1),(002,1,1),(11\overline{1},0,0)\},
\]
with $V=0$ on all remaining core states.
Equivalently, setting $\widehat V:=1-V$, then $\widehat V$ takes the value $0$ on the above $4$ core states and $1$ on the remaining $16$, and
\(
g(e)=\chi(e)+\widehat V(u)-\widehat V(v)\in\{0,1\}
\)
(fully consistent with the ``edge-wise non-negativization'' finite certificate caliber).
Therefore, for any periodic orbit (in particular, primitive) $\gamma$,
\[
\chi(\gamma)=\sum_{e\in\gamma}\chi(e)=\sum_{e\in\gamma}g(e)\ge 0.
\]
\end{theorem}

\begin{proof}
The core graph is finite and $V$ has been given explicitly; verifying $g(e)\in\{0,1\}$ edge by edge yields a finite certificate.
For any periodic orbit $\gamma$,
\[
\sum_{e\in\gamma}g(e)=\sum_{e\in\gamma}\chi(e)+\sum_{e:u\to v\in\gamma}\bigl(V(v)-V(u)\bigr)=\chi(\gamma),
\]
where the potential function terms telescope on a closed loop. Hence $\chi(\gamma)=\sum_{e\in\gamma}g(e)\ge 0$.
The auditable reproduction script is
\texttt{scripts/exp\_sync\_kernel\_real\_input\_40\_arity\_charge\_closed\_form.py}.
\end{proof}

\begin{corollary}[Prefix debt depth is $1$ (for any finite path)]\label{cor:real-input-40-arity-charge-prefix-debt}
For any finite allowed path \(\pi:u\leadsto v\) on the essential core,
\[
\sum_{e\in\pi}\chi(e)\ge -1.
\]
Equivalently, if one records the partial sums \(S_n:=\sum_{k=0}^{n-1}\chi(e_k)\) along a legal orbit, then for all \(n\) one has \(S_n\ge -1\):
the arity debt accumulates at most one unit at any time.
\end{corollary}

\begin{proof}
By Theorem \ref{thm:real-input-40-arity-charge-coboundary}, for each edge \(e:u\to v\) one has
\(
\chi(e)=g(e)-V(v)+V(u)
\)
with \(g(e)\in\{0,1\}\).
Summing along the path \(\pi\) gives
\[
\sum_{e\in\pi}\chi(e)
=\sum_{e\in\pi}g(e)+V(u)-V(v)\ \ge\ -1,
\]
where \(\sum g(e)\ge 0\) and \(V(u)-V(v)\ge -1\) since \(V\) takes values in \(\{0,1\}\).
\end{proof}

\begin{remark}[Notation alignment: $0/1$ binarization of \texorpdfstring{$\widetilde\chi$}{chi-tilde}]\label{rem:real-input-40-arity-charge-chi-tilde}
Set \(H:=V\), and for each edge \(e:u\to v\) define
\[
\widetilde\chi(e):=\chi(e)+H(v)-H(u).
\]
Then by Theorem \ref{thm:real-input-40-arity-charge-coboundary}, \(\widetilde\chi(e)=g(e)\in\{0,1\}\).
This formulation is fully equivalent to the ``edge-wise binarized coboundary normal form'': on closed orbits \(\sum\chi=\sum\widetilde\chi\ge 0\), and on finite paths the range of \(H\) directly yields the debt depth bound of Corollary \ref{cor:real-input-40-arity-charge-prefix-debt}.
\end{remark}

\begin{corollary}[(Originally numbered as Conjecture B.211) Non-negativity of closed orbit arity-charge]\label{cor:real-input-40-arity-charge-conjecture-b211}
On the essential $20$-state core of Proposition \ref{prop:real-input-40-essential-20}, for any periodic orbit \(\gamma\) (in particular, primitive),
\[
\chi(\gamma)\ge 0.
\]
\end{corollary}
\begin{proof}
By Theorem \ref{thm:real-input-40-arity-charge-coboundary}, for any periodic orbit \(\chi(\gamma)=\sum_{e\in\gamma}g(e)\ge 0\).
Therefore the original Conjecture B.211 holds; moreover, Theorem \ref{thm:real-input-40-arity-charge-coboundary} provides a stronger edge-wise \(0/1\) coboundary certificate.
\end{proof}

\begin{corollary}[Collision count does not exceed output count (at the closed orbit level)]\label{cor:real-input-40-arity-charge-collision-le-output}
On the same core, for any periodic orbit \(\gamma\) denote
\[
N_e(\gamma):=\sum_{k=1}^{|\gamma|} e_k,
\qquad
N_2(\gamma):=\sum_{k=1}^{|\gamma|}\mathbf{1}_{\{d_k=2\}}.
\]
Then \(N_2(\gamma)\le N_e(\gamma)\).
\end{corollary}
\begin{proof}
From \(\chi=e-\mathbf{1}_{\{d=2\}}\) one obtains \(\chi(\gamma)=N_e(\gamma)-N_2(\gamma)\).
Combining with Corollary \ref{cor:real-input-40-arity-charge-conjecture-b211}, \(\chi(\gamma)\ge 0\), yields \(N_2(\gamma)\le N_e(\gamma)\).
\end{proof}

\begin{remark}[Pure counting semantics of arity-charge]\label{rem:real-input-40-arity-charge-pure-counting}
From $\chi(\gamma)=\sum_{e\in\gamma}g(e)$ with $g(e)\in\{0,1\}$, for any periodic orbit $\gamma$ one has
\[
\chi(\gamma)=\#\{\,e\in\gamma:\ g(e)=1\,\},
\]
i.e., arity-charge (in the cohomological sense) equals the number of $g=1$ edges traversed along the orbit; consequently, no ``negative charge cancellation'' leading to Laurent negative powers can occur.
\end{remark}

\begin{corollary}[Primitive arity-charge polynomials have no Laurent negative powers]\label{cor:real-input-40-arity-charge-no-laurent}
For all $n\ge 1$,
\[
P_n(q)\in\ZZ_{\ge 0}[q].
\]
\end{corollary}

\begin{proof}
By Theorem \ref{thm:real-input-40-arity-charge-coboundary}, $\chi$ and $g$ have equal sums on periodic orbits, and $g$ takes values in $\{0,1\}$ on each edge.
Therefore each primitive orbit weight $q^{\chi(\gamma)}=q^{g(\gamma)}$ is a non-negative integer power, and the coefficients are counts.
\end{proof}

\begin{theorem}[Density upper bound and sharpness of arity-charge]\label{thm:real-input-40-arity-charge-density-bound}
For any primitive periodic orbit $\gamma$, let $n=|\gamma|$. Then
\[
0\le \chi(\gamma)\le \left\lfloor \frac{n}{2}\right\rfloor.
\]
Moreover, there exists a primitive orbit of length $2$ satisfying $\chi(\gamma)=1$, so the upper bound is attainable.
\end{theorem}

\begin{proof}
By Theorem \ref{thm:real-input-40-arity-charge-coboundary}, take $g(e)\in\{0,1\}$ such that $\chi(\gamma)=\sum_{e\in\gamma}g(e)$.
Set
\[
w(e):=1-2g(e)\in\{+1,-1\}.
\]
One may choose a potential function
\[
h:\Sigma_{\mathrm{ess}}\to\{-2,-1,0\}
\]
such that for every core edge $e:u\to v$,
\[
w(e)+h(u)-h(v)\ge 0.
\]
For example, the explicit assignment of $h$ on the $20$ core states may be taken as:
\[
\begin{aligned}
h=-2&\quad\Longleftrightarrow\quad (s,m)\in\{(000,00),(001,01),(001,10)\},\\
h=-1&\quad\Longleftrightarrow\quad (s,m)\in\{(000,01),(000,10),(010,00),(100,01),(100,10),(002,11),(11\overline{1},00)\},\\
h=0&\quad\Longleftrightarrow\quad \text{all remaining core states}.
\end{aligned}
\]
Verifying the above inequality edge by edge (there are $48$ edges in the core) yields a finite certificate.
Summing along a closed orbit gives $\sum_{e\in\gamma}w(e)\ge 0$, i.e., $\sum_{e\in\gamma}g(e)\le n/2$, from which the upper bound follows.
The attainability at length $2$ is witnessed in Table \ref{tab:real-input-40-rotation-polytope-vertices} by orbits with $\bar\chi=1/2$.
\end{proof}

\begin{theorem}[Parseval energy identity for the mean-square deviation of congruence classes]\label{thm:real-input-40-arity-charge-parseval-energy}
Let \(p_n:=P_n(1)=\sum_{r=0}^{m-1}N_{n,r}^{(m)}\). Then the exact identity holds:
\[
\sum_{r=0}^{m-1}\left|N_{n,r}^{(m)}-\frac{p_n}{m}\right|^2
=\frac{1}{m}\sum_{j=1}^{m-1}\left|P_n(\zeta_m^j)\right|^2.
\]
Equivalently, after normalization:
\[
\sum_{r=0}^{m-1}\left(\frac{N_{n,r}^{(m)}}{p_n}-\frac{1}{m}\right)^2
=\frac{1}{m\,p_n^2}\sum_{j=1}^{m-1}\left|P_n(\zeta_m^j)\right|^2.
\]
\end{theorem}
\begin{proof}
Set \(f(r):=N_{n,r}^{(m)}\) (\(r=0,1,\dots,m-1\)) and define its discrete Fourier transform
\[
\widehat f(j):=\sum_{r=0}^{m-1} f(r)\,\zeta_m^{jr}\qquad (j=0,1,\dots,m-1).
\]
From \(P_n(q)=\sum_k [q^k]P_n(q)\,q^k\) and the definition of \(f(r)\), one has
\[
\widehat f(j)=\sum_{r=0}^{m-1}\sum_{k\equiv r\ (\mathrm{mod}\ m)}[q^k]P_n(q)\,\zeta_m^{jr}
=\sum_k [q^k]P_n(q)\,\zeta_m^{jk}
=P_n(\zeta_m^j).
\]
Also \(p_n=\sum_r f(r)=\widehat f(0)\). Let \(\overline f:=p_n/m\) and \(g(r):=f(r)-\overline f\).
For \(j\neq 0\), \(\sum_{r=0}^{m-1}\overline f\,\zeta_m^{jr}=0\), so \(\widehat g(j)=\widehat f(j)=P_n(\zeta_m^j)\) (\(j=1,\dots,m-1\)), while \(\widehat g(0)=0\).
Applying the discrete Parseval identity to \(g\) yields
\[
\sum_{r=0}^{m-1}|g(r)|^2=\frac{1}{m}\sum_{j=0}^{m-1}|\widehat g(j)|^2
=\frac{1}{m}\sum_{j=1}^{m-1}|P_n(\zeta_m^j)|^2,
\]
which is the first identity. Dividing both sides by \(p_n^2\) gives the normalized form.
\end{proof}

\paragraph{Endpoint thermodynamics ($\theta\to\pm\infty$)}
\begin{proposition}[Endpoint pressure and extremal charge density]\label{prop:real-input-40-arity-charge-endpoints}
When $\theta\to-\infty$ (i.e., $q=e^\theta\to 0^+$),
\[
P_\chi(\theta)\to \log\kappa,
\]
where $\kappa>1$ is the Perron root of $\kappa^4-\kappa^3-1=0$ (same as in Proposition \ref{prop:real-input-40-arity-charge-zero-zeta}).
When $\theta\to+\infty$ (i.e., $q\to+\infty$),
\[
P_\chi(\theta)=\frac12\,\theta+\frac12\log 3+o(1),
\]
hence
\[
\sup_\mu\int\chi\,d\mu=\lim_{\theta\to+\infty}P_\chi'(\theta)=\frac12.
\]
\end{proposition}

\begin{proof}
Set $q=e^\theta$. When $\theta\to-\infty$, $M(q)=M_0+qM_1\to M_0$, and by continuity of the spectral radius,
$\rho(M(q))\to\rho(M_0)=\kappa$.
When $\theta\to+\infty$, the Perron root $\lambda(q)=\rho(M(q))$ satisfies the algebraic equation
$P_7(\lambda(q),q)=0$ from Theorem \ref{thm:real-input-40-arity-charge-det-closed}.
Setting $\lambda(q)=s\sqrt q$ and normalizing the equation by $q^{7/2}$, the leading-order balance gives
$s^3(s^4-5s^2+6)=0$, so the limiting branch satisfies $s\in\{\sqrt2,\sqrt3\}$.
Since the Perron root selects the largest positive real root branch, one obtains $s\to\sqrt3$, i.e., $\lambda(q)\sim \sqrt{3q}$,
hence $P_\chi(\theta)=\log\lambda(e^\theta)=\frac12\theta+\frac12\log 3+o(1)$.
The final identity follows from the endpoint slope.
\end{proof}

\begin{proposition}[Zero-temperature expansion: two-term closed form for \texorpdfstring{$q\to+\infty$}{q->+infty}]\label{prop:real-input-40-arity-charge-zero-temp-expansion}
When $q\to+\infty$, the Perron root $\lambda(q)=\rho(M(q))$ has the Puiseux asymptotic expansion
\[
\lambda(q)=\sqrt{3q}+\frac23+\frac{56}{81}\,q^{-1}+O(q^{-3/2}).
\]
Therefore the pressure function $P_\chi(\theta)=\log\lambda(e^\theta)$ satisfies
\[
P_\chi(\theta)=\frac12\,\theta+\frac12\log 3+\frac{2}{3\sqrt3}\,e^{-\theta/2}-\frac{2}{27}\,e^{-\theta}+O(e^{-3\theta/2}).
\]
\end{proposition}
\begin{proof}
By Theorem \ref{thm:real-input-40-arity-charge-det-closed}, $\lambda(q)$ is the largest positive real root branch of the algebraic equation $P_7(\lambda(q),q)=0$.
Set $t:=q^{-1/2}\to 0^+$ and substitute the Puiseux series
\[
\lambda(q)=\frac{\sqrt3}{t}+a_0+a_1t+a_2t^2+a_3t^3+\cdots.
\]
Substituting back into $P_7(\lambda(q),t^{-2})=0$ and matching coefficients of successive powers of $t$ yields $a_0=\frac23,\ a_1=0,\ a_2=\frac{56}{81}$,
giving $\lambda(q)=\sqrt{3q}+\frac23+\frac{56}{81}q^{-1}+O(q^{-3/2})$.
Taking the logarithm of
\(
P_\chi(\theta)=\log\lambda(e^\theta)
\)
with $\lambda(e^\theta)=\sqrt{3}\,e^{\theta/2}\bigl(1+\frac{2}{3\sqrt3}e^{-\theta/2}+O(e^{-\theta})\bigr)$ and expanding yields the stated pressure asymptotics.
\end{proof}
