\paragraph{Analytic Continuation of the Real-Parameter Moment Pressure: From Integer-Order Moment Spectra to Full Real-Line Convex Geometry}\label{par:pom-multiplicity-real-q-pressure}
In Definitions \ref{def:pom-multiplicity-composition-integer-moment-partition}--Proposition \ref{prop:pom-multiplicity-composition-moment-hierarchy-rational-growth}, the integer-order moment hierarchy induces growth constants \(\lambda_q\) through the zero structure of
\(
W_q(t)=\sum_{k\ge 1}F_{k+2}^{\,q}t^k
\).
Below we extend this ``moment pressure'' to all real parameters in \(q\), and establish its strict convexity, analyticity, and auditable expansions at the low- and high-temperature endpoints.

\begin{definition}[Real-parameter moment pressure and principal root]\label{def:pom-multiplicity-real-q-pressure}
For any real number \(q\ge 0\) define
\[
W_q(t):=\sum_{k\ge 1}F_{k+2}^{\,q}\,t^k,\qquad t\in(0,1).
\]
Let \(\rho(q)\in(0,1)\) be the unique positive root satisfying \(W_q(\rho(q))=1\), and define
\[
\boxed{\;\lambda(q):=\rho(q)^{-1}>1.\;}
\]
\end{definition}

\begin{theorem}[Uniqueness, analyticity, and strict convexity of the real-parameter moment pressure]\label{thm:pom-multiplicity-real-q-pressure}
Under the setting of Definition \ref{def:pom-multiplicity-real-q-pressure}:
\begin{enumerate}
  \item \textbf{(Uniqueness)} For each \(q\ge 0\), the equation \(W_q(t)=1\) has exactly one solution \(\rho(q)\) in \((0,1)\).
  \item \textbf{(Integer consistency)} When \(q\in\ZZ_{\ge 1}\), \(\lambda(q)\) coincides with \(\lambda_q\) from Proposition \ref{prop:pom-multiplicity-composition-moment-hierarchy-rational-growth}:
  \[
  \boxed{\;\lambda(q)=\lambda_q\qquad(q\in\ZZ_{\ge 1}).\;}
  \]
  \item \textbf{(Monotonicity and strict convexity)} The function \(q\mapsto \lambda(q)\) is strictly increasing on \([0,\infty)\), and
  \[
  \boxed{\;q\mapsto \log \lambda(q)\ \text{is strictly convex on }[0,\infty).\;}
  \]
  \item \textbf{(Analyticity and endpoints)} \(\lambda(q)\) is a real-analytic function on \([0,\infty)\), satisfying the endpoint values and limits
  \[
  \boxed{\;\lambda(0)=2,\qquad \lim_{q\to\infty}\frac{\lambda(q)}{2^{q}}=1.\;}
  \]
\end{enumerate}
\end{theorem}
\begin{proof}
\textbf{(1)} Fix \(q\ge 0\). By the positivity of coefficients \(F_{k+2}^{\,q}>0\), the map \(t\mapsto W_q(t)\) is continuous and strictly increasing on \((0,1)\), with \(W_q(t)\downarrow 0\) as \(t\downarrow 0\).
On the other hand, \(W_q(1^-)=\sum_{k\ge 1}F_{k+2}^{\,q}=+\infty\), so by the intermediate value theorem there exists a unique \(\rho(q)\in(0,1)\) with \(W_q(\rho(q))=1\).
\par
\textbf{(2)} For integer \(q\ge 1\), the generating function \(G_q(t)=\sum_{L\ge 0}Z_L^{(q)}t^L\) satisfies \(G_q(t)=1/(1-W_q(t))\) (see \eqref{eq:pom-multiplicity-composition-moment-hierarchy-gf}).
Since the coefficients are nonneg\-ative, Pringsheim's theorem asserts that the dominant singularity of \(G_q\) occurs on the positive real axis; moreover, the smallest positive root of the denominator is precisely the unique solution \(\rho(q)\) of \(W_q(t)=1\).
Therefore the growth constant \(\lambda_q=r_q^{-1}\) equals \(\lambda(q)=\rho(q)^{-1}\).
\par
\textbf{(3) Monotonicity:} If \(q_2>q_1\ge 0\), then for any \(t\in(0,1)\) we have \(W_{q_2}(t)>W_{q_1}(t)\).
By strict monotonicity of \(t\mapsto W_q(t)\), the solution of \(W_q(t)=1\) satisfies \(\rho(q_2)<\rho(q_1)\), hence \(\lambda(q_2)>\lambda(q_1)\).
\par
\textbf{(3) Strict convexity:} Take arbitrary \(q_1\ne q_2\) and \(\theta\in(0,1)\), let \(q=\theta q_1+(1-\theta)q_2\), and set
\(
t_i=\rho(q_i)
\),
\(
t=t_1^\theta t_2^{1-\theta}
\).
For each \(k\ge 1\),
\[
F_{k+2}^{\,q}t^{k}=\bigl(F_{k+2}^{\,q_1}t_1^k\bigr)^\theta\bigl(F_{k+2}^{\,q_2}t_2^k\bigr)^{1-\theta}.
\]
By H\"older's inequality,
\[
W_q(t)\le W_{q_1}(t_1)^\theta W_{q_2}(t_2)^{1-\theta}=1.
\]
Since \(t\mapsto W_q(t)\) is strictly increasing and \(W_q(\rho(q))=1\), it follows that \(\rho(q)\ge t\), i.e.,
\(
\log\rho(q)\ge \theta\log\rho(q_1)+(1-\theta)\log\rho(q_2)
\).
Hence \(q\mapsto -\log\rho(q)=\log\lambda(q)\) is convex.
Strictness follows from the strict case of H\"older's inequality: when \(q_1\ne q_2\), the sequences \(\{F_{k+2}^{q_1}t_1^k\}_k\) and \(\{F_{k+2}^{q_2}t_2^k\}_k\) cannot be proportional, so the inequality above is strict, yielding strict convexity.
\par
\textbf{(4) Analyticity:} Note that \(W_q(t)\) is term-wise real-analytic in \((q,t)\in[0,\infty)\times(0,1)\), with absolute uniform convergence on any compact set \(q\in[0,Q]\), \(t\in(0,\varphi^{-Q})\), so differentiation can be interchanged.
Moreover,
\(
\partial_t W_q(t)=\sum_{k\ge 1}kF_{k+2}^{\,q}t^{k-1}>0
\)
is strictly positive at \(t=\rho(q)\), so by the implicit function theorem, \(\rho(q)\) (and hence \(\lambda(q)=1/\rho(q)\)) is real-analytic on \([0,\infty)\).
\par
\textbf{(4) Endpoints:} When \(q=0\), \(W_0(t)=\sum_{k\ge 1}t^k=t/(1-t)\), so \(\rho(0)=1/2\) and \(\lambda(0)=2\).
Let \(x(q):=2^q\rho(q)=2^q/\lambda(q)\in(0,1]\); then from \(W_q(\rho(q))=1\),
\[
1=x(q)+\sum_{k\ge 2}\Bigl(\frac{F_{k+2}}{2^{k}}\Bigr)^{\!q}\,x(q)^k.
\]
For \(k\ge 2\), \(F_{k+2}<2^k\) (see the induction upper bound in Proposition \ref{prop:pom-path-component-multiplicity-refinement-monotone-extrema}), so the ratio inside the parentheses is strictly less than \(1\).
Therefore as \(q\to\infty\), the tail sum on the right tends to \(0\), hence \(x(q)\to 1\), i.e., \(\lambda(q)/2^q\to 1\). \qed
\end{proof}

\begin{proposition}[Sharp large-\texorpdfstring{$q$}{q} asymptotics]\label{prop:pom-multiplicity-lambdaq-large-q-asymptotic}
Let \(\lambda(q)\) be as in Theorem \ref{thm:pom-multiplicity-real-q-pressure}.
Then as \(q\to\infty\), the following rigorous asymptotic expansion holds:
\[
\boxed{\;
\lambda(q)=2^{q}\Bigl(1+\Bigl(\frac34\Bigr)^{q}+O\Bigl(\Bigl(\frac58\Bigr)^{q}\Bigr)\Bigr).\;}
\]
Equivalently,
\[
\boxed{\;
\log_2\lambda(q)=q+\frac{1}{\ln 2}\Bigl(\frac34\Bigr)^{q}+O\Bigl(\Bigl(\frac58\Bigr)^{q}\Bigr).\;}
\]
\end{proposition}
\begin{proof}
Adopting the notation \(x(q)=2^q/\lambda(q)\in(0,1]\) from the end of the proof of Theorem \ref{thm:pom-multiplicity-real-q-pressure},
\[
1=x(q)+\sum_{k\ge 2}r_k^{\,q}\,x(q)^k,\qquad
r_k:=\frac{F_{k+2}}{2^k}.
\]
Note that
\(
r_2=\frac34,\ r_3=\frac58
\), and for \(k\ge 4\), \(r_k\le \frac12\); therefore \(\sup_{k\ge 3}r_k=\frac58\).
Hence
\[
1-x(q)=\Bigl(\frac34\Bigr)^{q}x(q)^2+O\Bigl(\Bigl(\frac58\Bigr)^{q}\Bigr).
\]
From \(x(q)\in(0,1]\) we get \(1-x(q)=O((3/4)^q)\); substituting back yields
\(
1-x(q)=(3/4)^q+O((5/8)^q)
\).
Therefore
\(
\lambda(q)/2^q=1/x(q)=1+(3/4)^q+O((5/8)^q)
\), giving the first formula.
The second follows from \(\log_2(1+u)=u/\ln 2+O(u^2)\) with \(u=(3/4)^q+O((5/8)^q)\). \qed
\end{proof}

\begin{proposition}[Extreme low-temperature Gibbs limit: the \texorpdfstring{$(3/4)^q$}{(3/4)^q} sparse excitation law for part length distributions]\label{prop:pom-multiplicity-low-temperature-gibbs}
For any real number \(q\ge 0\) and \(L\ge 1\), define the \(q\)-tilted fiber-weight measure on \(\mathcal{C}_L\) by
\[
P^{(q)}_{L}(\boldsymbol{\ell})
:=\frac{w(\boldsymbol{\ell})^{q}}{Z_L^{(q)}},
\qquad
w(\boldsymbol{\ell})=\prod_{i}F_{\ell_i+2},
\qquad
Z_L^{(q)}=\sum_{\boldsymbol{\ell}\in\mathcal{C}_L}w(\boldsymbol{\ell})^{q}.
\]
Let \(\rho(q)=\lambda(q)^{-1}\).
Then for any fixed \(r\ge 1\), the random prefix \((\ell_1,\dots,\ell_r)\) converges in the \(L\to\infty\) limit to an i.i.d.\ process whose marginal distribution is
\begin{equation}\label{eq:pom-multiplicity-low-temp-iid-law}
\boxed{\;
\mathbb{P}_q(\ell=k)=F_{k+2}^{\,q}\,\rho(q)^{k}\qquad(k\ge 1),\;}
\end{equation}
satisfying \(\sum_{k\ge 1}\mathbb{P}_q(\ell=k)=1\).
Furthermore, as \(q\to\infty\), the excitation probabilities satisfy the asymptotics
\[
\boxed{\;
\mathbb{P}_q(\ell=1)=1-\Bigl(\frac34\Bigr)^{q}+O\Bigl(\Bigl(\frac58\Bigr)^{q}\Bigr),\qquad
\mathbb{P}_q(\ell=2)=\Bigl(\frac34\Bigr)^{q}+O\Bigl(\Bigl(\frac58\Bigr)^{q}\Bigr).\;}
\]
\end{proposition}
\begin{proof}
By the concatenation structure (following the same argument as in Proposition \ref{prop:pom-multiplicity-composition-partition-rational} and Proposition \ref{prop:pom-multiplicity-composition-moment-hierarchy-rational-growth}), for any fixed prefix \((k_1,\dots,k_r)\) (with \(s=k_1+\cdots+k_r\)) the exact identity holds:
\[
P^{(q)}_{L}(\ell_1=k_1,\dots,\ell_r=k_r)
=\frac{F_{k_1+2}^{\,q}\cdots F_{k_r+2}^{\,q}\,Z_{L-s}^{(q)}}{Z_{L}^{(q)}}.
\]
From \(G_q(t)=\sum_{L\ge 0}Z_L^{(q)}t^L=1/(1-W_q(t))\) and \(W_q(\rho(q))=1\) (Definition \ref{def:pom-multiplicity-real-q-pressure}), together with
\(
W_q'(\rho(q))>0
\)
implying that \(t=\rho(q)\) is a simple pole, the standard positive-coefficient pole transfer theorem gives
\(
Z_{L-s}^{(q)}/Z_L^{(q)}\to \rho(q)^{s}
\) (\(L\to\infty\), \(s\) fixed).
Substituting yields the joint distribution limit as \(\prod_i F_{k_i+2}^{\,q}\rho(q)^{k_i}\); since \(W_q(\rho(q))=1\) confirms this marginal is a probability distribution, we obtain \eqref{eq:pom-multiplicity-low-temp-iid-law} together with the i.i.d.\ property.
\par
Finally, from \(\mathbb{P}_q(\ell=1)=2^q\rho(q)=2^q/\lambda(q)\) and \(\mathbb{P}_q(\ell=2)=3^q\rho(q)^2\), combined with the asymptotic expansion of Proposition \ref{prop:pom-multiplicity-lambdaq-large-q-asymptotic}, the stated excitation law follows. \qed
\end{proof}

\begin{proposition}[Taylor constants at the high-temperature endpoint: second-order expansion of \texorpdfstring{$\log_2\lambda(q)$}{log2 lambda(q)} at \texorpdfstring{$q=0$}{q=0}]\label{prop:pom-multiplicity-lambdaq-taylor-q0}

S_1:=\sum_{k\ge 1}2^{-k}\log_2 F_{k+2},\qquad
S_2:=\sum_{k\ge 1}2^{-k}\bigl(\log_2 F_{k+2}\bigr)^2,\qquad
S_{1,1}:=\sum_{k\ge 1}k\,2^{-k}\log_2 F_{k+2}.

Then \(\log_2\lambda(q)\) admits a second-order expansion at \(q=0\):

\log_2\lambda(q)=1+\frac{S_1}{2}\,q+\frac12\,c_2\,q^2+O(q^3),\qquad(q\to 0),\;}
\]
where the second-order coefficient \(c_2\) has the fully closed form

=(\ln 2)\Bigl[\frac12\bigl(S_2-S_{1,1}S_1\bigr)+\frac34 S_1^2\Bigr].\;}
\]

\begin{proof}
From \(W_q(\rho(q))=1\) and \(W_0(t)=t/(1-t)\), we get \(\rho(0)=1/2\).
Performing second-order implicit differentiation of the identity \(W_q(\rho(q))\equiv 1\) at \(q=0\), using

\partial_t^2 W_0(1/2)=16

\begin{proposition}[Explicit ratio formula for the energy density: Gibbs representation of \texorpdfstring{$\frac{d}{dq}\log\lambda(q)$}{d/dq log lambda(q)}]\label{prop:pom-multiplicity-lambdaq-derivative-gibbs}
Under the setting of Definition \ref{def:pom-multiplicity-real-q-pressure}, let
\[
p_q(k):=F_{k+2}^{\,q}\,\rho(q)^k\qquad(k\ge 1),
\qquad
\sum_{k\ge 1}p_q(k)=1.
\]
Then the derivative exists and satisfies the closed form
\[
\boxed{\;
\frac{d}{dq}\log\lambda(q)
=\frac{\sum_{k\ge 1}p_q(k)\,\log F_{k+2}}{\sum_{k\ge 1}k\,p_q(k)}.\;}
\]
In particular,
\[
\lim_{q\to\infty}\frac{d}{dq}\log\lambda(q)=\log 2,
\qquad
\left.\frac{d}{dq}\log_2\lambda(q)\right|_{q=0}=\frac{S_1}{2}
\quad(\text{Proposition \ref{prop:pom-multiplicity-lambdaq-taylor-q0}}).
\]
\end{proposition}
\begin{proof}
From \(W_q(\rho(q))=1\) and implicit differentiation,
\[
0=\partial_q W_q(\rho)+\rho'(q)\,\partial_t W_q(\rho),
\qquad \rho=\rho(q).
\]
Note that
\[
\partial_q W_q(\rho)=\sum_{k\ge 1}F_{k+2}^{\,q}\rho^k\log F_{k+2}=\sum_{k\ge 1}p_q(k)\log F_{k+2},
\]
\[
\partial_t W_q(\rho)=\sum_{k\ge 1}kF_{k+2}^{\,q}\rho^{k-1}=\rho^{-1}\sum_{k\ge 1}k\,p_q(k).
\]
Therefore
\(
\rho'(q)=-\rho\,\frac{\sum p_q(k)\log F_{k+2}}{\sum k p_q(k)}
\).
From \(\log\lambda(q)=-\log\rho(q)\) we get
\(
\frac{d}{dq}\log\lambda(q)=-\rho'/\rho
\),
and rearranging gives the stated ratio formula.
The limit \(\lim_{q\to\infty}\frac{d}{dq}\log\lambda(q)=\log 2\) follows from Proposition \ref{prop:pom-multiplicity-lambdaq-large-q-asymptotic} (or directly from \(p_q(1)\to 1\)). \qed
\end{proof}
