\section{折叠多重度的渐近与"分辨率不足"}\label{app:fold-multiplicity}

平均折叠多重度
\[
d_m=\frac{|\Omega_m|}{|X_m|}=\frac{2^m}{F_{m+2}}
\]
满足
\[
d_m\sim C\left(\frac{2}{\varphi}\right)^m,\quad C=\frac{\sqrt5}{\varphi^2}.
\]
因此对固定窗口长度 $m$，观测将大量微态压缩为同一稳定类型；当 $m$ 增大时，稳定类型虽增多，但微态增长更快，从而"可区分细节"在有限分辨率下持续被折叠。这给出"规律不可读出可归因于分辨率不足"的数学版本：不可读出对应不同微态在折叠后不可区分，而稳定规律对应 $X_m$ 上的递归计数与不变分布仍可复核。

\subsection{折叠同余与有限分辨率半环}
为把折叠多重度与第 7 节的有限分辨率算术对接，这里给出折叠同余的代数化表述。

\begin{definition}[折叠同余与值模映射]
记 $\Omega_m:=\{0,1\}^m$，并定义折叠同余
\[
\omega\equiv_m\omega'\quad\Longleftrightarrow\quad\Fold_m(\omega)=\Fold_m(\omega').
\]
令
\[
N(\omega):=\sum_{k=1}^m \omega_k F_{k+1},
\qquad
\pi_m(\omega):=N(\omega)\bmod F_{m+2}.
\]
并记 $w(r):=\pi_m(Z(r))\in X_m\subset\Omega_m$（$0\le r\le F_{m+2}-1$）。
\end{definition}

\begin{theorem}[折叠同余的模半环表示]
\label{thm:fold-congruence-mod-semirings}
映射 $\pi_m$ 在 $\equiv_m$ 上良定义并诱导双射
\[
\overline\pi_m:\Omega_m/\!\equiv_m\;\longrightarrow\;\ZZ/(F_{m+2}\ZZ).
\]
进一步，若在 $\Omega_m/\!\equiv_m$ 上定义
\[
[\omega]\,\boxplus_m\,[\eta]:=[w(\overline\pi_m([\omega])\!+\!\overline\pi_m([\eta])\bmod F_{m+2})],
\]
\[
[\omega]\,\boxtimes_m\,[\eta]:=[w(\overline\pi_m([\omega])\,\overline\pi_m([\eta])\bmod F_{m+2})],
\]
则 $\overline\pi_m$ 给出半环同构
\[
(\Omega_m/\!\equiv_m,\boxplus_m,\boxtimes_m)\cong\bigl(\ZZ/(F_{m+2}\ZZ),+,\times\bigr).
\]
\end{theorem}

\textbf{证明：}
若 $\omega\equiv_m\omega'$，则 $\Fold_m(\omega)=\Fold_m(\omega')=:x\in X_m$。
由引理 \ref{lem:zeckendorf-range-bijection}，$V_m(x)$ 唯一确定模 $F_{m+2}$ 的剩余类，且
\(V_m(x)\equiv N(\omega)\equiv N(\omega')\pmod{F_{m+2}}\)。
故 $\pi_m$ 在 $\equiv_m$ 上良定义并诱导 $\overline\pi_m$。又对任意 $r$，取 $w(r)\in X_m\subset\Omega_m$，有
\(\overline\pi_m([w(r)])=r\)，故 $\overline\pi_m$ 为双射。由定义可见 $\overline\pi_m$ 保持加法与乘法，故为半环同构。证毕。

\subsection{多重度谱与方差能量恒等式}
将折叠多重度视为模环上的类函数，并给出其 $\ell^2$ 能量与频谱的精确对应。

\begin{definition}[多重度谱]
对 $r\in\ZZ/(F_{m+2}\ZZ)$ 定义多重度剖面
\[
d_m(r):=\bigl|\{\omega\in\Omega_m:\pi_m(\omega)=r\}\bigr|.
\]
记其离散 Fourier 变换为
\[
\widehat d_m(k):=\sum_{r\in\ZZ/(F_{m+2}\ZZ)} d_m(r)\,e^{-2\pi i kr/F_{m+2}},
\qquad k\in\ZZ/(F_{m+2}\ZZ).
\]
并记均值
\(
\overline d_m:=\frac{1}{F_{m+2}}\sum_r d_m(r)=\frac{2^m}{F_{m+2}}.
\)
\end{definition}

\begin{theorem}[多重度能量与方差的谱表示]
\label{thm:fold-multiplicity-energy}
设 $d_m(r)$ 如上，则有精确恒等式
\[
\sum_{r} d_m(r)^2=\frac{1}{F_{m+2}}\sum_{k}|\widehat d_m(k)|^2,
\]
并且方差满足
\[
\frac{1}{F_{m+2}}\sum_r\bigl(d_m(r)-\overline d_m\bigr)^2
=\frac{1}{F_{m+2}^2}\sum_{k\ne 0}|\widehat d_m(k)|^2.
\]
其中 $k=0$ 项给出 $\widehat d_m(0)=2^m$。
\end{theorem}

\textbf{证明：}
在有限交换群 $\ZZ/(F_{m+2}\ZZ)$ 上应用离散 Parseval 恒等式即得
\(
\sum_r d_m(r)^2=\frac{1}{F_{m+2}}\sum_k|\widehat d_m(k)|^2
\)。
又 $\widehat d_m(0)=\sum_r d_m(r)=|\Omega_m|=2^m$，代入
\(
\sum_r(d_m(r)-\overline d_m)^2=\sum_r d_m(r)^2-F_{m+2}\,\overline d_m^2
\)
并整理即可。证毕。

\begin{definition}[折叠碰撞概率]
对独立均匀的 $\omega,\eta\in\Omega_m$，定义
\[
\mathsf{Col}_m:=\mathbb{P}\bigl(\pi_m(\omega)=\pi_m(\eta)\bigr)
=\frac{1}{2^{2m}}\sum_r d_m(r)^2.
\]
称 $\mathsf{Col}_m-1/F_{m+2}$ 为折叠的均匀性缺口。
\end{definition}

\begin{theorem}[碰撞概率的谱恒等式]
\label{thm:fold-collision-spectrum}
折叠碰撞概率满足精确恒等式
\[
\mathsf{Col}_m=\frac{1}{F_{m+2}}+\frac{1}{2^{2m}F_{m+2}^2}\sum_{k\ne 0}|\widehat d_m(k)|^2.
\]
因此 $\mathsf{Col}_m\ge 1/F_{m+2}$，且等号当且仅当 $d_m(r)\equiv 2^m/F_{m+2}$ 为常数。
\end{theorem}

\textbf{证明：}
由定理 \ref{thm:fold-multiplicity-energy} 的方差恒等式，
\(
\sum_r d_m(r)^2=F_{m+2}\,\overline d_m^2+\frac{1}{F_{m+2}}\sum_{k\ne 0}|\widehat d_m(k)|^2
\)，
再代入 $\overline d_m=2^m/F_{m+2}$ 即得结论。

\begin{theorem}[多重度谱的乘法分解]
\label{thm:fold-spectrum-factorization}
对任意 $k\in\ZZ/(F_{m+2}\ZZ)$，有
\[
\widehat d_m(k)=\prod_{j=1}^m\Bigl(1+e^{-2\pi i k F_{j+1}/F_{m+2}}\Bigr).
\]
从而
\[
|\widehat d_m(k)|=2^m\prod_{j=1}^m\Bigl|\cos\Bigl(\pi k\,F_{j+1}/F_{m+2}\Bigr)\Bigr|.
\]
\end{theorem}

\textbf{证明：}
按定义
\(
\widehat d_m(k)=\sum_{\omega\in\Omega_m}e^{-2\pi i k\pi_m(\omega)/F_{m+2}}
\)。
又 $\pi_m(\omega)\equiv \sum_{j=1}^m \omega_j F_{j+1}\pmod{F_{m+2}}$，故
\[
\widehat d_m(k)=\sum_{\omega\in\Omega_m}\prod_{j=1}^m e^{-2\pi i k\omega_j F_{j+1}/F_{m+2}}
=\prod_{j=1}^m\bigl(1+e^{-2\pi i k F_{j+1}/F_{m+2}}\bigr).
\]
利用 $1+e^{-i\theta}=e^{-i\theta/2}\,2\cos(\theta/2)$ 得到模长公式。证毕。

\begin{corollary}[非平凡频谱的乘性上界]
\label{cor:fold-spectrum-nontrivial-bound}
对任意 $k\ne 0$，有
\[
|\widehat d_m(k)|\le 2^m\prod_{j=1}^m\Bigl|\cos\Bigl(\pi k\,F_{j+1}/F_{m+2}\Bigr)\Bigr|<2^m.
\]
因此碰撞缺口满足
\[
\mathsf{Col}_m-\frac{1}{F_{m+2}}=\frac{1}{2^{2m}F_{m+2}^2}\sum_{k\ne 0}|\widehat d_m(k)|^2
\le\frac{1}{F_{m+2}^2}\sum_{k\ne 0}\prod_{j=1}^m\cos^2\Bigl(\pi k\,F_{j+1}/F_{m+2}\Bigr).
\]
\end{corollary}

\begin{theorem}[奇偶性诱导的谱零点]
\label{thm:fold-spectrum-parity-zero}
若 $F_{m+2}$ 为偶数（等价于 $m+2\equiv 0\pmod 3$），则
\[
\widehat d_m\Bigl(\frac{F_{m+2}}{2}\Bigr)=0.
\]
换言之，频率 $k=F_{m+2}/2$ 为折叠多重度谱的一个严格零点。
\end{theorem}

\textbf{证明：}
由定理 \ref{thm:fold-spectrum-factorization}，
\[
\widehat d_m\Bigl(\frac{F_{m+2}}{2}\Bigr)=\prod_{j=1}^m\Bigl(1+e^{-\pi i F_{j+1}}\Bigr)
=\prod_{j=1}^m\bigl(1+(-1)^{F_{j+1}}\bigr).
\]
当 $m+2\equiv 0\pmod 3$ 时，$F_{m+2}$ 为偶数，$k=F_{m+2}/2$ 为整数。又 Fibonacci 数奇偶性以 3 为周期，存在 $j$ 使 $F_{j+1}$ 为奇数，故因子 $1+(-1)^{F_{j+1}}=0$，从而乘积为零。证毕。

\begin{theorem}[谱零点的同余判别]
\label{thm:fold-spectrum-zero-criterion}
对任意 $k\in\ZZ/(F_{m+2}\ZZ)$，有
\[
\widehat d_m(k)=0\quad\Longleftrightarrow\quad\exists\,j\in\{1,\dots,m\}\ \text{s.t.}\ 2kF_{j+1}\equiv F_{m+2}\pmod{2F_{m+2}}.
\]
等价地，存在 $j$ 使得
\(
e^{-2\pi i k F_{j+1}/F_{m+2}}=-1
\)。
\end{theorem}

\textbf{证明：}
由定理 \ref{thm:fold-spectrum-factorization}，
\(
\widehat d_m(k)=\prod_{j=1}^m(1+e^{-2\pi i k F_{j+1}/F_{m+2}})
\)。
乘积为零当且仅当存在某个 $j$ 使得因子为零，即
\(
e^{-2\pi i k F_{j+1}/F_{m+2}}=-1
\)。
将其改写为同余条件得到结论。证毕。

\begin{corollary}[零点类的存在与计数]
\label{cor:fold-spectrum-zero-count}
若 $m+2\equiv 0\pmod 3$，则谱零点集合至少包含
\[
\mathcal{Z}_m:=\left\{\frac{F_{m+2}}{2}\right\},
\]
并且对任意 $t\in\NN$，若存在 $j$ 使 $F_{j+1}\equiv 1\pmod{2t}$ 且 $2t\mid F_{m+2}$，则
\[
k=\frac{F_{m+2}}{2t}
\]
给出新的零点。特别地，零点数至少不小于满足 $2t\mid F_{m+2}$ 且某个 $F_{j+1}\equiv 1\pmod{2t}$ 的 $t$ 的个数。
\end{corollary}

\textbf{证明：}
由定理 \ref{thm:fold-spectrum-zero-criterion}，
\(
2kF_{j+1}\equiv F_{m+2}\pmod{2F_{m+2}}
\)
等价于 $e^{-2\pi i k F_{j+1}/F_{m+2}}=-1$。当 $2t\mid F_{m+2}$ 且 $F_{j+1}\equiv 1\pmod{2t}$ 时，取 $k=F_{m+2}/(2t)$ 得
\(
2kF_{j+1}=F_{m+2}\cdot F_{j+1}/t\equiv F_{m+2}\pmod{2F_{m+2}}
\)，
从而 $k$ 为零点。首项 $t=1$ 即给出 $k=F_{m+2}/2$。证毕。

\begin{theorem}[零点削减的碰撞上界]
\label{thm:fold-collision-zero-reduction}
设 $\mathcal{Z}_m:=\{k\in\ZZ/(F_{m+2}\ZZ):\widehat d_m(k)=0\}$，则
\[
\mathsf{Col}_m\le \frac{1}{F_{m+2}}+\frac{F_{m+2}-1-|\mathcal{Z}_m|}{F_{m+2}^2}.
\]
特别地，当 $|\mathcal{Z}_m|\ge 1$ 时，
\[
\mathsf{Col}_m\le \frac{1}{F_{m+2}}+\frac{F_{m+2}-2}{F_{m+2}^2}.
\]
\end{theorem}

\textbf{证明：}
由定理 \ref{thm:fold-collision-spectrum}，
\(
\mathsf{Col}_m-\frac{1}{F_{m+2}}=\frac{1}{2^{2m}F_{m+2}^2}\sum_{k\ne 0}|\widehat d_m(k)|^2
\)。
而 $|\widehat d_m(k)|\le\sum_r d_m(r)=2^m$，且对 $k\in\mathcal{Z}_m$ 取值为 0，故
\(
\sum_{k\ne 0}|\widehat d_m(k)|^2\le (F_{m+2}-1-|\mathcal{Z}_m|)\,2^{2m}
\)。
代入即得结论。证毕。

\begin{corollary}[整除结构给出的零点下界]
\label{cor:fold-spectrum-zero-divisor}
设 $d(n)$ 为正整数 $n$ 的除子个数。若 $m+2\equiv 0\pmod 3$，则
\[
|\mathcal{Z}_m|\ge d\Bigl(\frac{F_{m+2}}{2}\Bigr).
\]
从而
\[
\mathsf{Col}_m\le \frac{1}{F_{m+2}}+\frac{F_{m+2}-1-d(F_{m+2}/2)}{F_{m+2}^2}.
\]
\end{corollary}

\textbf{证明：}
当 $m+2\equiv 0\pmod 3$ 时，$F_{m+2}$ 为偶数。取任意 $t\mid F_{m+2}/2$，则 $2t\mid F_{m+2}$。
又 $F_2=1\equiv 1\pmod{2t}$，由推论 \ref{cor:fold-spectrum-zero-count} 可知
\(k=F_{m+2}/(2t)\) 为零点。不同 $t$ 给出不同的 $k$，故零点数至少为 $d(F_{m+2}/2)$。
再代入定理 \ref{thm:fold-collision-zero-reduction} 即得上界。证毕。

\subsection{二进制区间到 Zeckendorf 前缀的折叠：最大纤维与饱和判别}\label{app:fold-binary-prefix}

前文主要讨论 \(\Fold_m:\Omega_m=\{0,1\}^m\to X_m\)（Fibonacci 权重口径）诱导的折叠多重度。
为刻画“\(\,2^m\) 个二进制地址在 Zeckendorf（\(\varphi\)-基）分层中的投影压缩”，这里引入一个与 \(|\Omega_m|=2^m\) 同规模、但以\emph{整数区间}为输入的折叠投影。

\begin{definition}[二进制区间折叠 \(\mathrm{Fold}^{\mathrm{bin}}_m\)]\label{def:fold-bin}
设 \(Z:\ZZ_{\ge 0}\to\{0,1\}^{\NN}\) 为 Zeckendorf 编码：对每个 \(n\ge 0\)，存在唯一的合法位串
\[
Z(n)=(z_1,z_2,\dots),\qquad z_k\in\{0,1\},\quad z_k z_{k+1}=0,
\]
使得
\[
n=\sum_{k\ge 1} z_k F_{k+1}.
\]
记截断算子 \(\pi_m(z_1,z_2,\dots):=(z_1,\dots,z_m)\in X_m\)。定义
\[
\mathrm{Fold}^{\mathrm{bin}}_m:\{0,1,\dots,2^m-1\}\to X_m,\qquad
\mathrm{Fold}^{\mathrm{bin}}_m(n):=\pi_m(Z(n)).
\]
\end{definition}

\begin{definition}[\(K(m)\) 与 Zeckendorf 值]\label{def:K-of-m}
令 \(K(m)\) 为满足
\begin{equation}\label{eq:K-of-m}
F_{K(m)+1}\le 2^m-1 < F_{K(m)+2}
\end{equation}
的唯一整数。对 \(w=(w_1,\dots,w_m)\in X_m\) 定义其 Zeckendorf 值
\[
V_m(w):=\sum_{k=1}^m w_k F_{k+1}.
\]
\end{definition}

\begin{proposition}[纤维的尾部表示]\label{prop:fold-bin-fiber-tail}
设 \(m\ge 1\)，\(w\in X_m\)，并令 \(K=K(m)\)。记尾部长度 \(L:=K-m\ge 0\)。
则 \(\mathrm{Fold}^{\mathrm{bin}}_m{}^{-1}(w)\) 与满足下列约束的尾部位串 \((c_{m+1},\dots,c_K)\in\{0,1\}^{L}\) 一一对应：
\[
n
=V_m(w)+\sum_{k=m+1}^{K} c_k F_{k+1},
\]
其中
\begin{enumerate}
  \item \textbf{相邻约束：}\quad \(c_k c_{k+1}=0\)（对 \(m+1\le k\le K-1\)）；
  \item \textbf{边界约束：}\quad \(w_m\,c_{m+1}=0\)；
  \item \textbf{阈值约束：}\quad \(V_m(w)+\sum_{k=m+1}^{K} c_k F_{k+1}\le 2^m-1\)。
\end{enumerate}
从而纤维多重度
\(
d^{\mathrm{bin}}_m(w):=\bigl|\mathrm{Fold}^{\mathrm{bin}}_m{}^{-1}(w)\bigr|
\)
只依赖于 \((V_m(w),w_m)\)。
\end{proposition}

\begin{corollary}[最大纤维、组合上界与饱和判别]\label{cor:fold-bin-saturation}
对任意 \(m\ge 1\)，最大纤维由 \(w=0^m\) 取得，并满足组合上界
\begin{equation}\label{eq:fold-bin-upper}
d^{\mathrm{bin}}_{\max}(m):=d^{\mathrm{bin}}_m(0^m)\le |X_{K(m)-m}|=F_{K(m)-m+2}.
\end{equation}
记 \(K=K(m)\)、\(L=K-m\)，并定义在相邻约束下的\emph{最大尾和}
\[
S_{\max}(m):=\max\Bigl\{\sum_{k=m+1}^{K} c_k F_{k+1}:\ (c_{m+1},\dots,c_K)\in\{0,1\}^{L},\ c_k c_{k+1}=0\Bigr\}.
\]
则对 \(w=0^m\)，上界 \eqref{eq:fold-bin-upper} 饱和当且仅当阈值约束对所有合法尾部都是冗余的，即
\begin{equation}\label{eq:fold-bin-sat-iff}
d^{\mathrm{bin}}_{\max}(m)=F_{K(m)-m+2}\quad\Longleftrightarrow\quad 2^m-1\ge S_{\max}(m).
\end{equation}
并且 \(S_{\max}(m)\) 具有完全显式的闭式（由“1010\(\cdots\)”极大尾部取得）：
\begin{equation}\label{eq:fold-bin-Smax-closed}
S_{\max}(m)=
\begin{cases}
F_{K(m)+2}-F_{m+1}, & L=K(m)-m\ \text{为奇数},\\
F_{K(m)+2}-F_{m+2}, & L=K(m)-m\ \text{为偶数}.
\end{cases}
\end{equation}
等价地，饱和判别可改写为
\[
F_{K(m)+2}-2^m\le
\begin{cases}
F_{m+1}-1,& L\ \text{奇},\\
F_{m+2}-1,& L\ \text{偶}.
\end{cases}
\]
\end{corollary}

\begin{remark}[指数级 Diophantine 近似与“仅有限次饱和”]\label{rem:fold-bin-finite}
由 \eqref{eq:fold-bin-Smax-closed} 可见，饱和条件强迫 \(2^m\) 与某个 Fibonacci 数 \(F_{K(m)+2}\) \emph{异常接近}。
用 Binet 公式 \(F_n=\varphi^n/\sqrt5+O(\varphi^{-n})\) 将其改写，可导出一个指数级小的线性对数形式约束：
\[
\Bigl|(K(m)+2)\log\varphi - m\log 2 - \log\sqrt5\Bigr|
\ \lesssim\ \Bigl(\frac{\varphi}{2}\Bigr)^m.
\]
由于 \(\varphi\) 为代数数而 \(2\) 为有理数，上式左侧是非零的对数线性形式；而 Baker 型下界（例如 Matveev 的显式形式）给出其\emph{至多多项式级}的小下界（见 \cite{Matveev2000LinearFormsLogs,BakerWustholz2007LogarithmicForms}），从而与右侧的指数级小量矛盾，推出：
\[
\text{上界饱和只能发生有限次。}
\]
本结论在本文的可复现管线中可直接数值审计（见表 \ref{tab:fold_binary_saturation_points}）。
\end{remark}

\begin{proposition}[计算 \(\boldsymbol{d^{\mathrm{bin}}_m(w)}\) 的 \(O(m)\) Fibonacci-digit DP]\label{prop:fold-bin-digit-dp}
给定 \(m\ge 1\) 与 \(w\in X_m\)。令 \(K=K(m)\) 并记 \(V=V_m(w)\)，再令 \(\mathrm{slack}:=2^m-1-V\)。
将 \(\mathrm{slack}\) 写为 Zeckendorf 位串 \(Z(\mathrm{slack})=(b_1,\dots,b_K)\)（合法，无相邻 \(11\)）。
则 \(d^{\mathrm{bin}}_m(w)\) 可由如下 digit DP 在 \(O(K-m)=O(m)\) 时间内精确计算：
\[
\mathrm{DP}(k,\mathrm{prev},\mathrm{tight})
\quad(k=K,K-1,\dots,m+1),
\]
其中 \(\mathrm{prev}\in\{0,1\}\) 记录上一个（更高位）是否取 \(1\)，\(\mathrm{tight}\in\{0,1\}\) 表示当前前缀是否仍与 \((b_k,\dots,b_K)\) 相同；
转移时允许选择 \(c_k\in\{0,1\}\)，满足
\[
c_k c_{k+1}=0,\qquad
\mathrm{tight}=1\Rightarrow c_k\le b_k,\qquad
w_m=1\Rightarrow c_{m+1}=0,
\]
并按是否保持等号更新 \(\mathrm{tight}\)。该 DP 的输出即为满足 \(\sum_{k=m+1}^K c_k F_{k+1}\le \mathrm{slack}\) 的尾部数目，从而等于 \(d^{\mathrm{bin}}_m(w)\)。
\end{proposition}
