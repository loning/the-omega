\paragraph{Ap\'ery Reduction of the Tail-Additive Closure: Two-Generator Normal Form, Congruence Truncation Ledger, and Reset Clock Moment Invariants}
This paragraph, building on the tail-additive mechanism of Theorem \ref{thm:xi-window6-tail-semigroup-frobenius-threshold}, further reduces reachability from a "generating set" to a family of congruence thresholds, and provides the equivalent normal form, a finite truncation partition of the unreachable set, and moment invariants of the reset interval clock.
\par

\begin{theorem}[Two-Generator Reduction and Canonical Solution of the Tail-Additive Mechanism]\label{thm:xi-window6-tail-semigroup-two-generator-normal-form}
Under the tail offset additive closure framework of the window-\(6\) uplift, let the additive generating set be \(\{21,34,55\}\).
Since \(55=21+34\), the corresponding reachable semigroup satisfies
\[
S_6=\langle 21,34,55\rangle=\langle 21,34\rangle\subset\mathbb N.
\]
For any \(n\in\mathbb N\), let \(r\equiv n\ (\mathrm{mod}\ 21)\).
Define
\[
b_r:=13r\ (\mathrm{mod}\ 21)\in\{0,1,\dots,20\},
\qquad
w_r:=34\,b_r,
\]
then \(n\in\langle 21,34\rangle\) if and only if \(n\ge w_r\).
Moreover, in the reachable case, there exists a unique pair
\(
(a,b)\in\mathbb N\times\{0,1,\dots,20\}
\)
satisfying
\[
n=21a+34b,\qquad b=b_r,\qquad a=\frac{n-w_r}{21}.
\]
Therefore, reachability admits a constant-time criterion on each congruence class \(r\), and induces a unique "mod \(21\) normal form" (reducing \(b\) to \(\{0,\dots,20\}\)).
\end{theorem}

\begin{proof}
The identity \(55=21+34\) implies \(\langle 21,34,55\rangle=\langle 21,34\rangle\).
Since \(34\equiv 13\ (\mathrm{mod}\ 21)\) and \(13^2\equiv 1\ (\mathrm{mod}\ 21)\), \(13\) is the modular inverse of \(34\) modulo \(21\), so
\(
34\,b_r\equiv r\ (\mathrm{mod}\ 21)
\)
holds, and \(b_r\) is unique within \(\{0,\dots,20\}\).
\par
If \(n=21a+34b\) is reachable, then necessarily \(b\equiv b_r\ (\mathrm{mod}\ 21)\), so \(b=b_r+21k\) (\(k\in\mathbb N\)), which gives
\(
n=w_r+21(a+34k)\ge w_r
\).
Conversely, if \(n\equiv r\ (\mathrm{mod}\ 21)\) and \(n\ge w_r\), then \(n-w_r\) is a nonnegative multiple of \(21\); setting \(a=(n-w_r)/21\ge 0\) and \(b=b_r\) yields the decomposition.
Uniqueness is forced by the reduction condition \(b\in\{0,\dots,20\}\): if \(21a+34b=21a'+34b'\) with \(b,b'\in\{0,\dots,20\}\), then \(34(b-b')\) is a multiple of \(21\), and since \(\gcd(34,21)=1\), we get \(b=b'\), hence \(a=a'\).
\end{proof}

\begin{proposition}[Complete Congruence-Union Characterization of the Reachable Dimension Set with Finite Truncation]\label{prop:xi-window6-dimension-reachability-congruence-union}
Write the total dimension as \(D=12+n\) (where \(n\) is the tail-additive increment).
Then under the framework of Theorem \ref{thm:xi-window6-tail-semigroup-two-generator-normal-form}:
\begin{enumerate}
  \item \(D\) is reachable if and only if there exists \(r\in\{0,\dots,20\}\) such that \(n\equiv r\ (\mathrm{mod}\ 21)\) and \(n\in w_r+21\mathbb N\);
  equivalently,
  \[
  D\in\bigcup_{r=0}^{20}\bigl(12+w_r+21\mathbb N\bigr).
  \]
  \item \(D\) is unreachable if and only if there exists \(r\) such that \(n=r+21k\) with \(0\le k<g_r\);
  therefore the unreachable set is a finite truncated union of \(21\) arithmetic progressions, with truncation lengths completely given by \(\{g_r\}\).
\end{enumerate}
\end{proposition}

\begin{proposition}[Self-Duality of the Congruence Permutation: \texorpdfstring{$r\mapsto b_r$}{r->br} Is an Involution]\label{prop:xi-window6-congruence-involution}
The map \(r\mapsto b_r=13r\ (\mathrm{mod}\ 21)\) satisfies \(13^2\equiv 1\ (\mathrm{mod}\ 21)\), and hence is a self-inverse permutation:
\[
b_{b_r}=r.
\]
Therefore the Ap\'ery threshold family satisfies the exchange law
\[
w_{b_r}=34\,b_{b_r}=34\,r.
\]
\end{proposition}
