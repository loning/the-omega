
\section{Audit closure checklist}
\label{app:audit_checklist}

\subsection{Closed layer (self-contained within this paper)}
\label{app:closed_layer_list}

\begin{itemize}
\item \textbf{Protocol objects and scan orbits.} Definition~\ref{def:protocol} specifies Kronecker scan protocols.
\item \textbf{Finite-horizon readout.} Definition~\ref{def:readout} defines empirical averages as the sole protocol output.
\item \textbf{Period interface.} Definition~\ref{def:period_datum} defines period data and ideal values; Definition~\ref{def:hsp} defines the scan-to-period mapping.
\item \textbf{Wish.} Definition~\ref{def:wish} defines Wish as protocol-stable period data with auditable error budgets.
\item \textbf{Realization.} Theorem~\ref{thm:period_realization} states scan realization of periods under rational independence.
\item \textbf{Error certificates.} Equation~\ref{eq:error_decomposition} decomposes total error; Definitions~\ref{def:reg_certificate} and \ref{def:admissible_regularization} formalize auditable regularization certificates; Definition~\ref{def:audit_compatible_protocol} packages the finite-horizon audit-compatible conditions; Theorem~\ref{thm:koksma_hlawka} yields discrepancy--variation bounds; Definitions~\ref{def:star_discrepancy} and \ref{def:stabN} provide certificate objects.
\item \textbf{Explicit discrepancy certificates.} Equation~\ref{eq:golden_discrepancy_bound} gives the one-dimensional golden-branch bound; Theorem~\ref{thm:etk} provides the multi-dimensional ETK audit form.
\item \textbf{Certified dynamics.} Theorem~\ref{thm:teleology_dynamics} and Proposition~\ref{prop:energy_decay} derive and analyze the dissipative variational dynamics on protocol space, with the nonsmooth certificate extension in Proposition~\ref{prop:nonsmooth_energy}.
\end{itemize}

\subsection{Programmatic layer (falsifiable proposals, not used as theorem inputs)}
\label{app:programmatic_layer_list}

\begin{itemize}
\item \textbf{Selection functional.} Definition~\ref{def:certified_selection_functional} specifies an auditable cost $J_N(\mathcal{P};W_0)$; Conjecture~\ref{conj:selection} posits a falsifiable near-minimality behavior under a fixed audit convention.
\item \textbf{Constants as near-minimizers.} The mapping from physical constants to minimizers of $J_N$ is a model hypothesis.
\item \textbf{Teleology and observer language.} Any psychological or agency interpretation of kernel choice remains interpretation-layer only.
\end{itemize}
