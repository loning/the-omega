\section{Multiplicative phase \texorpdfstring{$\theta_\times$}{theta\_times} and the holographic polar embedding \texorpdfstring{$\mathcal{Z}$}{Z}}
\label{sec:embedding}

\subsection{Definition of multiplicative phase}
Fix a phase weight $\beta:\PP\to\RR$. Define the multiplicative phase (valued in $\RR/2\pi\ZZ$) by
$$
\theta_\times(n)\equiv \sum_{p\in\PP} a_p(n)\,\beta(p)\pmod{2\pi}.
$$

\begin{proposition}[Phase additivity under multiplication]
\label{prop:phase_hom}
For all $m,n\in M$,
$$
\theta_\times(mn)\equiv\theta_\times(m)+\theta_\times(n)\pmod{2\pi}.
$$
\end{proposition}

\begin{proof}
This follows from $a_p(mn)=a_p(m)+a_p(n)$ and additivity of the sum, exactly as in Proposition~\ref{prop:radial_hom}.
\end{proof}

A minimal parametrized choice is $\beta(p)=\beta_0\log p$ for some real constant $\beta_0$, in which case
$$
\theta_\times(n)\equiv \beta_0\log n\pmod{2\pi}.
$$

\subsection{Holographic polar embedding}
\begin{definition}[HPA embedding]
\label{def:Z_embedding}
Define a map $\mathcal{Z}:M\to\CC^*$ by
$$
\mathcal{Z}(n):=\rho(n)\,\e^{\iu\theta_\times(n)}.
$$
\end{definition}

Under the minimal choices $\rho(n)=n$ and $\theta_\times(n)\equiv\beta_0\log n\ (\mathrm{mod}\ 2\pi)$,
$$
\mathcal{Z}(n)=n\,\e^{\iu\beta_0\log n}=n^{1+\iu\beta_0},
$$
so $\mathcal{Z}(M)$ lies on a family of logarithmic spirals in the complex plane.

\begin{theorem}[Multiplicative homomorphism of the embedding]
\label{thm:Z_mult}
For all $m,n\in M$,
$$
\mathcal{Z}(mn)=\mathcal{Z}(m)\mathcal{Z}(n).
$$
\end{theorem}

\begin{proof}
By Proposition~\ref{prop:radial_hom} and Proposition~\ref{prop:phase_hom},
$$
\mathcal{Z}(mn)=\rho(mn)\e^{\iu\theta_\times(mn)}=\rho(m)\rho(n)\e^{\iu(\theta_\times(m)+\theta_\times(n))}=\mathcal{Z}(m)\mathcal{Z}(n).
$$
\end{proof}

Theorem~\ref{thm:Z_mult} is the first structural pillar: multiplicativity is decomposed into radial scaling and angular additivity.

\subsection{Octonionic prime fields: a non-associative lift of phase}
The complex embedding $\mathcal{Z}:M\to\CC^*$ resolves multiplicative structure into a radial coordinate and an $S^1$-phase.
To model an internal ``phase'' with higher-dimensional non-abelian symmetry, one can replace $S^1$ by the unit sphere $S^7$ of the octonions.

\subsubsection{Octonions, conjugation, and the unit sphere}
\begin{definition}[Octonions, conjugation, and norm]
\label{def:octonions}
Let $\OO$ denote the real octonion algebra \cite{Baez2002}.
Write an octonion in the standard basis as
$
x=x_0+\sum_{i=1}^7 x_i e_i
$
with $x_i\in\RR$.
Define octonionic conjugation by
$
\bar{x}:=x_0-\sum_{i=1}^7 x_i e_i
$
and the norm by
$
N(x):=x\bar{x}=\bar{x}x\in\RR_{\ge 0}.
$
Define $\|x\|:=\sqrt{N(x)}$ and the unit sphere
$
S^7:=\{u\in\OO:\ N(u)=1\}.
$
\end{definition}

\begin{proposition}[Composition]
\label{prop:octonion_composition}
For all $x,y\in\OO$ one has
$$
N(xy)=N(x)N(y).
$$
See, for example, \cite{Baez2002}.
\end{proposition}

\subsubsection{A discrete lattice: integral octonions and prime shells}
\begin{definition}[Hurwitz/Octavian integer lattice]
\label{def:ohur}
Let $E_8\subset\RR^8$ be the root lattice
$$
E_8:=\Big\{v\in\ZZ^8:\ \sum_{i=0}^7 v_i\equiv 0\ (\mathrm{mod}\ 2)\Big\}\ \cup\
\Big\{v\in\big(\ZZ+\tfrac12\big)^8:\ \sum_{i=0}^7 v_i\equiv 0\ (\mathrm{mod}\ 2)\Big\}.
$$
Let $\OHur\subset\OO$ be a maximal order (integral octonions). As a Euclidean lattice with respect to the octonion norm $N$, $\OHur$ is isometric to the scaled lattice $\frac{1}{\sqrt{2}}E_8$ (equivalently, $\sqrt{2}\,\OHur$ is an $E_8$ root lattice). This $\OHur$ is often called the octavian integers; see \cite{ConwaySmith2003}.
\end{definition}

\begin{definition}[Prime shells]
\label{def:prime_shells}
For a rational prime $p\in\PP$, define the norm-$p$ shell
$$
\OO_p:=\{\pi\in\OHur:\ N(\pi)=p\}.
$$
\end{definition}

\begin{proposition}[Prime-shell count]
\label{prop:prime_shell_count}
For every rational prime $p$ one has
$$
|\OO_p| = 240(1+p^3).
$$
See, for example, \cite{ConwaySmith2003}.
\end{proposition}

\begin{remark}[Gauge freedom and G2 symmetry]
The automorphism group $\mathrm{Aut}(\OO)\cong\Gtwo$ preserves multiplication and the norm. Hence it acts on each shell $\OO_p$.
Consequently, specifying a distinguished representative in $\OO_p$ is a choice of additional structure (a ``gauge''); any candidate observable should be formulated in a $\Gtwo$-covariant or $\Gtwo$-invariant way.
\end{remark}

\begin{remark}[Series impedance postulate (coupling viewpoint)]
The multiplicity of internal channels and the presence of natural submanifolds (bulk/boundary/path) motivate additive ``impedance'' postulates for effective couplings in an interpretation layer.
For example, in the octonionic setting one may postulate that inverse couplings add as series impedances of geometric channels; see Assumption~\ref{assump:alpha_geo_impedance}.
\end{remark}

\subsubsection{A lift map and an octonionic embedding}
\begin{definition}[Octonionic prime lift]
\label{def:lift_map}
An \emph{octonionic lift} is a map $\mathcal{L}:\PP\to\OHur$ such that $\mathcal{L}(p)\in\OO_p$ for all primes $p$.
\end{definition}

\begin{definition}[Canonical left-normed prime product]
\label{def:Pi_L}
Let $n\in\NN_{>0}$ and write its prime factor multiset as an ordered list with multiplicity
$$
n=p_1p_2\cdots p_k,\qquad p_1\le p_2\le\cdots\le p_k.
$$
Given a lift $\mathcal{L}$, define the \emph{canonical product}
$$
\Pi_{\mathcal{L}}(n):=\big(\cdots((\mathcal{L}(p_1)\mathcal{L}(p_2))\mathcal{L}(p_3))\cdots \big)\mathcal{L}(p_k)\ \in\ \OO,
$$
with left-normed bracketing (and set $\Pi_{\mathcal{L}}(1):=1$).
Define the associated unit direction
$$
U_{\mathcal{L}}(n):=\frac{\Pi_{\mathcal{L}}(n)}{\sqrt{n}}\ \in\ S^7.
$$
\end{definition}

\begin{proposition}[Norm consistency]
\label{prop:norm_consistency}
For all $n\in\NN_{>0}$ one has $N(\Pi_{\mathcal{L}}(n))=n$ and hence $U_{\mathcal{L}}(n)\in S^7$.
\end{proposition}

\begin{proof}
By Proposition~\ref{prop:octonion_composition} and $N(\mathcal{L}(p_i))=p_i$,
$$
N(\Pi_{\mathcal{L}}(n)) = \prod_{i=1}^k N(\mathcal{L}(p_i)) = \prod_{i=1}^k p_i = n.
$$
\end{proof}

\begin{definition}[Octonionic HPA embedding]
\label{def:Z_O_embedding}
Define $\mathcal{Z}_{\OO}:M\to\OO\setminus\{0\}$ by
$$
\mathcal{Z}_{\OO}(n):=\rho(n)\,U_{\mathcal{L}}(n).
$$
\end{definition}

\begin{remark}[Associative reductions]
If the image of $\mathcal{L}$ is contained in a fixed associative subalgebra of $\OO$ (e.g.\ a copy of $\CC$ or $\mathbb{H}$), then the dependence on bracketing disappears and $\mathcal{Z}_{\OO}$ reduces to an ordinary polar embedding in that subalgebra.
\end{remark}

\subsubsection{Associator defects and a concrete example}
\begin{definition}[Associator]
\label{def:associator}
For $x,y,z\in\OO$, define the associator
$$
\mathcal{A}(x,y,z):=(xy)z-x(yz).
$$
\end{definition}

\begin{proposition}[Alternation and associative subalgebras]
\label{prop:associator_properties}
The associator $\mathcal{A}$ is alternating in $x,y,z$.
Moreover, if $x,y,z$ lie in a common associative subalgebra of $\OO$ (in particular, in a quaternionic subalgebra), then $\mathcal{A}(x,y,z)=0$.
See, for example, \cite{Baez2002}.
\end{proposition}

\begin{remark}[A nonzero associator in a fixed Fano convention]
\label{ex:associator_example}
Fix a basis $(1,e_1,\dots,e_7)$. Encode multiplication by the oriented Fano cycles
$$
\begin{aligned}
&(1,2,3),\ (1,4,5),\ (1,7,6),\ (2,4,6),\\
&(2,5,7),\ (3,4,7),\ (3,6,5).
\end{aligned}
$$
meaning $e_ie_j=e_k$ when $(i,j,k)$ is cyclic in one of these triples and $e_je_i=-e_k$.
Then
$$
(e_1e_2)e_4=e_7,\qquad e_1(e_2e_4)=-e_7,
$$
so
$$
\mathcal{A}(e_1,e_2,e_4)=2e_7.
$$
\end{remark}
