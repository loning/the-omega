\section{Free energy and limit laws}\label{sec:free}

\subsection{Free energy and the dominant spectral branch}

For $y>0$, define the (per-step) free energy, whenever the limit exists,
\begin{equation}\label{eq:free-energy}
f(y)=\lim_{m\to\infty}\frac{1}{m}\log Z_m(y).
\end{equation}
By the order-$4$ recurrence \eqref{eq:recurrence}, we have a spectral decomposition of the form
\[
Z_m(y)=\sum_{i=1}^4 a_i(y)\lambda_i(y)^m,
\]
where $\lambda_i(y)$ are the roots of $\Pi(\lambda,y)=0$ and $a_i(y)$ are determined by the initial conditions.
When a unique dominant root (in modulus) exists and its coefficient does not vanish, it controls the exponential growth rate.

\begin{lemma}[Dominant-root separation near $y=1$]\label{lem:dominant-root-separation}
At $y=1$, the roots of $\Pi(\lambda,y)$ are $2,1,-1,-1$.
Hence exactly one root has modulus strictly larger than $\frac{3}{2}$.
There exists a neighborhood $U$ of $1$ such that for every $y\in U$:
\begin{enumerate}
  \item there is exactly one root $\lambda_+(y)$ of $\Pi(\lambda,y)$ with $|\lambda_+(y)|>\frac32$;
  \item the other three roots satisfy $|\lambda_i(y)|<\frac32$;
  \item $\lambda_+(y)$ is analytic in $y$ with $\lambda_+(1)=2$.
\end{enumerate}
\end{lemma}

\begin{proof}
At $(\lambda,y)=(2,1)$, $\partial_\lambda\Pi(2,1)=9\neq0$, so the implicit function theorem gives a unique analytic branch $\lambda_+(y)$ with $\lambda_+(1)=2$.
The remaining roots depend continuously on $y$ as well.
Since at $y=1$ all non-dominant roots satisfy $|\lambda|<\frac32$ and the root $\lambda_+(1)=2$ satisfies $|\lambda_+(1)|>\frac32$, continuity of roots implies a preserved modulus gap for $y$ close to $1$.
\end{proof}

\begin{lemma}[Dominant coefficient is nonzero]\label{lem:dominant-coeff}
For $y$ in a sufficiently small punctured neighborhood $U\setminus\Branch$ around $1$, let
\[
Z_m(y)=\sum_{i=1}^4 a_i(y)\lambda_i(y)^m,\qquad \lambda_+(y):=\lambda_1(y),
\]
be the spectral decomposition with $\lambda_i(y)$ the four roots of $\Pi(\lambda,y)$.
Then $a_+(y):=a_1(y)$ is nonzero on that neighborhood and $a_+(1)=1$.

\begin{proof}
For $y\in U\setminus\Branch$, the roots are simple, so $(a_1,\dots,a_4)$ solves a Vandermonde linear system from
\[
\bigl(Z_0(y),\dots,Z_3(y)\bigr)^T=V(y)\,(a_1(y),\dots,a_4(y))^T,\quad
V_{kj}(y)=\lambda_j(y)^k,
\]
hence each $a_i(y)$ is a rational function of $\lambda_j(y)$ with denominator $\det V(y)\neq0$.
Thus each $a_i(y)$ is analytic on $U\setminus\Branch$.

At $y=1$, the roots are $2,1,-1,-1$ and the explicit specialization $Z_m(1)=2^m$ forces the expansion
\[
2^m=\sum_{i=1}^4 a_i(1)\lambda_i(1)^m=a_1(1)2^m+(a_3(1)+a_4(1)m)(-1)^m+a_2(1),
\]
so $a_1(1)=1$.
By continuity, $a_+(y)$ remains nonzero for $y$ close to $1$ outside $\Branch$.
\end{proof}
\end{lemma}

In the following theorem, we write $\lambda_+(y)$ for the branch from
Lemma~\ref{lem:dominant-root-separation} and $a_+(y)$ for its coefficient in the spectral expansion.
\begin{theorem}[Dominant spectral branch]\label{thm:lambda-plus}
There exists an open neighborhood $U\subset\CC$ of $y=1$ and a function $\lambda_+(y)$
analytic on $U$ such that
\[
\Pi(\lambda_+(y),y)=0,\qquad \lambda_+(1)=2.
\]
Moreover, for all $y\in U\cap(0,\infty)\setminus\Branch$, the free energy exists and satisfies
\[
f(y)=\log \lambda_+(y).
\]
\end{theorem}

\begin{proof}
By Lemma~\ref{lem:dominant-root-separation}, shrink $U$ so that for every
$y\in U\setminus\Branch$ there is a unique root $\lambda_+(y)$ with strict
modulus dominance over the three other roots.
Lemma~\ref{lem:dominant-coeff} gives $a_+(1)=1$ and $a_+(y)\neq0$ near $1$.

For such $y$, let $\lambda_2(y),\lambda_3(y),\lambda_4(y)$ denote the other
three roots and write the spectral decomposition
\[
Z_m(y)=a_+(y)\lambda_+(y)^m+\sum_{i=2}^4 a_i(y)\lambda_i(y)^m.
\]
Define
\[
r(y):=\max_{i=2,3,4}\left|\frac{\lambda_i(y)}{\lambda_+(y)}\right|<1.
\]
Then
\[
Z_m(y)=a_+(y)\lambda_+(y)^m\!\left(1+\sum_{i=2}^4 \frac{a_i(y)}{a_+(y)}\left(\frac{\lambda_i(y)}{\lambda_+(y)}\right)^m\right),
\]
so
\[
\frac{1}{m}\log Z_m(y)=\log\lambda_+(y)+\frac{1}{m}\log\!\left(1+O(r(y)^m)\right)\to\log\lambda_+(y).
\]
The free energy formula follows.
\end{proof}

\subsection{An exponential family and the limiting log-mgf}

The probabilistic statements below use Theorem~\ref{thm:coeff-nonneg}, which gives nonnegative coefficients $a_{m,k}\ge0$, so $Z_m(e^s)/Z_m(1)$ is a true moment generating function.
Let $H_m$ be a random variable taking values in $\{0,1,\dots,\lfloor (m+3)/2\rfloor\}$ whose moment generating function is defined by
\begin{equation}\label{eq:mgf-def}
\mathbb{E}\bigl[e^{sH_m}\bigr]=\frac{Z_m(e^s)}{Z_m(1)}.
\end{equation}
Since $Z_m(1)=2^m$ and $Z_m(y)=\sum_{k=0}^{\lfloor (m+3)/2\rfloor} a_{m,k}y^k$ with $a_{m,k}\ge 0$ (Theorem~\ref{thm:coeff-nonneg} and Appendix~\ref{app:degeneracies}), we have
\[
\frac{Z_m(e^s)}{Z_m(1)}=\sum_{k=0}^{\lfloor (m+3)/2\rfloor} \frac{a_{m,k}}{2^m}\,e^{sk},
\]
and the weights $p_{m,k}:=a_{m,k}/2^m$ form a probability distribution on $\{0,1,\dots,\lfloor (m+3)/2\rfloor\}$ (since they are nonnegative and sum to~$1$).
Thus $H_m$ is a well-defined random variable with $\Pr(H_m=k)=a_{m,k}/2^m$.

Define the limiting log-moment generating function, when it exists,
\[
\Lambda(s)=\lim_{m\to\infty}\frac{1}{m}\log \mathbb{E}\bigl[e^{sH_m}\bigr].
\]

\begin{theorem}[Log-mgf]\label{thm:lambda-mgf}
There exists a complex neighborhood $V$ of $s=0$ such that for all $s\in V$,
\[
\Lambda(s)=\log \lambda_+(e^s)-\log \lambda_+(1)=\log \lambda_+(e^s)-\log 2.
\]
\end{theorem}

\begin{proof}
By Theorem~\ref{thm:lambda-plus} applied with $y=e^s$, we have
$\log Z_m(e^s)=m\log \lambda_+(e^s)+O(1)$ uniformly for $s$ in a small neighborhood of $0$ (where $e^s$ lies in the domain~$U$ of $\lambda_+$).
The same holds at $s=0$ with $\lambda_+(1)=2$.
Subtracting and dividing by $m$ in \eqref{eq:mgf-def} yields the claim.
\end{proof}

\subsection{Closed-form constants for LLN and CLT}

Let $\mu=\Lambda'(0)$ and $\sigma^2=\Lambda''(0)$.

\begin{theorem}[LLN and CLT]\label{thm:lln-clt}
We have
\[
\mu=\frac{5}{18},\qquad \sigma^2=\frac{67}{972}.
\]
Moreover:
\begin{enumerate}
  \item $m^{-1}H_m\to \mu$ in probability as $m\to\infty$;
  \item $\displaystyle \frac{H_m-\mu m}{\sqrt{m}}\Rightarrow \mathcal{N}(0,\sigma^2)$ in distribution.
\end{enumerate}
\end{theorem}

\begin{proof}
The constants follow from implicit differentiation of $\Pi(\lambda_+(y),y)=0$ at $y=1$ and the chain rule $y=e^s$; the explicit computation is given in Appendix~\ref{app:implicit}, yielding $\lambda_+'(1)=5/9$ and $\lambda_+''(1)=-64/243$.

For the convergence statements, observe that $\Lambda(s)=\log\lambda_+(e^s)-\log 2$ is analytic in a complex neighborhood of $s=0$ (Theorem~\ref{thm:lambda-mgf}), with $\Lambda(0)=0$, $\Lambda'(0)=\mu>0$, and $\Lambda''(0)=\sigma^2>0$.
These are precisely the hypotheses of the quasi-powers theorem of Hwang~\cite{Hwang1998} (see also Flajolet and Sedgewick~\cite[Chapter~IX, Theorem~IX.8]{FlajoletSedgewick2009}), which gives both the law of large numbers and the central limit theorem.
\end{proof}

\subsection{Large deviations and higher-order cumulants (supplement)}

\begin{theorem}[Local LDP for \(H_m/m\)]\label{thm:local-ldp}
Let \(H_m\) be as in the exponential family above, and let \(\Lambda(s)\) be as in Theorem~\ref{thm:lambda-mgf}.
There exists an interval \((s_-,s_+)\) containing \(0\) on which \(\Lambda\) is finite and differentiable.
Then \(H_m/m\) satisfies a large-deviation principle with speed \(m\) on
\[
\mathcal J:=\Lambda'((s_-,s_+))\subset\mathbb R,
\]
with rate function
\[
I(x)=\sup_{s\in\mathbb R}\{sx-\Lambda(s)\}.
\]
For \(s\in(s_-,s_+)\), writing \(y=e^s\) and
\[
x(s)=\Lambda'(s)=\frac{y\,\lambda_+'(y)}{\lambda_+(y)},
\]
one has the parametric form
\[
I\!\bigl(x(s)\bigr)=x(s)\,s-\Lambda(s)=x(s)\log y-\log\lambda_+(y)+\log2.
\]
\end{theorem}

\begin{remark}[Third-order cumulant]
Under the same differentiability regime, the first three derivatives at \(s=0\) are
\[
\Lambda'(0)=\mu,\qquad
\Lambda''(0)=\sigma^2,\qquad
\Lambda'''(0)=\frac{457}{2187}.
\]
Hence the standardized skewness constant is
\[
\gamma_1=\frac{\Lambda'''(0)}{\Lambda''(0)^{3/2}}.
\]
The value \(\Lambda'''(0)\) comes from the third implicit differentiation of
\(\Pi(\lambda,y)=0\) at \((\lambda,y)=(2,1)\), equivalently \(\lambda_+'''(1)=814/2187\);
an explicit closed form can be added to Appendix~\ref{app:implicit} as a third-order Edgeworth constant.
\end{remark}
