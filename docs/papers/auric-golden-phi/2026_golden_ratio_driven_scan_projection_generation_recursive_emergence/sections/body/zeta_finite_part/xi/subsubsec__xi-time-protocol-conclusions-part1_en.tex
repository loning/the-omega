\paragraph{"Temporalization" of fiber uniformization: explicit continuous semigroup and exponential certificates generated by idempotent Markov reflector}
Fix resolution \(m\). Let the folding readout be \(\Fold_m:\Omega_m\to X_m\), pushforward (visible readout) be \(P_m:=(\Fold_m)_\ast\), fiber uniform lifting (maximal entropy lifting) be \(Q_m\), and the fiber reflector
\[
K_m:=Q_m\circ P_m
\]
as defined in Theorem \ref{thm:pom-fiber-reflector}; it is equivalently the conditional expectation interface of "averaging over folding fibers" (Proposition \ref{prop:pom-Km-bistochastic}), and satisfies spectral dichotomy \(\mathrm{spec}(K_m)\subset\{0,1\}\) (Corollary \ref{cor:pom-Km-spectrum}).
This paragraph externalizes "information forgetting within fibers" as continuous time parameter \(t\ge 0\), and gives its auditable closure with exponential scale \(\log\mathrm{Ind}(E_m)\) (Proposition \ref{prop:fold-condexp-index-maxfiber}).
\par

\begin{theorem}[Explicit continuous-time semigroup for fiber uniformization]\label{thm:xi-fiber-uniformization-markov-semigroup-explicit}
Let the generator \(L_m:=K_m-I\) (\(I\) being the identity operator).
Then
\[
S_m(t):=\exp\!\bigl(t(K_m-I)\bigr)\qquad (t\ge 0)
\]
defines a family of strongly continuous Markov semigroups (equivalent to uniformly continuous in finite-dimensional spaces), with closed-form expression
\[
\boxed{\;
S_m(t)=e^{-t}I+(1-e^{-t})K_m,\qquad t\ge 0.
\;}
\]
In particular, \(S_m(0)=I\), and as \(t\to\infty\), \(S_m(t)\to K_m\) (in operator norm sense).
\end{theorem}
\begin{proof}
By \(K_m^2=K_m\) (Theorem \ref{thm:pom-fiber-reflector}), we obtain
\(
\exp(tK_m)=I+(e^{t}-1)K_m
\).
Thus
\[
\exp\!\bigl(t(K_m-I)\bigr)
=e^{-t}\exp(tK_m)
=e^{-t}I+(1-e^{-t})K_m.
\]
Semigroup law and strong continuity hold directly for finite-dimensional linear operator exponentials.
Since \(K_m\) and \(I\) both preserve non-negativity and total mass (Proposition \ref{prop:pom-Km-bistochastic}), \(S_m(t)\) is also a Markov operator. Proven.
\end{proof}

\begin{proposition}[Exact convergence rate and resolution-independent time constant from spectral dichotomy]\label{prop:xi-fiber-uniformization-spectral-decay-equals-exp}
For any vector \(v\) (distribution or observable), let
\[
v_{\mathrm{im}}:=K_m v,\qquad v_{\ker}:=(I-K_m)v.
\]
Then for all \(t\ge 0\) there holds the strict decomposition
\[
\boxed{\;
S_m(t)v=v_{\mathrm{im}}+e^{-t}v_{\ker}.
\;}
\]
Therefore, all components "deviating from fiber uniformity" decay at exact rate \(e^{-t}\); this time constant is \(1\), independent of resolution \(m\).
\end{proposition}
\begin{proof}
By Theorem \ref{thm:xi-fiber-uniformization-markov-semigroup-explicit},
\(
S_m(t)=e^{-t}I+(1-e^{-t})K_m
\).
Substituting and rearranging yields
\(
S_m(t)v=K_mv+e^{-t}(I-K_m)v
\), i.e., the stated decomposition.
Spectral dichotomy \(\mathrm{spec}(K_m)\subset\{0,1\}\) (Corollary \ref{cor:pom-Km-spectrum}) indicates this decomposition corresponds to eigendecomposition in image/kernel directions. Proven.
\end{proof}

\begin{proposition}[Time--index unified upper bound: explicit exponential decay and index cap for relative entropy gap]\label{prop:xi-fiber-uniformization-kl-gap-exp-decay-index-cap}
For any \(\mu\in\Delta(\Omega_m)\), denote the limiting fiber uniform state \(\mu_\infty:=K_m\mu\), and let
\[
\mu_t:=S_m(t)\mu=e^{-t}\mu+(1-e^{-t})\mu_\infty.
\]
Then the relative entropy gap satisfies the strict exponential decay inequality
\[
\boxed{\;
D_{\mathrm{KL}}(\mu_t\|\mu_\infty)\le e^{-t}\,D_{\mathrm{KL}}(\mu\|\mu_\infty).\;}
\]
Moreover, there exists a uniform cap controlled by the Jones/Pimsner--Popa index:
\[
\boxed{\;
D_{\mathrm{KL}}(\mu_t\|\mu_\infty)\le e^{-t}\log\mathrm{Ind}(E_m).\;}
\]
\end{proposition}
\begin{proof}
The first formula is given by convexity of relative entropy in the first variable:
since \(\mu_t=e^{-t}\mu+(1-e^{-t})\mu_\infty\), we have
\(
D_{\mathrm{KL}}(\mu_t\|\mu_\infty)\le e^{-t}D_{\mathrm{KL}}(\mu\|\mu_\infty)
\).
\par
The second formula follows from
\(
D_{\mathrm{KL}}(\mu\|\mu_\infty)\le D_\infty(\mu\|\mu_\infty)
\)
(monotonicity of Rényi relative entropy) and Pimsner--Popa type domination: for each atom \(a\in\Omega_m\), if its fiber size is \(d_m(\Fold_m(a))\), then
\(
\mu(a)\le d_m(\Fold_m(a))\,\mu_\infty(a)\le D_m\,\mu_\infty(a)
\)
(where \(D_m:=\max_x d_m(x)\)).
Thus \(D_\infty(\mu\|\mu_\infty)\le \log D_m\).
Again by \(\mathrm{Ind}(E_m)=D_m\) (Proposition \ref{prop:fold-condexp-index-maxfiber}), we get
\(
D_{\mathrm{KL}}(\mu\|\mu_\infty)\le \log\mathrm{Ind}(E_m)
\).
Combining with the first formula yields the conclusion. Proven.
\end{proof}

\begin{corollary}[Linear amplification of gap and randomness cost in Moment semantics]\label{cor:xi-momq-gap-linear-amplification-randomness-cost}
Let the index gap of some delinking/unlinking interface be \(\Gamma\ge 0\) (Appendix \ref{subsubsec:op-algebra-jones-scalar-twirl-cost-renyi-flatness}), and form amplified gap \(\Gamma^{(q)}\) under \(q\)-fold juxtaposition semantics in \(\mathrm{MOM}_q\).
Then under tensorization/juxtaposition semantics there holds the linear law
\[
\boxed{\;\Gamma^{(q)}=q\,\Gamma.\;}
\]
Moreover, if we require compressing the leakage to \(\varepsilon\) (characterized by conditional mutual information or equivalent index), then any mechanism using an external random register \(R\) must satisfy
\[
\boxed{\;H(R)\ge \Gamma^{(q)}-\varepsilon=q\Gamma-\varepsilon.\;}
\]
\end{corollary}
\begin{proof}[Proof sketch]
Index gap is strictly additive under tensorization (multiplicativity of logarithmic index), so \(\Gamma^{(q)}=q\Gamma\).
Randomness cost lower bound is given by the strong duality inequality of optimal delinking curve (Theorem \ref{thm:op-algebra-optimal-delinking-curve}), substituting the gap with \(\Gamma^{(q)}\) yields the result.
\end{proof}

\begin{corollary}[Strong converse with dual time: bilinear leakage lower bound under fixed randomness budget]\label{cor:xi-momq-strong-converse-bilinear-leakage}
Under the setting of Corollary \ref{cor:xi-momq-gap-linear-amplification-randomness-cost}, if randomness budget per round is limited to \(H(R)\le h\), and the amplified interface is cascaded for \(n\) rounds, then the leakage of any mechanism satisfies the strong converse lower bound
\[
\boxed{\;
\mathrm{Leak}_n\ \ge\ n\max\{q\Gamma-h,\,0\}. \;}
\]
\end{corollary}
\begin{proof}[Proof sketch]
Substituting the \(q\)-amplified gap \(\Gamma^{(q)}=q\Gamma\) into the multi-round strong converse theorem (Theorem \ref{thm:op-algebra-multi-round-strong-converse-linear-leakage}) yields the result.
\end{proof}

\begin{proposition}[Stable layer invariance and time orthogonal decomposition: fiber mixing time completely invisible to visible compression]\label{prop:xi-fiber-mixing-time-invisible-on-stable-compression}
Let \(P_m,Q_m,K_m\) be as above, and take any linear operator \(T:\Delta(\Omega_m)\to\Delta(\Omega_m)\).
For any \(t\ge 0\), define "fiber time regularization"
\[
T^{(t)}:=S_m(t)\,T\,S_m(t).
\]
Then its stable layer compression (visible operator) is strictly independent of \(t\):
\[
\boxed{\;
P_m\,T^{(t)}\,Q_m\ =\ P_m\,T\,Q_m.\;}
\]
\end{proposition}
\begin{proof}
By \(P_mQ_m=\mathrm{id}\) and \(K_m=Q_mP_m\), we obtain \(P_mK_m=P_m\), \(K_mQ_m=Q_m\) (also see \S\ref{subsubsec:xi-fixed-slice-audit-chain}).
Again by the closed form in Theorem \ref{thm:xi-fiber-uniformization-markov-semigroup-explicit},
\(
S_m(t)=e^{-t}I+(1-e^{-t})K_m
\),
thus
\[
P_mS_m(t)=e^{-t}P_m+(1-e^{-t})P_mK_m=P_m,
\qquad
S_m(t)Q_m=e^{-t}Q_m+(1-e^{-t})K_mQ_m=Q_m.
\]
Hence
\(
P_mT^{(t)}Q_m=P_mS_m(t)TS_m(t)Q_m=P_mTQ_m
\).
Proven.
\end{proof}

\begin{corollary}[Rényi order-independent time--index certificate preserved under \texorpdfstring{$q$}{q}-amplification]\label{cor:xi-renyi-flatness-under-momq-amplification}
Let \(E\) be a finite-index conditional expectation.
For any \(\alpha\in[1,\infty]\), let \(D_\alpha\) be the (sandwiched) Rényi relative entropy, and denote
\(
L_\alpha(\rho):=D_\alpha(\rho\|\rho\circ E)
\).
Then the worst-case loss satisfies the uniform extremal formula
\[
\sup_{\rho}L_\alpha(\rho)=\log\mathrm{Ind}(E)

Under juxtaposition semantics in \(\mathrm{MOM}_q\), the index scale linearly amplifies by \(q\), thus for all Rényi orders we synchronously obtain
\[
\boxed{\;
\sup_{\rho}D_\alpha(\rho\|\rho\circ E^{(q)})=q\log\mathrm{Ind}(E),
\qquad \forall\,\alpha\in[1,\infty].\;}
\]
\end{corollary}
