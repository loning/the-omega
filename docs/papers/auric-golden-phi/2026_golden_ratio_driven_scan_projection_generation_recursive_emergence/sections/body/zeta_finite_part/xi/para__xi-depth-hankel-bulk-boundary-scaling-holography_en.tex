\begin{corollary}[Double Scaling Rate of Bulk--Boundary: Boundary Linear Slope \texorpdfstring{$\kappa$}{kappa}, Bulk Limit Slope \texorpdfstring{$\kappa^2$}{kappa^2}]\label{cor:xi-bulk-boundary-double-scaling-holography}
Under the finite-atom setting of Proposition \ref{prop:xi-hankel-det-scaling-covariance}, define the bulk entropy potential
\[
S_{\mathrm{bulk}}(m):=-\log\det \widetilde{\mathsf H}_{\kappa-1},
\]
where \(\widetilde{\mathsf H}_{\kappa-1}\) is the rescaled Hankel principal block.
Then as \(m\to\infty\), the asymptotic holds
\[
\boxed{\;
S_{\mathrm{bulk}}(m)=\kappa^2\log m+O(1).\;}
\]
Moreover, if the boundary defect entropy \(S_{\mathrm{gen}}(m)\) is taken as the logarithmic-scale family in Corollary \ref{cor:xi-pick-poisson-entropy-derivative-visible-dimension}, then on the logarithmic scale \(s=\log m\to\infty\), the proportionality law holds
\[
\boxed{\;
S_{\mathrm{bulk}}(e^{s})=\kappa\,S_{\mathrm{gen}}(e^{s})+O(1).\;}
\]
\end{corollary}
\begin{proof}
By the covariance law of Proposition \ref{prop:xi-hankel-det-scaling-covariance},
\(
\det \widetilde{\mathsf H}_{\kappa-1}=\det \mathsf H_{\kappa-1}\cdot J_{\kappa-1}(m;\delta)
\),
hence
\(
S_{\mathrm{bulk}}(m)=S_{\mathrm{bulk}}(1)-\log J_{\kappa-1}(m;\delta)
\).
From
\[
\log J_{\kappa-1}(m;\delta)
=\kappa(\kappa-1)\log m
+(2\kappa-1)\sum_{j=1}^{\kappa}\Bigl(\log(1+\delta_j)-\log(1+m\delta_j)\Bigr)
\]
and using the asymptotic \(\log(1+m\delta_j)=\log m+\log\delta_j+O(m^{-1})\), we obtain
\(
\log J_{\kappa-1}(m;\delta)=-\kappa^2\log m+O(1)
\),
thus
\(
S_{\mathrm{bulk}}(m)=\kappa^2\log m+O(1)
\).
On the other hand, Corollary \ref{cor:xi-pick-poisson-entropy-derivative-visible-dimension} gives
\(
S_{\mathrm{gen}}(e^s)=S_{\mathrm{gen}}(1)+\kappa s
\).
Combining the two yields
\(
S_{\mathrm{bulk}}(e^s)-\kappa S_{\mathrm{gen}}(e^s)=O(1)
\). QED.
\end{proof}

\begin{remark}[Quadratic Universal Exponent at Boundary Criticality: Entropy Singularity Coefficient from Collision Triggering Is Identically \(2\)]\label{rem:xi-collision-universal-exponent-two}
In the bulk regime, Proposition \ref{prop:xi-depth-hankel-determinant-vandermonde-square} gives
\(
\det \mathsf H_{\kappa-1}
=\bigl(\prod_j\widetilde w_j\bigr)\prod_{j<k}(b_k-b_j)^2
\);
in the boundary regime, Proposition \ref{prop:xi-pick-poisson-det-factorization} gives
\(
\det\mathsf P=(\prod_j p_j)\prod_{j<k}\rho_{jk}^2
\).
Thus when some pair of parameters collides \(b_j\to b_k\) or \(\rho_{jk}\to 0\) (with others held in a compact domain),
the corresponding entropy potential exhibits a universal logarithmic divergence
\[
S_{\mathrm{bulk}}
:=-\log\det \mathsf H_{\kappa-1}
=2\log\frac{1}{|b_k-b_j|}+O(1),
\qquad
S_{\mathrm{gen}}
:=-\log\det\mathsf P
=2\log\frac{1}{\rho_{jk}}+O(1).
\]
This coefficient \(2\) is forced by the squared-degeneracy structure of the Gram determinant, independent of weights and the specific configuration of other points.
\end{remark}
