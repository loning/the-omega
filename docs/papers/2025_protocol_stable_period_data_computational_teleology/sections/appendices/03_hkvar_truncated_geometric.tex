
\section{Hardy--Krause variation of the truncated geometric kernel}
\label{app:hkvar_truncated_geometric}

This appendix justifies Proposition~\ref{prop:geom_hkvar}.

\paragraph{Hardy--Krause variation for smooth functions.}
If $f$ has continuous mixed partial derivatives up to order $d$ on $[0,1]^d$, then its Hardy--Krause variation admits the representation
\begin{equation}
\label{eq:hkvar_smooth_formula}
\HKVar(f)
=
\sum_{\emptyset\neq u\subseteq\{1,\dots,d\}}
\int_{[0,1]^{|u|}}
\left|
\frac{\partial^{|u|}}{\prod_{j\in u}\partial x_j}
f\big(x_u; \mathbf{1}_{-u}\big)
\right|\,
\dd x_u,
\end{equation}
where $x_u$ denotes the subvector of coordinates in $u$ and $\mathbf{1}_{-u}$ sets all complementary coordinates to $1$.
See, e.g., \cite[\S 3.3]{DickPillichshammer2010}.

\begin{proof}[Proof of Proposition~\ref{prop:geom_hkvar}]
Recall
\[
g_M(x_1,\dots,x_d)=\sum_{n=0}^{M-1}\left(\prod_{j=1}^d x_j\right)^n.
\]
Fix a nonempty subset $u\subseteq\{1,\dots,d\}$ with $|u|=k$.
On the boundary slice where $x_j=1$ for $j\notin u$, we have
\[
g_M(x_u;\mathbf{1}_{-u})=\sum_{n=0}^{M-1}\left(\prod_{j\in u} x_j\right)^n.
\]
For $n\ge 1$,
\[
\frac{\partial^k}{\prod_{j\in u}\partial x_j}
\left(\prod_{j\in u} x_j\right)^n
=
n^k\prod_{j\in u} x_j^{n-1},
\]
while the $n=0$ term differentiates to $0$.
Therefore the integrand in \eqref{eq:hkvar_smooth_formula} equals
\[
\left|
\sum_{n=1}^{M-1} n^k \prod_{j\in u} x_j^{n-1}
\right|
=
\sum_{n=1}^{M-1} n^k \prod_{j\in u} x_j^{n-1},
\]
since all terms are nonnegative on $[0,1]^k$.
Integrating termwise gives
\[
\int_{[0,1]^k}
n^k\prod_{j\in u} x_j^{n-1}\,\dd x_u
=
n^k\prod_{j\in u}\int_0^1 x^{n-1}\,\dd x
=
n^k\left(\frac{1}{n}\right)^k
=1.
\]
Summing over $n=1,\dots,M-1$ yields the contribution $M-1$ for each nonempty $u$.
There are $2^d-1$ nonempty subsets, hence
\[
\HKVar(g_M)=(2^d-1)(M-1).
\]
\end{proof}
