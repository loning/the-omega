
\section{Reproducible experiments (pure Python)}
\label{sec:reproducible_experiments}

This section provides reproducible numerical demonstrations of the paper's closed claims:
(i) scan protocols realize period values and (ii) finite-resource error decompositions admit auditable certificates.
The experiments do not ``prove'' equidistribution or period closure properties; they validate that the proposed implementations behave as predicted under finite resources.

\subsection{Reproducibility protocol}
\label{sec:reproducibility_protocol}

All experiments are implemented in pure Python~3 with no third-party dependencies.
To reproduce the tables, run the following scripts from this paper directory:
\begin{itemize}
\item \path{python3 scripts/exp_period_1d.py}
\item \path{python3 scripts/exp_zeta_truncated_geometric.py}
\item \path{python3 scripts/alpha_inverse_search.py}
\item \path{python3 scripts/teleology_dynamics.py}
\end{itemize}
The scripts write \LaTeX{} row files into \path{sections/generated/}, which are included below.
All randomness is controlled by fixed seeds.

\subsection{Experiment 1: one-dimensional scans for \texorpdfstring{$\log 2$}{log 2} and \texorpdfstring{$\pi$}{pi}}
\label{sec:exp_1d}

We use the golden-branch slope $\alpha=\varphi^{-1}$ and bounded-variation kernels
\[
f_{\log}(x)=\frac{1}{1+x},
\qquad
f_{\pi}(x)=\frac{4}{1+x^2},
\]
with exact period values
\[
\int_0^1 f_{\log}(x)\,\dd x=\log 2,
\qquad
\int_0^1 f_{\pi}(x)\,\dd x=\pi.
\]
The scripts compute the one-dimensional star discrepancy $\DNstar(P_N)$ exactly from its definition and report the deterministic Koksma certificate
\[
|\avgN{\mathcal{P}}-\per(\HSP(\mathcal{P}))|\le \Var(f)\,\DNstar(P_N).
\]

\subsection{Experiment 2: truncated geometric kernels for \texorpdfstring{$\zeta(2)$}{zeta(2)} and \texorpdfstring{$\zeta(3)$}{zeta(3)}}
\label{sec:exp_zeta}

\begin{sloppypar}
We use the truncated geometric kernel $g_M$ (Definition~\ref{def:geom_kernel}) with exact periods $H_M^{(d)}$ (Lemma~\ref{lem:geom_period}).
For $d=2,3$ the truncation remainders satisfy Proposition~\ref{prop:zeta_truncation}.
The scan slopes are
\[
\alpha_{2}=(\varphi^{-1},\sqrt{2}-1),
\qquad
\alpha_{3}=(\varphi^{-1},\sqrt{2}-1,\sqrt{3}-1).
\]
To obtain explicit discrepancy certificates in dimensions $2$ and $3$, the scripts compute the ETK bracket term $B_{N,H}(\alpha)$ (Definition~\ref{def:kronecker_bracket}), choose $H$ from fixed candidate sets by minimizing it, and use the explicit constant $C_d=(3/2)^d$ (Theorem~\ref{thm:etk}).
To preserve the inequality direction under finite-precision evaluation of $h\cdot\alpha$, the implementation computes $\distZ{h\cdot\alpha}$ with a conservative floating-point error guard, so that the reported $B_{N,H}(\alpha)$ remains an auditable upper bound at the code level \cite{Higham2002}.
\end{sloppypar}
