\subsubsection{Exact splitting of the primitive orbit spectrum of the real input kernel: \texorpdfstring{$M=(F\otimes F)\ \boxplus\ (-F)$}{M=(F⊗F) boxplus (-F)}}\label{subsec:real-input-40-prime-splitting}
Proposition \ref{prop:real-input-40-fib-tensor} has already provided a factorization at the \(\zeta\)-level:
\(
\det(I-zM)=(1-z)^2(1+z)\det(I-z(F\otimes F))\det(I-z(-F))
\).
This subsection extends the decomposition to the primitive orbit spectrum level: we give an \textbf{exact difference formula} expressing \(p_n(M)\) in terms of \(p_n(F\otimes F)\) and \(p_n(F)\), and derive from it a fast-converging expression for the Abel--Mertens constant.

\paragraph{Notation}
Let
\[
F:=\begin{pmatrix}1&1\\[2pt]1&0\end{pmatrix},\qquad
B:=F\otimes F,
\qquad
\lambda:=\rho(M)=\rho(B)=\varphi^2,
\qquad
z_\star:=\lambda^{-1}=\varphi^{-2}.
\]
For any integer adjacency matrix \(T\), let \(p_n(T)\) denote the number of primitive orbits of length \(n\) (counted by orbit, satisfying \(a_n(T)=\Tr(T^n)=\sum_{d\mid n} d\,p_d(T)\)).
We further write
\[
\pi_n:=p_n(F),\qquad r_n:=p_n(B)=p_n(F\otimes F).
\]

\begin{theorem}[Artin-type prime spectrum splitting: exact difference of \texorpdfstring{$p_n(M)$}{p_n(M)}]\label{thm:real-input-40-prime-artin-splitting}
For all \(n\ge 3\), the following exact identity holds:
\[
\boxed{\;
p_n(M)=r_n+(-1)^n\,\pi_n+\mathbf{1}_{\{n\equiv 2\!\!\pmod 4\}}\ \pi_{n/2}\;}
\]
where \(\mathbf{1}_{\{n\equiv 2\!\!\pmod 4\}}\) is the indicator function.
Equivalently, for \(n\ge 3\):
\begin{itemize}
  \item if \(n\) is odd, then \(p_n(M)=r_n-\pi_n\);
  \item if \(4\mid n\), then \(p_n(M)=r_n+\pi_n\);
  \item if \(n\equiv 2\pmod 4\), then \(p_n(M)=r_n+\pi_n+\pi_{n/2}\).
\end{itemize}
Furthermore, the low-order terms satisfy
\[
p_1(M)=r_1,
\qquad
p_2(M)=r_2+\pi_2+\pi_1+1.
\]
\end{theorem}
\begin{proof}
By Proposition \ref{prop:real-input-40-fib-tensor} and the additivity of traces (summing over block-diagonal and spectral decomposition), one obtains for all \(n\ge 1\):
\[
a_n(M):=\Tr(M^n)
=\Tr(B^n)+\Tr((-F)^n)+2\cdot 1^n+(-1)^n.
\]
Denoting the Lucas numbers by \(L_n:=\Tr(F^n)\) (see the definition in \S\ref{app:abel-mertens-universal} and the formula near \((\ref{app:abel-mertens-universal})\)), we have
\[
\Tr(B^n)=\Tr((F\otimes F)^n)=\Tr(F^n)^2=L_n^2,
\qquad
\Tr((-F)^n)=(-1)^n L_n.
\]
This yields the closed form
\[
a_n(M)=L_n^2+(-1)^n(L_n+1)+2.
\]
On the other hand, \(a_n(B)=L_n^2\), so we define the ``vertical correction trace''
\[
b_n:=a_n(M)-a_n(B)=2+(-1)^n+(-1)^nL_n.
\]
By M\"obius inversion (\(\,p_n=\frac1n\sum_{d\mid n}\mu(d)a_{n/d}\,\), see the end of \S\ref{prop:real-input-40-essential-20}), we have
\[
p_n(M)-r_n=\frac{1}{n}\sum_{d\mid n}\mu(d)\,b_{n/d}.
\]
We decompose \(b_m=2+(-1)^m+(-1)^mL_m\) into three parts:
\begin{itemize}
  \item The M\"obius transform of the constant term \(2\) contributes only at \(n=1\);
  \item The M\"obius transform of the pure sign term \((-1)^m\) contributes only at \(n=1,2\);
  \item For \(n\ge 3\), only the term
  \[
  \frac{1}{n}\sum_{d\mid n}\mu(d)(-1)^{n/d}L_{n/d}
  \]
  remains.
\end{itemize}
Thus for \(n\ge 3\), it suffices to prove
\[
\frac{1}{n}\sum_{d\mid n}\mu(d)(-1)^{n/d}L_{n/d}
=(-1)^n\pi_n+\mathbf{1}_{\{n\equiv 2\!\!\pmod 4\}}\pi_{n/2}.
\]
\textbf{(i) \(n\) odd:} In this case, for all \(d\mid n\) the quotient \(n/d\) is odd, so \((-1)^{n/d}=-1\) is constant, yielding
\(
\frac{1}{n}\sum_{d\mid n}\mu(d)(-1)^{n/d}L_{n/d}=-\pi_n
\), which is the desired conclusion.
\textbf{(ii) \(n=2m\) even:} Since \(\mu(d)=0\) whenever \(4\mid d\), only two types of terms contribute to the sum: those with \(d\) odd and those with \(d=2d'\) (\(d'\) odd). Thus
\[
\frac{1}{2m}\sum_{d\mid 2m}\mu(d)(-1)^{2m/d}L_{2m/d}
=\frac{1}{2m}\sum_{\substack{d\mid m\\ d\ \mathrm{odd}}}\mu(d)\,L_{2m/d}
\ -\ \frac{1}{2m}\sum_{\substack{d\mid m\\ d\ \mathrm{odd}}}\mu(d)\,(-1)^{m/d}L_{m/d}.
\]
If \(m\) is even (i.e., \(4\mid n\)), then \(m/d\) remains even, so \((-1)^{m/d}=+1\), and the expression equals
\(
\frac{1}{2m}\sum_{d\mid 2m}\mu(d)L_{2m/d}=\pi_{2m}
\)
(again because the terms with \(4\mid d\) are annihilated by \(\mu\)). If \(m\) is odd (i.e., \(n\equiv 2\pmod 4\)), then \(m/d\) is odd, so \((-1)^{m/d}=-1\), and the expression becomes
\[
\frac{1}{2m}\sum_{\substack{d\mid m\\ d\ \mathrm{odd}}}\mu(d)\,L_{2m/d}
\ +\ \frac{1}{2m}\sum_{\substack{d\mid m\\ d\ \mathrm{odd}}}\mu(d)\,L_{m/d}
=\pi_{2m}+\pi_m.
\]
Combining the two cases yields the conclusion for even \(n\ge 4\). Finally, the low-order corrections at \(n=1,2\) are listed separately. \qed
\end{proof}
\begin{corollary}[Space/time/boundary three-fold decomposition (term-by-term verifiable)]\label{cor:real-input-40-space-twist-boundary-split}
Retaining the notation \(F=\bigl(\begin{smallmatrix}1&1\\[2pt]1&0\end{smallmatrix}\bigr)\), \(B=F\otimes F\), and Lucas numbers \(L_n=\Tr(F^n)\).
Define the ``boundary short-period'' and ``time-twist block'' primitive coefficients as
\[
b_n:=\mathbf{1}_{\{n=1\}}+\mathbf{1}_{\{n=2\}},
\qquad
\tau_n:=p_n(-F)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,(-1)^{n/d}L_{n/d}.
\]
Then for all \(n\ge 1\), the following strict identity holds:
\[
\boxed{\; p_n(M)=p_n(B)+\tau_n+b_n\; }.
\]
Moreover, \(b_n=0\) for all \(n\ge 3\), so for \(n\ge 3\),
\(
p_n(M)-p_n(B)=p_n(-F)
\)
is given exactly by the twist block (see the first \(10\) terms in Table \ref{tab:real_input_40_primitive_artin_split}).
\end{corollary}
\begin{corollary}[Product rule at prime lengths (auditable additive-multiplicative unification)]\label{cor:real-input-40-prime-length-product}
Let \(p\) be an odd prime, and denote the Lucas number \(L_p=\Tr(F^p)\) and \(\pi_p=p_p(F)\). Then the real input kernel satisfies, at prime lengths,
\[
\boxed{\;
p_p(M)=\frac{L_p(L_p-1)}{p}=L_p\cdot \pi_p\; }.
\]
\end{corollary}
\begin{proof}
For an odd prime \(p\), from \(a_p(M)=L_p^2- (L_p+1)+2=L_p^2-L_p+1\) and \(a_1(M)=1\) we obtain
\[
p_p(M)=\frac{1}{p}\bigl(a_p(M)-a_1(M)\bigr)=\frac{L_p^2-L_p}{p}=\frac{L_p(L_p-1)}{p}.
\]
On the other hand, \(\pi_p=\frac{1}{p}(L_p-L_1)=\frac{L_p-1}{p}\), so \(p_p(M)=L_p\pi_p\). \qed
\end{proof}
\begin{corollary}[Closed-form identity for the primitive generating function]\label{cor:real-input-40-prime-genfun-identity}
Let
\[
P_M(z):=\sum_{n\ge 1}p_n(M)z^n,\qquad
P_B(z):=\sum_{n\ge 1}r_n z^n,\qquad
P_F(z):=\sum_{n\ge 1}\pi_n z^n.
\]
Then the following identity holds:
\[
\boxed{\;
P_M(z)-P_B(z)= z+z^2+P_F(-z)+\frac{P_F(z^2)-P_F(-z^2)}{2}\;}
\qquad(|z|\ \text{sufficiently small}).
\]
\end{corollary}
\begin{proof}
This follows by comparing coefficients. By Theorem \ref{thm:real-input-40-prime-artin-splitting}, for \(n\ge 3\) we have
\(
[z^n](P_M-P_B)=(-1)^n\pi_n+\mathbf{1}_{\{n\equiv 2\pmod 4\}}\pi_{n/2}
\), while
\(
[z^n]P_F(-z)=(-1)^n\pi_n
\).
On the other hand,
\(
\frac{P_F(z^2)-P_F(-z^2)}{2}
\)
has coefficient at \(z^{2m}\) equal to \(\frac{1-(-1)^m}{2}\pi_m\), which equals \(\pi_{n/2}\) only when \(m\) is odd (i.e., \(n\equiv 2\pmod 4\)), and is \(0\) otherwise.
Finally, \(z+z^2\) absorbs the low-order corrections at \(n=1,2\). \qed
\end{proof}
\begin{corollary}[Fast-converging expression for the Abel--Mertens constant (difference/three-point-value formula)]\label{cor:real-input-40-logM-fast-prime}
Retaining the notation of Theorem \ref{thm:abel-mertens-universal}, \(\log\mathfrak{M}_M\) and \(\log\mathfrak{M}_{\mathrm{in}}\) correspond to \(M\) and the input skeleton \(B=F\otimes F\), respectively.
The difference satisfies the absolutely convergent series
\[
\boxed{\;
\log\mathfrak{M}_M-\log\mathfrak{M}_{\mathrm{in}}
=\sum_{n\ge 1}\frac{p_n(M)-r_n}{\lambda^n}\;}
\]
and by Corollary \ref{cor:real-input-40-prime-genfun-identity}, one obtains the three-point-value formula
\[
\boxed{\;
\log\mathfrak{M}_M-\log\mathfrak{M}_{\mathrm{in}}
=\frac{1}{\lambda}+\frac{1}{\lambda^2}
P_F\!\left(-\frac{1}{\lambda}\right)
\frac{
P_F\!\left(\frac{1}{\lambda^2}\right)-P_F\!\left(-\frac{1}{\lambda^2}\right)
}{2}\; }.
\]
\end{corollary}
\begin{proof}
By the proof of Corollary \ref{cor:abel-mertens-constant-universal} (the formula \(\log\mathfrak{M}=\sum_{n\ge 1}(p_n/\lambda^n-1/n)\)), one obtains
\[
\log\mathfrak{M}_M-\log\mathfrak{M}_{\mathrm{in}}
=\sum_{n\ge 1}\left(\frac{p_n(M)}{\lambda^n}-\frac{1}{n}\right)
-\sum_{n\ge 1}\left(\frac{r_n}{\lambda^n}-\frac{1}{n}\right)
=\sum_{n\ge 1}\frac{p_n(M)-r_n}{\lambda^n},
\]
whose absolute convergence follows from \(p_n(M)-r_n=O(\varphi^n/n)\) and \(\lambda=\varphi^2\).
Substituting the right-hand side of Corollary \ref{cor:real-input-40-prime-genfun-identity} at \(z=z_\star=\lambda^{-1}\) yields the three-point-value formula. \qed
\end{proof}
\begin{remark}[End-to-end auditable algorithm: using only Lucas trace samples]\label{rem:real-input-40-lucas-pipeline}
Theorem \ref{thm:real-input-40-prime-artin-splitting} and Corollary \ref{cor:real-input-40-logM-fast-prime} can be combined into an end-to-end compilation pipeline with ``minimal data dependency'': the only input is the Lucas trace samples \(L_n=\Tr(F^n)\) (generated by the recurrence \(L_1=1,L_2=3,L_n=L_{n-1}+L_{n-2}\)), which suffice to reconstruct the real kernel primitive spectrum and the Abel constant difference without explicitly enumerating primitive orbits.
\begin{enumerate}
  \item \textbf{(ghost\(\to\)Witt)} Compute via M\"obius/Witt inversion:
  \[
  \pi_n=p_n(F)=\frac1n\sum_{d\mid n}\mu(d)\,L_{n/d},
  \qquad
  r_n=p_n(F\otimes F)=\frac1n\sum_{d\mid n}\mu(d)\,L_{n/d}^2.
  \]
  \item \textbf{(mod 4 selection rule)} Compute \(p_n(M)\) using Theorem \ref{thm:real-input-40-prime-artin-splitting}.
  \item \textbf{(Three-point evaluation)} Compute
  \(\log\mathfrak{M}_M-\log\mathfrak{M}_{\mathrm{in}}\) using Corollary \ref{cor:real-input-40-logM-fast-prime}.
\end{enumerate}
\par
\textbf{Error budget (explicit tail bound).} The three-point-value formula involves only \(P_F(z)=\sum_{n\ge 1}\pi_n z^n\) evaluated at
\(z\in\{-\lambda^{-1},\pm\lambda^{-2}\}\). Since for all \(n\ge 1\) we have
\(
n\pi_n\le a_n(F)=L_n\le \varphi^n+1
\),
the following auditable upper bound holds for \(|z|<\varphi^{-1}\):
\[
\sum_{n>N}\pi_n|z|^n
\le \sum_{n>N}\frac{(\varphi|z|)^n}{n}+\sum_{n>N}\frac{|z|^n}{n}
\le \frac{(\varphi|z|)^{N+1}}{(N+1)(1-\varphi|z|)}+\frac{|z|^{N+1}}{(N+1)(1-|z|)}.
\]
Accordingly, define
\[
\mathrm{Tail}(N;|z|):=\frac{(\varphi|z|)^{N+1}}{(N+1)(1-\varphi|z|)}+\frac{|z|^{N+1}}{(N+1)(1-|z|)}.
\]
Then when \(P_F\) is truncated to \(N\) terms at all three evaluation points, the total error satisfies
\[
\bigl|\mathrm{Err}_N\bigr|
\le \mathrm{Tail}(N;\lambda^{-1})+\mathrm{Tail}(N;\lambda^{-2}).
\]
From \(\varphi\lambda^{-1}=\varphi^{-1}\) and \(\varphi\lambda^{-2}=\varphi^{-3}\), one obtains the minimum sample length conclusion
\(
N(\varepsilon)=O(\log(1/\varepsilon))
\)
(so that only \(L_1,\dots,L_{N(\varepsilon)}\) are needed).
\par
\textbf{Computational complexity (sieve implementation).} For a given \(N\), the M\"obius inversion can be implemented using a linear sieve for \(\mu(1),\dots,\mu(N)\) and divisor convolution (accumulating each \(d\) to its multiples), so that generating \(\{\pi_n\}_{n\le N}\) and \(\{r_n\}_{n\le N}\) has time complexity \(O(N\log N)\) and space \(O(N)\); the subsequent mod \(4\) selection rule and three-point evaluation are linear time.
\par
\textbf{Reproducible implementation:} The script \texttt{scripts/exp\_real\_input\_40\_lucas\_pipeline.py} implements the above pipeline, simultaneously auditing:
(i) consistency between the mod \(4\) selection rule and the M\"obius(trace) computation;
(ii) the product rule at prime lengths \(p_p(M)=L_p\pi_p\) (odd primes \(p\));
(iii) consistency between the three-point-value formula and the series \(\sum_{n\ge 1}(p_n(M)-r_n)\lambda^{-n}\) within the tail bound.
\end{remark}
