\begin{corollary}[单位圆扭曲的相位--幅度分离：\texorpdfstring{$P(i\theta)=\frac{i\theta}{2}+\log|\Lambda(s)|$}{P(i theta)=i theta/2 + log|Lambda(s)|}]\label{cor:sync-kernel-weighted-phase-amplitude}
取单位圆扭曲 \(u=e^{i\theta}\)（\(\theta\in\RR\)），并令
\[
r:=e^{i\theta/2},\qquad s:=r+r^{-1}=2\cos(\theta/2)\in[-2,2].
\]
设 \(w_\star(s)\) 为方程 \(\widehat\Delta(w,s)=0\) 的根中满足 \(|w|\) 最小者，并定义
\[
\Lambda(s):=\frac{1}{w_\star(s)}.
\]
则最小极点与 Perron 根满足
\[
\boxed{\ z_\star(e^{i\theta})=\frac{w_\star(s)}{r},\qquad \lambda(e^{i\theta})=\frac{1}{z_\star(e^{i\theta})}=r\cdot \Lambda(s)\ },
\]
从而压力出现严格的“相位--幅度分离”：
\[
\boxed{\ P(i\theta)=\log\lambda(e^{i\theta})=\frac{i\theta}{2}+\log|\Lambda(2\cos(\theta/2))|\ }.
\]
\end{corollary}
\begin{proof}
由命题 \ref{prop:sync-kernel-weighted-hat-delta}，在 \(u=r^2\) 下有
\(
\Delta(z,u)=\widehat\Delta(zr,s)
\)。
对单位圆权重 \(|u|=1\) 有 \(|r|=1\)，因此对任意 \(w=zr\) 有 \(|z|=|w|\)。
故 \(\Delta(z,u)=0\) 在 \(z\)-平面上距离原点最近的零点（最小极点）与 \(\widehat\Delta(w,s)=0\) 在 \(w\)-平面上距离原点最近的零点一一对应，并满足 \(z_\star=w_\star/r\)。
由定义 \(\lambda(u)=1/z_\star(u)\) 得 \(\lambda(e^{i\theta})=r/w_\star(s)=r\Lambda(s)\)。取对数即得结论。证毕。
\end{proof}

\begin{theorem}[自对偶强制的 Gallavotti--Cohen 涨落对称（全域形式）]\label{thm:sync-kernel-gallavotti-cohen}
令 \(u=e^{\theta}\) 且 \(P(\theta)=\log\lambda(e^\theta)\)。定义完成化后的中心化累积母函数
\[
\Lambda(\theta):=P(\theta)-\frac{\theta}{2}.
\]
则对所有实 \(\theta\) 有精确恒等式
\[
\boxed{\ \Lambda(\theta)=\Lambda(-\theta)\ }.
\]
等价地，
\[
\boxed{\ P(\theta)=\theta+P(-\theta)\ }.
\]
此外：
\begin{enumerate}
  \item \textbf{（有限长度镜像对称）} 对任意 \(n\ge 2\) 有 \(p_{n,k}=p_{n,n-k}\)；
  \item \textbf{（率函数对称）} 若 \(I(\alpha)\) 为 \(\alpha=k/n\) 的大偏差率函数，则在其定义域上有
  \(\boxed{\,I(\alpha)=I(1-\alpha)\,}\)；
  \item \textbf{（奇阶累积量消失）} \(\Lambda\) 为偶函数蕴含 centered current 的所有奇阶累积量严格为零：
  \(P^{(2j+1)}(0)=0\ (j\ge 1)\)（参见命题 \ref{prop:fluctuation-even}）。
\end{enumerate}
\end{theorem}
\begin{proof}
\(\Lambda(\theta)\) 为偶函数与 \(P(\theta)=\theta+P(-\theta)\) 在代数上等价。
而 \(P(\theta)=\theta+P(-\theta)\) 正是本节自对偶相似恒等式 \(\Pi^{-1}B(u)\Pi=uB(1/u)\) 对 Perron 根取对数的直接推论。
有限长度镜像对称 \(p_{n,k}=p_{n,n-k}\) 已在上文给出；由 Legendre 对偶 \(I(\alpha)=\sup_{\theta}(\alpha\theta-P(\theta))\) 与 \(P(\theta)=\theta+P(-\theta)\) 即得 \(I(\alpha)=I(1-\alpha)\)。
最后奇阶累积量的结论由 \(\Lambda\) 偶性对 \(\theta=0\) 的导数给出，并与命题 \ref{prop:fluctuation-even} 的表述一致。证毕。
\end{proof}

\begin{corollary}[自对偶压力的斜率反演：\texorpdfstring{\(\alpha(\theta)+\alpha(-\theta)=1\)}{alpha(theta)+alpha(-theta)=1}]\label{cor:sync-kernel-gc-slope-involution}
在定理 \ref{thm:sync-kernel-gallavotti-cohen} 的设定下，若 \(\alpha(\theta):=P'(\theta)\) 在实轴上存在，则对所有实 \(\theta\) 成立
\[
\boxed{\ \alpha(\theta)+\alpha(-\theta)=1\ }.
\]
\end{corollary}
\begin{proof}
对恒等式 \(P(\theta)=\theta+P(-\theta)\) 两边求导，得到
\(
\alpha(\theta)=1-\alpha(-\theta)
\)，等价地即为所示对称性。证毕。
\end{proof}

\begin{proposition}[表示级自对偶：一维角色扭曲的函数方程]\label{prop:kernel-self-dual-character}
沿用 \S\ref{app:kernel-self-dual} 的记号。设 \(G\) 为有限阿贝尔群，\(\chi:G\to\CC^\times\) 为一维角色，且假设群标签只依赖输出位 \(e\in\{0,1\}\)：对 \(e=0\) 取标签为单位元、对 \(e=1\) 取某固定元 \(g_1\in G\)。
则相应的角色扭曲矩阵可写成
\[
B_\chi(u)=B_0+u\,\chi(g_1)\,B_1.
\]
在自对偶对称 \(\Pi^{-1}B_0\Pi=B_1,\ \Pi^{-1}B_1\Pi=B_0\) 下，有相似恒等式
\[
\boxed{\;
\Pi^{-1}B_\chi(u)\Pi
=u\,\chi(g_1)\,B_{\chi^{-1}}(1/u)\; }.
\]
因此行列式与 \(\zeta\)-函数满足表示级函数方程
\[
\boxed{\;
\Delta_\chi(z,u):=\det(I-zB_\chi(u))
=\Delta_{\chi^{-1}}\!\bigl(u\,\chi(g_1)\,z,\ 1/u\bigr)\; },
\]
\[
\boxed{\;
\zeta_\chi(z,u):=\Delta_\chi(z,u)^{-1}
=\zeta_{\chi^{-1}}\!\bigl(u\,\chi(g_1)\,z,\ 1/u\bigr)\; }.
\]
\end{proposition}
\begin{proof}
由 \(\Pi^{-1}B_0\Pi=B_1,\ \Pi^{-1}B_1\Pi=B_0\) 得
\[
\Pi^{-1}B_\chi(u)\Pi
=B_1+u\,\chi(g_1)\,B_0
=u\,\chi(g_1)\Bigl(B_0+\frac{1}{u}\chi(g_1)^{-1}B_1\Bigr)
=u\,\chi(g_1)\,B_{\chi^{-1}}(1/u).
\]
对两边取 \(\det(I-z\,\cdot)\) 并用 \(\det(I-z\alpha A)=\det(I-(\alpha z)A)\) 即得所示行列式函数方程；\(\zeta\) 版本由取倒数得到。证毕。
\end{proof}

\begin{remark}[表示级函数方程的二面体对称：\texorpdfstring{$(\chi,u,z)\mapsto(\chi^{-1},1/u,u\chi(g_1)z)$}{(chi,u,z)->(chi^{-1},1/u,u chi(g1) z)} 的闭合]\label{rem:kernel-self-dual-character-dihedral}
命题 \ref{prop:kernel-self-dual-character} 给出一个严格对合变换
\[
(\chi,u,z)\longmapsto\bigl(\chi^{-1},\,1/u,\,u\,\chi(g_1)\,z\bigr),
\qquad \iota^2=\mathrm{id},
\]
并满足 \(\Delta_\chi(z,u)=\Delta_{\chi^{-1}}(u\chi(g_1)z,1/u)\) 与 \(\zeta_\chi(z,u)=\zeta_{\chi^{-1}}(u\chi(g_1)z,1/u)\)。
当 \(|u|=1\) 且 \(|\chi(g_1)|=1\)（酉角色）时，尺度因子 \(u\chi(g_1)\) 的模长为 \(1\)，从而零点集合在模长意义下保持不变：\(\{\,|z|:\Delta_\chi(z,u)=0\,\}\) 与 \(\{\,|z|:\Delta_{\chi^{-1}}(z,1/u)=0\,\}\) 一致。
因此，对单位圆扭曲上的谱半径/prime-shadow 混合指数、以及任何由 \(\log\det(I-zB_\chi(u))\)（或其 Möbius--\(\log\zeta\) 常数项）函子化构造的异常签名，均在该对合下闭合。
特别地，全角色扫描可在商空间 \(\widehat G/\{\chi\sim\chi^{-1}\}\) 上完成而不丢失最坏情形。
\end{remark}

\begin{remark}[函数方程生成器与 Ramanujan 型谱界：核版 (G)RH 的统一母式]\label{rem:kernel-self-dual-character-ramanujan}
命题 \ref{prop:kernel-self-dual-character} 表明：一旦“状态级自对偶”可提升为表示级相似恒等式，则通道行列式 \(\Delta_\chi\) 的二变量函数方程自动生成。
因此对任意底 \(L>1\)，沿 \S\ref{subsec:xi-from-self-dual} 的 Dirichlet 截面定义
\[
\Xi_{\chi,L}(s):=\Delta_\chi(L^{-s},L^{2s-1}),
\]
即可得到（带 \(\chi\mapsto\chi^{-1}\) 的）严格函数方程与临界线实值化。
在这一层级上，真正需要额外输入的部分不再是“函数方程/完成化对称”，而是与核版 (G)RH 同型的\textbf{统一谱界}：
设 \(\chi_0\) 为平凡角色并记 \(\lambda:=\rho(B_{\chi_0}(1))\)。
则对所有非平凡 \(\chi\neq\chi_0\) 的 Ramanujan 型命题
\[
\max_{\mu\in\mathrm{spec}(B_\chi(1))}|\mu|\ \le\ \sqrt{\lambda}
\]
（或其在单位圆扭曲族上的版本）等价于“通道截面 \(\Xi_{\chi,L}\) 的非平凡零点/极点满足临界线约束”，并与正文定义 \ref{def:kernel-rh} 的核版 RH/GRH 判据在结构上完全一致：它把 (G)RH 的核心压缩为一个纯谱论不等式（参见命题 \ref{prop:finite-rh-triangle-dict} 的谱 \(\leftrightarrow\) 极点字典与注 \ref{rem:xi-artin-channel-pipeline} 的通道化完成化判据）。
\end{remark}

\begin{corollary}[符号扭曲 \texorpdfstring{$\chi(g_1)=-1$}{chi(g1)=-1} 的自对偶函数方程与 \texorpdfstring{$u=1$}{u=1} 的“纯奇部”读出]\label{cor:kernel-self-dual-sign-twist}
在命题 \ref{prop:kernel-self-dual-character} 的设定下，取 \(G=\ZZ/2\ZZ\) 且令 \(\chi(g_1)=-1\)。
记相应的符号扭曲矩阵为
\[
B_{-}(u):=B_0-uB_1.
\]
则有相似恒等式
\[
\boxed{\ \Pi^{-1}B_{-}(u)\Pi=-u\,B_{-}(1/u)\ },
\]
从而行列式与 \(\zeta\)-函数满足带符号的函数方程
\[
\boxed{\ \Delta_{-}(z,u):=\det(I-zB_{-}(u))=\Delta_{-}(-uz,1/u)\ },
\]
\[
\boxed{\ \zeta_{-}(z,u):=\Delta_{-}(z,u)^{-1}=\zeta_{-}(-uz,1/u)\ }.
\]
并且在 \(u=1\) 处有
\[
\boxed{\ B_{-}(1)=B_0-B_1=-2B_{\mathrm{asym}}\ },
\]
因此该符号扭曲在无偏切片上仅测量 \(\Pi\)-奇分量 \(B_{\mathrm{asym}}\)。
\end{corollary}

\begin{proof}
由命题 \ref{prop:kernel-self-dual-character} 取 \(\chi=\chi^{-1}\) 且 \(\chi(g_1)=-1\) 立得
\(
\Pi^{-1}B_{-}(u)\Pi=u\chi(g_1)B_{-}(1/u)=-uB_{-}(1/u)
\)
与对应的行列式函数方程。
最后一式由引理 \ref{lem:sync-kernel-even-odd-splitting} 的定义 \(B_{\mathrm{asym}}=(B_1-B_0)/2\) 直接得到。证毕。
\end{proof}

\begin{proposition}[周期迹多项式的回文对称]\label{prop:trace-palindrome}
记
\[
a_n(u):=\Tr(B(u)^n)=\sum_{k=0}^{n}a_{n,k}u^k.
\]
则对所有 $n\ge 1$ 有
\[
a_n(u)=u^n a_n(1/u)
\quad\Longleftrightarrow\quad
a_{n,k}=a_{n,n-k}.
\]
\end{proposition}

\begin{proof}
由自对偶 $\Pi^{-1}B(u)\Pi=uB(1/u)$ 与迹不变性，
\[
a_n(u)=\Tr(B(u)^n)=\Tr\!\bigl((uB(1/u))^n\bigr)=u^n\Tr(B(1/u)^n)=u^n a_n(1/u),
\]
比较系数即得结论。证毕。
\end{proof}

\begin{corollary}[奇阶消失律]\label{cor:trace-odd-vanish}
若 $n$ 为奇数，则
\[
a_n(-1)=0,
\qquad
(u+1)\mid a_n(u)\ \ \text{于}\ \mathbb{Z}[u].
\]
\end{corollary}
\begin{proof}
由命题 \ref{prop:trace-palindrome}，$a_n(u)=u^n a_n(1/u)$。取 $u=-1$ 得 $a_n(-1)=(-1)^n a_n(-1)$，奇数 $n$ 迫使 $a_n(-1)=0$，从而 $(u+1)\mid a_n(u)$。证毕。
\end{proof}

\begin{corollary}[周期层的中心矩恒等式：奇阶严格为零]\label{cor:trace-central-moments-zero}
沿用命题 \ref{prop:trace-palindrome} 的记号。对任意整数 \(m\ge 0\)，都有中心矩恒等式
$$
\boxed{\ \sum_{k=0}^{n}\left(k-\frac{n}{2}\right)^{2m+1}a_{n,k}=0\ }\qquad(\text{于 }\QQ).
$$
特别地，取 \(m=0\) 得
$$
\sum_{k=0}^{n}k\,a_{n,k}=\frac{n}{2}\sum_{k=0}^{n}a_{n,k}=\frac{n}{2}\,a_n(1).
$$
若进一步把系数正规化为 \(\pi^{\mathrm{per}}_{n,k}:=a_{n,k}/a_n(1)\)（当 \(a_n(1)\neq 0\)），则 \(\{\pi^{\mathrm{per}}_{n,k}\}_k\) 关于 \(n/2\) 精确对称，所有奇阶中心矩严格为零。
\end{corollary}
\begin{proof}
由命题 \ref{prop:trace-palindrome} 有 \(a_{n,k}=a_{n,n-k}\)。
对任意函数 \(f\) 满足 \(f(x)=-f(-x)\)（奇函数），把求和按 \(k\leftrightarrow n-k\) 配对得
$$
\sum_{k=0}^{n} f\!\left(k-\frac{n}{2}\right)a_{n,k}
=\frac12\sum_{k=0}^{n}\Bigl(
f\!\left(k-\frac{n}{2}\right)a_{n,k}
f\!\left((n-k)-\frac{n}{2}\right)a_{n,n-k}
\Bigr)=0,
$$
因为 \((n-k)-\frac{n}{2}=-(k-\frac{n}{2})\) 且 \(a_{n,n-k}=a_{n,k}\)。
取 \(f(x)=x^{2m+1}\) 即得所示恒等式；\(m=0\) 的线性式为其特例。证毕。
\end{proof}

\begin{corollary}[$\Delta(z,u)$ 的系数回文结构]\label{cor:delta-coeff-palindrome}
写
\[
\Delta(z,u)=\det(I-zB(u))=\sum_{j=0}^{10}c_j(u)\,z^j.
\]
则对每个 $j$ 都有
\[
c_j(u)=u^j c_j(1/u),
\]
即 $c_j(u)$ 为次数 $j$ 的回文多项式（完成化后等价为 $u^{j/2}$ 乘以偶函数）。
\end{corollary}

\begin{remark}[不变量表示：系数仅依赖 $u+u^{-1}$]
取 $u=e^\theta$，则 $u^{-j/2}c_j(u)$ 为 $\theta$ 的偶函数，因而可写成 $u+u^{-1}$ 的多项式；等价地，完成化后 $\widehat{\Delta}(w,u)$ 的系数落在不变量子环 $\QQ[u+u^{-1}]$ 中。
\end{remark}

\begin{corollary}[完成化系数的“临界线实值性”]\label{cor:completion-real-line}
取 $u=e^{it}$（$|u|=1$），并定义
\[
d_j(u):=u^{-j/2}c_j(u).
\]
则
\[
\overline{d_j(u)}=d_j(u),
\]
即完成化后的系数在单位圆上为实数。
\end{corollary}
