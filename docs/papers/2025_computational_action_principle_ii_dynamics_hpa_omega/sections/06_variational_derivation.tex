\section{Field equations from variation}
\label{sec:variational_derivation}

We record the dynamical equations implied by the action \eqref{eq:omega_action_cap2}.
Full variational details are deferred to Appendix~\ref{app:variation_details}.

\subsection{Metric variation and the information stress tensor}
\label{subsec:metric_variation}

\begin{theorem}[Einstein equation with information stress]
\label{thm:einstein_with_info}
Varying \eqref{eq:omega_action_cap2} with respect to the metric yields
\begin{equation}
  G_{\mu\nu}+\Lambda g_{\mu\nu}
  = 8\pi G\left(T^{(m)}_{\mu\nu}+T^{\mathrm{info}}_{\mu\nu}\right),
  \label{eq:einstein_info_cap2}
\end{equation}
where $T^{(m)}_{\mu\nu}$ is the matter stress tensor and
\begin{equation}
  T^{\mathrm{info}}_{\mu\nu}
  = 2\lambda_F\left(\nabla_\mu\chi\,\nabla_\nu\chi-\frac12 g_{\mu\nu}(\nabla\chi)^2\right)
  - g_{\mu\nu}V(\chi^2).
  \label{eq:info_stress_cap2}
\end{equation}
\end{theorem}

\subsection{Amplitude equation and conservation}
\label{subsec:chi_equation}

\begin{proposition}[Fisher-amplitude equation]
\label{prop:chi_equation}
Varying \eqref{eq:omega_action_cap2} with respect to $\chi$ yields
\begin{equation}
  2\lambda_F\,\square\chi-\frac{\dd V}{\dd\chi}=0.
  \label{eq:chi_eom_cap2}
\end{equation}
\end{proposition}

By diffeomorphism invariance one has $\nabla^\mu G_{\mu\nu}=0$, hence the total stress tensor is covariantly conserved:
\begin{equation}
  \nabla^\mu\left(T^{(m)}_{\mu\nu}+T^{\mathrm{info}}_{\mu\nu}\right)=0.
  \label{eq:stress_conservation_cap2}
\end{equation}
When \eqref{eq:chi_eom_cap2} holds and the matter sector is covariantly conserved (or exchanges flow through explicit couplings in $\mathcal{L}_m$), the split into matter and information sectors is consistent.

\subsection{\texorpdfstring{$\kappa$}{kappa}-formulation under the cost-to-density map}
\label{subsec:kappa_formulation}

The effective dynamics becomes explicitly closed in terms of the auditable routing overhead once one fixes the cost-to-density identification (Assumption~\ref{ass:cost_to_density}) and a potential $V$.
Write
\begin{equation}
  u(x,t):=\log\!\left(\frac{\kappa(x,t)}{\kappa_0}\right),
  \qquad
  \rho=\rho_0\,\e^{p u},
  \qquad
  \chi=\sqrt{\rho_0}\,\e^{\frac{p}{2}u}.
  \label{eq:u_rho_chi_kappa}
\end{equation}

\begin{proposition}[Closed scalar equation for $u=\log(\kappa/\kappa_0)$]
\label{prop:u_equation}
Assume Assumptions~\ref{ass:cost_to_density}--\ref{ass:fisher_amplitude}.
Then the Fisher-amplitude equation \eqref{eq:chi_eom_cap2} is equivalent to
\begin{equation}
  \lambda_F\,p\,\square u + \lambda_F\,\frac{p^2}{2}\,(\nabla u)^2
  = \frac{1}{\chi}\,\frac{\dd V}{\dd\chi},
  \qquad \chi=\sqrt{\rho_0}\,\e^{\frac{p}{2}u},
  \label{eq:u_equation}
\end{equation}
where $(\nabla u)^2:=g^{\mu\nu}\nabla_\mu u\nabla_\nu u$.
\begin{proof}
This is a direct chain-rule substitution for $\chi=\sqrt{\rho_0}\,\e^{\frac{p}{2}u}$; see Appendix~\ref{app:u_chain_rule}.
\end{proof}
\end{proposition}

\begin{proposition}[Potential-derivative reconstruction from $u$]
\label{prop:potential_reconstruction}
Assume $V$ depends only on $\rho=\chi^2$ (Assumption~\ref{ass:fisher_amplitude}), so $V(\chi^2)=\widehat V(\rho)$.
Then \eqref{eq:u_equation} is equivalently
\begin{equation}
  \widehat V'(\rho)
  = \frac12\left(\lambda_F\,p\,\square u + \lambda_F\,\frac{p^2}{2}\,(\nabla u)^2\right),
  \qquad \rho=\rho_0\,\e^{p u},
  \label{eq:Vprime_reconstruction}
\end{equation}
where $\widehat V'(\rho)=\dd\widehat V/\dd\rho$.
In particular, given a solution $(g,u)$ one can reconstruct $\widehat V'$ along the image of $\rho$.
\end{proposition}

\begin{remark}[Normalization of $\widehat V$]
\label{rem:V_normalization}
The additive constant in $\widehat V$ is not physically observable at the level of equations of motion: it shifts the effective cosmological term by a constant.
Accordingly, one may impose a normalization condition such as $\widehat V(\rho_0)=0$ without loss of generality, absorbing any constant offset into $\Lambda$.
\end{remark}

\begin{proposition}[Information stress in terms of $u$]
\label{prop:info_stress_u}
With the same identifications, the information stress tensor \eqref{eq:info_stress_cap2} can be written as
\begin{equation}
  T^{\mathrm{info}}_{\mu\nu}
  = \lambda_F\,\frac{p^2}{2}\,\chi^2
  \left(\nabla_\mu u\,\nabla_\nu u-\frac12 g_{\mu\nu}(\nabla u)^2\right)
  - g_{\mu\nu}V(\chi^2),
  \qquad \chi=\sqrt{\rho_0}\,\e^{\frac{p}{2}u}.
  \label{eq:info_stress_u}
\end{equation}
\end{proposition}

\paragraph{Quantitative closure and fitting viewpoint.}
Equations \eqref{eq:einstein_info_cap2} and \eqref{eq:u_equation} form a closed covariant system for $(g_{\mu\nu},u)$ once $(p,\rho_0,\lambda_F,V)$ and the matter sector are specified.
Operationally, $\kappa$ is measured (by compilation logs or via the scattering proxy in Section~\ref{sec:wigner_smith}), hence $u$ is data.
CAP-II uses this to impose a self-consistency fit: parameters are chosen so that the solution's lapse (in a chosen gauge) matches the operational lapse $\lapse=\kappa_0/\kappa$ after coarse-graining.

\subsection{Minimal potential families and long-range behavior}
\label{subsec:potential_families}

To make the closure predictive, one must specify a family for the local potential $V(\chi^2)$.
A minimal requirement is \emph{stability} of the reference background $u=0$ (equivalently $\kappa=\kappa_0$), which suggests that $V$ has a local minimum at $\chi^2=\rho_0$.
One convenient parametrization is to write $V(\chi^2)=\widehat V(\rho)$ with $\rho=\chi^2$ and expand around $\rho_0$:
\begin{equation}
  \widehat V(\rho)
  = \widehat V(\rho_0) + \frac{m_\rho^2}{2}(\rho-\rho_0)^2 + O\!\left((\rho-\rho_0)^3\right),
  \label{eq:V_rho_expansion}
\end{equation}
where $m_\rho$ sets an effective stiffness scale.

\begin{remark}[Massless versus massive regime]
\label{rem:massless_massive_regime}
In the linearized regime $|u|\ll 1$ one has $\rho-\rho_0 \approx \rho_0 p\,u$.
If $m_\rho>0$, the scalar sector has an effective correlation length (Compton scale) and deviations sourced through $u$ are suppressed beyond that scale.
If instead the quadratic term vanishes (effectively massless), then static spherically symmetric solutions generically carry scalar hair and differ from Schwarzschild; see the classical Fisher/JNW family for Einstein gravity coupled to a massless scalar \cite{Fisher1948,JanisNewmanWinicour1968}.
CAP-II treats the Schwarzschild lapse as a benchmark fit target in a regime where scalar backreaction is either negligible or short-ranged compared to the exterior window; deviations can be used to constrain the potential family.
\end{remark}
