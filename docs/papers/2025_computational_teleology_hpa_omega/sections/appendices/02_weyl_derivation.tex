\section{From Weyl pairs to resource complementarity: inequality and operational meaning}
\label{app:weyl_derivation}

We record the minimal objects behind the variance-form uncertainty inequality used in Theorem~\ref{thm:weyl_uncertainty}.
The inequality (in essentially this form) is due to Massar--Spindel \cite{MassarSpindel2008} for unitary operators obeying a Weyl commutation relation; for completeness we include a self-contained derivation (following their proof of Theorem~1 in \cite{MassarSpindel2008}).

\paragraph{Weyl relation.}
Let $U,V$ be unitaries satisfying $UV=\e^{\iu\Phi}VU$.
Nontrivial $\Phi$ implies $U$ and $V$ cannot be simultaneously diagonalized.

\paragraph{Unitary variances.}
For a normalized state $|\psi\rangle$, define
\begin{equation}
\Delta_U^2=1-|\langle U\rangle|^2,\qquad \Delta_V^2=1-|\langle V\rangle|^2,
\end{equation}
where $\langle U\rangle:=\langle\psi|U|\psi\rangle$.
These vanish exactly on eigenstates.

\subsection{Proof of Theorem~\ref{thm:weyl_uncertainty} (Massar--Spindel inequality)}
\label{app:weyl_proof}
We assume the phase is chosen so that $0\le \Phi\le \pi$ (this can always be arranged without changing $\Delta_U,\Delta_V$ by replacing $U$ or $V$ with its adjoint, which flips $\Phi\mapsto-\Phi$ modulo $2\pi$).
Set
\begin{equation}
A:=\tan(\Phi/2)\ \ge\ 0.
\end{equation}

\paragraph{Sine/cosine operators.}
Introduce the Hermitian operators
\begin{equation}
C_U:=\frac{U+U^{\dagger}}{2},\quad S_U:=\frac{U-U^{\dagger}}{2\iu},
\qquad
C_V:=\frac{V+V^{\dagger}}{2},\quad S_V:=\frac{V-V^{\dagger}}{2\iu}.
\label{eq:weyl_sine_cosine}
\end{equation}
They satisfy $C_U^2+S_U^2=\mathds{1}$ and $C_V^2+S_V^2=\mathds{1}$.
Moreover,
\begin{equation}
\Delta_U^2=\Delta C_U^2+\Delta S_U^2,\qquad \Delta_V^2=\Delta C_V^2+\Delta S_V^2,
\end{equation}
where $\Delta X^2:=\langle X^2\rangle-\langle X\rangle^2$ for Hermitian $X$.

\paragraph{Robertson reduction.}
Since $\Delta_U^2\ge \Delta S_U^2$ and $\Delta_V^2\ge \Delta S_V^2$, one has
\begin{equation}
\Delta_U^2\Delta_V^2\ \ge\ \Delta S_U^2\,\Delta S_V^2.
\end{equation}
By the Robertson uncertainty relation, for any Hermitian $A,B$,
\begin{equation}
\Delta A\,\Delta B\ \ge\ \frac12\bigl|\langle[A,B]\rangle\bigr|,
\end{equation}
and therefore
\begin{equation}
\Delta_U^2\Delta_V^2\ \ge\ \frac14\bigl|\langle[S_U,S_V]\rangle\bigr|^2.
\label{eq:weyl_robertson_step}
\end{equation}

\begin{lemma}[Sine-commutator identity {\cite[Proof of Theorem~1]{MassarSpindel2008}}]
\label{lem:weyl_sine_commutator}
If $U,V$ satisfy $UV=\e^{\iu\Phi}VU$ with $0\le\Phi\le\pi$ and $A=\tan(\Phi/2)$, then
\begin{equation}
[S_U,S_V]=-\iu A\,(C_UC_V+C_VC_U).
\label{eq:weyl_sine_commutator}
\end{equation}
\end{lemma}

\begin{proof}
Using \eqref{eq:weyl_sine_cosine} one expands
\begin{align}
S_U S_V
&=-\frac14\,(UV+U^{\dagger}V^{\dagger})+\frac14\,(U^{\dagger}V+UV^{\dagger}),\\
S_V S_U
&=-\frac14\,(VU+V^{\dagger}U^{\dagger})+\frac14\,(V^{\dagger}U+VU^{\dagger}),\\
C_U C_V
&=\frac14\,(UV+U^{\dagger}V^{\dagger})+\frac14\,(U^{\dagger}V+UV^{\dagger}),\\
C_V C_U
&=\frac14\,(VU+V^{\dagger}U^{\dagger})+\frac14\,(V^{\dagger}U+VU^{\dagger}).
\end{align}
Using $UV=\e^{\iu\Phi}VU$ and $U^{\dagger}V=VU^{\dagger}\e^{-\iu\Phi}$ to rewrite terms yields two equivalent forms:
\begin{align}
[S_U,S_V]
&=-\iu\sin\Phi\, C_V C_U-2\sin^2(\Phi/2)\,S_VS_U,\\
[S_U,S_V]
&=-\iu\sin\Phi\, C_U C_V+2\sin^2(\Phi/2)\,S_US_V.
\end{align}
Combining them gives
\begin{equation}
[S_U,S_V]=-\iu\,\frac{\sin\Phi}{2\cos^2(\Phi/2)}\,(C_UC_V+C_VC_U)
\ =\ -\iu\tan(\Phi/2)\,(C_UC_V+C_VC_U),
\end{equation}
as claimed.
\end{proof}

\begin{lemma}[Lower bound on $\langle C_UC_V\rangle$ {\cite[Proof of Theorem~1]{MassarSpindel2008}}]
\label{lem:weyl_cucu_bound}
By rephasing $U\mapsto \e^{-\iu\arg\langle U\rangle}U$ and $V\mapsto \e^{-\iu\arg\langle V\rangle}V$ one may assume $\langle U\rangle,\langle V\rangle\in\RR_{\ge 0}$, hence $\langle S_U\rangle=\langle S_V\rangle=0$ and $\langle C_U\rangle=\sqrt{1-\Delta_U^2}$, $\langle C_V\rangle=\sqrt{1-\Delta_V^2}$.
With this choice,
\begin{equation}
\bigl|\langle C_UC_V\rangle\bigr|\ \ge\ \sqrt{1-\Delta_U^2}\,\sqrt{1-\Delta_V^2}-\Delta_U\Delta_V.
\label{eq:cucu_bound}
\end{equation}
\end{lemma}

\begin{proof}
Write
$C_U|\psi\rangle=x_U|\psi\rangle+y_U|\psi^{\perp}\rangle$
where $|\psi^{\perp}\rangle$ is normalized and orthogonal to $|\psi\rangle$, and $x_U=\langle C_U\rangle$.
Since $\|C_U\|\le 1$, one has $x_U^2+y_U^2=\langle\psi|C_U^2|\psi\rangle\le 1$, hence
$y_U^2\le 1-x_U^2=\Delta_U^2$.
Similarly,
$C_V|\psi\rangle=x_V|\psi\rangle+y_V|\psi'^{\perp}\rangle$
with $x_V=\langle C_V\rangle$ and $y_V^2\le \Delta_V^2$.
Therefore
\begin{align}
\bigl|\langle C_VC_U\rangle\bigr|
&=\bigl|x_Ux_V+y_Uy_V\langle\psi'^{\perp}|\psi^{\perp}\rangle\bigr|\\
&\ge x_Ux_V-|y_Uy_V|\\
&\ge \sqrt{1-\Delta_U^2}\,\sqrt{1-\Delta_V^2}-\Delta_U\Delta_V,
\end{align}
which implies \eqref{eq:cucu_bound}.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:weyl_uncertainty}]
Using Lemma~\ref{lem:weyl_sine_commutator} in \eqref{eq:weyl_robertson_step} gives
\begin{equation}
\Delta_U\Delta_V\ \ge\ A\,\bigl|\langle C_UC_V\rangle\bigr|.
\end{equation}
Applying Lemma~\ref{lem:weyl_cucu_bound} yields
\begin{equation}
\Delta_U\Delta_V\ \ge\ A\Bigl(\sqrt{1-\Delta_U^2}\,\sqrt{1-\Delta_V^2}-\Delta_U\Delta_V\Bigr).
\label{eq:weyl_pre_squaring}
\end{equation}
Rearranging and squaring \eqref{eq:weyl_pre_squaring} gives \eqref{eq:weyl_variance_uncertainty}.
\end{proof}

\paragraph{Resource reading.}
In the scan model, $U=\Theta$ encodes access order (time), while $V$ encodes phase/readout channels (space).
The inequality asserts that sharpening readout localization (small $\Delta_V$) forces scan delocalization (large $\Delta_U$), and vice versa.
This is the formal kernel of time--space resource complementarity.

\subsection{Operational meaning: overlap, trace distance, and binary decision error}
\label{app:weyl_operational}
The variance quantities $\Delta_U,\Delta_V$ admit a direct operational meaning in quantum hypothesis testing.
Let $\rho:=|\psi\rangle\langle\psi|$ be a pure state and define the unitary conjugates
\begin{equation}
\rho_U:=U\rho U^{\dagger},\qquad \rho_V:=V\rho V^{\dagger}.
\end{equation}
Write the trace distance as $D(\rho,\sigma):=\tfrac12\|\rho-\sigma\|_1$, where $\|X\|_1:=\Tr\sqrt{X^{\dagger}X}$ \cite{Watrous2018}.

\begin{lemma}[Unitary variance equals trace distance for pure states]
\label{lem:variance_trace_distance}
For any normalized $|\psi\rangle$ and unitary $U$,
\begin{equation}
D(\rho,\rho_U)=\sqrt{1-\bigl|\langle\psi|U|\psi\rangle\bigr|^2}=\Delta_U.
\end{equation}
The analogous identity holds with $U$ replaced by $V$.
\end{lemma}

\begin{proof}
For pure states $\rho=|\psi\rangle\langle\psi|$ and $\sigma=|\phi\rangle\langle\phi|$ one has the standard formula
$D(\rho,\sigma)=\sqrt{1-|\langle\psi|\phi\rangle|^2}$ \cite{Watrous2018}.
Taking $|\phi\rangle:=U|\psi\rangle$ yields $D(\rho,\rho_U)=\sqrt{1-|\langle\psi|U|\psi\rangle|^2}$.
\end{proof}

\begin{corollary}[Helstrom error for distinguishing a unitary conjugate]
\label{cor:helstrom_error_unitary_kick}
Consider binary discrimination between $\rho$ and $\rho_U$ with equal prior probabilities.
The optimal (minimum) error probability is given by the Helstrom formula \cite{Helstrom1976,Watrous2018}
\begin{equation}
p_{\mathrm{err}}^{*}(U;\psi)=\frac12\left(1-D(\rho,\rho_U)\right)=\frac12\left(1-\Delta_U\right).
\label{eq:helstrom_error_deltaU}
\end{equation}
Equivalently, $\Delta_U=1-2p_{\mathrm{err}}^{*}(U;\psi)$.
The same statement holds for $V$ with $\Delta_V$.
\end{corollary}

Thus, $\Delta_U$ measures how distinguishable successive scan states are (one tick vs one tick plus a scan shift), while $\Delta_V$ measures how distinguishable the phase/readout channel is under conjugation by the phase operator.

\subsection{A task-level tradeoff induced by the Weyl inequality}
\label{app:weyl_task_tradeoff}
Combine Theorem~\ref{thm:weyl_uncertainty} with Corollary~\ref{cor:helstrom_error_unitary_kick}.
Define the optimal equal-prior error probabilities
\begin{equation}
p_U:=p_{\mathrm{err}}^{*}(U;\psi),\qquad p_V:=p_{\mathrm{err}}^{*}(V;\psi).
\end{equation}
Then $\Delta_U=1-2p_U$ and $\Delta_V=1-2p_V$.

\begin{proposition}[Binary-decision tradeoff for a Weyl pair]
\label{prop:weyl_binary_tradeoff}
Let $U,V$ satisfy $UV=\e^{\iu\Phi}VU$, with the phase chosen so that $0\le\Phi\le\pi$, and set $A=\tan(\Phi/2)$.
For any normalized state $|\psi\rangle$, the optimal discrimination errors $p_U,p_V$ satisfy
\begin{equation}
(1+2A)\,(1-2p_U)^2(1-2p_V)^2 + A^2\bigl((1-2p_U)^2+(1-2p_V)^2\bigr)\ \ge\ A^2.
\label{eq:weyl_binary_tradeoff}
\end{equation}
Equivalently, in terms of $\Delta_U,\Delta_V$ the inequality is exactly \eqref{eq:weyl_variance_uncertainty}.
\end{proposition}

\begin{remark}[How this functions as a resource lower bound]
\label{rem:weyl_tradeoff_resource_reading}
Equation~\eqref{eq:weyl_binary_tradeoff} is a concrete, task-level statement: one cannot simultaneously make both unitary conjugations $U$ and $V$ ``almost invisible'' to the state (i.e.\ $p_U\approx p_V\approx \tfrac12$).
In the scan--readout semantics, making the readout channel sharp pushes the state toward a regime where the $V$-conjugation is hard to detect (small $\Delta_V$), which forces scan shifts to be detectable (large $\Delta_U$), and vice versa.
This tradeoff underlies the impossibility of jointly optimizing access (scan) and distinction (readout) in a single state preparation.
\end{remark}

\subsection{From distinguishability to a sample-complexity lower bound}
\label{app:weyl_sample_complexity}
The preceding identities are single-shot statements.
To translate them into a time-like resource, one may consider repeating the same discrimination task with $n$ identical copies (or, equivalently, $n$ independent repetitions of state preparation and measurement).

\begin{proposition}[Helstrom error for $n$ copies of pure states]
\label{prop:helstrom_n_copies_pure}
Let $\rho_0=|\psi_0\rangle\langle\psi_0|$ and $\rho_1=|\psi_1\rangle\langle\psi_1|$ be pure states with equal priors.
For $n$ copies, the optimal (minimum) error probability for distinguishing $\rho_0^{\otimes n}$ from $\rho_1^{\otimes n}$ is
\begin{equation}
p_{\mathrm{err}}^{*(n)}(\psi_0,\psi_1)=\frac12\left(1-\sqrt{1-|\langle\psi_0|\psi_1\rangle|^{2n}}\right).
\label{eq:helstrom_n_copies_pure}
\end{equation}
\end{proposition}

\begin{proof}
The Helstrom formula gives $p_{\mathrm{err}}^{*(n)}=\tfrac12(1-D(\rho_0^{\otimes n},\rho_1^{\otimes n}))$ \cite{Helstrom1976,Watrous2018}.
For pure states one has $D(\rho,\sigma)=\sqrt{1-F(\rho,\sigma)}$ with $F(\rho,\sigma)=|\langle\psi|\phi\rangle|^2$ \cite{Watrous2018}.
Since $|\langle\psi_0^{\otimes n}|\psi_1^{\otimes n}\rangle|^2=|\langle\psi_0|\psi_1\rangle|^{2n}$, the claim follows.
\end{proof}

\begin{corollary}[Copies needed to detect a unitary conjugation]
\label{cor:copies_for_unitary_kick}
Let $\rho=|\psi\rangle\langle\psi|$ and $\rho_U=U\rho U^{\dagger}$ with overlap $F_U:=|\langle\psi|U|\psi\rangle|^2=1-\Delta_U^2$.
For any target error $\epsilon\in(0,\tfrac12)$, achieving $p_{\mathrm{err}}^{*(n)}(U;\psi)\le\epsilon$ requires
\begin{equation}
n\ \ge\ \frac{\log\!\bigl(1/(4\epsilon(1-\epsilon))\bigr)}{-\log F_U}
\ =\ \frac{\log\!\bigl(1/(4\epsilon(1-\epsilon))\bigr)}{-\log(1-\Delta_U^2)}.
\label{eq:copies_lower_bound_deltaU}
\end{equation}
If $\Delta_U=0$ (equivalently $F_U=1$), then $\rho=\rho_U$ and no finite $n$ can reduce the optimal error below $1/2$.
In particular, for $\Delta_U^2\ll 1$ one has $-\log(1-\Delta_U^2)=\Delta_U^2+O(\Delta_U^4)$, so
\begin{equation}
n\ \gtrsim\ \frac{\log(1/\epsilon)}{\Delta_U^2}.
\label{eq:copies_asymptotic_deltaU}
\end{equation}
\end{corollary}

\begin{proof}
By \eqref{eq:helstrom_n_copies_pure}, the condition $p_{\mathrm{err}}^{*(n)}\le\epsilon$ is equivalent to
$|\langle\psi|U|\psi\rangle|^{2n}\le 4\epsilon(1-\epsilon)$.
Taking logarithms yields \eqref{eq:copies_lower_bound_deltaU}.
The asymptotic expansion uses $-\log(1-x)=x+O(x^2)$.
\end{proof}

Corollary~\ref{cor:copies_for_unitary_kick} makes the ``small $\Delta_U$ implies expensive time'' intuition precise for a canonical task:
if a scan shift is almost invisible to the state (small $\Delta_U$), then the number of independent repetitions required to achieve error $\le\epsilon$ must grow at least on the order of $\log(1/\epsilon)/\Delta_U^2$.
