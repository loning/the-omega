\begin{conclusion}[Compressed multiplication operator: spectrum \texorpdfstring{$\{b_m\}$}{b\_m} with minimal polynomial \texorpdfstring{$y_{+}$}{y+}]\label{con:xi-terminal-wronskian-compressed-multiplication-spectrum-minpoly}
Following the notation of Conclusion~\ref{con:xi-terminal-wronskian-kernel-discrete-mercer-parseval}, let \(y_{+}(z)=\prod_{m=0}^{\kappa}(z-b_m)\) where \(b_m\in\RR\) are pairwise distinct, and define a linear operator \(M:\mathcal{H}_K\to\mathcal{H}_K\) by
\[
(Mf)(x):=\sum_{m=0}^{\kappa} b_m\,f(b_m)\,\frac{K(x,b_m)}{K(b_m,b_m)}.
\]
Then:
\begin{enumerate}
  \item \(M\) is self-adjoint and diagonalizable;
  \item Let \(e_m(x):=K(x,b_m)/\sqrt{K(b_m,b_m)}\); then \(\{e_m\}_{m=0}^{\kappa}\) is an orthonormal basis for \(\mathcal{H}_K\) and \(Me_m=b_m e_m\);
  \item The spectrum of \(M\) is exactly \(\{b_m\}_{m=0}^{\kappa}\), and the minimal polynomial of \(M\) equals \(y_{+}\), hence \(y_{+}(M)=0\).
\end{enumerate}
\end{conclusion}
\begin{proof}
By the Parseval structure of Conclusion~\ref{con:xi-terminal-wronskian-kernel-discrete-mercer-parseval}, \(\{K(\cdot,b_m)\}\) is an orthogonal basis for \(\mathcal{H}_K\), with
\(K(b_n,b_m)=0\) (\(n\neq m\)) and \(K(b_m,b_m)>0\).
Computing on the basis vector \(K(\cdot,b_m)\):
\[
M K(\cdot,b_m)
=\sum_{m'=0}^{\kappa} b_{m'}\,K(b_{m'},b_m)\,\frac{K(\cdot,b_{m'})}{K(b_{m'},b_{m'})}
=b_m\,K(\cdot,b_m),
\]
hence \(K(\cdot,b_m)\) (equivalently \(e_m\)) is an eigenvector with eigenvalue \(b_m\); normalization yields an orthonormal basis \(\{e_m\}\).
Thus \(M\) is diagonalized in this basis as \(\mathrm{diag}(b_0,\dots,b_\kappa)\), hence self-adjoint and diagonalizable with spectrum \(\{b_m\}\).
Finally, \(y_{+}\) is monic with zero set \(\{b_m\}\), so \(y_{+}(M)=0\); since \(\{b_m\}\) are distinct, the minimal polynomial must contain all linear factors \(z-b_m\), hence equals \(y_{+}\).
\end{proof}

\begin{conclusion}[Complete closed form of Weyl function: \texorpdfstring{$m(z)=-y/y_{+}$}{m(z)=-y/y+}]\label{con:xi-terminal-wronskian-weyl-function-closed-form}
Following the setting of Conclusion~\ref{con:xi-terminal-wronskian-compressed-multiplication-spectrum-minpoly}, let
\(
\psi:=y/\sqrt{\pi}\in\mathcal{H}_K
\).
For \(z\in\CC\setminus\{b_0,\dots,b_\kappa\}\), define the Weyl function
\[
m(z):=\langle (M-zI)^{-1}\psi,\psi\rangle_{\mathcal{H}_K}.
\]
Then the strict identity holds
\[
\boxed{\;m(z)=-\frac{y(z)}{y_{+}(z)}\;}
\]
and equivalently admits the complete partial fraction expansion
\[
-\frac{y(z)}{y_{+}(z)}=\sum_{m=0}^{\kappa}\frac{c_m}{b_m-z},
\qquad
c_m:=\frac{y(b_m)}{y_{+}'(b_m)}=\frac{y(b_m)^2}{A(b_m)}.
\]
\end{conclusion}
\begin{proof}
In the orthonormal eigenbasis \(\{e_m\}\) of Conclusion~\ref{con:xi-terminal-wronskian-compressed-multiplication-spectrum-minpoly},
\((M-zI)^{-1}e_m=(b_m-z)^{-1}e_m\), hence
\[
m(z)=\sum_{m=0}^{\kappa}\frac{|\langle \psi,e_m\rangle|^2}{b_m-z}.
\]
By the reproducing property \(f(b_m)=\langle f,K(\cdot,b_m)\rangle\) and \(e_m=K(\cdot,b_m)/\sqrt{K(b_m,b_m)}\), we have
\(
\langle \psi,e_m\rangle=\psi(b_m)/\sqrt{K(b_m,b_m)}
\).
Since \(\psi(b_m)=y(b_m)/\sqrt{\pi}\) and
\(
K(b_m,b_m)=A(b_m)/\pi=y(b_m)y_{+}'(b_m)/\pi
\),
we obtain
\[
|\langle \psi,e_m\rangle|^2
=\frac{y(b_m)^2/\pi}{A(b_m)/\pi}
=\frac{y(b_m)^2}{A(b_m)}
=\frac{y(b_m)}{y_{+}'(b_m)}
=:c_m.
\]
Thus \(m(z)=\sum_m c_m/(b_m-z)\).

Since \(m(z)=O(z^{-1})\) as \(z\to\infty\) and its only poles are simple poles at \(\{b_m\}\),
we may write \(m(z)=P(z)/y_{+}(z)\) with \(\deg P\le \kappa\).
Comparing the residue at \(z=b_m\):
\[
\frac{P(b_m)}{y_{+}'(b_m)}=\operatorname*{Res}_{z=b_m}m(z)=-c_m=-\frac{y(b_m)}{y_{+}'(b_m)},
\]
hence \(P(b_m)=-y(b_m)\) for all \(m\).
Since both \(P\) and \(-y\) are polynomials of degree \(\le\kappa\) agreeing at \(\kappa+1\) distinct points \(\{b_m\}\), they must coincide: \(P\equiv -y\), i.e., \(m(z)=-y(z)/y_{+}(z)\).
\end{proof}

\begin{conclusion}[Positive spectral weights and moment identities: \texorpdfstring{$\{c_m\}$}{c\_m} is a probability weight encoding the expansion of \texorpdfstring{$-y/y_{+}$}{-y/y+} at infinity]\label{con:xi-terminal-wronskian-spectral-weights-moment-identities}
Following the notation of Conclusion~\ref{con:xi-terminal-wronskian-weyl-function-closed-form},
\begin{enumerate}
  \item \(c_m>0\) for all \(m\), and \(\sum_{m=0}^{\kappa}c_m=1\);
  \item If
  \[
  -\frac{y(z)}{y_{+}(z)}=-\frac{1}{z}-\sum_{n\ge 1}\frac{\mu_n}{z^{n+1}}
  \qquad (z\to\infty),
  \]
  then each moment satisfies \(\mu_n=\sum_{m=0}^{\kappa}c_m b_m^{n}\) (\(n\ge 1\));
  \item If
  \[
  y(z)=z^{\kappa}+y_1 z^{\kappa-1}+y_2 z^{\kappa-2}+O(z^{\kappa-3}),\quad
  y_{+}(z)=z^{\kappa+1}+y_{+1}z^{\kappa}+y_{+2}z^{\kappa-1}+O(z^{\kappa-2}),
  \]
  then
  \[
  \sum_{m=0}^{\kappa}c_m b_m=y_1-y_{+1},\qquad
  \sum_{m=0}^{\kappa}c_m b_m^{2}=y_2-y_{+2}-y_1y_{+1}+y_{+1}^{2}.
  \]
\end{enumerate}
\end{conclusion}
\begin{proof}
By the Wronskian identity \(A=yy_{+}'-y'y_{+}\), we have
\[
w'(x)=\left(\frac{y_{+}}{y}\right)'(x)=\frac{A(x)}{y(x)^2}>0,
\qquad x\in\RR\setminus\{x_j\}.
\]
At \(x=b_m\), since \(y_{+}(b_m)=0\), we have
\(
w'(b_m)=y_{+}'(b_m)/y(b_m)>0
\), hence \(c_m=y(b_m)/y_{+}'(b_m)>0\).

On the other hand, by the partial fraction expansion of Conclusion~\ref{con:xi-terminal-wronskian-weyl-function-closed-form},
\[
\sum_{m=0}^{\kappa}\frac{c_m}{b_m-z}
=-\frac{1}{z}\sum_{m}c_m-\frac{1}{z^2}\sum_{m}c_m b_m-\cdots .
\]
Since \(-y(z)/y_{+}(z)=-z^{-1}+O(z^{-2})\) as \(z\to\infty\), comparing the \(z^{-1}\) coefficient yields \(\sum_m c_m=1\).
Further comparing general coefficients yields \(\mu_n=\sum_m c_m b_m^{n}\).

Finally, performing polynomial long division on \(-y/y_{+}\) yields
\[
-\frac{y(z)}{y_{+}(z)}
=-\frac{1}{z}-\frac{y_1-y_{+1}}{z^2}
-\frac{y_2-y_{+2}-y_1y_{+1}+y_{+1}^2}{z^3}+O(z^{-4}),
\]
and comparing with the moment expansion yields the two explicit formulas.
\end{proof}
