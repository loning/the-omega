\subsection{\texorpdfstring{$S_4$}{S4} Hurwitz 分块核、两换位碰撞边界与稳定退化 Jacobian 的显式结构}\label{subsec:conclusion-s4-hurwitz-block-kernel-boundary-cusp-stable-jacobian}
沿用结论 \ref{con:xi-terminal-zm-s4-nielsen-in-160-b5-transitive}、
\ref{con:xi-terminal-zm-s4-to-s3-nielsen-40-blocks-of-4}、
\ref{con:xi-terminal-zm-braid-monodromy-psp4f3-identification} 与命题
\ref{prop:xi-terminal-zm-hurwitz-generator-cycle-types-on-160-and-40} 的记号。
记
\[
\Gamma_{S_4}\leq \mathfrak S_{160},\qquad
\Gamma_{S_3}\leq \mathfrak S_{40},
\]
并令
\[
K:=\ker\!\bigl(\Gamma_{S_4}\twoheadrightarrow \Gamma_{S_3}\bigr).
\]

\begin{theorem}[块核 Sylow--\texorpdfstring{$3$}{3} 的规范分裂]\label{thm:conclusion-s4-block-kernel-sylow3-canonical-splitting}
在 \(160\) 点分解
\[
\bigsqcup_{j=1}^{40}B_j,\qquad |B_j|=4
\]
下，块核 \(K\) 的 Sylow--\(3\) 子群 \(P_3\) 满足
\[
P_3\cong (\ZZ/3\ZZ)^{40},
\]
并且存在唯一分解
\[
P_3=\prod_{j=1}^{40}P_{3,j},
\]
其中每个 \(P_{3,j}\cong \ZZ/3\ZZ\) 仅支撑在块 \(B_j\) 上（在 \(B_j\) 内为一个三循环，固定一元素），在其余块上恒等。
\end{theorem}
\begin{proof}
由结论 \ref{con:xi-terminal-zm-braid-monodromy-psp4f3-identification}，
\[
|K|=2^{96}\cdot 3^{40},
\]
故任一 Sylow--\(3\) 子群 \(P_3\) 的阶为 \(3^{40}\)。
又因 \(K\) 在块集合上作用平凡，存在自然忠实嵌入
\[
K\hookrightarrow \prod_{j=1}^{40}\mathfrak S(B_j)\cong \mathfrak S_4^{40}.
\]
于是对每个 \(j\)，投影 \(\pi_j(P_3)\) 是 \(\mathfrak S_4\) 的 \(3\)-子群，故
\(
|\pi_j(P_3)|\le 3
\)。
由忠实性，
\[
|P_3|
\le \prod_{j=1}^{40}|\pi_j(P_3)|
\le 3^{40}.
\]
与 \(|P_3|=3^{40}\) 比较可知上式处处取等，故每个 \(\pi_j(P_3)\) 阶恰为 \(3\)，并且
\[
P_3\cong \prod_{j=1}^{40}\pi_j(P_3)\cong (\ZZ/3\ZZ)^{40}.
\]
定义
\[
P_{3,j}:=P_3\cap \ker\!\Bigl(\prod_{i\neq j}\pi_i\Bigr),
\]
则 \(P_{3,j}\cong \ZZ/3\ZZ\)，且 \(P_3=\prod_j P_{3,j}\)。
该分解由块坐标投影唯一确定，故具有规范性。证毕。
\end{proof}

\begin{theorem}[两换位碰撞边界的三分型分解与分歧阶]\label{thm:conclusion-s4-two-transposition-collision-three-type-boundary}
设 \(\Delta\) 为“相邻两换位分支点碰撞”的一般边界分量，并令
\[
\tau:=\sigma_i^2
\]
为对应局部 braid 单值。则 \(S_4\)-Hurwitz 映射
\[
f:\mathcal H_{S_4}\to \mathcal M
\]
在 \(\Delta\) 一般点上满足：
\begin{enumerate}
  \item 归一化拉回 \(f^{-1}(\Delta)^\nu\) 恰分解为 \(88\) 个几何不可约余维 \(1\) 分量；
  \item 分量分为三类：
  \[
  \text{A 型 }28\text{ 个},\quad
  \text{B 型 }24\text{ 个},\quad
  \text{C 型 }36\text{ 个};
  \]
  \item 分歧指数分别为
  \[
  e_{\mathrm A}=1,\qquad e_{\mathrm B}=1,\qquad e_{\mathrm C}=3.
  \]
\end{enumerate}
其中 A/B/C 三型可按 \(p:=g_ig_{i+1}\) 的循环型区分：A 对应 \(p=1\)，B 对应 \(p\) 为双换位，C 对应 \(p\) 为三循环。
\end{theorem}
\begin{proof}
命题 \ref{prop:xi-terminal-zm-hurwitz-generator-cycle-types-on-160-and-40} 给出
\[
\sigma_i\sim 3^{36}2^{12}1^{28}\quad\text{（在 \(160\) 点 Nielsen 纤维上）}.
\]
对 \(\tau=\sigma_i^2\)：
\begin{itemize}
  \item \(1\)-循环保持为 \(1\)-循环，贡献 \(28\)；
  \item 每个 \(2\)-循环平方后分裂为两个 \(1\)-循环，共贡献 \(2\cdot 12=24\)；
  \item 每个 \(3\)-循环平方后仍为 \(3\)-循环，贡献 \(36\)。
\end{itemize}
故
\[
\tau\sim 1^{52}3^{36},
\]
其循环总数为 \(52+36=88\)，即对应 \(88\) 个不可约边界分量。

局部 Hurwitz 理论中，\(\tau\) 的循环长度即映射到 \(\Delta\) 的局部分歧指数：
固定点对应 \(e=1\)，三循环对应 \(e=3\)。
因此 \(36\) 个 C 型分量为三次分歧，其余 \(28+24\) 个分量无分歧。
再由相邻换位对 \((g_i,g_{i+1})\) 的局部分类：
\(g_i=g_{i+1}\) 给出 \(p=1\)（A 型），
互不相交换位给出双换位积（B 型），
相交换位给出三循环积（C 型），得到三型标记。证毕。
\end{proof}

\begin{corollary}[边界 \texorpdfstring{$\Delta$}{Delta} 上的不同指数]\label{cor:conclusion-s4-boundary-different-exponent-72}
在定理 \ref{thm:conclusion-s4-two-transposition-collision-three-type-boundary} 设定下，
\[
\operatorname{diff}_{\Delta}(f)
:=\sum_{D\subset f^{-1}(\Delta)^\nu}(e_D-1)
=36\cdot(3-1)=72.
\]
即 \(\Delta\) 上总判别指数完全由 \(36\) 个 C 型分量贡献。
\end{corollary}
\begin{proof}
由定理 \ref{thm:conclusion-s4-two-transposition-collision-three-type-boundary}，
仅 C 型满足 \(e_D=3\)，A/B 型均为 \(e_D=1\)。
逐分量求和即得
\(
\operatorname{diff}_\Delta(f)=36\cdot 2=72
\)。
证毕。
\end{proof}

\begin{theorem}[三 cusp 型稳定极限的归一化组合型、节点数与主分量属数]\label{thm:conclusion-s4-cusp-stable-limit-normalization-node-genus}
设 \(X\) 为本节 \(S_4\)-正则一般纤维，\(g(X)=16\)（见结论 \ref{con:xi-terminal-zm-s4-galois-closure-genus-tower-holomorphic-differentials}）。
在定理 \ref{thm:conclusion-s4-two-transposition-collision-three-type-boundary} 的三类边界点处，稳定极限 \(X_0\) 的归一化结构如下：
\begin{enumerate}
  \item A 型（\(p=1,\ H\cong C_2\)）：
  \[
  \#\text{尾分量}=12,\quad
  \#\text{节点}=24,\quad
  g(X_{\mathrm{main}})=4;
  \]
  \item B 型（\(|p|=2,\ H\cong V_4\)）：
  \[
  \#\text{尾分量}=6,\quad
  \#\text{节点}=12,\quad
  g(X_{\mathrm{main}})=10;
  \]
  \item C 型（\(|p|=3,\ H\cong S_3\)）：
  \[
  \#\text{尾分量}=4,\quad
  \#\text{节点}=8,\quad
  g(X_{\mathrm{main}})=12.
  \]
\end{enumerate}
并且三型中每个尾分量均与主分量通过恰好 \(2\) 个节点连接。
若 \(\Gamma_{X_0}\) 为对偶图，则
\[
b_1(\Gamma_{X_0})=
\begin{cases}
12,&\text{A 型},\\
6,&\text{B 型},\\
4,&\text{C 型},
\end{cases}
\qquad
g(X_0)=g(X_{\mathrm{main}})+b_1(\Gamma_{X_0})=16.
\]
\end{theorem}
\begin{proof}
设 \(G=S_4\)，令 \(p=g_ig_{i+1}\)，\(r:=|p|\in\{1,2,3\}\)，
并记尾部局部单值群 \(H:=\langle g_i,g_{i+1}\rangle\)。
对两换位生成子群，三种可能恰为
\[
(H,r)=(C_2,1),\ (V_4,2),\ (S_3,3).
\]
可容许覆盖的局部群论给出：
\begin{itemize}
  \item 尾分量数为指数 \([G:H]=24/|H|\)，故分别为 \(12,6,4\)；
  \item 节点惯性群阶为 \(r\)，主分量上节点预像数为 \(|G|/r=24/r\)，故分别为 \(24,12,8\)；
  \item 每个尾分量上的节点预像数为 \(|H|/r\)，三型均等于 \(2\)。
\end{itemize}
于是每个尾分量与主分量恰连两条边。

对偶图边数 \(E\) 等于节点数，顶点数 \(V=1+[G:H]\)（一个主分量加尾分量），
故
\[
b_1=E-V+1=
\begin{cases}
24-13+1=12,\\
12-7+1=6,\\
8-5+1=4.
\end{cases}
\]
三型尾分量均为三点分歧 \(H\)-覆盖的正规化，亏格为 \(0\)；
故稳定曲线算术亏格满足
\[
g(X_0)=g(X_{\mathrm{main}})+b_1(\Gamma_{X_0}).
\]
又 \(g(X_0)=g(X)=16\)（平坦退化下算术亏格守恒），遂得主分量属数分别为 \(4,10,12\)。证毕。
\end{proof}

\begin{corollary}[三 cusp 型下 Jacobian 退化的环面秩与阿贝尔部分维数]\label{cor:conclusion-s4-cusp-jacobian-toric-rank-abelian-dimension}
设 \(J(X_t)\) 为平滑纤维 Jacobian，\(t\to 0\) 收敛到定理
\ref{thm:conclusion-s4-cusp-stable-limit-normalization-node-genus} 的稳定极限。
则广义 Jacobian 的半稳定分解满足：
\begin{enumerate}
  \item 环面秩
  \[
  \operatorname{rk}_{\mathrm{tor}}=b_1(\Gamma_{X_0})=
  \begin{cases}
  12,&\text{A 型},\\
  6,&\text{B 型},\\
  4,&\text{C 型};
  \end{cases}
  \]
  \item 阿贝尔部分同构于归一化分量 Jacobian 乘积。
  因尾分量皆有理，故阿贝尔部分即 \(\operatorname{Jac}(X_{\mathrm{main}})\)，其维数分别为
  \[
  4,\ 10,\ 12.
  \]
\end{enumerate}
\end{corollary}
\begin{proof}
稳定曲线广义 Jacobian 的标准结构定理给出
\[
\operatorname{rk}_{\mathrm{tor}}=b_1(\Gamma_{X_0}),
\]
阿贝尔部分为各归一化不可约分量 Jacobian 的乘积。
将定理
\ref{thm:conclusion-s4-cusp-stable-limit-normalization-node-genus}
中的 \(b_1\) 与主分量属数代入即得。证毕。
\end{proof}

\begin{proposition}[三 cusp 型尾部覆盖的三点分歧刚性与边界模量来源]\label{prop:conclusion-s4-cusp-tail-three-point-rigidity-and-boundary-moduli}
在定理 \ref{thm:conclusion-s4-cusp-stable-limit-normalization-node-genus} 的三型中，尾部覆盖均为三点分歧的刚性 \(H\)-Galois 覆盖（模去 \(\mathrm{PGL}_2\) 后无连续模量）：
\begin{enumerate}
  \item A 型：\(H=C_2\)，分歧型 \((2,2,1)\)；
  \item B 型：\(H=V_4\)，分歧型 \((2,2,2)\)（对应三个非平凡元）；
  \item C 型：\(H=S_3\)，分歧型 \((2,2,3)\)。
\end{enumerate}
因此边界不可约分量的连续参数完全来自主分量上的分支与节点标记数据；尾部仅贡献离散多重度。
\end{proposition}
\begin{proof}
三点分歧 \(H\)-Galois 覆盖由生成三元组
\((h_1,h_2,h_3)\in H^3\)（满足 \(h_1h_2h_3=1\) 且生成 \(H\)）在同时共轭与三分支点重标记下分类。
对上述三型：
\begin{itemize}
  \item \(C_2\) 的 \((2,2,1)\) 三元组唯一；
  \item \(V_4\) 的 \((2,2,2)\) 生成三元组在自同构与重标记下唯一；
  \item \(S_3\) 的 \((2,2,3)\) 生成三元组在同时共轭下亦为唯一轨道。
\end{itemize}
故三型尾部皆刚性。
可容许覆盖边界局部是“主部数据 \(\times\) 尾部离散选择”的纤维化拼接，
既然尾部无连续模量，边界维数仅由主部分支与节点相对位置给出。证毕。
\end{proof}
