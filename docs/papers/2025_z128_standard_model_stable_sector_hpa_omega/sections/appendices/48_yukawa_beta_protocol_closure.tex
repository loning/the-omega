\section{Yukawa structure and \texorpdfstring{$\beta$}{beta}-function closure from protocol (OP5)}
\label{app:yukawa_beta_protocol_closure}

This appendix closes the OP5 open problem by recording:
(i) how the physical observables of the Yukawa sector (eigenvalue spectrum and mixing matrices) are determined by the closed protocol ingredients,
and (ii) how the one-loop Standard Model $\beta$-function coefficients follow from representation counting on the closed labeling.
The closure treats Yukawa observables as outputs of the tick+CAP framework (modulo stated interface calibrations), rather than as arbitrary matching-layer inputs.

\paragraph{Scope and layer discipline.}
\AuditTag The Yukawa matrices in a general basis contain redundant degrees of freedom (right-handed rotations, rephasing) that are not physical observables within the Standard Model gauge interactions.
We therefore define the \emph{physical Yukawa content} as the gauge-invariant and reparametrization-invariant data that enters observable predictions:
\begin{equation}\label{eq:yukawa_physical_observables}
\mathcal{Y}_{\mathrm{phys}} := \bigl\{\text{Yukawa singular-value spectra }(y_f),\;\; V_{\mathrm{CKM}},\;\; V_{\mathrm{PMNS}}\bigr\}.
\end{equation}
The matrix-element ``texture'' and right-handed rotation matrices are not observables in pure SM interactions and are therefore not required to be uniquely fixed by the protocol.

\subsection{Electroweak VEV from the closed $Z$-scale normalization (interface)}
\label{subsec:vev_from_closed_ew}

\begin{proposition}[Electroweak VEV from $m_Z$ and the closed electroweak normalization]\label{prop:vev_from_mz_closed_ew}
\MatchTag Let $m_Z$ denote the $Z$-pole reference mass recorded in Table~\ref{tab:mass_spectrum}.
\InterfaceTag In the standard electroweak EFT embedding (Appendix~\ref{app:protocol_eft_embedding}), the $Z$-boson mass satisfies
\begin{equation}\label{eq:mz_from_vev_g1g2}
m_Z=\frac{v}{2}\sqrt{g_1^2(\mu_Z)+g_2^2(\mu_Z)},
\end{equation}
where $g_1$ and $g_2$ are the hypercharge and weak-isospin couplings at the $Z$ scale.
Using the closed inverse couplings at $\mu_Z$ (Theorem~\ref{thm:weinberg_angle}),
\begin{equation}\label{eq:ew_inverse_alphas_recap}
\alpha_Y^{-1}(\mu_Z)=10\pi^2,\qquad \alpha_w^{-1}(\mu_Z)=3\pi^2,
\end{equation}
and the standard identification $g_i^2=4\pi\alpha_i$, \eqref{eq:mz_from_vev_g1g2} deterministically fixes the electroweak VEV to
\begin{equation}\label{eq:vev_from_mz_closed}
v:=\frac{2m_Z}{\sqrt{g_1^2(\mu_Z)+g_2^2(\mu_Z)}}=2m_Z\sqrt{\frac{15\pi}{26}}.
\end{equation}
\end{proposition}

\begin{remark}[Status of $v$ and $y_e$]\label{rem:vev_interface_status}
The electroweak VEV $v$ is fixed in this paper as an interface dictionary quantity by \eqref{eq:vev_from_mz_closed} once one commits to (i) the $Z$-scale reference mass $m_Z$ (Table~\ref{tab:mass_spectrum}) and (ii) the closed electroweak normalization at $\mu_Z$ (Theorem~\ref{thm:weinberg_angle}).
Equivalently, the electron Yukawa $y_e=\sqrt{2}m_e/v$ is fixed by the same interface dictionary.
\end{remark}

\subsection{Yukawa eigenvalues from the depth template (diagonal sector)}
\label{subsec:yukawa_eigenvalues_closure}

The mass spectrum has been closed via the resolution-depth template (Section~\ref{sec:mass_spectrum_closure}):
\begin{equation}\label{eq:mass_from_depth_recap}
m_f = m_e\,\varphi^{\,\widehat r(f)},
\end{equation}
where the normalized depth $\widehat r(f)=2\Delta V+5\Delta g+\Delta|w|_1$ is computed from the stable-type invariants of the labeling (Proposition~\ref{prop:rhat_simplified}).

\begin{proposition}[Yukawa eigenvalues from the depth template]\label{prop:yukawa_eigenvalue_closure}
\InterfaceTag Let $v$ denote the electroweak vacuum expectation value fixed by the $Z$-scale interface dictionary in Proposition~\ref{prop:vev_from_mz_closed_ew}.
Then the Yukawa eigenvalue for each fermion $f$ in the closed labeling is determined by:
\begin{equation}\label{eq:yukawa_from_depth}
y_f = \frac{\sqrt{2}\,m_f}{v} = \frac{\sqrt{2}\,m_e}{v}\,\varphi^{\,\widehat r(f)}.
\end{equation}
Equivalently, the Yukawa eigenvalue ratios relative to the electron are:
\begin{equation}\label{eq:yukawa_ratio_phi}
\frac{y_f}{y_e} = \varphi^{\,\widehat r(f)}.
\end{equation}
\end{proposition}
\begin{proof}
This follows directly from the standard relation $m_f = y_f v/\sqrt{2}$ in the SM Higgs mechanism \cite{PeskinSchroeder1995,PDGRPP2024} combined with the closed mass template \eqref{eq:mass_from_depth_recap}.
\end{proof}

\subsection{Mixing matrices from holonomy (off-diagonal sector)}
\label{subsec:mixing_from_holonomy}

The off-diagonal Yukawa structure manifests physically through the CKM and PMNS mixing matrices, which have been closed via the finite holonomy/phase-lift mechanisms:
\begin{itemize}
\item CKM: closed in Section~\ref{sec:couplings_cp} via bounded-complexity selection on mixing parameters (Proposition~\ref{prop:ckm_mixing_rigidity}).
\item PMNS: closed in Section~\ref{sec:pmns_neutrino_closure} via bounded-complexity selection and phase-lift mechanisms (Proposition~\ref{prop:pmns_mixing_rigidity}).
\end{itemize}

\begin{proposition}[Mixing matrices determine left-handed rotations]\label{prop:mixing_left_rotation}
\InterfaceTag In the standard convention, the mixing matrices are:
\begin{equation}
V_{\mathrm{CKM}} = U_u^{L\dagger} U_d^L, \qquad V_{\mathrm{PMNS}} = U_e^{L\dagger} U_\nu^L,
\end{equation}
where $U_f^L$ diagonalizes the left-handed component of the fermion mass matrix.
The physical mixing matrices, together with the mass eigenvalues, completely determine the observable flavor structure in SM gauge interactions.
\end{proposition}
\begin{proof}
Standard; see \cite{PeskinSchroeder1995,PDGRPP2024}.
The right-handed rotation matrices $U_f^R$ do not enter SM gauge vertices (which couple only to left-handed fields for charged currents) and are therefore physical only if new interactions beyond SM are present.
\end{proof}

\begin{remark}[Right-handed rotations and texture uniqueness]\label{rem:right_rotation_redundancy}
\AuditTag The protocol does not (and need not) fix the right-handed rotation matrices $U_u^R, U_d^R, U_e^R, U_\nu^R$ because they do not affect any SM observable.
This is not a failure of closure but a correct implementation of the minimal-input principle: the protocol determines exactly the physical content and no more.
If one wished to uniquely specify matrix-element textures (beyond what mixing and masses determine), an additional CAP selection principle would be required, which would constitute a new assumption beyond the declared input set.
\end{remark}

\subsection{CAP-minimal Higgs-doublet count for a renormalizable EFT embedding (interface)}
\label{subsec:minimal_higgs_doublet_count}

\begin{proposition}[CAP-minimal Higgs-doublet count]\label{prop:minimal_higgs_doublet_count}
Consider the bounded candidate family of Higgs-doublet counts
\[
N_H\in\{0,1,2,3\}.
\]
\InterfaceTag Impose the minimal EFT-embedding feasibility requirement: there exists a renormalizable, gauge-invariant Yukawa sector that generates nonzero Dirac masses for the closed charged fermion multiplets after electroweak symmetry breaking within the conventional SM EFT embedding (Appendix~\ref{app:protocol_eft_embedding}).
Then $N_H=0$ is infeasible, while $N_H\ge 1$ is feasible by including a single complex Higgs doublet $H\sim (1,2)_{1/2}$.
Under Axiom~\ref{ax:cap}, selecting the CAP-minimal feasible value yields the deterministic closure
\[
N_H=1.
\]
\end{proposition}
\begin{proof}
Without a Higgs doublet ($N_H=0$), there is no renormalizable gauge-invariant operator that produces Dirac masses for the chiral fermions.
With one Higgs doublet $H\sim(1,2)_{1/2}$, the standard renormalizable Yukawa terms exist and generate masses after electroweak symmetry breaking; see, e.g., \cite{PeskinSchroeder1995,PDGRPP2024}.
Within the stated finite family, CAP selects the minimal feasible count, hence $N_H=1$.
\end{proof}

\subsection{One-loop \texorpdfstring{$\beta$}{beta}-function coefficients from representation counting}
\label{subsec:beta_from_representation}

\paragraph{Convention.}
\MatchTag We adopt the standard normalization $Q=T_3+Y$ with the fundamental representation index $T(\text{fund})=1/2$ for $SU(N)$, and $Y$ normalized such that $Q_e=-1$.
The one-loop gauge $\beta$-functions are written as:
\begin{equation}\label{eq:beta_convention}
\frac{\mathrm{d}g_i}{\mathrm{d}\log\mu} = \frac{b_i}{16\pi^2}\,g_i^3, \qquad i\in\{1,2,3\}.
\end{equation}

\paragraph{Standard formula.}
The one-loop $\beta$-function coefficients for a gauge theory with gauge group $G$, Weyl fermions in representations $R_f$, and complex scalars in representations $R_s$ are given by \cite{MachacekVaughn1983,MachacekVaughn1984}:
\begin{equation}\label{eq:beta_one_loop_general}
b = -\frac{11}{3}C_2(G) + \frac{2}{3}\sum_{\text{Weyl }f} T(R_f) + \frac{1}{3}\sum_{\text{complex }s} T(R_s),
\end{equation}
where $C_2(G)$ is the quadratic Casimir of the adjoint representation and $T(R)$ is the Dynkin index of representation $R$.

\begin{proposition}[$\beta$-coefficients from closed labeling]\label{prop:beta_from_labeling}
\InterfaceTag
Given:
\begin{enumerate}
\item The gauge group $U(1)_Y\times SU(2)_L\times SU(3)_c$ closed via Proposition~\ref{prop:channel_to_gauge}.
\item The $18$ chiral fermion multiplets (three generations of quarks and leptons including $\nu_R$) closed via the labeling map $\mathcal{L}_{\mathrm{SM}}$ (Theorem~\ref{thm:labeling_unique}).
\item One complex Higgs doublet $(1,2)_{1/2}$ as the CAP-minimal renormalizable scalar uplift interface (Proposition~\ref{prop:minimal_higgs_doublet_count}; consistent with Remark~\ref{rem:higgs_not_in_21} and Appendix~\ref{app:scalar_interface_audits}).
\end{enumerate}
The one-loop $\beta$-coefficients are uniquely determined by representation counting:
\begin{equation}\label{eq:beta_sm_values}
(b_1, b_2, b_3) = \left(\frac{41}{6}, -\frac{19}{6}, -7\right).
\end{equation}
\end{proposition}
\begin{proof}
We compute each coefficient from \eqref{eq:beta_one_loop_general} using the closed matter content.

\paragraph{$SU(3)_c$ coefficient $b_3$.}
The adjoint Casimir is $C_2(SU(3))=3$.
The colored fermions from the closed labeling are (per generation): $Q_L=(3,2)$, $u_R=(3,1)$, $d_R=(3,1)$, giving $T(3)=1/2$ each.
With 3 generations and accounting for $SU(2)$ doublet multiplicity:
\[
b_3 = -\frac{11}{3}\times 3 + \frac{2}{3}\times 3\times\bigl(2\times\tfrac{1}{2} + \tfrac{1}{2} + \tfrac{1}{2}\bigr) + 0 = -11 + 4 = -7.
\]

\paragraph{$SU(2)_L$ coefficient $b_2$.}
The adjoint Casimir is $C_2(SU(2))=2$.
The $SU(2)$ doublets from the closed labeling are (per generation): $Q_L$, $L_L$, plus the Higgs doublet $H$.
Colored doublets contribute $T(2)=1/2$ times color multiplicity:
\[
b_2 = -\frac{11}{3}\times 2 + \frac{2}{3}\times 3\times\bigl(3\times\tfrac{1}{2} + \tfrac{1}{2}\bigr) + \frac{1}{3}\times\tfrac{1}{2}
= -\frac{22}{3} + 4 + \frac{1}{6} = -\frac{19}{6}.
\]

\paragraph{$U(1)_Y$ coefficient $b_1$.}
For $U(1)_Y$ with the standard normalization, $T(R)=Y^2$ for each field.
The hypercharge assignments from the closed labeling give (per generation, summing over colors and $SU(2)$ components):
\begin{align*}
\text{Fermions: } & Q_L: 6\times(\tfrac{1}{6})^2,\; u_R: 3\times(\tfrac{2}{3})^2,\; d_R: 3\times(-\tfrac{1}{3})^2, \\
& L_L: 2\times(-\tfrac{1}{2})^2,\; e_R: 1\times(-1)^2,\; \nu_R: 1\times 0^2.
\end{align*}
The Higgs doublet contributes $2\times(\tfrac{1}{2})^2=\tfrac{1}{2}$.
Summing:
\[
b_1 = \frac{2}{3}\times 3\times\bigl(\tfrac{6}{36}+\tfrac{12}{9}+\tfrac{3}{9}+\tfrac{2}{4}+1+0\bigr) + \frac{1}{3}\times\tfrac{1}{2}
= \frac{2}{3}\times 3\times\frac{10}{3} + \frac{1}{6} = \frac{20}{3} + \frac{1}{6} = \frac{41}{6}.
\]
\end{proof}

\begin{remark}[Higgs as uplift interface assumption]\label{rem:higgs_beta_interface}
\AuditTag The Higgs doublet enters the $\beta$-coefficient computation through the renormalizable protocol EFT embedding and is not derived as a primitive $m=6$ stable type (Remark~\ref{rem:higgs_not_in_21}).
Within the explicit bounded candidate family of scalar uplift completions, the minimal Higgs-doublet count is deterministically closed as $N_H=1$ (Proposition~\ref{prop:minimal_higgs_doublet_count}).
Extended Higgs sectors would modify $b_1$ and $b_2$.
\end{remark}

\begin{remark}[Relation to the $r$-coordinate RG dictionary]\label{rem:beta_r_coordinate}
With the $\beta$-coefficients derived from representation counting, the RG chain rule in the resolution coordinate (Appendix~\ref{app:running_couplings_resolution_flow}) becomes:
\begin{equation}
\frac{\mathrm{d}g_i}{\mathrm{d}r} = (\log\varphi)\frac{b_i}{16\pi^2}\,g_i^3.
\end{equation}
This closes the connection between the finite protocol structure (stable types $\to$ representations $\to$ $\beta$-coefficients) and the resolution-flow dynamics.
\end{remark}
