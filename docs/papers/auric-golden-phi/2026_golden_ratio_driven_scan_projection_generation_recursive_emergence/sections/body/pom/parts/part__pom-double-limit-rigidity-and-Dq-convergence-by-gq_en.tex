\begin{corollary}[Double-Limit Rigidity of the Main Exponent: \texorpdfstring{$\tfrac12\log\varphi$}{(1/2)logphi} as the Unique Unaliasable Volume Slope]\label{cor:pom-double-limit-rigidity-half-logphi}
For integer \(q\ge 2\) and resolution \(m\), let
\(
a_{m,q}:=\frac{1}{mq}\log S_q(m)
\).
Then under the verified Perron principal-scale regime (where \(r_q=\lim_{m\to\infty}S_q(m)^{1/m}\) exists), we have
\[
\lim_{q\to\infty}\ \lim_{m\to\infty}\ a_{m,q}
=\lim_{m\to\infty}\ \lim_{q\to\infty}\ a_{m,q}
=\frac12\log\varphi.
\]
\end{corollary}
\begin{proof}
For fixed \(q\), by \(S_q(m)=\Theta(r_q^m)\) we have
\(
\lim_{m\to\infty}a_{m,q}=\frac{1}{q}\log r_q
\).
Then by the endpoint theorem (Proposition \ref{prop:pom-rq-qinfty-endpoint})
\(
\lim_{q\to\infty}\frac{1}{q}\log r_q=\frac12\log\varphi
\),
yielding the first iterated limit.
\par
For fixed \(m\), by maximum-term dominance (the \(\ell^q\) limit as \(q\to\infty\))
\(
\lim_{q\to\infty}S_q(m)^{1/q}=D_m:=\max_{x\in X_m}d_m(x)
\),
hence
\(
\lim_{q\to\infty}a_{m,q}=\frac{1}{m}\log D_m
\).
Then by the maximum-fiber closed-form theorem (Theorem \ref{thm:pom-max-fiber})
\(
\lim_{m\to\infty}\frac{1}{m}\log D_m=\frac12\log\varphi
\),
yielding the second iterated limit. QED.
\end{proof}
\begin{corollary}[Free-Energy Excess Controls Dimension Convergence: Auditable Error Decomposition of Endpoint Dimension]\label{cor:pom-Dq-convergence-controlled-by-gq}
Let the free-energy excess \(g_q\) be as defined in Remark \ref{rem:pom-free-energy-excess-gq}:
\[
g_q:=\log r_q-\frac{q}{2}\log\varphi.
\]
Then the R\'enyi entropy rate satisfies the identity
\[
h_q=q\log\frac{2}{\sqrt{\varphi}}-g_q,
\]
hence under the geometric scale \(\varepsilon_m=\varphi^{-m}\), the R\'enyi dimension spectrum (Proposition \ref{prop:pom-renyi-dimension-spectrum}) can be written as
\[
\boxed{\;
D_q
=\frac{q}{q-1}\cdot\frac{\log(2/\sqrt{\varphi})}{\log\varphi}
-\frac{g_q}{(q-1)\log\varphi}\;}
\qquad(q\ge 2).
\]
Denoting the endpoint dimension
\(
D_\infty:=\frac{\log(2/\sqrt{\varphi})}{\log\varphi}
\),
the deviation admits the strict decomposition
\[
\boxed{\;
D_q-D_\infty=\frac{1}{q-1}\left(D_\infty-\frac{g_q}{\log\varphi}\right).}
\]
Thus any falsifiable statement about "whether the dimension spectrum rapidly approaches the endpoint" can be reduced to an audit of the decay pattern of \(g_q\) (whose numerical certificate has been given in Remark \ref{rem:pom-free-energy-excess-gq}).
\end{corollary}
\begin{proof}
From \(h_q=q\log 2-\log r_q\) and \(g_q=\log r_q-\frac{q}{2}\log\varphi\), direct manipulation yields
\(
h_q=q\log(2/\sqrt{\varphi})-g_q
\).
Substituting into \(D_q=\frac{h_q}{(q-1)\log\varphi}\) yields the first formula; rewriting the constant term as \(D_\infty\) and rearranging gives the deviation decomposition. QED.
