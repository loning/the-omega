\section{Cut-and-project geometry: from symbolic readout to discrete point sets}
\label{sec:cut_project}

\subsection{A cut-and-project scheme for an irrational slope}
Fix an irrational slope $\alpha\in(0,1)\setminus\QQ$. Consider the lattice $L:=\ZZ^2\subset\RR^2$. Let
$$
E_\parallel:=\mathrm{span}\{(1,\alpha)\},\qquad E_\perp:=\mathrm{span}\{(-\alpha,1)\},
$$
and let $\pi_\parallel$ and $\pi_\perp$ denote the orthogonal projections onto $E_\parallel$ and $E_\perp$, respectively. Given a bounded interval window $W\subset E_\perp$ with boundary of $\mu$-measure zero, the associated model set is
$$
\Lambda(W):=\{\pi_\parallel(\ell)\ :\ \ell\in L,\ \pi_\perp(\ell)\in W\}.
$$
This is the standard cut-and-project construction; see~\cite{Moody2000}. For the golden slope $\alpha=\varphi^{-1}$ and a standard half-open interval window in internal space, $\Lambda(W)$ is the Fibonacci model set, whose gap sequence yields the Fibonacci tiling (two spacings in a nonperiodic order).

\subsection{A canonical staircase coordinate from the readout}
The window coding $s_k\in\{0,1\}$ defines a natural integer coordinate by prefix summation. Set
$$
S_N:=\sum_{k=0}^{N-1} s_k,\qquad N\ge 1.
$$
Then the lattice path $\gamma_N:=(N,S_N)\in\ZZ^2$ advances by steps $(1,0)$ and $(1,1)$, encoding the local short/long pattern.

\begin{proposition}[Prefix-sum coordinate and the staircase line]
\label{prop:prefix_sum}
If $s_k$ is given by the mechanical word formula of Proposition~\ref{prop:window_mechanical}, then
$$
S_N=\big\lfloor N\alpha+\beta\big\rfloor-\big\lfloor \beta\big\rfloor.
$$
In particular, when $\beta\in[0,1)$ one has $S_N=\lfloor N\alpha+\beta\rfloor$, so the staircase $\{(N,S_N)\}$ is a digital approximation of the line $y=\alpha x+\beta$ and provides an explicit bridge between symbolic readout and lattice geometry.
\end{proposition}

\begin{proof}
By telescoping,
$$
S_N=\sum_{k=0}^{N-1}\Big(\big\lfloor (k+1)\alpha+\beta\big\rfloor-\big\lfloor k\alpha+\beta\big\rfloor\Big)
=\big\lfloor N\alpha+\beta\big\rfloor-\big\lfloor \beta\big\rfloor.
$$
\end{proof}
