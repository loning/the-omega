\section{Resolution folding: canonical forms and multiscale compatibility}\label{sec:folding}

This section defines the stable type space $X_m$, constructs the folding map $\Fold_m:\Omega_m\to X_m$, and proves that it is a local rewriting normalizer producing a unique canonical form.
The theorem-level output is a multiscale-compatible canonical interface that can be audited at finite $m$.

\subsection{Stable language $X_m$ (golden mean syntax)}\label{subsec:Xm}
Define the stable type space as the length-$m$ golden mean language:
\[
X_m:=\{x\in\{0,1\}^m:\ \text{$x$ contains no subword $11$}\}\subseteq \Omega_m.
\]
Equivalently, $X_m$ consists of all length-$m$ Zeckendorf-admissible digit blocks.

\subsection{Fibonacci value and folding}\label{subsec:fold-definition}
For $\omega=\omega_1\cdots\omega_m\in\Omega_m$, define its Fibonacci-weighted value
\[
N(\omega):=\sum_{k=1}^{m}\omega_k F_{k+1}\in\NN.
\]
Let $Z(N(\omega))=(c_1,c_2,\dots)$ be the Zeckendorf digit sequence of $N(\omega)$, and let
\[
\pi_m(c_1,c_2,\dots)=(c_1,\dots,c_m)\in X_m
\]
be the length-$m$ restriction. Define the resolution folding map
\[
\Fold_m:\Omega_m\to X_m,\qquad \Fold_m(\omega):=\pi_m(Z(N(\omega))).
\]

\subsection{Local rewriting implementation}\label{subsec:fold-rewrite}
To realize $\Fold_m$ as a local normalization procedure and to make the normal form auditable, we embed a word into a finite-support digit configuration.
Let
\[
\mathcal{D}:=\{a=(a_1,a_2,\dots):\ a_k\in\NN,\ \text{and only finitely many $a_k$ are nonzero}\},
\qquad
\mathrm{Val}(a):=\sum_{k\ge 1} a_k F_{k+1}.
\]
Embed $\omega\in\Omega_m$ into $\mathcal{D}$ by $\iota_m(\omega)=(\omega_1,\dots,\omega_m,0,0,\dots)$.
The folding normalizer is generated by value-preserving local rewriting rules that eliminate adjacent $11$ and digits $\ge 2$.

\begin{definition}[Value-preserving local rewrites]\label{def:fold-rewrite-rules}
For $a\in\mathcal{D}$ and standard basis vectors $e_k$, we allow the following local rules (when applicable):
\begin{enumerate}
  \renewcommand{\theenumi}{R\arabic{enumi}}
  \renewcommand{\labelenumi}{(\theenumi)}
  \item\label{rule:adj-merge}\textbf{Adjacent merge.} If $a_k\ge 1$ and $a_{k+1}\ge 1$, then
  \[
  a\ \to\ a-e_k-e_{k+1}+e_{k+2}.
  \]
  \item\label{rule:dedup-1}\textbf{Deduplicate $k=1$.} If $a_1\ge 2$, then
  \[
  a\ \to\ a-2e_1+e_2.
  \]
  \item\label{rule:dedup-2}\textbf{Deduplicate $k=2$.} If $a_2\ge 2$, then
  \[
  a\ \to\ a-2e_2+e_1+e_3.
  \]
  \item\label{rule:dedup-k}\textbf{Deduplicate $k\ge 3$.} If $a_k\ge 2$, then
  \[
  a\ \to\ a-2e_k+e_{k-2}+e_{k+1}.
  \]
\end{enumerate}
Each rewrite is local and preserves $\mathrm{Val}$ by Fibonacci identities.
\end{definition}

\begin{proposition}[Termination and confluence]\label{prop:fold-termination-confluence}
The rewriting system in Definition~\ref{def:fold-rewrite-rules} is strongly terminating and locally confluent on $\mathcal{D}$.
Hence it is confluent; every $a\in\mathcal{D}$ has a unique normal form $\mathrm{NF}(a)\in X_\infty:=\{0,1\}^{\NN}$ with no adjacent $11$, and $\mathrm{Val}(\mathrm{NF}(a))=\mathrm{Val}(a)$.
\end{proposition}

\begin{proof}
Termination is proved by a lexicographic decreasing measure, and confluence follows from value preservation combined with Zeckendorf uniqueness. See Appendix~\ref{app:rewrite-system} for a self-contained write-up.
\end{proof}

\begin{corollary}[Order-independent implementation of $\Fold_m$]\label{cor:fold-order-independent}
For any $\omega\in\Omega_m$,
\[
\Fold_m(\omega)=\pi_m\bigl(\mathrm{NF}(\iota_m(\omega))\bigr),
\]
and the right-hand side is independent of the choice and order of local rewrites.
\end{corollary}

\begin{proposition}[Idempotence and surjectivity]\label{prop:fold-idem-surj}
For all $m\ge 1$, the folding map satisfies:
\begin{enumerate}
  \item $\Fold_m(\omega)\in X_m$ for all $\omega\in\Omega_m$;
  \item $\Fold_m|_{X_m}=\mathrm{id}$ (idempotence on stable types);
  \item $\Fold_m:\Omega_m\twoheadrightarrow X_m$ is surjective.
\end{enumerate}
\end{proposition}

\begin{proof}
By definition, $Z(N(\omega))\in X_\infty$ and hence $\pi_m(Z(N(\omega)))\in X_m$.
If $\omega\in X_m$, then $\omega$ is already a length-$m$ Zeckendorf-admissible prefix, so Zeckendorf uniqueness implies $\Fold_m(\omega)=\omega$.
Surjectivity follows by taking $\omega=x\in X_m$.
\end{proof}

\subsection{Multiscale compatibility: inverse system and fold-aware restriction}\label{subsec:fold-multiscale}
Across resolutions, restriction on $X_m$ forms an inverse system.
However, naive truncation on $\Omega_m$ does not commute with folding, so we introduce a fold-aware restriction on microstates.

\begin{definition}[Restriction maps]\label{def:fold-restriction}
For $m_2\ge m_1$, define the prefix restriction on stable types
\[
\pi_{m_2\to m_1}:X_{m_2}\to X_{m_1},\qquad \pi_{m_2\to m_1}(x_1\cdots x_{m_2})=x_1\cdots x_{m_1}.
\]
On microstates, define naive truncation
\[
r_{m+1\to m}:\Omega_{m+1}\to\Omega_m,\qquad r_{m+1\to m}(\omega_1\cdots\omega_{m+1})=\omega_1\cdots\omega_m,
\]
and the fold-aware restriction
\[
\widetilde r_{m+1\to m}(\omega):=\pi_{m+1\to m}(\Fold_{m+1}(\omega))\in X_m\subset\Omega_m.
\]
\end{definition}

\begin{remark}[Naive truncation does not commute with folding]\label{rem:noncommute-truncation}
In general,
\(
\Fold_m\circ r_{m+1\to m}\neq \pi_{m+1\to m}\circ \Fold_{m+1}.
\)
For example, with $m=1$ and $\omega=11$, one has $r_{2\to 1}(11)=1$ and hence $(\Fold_1\circ r_{2\to 1})(11)=1$, whereas $\Fold_2(11)=00$ so $(\pi_{2\to 1}\circ\Fold_2)(11)=0$.
This motivates the fold-aware restriction $\widetilde r_{m+1\to m}$.
\end{remark}

\begin{definition}[Gauge anomaly]\label{def:gauge-anomaly}
For $m\ge 1$ and $\omega\in\Omega_{m+1}$, define
\[
G_m(\omega):=\bigl|\,r_{m+1\to m}(\omega)\ \oplus\ \widetilde r_{m+1\to m}(\omega)\,\bigr|_0,
\]
where $\oplus$ is bitwise XOR on $\{0,1\}^m$ and $|\cdot|_0$ is the Hamming weight.
\end{definition}

\begin{proposition}[Worst-case gauge anomaly is full length]\label{prop:gauge-anomaly-max}
For every $m\ge 1$, there exists $\omega^\star\in\Omega_{m+1}$ such that $G_m(\omega^\star)=m$.
\end{proposition}

\begin{proof}
Take $\omega^\star$ whose first $m$ bits follow the periodic pattern $110110\cdots$ and choose the last bit so that the value-preserving merges in rule~\eqref{rule:adj-merge} apply at disjoint positions.
After performing these disjoint merges, the normal form prefix flips to $001001\cdots$ on the first $m$ bits, yielding a bitwise mismatch on all $m$ coordinates. A detailed construction is standard from the local rewrite rules and is omitted.
\end{proof}

\begin{theorem}[Canonical folding with multiscale compatibility]\label{thm:fold-normal-form-multiscale}
For each $m\ge 1$, the folding map $\Fold_m:\Omega_m\to X_m$ admits a local rewriting implementation and produces a unique canonical form, independent of reduction order.
Moreover:
\begin{enumerate}
  \item (\textbf{Idempotent canonical projection}) $\Fold_m\circ\Fold_m=\Fold_m$, and $\Fold_m$ is the identity on $X_m$.
  \item (\textbf{Inverse system on stable types}) The family $\{X_m,\pi_{m_2\to m_1}\}$ is an inverse system; its inverse limit is naturally identified with
  \[
  X_\infty=\{(c_1,c_2,\dots)\in\{0,1\}^{\NN}:\ c_kc_{k+1}=0\ \forall k\}.
  \]
  \item (\textbf{Fold-aware commutative diagram}) For all $m\ge 1$, the diagram commutes:
  \[
  \begin{CD}
  \Omega_{m+1} @>{\Fold_{m+1}}>> X_{m+1}\\
  @V{\widetilde r_{m+1\to m}}VV @VV{\pi_{m+1\to m}}V\\
  \Omega_m @>{\Fold_m}>> X_m.
  \end{CD}
  \]
\end{enumerate}
\end{theorem}

\begin{proof}
Uniqueness of the normal form and order-independence follow from Proposition~\ref{prop:fold-termination-confluence}.
Item (1) is Proposition~\ref{prop:fold-idem-surj}.
Item (2) follows from the functoriality of prefix restriction and the characterization of $X_\infty$ as the space of all infinite admissible Zeckendorf digit sequences.
For (3), note that $\widetilde r_{m+1\to m}(\omega)\in X_m$, hence $\Fold_m(\widetilde r_{m+1\to m}(\omega))=\widetilde r_{m+1\to m}(\omega)$ by idempotence on $X_m$, which equals $\pi_{m+1\to m}(\Fold_{m+1}(\omega))$ by definition.
\end{proof}

\subsection{Compression and fibers}\label{subsec:fold-fibers}
The folding map is typically many-to-one; its fiber multiplicity is
\[
d_m(x):=\card{\Fold_m^{-1}(x)},\qquad x\in X_m.
\]
This quantity is the precise combinatorial carrier of the hidden ``fiber volume'' term in the information ledger (Section~\ref{sec:ledger}).
