\begin{definition}[Certificate object, refinement and verifiable equivalence]\label{def:xi-cert-refinement-equivalence}
Pin a verifiable real-valued function \(F:\RR\to\RR\) ​​on a unitary slice, and pin a deterministic verifier \(\mathrm{Ver}\);

\begin{enumerate}

\end{enumerate}
And define \textbf{verifiable equivalence} \(C\sim C'\) ​​if and only if there is a common refinement \(D\) such that \(D\preceq C\) and \(D\preceq C'\).
\end{definition}

\begin{theorem}[Inverse limit merging of accuracy consistency and implementation neutrality]\label{thm:xi-cert-inverse-limit-coherence}

\end{enumerate}

\begin{corollary}[Type negation: zero is not an address]\label{cor:xi-zero-not-address}

\begin{definition}[dyadic prefix compatibility depth and address hypermetric]\label{def:xi-dyadic-prefix-depth-ultrametric}
For any \(b\in\NN\) ​​and \(\theta\in\RR\), define a dyadic box
\[
J_b(\theta):=\Bigl[\frac{\lfloor 2^b\theta\rfloor}{2^b},\ \frac{\lfloor 2^b\theta\rfloor+1}{2^b}\Bigr),
\qquad |J_b(\theta)|=2^{-b}.
\]

\]
And agreed \(2^{-\infty}=0\).

\[
d_{\mathrm{addr}}(\theta,\theta'):=2^{-m(\theta,\theta')}.
\]
Under this convention, \(d_{\mathrm{addr}}\) ​​is ultrametric, and its closed sphere with radius \(2^{-b}\) is exactly a dyadic box: for any \(\theta\in\RR\),
\(

\end{definition}

\begin{remark}[Shared prefix and certificate refinement depth]\label{rem:xi-shared-prefix-cert-depth}

\subparagraph{The structural origin of endpoint compression and the precise determination of the "scale transfer" problem}

And let \(w=\mathcal{C}(z)\) ​​be the Cayley compactification, \(s(w)=\tfrac12+\mathrm{i}\,\mathcal{C}^{-1}(w)\) be the disk section, so the logarithmic derivative of the disk horizon

\begin{proposition}[Positivity judgment of extreme resonance and witness of finite-dimensional failure]\label{prop:xi-extreme-resonance-positivity}

  \item \textbf{(RH \texorpdfstring{$\Longleftrightarrow$}{<->} Positive \texorpdfstring{$\Longleftrightarrow$}{<->} Pole is pushed to the boundary)}

\end{proposition}

\end{proof}

\subparagraph{Carath\'eodory Schurization of impedance and Schur--OPUC certificate chain}
Proposition \ref{prop:xi-extreme-resonance-positivity} ​​has strictly aligned "the emergence of resonance poles within the stability domain" as Carath\'eodory positivity and failure of the Toeplitz--PSD hierarchy.
On top of this positive interface, the Cayley--Schur transformation further converts the positive real part of the horizon impedance into the Schur class, thus upgrading the "resonance = pole" dictionary into a completely discrete chain of reflection coefficient certificates, and giving the OPUC closed-form readout of the endpoint threshold bearing strength.
\par
Define the Cayley--Schur transformation
\begin{equation}\label{eq:xi-horizon-cayley-schur}
\boxed{\;
S_{\zeta}(w):=\frac{\mathscr{C}_{\zeta}(w)-1}{\mathscr{C}_{\zeta}(w)+1},
\qquad
\mathscr{C}_{\zeta}(w)=\frac{1+S_{\zeta}(w)}{1-S_{\zeta}(w)}.\;}
\end{equation}

\begin{theorem}[Carath\'eodory \texorpdfstring{$\Longleftrightarrow$}{<->} Schur and stable domain pole repulsion]\label{thm:xi-horizon-carath-schur-equivalence}
Let \(\mathscr{C}_{\zeta}\) ​​resolve within \(\mathbb{D}\) and satisfy \(\Re \mathscr{C}_{\zeta}(w)\ge 0\).
Then \(S_{\zeta}\) ​​is resolved within \(\mathbb{D}\) and satisfies \(|S_{\zeta}(w)|\le 1\), that is, \(S_{\zeta}\) belongs to the Schur class; vice versa.
Therefore, when \(\mathscr{C}_{\zeta}\) ​​is Carath\'eodory, the resonance pole within the stable domain \(\mathbb{D}\) is structurally excluded, and the allowed singularities can only be carried on \(|w|=1\) in the form of boundary spectral measures (represented by Herglotz/Carath\'eodory).
\end{theorem}

\begin{definition}[Schur algorithm and reflection coefficient]\label{def:xi-horizon-schur-parameters}
Execute Schur algorithm on Schur function \(S_{\zeta}\), defined
\[
S_0:=S_{\zeta},\qquad \alpha_n:=S_n(0),\qquad
S_{n+1}(w):=\frac{1}{w}\,\frac{S_n(w)-\alpha_n}{1-\overline{\alpha_n}S_n(w)}\qquad(n\ge 0).
\]
The sequence \((\alpha_n)_{n\ge 0}\) ​​is called the Schur/Verblunsky (reflection) coefficient.
\end{definition}

\begin{theorem}[Finite falsifier for reflection coefficient out-of-bounds]\label{thm:xi-horizon-reflection-finite-witness}
If \(\mathscr{C}_{\zeta}\) ​​is Carath\'eodory (equivalent to \(S_{\zeta}\) being Schur), then there is \(|\alpha_n|<1\) for all \(n\ge 0\).
Conversely, if there is some \(N\) ​​such that \(|\alpha_N|\ge 1\), then \(S_{\zeta}\) is not Schur, and therefore \(\mathscr{C}_{\zeta}\) is not Carath\'eodory;
Therefore, a resonance pole must appear in the stable domain, and equivalently trigger a certain finite-level Toeplitz--PSD failure witness (Proposition \ref{prop:xi-extreme-resonance-positivity} ​​(Article (3); see \cite{Simon2005OPUC1}).
\end{theorem}

\begin{theorem}[Conservative implementation and endpoint threshold bearer]\label{thm:xi-horizon-conservative-realization}
If \(\mathscr{C}_{\zeta}\) ​​is Carath\'eodory, then \(S_{\zeta}\) has a conservative (unitary) scattering implementation: there is a unitary colligation \(\mathcal U\) such that \(S_{\zeta}\) is its characteristic function.
Equivalently, there is a (finite/infinite) canonical realization of CMV generated by \((\alpha_n)\), and there are no scattering poles in the stable domain in this conservative phase; all "extreme resonances" can only degenerate into threshold carriers on the boundary \(|w|=1\) (see \cite{CanteroMoralVelazquez2003CMV,Simon2005OPUC1}, and with the proposition The limited window CMV scope of \ref{prop:xi-hilbert-polya-cmv} is the same type).
\end{theorem}

\begin{theorem}[Christoffel formula for endpoint atomic strength]\label{thm:xi-horizon-endpoint-atom-christoffel}
Let \(\mathscr{C}_{\zeta}\) ​​be Carath\'eodory, and let \(\mu_{\zeta}\) be its representation measure (theorem \ref{thm:app-horizon-weyl-herglotz}, corollary \ref{cor:app-horizon-carath-rh}).
Let \(\{\varphi_n\}_{n\ge 0}\) ​​be the unit circle orthogonal normalized polynomial (OPUC) of \(\mu_{\zeta}\), and define the Christoffel function
\[
\lambda_N(\xi):=\left(\sum_{n=0}^{N}|\varphi_n(\xi)|^2\right)^{-1},\qquad \xi\in\TT.
\]
Then it holds for any \(\xi\in\TT\)
\[
\mu_{\zeta}(\{\xi\})
=\left(\sum_{n=0}^{\infty}|\varphi_n(\xi)|^2\right)^{-1}
=\lim_{N\to\infty}\lambda_N(\xi),
\]
and \(\mu_{\zeta}(\{\xi\})=0\) ​​iff \(\lambda_N(\xi)\to 0\) (see \cite{Simon2005OPUC1}).
Take \(\xi=-1\) ​​to get the endpoint point state threshold intensity
\[
\mathfrak m_{-1}:=\mu_{\zeta}(\{-1\})=\lim_{N\to\infty}\lambda_N(-1).
\]
\end{theorem}

\begin{corollary}[Two-stage discrimination: reflection coefficient does not cross the boundary \(\times\) ​​Endpoint atoms are controllable]\label{cor:xi-horizon-extreme-resonance-two-stage}

\begin{theorem}[Quadratic radial compression law and precision potential attribution for deviation from critical line]\label{thm:xi-offcritical-quadratic-radial-compression}
Assume that there is a zero point \(\rho=\beta+\mathrm{i}\gamma\) ​​away from the critical line that satisfies \(0<\beta<\tfrac12\).
make
\(
\delta:=\tfrac12-\beta>0
\)
Let \(w_\rho=\mathcal{C}(\gamma+\mathrm{i}\delta)\in\mathbb{D}\) ​​be its disk image.
but:
\begin{enumerate}
  \item ​​\textbf{(Closed identity/horocycle implementation)}
  \[
  \delta=\mathcal{B}_{-1}(w_\rho)=\frac{1-|w_\rho|^2}{|1+w_\rho|^2},
  \qquad
  |w_\rho|^2=\frac{\gamma^2+(1-\delta)^2}{\gamma^2+(1+\delta)^2},
  \qquad
  1-|w_\rho|^2=\frac{4\delta}{\gamma^2+(1+\delta)^2}=\delta\,|1+w_\rho|^2
  \]
  (Corollary \ref{cor:app-offcritical-horocycle-busemann}).
  So when \(|\gamma|\to\infty\) ​​and \(\delta\) are bounded,
  \[
  1-|w_\rho|^{2}=\frac{4\delta}{\gamma^{2}}\bigl(1+o(1)\bigr),
  \qquad
  1-|w_\rho|=\frac{2\delta}{\gamma^{2}}\bigl(1+o(1)\bigr),
  \]
  Thus, at a fixed offset \(\delta\), off-slice information is compressed into the resolvable scale \(\Theta(\gamma^{-2})\) of the disk boundary layer, and its compression strength increases quadratically with height.
  This law is not an empirical geometry in the sense of "symmetry axis", but the structural consequence of "involution fixed slice \(|u|=1\) ​​+ compactification gate" in the visualization scope.
  \item ​​\textbf{(Endpoint second-order radial channel)}
  Record the endpoint depth \(\delta_\gamma:=\tfrac12-\pi^{-1}\arctan|\gamma|\downarrow 0\), then
  \[
  1-|w_\rho|^2
  =4\pi^2\delta\,\delta_\gamma^2\bigl(1+o(1)\bigr)\qquad(\delta_\gamma\downarrow 0)
  \]
  (Corollary \ref{cor:app-offcritical-radius-second-order-radial-channel}).
  Thus at the natural scale of the endpoint neighborhood \(|1+w|\ll 1\), \(\delta\) enters in the \emph{quadratic} term of the chordal distance \(|1+w|^2\).
  \item ​​\textbf{(Precision Potential Attribution)}
  Let the accuracy potential \(\mathcal{P}(x)=\log\pi+\log(1+x^2)\) ​​(define \ref{def:unit-circle-phase-precision-potential}), then
  \[
  1-|w_\rho|^2\sim 4\pi\,\delta\,e^{-\mathcal{P}(\gamma)}@@TK7 @@(|\gamma|\to\infty,\ \delta\ \text{bounded})
  \]
  (Note \ref{rem:app-offcritical-radius-precision-potential}).
  This attributes the "squeezing of disk visibility away from the critical line" exactly to an exponential factor \(e^{-\mathcal{P}(\gamma)}\) ​​of the accuracy potential.
\end{enumerate}
\end{theorem}

\begin{proof}
The three assertions are the direct restatement of the Corollary \ref{cor:app-offcritical-horocycle-busemann}, the Corollary \ref{cor:app-offcritical-radius-second-order-radial-channel}, and the note \ref{rem:app-offcritical-radius-precision-potential}. Certification completed.
\end{proof}

\begin{corollary}[Precision potential decomposition and “offset loading”: logarithmically additive structure of endpoint layer depth]\label{cor:xi-offcritical-endpoint-depth-log-decomposition}

\end{equation}
\end{corollary}

\begin{proof}

h(\rho)\sim 4\pi\,\delta\,e^{-\mathcal{P}(\gamma)}
\)

\end{proof}

\begin{remark}[Co-moving horizon localization: institutional costs of linear development and address prefixes]\label{rem:xi-comoving-horizon-localization-address-null}
The endpoint depth decomposition of the formula \eqref{eq:xi-offcritical-endpoint-product-law}--\eqref{eq:xi-offcritical-endpoint-depth-load-decomposition} is obtained under the fixed Cayley diagram \(\mathcal{C}\) (i.e. radial audit with fixed base points): when \(|\gamma|\to\infty\) and \(\delta\) are bounded, \(\delta\) is passed by the factor \(e^{-\mathcal{P}(\gamma)}\asymp \gamma^{-2}\) Quadratic radial channels that are structurally pressed into the neighborhood of endpoints (Theorem \ref{thm:xi-offcritical-quadratic-radial-compression}).
\par
However, this "second-order compression" is not an inherent obstacle to the horocycle structure.
The horizontal translation \(t\mapsto t-x_0\) ​​(\(x_0\in\RR\)) in the upper half-plane remains unchanged \(\Im(t)\), so it appears as an automorphic reselection of the disk along the same horocycle at the Busemann coordinates of the disk endpoint \(\delta=\mathcal{B}_{-1}(w)\) (corollary \ref{cor:app-offcritical-horocycle-busemann}).
The appendix theorem \ref{thm:app-comoving-horizon-localization} gives its closed-form effect: when taking the co-moving parameter \(x_0=\gamma\),
\[
1-|w_{\rho,\gamma}|^2=\frac{4\delta}{(1+\delta)^2}\sim 4\delta\qquad(\delta\downarrow 0),
\]
As a result, deviations from the critical line appear on a linear scale in the co-motion localization diagram, without second-order compression occurring with \(|\gamma|\).
\par
This localization geometrically eliminates secondary compression, but it is not without cost in the verifiable system: it is compatible with the "address before value" system In this paper, and the translation parameter \(x_0\) must be explicitly given as an addressable prefix (\S\ref{sec:recursive-addressing}); when the address is not given, the corresponding evaluation function is not defined at the object level and is read as \(\mathsf{NULL}\), and recursive addressing only reselects and organizes events within the existing visible domain without expanding the visible \(\sigma\)-algebra (corollary \ref{cor:recursive-addressing-visible-sigma-nonexpansion}).
Therefore, subsequent lower bounds on endpoint resolution and bit complexity (such as theorem \ref{thm:xi-offcritical-endpoint-resolution-lower-bound}) still describe the non-bypassable budget under the fixed visible interface: if the co-moving prefix is ​​not external, the offline offset must enter through the secondary radial channel in the high altitude range and trigger the corresponding capacity threshold.
\end{remark}

\begin{remark}[Addable load of endpoint depth]\label{rem:xi-offcritical-endpoint-depth-additive-load}

\begin{theorem}[Endpoint resolution—bit complexity barrier: irreducible budget lower bound derived from quadratic compression (explicit worst case)]\label{thm:xi-offcritical-endpoint-resolution-lower-bound}
Fixed \(\delta_0\in(0,\tfrac12)\) ​​and \(T>0\).
If there is a zero point away from the critical line \(\rho=\beta+\mathrm{i}\gamma\) ​​satisfies
\[
|\gamma|\le T,\qquad \beta\le \tfrac12-\delta_0,
\]
And still remember that its disk image is \(w_\rho=\mathcal{C}(\gamma+\mathrm{i}(\tfrac12-\beta))\), then the endpoint layer margin satisfies the explicit lower bound
\begin{equation}\label{eq:xi-offcritical-endpoint-margin-min}
\boxed{\;
1-|w_\rho|^2\ \ge\ \frac{4\delta_0}{T^2+(1+\delta_0)^2}\; }.
\end{equation}
Therefore, any verification strategy that attempts to exclude all offline zero points in the above-mentioned height window only by controlling the endpoint layer resolution threshold must satisfy if its minimum resolvable scale for the margin \(1-|w|^2\) ​​is \(2^{-p}\)
\begin{equation}\label{eq:xi-offcritical-endpoint-bit-lower-bound}
\boxed{\;
p\ \ge\ \log_2\!\Bigl(\frac{T^2+(1+\delta_0)^2}{4\delta_0}\Bigr)\; }.
\end{equation}
And there is an asymptotic lower bound when \(T\to\infty\)
\begin{equation}\label{eq:xi-offcritical-endpoint-bit-asymptotic}
\boxed{\;
p\ \ge\ 2\log_2 T+\log_2\!\frac{1}{\delta_0}+O(1)\; }.
\end{equation}
Using the accuracy potential \(\mathcal{P}(x)=\log\pi+\log(1+x^2)\), the asymptotic can also be written as
\begin{equation}\label{eq:xi-offcritical-endpoint-bit-precision-potential-form}
\boxed{\;
p\ \ge\ \frac{\mathcal{P}(T)+\log(1/\delta_0)-\log(4\pi)}{\log 2}+o(1)\qquad(T\to\infty)\; }.
\end{equation}
\end{theorem}

\begin{proof}
From the closed-form identity of theorem \ref{thm:xi-offcritical-quadratic-radial-compression} ​​Item (1),
\[
1-|w_\rho|^2=\frac{4\delta}{\gamma^2+(1+\delta)^2},
\qquad \delta=\tfrac12-\beta\in[\delta_0,\tfrac12).
\]
On \(\delta\in(0,1)\), the function \(\delta\mapsto \frac{4\delta}{\gamma^2+(1+\delta)^2}\) increases monotonically; and on \(|\gamma|\le T\) its right end decreases as \(|\gamma|\) increases.
Therefore, under the constraint \(\delta\ge\delta_0,|\gamma|\le T\), the minimum value of the margin is obtained by \(\delta=\delta_0\) and \(|\gamma|=T\), thus obtaining \eqref{eq:xi-offcritical-endpoint-margin-min}.
\par
If the minimum resolvable scale of the endpoint margin of the verification mechanism is \(2^{-p}\), then in order to achieve resolvable exclusion of all possible offline zero points at the endpoint layer within this height window, \(2^{-p}<\inf(1-|w_\rho|^2)\) must be satisfied. Combined with \eqref{eq:xi-offcritical-endpoint-margin-min}, we get \eqref{eq:xi-offcritical-endpoint-bit-lower-bound}.
\eqref{eq:xi-offcritical-endpoint-bit-asymptotic} Obtained from \(\log_2(T^2+(1+\delta_0)^2)=2\log_2 T+O(1)\).
Finally, from \(\mathcal{P}(T)=\log\pi+\log(1+T^2)\) ​​the principal term can be written in the form of precision potential, and we get \eqref{eq:xi-offcritical-endpoint-bit-precision-potential-form}. Certification completed.
\end{proof}

\begin{corollary}[Bit budget revenue law of co-moving prefix: secondary compression \texorpdfstring{$\Rightarrow$}{=>} constant budget]\label{cor:xi-comoving-prefix-bit-budget-gain}
Follow the setting of theorem \ref{thm:xi-offcritical-endpoint-resolution-lower-bound} and fix \(\delta_0\in(0,\tfrac12)\).
If the external addressable prefix \(x_0\) ​​is allowed at the object layer and the co-moving localization \(x_0=\gamma\) is taken (Appendix theorem \ref{thm:app-comoving-horizon-localization}; and see note \ref{rem:xi-comoving-horizon-localization-address-null} for the explanation of the system cost), then the detectable margin of the offline offset in the co-moving diagram satisfies
\[
h_{\gamma}(\rho)=1-|w_{\rho,\gamma}|^2=\frac{4\delta}{(1+\delta)^2}\ \ge\ \frac{4\delta_0}{(1+\delta_0)^2}\qquad(\delta\ge\delta_0).
\]
Therefore, if the verification mechanism uses the threshold \(2^{-p}\) ​​as the “detectable margin” (that is, requiring \(h_{\gamma}(\rho)\ge 2^{-p}\) to be true for all \(\delta\ge\delta_0\)), it must satisfy a highly independent constant lower bound
\begin{equation}\label{eq:xi-comoving-bit-threshold}
\boxed{\;
p\ \ge\ \log_2\!\Bigl(\frac{(1+\delta_0)^2}{4\delta_0}\Bigr)\; }.
\end{equation}
Comparing with \eqref{eq:xi-offcritical-endpoint-bit-asymptotic}, it can be seen that on the height window \(|\gamma|\le T\), the main stage budget "recycled" by the co-moving prefix is ​​\(2\log_2T+O(1)\).
\end{corollary}

\begin{proof}
From the appendix theorem \ref{thm:app-comoving-horizon-localization}, when taking \(x_0=\gamma\)
\(
h_{\gamma}(\rho)=\frac{4\delta}{(1+\delta)^2}
\)。
This function increases monotonically on \(\delta\in(0,1)\), so it has
\(
h_{\gamma}(\rho)\ge \frac{4\delta_0}{(1+\delta_0)^2}
\)。
If \(h_{\gamma}(\rho)\ge 2^{-p}\) ​​is required, \(2^{-p}\le \frac{4\delta_0}{(1+\delta_0)^2}\) is required, which is equivalent to \eqref{eq:xi-comoving-bit-threshold}.
The last sentence is obtained by subtracting \eqref{eq:xi-comoving-bit-threshold} ​​from \eqref{eq:xi-offcritical-endpoint-bit-asymptotic}. Certification completed.
\end{proof}
