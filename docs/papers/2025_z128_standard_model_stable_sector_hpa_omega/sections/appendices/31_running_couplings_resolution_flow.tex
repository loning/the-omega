\subsection{Running couplings as resolution flow in the $r$ coordinate (self-contained interface)}
\label{app:running_couplings_resolution_flow}

This appendix makes the RG dictionary in the Fibonacci resolution coordinate $r$ self-contained.
The main text already records the basic chain rule (Proposition~\ref{prop:rg_dictionary_r}); here we add the standard one-loop templates and the semantics of threshold matching and scheme shifts, in a form aligned with the tick\,+\,CAP stance (ratio-first, then matching).

\subsubsection{Scale map and chain rule}
\label{subsec:rg_chain_rule}

\paragraph{Resolution map.}
Work with the Fibonacci map
\[
\mu(r)=\mu_0\,\varphi^{\,r},
\]
so that $\log\mu=\log\mu_0+r\log\varphi$ and therefore $\dd/\dd r=(\log\varphi)\,\dd/\dd\log\mu$ (Proposition~\ref{prop:rg_dictionary_r}).

\paragraph{RG in $r$.}
If a coupling $g(\mu)$ satisfies an RG equation
\[
\frac{\dd g}{\dd\log\mu}=\beta(g),
\]
then in the $r$ coordinate,
\begin{equation}
\label{eq:rg_in_r}
\frac{\dd g}{\dd r}=(\log\varphi)\,\beta(g).
\end{equation}

\subsubsection{One-loop QED running (leading log)}
\label{subsec:qed_one_loop}

At one loop (mass-independent scheme, away from thresholds), QED running gives
\begin{equation}
\label{eq:qed_one_loop}
\alpha^{-1}(\mu)\approx \alpha^{-1}(\mu_0)-\frac{b}{2\pi}\log\!\left(\frac{\mu}{\mu_0}\right),
\qquad
b=\frac{2}{3}\sum_f N_c^{(f)}Q_f^2,
\end{equation}
so in the $r$ coordinate one obtains a linear flow
\begin{equation}
\label{eq:qed_one_loop_r}
\alpha^{-1}(r)\approx \alpha^{-1}(0)-\frac{b\log\varphi}{2\pi}\,r.
\end{equation}

\subsubsection{One-loop QCD running and dimensional transmutation}
\label{subsec:qcd_one_loop}

For QCD, the beta function has the opposite sign; at one loop,
\begin{equation}
\label{eq:qcd_one_loop}
\frac{\dd\alpha_s}{\dd\log\mu}=-\frac{b_0}{2\pi}\alpha_s^2,
\qquad
b_0=11-\frac{2}{3}n_f,
\end{equation}
so
\begin{equation}
\label{eq:qcd_solution}
\alpha_s(\mu)=\frac{2\pi}{b_0\log(\mu/\Lambda)},
\end{equation}
where $\Lambda$ is a scheme-dependent transmutation scale.
In the $r$ coordinate, a constant rescaling of $\Lambda$ is an additive shift of $r$, so ``scheme dependence'' becomes an explicit equivalence-class shift rather than an uncontrolled knob.

\subsubsection{Threshold matching as discrete uplifts}
\label{subsec:threshold_matching}

\InterfaceTag Effective-field-theory decoupling implies that charged fields contribute to running only above their thresholds; across a threshold one matches schemes by a finite jump/offset.
In the protocol language, thresholds are resolution-uplift points: as $m_{\mathrm{eff}}(\mu)$ changes discretely, effective degrees of freedom appear/disappear and induce discrete changes in $\beta$-coefficients.
This is the semantic link between RG matching and the discrete staircase used elsewhere in this paper (Section~\ref{subsec:p3_resolution_jumps}).
