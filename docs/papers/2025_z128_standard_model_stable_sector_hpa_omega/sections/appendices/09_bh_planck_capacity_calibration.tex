\subsection{Protocol screen capacity at finite resolution}
\label{subsec:bh_capacity_protocol}

\paragraph{Resolution parameters.}
\InterfaceTag At protocol resolution $(m,n)$, the local microstate alphabet at each site is $\Omega_m=\{0,1\}^m$ with $|\Omega_m|=2^m$,
and the $2$D screen contains $4^n$ sites under Hilbert addressing at order $n$.

\paragraph{Microstate capacity (bits).}
\InterfaceTag The maximal microstate outcome count on the screen is $|\Omega_m|^{4^n}$.
Accordingly, define the protocol screen capacity in bits as
\begin{equation}
\label{eq:bh_capacity_I_prot_def}
I_{\mathrm{prot}}(m,n)
:=
\log_2\!\left(|\Omega_m|^{4^n}\right)
=
m\,4^n.
\end{equation}
This quantity is protocol-native and does not depend on any continuum dictionary.

\subsection{Black-hole boundary capacity (external target)}
\label{subsec:bh_capacity_bh}

\paragraph{Boundary information in bits.}
\MatchTag Under the Planck-information dictionary (Assumption~\ref{ass:bh_planck_units_info}) and the semiclassical area law (Assumption~\ref{ass:bh_area_law}),
the black-hole boundary information capacity in bits is
\begin{equation}
\label{eq:bh_capacity_I_BH_def}
I_{\mathrm{BH}}(A)
:=
\frac{S_{\mathrm{BH}}}{k_B\ln 2}
=
\frac{A}{4\ell_P^2\ln 2}.
\end{equation}
When the Schwarzschild package is used (Assumption~\ref{ass:bh_schwarzschild_package}), this becomes an explicit function of $M$ via $A(M)$.

\subsection{Finite-family CAP calibration and capacity-calibrated uplift path}
\label{subsec:bh_capacity_cap_calibration}

\paragraph{Candidate family.}
\AuditTag We restrict attention to a bounded, auditable candidate family
\[
m\in\{6,8,10,12,14,16\},
\qquad
n\in\{3,4,5,6\}.
\]

\paragraph{Mismatch functional.}
\AuditTag For a target mass $M$ (or area $A(M)$), define the log-mismatch
\begin{equation}
\label{eq:bh_capacity_mismatch_def}
\Delta(m,n\mid M)
:=
\left|\log\!\left(\frac{I_{\mathrm{prot}}(m,n)}{I_{\mathrm{BH}}(A(M))}\right)\right|.
\end{equation}

\paragraph{Deterministic closure rule.}
\AuditTag For each $M$ in a declared finite test family, CAP selects the unique minimizer of the lexicographic key
\[
K_{\mathrm{cap}}(m,n\mid M)
:=
\Bigl(\Delta(m,n\mid M),\ n,\ m\Bigr),
\]
which enforces: (i) smallest mismatch first, then (ii) lower addressing complexity, then (iii) lower local alphabet length.
This yields an auditable mapping $M\mapsto (m^\ast(M),n^\ast(M))$ within the bounded family.

\paragraph{Capacity-calibrated uplift path.}
\AuditTag Fix the paper's anchor $(m_0,n_0)=(6,3)$.
Define the anchor capacity $I_{\mathrm{prot}}(6,3)$.
Within a finite dyadic mass family $M=\alpha m_P$ declared by the generator, select a reference mass $M_{\mathrm{ref}}$ as the smallest $\alpha$ such that
$I_{\mathrm{BH}}(M_{\mathrm{ref}})\ge I_{\mathrm{prot}}(6,3)$.
Then, for each uplift window $m\in\{6,8,10,12,14,16\}$, define a capacity-calibrated addressing choice $n(m)$ by minimizing the same log-mismatch \eqref{eq:bh_capacity_mismatch_def}
within the bounded $n$-family at fixed $m$ against the fixed reference budget $I_{\mathrm{BH}}(M_{\mathrm{ref}})$.
This produces a closed, finite uplift path $(m,n(m))$ that can be used as a reproducible interface calibration for uplift-dependent constructions.
