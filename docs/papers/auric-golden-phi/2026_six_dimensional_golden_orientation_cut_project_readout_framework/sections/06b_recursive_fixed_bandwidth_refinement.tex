\section{固定带宽递归读出与分割细化（\texorpdfstring{$m=6$}{m=6}）}

本节在主文固定带宽 $m=6$ 的前提下，将“递归观测”严格表述为协议层的分割细化过程：不通过增大 $m$，而通过对同一底系统施加逐级条件化读出，形成嵌套纤维链，从而提升有效分辨率。

\subsection{对象层定义}

\begin{definition}[第一级输出]\label{def:fw-rec-level0}
主文固定 $m=6$。记稳定化输出为
$$
x^{(0)}_t:=\Fold_6(\omega_t^{(6)})\in X_6^{\mathrm Z},
$$
其中 $\omega_t^{(6)}\in\{0,1\}^6$ 为长度 $6$ 的滑动窗口词。
\end{definition}

\begin{definition}[第 $k$ 级条件读出与输出]\label{def:fw-rec-levelk}
给定历史输出
$$
x^{(0:k-1)}=(x^{(0)},x^{(1)},\dots,x^{(k-1)})\in (X_6^{\mathrm Z})^{k},
$$
定义第 $k$ 级的条件读出协议为在该历史条件下的可测读出
$$
q_6^{(k)}:X\to \{0,1\}^6,
$$
并定义第 $k$ 级输出
$$
x^{(k)}=\Fold_6\!\left(q_6^{(k)}(\cdot)\right)\in X_6^{\mathrm Z}.
$$
\end{definition}

\begin{definition}[嵌套纤维链]\label{def:fw-rec-nested}
递归观测诱导嵌套链
$$
\mathcal{F}^{(0)}\supseteq \mathcal{F}^{(1)}\supseteq \cdots \supseteq \mathcal{F}^{(K)},
$$
其中 $\mathcal{F}^{(0)}=\Fold_6^{-1}(x^{(0)})$，而 $\mathcal{F}^{(k)}$ 表示在前级条件化下的可见等价类预像集合。
\end{definition}

\subsection{可核查命题}

\begin{proposition}[R1：递归读出等价于分割的逐级细化]\label{prop:fw-rec-R1}
递归读出产生的历史条件化等价于对可观测 $\sigma$-代数施加逐级细化：递归深度 $K$ 对应一个嵌套分割族，其柱集族随 $K$ 单调细化。
\end{proposition}

\begin{proposition}[R2：条件不确定性的单调性]\label{prop:fw-rec-R2}
令 $G$ 为目标细节量，则条件不确定性（例如条件熵）
$$
H(G\mid x^{(0:K)})
$$
随递归深度 $K$ 非增。
\end{proposition}

\begin{proposition}[R3：与增大 $m$ 的关系（仅信息量对照）]\label{prop:fw-rec-R3}
递归深度增加与一次性增大窗口长度 $m$ 均可提升可区分信息量。本文仅将二者作为实证对照，不在一般性上宣称严格等价。
\end{proposition}
