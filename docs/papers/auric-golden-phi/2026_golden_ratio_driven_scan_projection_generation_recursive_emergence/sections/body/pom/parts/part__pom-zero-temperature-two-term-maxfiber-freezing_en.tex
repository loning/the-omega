\paragraph{Zero-Temperature Two-Term Expansion and Maximum Fiber Set Freezing: Maximum Fiber Entropy, Top Gap, and Auditable Threshold}

\begin{definition}[Entropy Rate of Maximum Fiber Achievers]\label{def:pom-max-fiber-entropy-rate}
Following the notation for maximum fiber multiplicity \(M_m:=\max_{x\in X_m}d_m(x)\),
for each resolution \(m\) define the maximum fiber achiever set
\[
\mathcal{M}_m:=\{x\in X_m:\ d_m(x)=M_m\},
\qquad
N_m:=|\mathcal{M}_m|.
\]
Define its topological entropy (counting entropy rate)
\[
\boxed{\;
h_\ast:=\limsup_{m\to\infty}\frac{1}{m}\log N_m\; }.
\]
\end{definition}

\begin{theorem}[Zero-Temperature Two-Term Expansion: Linearization Intercept and Maximum Fiber Entropy for High-Order Pressure]\label{thm:pom-zero-temperature-two-term-expansion-maxfiber-entropy}
Under the spectral equality/Legendre duality regime of Theorem \ref{thm:pom-ldp-uniform-and-wm-exact-spectrum},
let \(f\) be the counting spectrum (multifractal spectrum) given by that theorem.
And let
\(
\alpha_\ast=\sup\{\alpha:\ f(\alpha)>-\infty\}
\)
be the spectrum endpoint (equivalently \(\alpha_\ast=\lim_{q\to\infty}P_q/q\), see Theorem \ref{thm:pom-multiplicative-refinement-telescope-alpha}).
Then the limit
\[
\lim_{q\to\infty}\bigl(P_q-q\alpha_\ast\bigr)
\]
exists and satisfies
\[
\boxed{\;
\lim_{q\to\infty}\bigl(P_q-q\alpha_\ast\bigr)
=
f(\alpha_\ast)
=
h_\ast\; }.
\]
Thus the zero-temperature two-term expansion holds
\[
\boxed{\;
P_q=q\alpha_\ast+h_\ast+o(1)\qquad(q\to\infty)\; }.
\]
\end{theorem}
\begin{proof}
By the spectral equality of Theorem \ref{thm:pom-ldp-uniform-and-wm-exact-spectrum},
\(\tau(q)=\lim_{m\to\infty}\frac{1}{m}\log S_q(m)\) exists and satisfies
\[
\tau(q)=\sup_{\alpha\in[0,\log 2]}\bigl(q\alpha+f(\alpha)\bigr).
\]
At integer points \(q\in\NN\) we have \(P_q=\tau(q)\), so
\[
P_q-q\alpha_\ast
=
\sup_{\alpha\le \alpha_\ast}\bigl(q(\alpha-\alpha_\ast)+f(\alpha)\bigr).
\]
Since \(f\) is an upper semicontinuous concave function, and \(\alpha\le \alpha_\ast\), when \(q\to\infty\) the supremum in the above expression can only be contributed by regions where \(\alpha\) approaches the endpoint \(\alpha_\ast\), thus the limit exists and equals \(f(\alpha_\ast)\).
\par
On the other hand, by Definition \ref{def:pom-max-fiber-entropy-rate},
\(\mathcal{M}_m\) is precisely the level set where \(Y_m(x)=\frac1m\log d_m(x)\) achieves its maximum:
if we denote \(\alpha_m:=\frac1m\log M_m=\max_x Y_m(x)\), then
\(
\mathcal{M}_m=\{x:\ Y_m(x)=\alpha_m\}
\)
and \(\alpha_m\to\alpha_\ast\).
Therefore the natural upper semicontinuous extension of the endpoint spectral value \(f(\alpha_\ast)\) can be taken as the counting entropy rate of this maximal level set, i.e., \(f(\alpha_\ast)=h_\ast\).
Substituting back yields the conclusion. \qed
\end{proof}

\begin{corollary}[Spectral Formula for Maximum Fiber Subshift]\label{cor:pom-max-fiber-subshift-spectral-entropy}
If the stable type language "achieving maximum local growth" is generated by some subshift of finite type \(\Sigma_{\max}\subseteq\Sigma\) (given by finite-step transition constraints),
and \(N_m\) is consistent with the length-\(m\) admissible word count of \(\Sigma_{\max}\) on the exponential scale,
then there exists a finite adjacency matrix \(A^{\max}\) such that
\[
\boxed{\;
h_\ast=\log\rho(A^{\max})\; }.
\]
\end{corollary}
\begin{proof}
The word count growth rate of a subshift of finite type is given by the Perron root of its adjacency matrix (Perron--Frobenius; see \cite{LindMarcus1995}),
and \(h_\ast\) is identified by the exponential rate consistency of Definition \ref{def:pom-max-fiber-entropy-rate} as this growth rate. \qed
\end{proof}

\begin{proposition}[Top Gap and Escort Freezing Threshold: Exponential Bound on Maximum Set Mass]\label{prop:pom-escort-freezing-gap-threshold}
For each \(m\) define the second-largest fiber multiplicity
\[
M_m^{(2)}:=\max\{d_m(x):\ x\in X_m\setminus\mathcal{M}_m\},
\]
and let the top gap exponent
\[
\delta:=\liminf_{m\to\infty}\frac{1}{m}\log\frac{M_m}{M_m^{(2)}}\ \in[0,\infty].
\]
For any fixed \(q>0\), let the escort distribution \(\pi_{m,q}(x):=d_m(x)^q/S_q(m)\).
Then we have the identity
\[
\pi_{m,q}(\mathcal{M}_m)=\frac{N_m M_m^q}{S_q(m)}.
\]
And under the type exponential rate gauge \(\lim_{m\to\infty}\frac{1}{m}\log|X_m|=\log\varphi\),
the upper bound holds
\[
\boxed{\;
1-\pi_{m,q}(\mathcal{M}_m)
\ \le\
\exp\!\Bigl(-m\bigl(q\delta-(\log\varphi-h_\ast)\bigr)+o(m)\Bigr)\; }.
\]
Therefore when \(q\delta>\log\varphi-h_\ast\), the escort family concentrates exponentially fast to the maximum fiber achiever set as \(m\to\infty\).
\end{proposition}
\begin{proof}
By the decomposition
\[
S_q(m)=\sum_{x\in\mathcal{M}_m}M_m^q+\sum_{x\notin\mathcal{M}_m}d_m(x)^q
=N_m M_m^q+\sum_{x\notin\mathcal{M}_m}d_m(x)^q
\]
and the upper bound \(d_m(x)\le M_m^{(2)}\) (when \(x\notin\mathcal{M}_m\)), we obtain
\[
\sum_{x\notin\mathcal{M}_m}d_m(x)^q\le |X_m|\,(M_m^{(2)})^q.
\]
Substituting into \(\pi_{m,q}(\mathcal{M}_m)=N_mM_m^q/S_q(m)\) gives
\[
1-\pi_{m,q}(\mathcal{M}_m)
\le
\frac{|X_m|}{N_m}\left(\frac{M_m^{(2)}}{M_m}\right)^q.
\]
Taking logarithm of the right side, dividing by \(m\) and taking \(\limsup\), then using
\(
\lim_{m\to\infty}\frac{1}{m}\log|X_m|=\log\varphi
\),
\(
\limsup_{m\to\infty}\frac{1}{m}\log N_m=h_\ast
\)
and the definition of \(\delta\) yields the shown exponential bound. \qed
\end{proof}

\begin{remark}[Top Gap Degeneracy and "Flat-Top Layer" Mechanism]\label{rem:pom-top-gap-degenerate-flat-top}
When \(\delta=0\), the exponential bound of Proposition \ref{prop:pom-escort-freezing-gap-threshold} degenerates to a trivial upper bound for any fixed \(q\),
thus "fixed temperature (fixed \(q\)) escort freezing to \(\mathcal{M}_m\) as \(m\to\infty\)" is no longer enforced by the top gap.
In this case, the limiting behavior of the escort depends on the counting spectrum \(f(\alpha)\) of the near-top layer sets and their endpoint geometry:
the zero-temperature intercept is still given by the endpoint value \(f(\alpha_\ast)=h_\ast\) (Theorem \ref{thm:pom-zero-temperature-two-term-expansion-maxfiber-entropy}),
while for fixed \(q\) the concentration/non-concentration is governed by the competition between "entropy (layer set complexity)--energy (exponential difference)."
\end{remark}
