\section{Scale map and small-loop regularity contract (CL2--CL3)}
\label{app:scale_map_and_small_loop_regularity_contract}

This appendix records the theorem-facing conditions under which finite loop observables can be interpreted as sampling \emph{small} plaquettes of side length $a$ in a continuum dictionary.
In the present manuscript, any numerical identification of loop observables with a continuum action density requires an explicit scale map and a regularity bundle; otherwise the use remains representative/matching only (CL2--CL3 in Appendix~\ref{app:minimal_failure_point_templates}).

\subsection{Scale map: from protocol loop scale to continuum length}
\begin{definition}[Scale map (protocol $\rightarrow$ continuum)]
\label{def:scale_map_protocol_to_continuum}
\MathTag Fix a refinement chain $(m_k,n_k)$ and a corresponding family of loop observables (unit plaquettes, $k\times k$ loops, etc.).
A \emph{scale map} is a choice of positive numbers $a_{k}>0$ (an ``effective plaquette side length'') assigned to the loop family at each scale $k$.
\end{definition}

\begin{assumption}[Bounded scale-map family (CL2)]
\label{ass:bounded_scale_map_family}
\AuditTag When continuum identification is attempted, the manuscript must declare a \emph{finite} candidate family of scale maps $\{a_k^{(j)}\}_{j\in\mathcal{J}}$ and report sensitivity under that family.
No unconstrained fitting of $a_k$ is permitted.
\end{assumption}

\paragraph{A canonical finite family on the balanced chain (normalized).}
\AuditTag For the balanced chain $m=2n$ used in Appendix~\ref{app:holonomy_sweeps_extended}, we fix a finite candidate family by normalizing to the anchor $a(3)=1$ (i.e.\ $(n,m)=(3,6)$) and declaring three scale-map candidates:
\begin{itemize}
\item \textbf{(S1) grid proxy:} $a(n):=2^{-(n-3)}$,
\item \textbf{(S2) window proxy:} $a(n):=2^{-(m-6)}=2^{-2(n-3)}$,
\item \textbf{(S3) geometric-mean proxy:} $a(n):=\sqrt{a_{\mathrm{S1}}(n)a_{\mathrm{S2}}(n)}=2^{-3(n-3)/2}$.
\end{itemize}

\subsection{Regularity bundle: when the small-plaquette expansion is valid}
\begin{assumption}[Small-loop regularity bundle (CL3)]
\label{ass:small_loop_regularity_bundle}
\MathTag Assume the continuum representative includes a Lie-algebra-valued connection one-form $A$ defined on a neighborhood of the point(s) being sampled, with:
\begin{itemize}
\item $A$ is $C^2$ in a coordinate chart (or admits the equivalent Sobolev regularity sufficient for the expansion),
\item local bounds exist on $A$ and its first two derivatives in the sampled neighborhood,
\item the loop family corresponds to axis-aligned plaquettes of side length $a$ in that chart, with $a$ in a regime where the Taylor/BCH remainders are controlled.
\end{itemize}
\end{assumption}

\paragraph{Where the regularity bundle can come from in this manuscript.}
\AuditTag In the AQFT/Wightman bridge, regularity is packaged by the reconstruction bundle (Assumption~\ref{ass:reconstruction_bundle_template}) and its energy-bound component (RC3) in Appendix~\ref{app:field_reconstruction_theorems}.
Positive-energy control enters via the spectrum condition (Appendix~\ref{app:microcausality_spectrum_covariance}) or its auditable windowed surrogate (Appendix~\ref{app:spectrum_surrogate_contract}).
If these inputs are not discharged in the stated window/regime, CL3 triggers and the small-loop expansion must be treated as a proxy-level dictionary only.

\begin{proposition}[Usable form of the small-plaquette bridge]
\label{prop:usable_small_plaquette_bridge}
\MathTag Under Assumption~\ref{ass:small_loop_regularity_bundle}, Theorem~\ref{thm:wilson_small_plaquette_expansion} applies with a remainder bound of the form
\[
\left|
1-\frac{1}{N}\RePart\Tr(U_\square)
-\frac{a^{4}}{2N}\Tr\!\bigl(F_{\mu\nu}(x)F^{\mu\nu}(x)\bigr)
\right|
\le C\,a^{6},
\]
where the constant $C$ depends only on local bounds on $A$ and its first two derivatives in the sampled neighborhood.
\end{proposition}

\AuditTag Proposition~\ref{prop:usable_small_plaquette_bridge} is the precise theorem-facing bridge point.
If the bundle cannot be discharged, the manuscript must treat $1-\frac1N\Re\Tr(U_\square)$ only as a finite proxy and record the downgrade as CL3.
