\begin{corollary}[Twofold law in the dense Toeplitz regime: condition-number exponent and volume-entropy-density exponent are of the same type, with ratio tending to \texorpdfstring{$2$}{2}]\label{cor:xi-hellinger-toeplitz-dense-2x-law}
Under equally spaced depths \(\sigma_j=\sigma_0+(j-1)\Delta\), let \(G_\kappa(\Delta)\) be the corresponding Toeplitz principal block, and define the volume-entropy density
\[
e(\Delta):=\lim_{\kappa\to\infty}-\frac1\kappa\log\det G_\kappa(\Delta)
=-\frac{1}{2\pi}\int_{-\pi}^{\pi}\log\sigma_\Delta(\theta)\,\dd\theta.
\]
Let the bilateral Toeplitz convolution operator \(T_\Delta:\ell^2(\ZZ)\to\ell^2(\ZZ)\) be defined by
\[
(T_\Delta x)_n:=\sum_{m\in\ZZ}A((n-m)\Delta)\,x_m
\]
Then its symbol is \(\sigma_\Delta\), and its condition number satisfies
\[
\mathrm{cond}(T_\Delta)=\frac{\max_{\theta\in[-\pi,\pi]}\sigma_\Delta(\theta)}{\min_{\theta\in[-\pi,\pi]}\sigma_\Delta(\theta)},
\qquad
c(\Delta):=\log\mathrm{cond}(T_\Delta).
\]
As \(\Delta\downarrow0\), one has
\[
\boxed{\;
e(\Delta)=\frac{\pi^2}{\Delta}+O\!\Bigl(\log\frac1\Delta\Bigr),\qquad
c(\Delta)=\frac{2\pi^2}{\Delta}+O\!\Bigl(\log\frac1\Delta\Bigr),\qquad
\frac{c(\Delta)}{e(\Delta)}\to 2.\;}
\]
Hence the exponential threshold for stable invertibility and the bulk log-volume (entropy density) are controlled by the same constant \(2\pi^2/\Delta\), and the boundary exponent is strictly twice the bulk exponent.
\end{corollary}
\begin{proof}
By Proposition \ref{prop:xi-hellinger-toeplitz-symbol-poisson},
\[
\sigma_\Delta(\theta)=\frac{1}{\Delta}\sum_{k\in\ZZ}\frac{\pi^{2}}{\cosh^{2}\!\Bigl(\pi\frac{\theta+2\pi k}{\Delta}\Bigr)}.
\]
Denote the principal term by
\(
\sigma_\Delta^{(0)}(\theta):=\frac{\pi^{2}}{\Delta}\cosh^{-2}(\pi\theta/\Delta)
\).
For \(\theta\in[-\pi,\pi]\), when \(k\neq0\), \(|\theta+2\pi k|\ge\pi\), and using \(\cosh t\ge \tfrac12 e^{|t|}\),
\[
0\le \sigma_\Delta(\theta)-\sigma_\Delta^{(0)}(\theta)
\le \frac{1}{\Delta}\sum_{k\neq0}4\pi^{2}\exp\!\Bigl(-2\pi\frac{|\theta+2\pi k|}{\Delta}\Bigr)
\le C\,\frac{\pi^{2}}{\Delta}\,e^{-2\pi^{2}/\Delta},
\]
where \(C>0\) is independent of \(\Delta\).
Also, by monotonicity of \(\cosh^{-2}\),
\(
\sigma_\Delta^{(0)}(\theta)\ge \sigma_\Delta^{(0)}(\pi)=\frac{\pi^{2}}{\Delta}\cosh^{-2}(\pi^{2}/\Delta)
\asymp \frac{\pi^{2}}{\Delta}e^{-2\pi^{2}/\Delta}
\),
hence \(\log\sigma_\Delta(\theta)=\log\sigma_\Delta^{(0)}(\theta)+O(1)\) uniformly, so \(e(\Delta)=e_0(\Delta)+O(1)\), where
\[
e_0(\Delta):=-\frac{1}{2\pi}\int_{-\pi}^{\pi}\log\sigma_\Delta^{(0)}(\theta)\,\dd\theta
=-\log\frac{\pi^{2}}{\Delta}+\frac{1}{\pi}\int_{-\pi}^{\pi}\log\cosh\!\Bigl(\pi\frac{\theta}{\Delta}\Bigr)\,\dd\theta.
\]
Using \(x-\log2\le \log\cosh x\le x\) (\(x\ge0\)),
\[
\frac{1}{\Delta}\int_{-\pi}^{\pi}|\theta|\,\dd\theta-\log\frac{4\pi^{2}}{\Delta}
\ \le\ e_0(\Delta)\ \le\
\frac{1}{\Delta}\int_{-\pi}^{\pi}|\theta|\,\dd\theta-\log\frac{\pi^{2}}{\Delta},
\]
and since \(\int_{-\pi}^{\pi}|\theta|\,\dd\theta=\pi^{2}\), one gets \(e(\Delta)=\pi^{2}/\Delta+O(\log(1/\Delta))\).
\par
For the condition number, note that \(T_\Delta\) is multiplication by \(\sigma_\Delta(\theta)\) on the Fourier side, hence \(\mathrm{cond}(T_\Delta)=\max\sigma_\Delta/\min\sigma_\Delta\).
From \(\sigma_\Delta(0)=\frac{\pi^{2}}{\Delta}+O\!\bigl(\frac{\pi^{2}}{\Delta}e^{-4\pi^{2}/\Delta}\bigr)\) and
\(
\sigma_\Delta(\pi)\asymp \frac{\pi^{2}}{\Delta}e^{-2\pi^{2}/\Delta}
\)
one obtains
\(
c(\Delta)=\log\frac{\max\sigma_\Delta}{\min\sigma_\Delta}
=\frac{2\pi^{2}}{\Delta}+O(\log(1/\Delta))
\),
hence \(c(\Delta)/e(\Delta)\to2\). QED.
\end{proof}

\begin{corollary}[Linear law between bit budget and volume-entropy density: minimal effective precision for stable inversion]\label{cor:xi-hellinger-toeplitz-bit-budget-linear-law}
Under the setting of Corollary \ref{cor:xi-hellinger-toeplitz-dense-2x-law}, if stable numerical inversion is required in relative-error sense and effective precision is measured by \(B\) bits, then a necessary condition is
\[
\boxed{\;
B\ \gtrsim\ \log_2\mathrm{cond}(T_\Delta)=\frac{c(\Delta)}{\ln 2}
\ \sim\ \frac{2\pi^{2}}{\Delta\ln 2}\qquad(\Delta\downarrow0).\;}
\]
Combining with \(c(\Delta)\sim 2e(\Delta)\) yields the parameter-free linear law
\[
B\ \gtrsim\ \frac{2}{\ln 2}\,e(\Delta).
\]
Therefore, in the dense limit, any scheme that attempts to increase resolution at fixed precision must trigger an exponential blow-up of the condition number.
\end{corollary}
\begin{proof}
The classical necessary condition for stable inversion is that machine precision \(\varepsilon_{\mathrm{mach}}\) satisfy \(\varepsilon_{\mathrm{mach}}\lesssim \mathrm{cond}(T_\Delta)^{-1}\).
With \(B\)-bit effective precision, \(\varepsilon_{\mathrm{mach}}\asymp 2^{-B}\), hence \(B\gtrsim \log_2\mathrm{cond}(T_\Delta)=c(\Delta)/\ln 2\).
Applying the asymptotics in Corollary \ref{cor:xi-hellinger-toeplitz-dense-2x-law} yields the claim. QED.
\end{proof}
