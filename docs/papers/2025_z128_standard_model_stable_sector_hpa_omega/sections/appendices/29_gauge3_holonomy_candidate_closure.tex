\section{Holonomy-driven candidate-family closure for the three-factor gauge identification (supplement)}
\label{app:gauge3_holonomy_candidate_closure}

\AuditTag This appendix records an audit-facing interface route that supplies a protocol-internal source for the candidate-family component (G3) in Proposition~\ref{prop:channel_to_gauge}.
It does not add new axioms beyond tick and CAP; it only fixes an explicit interface rule (a modeling dictionary) that maps finite holonomy diagnostics to constraints on a bounded continuous candidate family.
The resulting non-abelian factor choice is then closed by CAP within the declared bounded enumeration.

\subsection{Protocol inputs: nontrivial holonomy and a CP-odd invariant}
\label{subsec:gauge3_holonomy_inputs}

At the minimal holonomy anchor $(m,n)=(6,3)$, Section~\ref{sec:protocol_connections_holonomy} provides a deterministic $S_4$-valued edge connection and its plaquette holonomy distribution, summarized by cycle type in Table~\ref{tab:holonomy_cycle_types}.
The same section provides two minimal bridges:
\begin{itemize}
  \item a real $SO(3)\subset SU(3)$ representation skeleton (Table~\ref{tab:holonomy_su3_rotation}), in which the observed $3$- and $4$-cycles induce nontrivial rotation angles;
  \item a dyadic phase-register lift (Tables~\ref{tab:holonomy_phase_lift_j} and \ref{tab:holonomy_phase_lift_family}) producing effective $3\times 3$ unitary matrices together with a rephasing-invariant CP-odd quantity $J$.
\end{itemize}
In particular, the generated audit fragments show that:
\begin{itemize}
  \item nontrivial $3$- and $4$-cycle holonomies occur at $n=3$ (Table~\ref{tab:holonomy_cycle_types});
  \item the phase-lifted construction yields nonzero $J$ values supported on these nontrivial cycle types (Table~\ref{tab:holonomy_phase_lift_j}), and a bounded sweep over dyadic denominators exhibits a stable nonzero regime (Table~\ref{tab:holonomy_phase_lift_family}).
\end{itemize}

\subsection{A minimal interface rule: holonomy diagnostics constrain a bounded compact candidate family}
\label{subsec:gauge3_interface_rule}

\begin{definition}[Holonomy-to-candidate-family interface rule]
\label{def:holonomy_to_candidate_family_rule}
\InterfaceTag Fix the finite holonomy diagnostics at the anchor $(m,n)=(6,3)$ as recorded in Section~\ref{sec:protocol_connections_holonomy}.
We define the continuous gauge-factor candidate family by the following protocol-to-continuum dictionary constraints:
\begin{itemize}
  \item \textbf{(H1) An abelian rephasing sector.} Local rephasing redundancy is modeled by an abelian $U(1)$ connection sector (Proposition~\ref{prop:local_rephasing_connection}).
  \item \textbf{(H2) A non-abelian rotation skeleton.} Since the plaquette holonomy distribution contains $3$- and $4$-cycles (Table~\ref{tab:holonomy_cycle_types}) and these induce nontrivial rotations under the sign-twisted standard representation (Table~\ref{tab:holonomy_su3_rotation}), the continuum dictionary includes at least one connected compact non-abelian factor that admits a nontrivial $3$-dimensional real orthogonal action (rotation skeleton) compatible with these holonomy diagnostics.
  \item \textbf{(H3) A genuinely complex $3\times 3$ unitary sector with CP-odd support.} Since the dyadic phase-lift construction yields an effective $3\times 3$ unitary matrix with a nonzero CP-odd invariant $J$ on nontrivial plaquettes (Table~\ref{tab:holonomy_phase_lift_j}), the continuum dictionary includes a connected compact non-abelian factor that admits a $3$-dimensional complex unitary action supporting such a CP-odd invariant (in particular, not reducible to a purely real orthogonal $SO(3)$ action).
\end{itemize}
Within a declared bounded list of compact simple factors (Section~\ref{subsec:gauge3_bounded_enumeration}), we select the CAP-minimal non-abelian factors compatible with (H2)--(H3) under explicit complexity labels and tie-breaks (Appendix~\ref{app:gauge_complexity_sensitivity}).
\end{definition}

\subsection{Bounded enumeration and CAP closure (audit)}
\label{subsec:gauge3_bounded_enumeration}

We then apply deterministic CAP-style minimization with declared tie-breaks:
\begin{itemize}
  \item \textbf{(H2 proxy)} minimize $\dim(\mathfrak{g})$ among compact non-abelian simple factors (rotation-skeleton minimality);
  \item \textbf{(H3 proxy)} minimize $d_{\min}$, the minimal dimension of a faithful complex representation (up to finite quotients), among factors not isomorphic to the (H2) minimizer (genuinely complex $3\times 3$ minimality).

\subsection{Minimal failure points and falsifiability}
\label{subsec:gauge3_failure_falsifiability}

\AuditTag The interface rule of Definition~\ref{def:holonomy_to_candidate_family_rule} is falsifiable at the protocol layer; it can fail at explicit minimal points.
\begin{itemize}
  \item \textbf{(F1) Trivial holonomy.} If, under the declared deterministic connection rule and within the bounded counterfactual transport-rule audit, the plaquette holonomy distribution contains no nontrivial $3$- or $4$-cycles, then the rotation-skeleton requirement (H2) is unsupported. This failure mode is audited by the bounded sensitivity sweeps in Appendix~\ref{app:holonomy_sweeps_extended} (e.g.\ Table~\ref{tab:holonomy_transport_rule_sensitivity}) and by the balanced-chain sweep (Appendix~\ref{app:holonomy_sweeps_extended}).
  \item \textbf{(F2) CP-odd collapse under phase lift.} If, across the declared bounded phase-map family and dyadic denominator family, the phase-lifted construction yields either only Gram--Schmidt failures or only $J=0$ on nontrivial plaquettes, then the genuinely complex CP-support requirement (H3) is unsupported. This failure mode is audited by the phase-map and denominator sweeps (Table~\ref{tab:holonomy_phase_lift_family} and Appendix~\ref{app:holonomy_sweeps_extended}).
\end{itemize}
In either failure mode, one must revert to a conditional candidate-family route (e.g.\ Appendix~\ref{app:physics_consensus_inputs} or Appendix~\ref{app:internal_fiber_g2_optional}) for the gauge-factor identification.
