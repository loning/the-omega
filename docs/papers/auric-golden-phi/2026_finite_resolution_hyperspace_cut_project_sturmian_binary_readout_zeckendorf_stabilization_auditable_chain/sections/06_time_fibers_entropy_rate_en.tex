\section{Time Fibers: Preimage Structure and Information Rate}

\subsection{Time Fibers as ``Hidden Degrees of Freedom'' Container}

\begin{definition}[Time fiber]\label{def:time-fiber}
For $x\in X_m^{\mathrm Z}$, define its fiber (preimage set)
\[
\mathcal{F}_m(x):=\mathrm{Fold}_m^{-1}(x)\subset\{0,1\}^m.
\]
\end{definition}

\begin{remark}
$\mathcal{F}_m(x)$ records the set of original window readout microstates that could produce the given stable output $x$ under observation. The nontrivial nature of the fiber is the rigorous form of ``observation-induced irreversible compression.''
\end{remark}

\subsection{Historical Consistency Cylinder Sets and Posterior Contraction}

\begin{definition}[Historical consistency cylinder set]\label{def:cylinder-set}
Let the base system be $(\Omega,\mathcal{B},\mu,T)$, and the readout map produce the process $x_t$. For a given observation history $(x_0,\dots,x_t)$, define the cylinder set
\[
C_t(x_0^t):=\{\omega\in\Omega:\ x_0(\omega)=x_0,\dots,x_t(\omega)=x_t\}.
\]
\end{definition}

\begin{definition}[Time as internal timescale of information refinement]\label{def:internal-time-scale}
Let the prior be $\mu$. Define the posterior uncertainty
\[
H_t:=-\log\mu\bigl(C_t(x_0^t)\bigr).
\]
If $H_t$ exhibits linear growth in $t$ on typical orbits, then its growth rate serves as the ``internal time information rate'' characterizing observation refinement speed.
\end{definition}

\begin{remark}[Objectification caliber of the ``future'']
In this paper, the ``future'' is not treated as an external entity, but is objectified as the magnitude of conditional distribution uncertainty, for instance
$
H(x_{t+1}\mid x_{\le t})
$
or more generally the contraction degree of posterior cylinder set measure.
\end{remark}

\subsection{Theorem Characterization of Information Rate}

\begin{theorem}[Exponential law of cylinder set measure: SMB-type conclusion]\label{thm:smb}
If $T$ is ergodic on $(\Omega,\mu)$ and $x_t$ comes from a finite generating partition (or equivalently a generating $\sigma$-algebra), then for $\mu$-almost every orbit,
\[
\lim_{t\to\infty}\frac{1}{t}\Bigl(-\log\mu(C_t(x_0^t))\Bigr)=h_\mu(T),
\]
where $h_\mu(T)$ is the metric entropy.
\end{theorem}

\begin{corollary}[Protocol-internal characterization of the arrow of time]\label{cor:time-arrow}
Under the conditions of Theorem \ref{thm:smb}, cylinder set measures corresponding to typical observation histories contract at rate $e^{-t h_\mu}$; therefore, ``growth of time'' and ``exponential decrease of posterior set volume'' are equivalent within the protocol, quantitatively controlled by the entropy rate.
\end{corollary}

\begin{remark}
This conclusion belongs to the closed-layer theorem system; its physical interpretation (for instance, relating entropy rate to macroscopic irreversibility or thermodynamic arrow of time) must be handled as an interpretation-layer proposition, not entering the theorem chain.
\end{remark}
