\subsubsection{Contraction Constant Rigidity of the Golden Complementary Binary Channel: TV Isometry, Maximal Correlation, and Mutual Information SDPI}\label{subsubsec:pom-golden-channel-contraction-sdpi}

This paragraph treats the ``hidden bit observation'' as a fixed binary channel and compresses its contraction properties into three classes of auditable constants:
(i) the Dobrushin coefficient (TV contraction); (ii) the Hirschfeld--Gebelein--R\'enyi (HGR) maximal correlation; (iii) the strong data processing inequality (SDPI) constant for mutual information.
For the golden complementary pair \(p=\varphi^{-1}\), \(q=\varphi^{-2}\) (\(p-q=\varphi^{-3}\)), these constants undergo rigid collapse: the TV/correlation contraction factor is \(\varphi^{-3}\), and the mutual information contraction factor is \(\varphi^{-6}\).

\begin{theorem}[Contraction constant rigidity of the golden complementary binary channel: \texorpdfstring{$\delta=\rho_{\mathrm{m}}=\varphi^{-3}$}{delta=rho_m=phi^{-3}} and \texorpdfstring{$I$}{I}-contraction \texorpdfstring{$\varphi^{-6}$}{phi^{-6}}]\label{thm:pom-golden-binary-channel-contraction}
Let \(\varphi=\frac{1+\sqrt5}{2}\), and consider a channel \(W\) with binary input \(\Theta\in\{+,-\}\) and binary output \(B\in\{0,1\}\):
\[
W(B=1\mid \Theta=+)=\varphi^{-1},\qquad
W(B=1\mid \Theta=-)=\varphi^{-2}.
\]
Then the following hold:
\begin{enumerate}
  \item \textbf{(TV contraction/Dobrushin coefficient)}
  \[
  \delta(W)
  :=\sup_{\theta\neq\theta'}\mathrm{TV}\bigl(W(\cdot\mid\theta),W(\cdot\mid\theta')\bigr)
  =|\varphi^{-1}-\varphi^{-2}|
  =\varphi^{-3}.
  \]
  \item \textbf{(HGR maximal correlation)} The Hirschfeld--Gebelein--R\'enyi maximal correlation coefficient satisfies
  \[
  \rho_{\mathrm{m}}(W)=\varphi^{-3}.
  \]
  \item \textbf{(Exact TV scaling)} For any two input priors \(\mu,\nu\) (defined on \(\{+,-\}\)), the strict identity holds:
  \[
  \mathrm{TV}(\mu W,\nu W)=\varphi^{-3}\,\mathrm{TV}(\mu,\nu).
  \]
  \item \textbf{(Mutual information SDPI, optimal constant)} For any Markov chain \(U\to \Theta\to B\) (\(U\) arbitrary measurable), the sharp constant form holds:
  \[
  I(U;B)\le \varphi^{-6}\,I(U;\Theta),
  \]
  where \(\varphi^{-6}\) is the optimal (non-improvable) contraction factor for this channel, satisfying
  \[
  \varphi^{-6}=\rho_{\mathrm{m}}(W)^2=\delta(W)^2.
  \]
\end{enumerate}
\end{theorem}

\begin{proof}
Let \(p:=\varphi^{-1}\), \(q:=\varphi^{-2}=1-p\), and write \(\Delta:=p-q=\varphi^{-3}\).
\par
\textbf{(1)} For two-point distributions, \(\mathrm{TV}(\mathrm{Bern}(p),\mathrm{Bern}(q))=|p-q|=|\Delta|\), so \(\delta(W)=|\Delta|=\varphi^{-3}\).
\par
\textbf{(3)} Take any \(\mu(+)=a\), \(\nu(+)=b\), then
\(
\mu W=\mathrm{Bern}(q+a\Delta)
\), \(
\nu W=\mathrm{Bern}(q+b\Delta)
\).
Therefore
\[
\mathrm{TV}(\mu W,\nu W)=|(q+a\Delta)-(q+b\Delta)|=|\Delta|\,|a-b|=\varphi^{-3}\,\mathrm{TV}(\mu,\nu).
\]
\par
\textbf{(2)} Let \(X\in\{\pm 1\}\) be the sign encoding of \(\Theta\) (\(X=+1\iff \Theta=+\)), and let \(Y:=2B-1\in\{\pm 1\}\).
Then \(\mathbb{E}[Y\mid X=+1]=2p-1=\Delta\), \(\mathbb{E}[Y\mid X=-1]=2q-1=-\Delta\), i.e.,
\(\mathbb{E}[Y\mid X]=\Delta X\).
Under the uniform prior \(\mathbb{P}(X=+1)=\mathbb{P}(X=-1)=\tfrac12\), we immediately obtain
\(\mathrm{Corr}(X,Y)=\Delta\), whence by the definition of the HGR coefficient (taking the supremum over input priors) we have \(\rho_{\mathrm{m}}(W)\ge|\Delta|\).
On the other hand, for the binary symmetric channel the singular value computation of the \(2\times 2\) channel matrix yields \(\rho_{\mathrm{m}}(W)=|2p-1|=|p-q|=|\Delta|\).
Therefore \(\rho_{\mathrm{m}}(W)=|\Delta|=\varphi^{-3}\).
\par
\textbf{(4)} For any Markov chain \(U\to\Theta\to B\), the strong data processing inequality gives
\(
I(U;B)\le \rho_{\mathrm{m}}(W)^2 I(U;\Theta)
\).
Substituting \(\rho_{\mathrm{m}}(W)=\varphi^{-3}\) yields
\(
I(U;B)\le \varphi^{-6} I(U;\Theta)
\).
Optimality follows from the SDPI constant computation for the binary symmetric channel (equivalently from the contraction ratio of the mutual information quadratic form in the small perturbation limit); this constant equals exactly \(\rho_{\mathrm{m}}(W)^2\). \qed
\end{proof}

\subsubsection{\texorpdfstring{$\varphi$}{phi}-Logit Lattice: Translation Dynamics of Bayesian Updates on \texorpdfstring{$\mathbb{Z}$}{Z}}\label{subsubsec:pom-phi-logit-lattice-dynamics}

\begin{corollary}[\texorpdfstring{$\varphi$}{phi}-logit lattice: translation dynamics of Bayesian updates on \texorpdfstring{$\mathbb{Z}$}{Z}]\label{cor:pom-phi-logit-lattice-translation}
In the two-phase model of Proposition \ref{prop:pom-phi-logistic-posterior-grid}, take equal priors
\(\mathbb{P}(\Theta=\theta_+)=\mathbb{P}(\Theta=\theta_-)=\tfrac12\).
Define the posterior log-odds process
\[
Z_n:=\log_{\varphi}\frac{\mathbb{P}(\Theta=\theta_+\mid B_1^n)}{\mathbb{P}(\Theta=\theta_-\mid B_1^n)}.
\]
Then \(Z_n\in\mathbb{Z}\) and there exists an integer-valued representation with translation recursion:
\[
\boxed{\;Z_n=\sum_{i=1}^{n}(2B_i-1)\in\mathbb{Z},\qquad
Z_{n+1}=Z_n+(2B_{n+1}-1).\;}
\]
Consequently the posterior family forms a discrete orbit
\[
\boxed{\;\frac{\mathbb{P}(\Theta=\theta_+\mid B_1^n)}{\mathbb{P}(\Theta=\theta_-\mid B_1^n)}=\varphi^{Z_n},\qquad
\mathbb{P}(\Theta=\theta_+\mid B_1^n)=\frac{1}{1+\varphi^{-Z_n}}.\;}
\]
Therefore the inference process is algebraically equivalent to a multiplicative random walk over integer powers of the fundamental unit \(\varphi\); the reachable set of posteriors is the orbit image of \(\varphi^{\mathbb{Z}}\), exhibiting strict ``arithmetic auditability.''
\end{corollary}

\begin{proof}
By Proposition \ref{prop:pom-phi-logistic-posterior-grid},
\(
\frac{\mathbb{P}(\theta_+\mid B_1^n)}{\mathbb{P}(\theta_-\mid B_1^n)}=\varphi^{K_n}
\), where
\(
K_n=\sum_{i=1}^n(2B_i-1)
\).
Hence \(Z_n=\log_\varphi(\text{odds})=K_n\in\mathbb{Z}\), and the recursion and posterior closed form follow immediately. \qed
\end{proof}

\subsubsection{Chebyshev Certificate for Symmetric Double-Threshold Stopping Times: Generating Functions, Independence, and Finite Spectral Decomposition}\label{subsubsec:pom-chebyshev-stopping-time-spectral}

\begin{theorem}[Probability generating function of the symmetric double-threshold stopping time: Chebyshev preimage closed form]\label{thm:pom-symmetric-double-threshold-pgf-chebyshev}
Under the setting of Theorem \ref{thm:pom-sprt-symmetric-threshold-error}, let
\[
\tau_T:=\inf\{n\ge 0:\ |K_n|=T\}\qquad(T\in\mathbb{N}),
\]
and denote the Chebyshev polynomial of the first kind by \(T_T(x)=\cos\!\bigl(T\arccos x\bigr)\).
Then under the true phase \(\theta_+\), the probability generating function of \(\tau_T\) satisfies the strict closed form on \(|u|\le 1\):
\[
\boxed{\;
\mathbb{E}_{\theta_+}[u^{\tau_T}]
=\frac{\cosh\!\bigl(\tfrac{T}{2}\ln\varphi\bigr)}{T_T\!\bigl(\tfrac{\varphi^{3/2}}{2u}\bigr)}.\;}
\]
Equivalently, setting
\[
\xi(u):=\frac{\varphi^{3/2}}{2u}
\;+\;\sqrt{\Bigl(\frac{\varphi^{3/2}}{2u}\Bigr)^2-1}\qquad(|u|\le 1),
\]
we have
\[
\boxed{\;
\mathbb{E}_{\theta_+}[u^{\tau_T}]
=\frac{\varphi^{T/2}+\varphi^{-T/2}}{\xi(u)^T+\xi(u)^{-T}}.\;}
\]
\end{theorem}

\begin{proof}
Let \(p:=\mathbb{P}_{\theta_+}(Y_1=+1)=\varphi^{-1}\), \(q:=\mathbb{P}_{\theta_+}(Y_1=-1)=\varphi^{-2}\) (where \(Y_i:=2B_i-1\)), and for \(|k|<T\) define
\(
F_k(u):=\mathbb{E}_{\theta_+}[u^{\tau_T}\mid K_0=k]
\).
The boundary conditions are \(F_{\pm T}(u)=1\), while for \(|k|<T\) the one-step decomposition yields the second-order linear difference equation
\[
F_k(u)=u\bigl(pF_{k+1}(u)+qF_{k-1}(u)\bigr).
\]
The general solution takes the form \(F_k(u)=A r^k+B r^{-k}\), where \(r=r(u)\) satisfies the characteristic equation
\(
pu r^2-r+qu=0
\).
Rearranging to
\(
r+r^{-1}=\frac{1}{u\sqrt{pq}}=\frac{\varphi^{3/2}}{u}
\)
and setting
\(
\xi(u):=\frac{\varphi^{3/2}}{2u}+\sqrt{\bigl(\frac{\varphi^{3/2}}{2u}\bigr)^2-1}
\)
gives the two roots
\(
r_\pm=\sqrt{q/p}\,\xi(u)^{\pm 1}
\).
Substituting the boundary conditions and taking \(K_0=0\) (symmetric initial condition) yields
\[
F_0(u)=\frac{\cosh(\tfrac{T}{2}\ln(p/q))}{\frac12(\xi(u)^T+\xi(u)^{-T})}
=\frac{\varphi^{T/2}+\varphi^{-T/2}}{\xi(u)^T+\xi(u)^{-T}},
\]
where we used \(p/q=\varphi\). Applying the Chebyshev identity
\(
T_T(x)=\frac12\bigl(\xi^T+\xi^{-T}\bigr)
\)
(when \(x=\frac12(\xi+\xi^{-1})\)) and noting
\(
\frac12(\xi(u)+\xi(u)^{-1})=\frac{\varphi^{3/2}}{2u}
\)
yields the first form. \qed
\end{proof}

\begin{proposition}[Complete separation of the joint generating function: independence and constant ratio law for \texorpdfstring{$(\tau_T,K_{\tau_T})$}{(tauT,KtauT)}]\label{prop:pom-tau-terminal-sign-independence}
Under the setting of Theorem \ref{thm:pom-symmetric-double-threshold-pgf-chebyshev},
for any \(n\equiv T\pmod 2\) with \(n\ge T\), the strict constant ratio identity holds:
\[
\boxed{\;
\frac{\mathbb{P}_{\theta_+}(\tau_T=n,\ K_{\tau_T}=+T)}{\mathbb{P}_{\theta_+}(\tau_T=n,\ K_{\tau_T}=-T)}=\varphi^{T}\; }
\]
and this ratio is independent of \(n\).
Therefore the random variables \(\tau_T\) and the terminal sign
\(
S_T:=\mathrm{sign}(K_{\tau_T})\in\{+,-\}
\)
are \textbf{mutually independent}, and the joint probability generating function separates strictly as
\[
\boxed{\;
\mathbb{E}_{\theta_+}\!\left[u^{\tau_T}w^{K_{\tau_T}}\right]
=\mathbb{E}_{\theta_+}[u^{\tau_T}]\cdot\Bigl(\pi_T\,w^{T}+(1-\pi_T)\,w^{-T}\Bigr),\;}
\]
where
\(
\pi_T:=\mathbb{P}_{\theta_+}(K_{\tau_T}=+T)=\frac{\varphi^{T}}{1+\varphi^{T}}
\)
(see Theorem \ref{thm:pom-sprt-symmetric-threshold-error}).
In particular, for any integrable function \(f\) we have
\(
\mathbb{E}_{\theta_+}[f(\tau_T)\mid S_T]=\mathbb{E}_{\theta_+}[f(\tau_T)]
\).
\end{proposition}

\begin{proof}
Fix \(n\equiv T\ (\mathrm{mod}\ 2)\) with \(n\ge T\).
Consider the set \(\mathcal{P}_{n}^{+}\) of all paths starting from \(0\) that first reach \(+T\) at time \(n\), and the corresponding set \(\mathcal{P}_{n}^{-}\) of paths first reaching \(-T\).
Negating the step sequence of each path (\(+1\leftrightarrow -1\)) yields a bijection \(\mathcal{P}_{n}^{+}\leftrightarrow \mathcal{P}_{n}^{-}\).
For \(\omega\in\mathcal{P}_n^{+}\), let \(N_{+}\) and \(N_{-}\) denote the counts of positive and negative steps respectively, so \(N_{+}-N_{-}=T\) and
\(
\mathbb{P}_{\theta_+}(\omega)=p^{N_{+}}q^{N_{-}}
\).
Mapping to \(\omega'\in\mathcal{P}_n^{-}\) exchanges the step counts, hence
\[
\frac{\mathbb{P}_{\theta_+}(\omega)}{\mathbb{P}_{\theta_+}(\omega')}
=\Bigl(\frac{p}{q}\Bigr)^{N_{+}-N_{-}}
=\Bigl(\frac{p}{q}\Bigr)^{T}
=\varphi^{T}.
\]
Summing over \(\mathcal{P}_n^\pm\) yields the constant ratio law.
\par
From this ratio and
\(
\pi_T/(1-\pi_T)=\varphi^T
\)
(Theorem \ref{thm:pom-sprt-symmetric-threshold-error}),
we deduce that
\(
\mathbb{P}_{\theta_+}(\tau_T=n,S_T=+)/\mathbb{P}_{\theta_+}(\tau_T=n,S_T=-)
\)
equals \(\pi_T/(1-\pi_T)\) for every \(n\), whence
\(
\mathbb{P}_{\theta_+}(\tau_T=n,S_T=\pm)=\mathbb{P}_{\theta_+}(\tau_T=n)\mathbb{P}_{\theta_+}(S_T=\pm)
\), establishing independence.
The joint generating function separation is a direct equivalent reformulation of independence. \qed
\end{proof}

\begin{proposition}[Exact probability mass function of \texorpdfstring{$\tau_T$}{tauT}: finite spectral decomposition with explicit coefficients]\label{prop:pom-tau-pmf-finite-spectral-decomposition}
Let
\[
\theta_j:=\frac{(2j-1)\pi}{2T},\qquad
\lambda_j:=2\varphi^{-3/2}\cos\theta_j\quad (j=1,\dots,T),
\]
and define the coefficients
\[
c_j:=\frac{\cosh\!\bigl(\tfrac{T}{2}\ln\varphi\bigr)}{T}\,(-1)^{j-1}\tan\theta_j.
\]
Then for all \(n\ge 0\) the strict finite spectral decomposition holds:
\[
\boxed{\;\mathbb{P}_{\theta_+}(\tau_T=n)=\sum_{j=1}^{T} c_j\,\lambda_j^{n}.\;}
\]
Moreover, spectral symmetry automatically yields the support congruence constraint: when \(n\not\equiv T\pmod 2\) or \(n<T\), the right-hand side undergoes complete cancellation, so that \(\mathbb{P}_{\theta_+}(\tau_T=n)=0\).
\end{proposition}

\begin{proof}
By Theorem \ref{thm:pom-symmetric-double-threshold-pgf-chebyshev},
\(
G(u):=\mathbb{E}_{\theta_+}[u^{\tau_T}]
\)
is analytic on \(|u|\le 1\) and extends to the rational function
\[
G(u)=\frac{\cosh(\tfrac{T}{2}\ln\varphi)}{T_T(\varphi^{3/2}/(2u))}.
\]
The zeros of the Chebyshev polynomial are
\(
\cos\theta_j\ (j=1,\dots,T)
\), so \(G(u)\) has simple poles at
\(
u=1/\lambda_j=\frac{\varphi^{3/2}}{2\cos\theta_j}
\).
Performing a partial fraction decomposition of \(G(u)\) yields
\[
G(u)=\sum_{j=1}^{T}\frac{c_j}{1-\lambda_j u},
\]
where \(c_j\) can be computed explicitly via residues and simplified to the stated expression.
Expanding \(\frac{1}{1-\lambda_j u}=\sum_{n\ge 0}\lambda_j^n u^n\) and comparing power series coefficients gives
\(
\mathbb{P}_{\theta_+}(\tau_T=n)=\sum_j c_j\lambda_j^n
\).
The congruence/lower bound constraint follows from the spectral symmetry \(\lambda_{T+1-j}=-\lambda_j\) via parity cancellation. \qed
\end{proof}

\begin{proposition}[Second moment and variance closed form for \texorpdfstring{$\tau_T$}{tauT}: power-level rigidity of \texorpdfstring{$\varphi$}{phi}]\label{prop:pom-tau-second-moment-variance-closed}
Let \(T\ge 1\).
Then the variance of the stopping time \(\tau_T\) satisfies the strict closed form
\[
\boxed{\;
\mathrm{Var}_{\theta_+}(\tau_T)
=\frac{4T\,\varphi^{6}\bigl(\varphi^{2T}-1-T\varphi^{T}\bigr)}{(\varphi^{T}+1)^2}.\;}
\]
Furthermore, the second moment admits the closed form
\[
\boxed{\;
\mathbb{E}_{\theta_+}[\tau_T^2]
=\frac{\varphi^{6}}{(\varphi^{T}+1)^2}\Bigl(T^2(\varphi^{T}-1)^2+4T(\varphi^{2T}-1-T\varphi^{T})\Bigr).\;}
\]
In particular, as \(T\to\infty\) we have \(\mathrm{Var}_{\theta_+}(\tau_T)\sim 4\varphi^{6}T\).
\end{proposition}

\begin{proof}
By Theorem \ref{thm:pom-symmetric-double-threshold-pgf-chebyshev}, \(\mathbb{E}_{\theta_+}[u^{\tau_T}]\) is analytic in a neighborhood of \(u=1\), and
\[
\mathbb{E}_{\theta_+}[\tau_T]=\left.\frac{d}{du}\mathbb{E}_{\theta_+}[u^{\tau_T}]\right|_{u=1},\qquad
\mathbb{E}_{\theta_+}[\tau_T(\tau_T-1)]=\left.\frac{d^2}{du^2}\mathbb{E}_{\theta_+}[u^{\tau_T}]\right|_{u=1}.
\]
Differentiating the generating function at \(u=1\) and simplifying (using the explicit expressions for \(T_T'(x)\), \(T_T''(x)\), and \(\varphi^{3/2}/2=\tfrac12(\sqrt\varphi+\varphi^{-1/2})\)) yields the closed forms for \(\mathbb{E}_{\theta_+}[\tau_T^2]\) and \(\mathrm{Var}_{\theta_+}(\tau_T)\).
The asymptotic \(\mathrm{Var}_{\theta_+}(\tau_T)\sim 4\varphi^{6}T\) is read directly from the leading term. \qed
\end{proof}

\begin{theorem}[Central limit theorem in the large threshold limit: normalized fluctuation scale \texorpdfstring{$2\varphi^{3}\sqrt{T}$}{2 phi^3 sqrt(T)}]\label{thm:pom-tau-clt-large-threshold}
As \(T\to\infty\), the stopping time \(\tau_T\) (under the true phase \(\theta_+\)) satisfies the central limit theorem
\[
\boxed{\;
\frac{\tau_T-\mathbb{E}_{\theta_+}[\tau_T]}{2\varphi^{3}\sqrt{T}}\ \Longrightarrow\ \mathcal{N}(0,1).\;}
\]
Moreover, convergence holds in the second-moment sense:
\[
\boxed{\;\frac{\tau_T}{T}\xrightarrow{L^2}\varphi^{3}.\;}
\]
Therefore the complexity scale and random fluctuation scale for threshold \(T\) are \(T\) and \(\sqrt{T}\) respectively, with constants fully determined by \(\varphi\).
\end{theorem}

\begin{proof}
Let \(\mu:=\mathbb{E}_{\theta_+}[Y_1]=\varphi^{-3}\) and \(\sigma^2:=\mathrm{Var}_{\theta_+}(Y_1)=1-\mu^2=1-\varphi^{-6}\).
By the functional CLT for random walks with drift (Donsker's invariance principle) and the continuous mapping theorem for first passage times, the leading-order scale of \(\tau_T\) is \(T/\mu=\varphi^{3}T\), and the normalized fluctuation is
\(
\sigma\mu^{-3/2}\sqrt{T}=2\varphi^{3}\sqrt{T}
\)
(noting that \(\sigma^2\mu^{-3}=4\varphi^6\)).
This yields the stated CLT.
\par
The \(L^2\) convergence follows from \(\mathbb{E}[\tau_T]=\varphi^{3}T+o(T)\) (Theorem \ref{thm:pom-sprt-symmetric-threshold-mean-time}) combined with
\(\mathrm{Var}(\tau_T)=O(T)\) (Proposition \ref{prop:pom-tau-second-moment-variance-closed}). \qed
\end{proof}

\begin{proposition}[Exponential tail and analytic domain: principal eigenvalue \texorpdfstring{$2\varphi^{-3/2}\cos(\pi/2T)$}{2 phi^-3/2 cos(pi/2T)}]\label{prop:pom-tau-exponential-tail-analytic-disk}
Let
\[
\rho_T:=\max_{1\le j\le T}|\lambda_j|
=2\varphi^{-3/2}\cos\!\Bigl(\frac{\pi}{2T}\Bigr)\in(0,1),
\]
where \(\lambda_j\) is as defined in Proposition \ref{prop:pom-tau-pmf-finite-spectral-decomposition}.
Then there exist constants \(0<C_1(T)\le C_2(T)<\infty\) such that for all \(n\ge 0\) the two-sided exponential bound holds:
\[
\boxed{\;
C_1(T)\,\rho_T^{n}\ \le\ \mathbb{P}_{\theta_+}(\tau_T>n)\ \le\ C_2(T)\,\rho_T^{n}.\;}
\]
Equivalently, the probability generating function \(\mathbb{E}_{\theta_+}[u^{\tau_T}]\) has maximal analytic disk \(|u|<\rho_T^{-1}\) in the complex plane, with nearest singularity given by
\[
\boxed{\;
u_*=\rho_T^{-1}=\frac{\varphi^{3/2}}{2\cos(\pi/2T)}\;}
\]
\end{proposition}

\begin{proof}
By the spectral decomposition of Proposition \ref{prop:pom-tau-pmf-finite-spectral-decomposition},
\(
\mathbb{P}(\tau_T=n)
\)
is a linear combination of finitely many geometric terms \( \lambda_j^n \), with the largest modulus eigenvalue being \(\rho_T\) (corresponding to \(j=1\)).
Therefore the tail sum \(\mathbb{P}(\tau_T>n)=\sum_{m>n}\mathbb{P}(\tau_T=m)\) is likewise dominated by \(\rho_T^{n}\), and the finite spectral decomposition provides the upper and lower bound constants \(C_1(T),C_2(T)\).
\par
The analytic disk radius of the generating function is determined by its nearest pole; from the pole locations \(u=1/\lambda_j\) in the proof of Proposition \ref{prop:pom-tau-pmf-finite-spectral-decomposition}, the nearest singularity is \(u_*=1/\rho_T=\frac{\varphi^{3/2}}{2\cos(\pi/2T)}\). \qed
\end{proof}
