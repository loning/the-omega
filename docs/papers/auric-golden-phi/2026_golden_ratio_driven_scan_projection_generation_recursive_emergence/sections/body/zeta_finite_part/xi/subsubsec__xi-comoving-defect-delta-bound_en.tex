\subsubsection{Defect–Offset Quantification Under Comoving Horizon: Eliminating Height Redshift in Jensen Certificates}\label{subsubsec:xi-comoving-defect-delta-bound}
The previous subsection (\S\ref{subsubsec:xi-horizon-spectral-measure-tomography}) introduced the comoving translation parameter \(x_0\) and pointed out: the fixed chart at endpoint \(-1\) compresses off-critical-line offset into a boundary layer of thickness \(\Theta(\gamma^{-2})\), while comoving choice \(x_0=\gamma\) can lift this compression to first-order scale (see Remark \ref{rem:xi-horizon-comoving-vs-redshift}).
This subsection gives a direct corollary in the \emph{Jensen defect ledger} caliber under this localization: for disk-compactified completion functions constructed by translation families, their defect upper bound translates in comoving charts to an explicit offset upper bound \emph{without height factor}, yielding an equivalent criterion for RH.
\par
To avoid ambiguity, this section only uses the disk completion and Jensen defect general formulas already established within the text (Appendix \ref{app:unit-circle-phase-gate}), without introducing new external assumptions; comoving translation only changes charts, not the zero set of the object itself.

\paragraph{Shifted Cayley Charts and Translation Family Completion}
Continue with the disk-compactified completion interface of Appendix \ref{app:unit-circle-phase-gate}: the Cayley compactification \(\mathcal{C}:\mathbb{H}\to\mathbb{D}\) is given by \eqref{eq:app-cayley-disk}, and let \(s(w)=\tfrac12+\mathrm{i}\mathcal{C}^{-1}(w)\) (\(w\in\mathbb{D}\)).
For any \(x_0\in\RR\), define the shifted chart (equivalent to translation family \(\mathcal{C}_{x_0}(z):=\mathcal{C}(z-x_0)\))
\begin{equation}\label{eq:xi-comoving-shifted-s}
s_{x_0}(w):=\frac12+\mathrm{i}\,\bigl(x_0+\mathcal{C}^{-1}(w)\bigr)\qquad(w\in\mathbb{D}),
\end{equation}
and define the corresponding disk-compactified completion function
\begin{equation}\label{eq:xi-comoving-shifted-Fzeta}
F_{\zeta,x_0}(w):=\Xi_{\zeta}\!\bigl(s_{x_0}(w)\bigr)\qquad(w\in\mathbb{D}).
\end{equation}
Since \(\Xi_{\zeta}\) is entire and \(s_{x_0}\) is analytic in \(\mathbb{D}\), \(F_{\zeta,x_0}\) is analytic in \(\mathbb{D}\); and when \(x_0=0\) it degenerates to \(F_{\zeta}\) (formula \eqref{eq:xi-ramanujan-horizon-Fzeta}).
\par
For \(0<\varrho<1\), define the Jensen defect of translation family
\begin{equation}\label{eq:xi-comoving-shifted-jensen-defect}
D_{\zeta,x_0}(\varrho)
:=\frac{1}{2\pi}\int_{0}^{2\pi}\log\bigl|F_{\zeta,x_0}(\varrho e^{\mathrm{i}\theta})\bigr|\,\dd\theta-\log\bigl|F_{\zeta,x_0}(0)\bigr|\ \in\RR_{\ge 0}.
\end{equation}

\paragraph{Comoving Defect \texorpdfstring{$\Rightarrow$}{=>} Height-Free Offset Bound}
If \(\rho=\beta+\mathrm{i}\gamma\) is an off-critical-line zero of \(\Xi_{\zeta}\) with \(\beta<\tfrac12\), write \(\delta:=\tfrac12-\beta>0\).
By \eqref{eq:xi-comoving-shifted-s}--\eqref{eq:xi-comoving-shifted-Fzeta}, direct verification shows: its disk image under shifted chart \(x_0\) is
\begin{equation}\label{eq:xi-comoving-disk-image}
w_{\rho,x_0}=\mathcal{C}\bigl((\gamma-x_0)+\mathrm{i}\delta\bigr)\in\mathbb{D},
\end{equation}
satisfying \(F_{\zeta,x_0}(w_{\rho,x_0})=0\).
In particular, under comoving choice \(x_0=\gamma\),
\begin{equation}\label{eq:xi-comoving-pure-delta}
w_{\rho,\gamma}=\mathcal{C}(\mathrm{i}\delta)=\frac{1-\delta}{1+\delta}\in(0,1),
\qquad
\abs{w_{\rho,\gamma}}=\frac{1-\delta}{1+\delta}.
\end{equation}

\begin{theorem}[Comoving Defect Bound Implies Height-Free Offset Bound]\label{thm:xi-comoving-defect-implies-delta-bound}
Take \(x_0\in\RR\) and let \(F_{\zeta,x_0}\), \(D_{\zeta,x_0}\) be as defined in \eqref{eq:xi-comoving-shifted-Fzeta}--\eqref{eq:xi-comoving-shifted-jensen-defect}.
If there exist \(0<\varrho<1\) and \(\varepsilon\ge 0\) such that
\[
D_{\zeta,x_0}(\varrho)\le \varepsilon,
\]
then for any off-critical-line zero \(\rho=\beta+\mathrm{i}\gamma\) satisfying \(\beta<\tfrac12\) and \(\gamma=x_0\), its offset \(\delta=\tfrac12-\beta\) satisfies the explicit upper bound
\begin{equation}\label{eq:xi-comoving-delta-bound}
\boxed{\;
\delta\ \le\ \frac{1-\varrho e^{-\varepsilon}}{1+\varrho e^{-\varepsilon}}\,.
\;}
\end{equation}
\end{theorem}

\begin{proof}
Let \(\rho=\beta+\mathrm{i}x_0\) be an off-critical-line zero and write \(\delta=\tfrac12-\beta>0\).
By \eqref{eq:xi-comoving-disk-image} and \(\gamma=x_0\), we get that \(F_{\zeta,x_0}\) has a zero at point \(a:=w_{\rho,x_0}=w_{\rho,\gamma}\).
Applying Proposition \ref{prop:app-jensen-single-zero-lower-bound} (single zero Jensen lower bound) to the analytic function \(F_{\zeta,x_0}\), for any \(0<\varrho<1\), if \(|a|<\varrho\), then
\[
D_{\zeta,x_0}(\varrho)\ \ge\ \log\!\Bigl(\frac{\varrho}{|a|}\Bigr).
\]
Combining with the assumption \(D_{\zeta,x_0}(\varrho)\le\varepsilon\), we get \(|a|\ge \varrho e^{-\varepsilon}\).
On the other hand, by \eqref{eq:xi-comoving-pure-delta} we have \(|a|=\frac{1-\delta}{1+\delta}\).
Hence \(\frac{1-\delta}{1+\delta}\ge \varrho e^{-\varepsilon}\), rearranging yields \eqref{eq:xi-comoving-delta-bound}. This completes the proof.
\end{proof}

\begin{remark}[From Quadratic Redshift to First-Order Visibility: Comoving Upgrade of Defect Certificate]\label{rem:xi-comoving-defect-visibility-upgrade}
The informational meaning of Theorem \ref{thm:xi-comoving-defect-implies-delta-bound} is consistent with the depth comparison in Remark \ref{rem:xi-horizon-comoving-vs-redshift}: under fixed chart, off-line offset is compressed into endpoint boundary layer carrying \(\gamma^{-2}\) redshift suppression, while under comoving chart \(x_0=\gamma\), the offset quantity manifests at first-order scale via \(\abs{w_{\rho,\gamma}}=\frac{1-\delta}{1+\delta}\), thus translating the certificate of "defect upper bound \(\Rightarrow\) repulsion radius lower bound" (Theorem \ref{thm:dephys-defect-repulsion-radius}) into explicit offset upper bound \eqref{eq:xi-comoving-delta-bound} without height factor.
\end{remark}

\paragraph{RH's Comoving Defect Equivalence Criterion}
From Theorem \ref{thm:xi-comoving-defect-implies-delta-bound}, the following equivalent form follows directly; its structural point is: for any potential off-line zero \(\rho=\beta+\mathrm{i}\gamma\), one can take comoving parameter \(x_0=\gamma\) to read out its offset as disk one-dimensional radial quantity \(\abs{w_{\rho,\gamma}}\), thus downgrading "high-altitude shielding" from structural difficulty to chart choice problem.

\begin{theorem}[RH's Jensen Defect Uniform Smallness Criterion: Equivalent Certificate from Disk Sampling to Zero Exclusion]\label{thm:xi-rh-iff-uniform-comoving-defect}
The following propositions are equivalent:
\begin{enumerate}
  \item[(i)] RH holds;
  \item[(ii)] There exist \(\varrho_n\uparrow 1\), \(\varepsilon_n\downarrow 0\), such that for all \(n\) there is a uniform bound
  \[
  \sup_{x_0\in\RR}D_{\zeta,x_0}(\varrho_n)\le \varepsilon_n.
  \]
\end{enumerate}
\end{theorem}

\begin{proof}
\textbf{(i)\(\Rightarrow\)(ii).}
If RH holds, then all zeros of \(\Xi_{\zeta}\) satisfy \(\Re(\rho)=\tfrac12\), i.e., \(\delta=0\).
By \eqref{eq:xi-comoving-disk-image}, for any \(x_0\), zero images all fall on \(|w|=1\) without entering the interior of \(\mathbb{D}\), so \(F_{\zeta,x_0}\) has no zeros in \(\mathbb{D}\).
By Jensen summation formula (Proposition \ref{prop:app-jensen-formula-defect}), we get \(D_{\zeta,x_0}(\varrho)\equiv 0\) (\(\forall\,x_0,\varrho\)), thus (ii) holds.
\par
\textbf{(ii)\(\Rightarrow\)(i).}
Assume by contradiction RH does not hold, then there exists off-critical-line zero \(\rho=\beta+\mathrm{i}\gamma\) with \(\beta\neq\tfrac12\).
By functional equation symmetry, we can take \(\beta<\tfrac12\), thus \(\delta=\tfrac12-\beta>0\).
For any \(n\), take \(x_0=\gamma\) and by (ii) get \(D_{\zeta,\gamma}(\varrho_n)\le\varepsilon_n\).
Then by Theorem \ref{thm:xi-comoving-defect-implies-delta-bound},
\[
\delta\ \le\ \frac{1-\varrho_n e^{-\varepsilon_n}}{1+\varrho_n e^{-\varepsilon_n}}\xrightarrow[n\to\infty]{}0,
\]
contradicting \(\delta>0\).
Hence off-critical-line zeros do not exist, i.e., RH holds. This completes the proof.
\end{proof}
