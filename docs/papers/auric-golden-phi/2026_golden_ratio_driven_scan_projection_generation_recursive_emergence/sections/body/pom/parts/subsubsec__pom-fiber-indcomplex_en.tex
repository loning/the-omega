\subsubsection{Homotopy type of fiber independence complexes: the sphere--contractible dichotomy}\label{subsubsec:pom-fiber-indcomplex}

Under the sole structural input of the folding map \(\Fold_m:\Omega_m=\{0,1\}^m\twoheadrightarrow X_m\) generated by the local rewriting kernel \((011\leftrightarrow 100)\), this section extracts a collection of self-contained results:
the statements depend neither on the degenerate spectral histogram nor on any other extrinsic anchor or interpretive layer; they rely solely on the invertible-site structure of the induced graph \(\Gamma_x\) and the hypercube adjacency (random walk baseline).
First, the independent set system from Theorem \ref{thm:pom-fiber-indset-factorization} is lifted to the topological level: the family of independent sets is viewed as an abstract simplicial complex, and the topological complexity of the fiber is compressed via homotopy type and homology.
Under this kernel, the components of the candidate-site induced graph \(\Gamma_x\) are all paths, so the independence complex exhibits an extremely strong dichotomy: \emph{either contractible, or homotopy equivalent to a single sphere},
with the sphere dimension given by a closed-form parameter \(\tau(x)\) of the component lengths \((\ell_i)\) (Theorem \ref{thm:pom-fiber-independence-complex-classification}).

\begin{definition}[Independence complex]\label{def:pom-independence-complex}
For a finite simple graph \(G=(V,E)\), define its \textbf{independence complex} as the abstract simplicial complex
\[
\mathrm{Ind}^{\Delta}(G):=\{S\subseteq V:\ S\ \text{is an independent set of}\ G\},
\]
and denote its geometric realization by \(|\mathrm{Ind}^{\Delta}(G)|\).
Note that, by definition, the face set of \(\mathrm{Ind}^{\Delta}(G)\) coincides with the independent set family \(\mathsf{Ind}(G)\),
but in this section it is treated as a simplicial complex (rather than a mere set family) in order to discuss homotopy type and homology.
\end{definition}

\begin{lemma}[Independence complex of a path: the \(\bmod 3\) triperiodic dichotomy]\label{lem:pom-independence-complex-path}
Let \(P_{\ell}\) denote the path graph with \(\ell\) vertices (\(\ell\ge 0\)).
Then the homotopy type of \(|\mathrm{Ind}^{\Delta}(P_{\ell})|\) admits the following complete classification:
\[
|\mathrm{Ind}^{\Delta}(P_{\ell})|
\ \simeq\
\begin{cases}
\ast, & \ell\equiv 1\ (\mathrm{mod}\ 3),\\[2pt]
S^{\lfloor(\ell+1)/3\rfloor-1}, & \ell\not\equiv 1\ (\mathrm{mod}\ 3),
\end{cases}
\]
where \(\ast\) denotes a contractible space, and \(S^{-1}\) is used as endpoint notation for the \((-1)\)-dimensional sphere type of the empty complex (solely for unified notation).
\end{lemma}

\begin{proof}
This result can be obtained via the following recursive reduction; here we present an auditable recursive proof outline (the detailed steps can be completed using discrete Morse matching).
\par
\textbf{(1) Recursive reduction.}
For the path \(P_{\ell}\), take the endpoint \(v_1\) and its neighbor \(v_2\). Decomposing the independence complex according to whether the endpoint is included yields:
\[
\mathrm{Ind}^{\Delta}(P_{\ell})
\ =\
\mathrm{Ind}^{\Delta}(P_{\ell}\setminus v_1)\ \cup\
\bigl(v_1\ast \mathrm{Ind}^{\Delta}(P_{\ell}\setminus\{v_1,v_2\})\bigr),
\]
with the intersection being
\(
\mathrm{Ind}^{\Delta}(P_{\ell}\setminus\{v_1,v_2\})
\).
Applying a constructible homotopy equivalence to this three-part decomposition (viewing the right-hand cone as ``attaching a contractible cell''),
the homotopy type of \(\mathrm{Ind}^{\Delta}(P_{\ell})\) can be recursively reduced to a single suspension of \(\mathrm{Ind}^{\Delta}(P_{\ell-3})\), thereby yielding the triperiodic phenomenon.
\par
\textbf{(2) Base case verification.}
Direct enumeration gives the three homotopy types for \(\ell=0,1,2\) as
\(
S^{-1},\ \ast,\ S^{0}
\),
consistent with the stated classification.
\par
\textbf{(3) Inductive closure.}
From the recursive rule in (1) (``every three steps yield one suspension'') and the base cases in (2),
the complete \(\ell\bmod 3\) dichotomy and sphere dimension closed form follow.
This completes the proof.
\end{proof}

\begin{lemma}[Disjoint union and join: multiplicative structure of independence complexes]\label{lem:pom-independence-complex-join}
If \(G=\bigsqcup_{i=1}^r G_i\) is a disjoint union of graphs, then
\[
\mathrm{Ind}^{\Delta}(G)\ \cong\ \mathrm{Ind}^{\Delta}(G_1)\ast\cdots\ast \mathrm{Ind}^{\Delta}(G_r),
\]
where \(\ast\) denotes the simplicial join.
\end{lemma}
\begin{proof}
For a disjoint union \(G=\bigsqcup_i G_i\), an independent set \(S\subseteq V(G)\) is equivalent to choosing an independent set \(S_i\subseteq V(G_i)\) for each component and taking their union \(S=\bigsqcup_i S_i\).
This is precisely the face description of the join: \(\sigma\in K_1\ast\cdots\ast K_r\) if and only if \(\sigma=\bigsqcup_i \sigma_i\) with \(\sigma_i\in K_i\).
Hence the isomorphism holds. This completes the proof.
\end{proof}

\begin{theorem}[Complete classification of fiber independence complexes: single sphere or contractible]\label{thm:pom-fiber-independence-complex-classification}
Take any \(x\in X_m\), and adopt the candidate-site induced graph
\(\Gamma_x\) from Theorem \ref{thm:pom-fiber-indset-factorization}.
Suppose its connected components are isomorphic to paths \(P_{\ell_1},\dots,P_{\ell_r}\), respectively.
Define
\[
\tau(x):=\sum_{i=1}^r \left\lfloor\frac{\ell_i+1}{3}\right\rfloor.
\]
Let
\(
K_x:=\mathrm{Ind}^{\Delta}(\Gamma_x)
\)
denote the independence complex of \(\Gamma_x\). Then the following dichotomy holds:
\begin{enumerate}
  \item If some component satisfies \(\ell_i\equiv 1\pmod 3\), then \(|K_x|\) is contractible;
  \item Otherwise, \(|K_x|\simeq S^{\tau(x)-1}\), i.e., it is homotopy equivalent to a single sphere, with sphere dimension given in closed form by \(\tau(x)\).
\end{enumerate}
\end{theorem}
\begin{proof}
By Lemma \ref{lem:pom-independence-complex-join},
\(
K_x\cong \mathrm{Ind}^{\Delta}(P_{\ell_1})\ast\cdots\ast \mathrm{Ind}^{\Delta}(P_{\ell_r})
\).
If some component satisfies \(\ell_i\equiv 1\pmod 3\), then by Lemma \ref{lem:pom-independence-complex-path} that component is contractible, and the join of a contractible space with any space remains contractible,
so \(|K_x|\) is contractible.
\par
Otherwise, each component is homotopy equivalent to a sphere \(S^{\lfloor(\ell_i+1)/3\rfloor-1}\).
The join of spheres satisfies
\(
S^{a}\ast S^{b}\simeq S^{a+b+1}
\);
iterating over \(r\) components yields
\[
|K_x|\ \simeq\ S^{\sum_i(\lfloor(\ell_i+1)/3\rfloor-1)+(r-1)}
\ =\ S^{\tau(x)-1}.
\]
This completes the proof.
\end{proof}

\begin{corollary}[Simplicial homology rigidity of fibers: unique dimension, rank \(1\)]\label{cor:pom-fiber-independence-complex-homology}
Under the setting of Theorem \ref{thm:pom-fiber-independence-complex-classification},
if \(|K_x|\) is non-contractible, then its reduced homology satisfies
\[
\widetilde H_j\bigl(|K_x|;\mathbb Z\bigr)\ \cong\
\begin{cases}
\mathbb Z,& j=\tau(x)-1,\\
0,& j\neq \tau(x)-1.
\end{cases}
\]
In particular, when \(|K_x|\) is non-contractible, its reduced Euler characteristic is \(\widetilde\chi(|K_x|)=(-1)^{\tau(x)-1}\).
\end{corollary}
\begin{proof}
By Theorem \ref{thm:pom-fiber-independence-complex-classification}, in the non-contractible case \(|K_x|\simeq S^{\tau(x)-1}\);
the reduced homology and Euler characteristic of spheres are standard results. This completes the proof.
\end{proof}

\subsubsection{Distribution of inverse substitution counts: component independence, \texorpdfstring{$\varphi$}{phi}-closed-form densities, and real-rootedness}\label{subsubsec:pom-fiber-rewrite-statistics}

Theorem \ref{thm:pom-fiber-indset-factorization} identifies the fiber \(\Fold_m^{-1}(x)\) completely with the independent set system of the induced graph \(\Gamma_x\),
and provides a multiplicative factorization of the generating function \(Z_x(t)\).
This section reads this as an interface to ``intra-fiber statistical mechanics'': under the uniform distribution on the fiber,
the inverse substitution count \(r(\omega)\) (equivalent to the independent set size) exhibits strict component independence,
and in the large-path limit yields mean and variance densities in \(\varphi\)-algebraic form.

\begin{definition}[Intra-fiber inverse substitution count random variable]\label{def:pom-fiber-rewrite-rv}
Fix \(x\in X_m\) and equip the fiber \(\Fold_m^{-1}(x)\) with the uniform distribution.
Define the random variable
\[
R_x(\omega):=r(\omega)\qquad(\omega\in\Fold_m^{-1}(x)),
\]
i.e., the number of inverse substitutions \(100\mapsto 011\) used to generate \(\omega\) from \(x\).
Under the canonical bijection of Theorem \ref{thm:pom-fiber-indset-factorization}, \(R_x\) equals the size of the corresponding independent set.
\end{definition}

\begin{proposition}[Component-independent decomposition: \(R_x\) as a sum of independent terms]\label{prop:pom-fiber-rewrite-independent-sum}
Let \(\Gamma_x\cong\bigsqcup_{i=1}^r P_{\ell_i}\).
Let \(R_{\ell}\) denote the independent set size under the uniform distribution on the independent set family \(\mathsf{Ind}(P_{\ell})\) of the path \(P_{\ell}\).
Then there exist mutually independent random variables \(R_{\ell_1},\dots,R_{\ell_r}\) such that
\[
R_x\ \overset{d}{=}\ \sum_{i=1}^r R_{\ell_i}.
\]
\end{proposition}
\begin{proof}
By Theorem \ref{thm:pom-fiber-indset-factorization},
\(
\Fold_m^{-1}(x)\cong \mathsf{Ind}(\Gamma_x)
\),
and when \(\Gamma_x=\bigsqcup_i P_{\ell_i}\) there is a set decomposition
\(
\mathsf{Ind}(\Gamma_x)\cong \prod_i \mathsf{Ind}(P_{\ell_i})
\).
Therefore the uniform distribution on the fiber is equivalent to the uniform distribution on this Cartesian product, so the component projections are independent and uniform;
since the independent set size is additive over components, the conclusion follows. This completes the proof.
\end{proof}

\begin{proposition}[Thermodynamic closed forms for path components: principal eigenvalue \(\lambda(t)\) and density constants]\label{prop:pom-path-indset-thermo-constants}
For the path independent set polynomial \(I_{\ell}(t)\) (Definition \ref{def:pom-fiber-adm-path}), the recurrence
\[
I_{\ell}(t)=I_{\ell-1}(t)+t\,I_{\ell-2}(t)\qquad(\ell\ge 2),
\qquad
I_0(t)=1,\ I_1(t)=1+t
\]
holds. Let
\[
\lambda(t):=\frac{1+\sqrt{1+4t}}{2}\qquad(t>0),
\]
then for fixed \(t>0\), \(I_{\ell}(t)=\lambda(t)^{\ell+O(1)}\).
Under the hard-core weight distribution
\(
\mathbb P_t(S)\propto t^{|S|}
\)
(\(S\in\mathsf{Ind}(P_{\ell})\)), the independent set size \(R_{\ell}^{(t)}:=|S|\) satisfies the linear scaling law
\[
\mathbb{E}_t[R_{\ell}^{(t)}]=\rho(t)\,\ell+O(1),\qquad
\mathrm{Var}_t(R_{\ell}^{(t)})=\sigma^2(t)\,\ell+O(1),
\]
where the density constants are
\[
\rho(t)=\frac{t\lambda'(t)}{\lambda(t)},\qquad
\sigma^2(t)=\bigl(t\partial_t\bigr)^2\log\lambda(t).
\]
In particular, for the uniform independent set distribution (\(t=1\)) the closed forms are
\[
\rho=\rho(1)=\frac{1}{\varphi^2+1},
\qquad
\sigma^2=\sigma^2(1)=\frac{3}{5\sqrt5\,\varphi}-\frac{1}{5\varphi^2},
\]
where \(\varphi=(1+\sqrt5)/2\).
\end{proposition}
\begin{proof}
The recurrence follows from a case analysis on whether the terminal vertex is selected (see also the proof of Theorem \ref{thm:pom-fiber-signed-index-periodicity}).
Its characteristic equation is \(z^2=z+t\), with principal root \(\lambda(t)\), so
\(
I_{\ell}(t)=A(t)\lambda(t)^{\ell}+B(t)\lambda_-(t)^{\ell}
\)
with \(|\lambda_-(t)|<\lambda(t)\) (\(t>0\)), giving \(\log I_{\ell}(t)=\ell\log\lambda(t)+O(1)\).
\par
For the distribution \(\mathbb P_t\), the standard identities
\(
\mathbb E_t[|S|]=t(\partial_t\log I_{\ell})(t)
\),
\(
\mathrm{Var}_t(|S|)=(t\partial_t)^2\log I_{\ell}(t)
\)
yield
\(
\mathbb E_t[|S|]=\ell\,t(\partial_t\log\lambda)(t)+O(1)
\),
\(
\mathrm{Var}_t(|S|)=\ell\,(t\partial_t)^2\log\lambda(t)+O(1)
\),
which is the stated density form.
Finally, substituting \(\lambda'(t)=1/\sqrt{1+4t}\) and evaluating at \(t=1\) gives the closed-form constants. This completes the proof.
\end{proof}

\begin{corollary}[The leading terms of mean and variance for general fibers are controlled by \(\sum \ell_i\)]\label{cor:pom-fiber-rewrite-mean-var-density}
Under the uniform fiber measure of Definition \ref{def:pom-fiber-rewrite-rv}, if \(\Gamma_x\cong\bigsqcup_{i=1}^r P_{\ell_i}\), then
\[
\mathbb E[R_x]=\rho\sum_{i=1}^r \ell_i+O(r),
\qquad
\mathrm{Var}(R_x)=\sigma^2\sum_{i=1}^r \ell_i+O(r),
\]
where \(\rho,\sigma^2\) are the constants from Proposition \ref{prop:pom-path-indset-thermo-constants} (at \(t=1\)).
\end{corollary}
\begin{proof}
By the independent sum decomposition of Proposition \ref{prop:pom-fiber-rewrite-independent-sum} and the linear scaling law of Proposition \ref{prop:pom-path-indset-thermo-constants}, summation over components yields the result. This completes the proof.
\end{proof}

\begin{theorem}[Real-rootedness \(\Rightarrow\) ultra-log-concavity: strong unimodality of the fiber inverse substitution count distribution]\label{thm:pom-fiber-rewrite-realroot-ulc}
For any \(\ell\ge 0\), the path independent set polynomial \(I_{\ell}(t)\) has only real negative roots.
Consequently, for any \(x\in X_m\), the fiber generating function
\(
Z_x(t)=\prod_{i=1}^r I_{\ell_i}(t)
\)
also has only real negative roots.
Therefore, the coefficient sequence \(\bigl([t^k]Z_x(t)\bigr)_{k\ge 0}\) is \textbf{ultra-log-concave},
and in particular satisfies Newton-type inequalities and (in the non-degenerate case) is strictly unimodal.
\end{theorem}
\begin{proof}
For the path \(P_{\ell}\), the real-rootedness of the independent set polynomial can be established via several equivalent routes:
one standard approach uses the fact that ``the independent set polynomial equals the matching polynomial on the line graph'' and invokes the real-rootedness theorem for matching polynomials;
alternatively, one may directly cite the real-rootedness result for independent set polynomials of claw-free graphs (paths being a special case).
Therefore \(I_{\ell}(t)\) has only real negative roots.
\par
By the factorization \(Z_x(t)=\prod_i I_{\ell_i}(t)\) from Theorem \ref{thm:pom-fiber-indset-factorization},
the union property of root sets immediately gives that \(Z_x(t)\) also has only real negative roots.
Finally, for polynomials with nonnegative coefficients and roots on the negative real axis, the Newton inequalities hold, yielding ultra-log-concavity and unimodality. This completes the proof.
\end{proof}

\begin{corollary}[Intra-fiber local CLT: normal discrete approximation in the central interval]\label{cor:pom-fiber-rewrite-local-clt}
Adopt the notation of Definition \ref{def:pom-fiber-rewrite-rv}.
If \(\mathrm{Var}(R_x)\to\infty\) (for example, when \(\sum_i \ell_i\to\infty\) with non-degenerate component sizes),
then the centered and standardized \(R_x\) satisfies the central limit theorem; furthermore, within the central window, a local limit theorem holds:
there exists an error bound such that \(\mathbb P(R_x=k)\) is uniformly approximated by the discretization of the corresponding normal density,
with leading-order error \(O(\mathrm{Var}(R_x)^{-1/2})\).
\end{corollary}
\begin{proof}
By Proposition \ref{prop:pom-fiber-rewrite-independent-sum}, \(R_x\) is a sum of independent components;
Proposition \ref{prop:pom-path-indset-thermo-constants} provides the linear growth density of the variance.
In the regime \(\mathrm{Var}(R_x)\to\infty\),
the CLT/local CLT criteria for independent sums apply;
alternatively, by exploiting the real-rootedness from Theorem \ref{thm:pom-fiber-rewrite-realroot-ulc},
one may view the distribution as coefficients of a real-rooted generating function and invoke the corresponding local limit theorem for real-rooted polynomials.
This completes the proof.
\end{proof}

\subsubsection{Fiber reconstruction graphs and the Fibonacci cube: Cartesian decomposition, cubical face counts, and Gray--Hamilton structure}\label{subsubsec:pom-fiber-reconstruction-cubical}

The independent set model not only provides fiber counts and distributions but also yields an \emph{auditable internal geometric object}:
taking ``single inverse substitution toggles'' as the edge relation, the fiber naturally carries a cubical graph structure.
Under the \((011\leftrightarrow 100)\) kernel, this structure decomposes strictly into a Cartesian product of Fibonacci cubes,
thereby automatically inheriting median/CAT(0) cubical geometry and an enumerable Hamilton structure.

\begin{definition}[Fiber reconstruction graph]\label{def:pom-fiber-reconstruction-graph}
Fix \(x\in X_m\). Under the canonical isomorphism
\(\Fold_m^{-1}(x)\cong \mathsf{Ind}(\Gamma_x)\)
from Theorem \ref{thm:pom-fiber-indset-factorization}, define the \textbf{fiber reconstruction graph} \(\mathcal F_x\) as the following undirected graph:
the vertex set is \(\mathsf{Ind}(\Gamma_x)\), and two vertices \(S,S'\) are adjacent if and only if their symmetric difference satisfies
\(
|S\triangle S'|=1
\)
(i.e., a single legal ``toggle'' changes the selection state of exactly one candidate site).
Via the above isomorphism, \(\mathcal F_x\) may also be viewed as the adjacency structure defined on \(\Fold_m^{-1}(x)\).
\end{definition}

\begin{definition}[Fibonacci cube]\label{def:pom-fibonacci-cube}
For an integer \(n\ge 0\), define the \textbf{Fibonacci cube} \(\Gamma_n\) as the following graph:
its vertices are binary words \(b_1\cdots b_n\in\{0,1\}^n\) of length \(n\) with no adjacent \(11\);
two vertices are adjacent if and only if they differ in exactly one position (Hamming distance \(1\)) and the result of the flip remains valid.
\end{definition}

\begin{theorem}[Cartesian product decomposition and cubical geometry closed forms]\label{thm:pom-fiber-reconstruction-cartesian-product}
Let \(\Gamma_x\cong\bigsqcup_{i=1}^r P_{\ell_i}\).
Then the fiber reconstruction graph admits the natural isomorphism
\[
\mathcal F_x\ \cong\ \prod_{i=1}^r \Gamma_{\ell_i},
\]
where the right-hand side is the Cartesian product of graphs.
Furthermore, if \(\dim_{\square}(G)\) denotes the dimension of the largest cubical subgraph of \(G\) (the largest \(k\) such that \(Q_k\) is an induced subgraph), then the closed forms hold:
\[
\dim_{\square}(\mathcal F_x)=\sum_{i=1}^r \left\lceil\frac{\ell_i}{2}\right\rceil,
\qquad
\mathrm{diam}(\mathcal F_x)=\sum_{i=1}^r \ell_i.
\]
\end{theorem}
\begin{proof}
For the disjoint union \(\Gamma_x=\bigsqcup_i P_{\ell_i}\), the independent set family decomposes as a Cartesian product,
and the ``symmetric difference of size \(1\)'' adjacency relation corresponds precisely to flipping one position in a single component while keeping the others unchanged;
hence \(\mathcal F_x\) is the Cartesian product of the per-component independent set graphs.
On the other hand, the independent sets of the path \(P_{\ell}\) can be encoded as valid binary words of length \(\ell\) (whether each vertex is selected), adjacent if and only if one bit is flipped and the result remains valid,
so the component graph is isomorphic to \(\Gamma_{\ell}\), yielding the stated decomposition.
\par
For the dimension: a \(k\)-cubical subgraph of \(\Gamma_{\ell}\) corresponds to selecting \(k\) pairwise non-adjacent vertices in \(P_{\ell}\) as ``independently togglable coordinates,''
so the maximum \(k\) equals the maximum independent set size of the path, which is \(\lceil \ell/2\rceil\).
The cubical dimension of a Cartesian product is additive, giving \(\dim_{\square}(\mathcal F_x)=\sum_i\lceil\ell_i/2\rceil\).
\par
For the diameter: the two extremal vertices of \(\Gamma_{\ell}\) can be taken as the alternating words \(1010\cdots\) and \(0101\cdots\), which differ in all \(\ell\) positions,
and a legal path of length \(\ell\) can be constructed by flipping one position at a time, so \(\mathrm{diam}(\Gamma_{\ell})=\ell\);
the diameter of a Cartesian product equals the sum of the component diameters, giving \(\mathrm{diam}(\mathcal F_x)=\sum_i\ell_i\).
This completes the proof.
\end{proof}

where \(\epsilon\) is the empty word, \(\mathrm{rev}\) denotes reversal,
and \(0\,\mathrm{rev}(\mathcal G_{n-1})\) (resp.\ \(10\,\mathrm{rev}(\mathcal G_{n-2})\)) means first reversing the sequence and then prepending a common prefix to each word;
the concatenation symbol ``\(\cdot\)'' denotes sequence concatenation.
Then \(\mathcal G_n\) yields a Hamilton path of \(\Gamma_n\): the sequence covers all vertices and consecutive terms differ in exactly one position.

Denote by \(\mathrm{first}(\mathcal G_n)\) and \(\mathrm{last}(\mathcal G_n)\) the first and last terms of the sequence.
From the recursive formula and the base cases \(\mathcal G_0=(\epsilon)\), \(\mathcal G_1=(0,1)\), one can inductively establish the endpoint identity
\(
\mathrm{first}(\mathcal G_{n-1})=0\,\mathrm{last}(\mathcal G_{n-2})
\).
Therefore the last term of the first segment is
\(
0\,\mathrm{first}(\mathcal G_{n-1})=00\,\mathrm{last}(\mathcal G_{n-2})
\),
and the first term of the second segment is
\(
10\,\mathrm{last}(\mathcal G_{n-2})
\),
which differ only in the first position, hence have Hamming distance \(1\).
Thus concatenation yields a Hamilton path covering all vertices with consecutive entries differing in one position. This completes the proof.
\end{proof}

\subsubsection{Fold factor chain of the hypercube random walk: reversibility, zero self-loops, and spectral constant lower bounds}\label{subsubsec:pom-fold-factor-chain}

This section provides a probability-theoretic interface decoupled from ``folding details'': taking the simple random walk on \(\Omega_m=\{0,1\}^m\) as the microscopic baseline,
the folding map \(\Fold_m\) is treated as an observation map, and the factor chain is defined on the stable layer \(X_m\).
The results show that under the natural fiber-uniform conditioning, this factor chain is not only reversible and loop-free,
but its Poincar\'e and log-Sobolev constants are strictly \emph{no worse than} those of the hypercube random walk itself.

\begin{definition}[Hypercube simple random walk and fold factor chain]\label{def:pom-fold-factor-chain}
On the microstate space \(\Omega_m=\{0,1\}^m\), take the loop-free simple random walk kernel \(P^{\square}_m\):
for \(\omega,\omega'\in\Omega_m\), let
\[
P^{\square}_m(\omega,\omega')
:=
\begin{cases}
1/m,& \mathrm{Ham}(\omega,\omega')=1,\\
0,& \text{otherwise}.
\end{cases}
\]
This chain has the uniform distribution \(\mu_m(\omega)=2^{-m}\) as its stationary distribution and is reversible.
\par
Define the \textbf{fold factor chain} on the stable layer as the pushforward--fiber-uniform-lift compression kernel
\[
\overline P^{\square}_m\ :=\ P_m\circ P^{\square}_m\circ Q_m:\Delta(X_m)\to\Delta(X_m),
\]
where \(P_m=(\Fold_m)_*\) and \(Q_m\) is the fiber-uniform lift (see \S\ref{subsec:pom-information-ledger}).
In terms of transition probabilities, this is equivalent to: first drawing a uniform microstate \(\omega\in\Fold_m^{-1}(x)\) from the fiber of the current stable type \(x\),
performing one step of the hypercube flip, and then reading off the stable type of the next step.
\end{definition}

\begin{proposition}[Reversibility and zero self-loops]\label{prop:pom-fold-factor-chain-reversible-noloop}
Let
\(
\pi_m(x):=d_m(x)/2^m
\)
be the pushforward distribution of the uniform microstate under folding.
Then \(\pi_m\) is the stationary distribution of \(\overline P^{\square}_m\) and satisfies detailed balance (reversibility).
Moreover, for any \(x\in X_m\),
\[
\overline P^{\square}_m(x,x)=0.
\]
\end{proposition}
\begin{proof}
By Definition \ref{def:pom-fold-factor-chain},
\(
Q_m\pi_m
\)
is precisely the uniform distribution on \(\Omega_m\): for any \(\omega\),
\(
(Q_m\pi_m)(\omega)=\pi_m(\Fold_m(\omega))/d_m(\Fold_m(\omega))=(d_m/2^m)/d_m=2^{-m}
\).
Therefore
\[
\pi_m\overline P^{\square}_m
=
P_m\bigl((Q_m\pi_m)P^{\square}_m\bigr)
=
P_m(Q_m\pi_m)
=\pi_m,
\]
i.e., \(\pi_m\) is stationary.
Similarly, from the reversibility of \(P^{\square}_m\) under \(\mu_m\) and the compression form \(\overline P^{\square}_m=P_mP^{\square}_mQ_m\),
for any \(x,y\) the flow \(\pi_m(x)\overline P^{\square}_m(x,y)\) can be written as a cross-fiber undirected edge count and symmetry yields detailed balance.
\par
For zero self-loops: if \(\omega,\omega'\in\Omega_m\) are hypercube-adjacent (Hamming distance \(1\)), then
\(
N(\omega')-N(\omega)=\pm F_{k+1}
\)
for some \(1\le k\le m\).
By Lemma \ref{lem:pom-one-bit} (single hidden bit decomposition), the value difference between two points in the same stable type \(x\) fiber can only be \(0\) or \(\pm F_{m+2}\).
Since \(F_{k+1}<F_{m+2}\), it is impossible that \(\Fold_m(\omega')=\Fold_m(\omega)\).
Therefore a single hypercube flip necessarily changes the stable type, equivalent to \(\overline P^{\square}_m(x,x)=0\). This completes the proof.
\end{proof}

\begin{definition}[Spectral gap and log-Sobolev constant]\label{def:pom-gap-lsi}
For a finite-state reversible Markov kernel \(P\) (stationary distribution \(\pi\)), define the Dirichlet form
\[
\mathcal E_P(f,f):=\frac12\sum_{x,y}\pi(x)P(x,y)\bigl(f(x)-f(y)\bigr)^2.
\]
The \textbf{spectral gap} and \textbf{log-Sobolev constant} are defined as
\[
\mathrm{gap}(P):=\inf_{f\not\equiv \mathrm{const}}\frac{\mathcal E_P(f,f)}{\mathrm{Var}_{\pi}(f)},
\qquad
\alpha(P):=\inf_{f\not\equiv \mathrm{const}}\frac{\mathcal E_P(f,f)}{\mathrm{Ent}_{\pi}(f^2)},
\]
where
\(
\mathrm{Ent}_{\pi}(g):=\mathbb E_{\pi}[g\log g]-\mathbb E_{\pi}[g]\log\mathbb E_{\pi}[g]
\).
\end{definition}

\begin{theorem}[Spectral constant domination of the factor chain: no worse than the hypercube]\label{thm:pom-fold-factor-chain-gap-lsi}
Under the setting of Definition \ref{def:pom-fold-factor-chain},
\[
\mathrm{gap}(\overline P^{\square}_m)\ \ge\ \mathrm{gap}(P^{\square}_m)=\frac{2}{m},
\qquad
\alpha(\overline P^{\square}_m)\ \ge\ \alpha(P^{\square}_m)=\frac{1}{m}.
\]

\end{theorem}
\begin{proof}
Take any \(f:X_m\to\mathbb R\) and define its fiber-constant lift \(\widetilde f:=f\circ \Fold_m:\Omega_m\to\mathbb R\).
From \(\mu_m=Q_m\pi_m\) and the definition of the compression kernel \(\overline P^{\square}_m=P_mP^{\square}_mQ_m\),
one can directly verify (by expanding the summation fiber by fiber):
\[
\mathrm{Var}_{\pi_m}(f)=\mathrm{Var}_{\mu_m}(\widetilde f),\qquad
\mathrm{Ent}_{\pi_m}(f^2)=\mathrm{Ent}_{\mu_m}(\widetilde f^{\,2}),
\qquad
\mathcal E_{\overline P^{\square}_m}(f,f)=\mathcal E_{P^{\square}_m}(\widetilde f,\widetilde f).
\]
Therefore, for any non-constant \(f\), the Rayleigh quotient satisfies
\[
\frac{\mathcal E_{\overline P^{\square}_m}(f,f)}{\mathrm{Var}_{\pi_m}(f)}
=
\frac{\mathcal E_{P^{\square}_m}(\widetilde f,\widetilde f)}{\mathrm{Var}_{\mu_m}(\widetilde f)}
