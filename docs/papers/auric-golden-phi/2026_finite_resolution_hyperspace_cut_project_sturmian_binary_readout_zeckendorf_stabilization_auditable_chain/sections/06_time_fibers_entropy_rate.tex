\section{时间纤维：预像结构与信息率}

\subsection{预像纤维作为“隐藏自由度”容器}

\begin{definition}[时间纤维]\label{def:time-fiber}
对 $x\in X_m^{\mathrm Z}$，定义其纤维（预像集合）
\[
\mathcal{F}_m(x):=\mathrm{Fold}_m^{-1}(x)\subset\{0,1\}^m.
\]
\end{definition}

\begin{remark}
$\mathcal{F}_m(x)$ 记录在给定稳定输出 $x$ 的条件下，可能产生该输出的原始窗口读出微观态集合。纤维的非平凡性是“观测不可逆压缩”的严格形式。
\end{remark}

\subsection{历史一致性柱集与后验收缩}

\begin{definition}[历史一致性柱集]\label{def:cylinder-set}
设底层系统为 $(\Omega,\mathcal{B},\mu,T)$，读出映射产生过程 $x_t$。
对给定观测历史 $(x_0,\dots,x_t)$，定义柱集
\[
C_t(x_0^t):=\{\omega\in\Omega:\ x_0(\omega)=x_0,\dots,x_t(\omega)=x_t\}.
\]
\end{definition}

\begin{definition}[时间作为信息精化的内部标尺]\label{def:internal-time-scale}
设先验为 $\mu$。定义后验不确定度
\[
H_t:=-\log\mu\bigl(C_t(x_0^t)\bigr).
\]
若 $H_t$ 在典型轨道上随 $t$ 增长呈线性增长，则其增长率作为“内部时间信息率”刻画观测精化速度。
\end{definition}

\begin{remark}[“未来”的对象化口径]
在本文中，“未来”不被当作外加实体，而被对象化为条件分布的不确定性规模，例如
$
H(x_{t+1}\mid x_{\le t})
$
或更一般的后验柱集测度的收缩程度。
\end{remark}

\subsection{信息率的定理刻画}

\begin{theorem}[柱集测度的指数律：SMB 型结论]\label{thm:smb}
若 $T$ 在 $(\Omega,\mu)$ 上遍历，且 $x_t$ 来自有限生成分割（或等价的生成 $\sigma$-代数），则对 $\mu$-几乎处处的轨道，
\[
\lim_{t\to\infty}\frac{1}{t}\Bigl(-\log\mu(C_t(x_0^t))\Bigr)=h_\mu(T),
\]
其中 $h_\mu(T)$ 为度量熵。
\end{theorem}

\begin{corollary}[时间箭头的协议内刻画]\label{cor:time-arrow}
在定理 \ref{thm:smb} 条件下，典型观测历史对应的柱集测度按 $e^{-t h_\mu}$ 速率收缩；因此，“时间的增长”与“后验集合体积的指数级下降”在协议内等价，并由熵率定量控制。
\end{corollary}

\begin{remark}
该结论属于闭层定理体系；其物理解释（例如将熵率与宏观不可逆性或热力学时间箭头关联）须作为解释层命题处理，不进入定理链条。
\end{remark}
