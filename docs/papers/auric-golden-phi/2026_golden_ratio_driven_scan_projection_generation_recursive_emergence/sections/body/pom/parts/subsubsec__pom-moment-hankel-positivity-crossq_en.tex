\subsubsection{Moment-Order Positivity and Cross-Order Convexity: Hankel Positive Semi-Definiteness, Rational Generating Functions, and Spectral Constraints}\label{subsubsec:pom-moment-hankel-positivity-crossq}
This paragraph characterizes the \textbf{moment-order (\(q\)-axis) structure} at fixed resolution \(m\): the power-sum moments \(\{S_q(m)\}_{q\ge 0}\) are interpreted as the moment sequence of a discrete measure,
from which Hankel positivity yields cross-order discrete convexity, aligning with the pressure/spectral radius fingerprints of the main text (Proposition \ref{prop:pom-pressure-convexity}).
In addition, we record the standard spectral decomposition protocol ``finite-state matrix power readout \(\Rightarrow\) rational generating function \(\Rightarrow\) full-order Jordan expansion,''
so that the consequences of the recurrence/kernel realization can be directly organized into a verifiable chain of asymptotic formulas.

\begin{proposition}[Hankel positive semi-definiteness and nonnegativity of principal minors for power-sum moments]\label{prop:pom-power-sum-hankel-psd}
Fix a resolution \(m\ge 1\), and let the fiber multiplicity of the folding \(\Fold_m:\Omega_m\twoheadrightarrow X_m\) be
\(
d_m(x):=\card{\Fold_m^{-1}(x)}\ge 1
\).
For integer \(q\ge 0\) define the power-sum moment
\[
S_q(m):=\sum_{x\in X_m} d_m(x)^q,
\qquad
S_0(m)=|X_m|,
\qquad
S_1(m)=\sum_x d_m(x)=|\Omega_m|=2^m.
\]
Then for any integer \(r\ge 0\), the Hankel matrix
\[
H^{(m)}_r:=\bigl[S_{i+j}(m)\bigr]_{0\le i,j\le r}
\]
is positive semi-definite; consequently all its principal minors are nonnegative.
\end{proposition}
\begin{proof}
For any \(c=(c_0,\dots,c_r)^{\mathsf T}\in\RR^{r+1}\),
\[
c^{\mathsf T}H^{(m)}_r c
=\sum_{i,j=0}^{r} c_ic_jS_{i+j}(m)
=\sum_{x\in X_m}\Bigl(\sum_{i=0}^{r}c_i\,d_m(x)^i\Bigr)^2
\ge 0,
\]
whence \(H^{(m)}_r\succeq 0\).
Any principal submatrix of a positive semi-definite matrix is itself positive semi-definite, so its determinant is nonnegative. \qed
\end{proof}

\begin{corollary}[Cross-\(q\) discrete log-convexity chain and ratio monotonicity]\label{cor:pom-crossq-logconvex-chain}
Under the setting of Proposition \ref{prop:pom-power-sum-hankel-psd}, for any integer \(q\ge 1\),
\[
\boxed{\;
S_q(m)^2\ \le\ S_{q-1}(m)\,S_{q+1}(m)\; }.
\]
More generally, for any integers \(q\ge r\ge 1\),
\[
\boxed{\;
S_q(m)^2\ \le\ S_{q-r}(m)\,S_{q+r}(m)\; }.
\]
Hence the ratio
\(
R_q(m):=S_{q+1}(m)/S_q(m)
\)
is nondecreasing in \(q\ge 0\) (i.e., \(R_{q}(m)\ge R_{q-1}(m)\) for all \(q\ge 1\)).
\end{corollary}
\begin{proof}
The first inequality follows directly from the nonnegativity of the \(2\times 2\) principal minor \(\det\bigl[\begin{smallmatrix}S_{q-1}(m)&S_q(m)\\ S_q(m)&S_{q+1}(m)\end{smallmatrix}\bigr]\ge 0\).
For general \(r\ge 1\), applying the Cauchy--Schwarz inequality to the vectors \((d_m(x)^{(q-r)/2})_x\) and \((d_m(x)^{(q+r)/2})_x\) yields
\[
S_q(m)=\sum_x d_m(x)^{(q-r)/2}d_m(x)^{(q+r)/2}
\le \sqrt{S_{q-r}(m)}\,\sqrt{S_{q+r}(m)},
\]
and squaring gives the second inequality.
Finally, from \(S_q(m)^2\le S_{q-1}(m)S_{q+1}(m)\) we obtain
\(
S_{q+1}(m)/S_q(m)\ge S_q(m)/S_{q-1}(m)
\),
i.e., \(R_q(m)\ge R_{q-1}(m)\). \qed
\end{proof}

\paragraph{Prony--Hankel reconstruction: the fiber spectrum at fixed \(m\) is completely determined by finitely many \(q\)-moments}
Fix a resolution \(m\) and consider the value set of fiber multiplicities
\[
\mathcal{D}_m:=\{d_m(x):\ x\in X_m\}=\{a_1<\cdots<a_{r_m}\}\subset\NN,
\qquad
n_i:=\#\{x\in X_m:\ d_m(x)=a_i\}.
\]
Then for all integers \(q\ge 0\), there is a finite exponential-sum representation
\[
S_q(m)=\sum_{i=1}^{r_m} n_i\,a_i^{\,q}.
\]

\begin{theorem}[Minimal order and Hankel rank]\label{thm:pom-fiber-spectrum-prony-hankel-rank}
In the above setting, \(r_m\) equals the rank of the infinite Hankel matrix \(\bigl[S_{i+j}(m)\bigr]_{i,j\ge 0}\).
Equivalently, \(r_m\) is the order of the minimal-order linear recurrence satisfied by the sequence \(q\mapsto S_q(m)\) (with \(q\) as the time variable):
there exist a unique minimal integer \(r=r_m\) and constants \(c_0,\dots,c_{r-1}\) such that for all \(q\ge 0\),
\[
\boxed{\;
S_{q+r}(m)+c_{r-1}S_{q+r-1}(m)+\cdots+c_0 S_q(m)=0\; }.
\]
\end{theorem}
\begin{proof}
From the representation \(S_q(m)=\sum_{i=1}^{r_m}n_i a_i^q\) (with \(a_i\) pairwise distinct and nonzero), the sequence \((S_q(m))_{q\ge 0}\) is a finite sum of \(r_m\) exponential terms,
hence satisfies a linear recurrence of order at most \(r_m\): take
\(
Q_m(t):=\prod_{i=1}^{r_m}(t-a_i)=t^{r_m}+c_{r_m-1}t^{r_m-1}+\cdots+c_0
\),
then for each \(i\), \(a_i^{q+r_m}+c_{r_m-1}a_i^{q+r_m-1}+\cdots+c_0 a_i^q=0\);
multiplying by \(n_i\) and summing over \(i\) yields the recurrence.
\par
Conversely, if there exists a recurrence of order \(r\), then the column vectors of the Hankel matrix satisfy a linear dependence relation of length \(r\), so its rank is at most \(r\);
hence the minimal recurrence order equals the Hankel rank.
For a finite exponential sum, the Hankel rank equals precisely the number of exponential terms \(r_m\) (classical Prony/Hankel theory). \qed
\end{proof}

\begin{theorem}[Complete reconstruction from \texorpdfstring{$2r_m$}{2rm} moments]\label{thm:pom-fiber-spectrum-prony-hankel-2r-reconstruction}
The finite data \(\{S_q(m)\}_{q=0}^{2r_m-1}\) alone uniquely determines all atoms \(\{a_i\}_{i=1}^{r_m}\) and their multiplicities \(\{n_i\}_{i=1}^{r_m}\).
More specifically, there exists a unique monic polynomial
\[
Q_m(t)=t^{r_m}+c_{r_m-1}t^{r_m-1}+\cdots+c_0=\prod_{i=1}^{r_m}(t-a_i)
\]
such that for all \(q\ge 0\), the recurrence identity
\(
S_{q+r_m}(m)+c_{r_m-1}S_{q+r_m-1}(m)+\cdots+c_0S_q(m)=0
\)
holds.
The coefficients \((c_j)\) are uniquely solved from the linear system
\[
\sum_{j=0}^{r_m} c_j\,S_{q+j}(m)=0,\qquad q=0,\dots,r_m-1,\qquad c_{r_m}=1;
\]
subsequently, \(\{a_i\}\) are the roots of \(Q_m\) (counted with multiplicity), and \(\{n_i\}\) are uniquely recovered from the Vandermonde system
\[
\sum_{i=1}^{r_m} n_i\,a_i^{\,q}=S_q(m),\qquad q=0,\dots,r_m-1.
\]
\end{theorem}
\begin{proof}
From the minimal recurrence characterization of Theorem \ref{thm:pom-fiber-spectrum-prony-hankel-rank},
\((c_0,\dots,c_{r_m-1})\) satisfies the stated linear system.
When \(r_m\) is taken as the minimal order, the corresponding Hankel principal minor \(\det H^{(m)}_{r_m-1}>0\), so this linear system is invertible and yields a unique solution.
After constructing \(Q_m\) from the recurrence, the roots of \(Q_m\) are precisely the base set \(\{a_i\}\) of the exponential sum;
when the roots are distinct, the Vandermonde matrix is invertible, hence the multiplicities \(\{n_i\}\) are uniquely determined.
(If multiple roots appear, the corresponding exponential terms must be multiplied by polynomial factors and enter the Jordan-type generalized Prony; in the fiber counting framework of this paper, \(a_i\) are distinct integer values, so there are no multiple roots.) \qed
\end{proof}

\begin{theorem}[Sharpness of the information completeness threshold: in general, \texorpdfstring{$2r$}{2r} moments cannot be reduced]\label{thm:pom-fiber-spectrum-prony-hankel-threshold-sharp}
For any integer \(r\ge 2\), there exist two distinct finite atomic spectra
\[
\{(a_i,n_i)\}_{i=1}^{r}\ \neq\ \{(a'_i,n'_i)\}_{i=1}^{r}
\]
(where \(a_i,a'_i\in\NN_{\ge 1}\) are pairwise distinct and \(n_i,n'_i\in\NN_{\ge 1}\)), such that their power-sum moments satisfy
\[
\sum_{i=1}^{r}n_i a_i^{\,k}
\;=\;
\sum_{i=1}^{r}n'_i (a'_i)^{\,k}
\qquad(0\le k\le 2r-2),
\]
but are unequal at \(k=2r-1\).
Therefore, without additional structural assumptions, any method relying solely on \(\{S_k\}_{k=0}^{2r-2}\) cannot uniquely determine the \(r\)-atomic spectrum;
in other words, the \(2r\)-moment threshold of Theorem \ref{thm:pom-fiber-spectrum-prony-hankel-2r-reconstruction} is sharp in the general sense.
\end{theorem}
\begin{proof}
Take \(n:=2r-1\), and consider the binomial coefficient weights \(w_j:=\binom{n}{j}\) (\(0\le j\le n\)).
Define two atomic spectra (each containing \(r\) distinct atoms):
\[
a_i:=2i-1,\quad n_i:=\binom{n}{2i-2}\qquad(1\le i\le r),
\]
\[
a'_i:=2i,\quad n'_i:=\binom{n}{2i-1}\qquad(1\le i\le r).
\]
For each integer \(k\ge 0\), consider the alternating sum
\[
\Delta_k:=\sum_{j=0}^{n}(-1)^j\binom{n}{j}(j+1)^k.
\]
This is the value of the \(n\)-th order finite difference of the polynomial \(p(x)=x^k\) at the point \(x=1\) (the difference operator acting on a polynomial reduces its degree), hence when \(k\le n-1\), \(\Delta_k=0\).
From \(\Delta_k=0\) we immediately deduce
\[
\sum_{j\ \mathrm{even}}\binom{n}{j}(j+1)^k
\;=\;
\sum_{j\ \mathrm{odd}}\binom{n}{j}(j+1)^k
\qquad(0\le k\le n-1),
\]
which is precisely the moment equality for the two spectra at \(k\le 2r-2\).
\par
When \(k=n\), \(\Delta_n\) equals the action of the \(n\)-th order difference on the leading term \(x^n\), satisfying \(\Delta_n=(-1)^n n!\neq 0\),
so the two spectra must have unequal moments at \(k=2r-1\).
This yields the stated family of counterexamples, and the conclusion holds. \qed
\end{proof}

\begin{corollary}[Finite discrimination in the two-spectrum case]\label{cor:pom-fiber-spectrum-two-atom-complete-invariant}
If the fiber multiplicity at fixed \(m\) takes only two values (\(r_m=2\)),
then four moment data \(\{S_q(m)\}_{q=0}^{3}\) suffice to completely recover the fiber spectrum \(\{(a_1,n_1),(a_2,n_2)\}\) at this resolution.
Therefore, in two-spectrum solvable models (such as the bin-fold solvable framework in the appendix; see Remark \ref{rem:fold-bin-solvable-model}),
the ``two-spectrum data'' at each fixed \(m\) constitutes a complete invariant.
\end{corollary}

\begin{corollary}[Discrete convexity of the pressure and Perron root inequality]\label{cor:pom-crossq-pressure-convexity}
Let \(\tau(q):=\limsup_{m\to\infty}\frac{1}{m}\log S_q(m)\) be the moment pressure from Definition \ref{def:pom-moment-pressure}.
Then for all integers \(q\ge 1\) the discrete convexity inequality holds:
\[
\boxed{\;
2\tau(q)\ \le\ \tau(q-1)+\tau(q+1)\; }.
\]
If, moreover, in the verified regime \(\tau(q)=\log r_q\) (where \(r_q=\rho(A_q)\) is the Perron root of the \(q\)-th order collision kernel; see Theorem \ref{thm:pom-collision-kernel-ak} and Definition \ref{def:pom-moment-pressure}), then the cross-order spectral radius constraint
\[
\boxed{\;
r_q^{\,2}\ \le\ r_{q-1}\,r_{q+1}\; }
\]
holds.
\end{corollary}
\begin{proof}
From \(S_q(m)^2\le S_{q-1}(m)S_{q+1}(m)\) (Corollary \ref{cor:pom-crossq-logconvex-chain}), taking logarithms, dividing by \(m\), and passing to \(\limsup\) yields the first inequality.
The second follows by substituting \(\tau(q)=\log r_q\) and exponentiating. \qed
\end{proof}

\begin{proposition}[Rational generating function and full-order Jordan spectral expansion of the moment sequence]\label{prop:pom-moment-genfunc-jordan}
Fix an integer \(q\ge 2\).
Suppose there exist a finite-dimensional nonnegative matrix \(A_q\) and vectors \(u_q,v_q\ge 0\), together with an integer offset \(m_0\ge 0\), such that for all \(n\ge 0\),
\[
S_q(n+m_0)=u_q^{\mathsf T}A_q^{\,n}v_q.
\]
Then the power series
\[
G_q(z):=\sum_{n\ge 0} S_q(n+m_0)\,z^n
\]
is a rational function in the formal sense, satisfying the resolvent identity
\[
\boxed{\;
G_q(z)=u_q^{\mathsf T}(I-zA_q)^{-1}v_q\; }.
\]
Its denominator can be taken as a factor of \(\det(I-zA_q)\).
\par
Furthermore, let \(\mathrm{Spec}(A_q)=\{\lambda_1,\dots,\lambda_s\}\) (counted with Jordan structure) be the spectrum of \(A_q\).
Then there exist polynomials \(P_j(n)\) (of degree strictly less than the corresponding Jordan block size) such that for all \(n\ge 0\),
\[
\boxed{\;
S_q(n+m_0)=\sum_{j=1}^{s} P_j(n)\,\lambda_j^{\,n}\; }.
\]
If \(A_q\) is primitive, write \(r_q=\rho(A_q)>0\) for its Perron root; then there exist constants \(c_q>0\) and \(0<\theta_q<r_q\) such that
\[
S_q(n+m_0)=c_q\,r_q^{\,n}+O(\theta_q^{\,n})\qquad(n\to\infty).
\]
If \(A_q\) is irreducible but not primitive, with period \(p\), then there exist \(p\)-periodic coefficient sequences \(\{c_{q,r}\}_{r=0}^{p-1}\) such that
\[
S_q(n+m_0)=r_q^{\,n}\bigl(c_{q,n\bmod p}+o(1)\bigr)\qquad(n\to\infty).
\]
\end{proposition}
\begin{proof}
By the Neumann series \((I-zA_q)^{-1}=\sum_{n\ge 0} z^nA_q^n\) (in the sense of formal power series),
\(
u_q^{\mathsf T}(I-zA_q)^{-1}v_q=\sum_{n\ge 0}u_q^{\mathsf T}A_q^n v_q\,z^n
\), yielding the rational representation of \(G_q\) with denominator controlled by \(\det(I-zA_q)\).
The Jordan expansion is a standard result from finite-dimensional linear algebra: placing \(A_q\) in Jordan normal form and expanding \(u_q^{\mathsf T}A_q^{n}v_q\) gives the polynomial-times-exponential sum;
the primitive and periodic cases follow from the Perron--Frobenius characterization of the peripheral spectrum. \qed
\end{proof}

\paragraph{Finite-state compression of the bivariate \texorpdfstring{$\zeta$}{zeta} and commutative spectral surface}
For each fixed moment order \(q\), Proposition \ref{prop:pom-moment-genfunc-jordan} yields a rational-type generating function in the \(m\)-direction;
for each fixed resolution \(m\), Theorems \ref{thm:pom-fiber-spectrum-prony-hankel-rank}--\ref{thm:pom-fiber-spectrum-prony-hankel-2r-reconstruction} yield a finite exponential-sum structure in the \(q\)-direction.
When there exists a \emph{unified finite-state bivariate compilation} that compresses the entire table \(\{S_q(m)\}_{q,m}\) into a fixed finite-dimensional kernel family, both axes can be simultaneously packaged into a single Fredholm determinant object.

\begin{definition}[Bivariate moment-\texorpdfstring{$\zeta$}{zeta} (encapsulation of the full collision-moment table)]\label{def:pom-bivariate-moment-zeta}
Define the formal power series (bivariate generating function)
\[
\mathfrak{Z}(z,u)
:=\exp\!\Bigl(\sum_{m\ge 1}\sum_{q\ge 1}\frac{S_q(m)}{m}\,u^{q}z^{m}\Bigr)
\in \QQ[[z,u]].
\]
By coefficient extraction identity,
\[
[z^m][u^q]\ \log\mathfrak{Z}(z,u)=\frac{S_q(m)}{m}\qquad(m,q\ge 1).
\]
\end{definition}

\begin{theorem}[Bivariate Fredholm determinant representation and rationality]\label{thm:pom-bivariate-moment-zeta-fredholm}
Suppose there exists a matrix family \(\mathcal{A}(u)\in\Mat_{D\times D}(\QQ(u))\) of fixed dimension \(D\), analytic at \(u=0\) with \(\mathcal{A}(0)=0\), satisfying the trace identity for all \(m\ge 1\):
\[
\Tr\!\bigl(\mathcal{A}(u)^m\bigr)=\sum_{q\ge 1} S_q(m)\,u^{q}\ \in\ \QQ[[u]].
\]
Then the \(\mathfrak{Z}(z,u)\) of Definition \ref{def:pom-bivariate-moment-zeta} satisfies the strict identity
\[
\boxed{\;
\mathfrak{Z}(z,u)=\det\bigl(I-z\,\mathcal{A}(u)\bigr)^{-1}\; }.
\]
Consequently, \(\mathfrak{Z}\) is a rational function in \((z,u)\) in the formal sense; and \(\{S_q(m)\}\) can be strictly recovered by coefficient extraction from \(\log\mathfrak{Z}\).
\end{theorem}
\begin{proof}
By the definition of the Fredholm determinant (see the finite-dimensional analogue in Definition \ref{def:fredholm-determinant}), for any matrix \(T\) we have the identity
\[
\det(I-zT)=\exp\!\left(-\sum_{m\ge 1}\frac{\Tr(T^m)}{m}z^m\right).
\]
Taking \(T=\mathcal{A}(u)\) and substituting the assumed trace expansion yields
\[
\det(I-z\mathcal{A}(u))^{-1}
=\exp\!\Bigl(\sum_{m\ge 1}\frac{\Tr(\mathcal{A}(u)^m)}{m}z^m\Bigr)
=\exp\!\Bigl(\sum_{m\ge 1}\sum_{q\ge 1}\frac{S_q(m)}{m}u^q z^m\Bigr)
=\mathfrak{Z}(z,u).
\]
\qed
\end{proof}

\begin{theorem}[Commutative spectral surface of the two-dimensional recurrence algebra]\label{thm:pom-commutative-spectral-surface}
Under the hypotheses of Theorem \ref{thm:pom-bivariate-moment-zeta-fredholm}, let
\[
D(z,u):=\det(I-z\,\mathcal{A}(u))\in\QQ(z,u),
\]
and take a monic polynomial representative \(\widetilde D(z,u)\in\ZZ[z,u]\) after clearing denominators (removing overall common factors).
Define the algebraic set
\[
\mathcal{V}:=\{(z,u)\in\CC^{2}:\ \widetilde D(z,u)=0\}.
\]
Then \(\mathcal{V}\) can be viewed as the \textbf{commutative spectral surface} of this bivariate compilation: it compresses the poles/spectrum in the \(m\)-direction (controlled by \(z\)) and the algebraic singularities/recurrence in the \(q\)-direction (controlled by \(u\)) onto the same algebraic certificate \(\widetilde D=0\);
correspondingly, the shift operators along \(m\) and along \(q\) commute in the coordinate algebra generated by this certificate, thereby collapsing the degrees of freedom of the seemingly ``two-dimensional noncommutative recurrence'' into a commutative finitely-generated algebraic object.
\end{theorem}
\begin{proof}
By Theorem \ref{thm:pom-bivariate-moment-zeta-fredholm}, \(\mathfrak{Z}=1/D\) is a bivariate rational function, whose zero-pole structure (and thereby the induced two-dimensional linear constraints/elimination relations) is completely characterized by the cleared-denominator polynomial \(\widetilde D\).
Viewing the shifts along \(m\) and along \(q\) as algebraic actions induced by multiplication and coefficient extraction on \(\QQ[[z,u]]\), the two actions arise from the multiplication action of the commuting variables \(z,u\), and hence remain commutative in the quotient algebra defined by \(\widetilde D\); the maximal spectrum of this quotient algebra is naturally contained in \(\mathcal{V}\).
\qed
\end{proof}

\begin{proposition}[Chernoff-type threshold upper bounds: unified form for counting and pushforward measures]\label{prop:pom-chernoff-threshold-bounds}
Fix a resolution \(m\) and define the threshold set
\[
A_m(\alpha):=\{x\in X_m:\ \log d_m(x)\ge \alpha m\}\qquad(\alpha\ge 0).
\]
Then for any integer \(q\ge 1\) and any \(\alpha\ge 0\), we have the deterministic upper bound
\[
\boxed{\;
|A_m(\alpha)|\ \le\ e^{-q\alpha m}\,S_q(m)\; }.
\]
Furthermore, let \(w_m(x):=d_m(x)/2^m\) be the folding pushforward of the uniform microstate measure, and sample \(X_m\sim w_m\).
Then for any integer \(q\ge 1\) and any \(\alpha\ge 0\),
\[
\boxed{\;
\mathbb{P}\bigl(\log d_m(X_m)\ge \alpha m\bigr)
\ \le\ e^{-q\alpha m}\,\frac{S_{q+1}(m)}{2^m}\; }.
\]
The two bounds recover, at exponential scale, the spectral upper bound of Proposition \ref{prop:pom-spectrum-legendre-upper} (counting measure) and the Chernoff/Markov tail bounds of \S\ref{subsec:pom-spectral-resource-type} (output measure tail budget), respectively.
\end{proposition}
\begin{proof}
For the first bound, if \(x\in A_m(\alpha)\) then \(d_m(x)^q\ge e^{q\alpha m}\), so
\(
|A_m(\alpha)|e^{q\alpha m}\le \sum_{x\in A_m(\alpha)}d_m(x)^q\le S_q(m)
\).
For the second bound, on \(A_m(\alpha)\) we have \(d_m(x)^{q+1}\ge e^{q\alpha m}d_m(x)\), hence
\(
\sum_{x\in A_m(\alpha)}d_m(x)\le e^{-q\alpha m}\sum_{x\in A_m(\alpha)}d_m(x)^{q+1}\le e^{-q\alpha m}S_{q+1}(m)
\),
and dividing by \(2^m\) yields the result. \qed
\end{proof}

\begin{remark}[Fibonacci baseline, strictness, and endpoint extremal spectrum]\label{rem:pom-fib-baseline-and-endpoint}
For the Fibonacci-scale universal lower bound on the moment pressure and its strictness criterion, see directly Proposition \ref{prop:pom-rq-universal-bounds};
the equality condition at fixed \(m\) is equivalent to all fibers having equal size (i.e., \(d_m(\cdot)\) is constant everywhere), corresponding to strict uniformity of the pushforward distribution \(w_m\).
\par
For the characterization of the extreme fiber exponent (maximal fiber growth rate) and the endpoint of the higher-order pressure slope,
the main text has already provided this via Proposition \ref{prop:pom-rq-qinfty-endpoint} and Proposition \ref{prop:pom-maxfiber-exponent-equals-renyi-endpoint}:
the core mechanism is that the basic squeeze \(M_m^q\le S_q(m)\le |X_m|M_m^q\) compresses the constant-level contribution of \(|X_m|\) in the slope limit as \(q\to\infty\).
\end{remark}
