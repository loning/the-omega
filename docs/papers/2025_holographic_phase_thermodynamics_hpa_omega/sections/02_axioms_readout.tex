\section{Axioms and operational interface: scan--projection readout}
\label{sec:axioms_readout}

This paper uses one structural input: the layered HPA--$\Omega$ interface separating (i) an ontic unitary scan from (ii) an operational finite-resolution readout. We state it in an intentionally minimal form, compatible with the fuller developments in \cite{Ma2025HPA,Ma2025FoP,Ma2025HPOT}.

\subsection{Ontic layer: a static global state and intrinsic automorphism}

\begin{axiom}[O1: global state without external time]
The physical universe is described by a normalized global state $\omega_\Omega$ on an observable algebra $\mathcal{A}$. No externally imposed continuous time parameter is assumed.
\end{axiom}

\begin{axiom}[O2: finite information / holographic bound]
For any causally closed region, the effective accessible Hilbert-space dimension obeys a holographic bound
\begin{equation}
\dim \hilbert_{\mathrm{region}} \;\le\; \exp\!\left(\frac{A}{4 \ell_P^2}\right),
\end{equation}
where $A$ is an area measure associated with the boundary of the region \cite{tHooft1993,Susskind1995,Bousso2002}.
\end{axiom}

\begin{axiom}[O3: intrinsic update]
There exists a discrete-step automorphism $U:\mathcal{A}\to\mathcal{A}$ (implemented unitarily in controlled representations) such that correlations can be written as $\omega_\Omega(U^n(A))$ for $A\in\mathcal{A}$.
\end{axiom}

\begin{axiom}[O4: holographic encoding with approximate reconstruction]
There exists an encoding map $\Phi$ relating bulk and boundary operator algebras such that the effective readout observables can be represented on a boundary algebra, supporting approximate reconstruction consistent with quantum error correction \cite{AlmheiriDongHarlow2015,Pastawski2015,Harlow2017}.
\end{axiom}

\subsection{Operational layer: readout as a finite-resolution instrument}

\begin{axiom}[O5: scan--projection readout and induced probability]
Fix a resolution parameter $\varepsilon>0$. A finite observer accesses the system through an operational \emph{instrument} $\{\mathcal{I}_k^{(\varepsilon)}\}_k$ with associated POVM effects $\{E_k^{(\varepsilon)}\}_k\subset\mathcal{A}$ satisfying
\begin{equation}
\sum_k E_k^{(\varepsilon)}=\ind,\qquad E_k^{(\varepsilon)}\ge 0,
\end{equation}
and the induced outcome probabilities
\begin{equation}
P_k^{(\varepsilon)}=\omega_{\mathrm{eff}}\!\left(E_k^{(\varepsilon)}\right),
\end{equation}
where $\omega_{\mathrm{eff}}$ denotes the effective state on the accessible readout sector.

Concretely, one may model readout by a ``pointer'' unitary $V$ with spectral measure $\Pi_V$, and define effects using a window kernel (response function) $w_k^{(\varepsilon)}$ on $\mathbb{T}\cong \RR/\ZZ$:
\begin{equation}
E_k^{(\varepsilon)}=\int_{\mathbb{T}} w_k^{(\varepsilon)}(x)\, \dd \Pi_V(x).
\end{equation}
Discarding outcomes yields the unconditional readout channel (CPTP map)
\begin{equation}
\Lambda^{(\varepsilon)}(\rho)=\sum_k \mathcal{I}_k^{(\varepsilon)}(\rho),
\end{equation}
capturing operational irreversibility at fixed resolution \cite{Kraus1983,NielsenChuang2000}.
\end{axiom}

\subsection{Weyl complementarity: the structural origin of readout noise}

\begin{axiom}[O6: Weyl pair and intrinsic non-commutativity]
The scan is implemented by a unitary $U_{\mathrm{scan}}$, and the pointer phase by a unitary $V$, satisfying a Weyl commutation relation
\begin{equation}
U_{\mathrm{scan}} V = \e^{2\pi \iu \alpha}\, V U_{\mathrm{scan}},\qquad \alpha\in(0,1)\setminus\QQ.
\end{equation}
In the canonical representation on $L^2(\mathbb{T})$,
\begin{equation}
(U_{\mathrm{scan}}\psi)(x)=\psi(x+\alpha),\qquad (V\psi)(x)=\e^{2\pi \iu x}\psi(x),
\end{equation}
so the scan induces the irrational rotation orbit
\begin{equation}
x_n=x_0+n\alpha \pmod 1.
\end{equation}
\end{axiom}

\noindent The key point is operational: because $U_{\mathrm{scan}}$ and $V$ fail to commute, one cannot refine ``time access'' and ``phase readout'' simultaneously without cost. In this paper, \emph{thermal fluctuations and entropy production} are interpreted as the unavoidable residual mismatch generated when finite-resolution readout acts on this Weyl structure.

\subsection{Golden branch and canonical coding}
\label{subsec:golden_branch}

For a generic irrational slope $\alpha$, windowed readout of the rotation orbit produces a Sturmian mechanical word and is canonically encoded by an Ostrowski numeration system \cite{MorseHedlund1940,Lothaire2002,AlloucheShallit2003}. The \emph{golden branch} is the special choice
\begin{equation}
\alpha=\varphi^{-1}=\frac{\sqrt{5}-1}{2},\qquad \varphi=\frac{1+\sqrt{5}}{2},
\end{equation}
whose continued fraction has constant partial quotients $[0;1,1,1,\dots]$. In this case, Ostrowski representation reduces to Zeckendorf/Fibonacci coding: every natural number admits a unique decomposition as a sum of non-consecutive Fibonacci numbers \cite{Zeckendorf1972}. This provides a minimal-alphabet canonical discretization for readout depth and will serve as the normalization reference for ``arithmetic'' thermodynamic quantities in the next sections.
