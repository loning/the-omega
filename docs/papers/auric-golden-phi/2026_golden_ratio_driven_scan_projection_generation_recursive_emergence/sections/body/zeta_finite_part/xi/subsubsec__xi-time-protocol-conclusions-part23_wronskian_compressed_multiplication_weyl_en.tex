\begin{conclusion}[Compressed multiplication operator: spectrum \texorpdfstring{$\{b_m\}$}{b\_m} and minimal polynomial \texorpdfstring{$y_{+}$}{y+}]\label{con:xi-terminal-wronskian-compressed-multiplication-spectrum-minpoly}
Use the notation of Conclusion \ref{con:xi-terminal-wronskian-kernel-discrete-mercer-parseval}.
Let \(y_{+}(z)=\prod_{m=0}^{\kappa}(z-b_m)\) with pairwise distinct \(b_m\in\RR\), and define the linear operator \(M:\mathcal{H}_K\to\mathcal{H}_K\) by
\[
(Mf)(x):=\sum_{m=0}^{\kappa} b_m\,f(b_m)\,\frac{K(x,b_m)}{K(b_m,b_m)}.
\]
Then:
\begin{enumerate}
  \item \(M\) is self-adjoint and diagonalizable;
  \item with \(e_m(x):=K(x,b_m)/\sqrt{K(b_m,b_m)}\), the family \(\{e_m\}_{m=0}^{\kappa}\) is an orthonormal basis of \(\mathcal{H}_K\), and \(Me_m=b_m e_m\);
  \item the spectrum of \(M\) is exactly \(\{b_m\}_{m=0}^{\kappa}\), and the minimal polynomial of \(M\) equals \(y_{+}\); hence \(y_{+}(M)=0\).
\end{enumerate}
\end{conclusion}
\begin{proof}[Proof]
By the Parseval structure in Conclusion \ref{con:xi-terminal-wronskian-kernel-discrete-mercer-parseval}, \(\{K(\cdot,b_m)\}\) is an orthogonal basis of \(\mathcal{H}_K\), with
\(K(b_n,b_m)=0\) for \(n\neq m\), and \(K(b_m,b_m)>0\).
For a basis vector \(K(\cdot,b_m)\),
\[
M K(\cdot,b_m)
=\sum_{m'=0}^{\kappa} b_{m'}\,K(b_{m'},b_m)\,\frac{K(\cdot,b_{m'})}{K(b_{m'},b_{m'})}
=b_m\,K(\cdot,b_m),
\]
so \(K(\cdot,b_m)\) (equivalently \(e_m\)) is an eigenvector with eigenvalue \(b_m\). After normalization, \(\{e_m\}\) is orthonormal.
Hence \(M\) is the real diagonal matrix \(\mathrm{diag}(b_0,\dots,b_\kappa)\) in this basis, so it is self-adjoint and diagonalizable, with spectrum \(\{b_m\}\).
Finally, \(y_{+}\) is monic with zero set \(\{b_m\}\), so \(y_{+}(M)=0\). Since the \(b_m\) are pairwise distinct, the minimal polynomial must contain all linear factors \(z-b_m\), and therefore equals \(y_{+}\).
\end{proof}

\begin{conclusion}[Closed form of the Weyl function: \texorpdfstring{$m(z)=-y/y_{+}$}{m(z)=-y/y+}]\label{con:xi-terminal-wronskian-weyl-function-closed-form}
Use the setting of Conclusion \ref{con:xi-terminal-wronskian-compressed-multiplication-spectrum-minpoly}, and let
\(
\psi:=y/\sqrt{\pi}\in\mathcal{H}_K
\).
For \(z\in\CC\setminus\{b_0,\dots,b_\kappa\}\), define the Weyl function
\[
m(z):=\langle (M-zI)^{-1}\psi,\psi\rangle_{\mathcal{H}_K}.
\]
Then the exact identity holds:
\[
\boxed{\;m(z)=-\frac{y(z)}{y_{+}(z)}\;}
\]
and equivalently one has the full partial-fraction expansion
\[
-\frac{y(z)}{y_{+}(z)}=\sum_{m=0}^{\kappa}\frac{c_m}{b_m-z},
\qquad
c_m:=\frac{y(b_m)}{y_{+}'(b_m)}=\frac{y(b_m)^2}{A(b_m)}.
\]
\end{conclusion}
\begin{proof}[Proof]
In the orthonormal eigenbasis \(\{e_m\}\) from Conclusion \ref{con:xi-terminal-wronskian-compressed-multiplication-spectrum-minpoly}, one has
\((M-zI)^{-1}e_m=(b_m-z)^{-1}e_m\), hence
\[
m(z)=\sum_{m=0}^{\kappa}\frac{|\langle \psi,e_m\rangle|^2}{b_m-z}.
\]
By the reproducing identity \(f(b_m)=\langle f,K(\cdot,b_m)\rangle\) and \(e_m=K(\cdot,b_m)/\sqrt{K(b_m,b_m)}\),
\(
\langle \psi,e_m\rangle=\psi(b_m)/\sqrt{K(b_m,b_m)}
\).
Using \(\psi(b_m)=y(b_m)/\sqrt{\pi}\) and
\(
K(b_m,b_m)=A(b_m)/\pi=y(b_m)y_{+}'(b_m)/\pi
\), we obtain
\[
|\langle \psi,e_m\rangle|^2
=\frac{y(b_m)^2/\pi}{A(b_m)/\pi}
=\frac{y(b_m)^2}{A(b_m)}
=\frac{y(b_m)}{y_{+}'(b_m)}
=:c_m.
\]
Therefore \(m(z)=\sum_m c_m/(b_m-z)\).

Since \(m(z)=O(z^{-1})\) as \(z\to\infty\), and its poles are only simple poles at \(\{b_m\}\), one can write
\(m(z)=P(z)/y_{+}(z)\) with \(\deg P\le \kappa\).
Comparing residues at \(z=b_m\),
\[
\frac{P(b_m)}{y_{+}'(b_m)}=\operatorname*{Res}_{z=b_m}m(z)=-c_m=-\frac{y(b_m)}{y_{+}'(b_m)},
\]
thus \(P(b_m)=-y(b_m)\) for all \(m\).
Because both \(P\) and \(-y\) have degree at most \(\kappa\) and agree at \(\kappa+1\) distinct points \(\{b_m\}\), we get \(P\equiv -y\), i.e. \(m(z)=-y(z)/y_{+}(z)\).
\end{proof}

\begin{conclusion}[Positive spectral weights and moment identities: \texorpdfstring{$\{c_m\}$}{c\_m} is a probability family encoding the expansion of \texorpdfstring{$-y/y_{+}$}{-y/y+} at infinity]\label{con:xi-terminal-wronskian-spectral-weights-moment-identities}
Use the notation of Conclusion \ref{con:xi-terminal-wronskian-weyl-function-closed-form}.
Then:
\begin{enumerate}
  \item \(c_m>0\) for all \(m\), and \(\sum_{m=0}^{\kappa}c_m=1\);
  \item if
  \[
  -\frac{y(z)}{y_{+}(z)}=-\frac{1}{z}-\sum_{n\ge 1}\frac{\mu_n}{z^{n+1}}
  \qquad (z\to\infty),
  \]
  then \(\mu_n=\sum_{m=0}^{\kappa}c_m b_m^{n}\) for \(n\ge 1\);
  \item if
  \[
  y(z)=z^{\kappa}+y_1 z^{\kappa-1}+y_2 z^{\kappa-2}+O(z^{\kappa-3}),\quad
  y_{+}(z)=z^{\kappa+1}+y_{+1}z^{\kappa}+y_{+2}z^{\kappa-1}+O(z^{\kappa-2}),
  \]
  then
  \[
  \sum_{m=0}^{\kappa}c_m b_m=y_1-y_{+1},\qquad
  \sum_{m=0}^{\kappa}c_m b_m^{2}=y_2-y_{+2}-y_1y_{+1}+y_{+1}^{2}.
  \]
\end{enumerate}
\end{conclusion}
\begin{proof}[Proof]
From the Wronskian identity \(A=yy_{+}'-y'y_{+}\),
\[
w'(x)=\left(\frac{y_{+}}{y}\right)'(x)=\frac{A(x)}{y(x)^2}>0,
\qquad x\in\RR\setminus\{x_j\}.
\]
At \(x=b_m\), since \(y_{+}(b_m)=0\),
\(
w'(b_m)=y_{+}'(b_m)/y(b_m)>0
\), hence \(c_m=y(b_m)/y_{+}'(b_m)>0\).

On the other hand, by the partial-fraction expansion from Conclusion \ref{con:xi-terminal-wronskian-weyl-function-closed-form},
\[
\sum_{m=0}^{\kappa}\frac{c_m}{b_m-z}
=-\frac{1}{z}\sum_{m}c_m-\frac{1}{z^2}\sum_{m}c_m b_m-\cdots .
\]
Since \(-y(z)/y_{+}(z)=-z^{-1}+O(z^{-2})\) as \(z\to\infty\), comparing the \(z^{-1}\) coefficient yields \(\sum_m c_m=1\).
Comparing higher-order coefficients gives \(\mu_n=\sum_m c_m b_m^n\).

Finally, long-division of \(-y/y_{+}\) gives
\[
-\frac{y(z)}{y_{+}(z)}
=-\frac{1}{z}-\frac{y_1-y_{+1}}{z^2}
-\frac{y_2-y_{+2}-y_1y_{+1}+y_{+1}^2}{z^3}+O(z^{-4}),
\]
and comparison with the moment expansion immediately yields the two explicit formulas.
\end{proof}
