\begin{definition}[Isobaric Drift Capacity Vector]\label{def:real-input-40-isobaric-drift-vector}

\begin{proposition}[Isobaric Drift Capacity: The Strongest Drift Direction When First-Order Dominant Scale Is Fixed]\label{prop:real-input-40-isobaric-drift-capacity}

Therefore under the first-order isobaric constraint \(dP=\langle\nabla P(0),d\theta\rangle=0\), \(v_\perp\) gives the tangential direction along which \(d(\log\mathfrak{M})=\langle\nabla\log\mathfrak{M}(0),d\theta\rangle\) increases most rapidly.
\par
Under the three-dimensional observables of the real input 40-state kernel, the numerical values are
\[

v_\perp\approx(-0.0289971,\ 0.1598969,\ -0.1498087),
\qquad
\|v_\perp\|\approx 0.2210215.

\end{proposition}
\begin{proof}
By definition, \(\langle\nabla P(0),v_\perp\rangle=\langle\nabla P(0),\nabla\log\mathfrak{M}(0)\rangle-\mu\|\nabla P(0)\|^2=0\).
Under the constraint \(\langle\nabla P(0),d\theta\rangle=0\), the maximum of \(\langle\nabla\log\mathfrak{M}(0),d\theta\rangle\) is attained by the orthogonal projection of \(\nabla\log\mathfrak{M}(0)\) onto that tangent space, which is precisely \(v_\perp\). \qed
\end{proof}

\begin{theorem}[Near-Coboundary Quadratic Selection Law: The Worst Twist Index as the Shortest Vector in a Metric Lattice]\label{thm:real-input-40-near-coboundary-svp}
Following the notation \(\Sigma:=\nabla^2P(0)\succ 0\), and writing the three-dimensional observable vector as
\(
\Phi=(\varphi_e,\varphi_-,\varphi_2)
\) (see \S\ref{app:real-input-40-arity-3d} for the definition).
For a real vector \(\vartheta=(\vartheta_e,\vartheta_-,\vartheta_2)\in\RR^3\), define the unit-circle twist matrix
\[
(M_{i\vartheta})_{ij}:=\sum_{i\to j}\exp\bigl(i(\vartheta_e\varphi_e+\vartheta_-\varphi_-+\vartheta_2\varphi_2)\bigr),
\qquad
\rho(\vartheta):=\rho(M_{i\vartheta}),
\qquad
\lambda:=\rho(M_0).
\]
Then there exists \(\delta>0\) such that when \(\|\vartheta\|\le \delta\), the quadratic approximation holds:
\[
\boxed{\ \log\frac{\rho(\vartheta)}{\lambda}
\;=\;
-\frac{1}{2}\,\vartheta^{\mathsf T}\Sigma\,\vartheta\ +\ O(\|\vartheta\|^4)\ }.
\]
Therefore in the small-angle regime, ``slowest decay'' (largest \(\rho(\vartheta)/\lambda\)) is approximately equivalent to minimizing the quadratic form \(\vartheta^{\mathsf T}\Sigma\vartheta\).
\par
Furthermore, given a modulus tuple \((m_e,m_-,m_2)\) and a nonzero index \(j=(j_e,j_-,j_2)\) (each component taken modulo), let the wrapped angle vector be
\[
\vartheta(j):=\bigl(\mathrm{wrap}(2\pi j_e/m_e),\ \mathrm{wrap}(2\pi j_-/m_-),\ \mathrm{wrap}(2\pi j_2/m_2)\bigr)\in[-\pi,\pi]^3,
\]
where \(\mathrm{wrap}\) takes the principal-value representative. Then in the small-angle regime the selection law holds:
\[
j_\star\ \approx\ \arg\min_{j\neq 0}\ \vartheta(j)^{\mathsf T}\Sigma\,\vartheta(j),
\]
i.e., the worst twist index is approximately determined by the shortest vector problem on the discrete angle lattice
\(
\Lambda=\frac{2\pi}{m_e}\ZZ\times\frac{2\pi}{m_-}\ZZ\times\frac{2\pi}{m_2}\ZZ
\)
with respect to the metric \(\Sigma\).
\end{theorem}
\begin{proof}
Since \(M_\theta\) at \(\theta=0\) is a primitive positive matrix, its Perron eigenvalue is an algebraically simple root; by Kato's analytic perturbation theory, there exists a neighborhood in which the dominant eigenvalue branch \(\lambda(\theta)\) is analytic in the complex parameter \(\theta\), and \(P(\theta):=\log\lambda(\theta)\) is analytic in that neighborhood.
Performing a Taylor expansion along the purely imaginary direction \(\theta=i\vartheta\) and taking the real part, odd-order terms are purely imaginary and vanish, while the quadratic term gives
\[
\Re\bigl(P(i\vartheta)-P(0)\bigr)=-\frac12\,\vartheta^{\mathsf T}\nabla^2P(0)\,\vartheta+O(\|\vartheta\|^4).
\]
In a sufficiently small neighborhood, the dominant eigenvalue branch still dominates all other spectral values in modulus, so \(\rho(M_{i\vartheta})=|\lambda(i\vartheta)|\), and
\(
\log(\rho(\vartheta)/\lambda)=\Re(P(i\vartheta)-P(0))
\).
Combining gives the first boxed formula.
The discrete index conclusion follows by restricting \(\vartheta\) to the angle lattice and maximizing the above quadratic approximation (equivalently minimizing the quadratic form). \qed
\end{proof}

\begin{corollary}[The \texorpdfstring{$m^2$}{m2} Diffusion Barrier for Dirichlet Moduli (Locked by the Variance Density)]\label{cor:dirichlet-modulus-diffusion-barrier}
Under the setting of Theorem \ref{thm:real-input-40-near-coboundary-svp}, fix an integer-valued linear combination
\(
F:=a_e\varphi_e+a_-\varphi_-+a_2\varphi_2
\)
(\(a_e,a_-,a_2\in\ZZ\)), and take the one-dimensional unit-circle twist along this direction
\[
M_{itF}:=M_{i(t a_e,\ t a_-,\ t a_2)},\qquad t\in\RR.
\]
Let
\(
\sigma_F^2:=\mathrm{Var}_\infty(F)=a^{\mathsf T}\Sigma a
\)
be the covariance density of this combination (where \(a=(a_e,a_-,a_2)^{\mathsf T}\), \(\Sigma=\nabla^2P(0)\)).
Then as the modulus \(m\to\infty\), for the minimal nontrivial angle \(t_m:=2\pi/m\), the asymptotic holds:
\[
\boxed{\;
\log\frac{\rho(M_{it_mF})}{\lambda}
=-\frac{\sigma_F^2}{2}\left(\frac{2\pi}{m}\right)^2+O(m^{-4})
\;}
\]
whence
\[
\boxed{\;
\frac{\rho(M_{it_mF})}{\lambda}
=\exp\!\left(-\frac{2\pi^2\sigma_F^2}{m^2}+O(m^{-4})\right)
=1-\frac{2\pi^2\sigma_F^2}{m^2}+O(m^{-4}).
\;}
\]
Therefore, for the equidistribution error term \(O(\rho^n/n)\) across $m$ residue classes, the ``effective decay length'' satisfies
\[
n\ \gtrsim\ \frac{m^2}{2\pi^2\sigma_F^2},
\]
meaning that increasing the modulus induces an \(m^2\) diffusion barrier at the dominant scale, determined by the variance density.
\end{corollary}

\begin{proof}
By the boxed formula of Theorem \ref{thm:real-input-40-near-coboundary-svp}, taking \(\vartheta=t a\) gives
\(
\log(\rho(ta)/\lambda)=-\frac12 t^2 a^{\mathsf T}\Sigma a+O(t^4)
=-\frac12\sigma_F^2 t^2+O(t^4)
\).
Setting \(t=t_m=2\pi/m\) yields the conclusion.
\end{proof}

\qquad

\boxed{\;

\]
\end{proposition}
\begin{proof}
By the boxed formula of Theorem \ref{thm:real-input-40-near-coboundary-svp}, taking \(\vartheta=t v\) gives
\[
\log\frac{\rho(M_{itv})}{\lambda}
=-\frac{t^2}{2}\,v^{\mathsf T}gv+O(t^4).

On the other hand, \(\log\mathfrak{M}(\theta)\) is real-analytic in a neighborhood of \(\theta=0\) (Remark \ref{prop:logM-analytic}), and the Taylor expansion along the slice \(\theta(t)=itv\) gives
\[

=\log\mathfrak{M}(0)+it\,\langle\nabla\log\mathfrak{M}(0),v\rangle-\frac{t^2}{2}\,v^{\mathsf T}Hv+O(t^3).
\]
Taking the real part and noting that odd-order terms are purely imaginary gives
\(
\Delta_{\mathfrak{M},v}(t)=\frac{t^2}{2}v^{\mathsf T}Hv+O(t^4)
\).
Then from \(s_v(t)=P(itv)/\log\lambda\) and \(\Re P(itv)=\log\rho(M_{itv})\) (the dominant eigenvalue branch dominates other spectral values in modulus; see the proof of Theorem \ref{thm:real-input-40-near-coboundary-svp}), we obtain

=-\frac{1}{\log\lambda}\log\frac{\rho(M_{itv})}{\lambda}
=\frac{t^2}{2\log\lambda}\,v^{\mathsf T}gv+O(t^4).

\end{proof}

\begin{corollary}[Fourth-Order Spectral Gap Asymptotic Under Unit-Root Twists]\label{cor:real-input-40-collision-twist-fourth}

\begin{remark}[Large Deviation Parametrization of the Collision Frequency]\label{rem:real-input-40-collision-ldp-param}
From the cubic equation one can eliminate $u$ to obtain the parametric representation

Further denoting $\alpha(\theta)=P_2'(\theta)$, one can write

where \(\lambda=\rho(1)=\varphi^2\) (baseline pressure \(P_2(0)=\log\lambda\)).
\end{remark}

\begin{proposition}[All-Real Spectrum and Root Type Fixed on \texorpdfstring{$u>0$}{u>0}]\label{prop:real-input-40-collision-all-real}
For any \(u>0\), the cubic polynomial
\[
f(\lambda;u):=\lambda^3-2\lambda^2-(u+1)\lambda+u
\]
has discriminant
\[
\Delta(u)=4u^3+25u^2+88u+8>0.
\]
Therefore \(f(\lambda;u)\) always has three distinct real roots over \(\RR\); combined with the extra eigenvalue \(-1\), all eigenvalues of the matrix \(W(u)=D(u)A\) are real.
Moreover, the root type (by size/sign) satisfies:
\[
\lambda_+(u)>1,\qquad 0<\lambda_0(u)<1,\qquad \lambda_-(u)<0,
\]
where \(\lambda_+(u)\) is the Perron root (Proposition \ref{prop:real-input-40-collision-pressure}).
More specifically, when \(u\le 1\), \(\lambda_-(u)\in[-1,0)\), and when \(u>1\), \(\lambda_-(u)<-1\).
\end{proposition}
\begin{proof}
The discriminant is computed directly as \(\Delta(u)=4u^3+25u^2+88u+8>0\) (for \(u>0\), all terms on the right are nonneg and the constant term \(8>0\)), so the three roots are distinct and all real.
Moreover, \(f(0;u)=u>0,\ f(1;u)=1-2-(u+1)+u=-2<0\), so there is a root \(\lambda_0(u)\) in \((0,1)\).
Also \(f(\lambda;u)\to+\infty\) as \(\lambda\to+\infty\), combined with \(f(1;u)<0\), gives a root \(\lambda_+(u)>1\) in \((1,\infty)\).
By Vieta's relations, the product of the three roots is \(\lambda_+(u)\lambda_0(u)\lambda_-(u)=-u<0\), so the third root \(\lambda_-(u)\) is negative.
Finally, computing
\(
f(-1;u)=2(u-1)
\):
when \(u\le 1\), \(f(-1;u)\le 0\) and \(f(0;u)>0\), so the negative root lies in \([-1,0)\);
when \(u>1\), \(f(-1;u)>0\) and \(f(\lambda;u)\to-\infty\) as \(\lambda\to-\infty\), so the negative root satisfies \(\lambda_-(u)<-1\).
\qed
\end{proof}

\begin{proposition}[Kernel-RH/Ramanujan Window and Critical Quintic Equation for the Collision Potential]\label{prop:real-input-40-collision-rhk-window}
Let \(\lambda_1(u):=\lambda_+(u)\) be the Perron root, and denote the secondary spectral radius
\[
\Lambda(u):=\max\{|\lambda_0(u)|,|\lambda_-(u)|,1\}.
\]
Then the kernel-RH condition \(\textsf{RH}(u)\) (Definition \ref{def:kernel-rh}) on this \(4\times4\) collision potential is equivalent to
\[
\Lambda(u)\le \lambda_1(u)^{1/2}.
\]
There exists a unique critical parameter \(u_R>1\) such that
\[
u_R^5+4u_R^4+3u_R^3-96u_R^2-42u_R-14=0,
\]
and
\[
\textsf{RH}(u)\ \Longleftrightarrow\ 0<u\le u_R.
\]
Numerically \(u_R\approx 3.59262181344\) (\(\theta_R=\log u_R\approx 1.27888\)).
\end{proposition}
\begin{proof}
By Proposition \ref{prop:real-input-40-collision-all-real}, when \(u\le 1\), \(|\lambda_-(u)|\le 1\) and \(0<\lambda_0(u)<1\), so \(\Lambda(u)=1\), and \(\lambda_1(u)>1\) gives \(1\le \sqrt{\lambda_1(u)}\), so \(\textsf{RH}(u)\) holds automatically.

When \(u>1\), \(\lambda_-(u)<-1\), so \(\Lambda(u)=-\lambda_-(u)\).
Hence \(\textsf{RH}(u)\) is equivalent to
\[
(-\lambda_-(u))^2\le \lambda_+(u).
\]
Let \(b:=-\lambda_-(u)>1\). The critical equality occurs if and only if there exists \(b>1\) such that
\(
f(-b;u)=0
\)
and
\(
f(b^2;u)=0
\).
Eliminating the parameter \(b\) (taking the resultant over \(b\)) gives an algebraic equation in \(u\):
\(
u(u^5+4u^4+3u^3-96u^2-42u-14)=0
\);
since \(u>0\), the critical points are given by the positive real roots of the quintic polynomial. The sign sequence of the quintic coefficients is \((+,+,+,-,-,-)\), so by Descartes' rule there is exactly one positive real root, denoted \(u_R\).
Finally, by continuity of the roots and the fact that the inequality holds at \(u=1\), \(\textsf{RH}(u)\) holds on \((0,u_R]\) and fails on \((u_R,\infty)\). \qed
\end{proof}

\begin{remark}[RH-Sharp Equality Point: Two-Root Collision and Quintic Algebraic Characterization]\label{rem:collision-cubic-rhsharp-quintic}
For the pure collision cubic equation
\[
f(\lambda;u)=\lambda^3-2\lambda^2-(u+1)\lambda+u,
\]
consider the critical point \(u_\star:=u_R\) of Proposition \ref{prop:real-input-40-collision-rhk-window}, and denote
\[
s:=\abs{\lambda_-(u_\star)}=-\lambda_-(u_\star)>0.
\]
The RH-sharp equality \(\Lambda(u_\star)=\sqrt{\lambda_+(u_\star)}\) (equivalently \(\beta(u_\star)=\tfrac12\); see Corollary \ref{cor:real-input-40-collision-error-phase}), dominated by the negative root, forces
\[
\boxed{\ \lambda_+(u_\star)=s^2,\qquad \lambda_-(u_\star)=-s,\qquad \lambda_0(u_\star)=2-s^2+s\ }.
\]
Here \(\lambda_0(u_\star)\) follows immediately from Vieta's relation \(\lambda_+(u_\star)+\lambda_0(u_\star)+\lambda_-(u_\star)=2\).
Substituting the three roots back into the polynomial gives the factorization at threshold
\[
f(\lambda;u_\star)=(\lambda-s^2)(\lambda+s)\bigl(\lambda-(2-s^2+s)\bigr).
\]
Comparing constant and linear coefficients yields
\[
u_\star=s^3(2-s^2+s),
\qquad
s^5-3s^3-s^2+2s-1=0.
\]
This quintic has exactly one real root for \(s>0\) (verifiable e.g.\ by a Sturm chain), so the spectral triple at threshold is uniquely locked by a single algebraic number \(s\).
Eliminating \(s\) recovers the quintic algebraic equation for \(u_\star\) in Proposition \ref{prop:real-input-40-collision-rhk-window}; hence the critical \(u_R\) is an algebraic number.
\end{remark}

\begin{corollary}[Phase Transition of the Prime Orbit Counting Error Exponent: \texorpdfstring{$\beta(u)$}{beta(u)} Continuously Deteriorates from \texorpdfstring{$\le 1/2$}{<=1/2} to \texorpdfstring{$1$}{1}]\label{cor:real-input-40-collision-error-phase}
On the collision potential \(W(u)\), let \(\lambda(u):=\lambda_1(u)=\lambda_+(u)\) and denote \(\Lambda(u)\) as in Proposition \ref{prop:real-input-40-collision-rhk-window}.
Then the leading term of the prime-orbit counting is \(\lambda(u)^n/n\), and the error exponent is controlled by
\[
\beta(u):=\frac{\log \Lambda(u)}{\log \lambda(u)}:
\]
\begin{itemize}
  \item For \(0<u\le 1\), \(\Lambda(u)=1\), so \(\beta(u)=0\) (error is at most constant order);
  \item For \(1<u\le u_R\), \(\Lambda(u)\le \sqrt{\lambda(u)}\), so \(\beta(u)\le \tfrac12\) (square-root phase; Corollary \ref{cor:kernel-rh-sqrt-error}), with equality only at \(u=u_R\);
  \item For \(u>u_R\), \(\Lambda(u)>\sqrt{\lambda(u)}\), so \(\beta(u)>\tfrac12\), and as \(u\to\infty\), \(\beta(u)\to 1\).
\end{itemize}
\par
More specifically, when \(u=t^2\to\infty\), the Puiseux expansion of the cubic gives
\[
\boxed{\ \lambda_+(u)=t+\frac12+\frac{9}{8t}+O(t^{-2}),\qquad
\lambda_-(u)=-t+\frac12-\frac{9}{8t}+O(t^{-2})\ } ,
\]
whence for \(u>1\) (so that \(\Lambda(u)=\abs{\lambda_-(u)}=-\lambda_-(u)\)),
\[
\boxed{\ \beta(u)=\frac{\log\Lambda(u)}{\log\lambda_+(u)}
=1-\frac{2}{\sqrt u\,\log u}+O\!\left(\frac{1}{u\log u}\right)\ } ,
\]
and \(\Lambda(u)/\lambda_+(u)\to 1\) (spectral gap collapse).
\end{corollary}
