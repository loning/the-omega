  \begin{theorem}[Time factor of the Zeckendorf truncation folding: unique branch point and unique merge point]\label{thm:terminal-succ-unique-branch-merge}
  For each \(m\ge 2\), define two alternating extremal words
  \[
  u_m:=(u_{m,1},\dots,u_{m,m}),\qquad
  u_{m,k}:=
  \begin{cases}
    1,& k+m\ \text{is even},\\
    0,& \text{otherwise},
  \end{cases}
  \]
  \[
  b_m:=(b_{m,1},\dots,b_{m,m}),\qquad
  b_{m,k}:=
  \begin{cases}
    1,& k+m\ \text{is odd},\\
    0,& \text{otherwise}.
  \end{cases}
  \]
  Then \(u_m,b_m\in X_m\), with \(u_m\) having last bit \(1\) and \(b_m\) having last bit \(0\), and satisfying
  \(V_m(u_m)=F_{m+2}-1\), \(V_m(b_m)=F_{m+1}-1\).
  In the directed graph \(G^{\mathrm{Succ}}_m\) of Definition \ref{def:terminal-succ-m}, the following uniqueness properties hold:
  \begin{enumerate}
    \item If \(w\neq b_m\), then \(|\mathrm{Succ}_m(w)|=1\).
    \item \(\mathrm{Succ}_m(b_m)=\{0^m,0^{m-1}1\}\). Therefore \(G^{\mathrm{Succ}}_m\) has exactly one branch vertex with out-degree \(2\), namely \(b_m\).
    \item The unique merge vertex with in-degree \(2\) is \(0^m\), and its predecessors are exactly \(u_m\) and \(b_m\).
  \end{enumerate}
  \emph{Minimal proof structure:} By the tail parametrization of Definition \ref{def:terminal-fold-zeck-extension}, fixing \(w\in X_m\), its preimage is given by \(N=V_m(w)+\)(admissible tail combinations), subject to the boundary constraint \(w_m c_{m+1}=0\).
  If \(w_m=1\) then \(c_{m+1}=0\) is forced, so the add-one normalization does not cross the boundary to trigger ambiguity; hence \(\mathrm{Fold}^{\mathrm{zeck}}_m(N+1)\) is uniquely determined by the lower \(m\) bits.
  If \(w_m=0\) and \(w\neq b_m\), then since \(b_m\) attains the maximum Zeckendorf value among admissible words with last bit \(0\) (\(V_m(b_m)=F_{m+1}-1\)), one has \(V_m(w)\le F_{m+1}-2\), so for any admissible tail, \(V_m(w)+1\le F_{m+1}-1\), and the add-one does not trigger a ``ceiling carry'' structure at position \(m\) nor a cross-boundary adjacency conflict with the optional \(c_{m+1}=1\), so \(|\mathrm{Succ}_m(w)|=1\) holds.
  When \(w=b_m\), \(V_m(b_m)=F_{m+1}-1\).
  If the tail takes the \(c_{m+1}=0\) branch, then the lower \(m\)-bit add-one normalization yields \(0^{m-1}1\);
  if the tail allows \(c_{m+1}=1\) (the boundary constraint does not block this since \(b_m\) has last bit \(0\)), then
  \(N=(F_{m+1}-1)+F_{m+2}+\cdots\Rightarrow N+1=F_{m+3}+\cdots\), so the first \(m\) bits are all zero, i.e., \(\mathrm{Fold}^{\mathrm{zeck}}_m(N+1)=0^m\).
  Finally, since \(V_m(u_m)=F_{m+2}-1\), the canonical representative of \(u_m\) plus one necessarily reaches the \(F_{m+2}\) reset and gives all-zero first \(m\) bits; and \(b_m\) in the \(c_{m+1}=1\) tail branch also reaches \(0^m\) in one step. All other vertices have out-degree \(1\), ruling out a second incoming edge, so the merge uniqueness holds.
  \end{theorem}

  \begin{theorem}[Pushforward stable type Markov kernel of \(\mathrm{Fold}^{\mathrm{bin}}_6\): reversibility and stationary measure]\label{thm:terminal-foldbin6-pushforward-markov}
  Let \(Q_6\) be the \(6\)-dimensional hypercube graph with vertex set \(\Omega_6=\{0,1\}^6\), and let the labeling function
  \(
  F_6(\omega):=\mathrm{Fold}^{\mathrm{bin}}_6(\mathrm{int}_6(\omega))\in X_6
  \)
  be as in Theorem \ref{thm:terminal-foldbin6-cube-edge-separation}.
  For \(w,w'\in X_6\), define the pushforward adjacency count (directed edge count):
  \[
  A_6(w,w')
  :=\#\bigl\{(\omega,\omega'):\ \omega\sim\omega'\text{ in }Q_6,\ F_6(\omega)=w,\ F_6(\omega')=w'\bigr\},
  \]
  and define the Markov kernel on \(X_6\):
  \[
  P_6(w,w'):=\frac{A_6(w,w')}{6\,d^{\mathrm{bin}}_6(w)},
  \qquad d^{\mathrm{bin}}_6(w):=\bigl|(\mathrm{Fold}^{\mathrm{bin}}_6)^{-1}(w)\bigr|.
  \]
  Then \(P_6\) is a stochastic kernel on \(X_6\); and \((X_6,P_6)\) is reversible, with stationary distribution
  \[
  \pi_6(w)=\frac{d^{\mathrm{bin}}_6(w)}{64}.
  \]
  More strongly, for any \(w,w'\in X_6\), the detailed balance identity
  \(
  \pi_6(w)P_6(w,w')=\pi_6(w')P_6(w',w)
  \)
  holds.
  \end{theorem}
  \begin{proof}
  By definition, \(\sum_{w'}A_6(w,w')=6\,d^{\mathrm{bin}}_6(w)\), so \(P_6\) has row sums \(1\).
  On the other hand, undirected edges of \(Q_6\) appear in both \(A_6(w,w')\) and \(A_6(w',w)\) with the same count, so \(A_6(w,w')=A_6(w',w)\).
  Thus
  \[
  \pi_6(w)P_6(w,w')=\frac{d^{\mathrm{bin}}_6(w)}{64}\cdot\frac{A_6(w,w')}{6\,d^{\mathrm{bin}}_6(w)}
  =\frac{A_6(w,w')}{64\cdot 6}
  =\pi_6(w')P_6(w',w),
  \]
  yielding detailed balance and reversibility; stationarity follows from \(\sum_w \pi_6(w)P_6(w,w')=\pi_6(w')\). \qed
  \end{proof}
