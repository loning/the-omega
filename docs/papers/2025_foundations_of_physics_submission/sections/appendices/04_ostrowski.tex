\section{Ostrowski numeration, Sturmian coding, and Zeckendorf time}
\label{app:ostrowski}

This appendix records the Zeckendorf specialization used in Proposition~\ref{prop:ostrowski_zeckendorf} and summarizes the general Ostrowski framework associated with an irrational rotation (see, e.g., \cite{Zeckendorf1972,AlloucheShallit2003}).

\subsection{Ostrowski numeration associated with an irrational slope}
Let $\alpha_{\mathrm{rot}}\in(0,1)\setminus\mathbb{Q}$ have continued fraction expansion $\alpha_{\mathrm{rot}}=[0;a_1,a_2,\dots]$ with convergents $p_n/q_n$. Define $q_{-1}\equiv 0$, $q_0\equiv 1$ and
\begin{equation}
    q_{n+1}=a_{n+1}q_n+q_{n-1}\qquad (n\ge 0).
\end{equation}
The \emph{Ostrowski representation} associated with $\alpha_{\mathrm{rot}}$ states that every integer $t\in\mathbb{Z}_{\ge 0}$ admits a unique expansion
\begin{equation}
    t=\sum_{n=0}^{m} b_n q_n,
\end{equation}
with digits satisfying $0\le b_0<a_1$, $0\le b_n\le a_{n+1}$ for $n\ge 1$, and the admissibility constraint that if $b_n=a_{n+1}$ then $b_{n-1}=0$ (see, e.g., \cite{AlloucheShallit2003}).

\subsection{Golden case and the Zeckendorf rule}
For the golden-slope choice $\alpha_{\mathrm{rot}}=\phi^{-2}$ one has $\alpha_{\mathrm{rot}}=[0;2,1,1,1,\dots]$, so $a_1=2$ and $a_n=1$ for $n\ge 2$. The recurrence above then yields
\begin{equation}
    q_n = F_{n+2}\qquad (n\ge 0),
\end{equation}
with Fibonacci numbers defined by $F_0=0$, $F_1=1$, $F_{n+2}=F_{n+1}+F_n$. The Ostrowski digit constraints therefore reduce to $b_n\in\{0,1\}$ with no adjacent ones, exactly the Zeckendorf admissibility rule used in Proposition~\ref{prop:ostrowski_zeckendorf}.

\subsection{Zeckendorf's theorem (existence and uniqueness)}
\label{app:zeckendorf_proof}
\begin{theorem}[Zeckendorf]
Every $t\in\mathbb{Z}_{\ge 0}$ admits a unique representation
\begin{equation}
    t=\sum_{k=0}^{m} b_k F_{k+2},\qquad b_k\in\{0,1\},\qquad b_k b_{k+1}=0.
\end{equation}
\end{theorem}
\begin{proof}[Reference]
See, e.g., \cite{Zeckendorf1972,AlloucheShallit2003}.
\end{proof}
