\paragraph{谱零点的 \texorpdfstring{$(\ZZ_2)^r$}{(Z2)r} 轨道抵消、Helly--\texorpdfstring{$2$}{2} 刚性与密度--信息耦合}
在 Fibonacci 模零谱的余类刚性、六倍窗半阶零块以及最大纤维偏置的统一非均匀性链已经建立之后，还可把单频零化的逐点见证、零余类覆盖的高阶交结构、六倍窗归一化零点密度的精确指数率以及 KL 缺口对峰值纤维的确定性控制进一步压缩为下列五条结果。它们共同表明：谱零点的出现不仅可以由单坐标翻转的配对抵消逐点解释，而且其余类交结构没有隐藏的高阶障碍；更进一步，零点密度的半维指数率与输出分布的 KL 缺口又分别在频域与信息域给出完全可审计的数量级控制。

\begin{theorem}[谱零点的高阶 \texorpdfstring{$(\ZZ_2)^r$}{(Z2)r} 轨道抵消]\label{thm:xi-time-part70c-zero-spectrum-z2-orbit-cancellation}
记
\[
M:=F_{m+2},
\qquad
\widehat d_m(k):=\sum_{\omega\in\Omega_m}e^{-2\pi i kN(\omega)/M},
\]
并定义
\[
J(k):=\left\{j\in\{1,\dots,m\}:2kF_{j+1}\equiv M\pmod{2M}\right\}.
\]
对每个 \(j\in J(k)\)，令
\[
\iota_j(\omega):=\omega\oplus e_j
\]
为翻转第 \(j\) 位的对合。则有：
\begin{enumerate}
  \item 群
  \[
  G_k:=\langle \iota_j:\ j\in J(k)\rangle
  \]
  自然同构于 \((\ZZ_2)^{|J(k)|}\)，并在 \(\Omega_m\) 上自由作用；
  \item 若定义相位函数
  \[
  \Phi_k(\omega):=e^{-2\pi i kN(\omega)/M},
  \]
  则对任意
  \[
  g=\prod_{j\in S}\iota_j\in G_k
  \]
  都有
  \[
  \Phi_k(g\omega)=(-1)^{|S|}\Phi_k(\omega);
  \]
  \item 因而对每个 \(G_k\)-轨道 \(\mathcal O\subseteq\Omega_m\)，都有
  \[
  \sum_{\omega\in\mathcal O}\Phi_k(\omega)=0
  \qquad\text{当 }J(k)\neq\varnothing,
  \]
  从而
  \[
  \widehat d_m(k)=0
  \qquad\Longleftrightarrow\qquad
  J(k)\neq\varnothing.
  \]
\end{enumerate}
\end{theorem}
\begin{proof}
由定理 \ref{thm:fold-spectrum-zero-criterion}，
\[
\widehat d_m(k)=0
\iff
\exists\,j\in\{1,\dots,m\}\ \text{s.t.}\ 2kF_{j+1}\equiv M\pmod{2M},
\]
这正是 \(J(k)\neq\varnothing\) 的定义。另一方面，推论 \ref{cor:fold-spectrum-zero-involution} 说明：若 \(j\in J(k)\)，则 \(\iota_j\) 为无固定点对合，并满足
\[
\Phi_k(\iota_j\omega)=-\Phi_k(\omega).
\]

现取 \(j,j'\in J(k)\)。由于 \(\iota_j,\iota_{j'}\) 只是翻转不同坐标，故
\[
\iota_j^2=\mathrm{id},
\qquad
\iota_j\iota_{j'}=\iota_{j'}\iota_j.
\]
因此
\[
G_k\cong(\ZZ_2)^{|J(k)|}.
\]
任一非平凡元素都会翻转至少一个坐标，故不可能固定任何 \(\omega\in\Omega_m\)，从而该作用自由。

再取
\[
g=\prod_{j\in S}\iota_j\in G_k.
\]
由各 \(\iota_j\) 交换，并反复使用 \(\Phi_k(\iota_j\omega)=-\Phi_k(\omega)\)，得到
\[
\Phi_k(g\omega)=(-1)^{|S|}\Phi_k(\omega).
\]
于是若 \(J(k)\neq\varnothing\)，对任一轨道 \(\mathcal O=G_k\cdot \omega\) 有
\[
\sum_{\eta\in\mathcal O}\Phi_k(\eta)
=
\Phi_k(\omega)\sum_{S\subseteq J(k)}(-1)^{|S|}
=
\Phi_k(\omega)(1-1)^{|J(k)|}
=
0.
\]
对全部轨道求和即得 \(\widehat d_m(k)=0\)。反向蕴含已由定理 \ref{thm:fold-spectrum-zero-criterion} 给出。证毕。
\end{proof}

\begin{theorem}[零余类族的任意有限交截由两两兼容性完全控制]\label{thm:xi-time-part71-zero-coset-finite-intersection-pairwise-compatibility}
设 \(M\) 为偶整数，且
\[
g_1,\dots,g_r\mid \frac{M}{2}.
\]
对每个 \(g\mid M/2\)，定义
\[
S_g(M):=
\left\{
k\in \ZZ/(M\ZZ):
\ k\equiv \frac{M}{2g}\pmod{M/g}
\right\}.
\]
则对任意有限族 \(\{g_1,\dots,g_r\}\)，都有
\[
\bigcap_{j=1}^{r}S_{g_j}(M)\neq\varnothing
\iff
2\gcd(g_i,g_j)\mid (g_j-g_i)
\qquad
(1\le i<j\le r).
\]
并且一旦交截非空，就有精确基数公式
\[
\left|
\bigcap_{j=1}^{r}S_{g_j}(M)
\right|
=
\gcd(g_1,\dots,g_r).
\]
\end{theorem}
\begin{proof}
由定理 \ref{thm:fold-zero-coset-rigidity}，
\[
S_{g_j}(M)
=
\left\{
k\in \ZZ/(M\ZZ):
\ k\equiv \frac{M}{2g_j}\pmod{M/g_j}
\right\}
\qquad (1\le j\le r).
\]
因此
\[
\bigcap_{j=1}^{r}S_{g_j}(M)\neq\varnothing
\]
当且仅当同余组
\[
k\equiv \frac{M}{2g_j}\pmod{M/g_j}
\qquad (j=1,\dots,r)
\]
可解。广义中国剩余定理断言：这一线性同余组可解，当且仅当任意两式相容。另一方面，引理 \ref{lem:fold-zero-coset-intersection} 已给出两式相容的充要条件恰为
\[
2\gcd(g_i,g_j)\mid (g_j-g_i).
\]
于是得到第一条等价关系。

现设该系统可解。由于解集模数为
\[
\operatorname{lcm}\!\left(\frac{M}{g_1},\dots,\frac{M}{g_r}\right)
=
\frac{M}{\gcd(g_1,\dots,g_r)},
\]
故在模 \(M\) 的意义下，解恰形成一个单一同余类的提升，其基数为
\[
\frac{M}{M/\gcd(g_1,\dots,g_r)}
=
\gcd(g_1,\dots,g_r).
\]
证毕。
\end{proof}

\begin{corollary}[零点余类覆盖的 nerve 是 flag 复形]\label{cor:xi-time-part70c-zero-coset-nerve-flag-complex}
设 \(\mathcal G\subseteq\{g:\ g\mid M/2\}\) 为任意有限集合，并考虑覆盖
\[
\mathcal U_{\mathcal G}:=\{S_g(M):\ g\in\mathcal G\}.
\]
定义其 nerve 为
\[
\mathcal N(\mathcal U_{\mathcal G})
:=
\left\{
\sigma\subseteq \mathcal G:\ \bigcap_{g\in \sigma}S_g(M)\neq\varnothing
\right\}.
\]
则 \(\mathcal N(\mathcal U_{\mathcal G})\) 是一个 flag 复形：任意顶点集若两两有边，则自动张成一个单纯形。更明确地，\(\mathcal N(\mathcal U_{\mathcal G})\) 完全由算术图
\[
g\sim h
\iff
2\gcd(g,h)\mid (h-g)
\]
的 clique 结构决定。
\end{corollary}
\begin{proof}
由定理 \ref{thm:xi-time-part71-zero-coset-finite-intersection-pairwise-compatibility}，对任意有限子族 \(\sigma\subseteq \mathcal G\)，有
\[
\bigcap_{g\in \sigma}S_g(M)\neq\varnothing
\]
当且仅当 \(\sigma\) 中任意两点 \(g,h\) 都满足
\[
2\gcd(g,h)\mid (h-g).
\]
这恰好意味着 \(\sigma\) 是上述算术图中的一个 clique。因此 nerve 的全部高维单纯形都由其 \(1\)-骨架唯一决定，故 \(\mathcal N(\mathcal U_{\mathcal G})\) 为 flag 复形。证毕。
\end{proof}

\begin{theorem}[六倍窗上的零点密度具有精确指数率]\label{thm:xi-time-part70c-zero-density-exact-ratio-exponent-multiple6}
若
\[
m+2\equiv 0\pmod 6,
\qquad
d:=\frac{m+2}{2},
\qquad
M:=F_{m+2},
\qquad
\mathcal Z_m:=\{k\in \ZZ/(M\ZZ):\ \widehat d_m(k)=0\},
\]
则有极限
\[
\lim_{\substack{m\to\infty\\ m+2\equiv 0\ (6)}}
\frac1m\log\frac{|\mathcal Z_m|}{F_{m+2}}
=
-\frac12\log\varphi.
\]
等价地，在六倍窗子序列上，
\[
\frac{|\mathcal Z_m|}{F_{m+2}}
=
\varphi^{-m/2+o(m)}.
\]
\end{theorem}
\begin{proof}
下界方面，由推论 \ref{cor:fold-zero-half-index-multiple6}，
\[
|\mathcal Z_m|
\ge
F_d.
\]
因此
\[
\frac{|\mathcal Z_m|}{F_{m+2}}
\ge
\frac{F_d}{F_{m+2}}.
\]
由 Binet 公式 \(F_n=\varphi^{n+o(n)}\) 得
\[
\liminf_{\substack{m\to\infty\\ m+2\equiv 0\ (6)}}
\frac1m\log\frac{|\mathcal Z_m|}{F_{m+2}}
\ge
\lim_{m\to\infty}
\frac1m\log\frac{F_d}{F_{m+2}}
=
-\frac12\log\varphi.
\]

上界方面，由定理 \ref{thm:fold-zero-density-sparse}，
\[
\frac{|\mathcal Z_m|}{F_{m+2}}
\le
\tau(m+2)\frac{F_{\lfloor (m+2)/2\rfloor}}{F_{m+2}}.
\]
又由于
\[
\tau(n)=e^{o(n)},
\qquad
F_{\lfloor (m+2)/2\rfloor}= \varphi^{m/2+o(m)},
\qquad
F_{m+2}=\varphi^{m+o(m)},
\]
故
\[
\limsup_{\substack{m\to\infty\\ m+2\equiv 0\ (6)}}
\frac1m\log\frac{|\mathcal Z_m|}{F_{m+2}}
\le
-\frac12\log\varphi.
\]
上下界相合，即得结论。
\end{proof}

\begin{theorem}[最大纤维超额由 KL 缺口确定性控制]\label{thm:xi-time-part70c-kl-gap-controls-maxfiber-excess}
设
\[
p_m(r):=\frac{d_m(r)}{2^m},
\qquad
M:=F_{m+2},
\qquad
u(r):=\frac1M,
\]
并记
\[
M_m:=\max_r d_m(r),
\qquad
d_m^{\mathrm{avg}}:=\frac{2^m}{M}.
\]
则有确定性耦合不等式
\[
\boxed{\;
M_m-d_m^{\mathrm{avg}}
\le
2^m\,\|p_m-u\|_{\mathrm{TV}}
\le
2^m\sqrt{\frac12\,D_{\mathrm{KL}}(p_m\|u)}.\;}
\]
特别地，一旦 KL 缺口被控制，最大纤维相对于平均纤维的超额便被同步控制。
\end{theorem}
\begin{proof}
首先由
\[
M_m=2^m\max_r p_m(r),
\qquad
d_m^{\mathrm{avg}}=\frac{2^m}{M}=2^m u(r),
\]
可得
\[
M_m-d_m^{\mathrm{avg}}
=
2^m\left(\max_r p_m(r)-\frac1M\right)
\le
2^m\max_r|p_m(r)-u(r)|.
\]
另一方面，对任意概率分布 \(p,u\) 都有
\[
\max_r|p(r)-u(r)|
\le
\frac12\sum_r|p(r)-u(r)|
=
\|p-u\|_{\mathrm{TV}},
\]
故
\[
M_m-d_m^{\mathrm{avg}}
\le
2^m\,\|p_m-u\|_{\mathrm{TV}}.
\]
最后应用 Pinsker 不等式
\[
\|p_m-u\|_{\mathrm{TV}}
\le
\sqrt{\frac12\,D_{\mathrm{KL}}(p_m\|u)},
\]
便得到所示结论。证毕。
\end{proof}

\noindent
由此，谱零点结构在三个层级上同时完成闭合：在单频层级，它由 \((\ZZ_2)^r\) 轨道的符号特征逐点见证；在余类覆盖层级，它由 Helly--\(2\) 刚性与 flag nerve 压缩为纯图论对象；在密度与信息层级，它又一方面把六倍窗上的归一化零点密度锁定在 \(\varphi^{-m/2+o(m)}\)，另一方面把 KL 缺口与最大纤维超额耦合为单一确定性不等式。于是，轨道抵消、同余几何、半维稀疏率与信息偏置便被压缩进同一条完全可审计的频域主线。
