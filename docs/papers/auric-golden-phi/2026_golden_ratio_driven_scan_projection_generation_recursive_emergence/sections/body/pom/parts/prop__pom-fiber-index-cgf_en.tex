For $\gamma\ge 0$ and $\varepsilon>0$, define the "approximate layer" count:
$$
N_m(\gamma,\varepsilon):=\Bigl|\bigl\{x\in X_m:\ |\gamma_m(x)-\gamma|\le \varepsilon\bigr\}\Bigr|.
$$
Define the (upper) spectral function of fiber multiplicity:
$$
f(\gamma):=\lim_{\varepsilon\downarrow 0}\ \limsup_{m\to\infty}\ \frac{1}{m}\log N_m(\gamma,\varepsilon).
$$
\end{definition}

\begin{proposition}[Geometric meaning of collision spectrum: exponential moment generating function of fiber index]\label{prop:pom-fiber-index-cgf}
Following the output distribution notation $\pi_m(x)=d_m(x)/2^m$, treat $\gamma_m(X)$ as a random variable sampled from $X_m$ with $\pi_m$.
Then for any integer $q\ge 2$ the strict identity holds:
\[
\boxed{\;
\mathbb{E}_{\pi_m}\!\left[\exp\!\bigl((q-1)m\,\gamma_m(X)\bigr)\right]
=\frac{S_q(m)}{2^m}\; }.
\]
Thus:
\[
\frac{1}{m}\log \mathbb{E}_{\pi_m}\!\left[\exp\!\bigl((q-1)m\,\gamma_m(X)\bigr)\right]
=\frac{1}{m}\log S_q(m)-\log 2.
\]
In particular, if the Perron leading scale $S_q(m)=\Theta(r_q^m)$ exists (e.g., given by moment-kernel implementation), then the limiting cumulant generating function satisfies at integer points $t=q-1$:
\[
\lim_{m\to\infty}\frac{1}{m}\log \mathbb{E}_{\pi_m}\!\left[e^{t m\,\gamma_m(X)}\right]
=\log r_{t+1}-\log 2,
\qquad
h_{t+1}=(t+1)\log 2-\log r_{t+1}.
\]
\end{proposition}
\begin{proof}
Expanding by definition using $\pi_m(x)=d_m(x)/2^m$:
\[
\begin{aligned}
\mathbb{E}_{\pi_m}\!\left[\exp\!\bigl((q-1)m\,\gamma_m(X)\bigr)\right]
&=\sum_{x\in X_m}\pi_m(x)\,e^{(q-1)\log d_m(x)}\\
&=\sum_{x\in X_m}\frac{d_m(x)}{2^m}\,d_m(x)^{q-1}\\
&=\frac{1}{2^m}\sum_{x\in X_m}d_m(x)^q\\
&=\frac{S_q(m)}{2^m}.
\end{aligned}
\]
Taking logarithm and dividing by $m$ yields the second formula. The last part follows directly from the exponential scale $S_q(m)=\Theta(r_q^m)$ and the definition $h_q=\log(2^q/r_q)$. This completes the proof.
\end{proof}

\begin{remark}[Large deviation/multifractal inversion (interface + executable construction)]\label{rem:pom-fiber-index-ldp}

\begin{theorem}[Large deviation principle for fiber index (thermodynamic completion)]\label{thm:pom-fiber-index-ldp-thermo}

\begin{proposition}[Legendre upper bound: spectrum controlled by moment pressure]\label{prop:pom-spectrum-legendre-upper}
For any $q\ge 2$ and any $\gamma\ge 0$, we have the upper bound:
$$
f(\gamma)\ \le\ \inf_{q\ge 2}\bigl(\tau(q)-q\gamma\bigr).
$$
In particular, in the case $\tau(q)=\log r_q$:
$$
f(\gamma)\ \le\ \inf_{q\ge 2}\bigl(\log r_q-q\gamma\bigr).
$$
\end{proposition}

\begin{proof}
Fix $q\ge 2$ and $\gamma,\varepsilon$. For any $x$ satisfying $|\gamma_m(x)-\gamma|\le\varepsilon$, we have:
$$
d_m(x)^q=\exp(qm\gamma_m(x))\ \ge\ \exp\bigl(qm(\gamma-\varepsilon)\bigr).
$$
Therefore:
$$
S_q(m)=\sum_{x\in X_m} d_m(x)^q
\ \ge\ N_m(\gamma,\varepsilon)\,\exp\bigl(qm(\gamma-\varepsilon)\bigr),
$$
i.e.:
$$
\frac{1}{m}\log N_m(\gamma,\varepsilon)\ \le\ \frac{1}{m}\log S_q(m)\ -\ q(\gamma-\varepsilon).
$$
Taking $\limsup$ as $m\to\infty$ and then letting $\varepsilon\downarrow 0$ yields:
$$
f(\gamma)\le \tau(q)-q\gamma.
$$
Taking infimum over all $q\ge 2$ yields the conclusion. This completes the proof.
\end{proof}

\begin{remark}[Multifractal formalism (as testable target)]\label{rem:pom-spectrum-formalism}
Proposition~\ref{prop:pom-spectrum-legendre-upper} provides a strict upper bound: the convex conjugate of the moment pressure curve $\tau(q)$ controls the (upper) spectrum of fiber multiplicity.
Under standard thermodynamic formalism conditions (e.g., existence of a Gibbs structure dual to $\tau$ with sufficient quasi-multiplicativity), one often obtains the inequality in the opposite direction to achieve equality, i.e., "spectrum $f$ and pressure $\tau$ are Legendre transforms of each other".
In this framework, this equality can be viewed as a falsifiable structural target: it translates "space loss is a spectrum" into the computable proposition "$\{r_q\}$ family determines spectral shape".
\end{remark}

\begin{proposition}[Paley-Zygmund lower envelope: "heavy tail" exponent lower bound from only \texorpdfstring{$(r_q,r_{2q})$}{(rq,r2q)}]\label{prop:pom-spectrum-pz-lower}
Take any integer $q\ge 2$. Let $X$ be uniformly distributed on $X_m$, and let the non-negative random variable be:
\[
Y:=d_m(X)^q.
\]
Then the Paley-Zygmund inequality for threshold $\theta=\tfrac12$ yields:
\[
\#\Bigl\{x\in X_m:\ d_m(x)\ge \Bigl(\tfrac12\,\frac{S_q(m)}{|X_m|}\Bigr)^{1/q}\Bigr\}
\ \ge\ \frac14\cdot\frac{S_q(m)^2}{S_{2q}(m)}.
\]
Thus if the limits $r_k=\lim_{m\to\infty}S_k(m)^{1/m}$ exist at $k=q,2q$, and $\lim_{m\to\infty}\frac1m\log|X_m|=\log\varphi$, then defining:
\[
\gamma_q:=\frac{\log r_q-\log\varphi}{q},
\qquad
\underline f(q):=2\log r_q-\log r_{2q},
\]
we have the strict lower bound:
\[
\boxed{\;
\lim_{\varepsilon\downarrow 0}\ \liminf_{m\to\infty}\ \frac1m
\log \#\bigl\{x\in X_m:\ \gamma_m(x)\ge \gamma_q-\varepsilon\bigr\}
\ \ge\ \underline f(q)\; }.
\]
\end{proposition}
\begin{proof}
Expanding expectations under uniform sampling $X\sim \mathrm{Unif}(X_m)$:
\[
\mathbb{E}[Y]=\frac{1}{|X_m|}\sum_{x\in X_m}d_m(x)^q=\frac{S_q(m)}{|X_m|},
\qquad
\mathbb{E}[Y^2]=\frac{S_{2q}(m)}{|X_m|}.
\]
For any non-negative $Y$ and $\theta\in(0,1)$, the Paley-Zygmund inequality yields:
\[
\mathbb{P}\!\left(Y\ge \theta\,\mathbb{E}Y\right)\ \ge\ (1-\theta)^2\frac{(\mathbb{E}Y)^2}{\mathbb{E}[Y^2]}.
\]
Taking $\theta=\tfrac12$ and substituting the above expressions yields:
\[
\mathbb{P}\!\left(Y\ge \tfrac12\,\mathbb{E}Y\right)\ \ge\ \frac14\cdot\frac{S_q(m)^2}{|X_m|\,S_{2q}(m)}.
\]
Multiplying the probability by $|X_m|$ yields the counting lower bound:
\[
\#\Bigl\{x\in X_m:\ d_m(x)^q\ge \tfrac12\,\frac{S_q(m)}{|X_m|}\Bigr\}
\ \ge\ \frac14\cdot\frac{S_q(m)^2}{S_{2q}(m)},
\]
which is equivalent to the first inequality stated in the proposition.
\par
For the second part, taking logarithm, dividing by $m$, and letting $m\to\infty$:
the right-hand side yields:
\[
\liminf_{m\to\infty}\frac1m\log\!\left(\frac14\cdot\frac{S_q(m)^2}{S_{2q}(m)}\right)
=2\log r_q-\log r_{2q}=\underline f(q),
\]
while the threshold on the left-hand side satisfies:
\[
\frac1m\log\Bigl(\Bigl(\tfrac12\,\frac{S_q(m)}{|X_m|}\Bigr)^{1/q}\Bigr)
=\frac{1}{q}\Bigl(\frac1m\log S_q(m)-\frac1m\log|X_m|+\frac1m\log\tfrac12\Bigr)
\to \gamma_q.
\]
Therefore, for any $\varepsilon>0$, when $m$ is sufficiently large, the left-hand set is contained in $\{\gamma_m(x)\ge \gamma_q-\varepsilon\}$, yielding the stated $\liminf$ lower bound as $\varepsilon\downarrow 0$. This completes the proof.
\end{proof}

\begin{remark}[Two readouts of the same convex conjugate: \texorpdfstring{Legendre spectrum upper bound $\leftrightarrow$ Chernoff tail bound}{Legendre<->Chernoff}]\label{rem:pom-legendre-chernoff-same-dual}
Proposition~\ref{prop:pom-spectrum-legendre-upper} (geometric spectrum upper bound by counting measure) and Proposition~\ref{prop:pom-quantile-chernoff}--Corollary~\ref{cor:pom-quantile-dual-optimizer} (tail control by output measure $\pi_m$) are essentially given by the same convex duality kernel.
Let:
\[
\Phi(q):=\tau(q)-\log 2,
\qquad
\Lambda(t):=\Phi(t+1)=\tau(t+1)-\log 2\quad(t\ge 1),
\]
then the Chernoff rate function upper bound is:
\[
I(\gamma)=\sup_{t\ge 1}\bigl(t\gamma-\Lambda(t)\bigr)
\]
(Corollary~\ref{cor:pom-quantile-dual-optimizer}).
While the Legendre upper bound can be rewritten as:
\[
\begin{aligned}
\inf_{q\ge 2}\bigl(\tau(q)-q\gamma\bigr)
&=\log 2-\gamma+\inf_{t\ge 1}\bigl(\Lambda(t)-t\gamma\bigr)\\
&=\log 2-\gamma-\sup_{t\ge 1}\bigl(t\gamma-\Lambda(t)\bigr)
=\log 2-\gamma-I(\gamma).
\end{aligned}
\]
Therefore, both share the same convex conjugate $\sup_{t\ge 1}(t\gamma-\Lambda(t))$: the difference arises only from the $-\log 2$ normalization introduced by the base measure (counting vs.\ $\pi_m$) and the shift $q\leftrightarrow t=q-1$.
\par
Under the differentiable/strictly convex extension calibration (e.g., treating $q$ as a real parameter and extending $\tau$), the optimal moment order $q^\star(\gamma)$ (whether used for spectrum upper bound or tail bound budget) satisfies the same "support line condition":
\[
\gamma\ \in\ \partial \tau\bigl(q^\star(\gamma)\bigr),
\]
i.e., the subgradient of $\tau$ at $q^\star$ hits $\gamma$; together with discrete convexity of pressure (Proposition~\ref{prop:pom-pressure-convexity}), this upgrades "selecting $q$" from an empirical knob to a dual coordinate determined by convex geometry.
\end{remark}

\subsubsection{Irreducible lower bound for online delay}\label{subsec:pom-delay-lb}
\begin{proposition}[Minimality of single-pass online delay $3$]\label{prop:pom-delay-min}
Any single-pass online transducer implementing $\Fold_\infty:\{0,1,2\}^{\ZZ}\to X_\infty$ has minimum delay satisfying:
\[
\delta_{\min}(\Fold_\infty)\ge 3.
\]
Combined with Theorem~\ref{thm:online-delay-fold}, we obtain $\delta_{\min}(\Fold_\infty)=3$.
\end{proposition}
\begin{proof}
In the online kernel of Subsection~\ref{subsec:online-delay}, the output satisfies:
\(
e=\lfloor \varphi R(s)+d\varphi^{-3}\rfloor
\),
where $s$ is the synchronization state of length $3$, $d\in\{0,1,2\}$ is the current input. Taking state $s=002$ (which belongs to state set $Q$):
\[
R(s)=2\varphi^{-3},\qquad
e=\left\lfloor 2\varphi^{-2}+d\varphi^{-3}\right\rfloor.
\]
When $d=0$ then $e=0$, when $d=1$ then $e=1$, thus this bit output indeed depends on the "forward 3rd bit" input. Any single-pass online machine with delay $2$ has not yet read this input when outputting this bit, necessarily producing a conflict, and thus cannot implement $\Fold_\infty$. This completes the proof.
\end{proof}

\begin{remark}[Orthogonality of space defect and time defect]\label{rem:pom-space-time-orth}

\begin{definition}[Space-time-spectrum triple fingerprint]\label{def:pom-fingerprint}

\paragraph{Object: deterministic skew-product for real input}

\begin{proposition}[Deterministic extension does not increase entropy: $\rho(M)=\varphi^2$]\label{prop:real-input-40-entropy}

\[

\qquad
\rho(M)=e^{h_{\mathrm{top}}(\Sigma)}=\varphi^2.

\end{proposition}
\begin{proof}
The base system is input pair syntax $X\times X$, with adjacency matrix $A\otimes A$, thus $h_{\mathrm{top}}(X\times X)=\log\rho(A\otimes A)=2\log\varphi$.
On the other hand, given any base system symbol sequence $\{(x_k,y_k)\}$, then $d_k=x_k+y_k$ is uniquely determined, thus fiber update $s_{k+1}=\delta(s_k,d_k)$ is also uniquely determined.

Thus $h_{\mathrm{top}}(\Sigma)=h_{\mathrm{top}}(X\times X)$, hence $\rho(M)=\varphi^2$. This completes the proof.
\end{proof}
