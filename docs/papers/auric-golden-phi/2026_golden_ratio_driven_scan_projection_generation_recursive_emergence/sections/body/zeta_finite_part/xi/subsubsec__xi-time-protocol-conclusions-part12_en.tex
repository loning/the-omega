\begin{conclusion}[Discrete convexity and cross-order monotone law of collision pressure: monotone ascent of normalized pressure slope \texorpdfstring{$g(q)$}{g(q)} and consistency audit]\label{con:xi-terminal-collision-pressure-discrete-convexity-crossq-g}
Let
\(
S_q(m):=\sum_{x\in X_m}d_m(x)^q
\)
and denote by \(r_q\) its leading growth constant (e.g., in Perron--Frobenius regime \(S_q(m)=\Theta(r_q^m)\)).
Then for any fixed \(m\ge 2\), the function \(q\mapsto \log S_q(m)\) is convex on \(q\ge 0\);
therefore the pressure \(P(q):=\log r_q\) satisfies discrete convexity at integer points: for any integers \(q_0<q_1<q_2\), let
\(
\theta=\frac{q_2-q_1}{q_2-q_0}
\),
then
\[
P(q_1)\le \theta P(q_0)+(1-\theta)P(q_2),
\qquad
\text{equivalently } r_{q_1}\le r_{q_0}^{\theta}r_{q_2}^{1-\theta}.
\]
Meanwhile, the cross-order norm inequality induces: for any \(2\le p<q\), at scale \(|X_m|\sim\varphi^m\) there is the lower bound
\[
\log r_q \ \ge\ \frac{q}{p}\log r_p+\Bigl(1-\frac{q}{p}\Bigr)\log\varphi.
\]
Equivalently, let
\(
g(q):=\frac{\log r_q-\log\varphi}{q}
\),
then \(g(q)\) is non-decreasing on \(q\ge 2\), so that \((r_q/\varphi)^{1/q}\) ascends monotonically with \(q\);
this monotone law can serve as a cross-implementation regression consistency audit scalar.
\end{conclusion}
\begin{proof}[Proof]
See Proposition \ref{prop:pom-pressure-convexity} (convexity and discrete convexity) and Proposition \ref{prop:pom-crossq-g-monotone} (cross-order lower bound and \(g(q)\) monotonicity). Proof complete.
\end{proof}

\begin{conclusion}[Fibonacci ladder gaps of top spectrum: second and third maximal fiber multiplicity closed forms and parity limit constants]\label{con:xi-terminal-top-spectrum-fibonacci-ladder-gaps}

Then when \(k\ge 5\) there are closed forms
\[
D_{2k}^{(2)}=F_{k+2}-F_{k-4},\qquad
D_{2k+1}^{(2)}=2F_{k+1}-F_{k-4},
\]
and when \(k\ge 6\)
\[
D_{2k}^{(3)}=F_{k+2}-F_{k-3}.
\]
The odd branch enters exact closed form in the stable phase \(m=2k+1\ge 19\)
\[
D_{2k+1}^{(3)}=2F_{k+1}-\bigl(F_{k-4}+F_{k-8}\bigr).
\]
Thus the top gaps exhibit rigid Fibonacci ladder at exponential scale: for even branch
\(
D_{2k}-D_{2k}^{(2)}=F_{k-4}
\),
\(
D_{2k}-D_{2k}^{(3)}=F_{k-3}
\), and the top two-level gap for odd branch is also explicit Fibonacci combination.
Further, the top relative gap
\(

has parity limit constants
\[
\lim_{k\to\infty}G_{2k}=\varphi^{-6},\qquad
\lim_{k\to\infty}G_{2k+1}=\frac12\varphi^{-5},

with exponential convergence.

\begin{conclusion}[Cubical geometry pressure of time fiber: factorization of \texorpdfstring{$F_x(u)$}{F_x(u)} and growth constant \texorpdfstring{$r_\square(u)=\frac{1+\sqrt{5+4u}}{2}$}{r_square(u)=(1+sqrt(5+4u))/2}]\label{con:xi-terminal-timefiber-cubical-pressure-factorization}

Then \(C(n,k)\) satisfies two-dimensional recurrence
\[
C(n,k)=C(n-1,k)+C(n-2,k)+C(n-2,k-1),
\]
its generating function is rational
\[
\sum_{n\ge 0}C_n(u)z^n=\frac{1+z(1+u)}{1-z-(1+u)z^2},
\]
and has explicit coefficient formula
\[
C(n,k)=\sum_{j=k}^{\lceil n/2\rceil}\binom{n-j+1}{j}\binom{j}{k}.
\]

\[
\lim_{n\to\infty}\frac1n\log C_n(u)=\log r_\square(u),\qquad r_\square(u)=\frac{1+\sqrt{5+4u}}{2}.

so that "fiber cubical pressure" \(u\mapsto \log F_x(u)\) is completely determined by additive combination of \((\ell_i)\), and is uniformly controlled by the same closed-form constant \(r_\square(u)\) in the large-component limit.
\end{conclusion}

\begin{conclusion}[Exact closed form and one-sided sensitivity of Kronecker denominator tick \texorpdfstring{$W_1$}{W1}: good side quadratic, bad side linear]\label{con:xi-terminal-kronecker-w1-denominator-one-sided-sensitivity}
Let \(p/q\) be a reduced fraction, and \(|\alpha-p/q|<1/q^2\).
Write \(\delta=\alpha-p/q\), let the empirical measure of point set \(P_q(\alpha)=\{\{t\alpha\}:t=0,\dots,q-1\}\) be \(\mu_q\), and write
\(
\mathsf{T}_q(\alpha)=W_1(\mu_q,\lambda)
\).
Then there exists a complete closed-form denominator tick formula:
\begin{enumerate}
  \item If \(\delta<0\) (bad side, \(\alpha<p/q\)), then
  \[
  \mathsf{T}_q(\alpha)=\frac{1}{2q}+\frac{q-1}{2}\Bigl(\frac{p}{q}-\alpha\Bigr),
  \qquad
  \mathsf{T}_q(\alpha)=\frac12\,D_q^\ast(P_q(\alpha)).
  \]
  \item If \(\delta>0\) (good side, \(\alpha>p/q\)), then
  \[
  \mathsf{T}_q(\alpha)=\frac{1}{2q}-\frac{q-1}{2}\Bigl(\alpha-\frac{p}{q}\Bigr)
  +\frac{q(q-1)(2q-1)}{6}\Bigl(\alpha-\frac{p}{q}\Bigr)^2,
  \]
  and on this side star discrepancy degenerates to constant \(D_q^\ast(P_q(\alpha))=1/q\) (each box contains exactly one point).
\end{enumerate}
Therefore, on the same "denominator tick," the bad side's response to \(\delta\) is first-order linear (structural bias from double occupancy of first box/vacancy of last box), while the good side manifests only via quadratic term;
this one-sided sensitivity completely algebraizes "approximate rational" geometric error in the \(W_1\) regime, yielding an auditable deterministic closed-form budget.
\end{conclusion}
\begin{proof}[Proof]
See Theorem \ref{thm:kronecker-w1-denominator-closed-form} and Corollary \ref{cor:kronecker-w1-two-sided-interpretation}. Proof complete.
