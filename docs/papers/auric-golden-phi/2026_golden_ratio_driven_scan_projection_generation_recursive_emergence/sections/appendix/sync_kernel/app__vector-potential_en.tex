\subsubsection{Vectorized Extension of Multi-Dimensional Potentials}\label{app:vector-potential}
If one introduces a vector potential
\[
\boldsymbol{\varphi}(e)=\bigl(\varphi_1(e),\dots,\varphi_d(e)\bigr),
\qquad
B(\boldsymbol{\theta})_{ij}=\sum_{e:i\to j}\exp\!\bigl(\langle\boldsymbol{\theta},\boldsymbol{\varphi}(e)\rangle\bigr),
\]
then the pressure
\[
P(\boldsymbol{\theta})=\log\rho\bigl(B(\boldsymbol{\theta})\bigr)
\]
and the large deviation rate function
\[
I(\alpha)=\sup_{\boldsymbol{\theta}\in\RR^d}\bigl(\langle\boldsymbol{\theta},\alpha\rangle-(P(\boldsymbol{\theta})-P(0))\bigr)
\]
both remain within the ``finite matrix + determinant'' auditable caliber; two- and three-dimensional instances are given in subsequent subsections.

\subsubsection{Explicit Closed Form of the Negative-Carry Potential (Auditable Instance)}
Take
\[
Q_-:=\{0\overline{1}2,\,1\overline{1}2,\,01\overline{1},\,11\overline{1}\},
\qquad
\phi(s\to t):=\mathbf{1}_{\{t\in Q_-\}}.
\]
Let $u=e^\theta$. Then $B_\theta$ is equivalent to multiplying all columns landing in $Q_-$ by $u$, and admits the explicit factorization
\[
\det(I-zB_\theta)
\;=\;
\bigl(u z^3-u z^2+u z+1\bigr)\bigl(3u z^3+u z^2-u z-2z^2-2z+1\bigr).
\]
The dominant pole $z_\star(\theta)=\lambda(\theta)^{-1}$ is given by the second factor; substituting $z=1/\lambda$ yields the cubic equation
\[
\lambda^3-(u+2)\lambda^2+(u-2)\lambda+3u=0,\qquad u=e^\theta,
\]
so the pressure $P(\theta)=\log\lambda(\theta)$ is completely determined by the largest real root of this cubic.

\begin{proposition}[Algebraic large deviation of the negative-carry potential: rational parametrization of the slope and closed-form rate function]\label{prop:vector-potential-phi-minus-ldp-param}
Retaining the notation of this subsection, let \(u=e^{\theta}>0\), and let \(\lambda(u)\) be the Perron branch (largest positive real root) of the cubic equation
\[
F(\lambda,u):=\lambda^3-(u+2)\lambda^2+(u-2)\lambda+3u=0
\]
for \(u>0\).
Define the pressure \(P(\theta)=\log\lambda(e^\theta)\) and the slope \(\alpha(\theta):=P'(\theta)\), and define the rate function via Legendre transform:
\[
I(\alpha):=\sup_{\theta\in\RR}\bigl(\theta\alpha-(P(\theta)-P(0))\bigr).
\]
Then there exists an explicit rational parametrization in \(\lambda\in(1+\sqrt3,\infty)\) such that
\[
u(\lambda)=\frac{\lambda(\lambda^{2}-2\lambda-2)}{\lambda^{2}-\lambda-3},
\qquad
\alpha(\lambda)=\frac{\lambda^{4}-3\lambda^{3}-3\lambda^{2}+8\lambda+6}{\lambda^{4}-2\lambda^{3}-5\lambda^{2}+12\lambda+6},
\]
and in this parametric representation the rate function has the closed form
\[
I(\alpha(\lambda))=\alpha(\lambda)\log u(\lambda)-\log\lambda+\log 3.
\]
\end{proposition}
\begin{proof}
By viewing \(F(\lambda,u)=0\) as a linear equation in \(u\), we directly solve
\[
u=\frac{\lambda^{3}-2\lambda^{2}-2\lambda}{\lambda^{2}-\lambda-3}
=\frac{\lambda(\lambda^{2}-2\lambda-2)}{\lambda^{2}-\lambda-3}
=:u(\lambda).
\]
Taking \(u=e^{\theta}\) and applying implicit differentiation,
\[
\lambda_u=-\frac{\partial_u F}{\partial_\lambda F},
\qquad
\alpha(\theta)=\frac{\dd}{\dd\theta}\log\lambda(e^\theta)=\frac{u\,\lambda_u}{\lambda}
=-\frac{u\,\partial_u F(\lambda,u)}{\lambda\,\partial_\lambda F(\lambda,u)}\Bigg|_{\lambda=\lambda(u)}.
\]
For this example,
\(
\partial_u F(\lambda,u)=-(\lambda^2-\lambda-3)
\)
and
\(
\partial_\lambda F(\lambda,u)=3\lambda^2-2(u+2)\lambda+(u-2)
\).
Substituting and simplifying using \(u=u(\lambda)\) yields the stated rational expression for \(\alpha(\lambda)\).
\par
For the rate function, at an extremal point satisfying \(\alpha=P'(\theta)\) under the caliber,
\[
I(\alpha)=\theta\alpha-P(\theta)+P(0).
\]
Here \(\theta=\log u(\lambda)\) and \(P(\theta)=\log\lambda\). Moreover, when \(u=1\),
\(
F(\lambda,1)=\lambda^{3}-3\lambda^{2}-\lambda+3=(\lambda-3)(\lambda^{2}-1)
\),
so the Perron root is \(\lambda(1)=3\), whence \(P(0)=\log 3\).
Substituting yields
\(
I(\alpha(\lambda))=\alpha(\lambda)\log u(\lambda)-\log\lambda+\log 3
\).
\qed
\end{proof}

\subsubsection{Zero-Cost Generalization of Multi-Dimensional Big-Witt Congruence Guardrails: Upgrading the Coefficient Ring Does Not Alter the Divisibility/Congruence Structure}\label{app:multi-dim-witt-guardrails}
The two- and three-dimensional potentials of this appendix do not alter the primitive--periodic Big-Witt/necklace structure: the only difference is that the coefficient ring is upgraded from \(\ZZ[u]\) to a multivariate polynomial ring.
Therefore, the divisibility laws and Dwork-type congruences from Appendix \ref{app:witt-bridge-package} can be extended verbatim to multi-dimensional versions, immediately yielding a ``self-auditing guardrail'' that requires no additional computational overhead and can be directly recomputed.

\paragraph{Two-dimensional potential: \texorpdfstring{$R=\ZZ[u,v]$}{R=Z[u,v]}}
For the two-dimensional weighted matrix \(B(u,v)\) (\S\ref{app:vector-potential}), let
\[
a_n(u,v):=\Tr\bigl(B(u,v)^n\bigr)\in\ZZ[u,v],
\qquad
\psi^d:\ZZ[u,v]\to\ZZ[u,v],\ \ \psi^d(f)(u,v):=f(u^d,v^d).
\]
Then by Proposition \ref{prop:witt-necklace-dwork-R} (taking \(R=\ZZ[u,v]\)), the following hold:
\begin{enumerate}
  \item \textbf{(Multi-dimensional necklace divisibility)} For any \(n\ge 1\),
  \[
  \boxed{\ \sum_{d\mid n}\mu(d)\,a_{n/d}(u^d,v^d)\in n\,\ZZ[u,v]\ }.
  \]
  \item \textbf{(Multi-dimensional Dwork congruence)} For any prime \(p\) and \(k\ge 1\),
  \[
  \boxed{\ a_{p^k}(u,v)\equiv a_{p^{k-1}}(u^p,v^p)\pmod{p^k}\quad\text{in}\ \ZZ[u,v]\ }.
  \]
\end{enumerate}
