\section{算术的涌现：在稳定地址空间上的归一化运算}

本节证明：在稳定类型空间（Zeckendorf 语法）上可定义加法与乘法，使其与经典自然数算术同构；运算本身由规范形与可计算归一化过程给出，从而符合“运算为统计稳定读出规律”的要求。

\subsection{从 Fibonacci 关系呈示的尺度计数单子到自然数（商与不变量）}

本节的关键点是：\textbf{算术结构并非预置}，而是从“尺度计数的自由结构”在 Fibonacci 关系下取商（或取不变量）后被强制出来；Zeckendorf 规范形只是该商结构上的一个\textbf{可计算的规范代表系}。

令
\[
\mathcal{D}:=\Bigl\{a=(a_1,a_2,\dots): a_k\in\NN,\ \text{仅有限个 }a_k\neq 0\Bigr\}.
\]
将 $a\in\mathcal{D}$ 视为“在各尺度 $k$ 上出现 $a_k$ 次”的计数配置；$\mathcal{D}$ 在逐点加法下是自由交换幺半群（自由交换单子）。
记 $e_k\in\mathcal{D}$ 为第 $k$ 个标准基（$e_k$ 在第 $k$ 位为 $1$，其余为 $0$）。

\begin{definition}[Fibonacci 关系生成的同余]\label{def:fib-congruence}
记 $\equiv$ 为 $\mathcal{D}$ 上最小的交换幺半群同余，使得
\begin{enumerate}
  \item \textbf{（基准同余）}$e_2\equiv 2e_1$（把“同一权重的重复”折叠为下一尺度的单次出现）；
  \item \textbf{（递推同余）}对所有 $k\ge 1$ 都有
\[
e_{k+2}\ \equiv\ e_{k+1}+e_k.
\]
\end{enumerate}
等价地，$\equiv$ 由“基准折叠 + Fibonacci 递推”在“尺度计数”层面生成，并对加法封闭。
\end{definition}

\begin{definition}[值同态（不变量）]\label{def:val-on-D}
定义映射 $\mathrm{Val}:\mathcal{D}\to\NN$ 为
\[
\mathrm{Val}(a):=\sum_{k\ge 1} a_k F_{k+1}.
\]
这是交换幺半群同态，并与前述对 $X_\infty$ 的 $\mathrm{Val}$ 定义一致（将 $c\in X_\infty$ 视为 $a_k=c_k$ 的特殊情形）。
\end{definition}

\begin{proposition}[Fibonacci 同余的完备性]\label{prop:val-invariant}
若 $a\equiv b$，则 $\mathrm{Val}(a)=\mathrm{Val}(b)$。因此 $\mathrm{Val}$ 在商幺半群 $\mathcal{D}/\!\equiv$ 上诱导良定义的同态
\[
\overline{\mathrm{Val}}:\mathcal{D}/\!\equiv\ \to\ \NN.
\]
\end{proposition}

\textbf{证明：}
只需验证生成关系保持值。对递推同余 $e_{k+2}\equiv e_{k+1}+e_k$，
\[
\mathrm{Val}(e_{k+2})=F_{k+3}=F_{k+2}+F_{k+1}=\mathrm{Val}(e_{k+1}+e_k),
\]
对基准同余 $e_2\equiv 2e_1$，
\[
\mathrm{Val}(e_2)=F_3=2=2F_2=\mathrm{Val}(2e_1).
\]
其余由同余对加法封闭、以及 $\mathrm{Val}$ 的加法同态性推出。

\begin{theorem}[自然数的涌现：商幺半群的同构]\label{thm:monoid-quotient-is-N}
映射 $\overline{\mathrm{Val}}:\mathcal{D}/\!\equiv \to \NN$ 是交换幺半群同构。并且 $X_\infty$（Zeckendorf 合法序列）给出每个同余类的唯一规范代表：对任意 $a\in\mathcal{D}$，存在唯一 $c\in X_\infty$ 使得 $a\equiv c$ 且 $\mathrm{Val}(a)=\mathrm{Val}(c)$。
\end{theorem}

\textbf{证明：}
\begin{itemize}
  \item \textbf{满射性：}对任意 $N\in\NN$，取其 Zeckendorf 数字 $Z(N)=(c_1,c_2,\dots)\in X_\infty$（\cite{Zeckendorf1972}），则 $\mathrm{Val}(Z(N))=N$，故 $\overline{\mathrm{Val}}$ 满射。
  \item \textbf{同构性（用呈示把一切类压到单生成元）：}
  在商幺半群 $\mathcal{D}/\!\equiv$ 中，令 $\varepsilon:=[e_1]$。由基准同余有 $[e_2]=2\varepsilon$。再由递推同余归纳得对所有 $k\ge 1$，
  \[
  [e_{k+2}]=[e_{k+1}]+[e_k],
  \qquad
  \Rightarrow\qquad
  [e_k]=F_{k+1}\,\varepsilon.
  \]
  因此对任意 $a=\sum_{k\ge 1} a_k e_k\in\mathcal{D}$，
  \[
  [a]=\sum_{k\ge 1} a_k [e_k]=\left(\sum_{k\ge 1} a_k F_{k+1}\right)\varepsilon
  =\mathrm{Val}(a)\,\varepsilon.
  \]
  这说明每个同余类都由其值 $\mathrm{Val}(a)$ 完全刻画，且 $\overline{\mathrm{Val}}([a])=\mathrm{Val}(a)$ 正是把 $n\varepsilon$ 送到 $n$ 的单调同构。于是 $\overline{\mathrm{Val}}$ 为交换幺半群同构。
  \item \textbf{规范代表：}对任意 $a\in\mathcal{D}$，令 $N=\mathrm{Val}(a)$，取 $c=Z(N)\in X_\infty$。上一步给出
  \(
  [a]=N\varepsilon=[c],
  \)
  即 $a\equiv c$。唯一性由 Zeckendorf 唯一性推出：若 $c,c'\in X_\infty$ 且 $c\equiv c'$，则 $\mathrm{Val}(c)=\mathrm{Val}(c')$，从而 $c=c'$。
\end{itemize}

\begin{corollary}[折叠作为商映射的可计算截面]\label{cor:fold-as-section}
上一定理给出的结论可概括为：
\[
\mathcal{D}\ \twoheadrightarrow\ \mathcal{D}/\!\equiv\ \cong\ \NN
\quad\text{且}\quad
X_\infty \hookrightarrow \mathcal{D}
\]
是一个可计算的截面（选择规范代表）。这正是“折叠（folding）= 取不变量/取商 + 选规范形”的抽象机制在 Fibonacci 情形下的具体实现。
\end{corollary}

\textbf{证明：}
由定理 \ref{thm:monoid-quotient-is-N} 的同构性与唯一规范代表结论直接推出。

\subsection{群胚视角：值保持重写的轨道与 Zeckendorf 横截面}\label{sec:groupoid-zeckendorf}

为了把“折叠为何是涌现而非注入”说得更具体，我们把 Fibonacci 关系实现为一族\textbf{可逆局部变换}（群胚生成元），并把 Zeckendorf 合法序列理解为其轨道横截面。

\begin{definition}[Fibonacci 局部变换（可逆重写）]\label{def:fib-moves}
定义以下作用于 $\mathcal{D}$ 的部分可逆变换（在各自可行域上）：
\begin{itemize}
  \item \textbf{基准变换：}
  \[
  E_0(a)=a+2e_1-e_2\quad\text{当 }a_{2}\ge 1,
  \qquad
  E_0^{-1}(a)=a-2e_1+e_2\quad\text{当 }a_{1}\ge 2.
  \]
  \item \textbf{递推变换（$k\ge 1$）：}
  \[
  E_k(a)=a+e_k+e_{k+1}-e_{k+2}\quad\text{当 }a_{k+2}\ge 1,
  \]
  其逆变换 $E_k^{-1}$ 在 $a_k,a_{k+1}\ge 1$ 时定义为
  \[
  E_k^{-1}(a)=a-e_k-e_{k+1}+e_{k+2}.
  \]
\end{itemize}
令 $\mathcal{G}$ 为由 $\{E_k,E_k^{-1}\}_{k\ge 0}$ 生成的群胚（对象为 $\mathcal{D}$，态射为这些部分双射的复合）。
\end{definition}

\begin{proposition}[轨道不变量]\label{prop:orbits-preserve-val}
对任意群胚态射（由 $E_k$ 与 $E_k^{-1}$ 复合得到的可逆变换）$a\mapsto a'$，都有 $\mathrm{Val}(a)=\mathrm{Val}(a')$。
\end{proposition}

\textbf{证明：}
对生成元 $E_k$，
\[
\mathrm{Val}(E_k(a))-\mathrm{Val}(a)
=F_{k+1}+F_{k+2}-F_{k+3}=0,
\]
逆变换同理。复合保持不变量。
对基准生成元 $E_0$ 亦有 $\mathrm{Val}(E_0(a))-\mathrm{Val}(a)=2F_2-F_3=0$，故结论成立。

\begin{theorem}[轨道横截面与折叠的群论解释]\label{thm:zeckendorf-transversal}
对任意 $a\in\mathcal{D}$，其 $\mathcal{G}$-轨道与其值纤维一致：
\[
\mathcal{G}\cdot a \ =\ \{b\in\mathcal{D}:\mathrm{Val}(b)=\mathrm{Val}(a)\}.
\]
并且该轨道与 $X_\infty$ 恰好在一点相交，交点为 $Z(\mathrm{Val}(a))$。因此定义
\[
\Fold_\infty(a):=Z(\mathrm{Val}(a))\in X_\infty
\]
给出“沿轨道投影到横截面”的折叠算子；它把每个不可观测的值保持对称轨道压缩为唯一稳定代表。
\end{theorem}

\textbf{证明：}
由命题 \ref{prop:orbits-preserve-val} 得 $\mathcal{G}\cdot a$ 包含于值纤维。反向包含由定理 \ref{thm:monoid-quotient-is-N} 给出：若 $\mathrm{Val}(b)=\mathrm{Val}(a)$，则 $[b]=[a]$，即 $b\equiv a$。而 $\equiv$ 正是由基准同余 $e_2\equiv 2e_1$ 与递推同余 $e_{k+2}\equiv e_{k+1}+e_k$ 在配置空间上的闭包；相应的局部可逆重写由 $E_0$ 与 $E_k$（$k\ge 1$）实现，故存在群胚态射把 $a$ 送到 $b$。最后，值纤维中落在 $X_\infty$ 的元素按 Zeckendorf 唯一性至多一个，而 $Z(\mathrm{Val}(a))$ 显然属于该纤维，故交点唯一且为所述。

\subsubsection{定向化归一化重写（\texorpdfstring{$\Fold_\infty$}{Fold\_infty} 的编译表述）}\label{subsec:directed-rewrite}
定理 \ref{thm:zeckendorf-transversal} 已经把折叠解释为“沿值保持轨道投影到唯一稳定代表”。为了获得直接可执行的归一化过程，可将群胚生成元定向为一套\textbf{值保持重写}：
\begin{itemize}
  \item \textbf{(R\_adj) 相邻合并：}若 $a_k\ge 1$ 且 $a_{k+1}\ge 1$，则
  \[
  a\ \to\ a-e_k-e_{k+1}+e_{k+2}.
  \]
  \item \textbf{(R\_1) 底位去重：}若 $a_1\ge 2$，则
  \[
  a\ \to\ a-2e_1+e_2.
  \]
  \item \textbf{(R\_2) 第二位去重：}若 $a_2\ge 2$，则
  \[
  a\ \to\ a-2e_2+e_1+e_3.
  \]
  \item \textbf{(R\_k) 一般位去重（$k\ge 3$）：}若 $a_k\ge 2$，则
  \[
  a\ \to\ a-2e_k+e_{k-2}+e_{k+1}.
  \]
\end{itemize}
它们分别对应
\(
F_{k+1}+F_{k+2}=F_{k+3},
2F_2=F_3,
2F_3=F_2+F_4,
2F_{k+1}=F_{k-1}+F_{k+2}
\),
因此均保持 $\mathrm{Val}$ 不变。

\begin{proposition}[终止性与唯一正规形]\label{prop:directed-rewrite-termination}
上述重写系统对任意 $a\in\mathcal{D}$ 强终止，并且不可约项恰为 $X_\infty$；因此任意归约序列终止于唯一正规形，且该正规形等于 $\Fold_\infty(a)$。
\end{proposition}

\textbf{证明：}
设
\[
A(a)=\sum_{k\ge 1} a_k,\qquad
B(a)=\sum_{k\ge 1} k\,a_k,\qquad
C(a)=a_2,
\qquad
\mu(a)=(A(a),B(a),C(a))
\]
按字典序比较。则 (R\_adj) 与 (R\_1) 使 $A$ 减少 1；$(R_k,\,k\ge 3)$ 使 $B$ 减少 1；(R\_2) 保持 $A,B$ 不变但使 $C$ 减少 2。故每步重写都严格降低 $\mu$，系统终止。不可约项必满足 $a_k\le 1$ 且无相邻 $11$，即属于 $X_\infty$；反之 $X_\infty$ 中元素显然不可约。又因每步重写保持 $\mathrm{Val}$，终止形与初态同值；由定理 \ref{thm:zeckendorf-transversal} 的横截面唯一性知正规形唯一且为 $\Fold_\infty(a)$。证毕。

\begin{remark}[残差的两种自然落点]\label{rem:residual}
折叠 $\Fold_\infty$ 是值纤维到横截面的投影；因此“残差”可视为从 $a$ 到 $\Fold_\infty(a)$ 的群胚态射（重写路径/轨道坐标）。在有限分辨率 $m$ 下，则对应附录 D 中的折叠多重度 $d_m$：大量微态压缩到同一稳定类型，丢失的信息即是这些预像的区分标签。
\end{remark}

数值验证与统计复现见脚本 \texttt{scripts/exp\_fold\_infty\_rewrite.py}，其输出为 \texttt{artifacts/export/fold\_infty\_rewrite\_stats.json}。

\subsection{Zeckendorf 值与稳定地址的数值化}

对 $w=w_1\cdots w_m\in X_m$，定义其 Zeckendorf 值
\[
V_m(w)=\sum_{k=1}^m w_k F_{k+1}.
\]
当 $0\le N\le F_{m+2}-1$ 时，$N$ 的 Zeckendorf 数字 $Z(N)=(c_1,c_2,\dots)$ 在第 $m+1$ 位之后全为 $0$（引理 \ref{lem:zeckendorf-range-bijection}），故令
\[
w(N):=\pi_m(Z(N))=(c_1,\dots,c_m)\in X_m,
\]
则
\[
N=V_m\bigl(w(N)\bigr).
\]
因此 $X_m$ 在该范围上可作为自然数的规范“地址表示”。

\subsection{加法作为归一化：从规范形到交换幺半群}

考虑无限 Zeckendorf 地址空间
\[
X_\infty=\{(c_1,c_2,\dots): c_k\in\{0,1\},\ c_kc_{k+1}=0,\ \text{仅有限个 }1\}.
\]
定义值映射
\[
\mathrm{Val}(c)=\sum_{k\ge1}c_kF_{k+1}\in\NN.
\]

为避免将“合流/终止”作为额外假设，本文首先用 Zeckendorf 唯一性给出运算的数学定义（\cite{Zeckendorf1972}），随后再说明其可计算实现（附录 E）。

\begin{definition}[稳定加法]\label{def:stable-add}
对 $c,d\in X_\infty$，定义
\[
c\oplus d:=\mathrm{Val}^{-1}\bigl(\mathrm{Val}(c)+\mathrm{Val}(d)\bigr)\in X_\infty.
\]
\end{definition}

\begin{remark}[关于“同义反复”的澄清：定义性同构 vs. 可观测实现]\label{rem:arith-not-novel-as-abstract}
需要承认：定义 \ref{def:stable-add} 将运算直接拉回到 $(\NN,+)$ 上，因此“\(\mathrm{Val}\) 给出同构”这一结论在抽象代数层面是\textbf{定义性的}，并不产生一个与 \((\NN,+)\) 非同构的新幺半群结构。

本文把该同构写成定理的目的仅是：固定记号并强调“无需把 \(+\) 作为原语输入”。本节真正承担的数学内容是另外两类、与“扫描—投影读出”接口对齐的结论：
\begin{itemize}
  \item \textbf{（轨道/折叠的内容）}由群胚重写生成的轨道与值纤维一致，并存在唯一横截面（定理 \ref{thm:zeckendorf-transversal}）。这给出“折叠 = 沿值保持对称轨道投影到唯一稳定代表”的严格形式。
  \item \textbf{（纯数字层的内容）}运算可等价表述为“逐位叠加/卷积 $\Rightarrow$ 归一化折叠”（推论 \ref{cor:add-as-fold}），从而把抽象的 \(\mathrm{Val}^{-1}\) 定义转成一个可实现、可审计的归一化规则；其复杂度界见附录 E$'$。
\end{itemize}
\end{remark}

\begin{theorem}[Zeckendorf 加法的纯数字层实现（线性时间/有限状态）]\label{thm:zeckendorf-add-fst}
设 $c,d\in X_\infty$ 仅有限个 $1$，并令
\(
m:=\max\{k:\ c_k=1\text{ 或 }d_k=1\}
\)
为有效分辨率。则存在一个仅对 Zeckendorf 数字进行局部有限操作的算法（等价地：一个确定的有限状态换能器/有限延迟顺序换能器），在时间 $O(m)$ 内直接从 $(c_1,\dots,c_m)$ 与 $(d_1,\dots,d_m)$ 计算出 $c\oplus d$ 的 Zeckendorf 数字输出，而不需要显式构造整数 $\mathrm{Val}(c),\mathrm{Val}(d)$。
\end{theorem}

\begin{proof}
这是 Zeckendorf/黄金比例 Ostrowski 数制的标准“归一化可由有限自动机实现”结论在加法特例下的表述：加法可视为先逐位相加得到字母表 $\{0,1,2\}$ 上的非规范数字，再执行一个保持值不变的规范化（消去 $2$ 与相邻 $11$）得到唯一 Zeckendorf 形式。Pisot 基（特别是黄金比例）情形下，这类规范化映射可由有限自动机/顺序换能器实现，从而给出线性时间算法（见 \cite{Frougny1992NumbersAutomata,Frougny1999OnlineAddition}；对 Zeckendorf 情形的显式高效算法与实现细节可参见 \cite{AhlbachUsatineFrougnyPippenger2013}；Ostrowski 三遍扫描算法的黄金比例特例亦见 \cite{HieronymiTerry2018OstrowskiAddition}）。
\end{proof}

\begin{remark}[on-line 与不可并行化：时间读出语义的实现侧约束]\label{rem:online-not-parallel}
在“扫描—读出”的语义里，运算规则若能由一个 on-line 换能器实现，意味着读出可以沿时间单向流式推进并维持有限记忆状态，这与第 6 节的滑动块算子视角一致。另一方面，黄金比例基上的加法虽可 on-line 实现，但一般不可并行化（即不存在把输入划分为并行块并在常数轮数内完成的加法电路口径），这一现象在本文语境中可解释为：\textbf{稳定读出虽是有限记忆的，但仍携带跨尺度的进位/归一化依赖}（参见 \cite{Frougny1999OnlineAddition} 的精确结论与讨论）。
\end{remark}

\subsection{核对比：零式 \texorpdfstring{$+$}{+} $q$-场、carry-free 骨架与“素数影子”}\label{sec:A-kernel-compare-main}
本小节给出一个\textbf{可直接编码/BFS 构图}的统一接口，用以对比黄金比例基 $\tau=\frac{1+\sqrt5}{2}$ 上三类典型并行归一化核（$9/13/21$-local；Algorithm I/II/III \cite{SvobodovaAAMPVIIIParallelAddition}，细节规则见 \cite{FrougnyHellerPelantovaSvobodovaParallelAlgorithmsSlides}）。

\subsubsection{统一生成模板：零式 \texorpdfstring{$+$}{+} $q$-场}
固定 $\beta=\tau$。给定 Laurent 多项式
\[
Z(X)=\sum_{i=-h}^{k} b_i X^i,\qquad Z(\beta)=0,
\]
给定有界整数场 $q=(q_j)_{j\in\ZZ}$（取值于有限集合），定义一次“注入零式”的统一更新
\[
z'_j:=z_j-\sum_{i=-h}^{k} b_i\,q_{j-i}.
\]
因为 $Z(\beta)=0$，故该更新保持值 $\sum_j z'_j\beta^j=\sum_j z_j\beta^j$。

\paragraph{黄金比例三核只用到两种稀疏 stencil}
Algorithm I/II/III 仅使用下列两种稀疏零式（\cite{SvobodovaAAMPVIIIParallelAddition}；推导与伪代码见 \cite{FrougnyHellerPelantovaSvobodovaParallelAlgorithmsSlides}）：
\[
-\tau^4+7-\tau^{-4}=0\ \Rightarrow\ z'_j=z_j-7q_j+q_{j-4}+q_{j+4}\quad (9\text{-local}),
\]
\[
-\tau^2+3-\tau^{-2}=0\ \Rightarrow\ z'_j=z_j-3q_j+q_{j-2}+q_{j+2}\quad (13/21\text{-local}).
\]
三核的差异只在于 $q$ 的\textbf{取值域}与\textbf{触发策略}：一次（SRZ）、定步迭代（WRZ）、两相消元（A$\circ$B）。

\subsubsection{块级在线化：同余流解耦与最小状态规格（单流）}
由于 stencil 仅耦合 $\pm 4$ 或 $\pm 2$，系统按同余类解耦：$\pm 4$ 对应模 $4$ 的 $m=4$ 条流，$\pm 2$ 对应模 $2$ 的 $m=2$ 条流。每条流都可编译为确定性在线机（固定延迟），并可直接对其\textbf{可达状态空间}做 BFS 构图；全局块级在线机是各流的直积装配（见下）。

\begin{proposition}[单流在线机的最小状态变量（可 BFS）]\label{prop:min-state-spec-flow}
在单流坐标下（强核：$j=4n+r$；弱核：$j=2n+r$），三核可取以下最小状态变量，使得“读入一个输入符号 $\Rightarrow$ 状态更新 + 输出（固定延迟）”为确定性局部规则：

\subsubsection{两变量 \texorpdfstring{$\zeta_K(z,u)$}{zeta(z,u)} 与 primitive 分层谱 \texorpdfstring{$B_n(u)$}{B\_n(u)}}
对单流同步图 $\Gamma_K$，以每条边的原子活动度 $\kappa(e)\in\ZZ_{\ge 0}$（本步新决定的非零 $q$ 事件数）作为势，定义带权邻接矩阵族
\[
M_K(u)_{ij}:=\sum_{e:i\to j}u^{\kappa(e)}\in\ZZ[u].
\]
把有限图的 Bowen--Lanford/Brin--Stuck 行列式公式逐点应用到 $M_K(u)$，定义两变量 $\zeta$-指纹
\[
\zeta_K(z,u)
:=\exp\!\left(\sum_{n\ge 1}\frac{\Tr(M_K(u)^n)}{n}z^n\right)
=\frac{1}{\det(I-zM_K(u))}.
\]
这是 Ruelle 加权 $\zeta$ 在有限图情形的具体化（参见 \cite{Ruelle1976ZetaExpanding,ParryPollicott1990Zeta,BrinStuck2002}）。

令 $\mathcal{P}$ 为 $\Gamma_K$ 的 primitive 周期轨道集合，并记 $\kappa(\gamma)=\sum_{e\in\gamma}\kappa(e)$。则 $\zeta_K(z,u)$ 具有 Euler 乘积形式
\[
\zeta_K(z,u)=\prod_{\gamma\in\mathcal{P}}\bigl(1-z^{|\gamma|}u^{\kappa(\gamma)}\bigr)^{-1}.
\]
定义长度恰为 $n$ 的 primitive 轨道“分层素数谱”多项式
\[
B_n(u):=\sum_{\substack{\gamma\in\mathcal{P}\\|\gamma|=n}}u^{\kappa(\gamma)}\in\ZZ[u].
\]
若记 $a_n(u):=\Tr(M_K(u)^n)$，则 Witt/Möbius 反演给出可计算公式
\[
B_n(u)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}(u^d)
=\frac{1}{n}\sum_{d\mid n}\mu(d)\,\Tr\!\left(M_K(u^d)^{\,n/d}\right),
\]
这正把“周期闭环”分解为“primitive 闭环的幂重复”，从而把“素数影子”严格化。

在全局块级直积装配（$m$ 条同余流）且各流同构时，有
\[
M_K^{\mathrm{glob}}(u)=(M_K^{\mathrm{flow}}(u))^{\otimes m},\qquad
a_n^{\mathrm{glob}}(u)=a_n^{\mathrm{flow}}(u)^m,
\]
故只需单流的 $a_n(u)$ 即可装配全局的 $B_n^{\mathrm{glob}}(u)$（再做一次 Möbius 反演即可）。

\subsubsection{压力曲线 \texorpdfstring{$\lambda_K(u)$}{lambda(u)}、端点与 carry 密度}\label{subsec:kernel-pressure-curve}
对每个核的单流带权邻接矩阵族 $M_K(u)$，定义压力曲线（谱半径曲线）
\[
\lambda_K(u):=\rho(M_K(u)),\qquad P_K(u):=\log\lambda_K(u)\qquad(u\in[0,1]).
\]
它刻画“允许 carry 的程度（$u$）”对指数增长率的响应，可作为第二层谱指纹；其数值曲线与 carry-free 素数谱相图见附录 \ref{app:sync-kernel-A-compare} 的图 \ref{fig:parallel-addition-kernels-lambda-u} 与图 \ref{fig:parallel-addition-kernels-bn0}（脚本可复现）。

\begin{proposition}[端点、单调性与多流提升]\label{prop:kernel-pressure-endpoints}
设 $\kappa(e)\in\ZZ_{\ge 0}$，并令 $A_0:=M_K(0)$ 为 carry-free 骨架邻接矩阵。
则：
\begin{enumerate}
  \item \textbf{端点 $u=0$}：$\lambda_K(0)=\rho(A_0)$；
  \item \textbf{端点 $u=1$}：若在线机对每个输入符号都存在一条出边（确定性完备读入），则 $\lambda_K(1)=|B|$；
  \item \textbf{单调性/连续性}：$u\mapsto \lambda_K(u)$ 在 $[0,1]$ 上非减且连续；
  \item \textbf{多流提升}：若全局块级由 $m$ 条同构同余流张量装配，则
  \(
  \lambda^{\mathrm{glob}}_K(u)=\lambda^{\mathrm{flow}}_K(u)^m
  \)
  且
  \(
  P^{\mathrm{glob}}_K(u)=m\,P^{\mathrm{flow}}_K(u)
  \)。
\end{enumerate}
\end{proposition}

\begin{remark}[carry 密度与压力导数（热力学形式）]
把 $\kappa$ 视为边势并令 $\theta=\log u$，则 $P_K(u)=\log\rho(M_K(e^\theta))$ 是有限图移位的压力函数。
在不可约情形，$P_K$ 在 $(0,1)$ 上可微，并满足
\[
u\frac{\mathrm{d}}{\mathrm{d}u}\log\lambda_K(u)
=\frac{\mathrm{d}}{\mathrm{d}\theta}P_K(e^\theta)
=\mathbb{E}_{\mu_u}[\kappa],
\]
其中 $\mu_u$ 为势 $\theta\kappa$ 的平衡态（finite-type 情形的 Ruelle/Parry--Pollicott 口径；参见 \cite{Ruelle1978Thermodynamic,ParryPollicott1990Zeta}）。
这给出一个与“均匀输入驱动”\textbf{不同}但可比较的 carry 密度读出：前者来自外加输入分布，后者来自压力最优化（指纹层更稳定）。
\end{remark}

\begin{corollary}[折叠公式：加法即“叠加后归一化”]\label{cor:add-as-fold}
将 $c,d\in X_\infty$ 视为 $\mathcal{D}$ 中的 $0/1$ 配置，令其逐点和为
\[
(c+d)_k:=c_k+d_k\in\{0,1,2\},
\]
则
\[
c\oplus d=\Fold_\infty(c+d).
\]
\end{corollary}

\textbf{证明：}
由定义 \ref{def:val-on-D}，$\mathrm{Val}(c+d)=\mathrm{Val}(c)+\mathrm{Val}(d)$，再由定理 \ref{thm:zeckendorf-transversal} 的定义 $\Fold_\infty(a)=Z(\mathrm{Val}(a))=\mathrm{Val}^{-1}(\mathrm{Val}(a))$ 即得。

\begin{theorem}[加法的定义性同构（记号固定）]\label{thm:add-definitional}
$(X_\infty,\oplus)$ 为交换幺半群，且 $\mathrm{Val}:(X_\infty,\oplus)\to(\NN,+)$ 为同构。
\end{theorem}

\textbf{证明：}
由定义 \ref{def:stable-add} 直接有 $\mathrm{Val}(c\oplus d)=\mathrm{Val}(c)+\mathrm{Val}(d)$。又 $\mathrm{Val}$ 为双射（Zeckendorf 唯一性），故 $(X_\infty,\oplus)$ 的结合律、交换律与单位元均由 $(\NN,+)$ 传递得到，且 $\mathrm{Val}$ 给出同构。

该定理将加法解释为“从非规范叠加到稳定地址的投影读出”：原始叠加（逐位相加）一般不保持稳定语法，需要通过规范化投影到稳定地址空间。

\subsection{乘法作为二层归一化：卷积与规范形}

\begin{definition}[稳定乘法]\label{def:stable-mul}
对 $c,d\in X_\infty$，定义
\[
c\otimes d:=\mathrm{Val}^{-1}\bigl(\mathrm{Val}(c)\cdot\mathrm{Val}(d)\bigr)\in X_\infty.
\]
\end{definition}

\begin{theorem}[乘法的定义性同构（记号固定）]\label{thm:mul-definitional}
$(X_\infty,\oplus,\otimes)$ 为交换半环，且 $\mathrm{Val}$ 给出与 $(\NN,+,\times)$ 的半环同构。
\end{theorem}

\textbf{证明：}
由定义 \ref{def:stable-add} 与 \ref{def:stable-mul}，$\mathrm{Val}$ 同时保持加法与乘法。由于 $\mathrm{Val}$ 为双射，交换律、结合律与分配律均从 $(\NN,+,\times)$ 传递，因而得到交换半环同构。

由此，自然数、加法、乘法在稳定地址空间中完全由\textbf{语法约束 + 规范形归一化}定义，而不需将 $(+,\times)$ 作为原语预置。

\subsubsection{乘法由加法迭代内生}\label{subsec:mul-as-iterated-add}
\begin{theorem}[乘法可由加法与迭代内生]\label{thm:mul-by-iterated-add}
在 $X_\infty$ 上定义
\[
c\odot d:=\underbrace{c\oplus c\oplus\cdots\oplus c}_{\mathrm{Val}(d)\text{ 次}},
\qquad
\mathrm{Val}(d)=0\text{ 时取 }0_X,
\]
则对任意 $c,d\in X_\infty$，$c\odot d=c\otimes d$。
\end{theorem}

\textbf{证明：}
由定理 \ref{thm:add-definitional}，$\mathrm{Val}$ 保持 $\oplus$ 且双射，故
\[
\mathrm{Val}(c\odot d)=\mathrm{Val}(c)\mathrm{Val}(d)=\mathrm{Val}(c\otimes d),
\]
从而 $c\odot d=c\otimes d$。证毕。

\begin{corollary}[“合并”的第一种严格版本]\label{cor:mul-no-new-primitive}
$\otimes$ 由 $\oplus$ 与可迭代计数唯一确定，无需作为新的原语引入。
\end{corollary}

\subsubsection{后继与折叠：把 \texorpdfstring{$\oplus$}{oplus} 也消隐}\label{subsec:succ-and-fold}
\begin{definition}[后继与折叠]\label{def:successor-fold}
记 $0_X:=(0,0,\dots)\in X_\infty$，$e_1\in\mathcal{D}$ 为第 1 个标准基。定义
\[
S:X_\infty\to X_\infty,\qquad S(c):=\Fold_\infty(c+e_1).
\]
\end{definition}

\begin{theorem}[后继结构同构]\label{thm:successor-structure}
$\mathrm{Val}$ 给出结构同构
\[
\mathrm{Val}(0_X)=0,\qquad \mathrm{Val}(S(c))=\mathrm{Val}(c)+1,
\]
并且 $\mathrm{Val}$ 为双射。
\end{theorem}

\textbf{证明：}
$\mathrm{Val}(0_X)=0$。又
\[
\mathrm{Val}(S(c))=\mathrm{Val}(c+e_1)=\mathrm{Val}(c)+\mathrm{Val}(e_1)=\mathrm{Val}(c)+1,
\]
且 $\mathrm{Val}(e_1)=F_2=1$；双射性由 Zeckendorf 唯一性。证毕。

\begin{corollary}[加法由后继迭代给出]\label{cor:add-from-successor}
令
\[
c\oplus_S d:=S^{\circ\mathrm{Val}(d)}(c),
\]
则 $c\oplus_S d=c\oplus d$，且 $c\oplus d=\Fold_\infty(c+d)$。
\end{corollary}

\textbf{证明：}
由定理 \ref{thm:successor-structure}，$\mathrm{Val}(c\oplus_S d)=\mathrm{Val}(c)+\mathrm{Val}(d)$，由双射性得 $c\oplus_S d=c\oplus d$；折叠公式由推论 \ref{cor:add-as-fold}。证毕。

\begin{theorem}[乘法由后继迭代内生]\label{thm:mul-from-successor}
定义
\[
c\otimes_S d:=\underbrace{c\oplus_S c\oplus_S\cdots\oplus_S c}_{\mathrm{Val}(d)\text{ 次}},
\qquad \mathrm{Val}(d)=0\text{ 时取 }0_X,
\]
则 $c\otimes_S d=c\otimes d$。
\end{theorem}

\textbf{证明：}
由推论 \ref{cor:add-from-successor}，$\oplus_S=\oplus$，故
\[
\mathrm{Val}(c\otimes_S d)=\mathrm{Val}(c)\mathrm{Val}(d)=\mathrm{Val}(c\otimes d),
\]
由双射性得结论。证毕。

\begin{corollary}[极简原语集]\label{cor:primitive-minimal-successor}
稳定地址层的算术由后继 $S$ 与折叠 $\Fold_\infty$ 生成：$\oplus$ 为 $S$ 的一层迭代读出，$\otimes$ 为该迭代的二层叠加读出。
\end{corollary}

\subsubsection{单一原语 \texorpdfstring{$\circ$}{o} 的两层读出}\label{subsec:arith-composition}
本小节给出一个与“扫描迭代 $\Rightarrow$ 稳定编码 $\Rightarrow$ 归一化读出”链条对齐的定理化版本：把加法与乘法从原语层彻底移除，只保留\textbf{复合}这一最底层原语，并在稳定地址层通过折叠回到规范形。
取任意非空集合 $A$ 与自映射 $g:A\to A$。记
\[
S_0:=\langle g\rangle=\{g^{\circ n}:n\in\NN\}\subseteq \mathrm{End}(A),\qquad
S_1:=\{I_n:n\in\NN\}\subseteq \mathrm{End}(\mathrm{End}(A)),
\]
其中 $I_n(f):=f^{\circ n}$。两层对象的唯一运算均为复合 $\circ$。

\begin{theorem}[单一原语 \texorpdfstring{$\circ$}{o} 的两层读出同时生成 $(+,\times)$]\label{thm:composition-two-layer}
定义嵌入 $\iota_0:\NN\to S_0$ 为
\[
\iota_0(n):=g^{\circ n},\qquad g^{\circ 0}:=\mathrm{id}_A.
\]
并令 $\iota_1:\NN\to S_1$ 为 $\iota_1(n):=I_n$。则对任意 $m,n\in\NN$，成立
\[
\iota_0(m+n)=\iota_0(m)\circ \iota_0(n),
\qquad
\iota_1(mn)=\iota_1(m)\circ \iota_1(n).
\]
\end{theorem}

\textbf{证明：}
由迭代定义与复合结合律，
\[
g^{\circ(m+n)}=g^{\circ m}\circ g^{\circ n},
\]
故 $\iota_0(m+n)=\iota_0(m)\circ \iota_0(n)$。另一方面，对任意 $f\in\mathrm{End}(A)$，
\[
(\iota_1(m)\circ \iota_1(n))(f)=I_m(I_n(f))=I_m(f^{\circ n})=(f^{\circ n})^{\circ m}
=f^{\circ (nm)}=I_{mn}(f).
\]
由于对一切 $f$ 相等，故 $\iota_1(m)\circ \iota_1(n)=\iota_1(mn)$。证毕。

\begin{theorem}[稳定地址半环的“去原语化”实现]\label{thm:arith-composition}
沿用定义 \ref{def:stable-add} 与 \ref{def:stable-mul}，并记 $\mathrm{Val}:X_\infty\to\NN$ 为值映射。对任意定理 \ref{thm:composition-two-layer} 中的 $(A,g)$，定义两条“过程语义”映射
\[
\Phi_0:X_\infty\to \mathrm{End}(A),\quad \Phi_0(c):=g^{\circ\mathrm{Val}(c)},
\]
\[
\Phi_1:X_\infty\to \mathrm{End}(\mathrm{End}(A)),\quad \Phi_1(c):=I_{\mathrm{Val}(c)}.
\]
则对任意 $c,d\in X_\infty$，
\[
\Phi_0(c\oplus d)=\Phi_0(c)\circ\Phi_0(d),\qquad
\Phi_1(c\otimes d)=\Phi_1(c)\circ\Phi_1(d).
\]
也即：在稳定地址层的 $\oplus,\otimes$ 同时被同一原语 $\circ$ 在两层对象上实现。
\end{theorem}

\textbf{证明：}
由 $\oplus$ 定义与定理 \ref{thm:composition-two-layer} 的第一条，
\[
\Phi_0(c\oplus d)=g^{\circ\mathrm{Val}(c\oplus d)}
=g^{\circ(\mathrm{Val}(c)+\mathrm{Val}(d))}
 =g^{\circ\mathrm{Val}(c)}\circ g^{\circ\mathrm{Val}(d)}
=\Phi_0(c)\circ\Phi_0(d).
\]
同理由 $\otimes$ 定义与定理 \ref{thm:composition-two-layer} 的第二条，
\[
\Phi_1(c\otimes d)=I_{\mathrm{Val}(c\otimes d)}
=I_{\mathrm{Val}(c)\mathrm{Val}(d)}
=I_{\mathrm{Val}(c)}\circ I_{\mathrm{Val}(d)}
=\Phi_1(c)\circ\Phi_1(d).
\]
证毕。

\begin{corollary}[仅复合的读出落回稳定地址]\label{cor:composition-pullback}
记 $Z:=\mathrm{Val}^{-1}$，并定义
\[
\rho_0:S_0\to X_\infty,\quad \rho_0(g^{\circ n}):=Z(n),\qquad
\rho_1:S_1\to X_\infty,\quad \rho_1(I_n):=Z(n).
\]
则对任意 $c=Z(m),d=Z(n)\in X_\infty$，
\[
c\oplus d=\rho_0\bigl(\rho_0^{-1}(c)\circ \rho_0^{-1}(d)\bigr),
\qquad
c\otimes d=\rho_1\bigl(\rho_1^{-1}(c)\circ \rho_1^{-1}(d)\bigr).
\]
\end{corollary}

\textbf{证明：}
由定理 \ref{thm:composition-two-layer} 得 $\rho_0(g^{\circ m}\circ g^{\circ n})=Z(m+n)$ 与 $\rho_1(I_m\circ I_n)=Z(mn)$，再与 $\oplus,\otimes$ 的定义对齐即可。证毕。

\begin{corollary}[“合并掉 $(+,\times)$”的严格表述]\label{cor:primitive-free-arith}
在配置层，算术原语可压缩为以下最小生成集：
\begin{itemize}
  \item 计数配置的逐点叠加（自由交换幺半群结构）；
  \item Fibonacci 关系诱导的局部可逆重写（群胚生成元）；
  \item 选取规范形截面 $\Fold_\infty$（或等价的 $\mathrm{Val}$ 与 Zeckendorf 规范形 $Z$）。
\end{itemize}
在过程层，等价地可用复合运算 $\circ$ 与折叠 $\Fold_\infty$ 来表达：$\oplus$ 与 $\otimes$ 分别是复合在一层与二层对象上的读出，并经折叠归一化后落回 $X_\infty$。
等价地，运算层只需 $\circ$，可见层只需编码 $Z$（或 $\mathrm{Val}/\Fold_\infty$）。
\end{corollary}

\textbf{证明：}
配置层的三类原语由定理 \ref{thm:monoid-quotient-is-N}、定理 \ref{thm:zeckendorf-transversal} 与推论 \ref{cor:fold-as-section} 直接给出；过程层的两层复合读出由定理 \ref{thm:composition-two-layer} 与 \ref{thm:arith-composition} 给出。加法的折叠接口由推论 \ref{cor:add-as-fold} 保证，因而结论成立。

\begin{proposition}[可计算实现]\label{prop:zeckendorf-arith-computable}
对任意 $c,d\in X_\infty$，$c\oplus d$ 与 $c\otimes d$ 可由有限步算法计算：先在值层执行整数加法/乘法得到 $\mathrm{Val}(c)\pm\mathrm{Val}(d)$ 或 $\mathrm{Val}(c)\mathrm{Val}(d)$，再以 Zeckendorf 贪心算法取唯一规范形（附录 C）。
\end{proposition}

\subsection{显式算法与复杂度界（审稿级）}\label{subsec:zeckendorf-algorithms-complexity}

本小节把“可计算、可终止”的表述提升为\textbf{显式算法}与\textbf{复杂度界}。为保持依赖最小，算法采用最短闭环：\textbf{值层运算}（整数加法/乘法）+\textbf{Zeckendorf 规范化}（贪心）。

\subsubsection{Zeckendorf 规范化：从整数到合法数字}

给定 $N\in\NN$，目标是计算其唯一 Zeckendorf 数字 $Z(N)=(c_1,c_2,\dots)$（仅有限个 $1$ 且无相邻 $1$）使得 $N=\sum_{k\ge 1}c_kF_{k+1}$。

\begin{itemize}
  \item \textbf{输入：}整数 $N$。
  \item \textbf{输出：}Zeckendorf 合法数字序列 $Z(N)$。
\end{itemize}

\textbf{算法（贪心 + 跳邻位）：}
\begin{enumerate}
  \item 预生成 Fibonacci 权重 $F_{2},F_{3},\dots,F_{M+1}$，其中 $M$ 足够大使得 $F_{M+1}\le N<F_{M+2}$；
  \item 从 $k=M$ 递减到 $1$：若 $F_{k+1}\le N$，则置 $c_k:=1$、令 $N\leftarrow N-F_{k+1}$，并跳过相邻位（将 $k\leftarrow k-1$ 再继续递减）；否则置 $c_k:=0$；
  \item 输出 $c=(c_1,c_2,\dots,c_M)$（其后全为 $0$）。
\end{enumerate}

该算法的正确性与唯一性由 Zeckendorf 定理给出（\cite{Zeckendorf1972}；本稿中见附录 C）。

\subsubsection{稳定加法与稳定乘法：两步实现}

对输入 $c,d\in X_\infty$（仅有限个 $1$），令其有效分辨率为
\[
m:=\max\{k:\ c_k=1\text{ 或 }d_k=1\}.
\]
则
\begin{itemize}
  \item \textbf{稳定加法} $c\oplus d$：计算 $N=\mathrm{Val}(c)+\mathrm{Val}(d)$，再输出 $Z(N)$；
  \item \textbf{稳定乘法} $c\otimes d$：计算 $N=\mathrm{Val}(c)\mathrm{Val}(d)$，再输出 $Z(N)$。
\end{itemize}
这正是附录 E 的“值层运算 + 规范形”的最短闭环实现。

\subsubsection{复杂度（按分辨率 \texorpdfstring{$m$}{m} 与按比特两种口径）}

为了避免模型争议，本文在附录 E$'$ 中同时给出两种等价口径的复杂度界：分辨率口径（把 $m$ 视为输入长度）与比特口径（把整数按二进制长度计成本）。结论可概括如下（详证见附录 E$'$ 中的命题 \ref{prop:fold-complexity} 与 \ref{prop:stable-arith-complexity}）：

\begin{itemize}
  \item \textbf{折叠/规范化：}对长度 $m$ 的输入，Zeckendorf 规范化在分辨率口径为 $O(m)$ 步；在比特口径为 $O(m^2)$ 比特运算。
  \item \textbf{稳定加法：}分辨率口径 $O(m)$；比特口径 $O(m^2)$。
  \item \textbf{稳定乘法：}分辨率口径为“值层乘法成本 + $O(m)$”；比特口径为 $O(M(m)+m^2)$，其中 $M(m)$ 为 $\\Theta(m)$ 比特整数乘法成本（学校乘法为 $O(m^2)$）。
\end{itemize}

\subsubsection{纯局部归一化（可选扩展）}\label{subsec:local-normalize}
本方向用 Zeckendorf 数字上的局部 carry/normalize 直接实现 $\Fold_\infty$，避免显式值层运算。以逐位叠加得到的串 $s\in\{0,1,2\}^m$ 为输入，给出一个三趟局部规则体系（每步保持 $\mathrm{Val}$；规则可由 Fibonacci 递推验证，见 \cite{AhlbachUsatineFrougnyPippenger2013}）：
\begin{itemize}
  \item \textbf{第 1 趟（宽度 4，左到右）}：对 $x\in\{0,1,2\}$，
  \[
  020x\Rightarrow 100x^{+},\quad
  030x\Rightarrow 110x^{+},\quad
  021x\Rightarrow 110x,\quad
  012x\Rightarrow 101x,
  \]
  其中 $x^{+}:=x+1$。末端清理可取
  \[
  030\Rightarrow 111,\quad 020\Rightarrow 101,\quad 0120\Rightarrow 1010,\quad
  03\Rightarrow 11,\quad 02\Rightarrow 10,\quad 012\Rightarrow 101.
  \]
  \item \textbf{第 2--3 趟（宽度 3）}：用规则
  \[
  011\Rightarrow 100
  \]
  先右到左、再左到右各扫一遍，消去相邻 $11$。
\end{itemize}

\begin{theorem}[三趟局部归一化与 $\Fold_\infty$ 一致]\label{thm:local-normalize-fold}
令 $T:=T_3\circ T_2\circ T_1$ 为上述三趟归一化。则对任意 $s\in\{0,1,2\}^m$，
\[
T(s)=\Fold_\infty(s)\in X_m.
\]
\end{theorem}

\textbf{证明：}
每条规则保持 $\mathrm{Val}$（等价地：在群胚生成元轨道内移动），且第 2--3 趟保证输出无相邻 $11$，从而落在 $X_m$。由定理 \ref{thm:zeckendorf-transversal} 的横截面唯一性交点，终止形必为 $\Fold_\infty(s)$。证毕。

\begin{corollary}[有限状态实现与线性时间]\label{cor:local-normalize-fst}
上述三趟均可由有限状态转导器以常数延迟实现，故分辨率口径下归一化为 $O(m)$。例如第 1 趟可取状态集 $Q_1=\{0,1,2,3\}^3$（至多 $4^3$ 个状态），第 2--3 趟可取 $Q_2=\{0,1\}^2$（至多 $2^2$ 个状态）。\cite{AhlbachUsatineFrougnyPippenger2013,Frougny1999OnlineAddition}
\end{corollary}

\subsubsection{单趟在线延迟 \texorpdfstring{$3$}{3}：统一转导器}\label{subsec:online-delay}
三趟局部规则可视为转导器串联的产品自动机。更强的结果是存在\textbf{单趟在线}转导器（延迟 3）直接实现同一归一化；其同步部分可用“余量守恒”给出（\cite{Frougny1999OnlineAddition}）。

令 $\varphi=(1+\sqrt5)/2$，定义余量
\[
R(s):=s_1\varphi^{-1}+s_2\varphi^{-2}+s_3\varphi^{-3}\in[0,1),
\]
并令输入 $d\in\{0,1,2\}$、输出 $e\in\{0,1\}$。同步边 $(s\to t)$ 满足守恒式
\[
R(s)+d\varphi^{-4}=e\varphi^{-1}+t_1\varphi^{-2}+t_2\varphi^{-3}+t_3\varphi^{-4},
\]
等价地
\[
R(t)=\varphi R(s)+d\varphi^{-3}-e,\qquad R(t)\in[0,1).
\]
该同步部分的状态集可取为
\[
Q=\{000,001,002,010,100,101,0\overline{1}2,1\overline{1}2,01\overline{1},11\overline{1}\},
\]
并且出边（标记 $d/e$）可写为：
\[
\begin{aligned}
000&\xrightarrow{d/0}00d,\qquad 100\xrightarrow{d/1}00d\quad(d=0,1,2),\\
001&\xrightarrow{0/0}010,\ \xrightarrow{1/0}100,\ \xrightarrow{2/0}101,\\
002&\xrightarrow{0/0}11\overline{1},\ \xrightarrow{1/1}000,\ \xrightarrow{2/1}001,\\
010&\xrightarrow{0/0}100,\ \xrightarrow{1/0}101,\ \xrightarrow{2/1}0\overline{1}2,\\
101&\xrightarrow{0/1}010,\ \xrightarrow{1/1}100,\ \xrightarrow{2/1}101,\\
0\overline{1}2&\xrightarrow{0/0}01\overline{1},\ \xrightarrow{1/0}010,\ \xrightarrow{2/0}100,\\
1\overline{1}2&\xrightarrow{0/1}01\overline{1},\ \xrightarrow{1/1}010,\ \xrightarrow{2/1}100,\\
01\overline{1}&\xrightarrow{0/0}001,\ \xrightarrow{1/0}002,\ \xrightarrow{2/0}1\overline{1}2,\\
11\overline{1}&\xrightarrow{0/1}001,\ \xrightarrow{1/1}002,\ \xrightarrow{2/1}1\overline{1}2.
\end{aligned}
\]
末端函数可取
\[
\omega(s_1s_2s_3):=Z(s_1F_4+s_2F_3+s_3F_2),
\]
以吞吐延迟窗口的残余并落回 Zeckendorf 规范形（\cite{Frougny1999OnlineAddition}）。

\begin{theorem}[单趟在线延迟 3 的 $\Fold_\infty$ 实现]\label{thm:online-delay-fold}
以上在线转导器（含末端函数）对任意 $s\in\{0,1,2\}^m$ 输出 $\Fold_\infty(s)$；其同步图即附录 B 的 10 状态同步核。
\end{theorem}

\textbf{证明：}
守恒式保证值不变量，末端函数输出落在 $X_m$；由定理 \ref{thm:zeckendorf-transversal} 的横截面唯一性得 $\Fold_\infty(s)$。同步图与附录 B 的状态/转移一致。证毕。

\begin{remark}[操作数不一致与隐含属性]
原子生成元 $E_k,E_k^{-1}$ 具有尺度参数 $k$ 与守卫条件，体现为“隐含类型”。令 $\nu(a):=\sum_k a_k$ 表示块数，则
\[
\nu(E_k(a))=\nu(a)+1,\qquad \nu(E_k^{-1}(a))=\nu(a)-1,
\]
即局部裂解/合并天然呈现 $1\leftrightarrow 2$ 的操作数不一致性。
\end{remark}

\begin{remark}[primitive 轨道与“素数影子”]\label{rem:primitive-shadow}
上述在线转导器的同步图是有限有向图，其 primitive 周期轨道给出“不可约循环模式”。设其邻接矩阵为 $B$，则周期点计数 $a_n=\mathrm{Tr}(B^n)$，primitive 轨道数 $p_n$ 由
\[
a_n=\sum_{d\mid n} d\,p_d,\qquad
p_n=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}
\]
给出，并与第 9 节的 $\zeta$--欧拉乘积结构一致（参见 \cite{LindMarcus1995} 与附录 B 中的 $\zeta_B$、显式分解与数值谱）。
\end{remark}
