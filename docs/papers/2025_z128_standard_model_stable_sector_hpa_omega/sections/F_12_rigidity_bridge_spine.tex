\subsection{One constant, three readings: \texorpdfstring{$\log(2/\varphi)$}{log(2/phi)}}
\label{subsec:rigidity_log2phi_triple_reading}

\begin{proposition}[The constant $\log(2/\varphi)$ as a rigidity bridge]\label{prop:rigidity_log2phi_triple_reading}
\MathTag Let $\varphi=(1+\sqrt{5})/2$.
The same constant $\log(2/\varphi)$ admits the following three equivalent readings within this manuscript:
\begin{enumerate}
\item \textbf{Degeneracy growth (capacity gap).}
The mean degeneracy $d_m=2^m/F_{m+2}$ satisfies
\[
\lim_{m\to\infty}\frac{1}{m}\log d_m=\log(2/\varphi)
\]
(Lemma~\ref{lem:entropy_gap_hidden_exponent_cosmo}).
\item \textbf{Conditional entropy rate (finite observability).}
For $N$ uniform on $K_m=\{0,\dots,2^m-1\}$ and $W=\mathrm{Fold}_m(N)$, one has
\[
\lim_{m\to\infty}\frac{1}{m}H(N\mid W)=\log(2/\varphi)
\]
(Proposition~\ref{prop:folding_relative_entropy_decomposition}).
\item \textbf{Pressure gap (zero potential).}
For the full two-shift $\Sigma_2$ and the golden-mean shift $\Sigma_{\mathrm{GM}}$, the zero-potential pressures satisfy
\[
P_{\Sigma_2}(0)-P_{\Sigma_{\mathrm{GM}}}(0)=\log 2-\log\varphi=\log(2/\varphi)
\]
(Lemma~\ref{lem:full_shift_entropy_gap}).
\end{enumerate}
\end{proposition}

\begin{remark}[Why this matters for the narrative]\label{rem:rigidity_log2phi_why}
\AuditTag The paper uses the constant $\log(2/\varphi)$ in several places under different names
(\emph{entropy gap}, \emph{hidden exponent}, \emph{capacity gap}, or as an Abel/pole-barrier-aligned growth rate).
Proposition~\ref{prop:rigidity_log2phi_triple_reading} records that these are not independent numerical coincidences:
they are the same rigidity constant seen through three equivalent object choices.
\end{remark}

\begin{proposition}[Fiber-entropy and relative-entropy squeezing]\label{prop:fiber_entropy_relative_entropy_squeezing}
\MathTag Fix $m\ge 1$ and let $K_m=\{0,\dots,2^m-1\}$.
Let $N$ be uniform on $K_m$ and let $W:=\mathrm{Fold}_m(N)\in X_m$.
Write $P(w):=\mathrm{Fold}_m^{-1}(w)$ for fibers, define the mean degeneracy $d_m:=|K_m|/|X_m|=2^m/|X_m|$, and the maximal degeneracy $r_m:=\max_{w\in X_m}|P(w)|$.
Let $\mu_m$ denote the induced distribution $\mu_m(w)=|P(w)|/2^m$ on $X_m$ and let $u_m$ be uniform on $X_m$.
Then:
\begin{enumerate}
\item \textbf{Entropy squeezing:}
\[
\log d_m \;\le\; H(N\mid W) \;\le\; \log r_m.
\]
\item \textbf{Relative-entropy squeezing:}
\[
0\;\le\; D(\mu_m\Vert u_m)\;\le\;\log(r_m/d_m).
\]
\item \textbf{Rate consequence:} using $\log d_m=m\log(2/\varphi)+O(1)$ (Lemma~\ref{lem:entropy_gap_hidden_exponent_cosmo}) and $\log r_m=m\log(2/\varphi)+O(1)$ (Corollary~\ref{cor:rm_growth_rate}), one has
\[
D(\mu_m\Vert u_m)=O(1),
\qquad
\lim_{m\to\infty}\frac{1}{m}D(\mu_m\Vert u_m)=0.
\]
\end{enumerate}
\end{proposition}
\begin{proof}
By Proposition~\ref{prop:folding_relative_entropy_decomposition} one has $H(N\mid W)=\log d_m + D(\mu_m\Vert u_m)$, hence $H(N\mid W)\ge \log d_m$ and $D(\mu_m\Vert u_m)\ge 0$.
Also,
\[
D(\mu_m\Vert u_m)=\mathbb{E}_{\mu_m}\!\left[\log\!\left(\frac{|P(W)|}{d_m}\right)\right]
\le \log\!\left(\frac{r_m}{d_m}\right),
\]
since $|P(W)|\le r_m$ almost surely.
Finally, $H(N\mid W)=\mathbb{E}_{\mu_m}[\log|P(W)|]\le \log r_m$ because $\log|P(W)|\le \log r_m$ pointwise.
The rate consequence follows because $\log(r_m/d_m)=\log r_m-\log d_m=O(1)$ by the cited rigidity rates.
\end{proof}

\subsection{Determinants, traces, and zeta: a rigid bookkeeping identity}
\label{subsec:rigidity_det_trace_zeta}

\begin{lemma}[Log-determinant as a trace generating series]\label{lem:rigidity_logdet_trace_series}
\MathTag For any square matrix $M$, one has the formal identity
\[
-\log\det(I-zM)=\sum_{k\ge 1}\frac{z^k}{k}\,\mathrm{tr}(M^k),
\]
as a power series around $z=0$.
In the trace-class operator setting, the same identity holds on the disk of convergence (Remark~\ref{rem:logdet_trace_expansion}).
\end{lemma}
\begin{proof}
Expand $\log(I-zM)=-\sum_{k\ge 1} z^k M^k/k$ formally and use $\log\det=\mathrm{tr}\log$.
\end{proof}

\begin{remark}[Refinement operators as determinant/trace objects]\label{rem:rigidity_refinement_to_det_trace}
\AuditTag The refinement operators $T_L:=A^L$ (Definition~\ref{def:refinement_operators_TL}) are powers of the golden-mean transition matrix $A$.
Lemma~\ref{lem:rigidity_logdet_trace_series} implies that the traces $\mathrm{tr}(A^k)$ are rigidly packaged by $\det(I-zA)$.
This is the shared algebraic core behind (i) extension-count operator evaluations, (ii) periodic-orbit/zeta expansions, and (iii) pole-barrier normalizations.
For a graph-theoretic realization of the same determinant/prime-cycle packaging, see Appendix~\ref{app:graph_zeta_holonomy}.
\AuditTag For a fully finite-state refinement of this packaging that separates periodic-point counts into primitive-cycle counts via M\"obius inversion (a concrete ``prime cycle'' template for the golden-branch grammar), see Appendix~\ref{app:zeckendorf_shift_prime_cycles}.
\end{remark}

\begin{corollary}[Pole barrier as an exponential of pressure (matrix/SFT form)]\label{cor:pole_barrier_equals_exp_minus_pressure}
\MathTag Let $(X,\sigma)$ be a mixing shift of finite type with adjacency matrix $A$.
Then the dominant singularity (pole barrier) of the associated determinant/zeta package occurs at
\[
|z|_\star=\rho(A)^{-1}=\exp(-P(0)),
\]
where $P(0)=\log\rho(A)$ is the zero-potential pressure (Theorem~\ref{thm:pressure_spectral_radius_standard}).
Equivalently, the Abel normalization $z=r/\rho(A)$ places the leading pole barrier at $r=1$.
\end{corollary}
\begin{proof}
For a one-step SFT, $P(0)=\log\rho(A)$ (Theorem~\ref{thm:pressure_spectral_radius_standard}).
The determinant/zeta rational form is controlled by $\det(I-zA)$, whose smallest-modulus singularity lies at $|z|=\rho(A)^{-1}$.
\end{proof}

\subsection{Scattering delay as a \texorpdfstring{$\log\det$}{logdet} derivative (unitary rigidity)}
\label{subsec:rigidity_scattering_delay_logdet}

\begin{lemma}[Wigner--Smith trace as a frequency derivative of \texorpdfstring{$\log\det S$}{logdet S}]\label{lem:wigner_smith_trace_logdet}
\MathTag Let $S(\omega)$ be a differentiable family of \emph{unitary} matrices on a frequency band.
Define the Wigner--Smith matrix
\[
Q(\omega):=-\iu\,S(\omega)^\dagger\frac{\dd S}{\dd \omega}.
\]
Then
\[
\Tr Q(\omega)=-\iu\,\frac{\dd}{\dd\omega}\log\det S(\omega).
\]
\end{lemma}
\begin{proof}
For an invertible differentiable matrix $S(\omega)$ one has the standard identity
\[
\frac{\dd}{\dd\omega}\log\det S(\omega)=\Tr\!\left(S(\omega)^{-1}\frac{\dd S}{\dd\omega}\right).
\]
Under unitarity, $S^{-1}=S^\dagger$, hence
\[
-\iu\,\frac{\dd}{\dd\omega}\log\det S(\omega)
=-\iu\,\Tr\!\left(S^\dagger\frac{\dd S}{\dd\omega}\right)
=\Tr\!\left(-\iu\,S^\dagger\frac{\dd S}{\dd\omega}\right)
=\Tr Q(\omega).
\]
\end{proof}

\begin{remark}[Why this is a rigidity bridge]\label{rem:rigidity_ws_logdet_bridge}
\AuditTag Lemma~\ref{lem:wigner_smith_trace_logdet} makes the phase/delay interface in Appendix~\ref{app:time_mass_delay} algebraically compatible with the determinant/trace bookkeeping used throughout the $\e$-channel and the operator mother-space vocabulary:
delay is a derivative of a \emph{log-determinant}.
This is the minimal reason that ``delay audits'' can be discussed in the same certificate language as pole-barrier and logdet/trace expansions, without importing any additional dynamical axiom beyond the declared unitarity window.
\end{remark}

\subsection{Holonomy statistics: cycle-type rigidity for class-function readouts}
\label{subsec:rigidity_cycle_type_statistics}

\begin{proposition}[Cycle-type statistics rigidly determine class-function loop readouts]\label{prop:rigidity_cycle_type_class_functions}
\MathTag In the anchor holonomy setting (Section~\ref{sec:protocol_connections_holonomy}), loop holonomies are defined up to conjugation (gauge relabeling),
so any readout that depends on holonomy only through a class function is determined by the cycle-type histogram.
\end{proposition}
\begin{proof}
This is precisely Proposition~\ref{prop:cycle_type_stats_determine_class_sums}.
\end{proof}

\begin{remark}[Why this strengthens CP/mixing narratives]\label{rem:rigidity_cycle_type_strengthens_cp}
\AuditTag When a CP- or mixing-facing diagnostic is expressed as a class-function loop statistic (e.g.\ a character/trace readout of a representation of the holonomy group),
its aggregate behavior is not a free numerical target but is rigidly constrained by the reported cycle-type statistics.
This is the mechanism by which holonomy tables can support distributional, not only point-estimate, falsifiability hooks.
\end{remark}

\begin{proposition}[Cycle-type identifiability boundary for class-function observables]\label{prop:cycle_type_identifiability_boundary}
\AuditTag Fix a finite loop family $\mathcal{L}$ in the anchor holonomy setting so that each loop $L\in\mathcal{L}$ yields a holonomy $p_L\in S_4$, and summarize the multiset $\{p_L\}_{L\in\mathcal{L}}$ by its cycle-type histogram.
Then the following are equivalent:
\begin{enumerate}
\item the cycle-type histograms agree for two protocol realizations (same loop family, possibly different internal constructions);
\item for every class function $F:S_4\to\RR$, the aggregate sums $\sum_{L\in\mathcal{L}}F(p_L)$ agree;
\item for every representation $\rho$ of $S_4$, all trace-based loop summaries $\sum_{L\in\mathcal{L}}\mathrm{tr}(\rho(p_L))$ agree.
\end{enumerate}
Consequently, within any diagnostic pipeline that reports only cycle-type histograms (or, equivalently, only class-function summaries), two realizations with the same histogram are indistinguishable by those observables; distinguishing them requires enlarging the loop family or introducing a non-class-function readout.
\end{proposition}
\begin{proof}
$(1)\Rightarrow(2)$ is Proposition~\ref{prop:cycle_type_stats_determine_class_sums}.
$(2)\Rightarrow(3)$ holds because traces of representations are class functions.
For $(2)\Rightarrow(1)$, if the histograms differ then there exists a cycle type $C$ whose count differs; the indicator function $F=\mathbf{1}_{C}$ is a class function and yields different sums, contradicting (2).
\end{proof}
