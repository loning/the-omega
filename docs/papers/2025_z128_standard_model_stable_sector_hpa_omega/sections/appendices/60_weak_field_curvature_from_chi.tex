\section[Weak-field curvature from \texorpdfstring{$\chi$}{chi}]{Weak-field curvature from $\chi$: a theorem-level bridge from discrete reconstructions to continuum curvature}
\label{app:weak_field_curvature_from_chi}

This appendix records an explicit weak-field curvature bridge that upgrades the interface dictionary
$\chi\to N\to g_{00}\to \Phi$ (Appendix~\ref{app:overhead_to_gravity_closure})
to a theorem-level statement about a curvature component in a standard weak-field gauge.
It is designed to compose directly with the protocol-to-continuum error-control bounds in Appendix~\ref{app:protocol_to_continuum_error_control}.

\subsection{Weak-field metric ansatz (declared gauge)}
\label{subsec:weak_field_metric_ansatz}

\begin{assumption}[Static weak-field isotropic gauge for a single potential]
\label{ass:weak_field_isotropic_metric}
\MathTag Assume a static weak-field regime in which the continuum representative metric admits a gauge in which
\begin{equation}
\label{eq:weak_field_isotropic_metric}
\dd s^2
=-(1+2\phi(x))\,c^2\,\dd t^2+(1-2\phi(x))\,\delta_{ij}\,\dd x^i\dd x^j,
\qquad
|\phi(x)|\ll 1,
\end{equation}
with $\phi(x):=\Phi(x)/c^2$ and $\Phi$ a Newtonian potential representative.
\end{assumption}

\begin{remark}[Relation to the lapse dictionary]\label{rem:weak_field_lapse_dictionary_relation}
\InterfaceTag Under the lapse dictionary $g_{00}\approx -N^2$ and the weak-field expansion
$g_{00}\approx -(1+2\Phi/c^2)$ (Appendix~\ref{app:overhead_to_gravity_closure}),
Assumption~\ref{ass:weak_field_isotropic_metric} is a standard completion of the $g_{00}$ statement by a spatial gauge choice.
The curvature bridge below uses only the leading $O(\phi)$ terms and is therefore compatible with the minimal $g_{00}$-first interface stance.
\end{remark}

\subsection{Curvature component in the weak-field gauge}
\label{subsec:weak_field_curvature_component}

\begin{theorem}[Leading-order weak-field curvature: \texorpdfstring{$G_{00}$}{G00} as a Laplacian]
\label{thm:weak_field_G00_laplacian}
\MathTag Under Assumption~\ref{ass:weak_field_isotropic_metric}, the Einstein-tensor component satisfies
\begin{equation}
\label{eq:weak_field_G00_laplacian}
G_{00}(x)=2\,\Delta\phi(x)+O\!\left(\phi\,\Delta\phi+|\nabla\phi|^2\right),
\end{equation}
where $\Delta$ and $\nabla$ are taken with respect to the Euclidean spatial coordinates in \eqref{eq:weak_field_isotropic_metric}.
Equivalently, in terms of $\Phi=c^2\phi$,
\begin{equation}
\label{eq:weak_field_G00_laplacian_Phi}
G_{00}(x)=\frac{2}{c^2}\,\Delta\Phi(x)+O\!\left(\frac{\Phi\,\Delta\Phi}{c^4}+\frac{|\nabla\Phi|^2}{c^4}\right).
\end{equation}
\end{theorem}

\begin{proof}
Write the metric as $g_{\mu\nu}=\eta_{\mu\nu}+h_{\mu\nu}$ with signature $(-,+,+,+)$ and
$h_{00}=-2\phi$, $h_{ij}=-2\phi\,\delta_{ij}$, $h_{0i}=0$.
To first order in $h$ and for a static field, the Christoffel symbols are
\[
\Gamma^{i}_{00}=\partial_i\phi+O(\phi\,\partial\phi),
\qquad
\Gamma^{0}_{0i}=\partial_i\phi+O(\phi\,\partial\phi),
\qquad
\Gamma^{k}_{ij}=-(\delta^k{}_j\,\partial_i\phi+\delta^k{}_i\,\partial_j\phi-\delta_{ij}\,\partial^k\phi)+O(\phi\,\partial\phi),
\]
with all time derivatives vanishing.
Using $R_{\mu\nu}=\partial_\alpha\Gamma^\alpha_{\mu\nu}-\partial_\nu\Gamma^\alpha_{\mu\alpha}+\Gamma^\alpha_{\mu\nu}\Gamma^\beta_{\alpha\beta}-\Gamma^\alpha_{\mu\beta}\Gamma^\beta_{\nu\alpha}$,
one obtains at leading order
\[
R_{00}=\Delta\phi+O(\phi\,\Delta\phi+|\nabla\phi|^2),
\qquad
R=2\,\Delta\phi+O(\phi\,\Delta\phi+|\nabla\phi|^2),
\]
hence $G_{00}=R_{00}-\frac12 g_{00}R=2\,\Delta\phi+O(\phi\,\Delta\phi+|\nabla\phi|^2)$.
\end{proof}

\begin{corollary}[Weak-field curvature sourced by the overhead proxy \texorpdfstring{$\chi$}{chi}]
\label{cor:weak_field_G00_from_chi}
\MathTag Adopt the overhead-to-gravity dictionary of Appendix~\ref{app:overhead_to_gravity_closure}, in particular
\begin{equation}
\label{eq:Phi_from_chi_recalled}
\Phi(x)=-\gamma c^2\bigl(\chi(x)-\chi_0\bigr),
\end{equation}
with $\gamma>0$ and arbitrary reference $\chi_0$.
Under Assumption~\ref{ass:weak_field_isotropic_metric},
\begin{equation}
\label{eq:G00_from_chi}
G_{00}(x)=-2\gamma\,\Delta\chi(x)
\;+\;O\!\left(\gamma^2\,|\nabla\chi|^2+\gamma\,|\chi-\chi_0|\,|\Delta\chi|\right).
\end{equation}
In particular, at leading order one has the curvature--source identification
\begin{equation}
\label{eq:G00_rho_eff_identification}
G_{00}(x)\approx \frac{8\pi G}{c^2}\,\rho_{\mathrm{eff}}(x),
\qquad
\rho_{\mathrm{eff}}(x):=-\frac{\gamma c^2}{4\pi G}\,\Delta\chi(x),
\end{equation}
which matches the Poisson template used in Appendix~\ref{app:overhead_to_gravity_closure}.
\end{corollary}

\begin{proof}
Substitute \eqref{eq:Phi_from_chi_recalled} into \eqref{eq:weak_field_G00_laplacian_Phi}.
Since $\nabla\Phi=-\gamma c^2\nabla\chi$ and $\Delta\Phi=-\gamma c^2\Delta\chi$, the leading term gives $G_{00}=-(2\gamma)\Delta\chi$.
The remainder terms in \eqref{eq:weak_field_G00_laplacian_Phi} become $O(\gamma^2|\nabla\chi|^2+\gamma|\chi-\chi_0||\Delta\chi|)$.
Finally, combine the leading term with the definition of $\rho_{\mathrm{eff}}$ in Appendix~\ref{app:overhead_to_gravity_closure}.
\end{proof}

\subsection{A discrete curvature estimator and an error budget}
\label{subsec:weak_field_discrete_curvature_estimator}

\begin{definition}[Discrete weak-field curvature proxy from reconstructed \texorpdfstring{$\chi$}{chi}]
\label{def:discrete_G00_from_chi}
\textsf{\footnotesize[Prot]} Let $\widehat\chi_h$ be a reconstructed grid field at spacing $h$ (Appendix~\ref{app:chi_reconstruction_protocol}).
Define the discrete curvature proxy
\begin{equation}
\label{eq:discrete_G00_hat}
\widehat G_{00,h}(x)
:= -2\,\widehat\gamma\,(\Delta_h\widehat\chi_h)(x),
\end{equation}
where $\Delta_h$ is the central-difference Laplacian (Appendix~\ref{app:protocol_to_continuum_error_control}) and $\widehat\gamma$ is the declared (or fitted) coupling.
\end{definition}

\begin{corollary}[Error budget for \texorpdfstring{$\widehat G_{00,h}$}{G00-hat}]
\label{cor:discrete_G00_error_budget}
\MathTag Let $\widehat\chi_h=\chi_h+\eta$ with $|\eta|\le \epsilon_\chi$ on the Laplacian stencil and $\widehat\gamma=\gamma+\delta\gamma$.
Assume $\chi\in C^4$ near the grid point.
Then
\begin{equation}
\label{eq:discrete_G00_error_budget}
\bigl|\widehat G_{00,h}(x)-(-2\gamma\,\Delta\chi)(x)\bigr|
\;\le\;
2\Bigl(
|\gamma|\cdot|(\Delta_h\chi_h)(x)-(\Delta\chi)(x)|
\;+\;|\gamma|\cdot|(\Delta_h\eta)(x)|
\;+\;|\delta\gamma|\cdot|(\Delta_h\widehat\chi_h)(x)|
\Bigr),
\end{equation}
and the first two terms admit the explicit controls
$|(\Delta_h\chi_h)-(\Delta\chi)|\lesssim h^2$ (Theorem~\ref{thm:laplacian_truncation})
and $|(\Delta_h\eta)|\le (4d/h^2)\epsilon_\chi$ (Corollary~\ref{cor:laplacian_noise_amplification}).
\end{corollary}

\begin{proof}
This is the specialization of Corollary~\ref{cor:rho_eff_error_budget} to the proxy $\widehat G_{00,h}=-2\widehat\gamma\,\Delta_h\widehat\chi_h$.
\end{proof}
