\section{Generalization II: mass ratios, spectral gaps, and mixing as holonomy}
\label{sec:masses_mixing}

\subsection{Mass as internal winding and extra scan cycles}
In the HPA program, nonassociative defects act as phase-impedance centers. Resolving local algebraic obstructions requires additional scan cycles, increasing local scan density; macroscopically this is read as metric dilation and Shapiro-type delays \cite{Shapiro1964}. To avoid unit dependence, we restrict the primary geometricization target to \emph{dimensionless mass ratios}.

\begin{definition}[Mass-ratio invariants]
\label{def:mass_ratio}
For particles $i,j$ define
\begin{equation}
\frac{m_i}{m_j}=\frac{I_i(\mathcal{G})}{I_j(\mathcal{G})},
\end{equation}
where $I_i(\mathcal{G})$ is a process cost or spectral quantity induced by the defect associated with particle $i$ (e.g., associator tension, holonomy data, or a spectral gap).
\end{definition}

This definition fixes ratios; an overall scale is set by the reference choice. The task reduces to defining a computable family $\{I_i\}$ whose ratios match the PDG mass-ratio data within uncertainties \cite{PDGRPP2024}.

\subsection{Worked example: proton--electron mass ratio}
To make the program quantitative beyond $\alpha$, we record a second low-parameter worked example: the proton--electron mass ratio $\mu=m_p/m_e$.

We strengthen Definition~\ref{def:mass_ratio} by fixing a closed-theory invariant choice.

\begin{axiom}[Mass invariants as phase-volume costs]\label{ax:mass_volume}
For each particle class $i$, the mass invariant $I_i$ is given by a canonical phase-volume of an internal compact manifold $\mathcal{M}_i$ (or a finite disjoint union of such manifolds), normalized so that the electron reference satisfies $I_e=1$.

For a baryon in three-color QCD ($N_c=3$), the invariant is the sum over three color sectors \cite{PeskinSchroeder1995,PDGRPP2024}.
\end{axiom}

\begin{axiom}[Proton internal phase space]\label{ax:proton_phase_space}
The proton is assigned three identical color sectors ($N_c=3$), each modeled by the compact phase space \cite{PeskinSchroeder1995,PDGRPP2024}
\begin{equation}
\mathcal{M}_{q}\cong SO(3)\times SO(3)\times U(1),
\end{equation}
so that $\mathcal{M}_p$ is the disjoint union of three copies of $\mathcal{M}_q$.

This choice is equivalent, at the level of canonical volumes, to the alternative factorization $SO(3)\times SU(2)\times \mathbb{R}P^1$, since $\mathrm{Vol}(SO(3))\,\mathrm{Vol}(U(1))=\mathrm{Vol}(SU(2))\,\mathrm{Vol}(\mathbb{R}P^1)$ by the $\mathbb{Z}_2$ quotient relations in Appendix~\ref{app:group_volumes}. We select $SO(3)\times SO(3)\times U(1)$ as the projective-rotation convention: rotational sectors are projectivized (spinorial double cover removed) while the phase circle is kept unquotiented.
\end{axiom}

\begin{theorem}[Proton--electron mass ratio]\label{thm:mu_geo}
Under Axioms~\ref{ax:mass_volume}--\ref{ax:proton_phase_space}, the geometric prediction for the proton--electron mass ratio is
\begin{equation}
\mu_{\mathrm{geo}}:=\frac{m_p}{m_e}=6\pi^5\approx 1836.1181087117.
\end{equation}
\end{theorem}
\begin{proof}
By Axiom~\ref{ax:mass_volume}, $m_p/m_e=I_p/I_e=I_p$ since $I_e=1$. By Axiom~\ref{ax:proton_phase_space},
\begin{equation}
I_p=3\,\mathrm{Vol}(SO(3)\times SO(3)\times U(1)).
\end{equation}
Using $\mathrm{Vol}(SO(3))=\pi^2$ and $\mathrm{Vol}(U(1))=2\pi$ (Appendix~\ref{app:group_volumes}) gives
\begin{equation}
I_p=3\,(\pi^2)(\pi^2)(2\pi)=6\pi^5.
\end{equation}
\end{proof}

\begin{remark}[Factorization rigidity at fixed primitive volumes]
If one restricts the per-color sector to be a product of three primitive phase spaces drawn from $\{U(1),SU(2),SO(3),\mathbb{R}P^1\}$ with canonical volumes (Appendix~\ref{app:group_volumes}), then the only factorizations that realize $I_q=2\pi^5$ are
\begin{equation}
SO(3)\times SO(3)\times U(1)
\qquad\text{and}\qquad
SO(3)\times SU(2)\times \mathbb{R}P^1,
\end{equation}
which are related by the double-cover exchanges $SU(2)\leftrightarrow SO(3)$ and $U(1)\leftrightarrow \mathbb{R}P^1$. Thus the $\pi^5$ scaling and the coefficient $6$ are fixed by low-complexity phase-volume factorization; the remaining choice amounts to a projective convention, as recorded in Axiom~\ref{ax:proton_phase_space}.
\end{remark}

Numerically, CODATA 2022 gives $\mu_{\mathrm{exp}}\approx 1836.15267343$ \cite{CODATA2022RMP}, so
\begin{equation}
\Delta\mu:=\mu_{\mathrm{geo}}-\mu_{\mathrm{exp}}\approx -3.456\times 10^{-2},
\qquad
\frac{\Delta\mu}{\mu_{\mathrm{exp}}}\approx -1.88\times 10^{-5}.
\end{equation}
The offset $\Delta\mu$ is the matching correction between the canonical phase-volume value and the CODATA ratio; it collects QCD binding and radiative effects.

A standard external benchmark is that most of the proton mass is dynamical, arising from QCD scale generation and strong-interaction energy rather than from the light-quark current masses \cite{Wilczek1999MassWithoutMass,Collins2011,PDGRPP2024}. Accordingly, $\Delta\mu$ is identified as the QCD matching correction associated with strong binding and radiative structure.

As a rigidity check analogous to Section~\ref{sec:alpha_anchor}, we approximate $\mu_{\mathrm{exp}}$ by the five-term ansatz
\begin{equation}
\mu(a,b,c,d,e)=a\pi^5+b\pi^4+c\pi^3+d\pi^2+e\pi
\end{equation}
with small nonnegative integers.

\begin{proposition}[Uniqueness at low coefficient complexity for $\mu$]\label{prop:mu_integer_rigidity}
Within the domain $0\le a,b,c,d,e\le 12$ and $a+b+c+d+e\le 10$, the unique minimizer of $|\mu(a,b,c,d,e)-\mu_{\mathrm{exp}}|$ is $(a,b,c,d,e)=(6,0,0,0,0)$, i.e., $\mu_{\mathrm{geo}}=6\pi^5$, with relative error $1.88\times 10^{-5}$. The next-best combination in the same domain has relative error at least $4.44\times 10^{-4}$.
\end{proposition}
\begin{proof}
This is a finite check by exhaustive enumeration over all $(a,b,c,d,e)$ in the stated domain.
\end{proof}

\subsection{Mixing matrices as holonomy: from geometric phase to CKM/PMNS}
The $\Omega$ framework uses standard geometric-phase language (Berry connection and holonomy) on parameter manifolds \cite{Simon1983,Berry1984,WilczekZee1984,Nakahara2003}. We therefore adopt the following geometric implementation axiom.

\begin{axiom}[Mixing matrices as holonomy]
\label{ax:mixing_holonomy}
There exists a parameter manifold $\mathcal{M}$ associated with the readout protocol / gauge choice / internal fiber structure, together with a $U(1)$ or $U(n)$ connection $\mathcal{A}$ on $\mathcal{M}$, such that generation mixing can be represented as holonomy along basic loops:
\begin{equation}
U_{\mathrm{mix}}=\mathcal{P}\exp\!\left(\oint_{\gamma\subset\mathcal{M}} \mathcal{A}\right).
\end{equation}
Observable invariants (mixing angles, CP phases) correspond to conjugacy-class data of the holonomy.
\end{axiom}

This template upgrades ``angles/phases'' from fit parameters to geometric objects (connection, curvature, and topology). It matches the unitarity structure used in the standard CKM/\allowbreak PMNS parameterizations \cite{Cabibbo1963,KobayashiMaskawa1973,Pontecorvo1957,MakiNakagawaSakata1962}. The PDG provides systematic reviews and conventions \cite{PDGRPP2024}. A short geometric-phase reminder is included in Appendix~\ref{app:holonomy}.

\subsection{A quantitative CP-odd benchmark: the Jarlskog invariant}
For quark mixing, a basis-independent measure of CP violation is the Jarlskog invariant $J$, defined from the CKM matrix $V$ by \cite{Jarlskog1985}
\begin{equation}
J:=\mathrm{Im}\!\left(V_{ud}V_{cs}V_{us}^\ast V_{cd}^\ast\right),
\end{equation}
equivalently by any of its rephasing-invariant forms. The PDG quotes \cite{PDGRPP2024}
\begin{equation}
J_{\mathrm{PDG}}=(3.00\pm 0.15)\times 10^{-5}.
\end{equation}
In the Wolfenstein parametrization \cite{Wolfenstein1983}, one has at leading order
\begin{equation}\label{eq:jarlskog_wolfenstein}
J=A^2\lambda^6\bar{\eta}+O(\lambda^8),
\end{equation}
with the standard $(\lambda,A,\bar{\rho},\bar{\eta})$ conventions summarized by the PDG \cite{PDGRPP2024}.

To complement the holonomy axiom with a minimal quantitative rigidity check, we test the one-parameter $\pi$-power ansatz
\begin{equation}\label{eq:jarlskog_pi_ansatz}
J_{\mathrm{geo}}(a,n)=\frac{1}{a\pi^n},\qquad a\in\NNpos,\ n\in\NNpos,
\end{equation}
which is natural for volume-normalized geometric costs and their inverses.

\begin{proposition}[Low-complexity $\pi$-power rigidity for CKM CP violation]\label{prop:jarlskog_rigidity}
Let $J_{\mathrm{PDG}}=3.00\times 10^{-5}$ be the PDG central value. Among all pairs $(a,n)$ with $1\le a\le 50$ and $1\le n\le 20$, the unique minimizer of $|J_{\mathrm{geo}}(a,n)-J_{\mathrm{PDG}}|$ is $(a,n)=(11,7)$, yielding
\begin{equation}
J_{\mathrm{geo}}=\frac{1}{11\pi^7}\approx 3.009942547\times 10^{-5},
\end{equation}
with relative deviation $3.31\times 10^{-3}$. The next-best pair in the same domain has relative deviation at least $9.37\times 10^{-3}$.
\end{proposition}
\begin{proof}
This is a finite check by enumeration over $(a,n)$ in the stated domain.
\end{proof}

\begin{proposition}[Minimax rigidity over the PDG interval]\label{prop:jarlskog_minimax_rigidity}
Let the PDG benchmark interval be $J\in[J_-,J_+]$ with $J_-:=2.85\times 10^{-5}$ and $J_+:=3.15\times 10^{-5}$ \cite{PDGRPP2024}. Among all pairs $(a,n)$ with $1\le a\le 50$ and $1\le n\le 20$, the unique minimizer of the worst-case relative deviation
\begin{equation}
\max\!\left\{\left|\frac{J_{\mathrm{geo}}(a,n)-J_-}{J_-}\right|,\ \left|\frac{J_{\mathrm{geo}}(a,n)-J_+}{J_+}\right|\right\}
\end{equation}
is $(a,n)=(11,7)$, with worst-case value $5.61\times 10^{-2}$. The next-best pair in the same domain has worst-case value at least $5.65\times 10^{-2}$.
\end{proposition}
\begin{proof}
This is a finite check by enumeration over $(a,n)$ in the stated domain, evaluating the worst-case error at the endpoints $J_-$ and $J_+$.
\end{proof}

\paragraph{Geometric interpretation.}
The exponent $n=7$ matches a minimal CP-odd readout sector built from three projective rotation factors (one per generation) and one projective line:
\begin{equation}
\mathcal{M}_{\mathrm{CP}}\cong SO(3)\times SO(3)\times SO(3)\times \mathbb{R}P^1,
\qquad
\mathrm{Vol}(\mathcal{M}_{\mathrm{CP}})=(\pi^2)^3(\pi)=\pi^7
\end{equation}
under the canonical normalization (Appendix~\ref{app:group_volumes}). Thus $J_{\mathrm{geo}}=1/(11\pi^7)$ is read as the inverse phase volume of $\mathcal{M}_{\mathrm{CP}}$ times an $11$-fold discrete channel multiplicity. In particular,
\begin{equation}
11 = \dim(\mathfrak{su}(3))+\dim(\mathfrak{su}(2))=8+3,
\end{equation}
so the coefficient is the combined gauge-sector dimension, and it is fixed by the low-complexity rigidity principle rather than by continuous fitting, as in Remark~\ref{rem:holonomy_rigidity}.
