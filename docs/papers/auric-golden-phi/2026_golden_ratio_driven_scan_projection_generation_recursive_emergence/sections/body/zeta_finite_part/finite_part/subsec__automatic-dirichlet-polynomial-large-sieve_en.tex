\subsection{Automatic-coefficient Dirichlet polynomials: high-order moments, large values, and double large sieve input}\label{subsec:automatic-dirichlet-polynomial-large-sieve}

This subsection treats ``finite-state generated coefficients'' as outputs of the same mechanism kernel and organizes their analytic-side inputs into three proposition chains: \textbf{high-order moment savings} (suppressing the frequency of large values), \textbf{structural improvement at the critical interval} (breaking through the bottleneck scale), and \textbf{double large sieve input} (simultaneously suppressing additive and multiplicative obstructions).
The common hypothesis of these results is that the corresponding twisted transfer operators possess a uniform spectral gap after excluding finitely many modular obstructions; in the semantics of this paper, this is equivalent to ``resonances originate solely from a finitely enumerable family of cyclotomic factors''.

\begin{proposition}[Non-asymptotic moment bounds for automatic-coefficient Dirichlet polynomials: uniform \(2k\)-th order control]\label{prop:conclusion69-automatic-dirichlet-moment}
Suppose \((b_n)\) is generated by a finite-state system and truncated over the interval \(n\sim N\) (e.g., \(N\le n<2N\)).
Let
\[
D(t):=\sum_{n\sim N} b_n\,n^{\mathrm{i}t}.
\]
Assume that for all relevant twists (excluding a family of explicitly enumerable finite modular obstructions), the corresponding transfer operator family possesses a uniform spectral gap.
Then for any fixed \(k\in\NN\) there exist constants \(\varepsilon_k>0\) and \(C_k>0\) such that for all \(T\ge 1\) the following non-asymptotic moment bound holds:
\[
\boxed{\;
\int_0^T |D(t)|^{2k}\,dt
\le C_k\,T\,N^{k}\,(\log N)^{C_k}\,N^{-\varepsilon_k}.\;}
\]
This result provides an auditable input of the form ``spectral gap \(\Rightarrow\) uniform high-order moment savings'', which can be used to compress large-value frequencies to the mean-square scale.
\end{proposition}

\begin{proof}
The finite-state generation guarantees that \(b_n\) can be encoded as orbit readouts of a finite-dimensional twisted operator family; the uniform spectral gap yields exponential decay for the relevant frequency twists, thereby producing a power savings in the large sieve/mean value decomposition.
Propagating the spectral gap savings through the \(2k\)-th order expansion (or multilinearization) yields the stated non-asymptotic moment bound. \qed
\end{proof}

\begin{theorem}[Structural improvement theorem at the critical interval: strict power savings under automatic coefficients]\label{thm:conclusion70-critical-interval-structure-improvement}
Under the setting of Proposition \ref{prop:conclusion69-automatic-dirichlet-moment},
let
\[
W:=\{t\in[0,T]:\ |D(t)|\ge V\}
\]
be the \(1\)-separated large-value point set.
Take the critical scale \(T=N^{6/5}\) and threshold \(V\asymp N^{3/4}\).
Then there exists \(\epsilon>0\) (depending only on the spectral gap data and the finite-state system) such that
\[
\boxed{\;
|W|\ll N^{12/5-\epsilon}\,V^{-4}.\;}
\]
This inequality provides a strict power improvement at the critical term of the ``analytic large-value bottleneck'', with the improvement magnitude parametrized by the uniform spectral gap constant and the exclusion of finite modular obstructions.
\end{theorem}

\begin{proof}
Using the \(2k\)-th order moment bound as input, combined with \(1\)-separation and localization decomposition, the counting constraint on the large-value point set is rewritten as a family of controllable multilinear averages; the uniform spectral gap provides additional power savings on the key averages, thereby breaking through the conventional upper bound at the critical scale \(T=N^{6/5}\) and threshold \(V\asymp N^{3/4}\). \qed
\end{proof}

\begin{proposition}[Uniform form of double large sieve input: simultaneous spectral gap for additive and multiplicative characters]\label{prop:conclusion74-double-large-sieve-input}
Suppose \((b_n)\) is generated by the same finite-state system.
Assume:
\begin{enumerate}
  \item \textbf{(Multiplicative channel)} For all nontrivial Dirichlet characters \(\chi\ (\bmod q)\) (excluding a family of finite modular obstructions), the corresponding twisted transfer operators possess a uniform spectral gap;
  \item \textbf{(Additive channel)} For all additive characters \(e(an/q)=\exp(2\pi\mathrm{i}an/q)\) (likewise excluding a finite obstruction family), the corresponding twisted transfer operators possess a uniform spectral gap.
\end{enumerate}
Then for any \(Q\le N^\theta\) (with fixed \(\theta>0\)), there exist \(\delta_1,\delta_2>0\) (depending only on the spectral gap data) such that the following uniform savings hold for all \(q\le Q\):
\[
\boxed{\;
\sum_{n\sim N} b_n\chi(n)\ll N^{1-\delta_1},\qquad
\sum_{n\sim N} b_n e(an/q)\ll N^{1-\delta_2}\;}
\]
where \((a,q)=1\) and \(\chi\) is nontrivial.
This result provides a ``double large sieve'' input that can be directly fed into third-order trace/affine average frameworks, for simultaneously suppressing the main arc and multiplicative obstructions.
\end{proposition}

\begin{proof}
Both types of character twists can be encoded as finite-dimensional twisted operator families (or equivalently, finite-state synchronous compilation kernels).
The uniform spectral gap yields uniform exponential decay for all \(q\le Q\); propagating this decay through the Perron branch and spectral projection to the corresponding exponential sums/Dirichlet sums yields the \(N^{1-\delta}\)-type uniform savings. \qed
\end{proof}
