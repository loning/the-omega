\paragraph{Local arithmetic fingerprint of the Newman critical octic certificate}
The negative real channel at the critical point in Remark~\ref{rem:pom-collision-rh-break-a4-threshold} induces a sparse octic equation.
To fix notation, let
\[
\mathcal Z_4(z):=z^8-2z^6-2z^5-2z^4-2z^3-2\in\ZZ[z],
\]
let \(\theta\) be a root of \(\mathcal Z_4\), and set
\(
K:=\QQ(\theta)
\).
In the notation of Remark~\ref{rem:pom-collision-rh-break-a4-threshold}, the critical coordinate \(z_\star:=-\mu_\star>0\) satisfies \(\mathcal Z_4(z_\star)=0\), so \(K\) may be taken as \(\QQ(z_\star)\).
The following result compresses the corresponding number field certificate into auditable discriminant and local decomposition data, thereby providing the shortest interface for directly linking mod-\(p\) fingerprints with ramification structure.

\begin{proposition}[Power integral basis, discriminant, and signature]\label{prop:pom-a4t-newman-octic-field-basic}
\(\mathcal Z_4\) is an Eisenstein polynomial over the prime \(2\), hence irreducible in \(\QQ[z]\).
Moreover, its number field \(K=\QQ(\theta)\) satisfies
\[
\mathcal O_K=\ZZ[\theta],
\qquad
\mathrm{Disc}(K)=\mathrm{Disc}(\mathcal Z_4)=-2^{10}\cdot 7^2\cdot 23^2\cdot 1151.
\]
Thus \(\{1,\theta,\theta^2,\dots,\theta^7\}\) is a power integral basis for \(\mathcal O_K\).
Furthermore, the signature of \(K\) is \((r_1,r_2)=(2,3)\), so by Dirichlet's unit theorem the unit rank is \(r_1+r_2-1=4\).
\end{proposition}
\begin{proof}
Eisenstein's criterion yields irreducibility.
The remaining conclusions are obtained by symbolic verification via the round-two algorithm for maximal orders applied to \(\mathcal Z_4\): this algorithm returns the field discriminant and confirms that \(\ZZ[\theta]\) is already the maximal order (i.e., index 1), hence \(\mathrm{Disc}(\mathcal Z_4)=\mathrm{Disc}(K)\) and the power basis holds.
The signature is given by the number of real roots of \(\mathcal Z_4\) (two real roots and three pairs of conjugate non-real roots); the unit rank formula is a direct consequence of Dirichlet's theorem.
Reproducible script: \texttt{scripts/exp\_collision\_kernel\_A4\_newman\_octic\_field\_arithmetic.py}.
\end{proof}

\begin{remark}[Consistency of the quadratic subfield in the discriminant]\label{rem:pom-a4t-newman-octic-field-quadratic-subfield}
By Proposition~\ref{prop:pom-a4t-newman-octic-field-basic},
the square-free part of \(\mathrm{Disc}(\mathcal Z_4)\) is \(-1151\).
Hence the splitting field of \(\mathcal Z_4\) contains the canonical quadratic subfield \(\QQ(\sqrt{-1151})\) (corresponding to the kernel of the signature representation), which agrees with the quadratic subfield cut out by the discriminant of \(P_4(t)\) in Proposition~\ref{prop:pom-a4t-newman-k2-class-number}.
\end{remark}

\begin{proposition}[Ramified primes and prime ideal factorization]\label{prop:pom-a4t-newman-octic-field-prime-decomp}
Following the setting of Proposition~\ref{prop:pom-a4t-newman-octic-field-basic},
the ramified primes in \(K/\QQ\) are exactly
\(
\{2,7,23,1151\}
\).
Moreover, in \(\mathcal O_K\) (denoting by \((p,f(\theta))\) the prime ideal generated by \(p\) and \(f(\theta)\)), we have the complete factorizations:
\begin{align*}
(2)\ &=\ \mathfrak p_2^{\,8},\qquad &&\mathfrak p_2:=(2,\theta), &&e(\mathfrak p_2\mid 2)=8,\ f(\mathfrak p_2\mid 2)=1;\\
(7)\ &=\ \mathfrak p_7^{\,3}\,\mathfrak q_7,\qquad
&&\mathfrak p_7:=(7,\theta-2), &&e(\mathfrak p_7\mid 7)=3,\ f(\mathfrak p_7\mid 7)=1,\\
&&&&\mathfrak q_7:=(7,\theta^5-\theta^4+\theta^3+3\theta^2+3\theta+2), &&e(\mathfrak q_7\mid 7)=1,\ f(\mathfrak q_7\mid 7)=5;\\
(23)\ &=\ \mathfrak p_{23}^{\,3}\,\mathfrak a_{23}\,\mathfrak b_{23}\,\mathfrak c_{23},\qquad
&&\mathfrak p_{23}:=(23,\theta-6), &&e(\mathfrak p_{23}\mid 23)=3,\ f(\mathfrak p_{23}\mid 23)=1,\\
&&&&\mathfrak a_{23}:=(23,\theta-7), &&e(\mathfrak a_{23}\mid 23)=1,\ f(\mathfrak a_{23}\mid 23)=1,\\
&&&&\mathfrak b_{23}:=(23,\theta^2+10\theta+8), &&e(\mathfrak b_{23}\mid 23)=1,\ f(\mathfrak b_{23}\mid 23)=2,\\
&&&&\mathfrak c_{23}:=(23,\theta^2-8\theta+1), &&e(\mathfrak c_{23}\mid 23)=1,\ f(\mathfrak c_{23}\mid 23)=2;\\
(1151)\ &=\ \mathfrak p_{1151}^{\,2}\,\mathfrak b_{1151}\,\mathfrak c_{1151}\,\mathfrak d_{1151},\qquad
&&\mathfrak p_{1151}:=(1151,\theta+499), &&e(\mathfrak p_{1151}\mid 1151)=2,\ f(\mathfrak p_{1151}\mid 1151)=1,\\
&&&&\mathfrak b_{1151}:=(1151,\theta^2+200\theta-233), &&e(\mathfrak b_{1151}\mid 1151)=1,\ f(\mathfrak b_{1151}\mid 1151)=2,\\
&&&&\mathfrak c_{1151}:=(1151,\theta^2+330\theta+217), &&e(\mathfrak c_{1151}\mid 1151)=1,\ f(\mathfrak c_{1151}\mid 1151)=2,\\
&&&&\mathfrak d_{1151}:=(1151,\theta^2-377\theta+312), &&e(\mathfrak d_{1151}\mid 1151)=1,\ f(\mathfrak d_{1151}\mid 1151)=2.
\end{align*}
\end{proposition}
\begin{proof}[Sketch]
By the single-generator property \(\mathcal O_K=\ZZ[\theta]\) of Proposition~\ref{prop:pom-a4t-newman-octic-field-basic}, for each prime \(p\), Dedekind's factorization theorem translates the factorization pattern of \(\mathcal Z_4\bmod p\) into the prime ideal factorization of \((p)\); this is directly given by implementations of Cohen's algorithm.
Reproducible script as above.
\end{proof}

\begin{corollary}[Purely tame ramification at odd ramified primes and discriminant exponent]\label{cor:pom-a4t-newman-octic-field-tame}
For each odd ramified prime \(p\in\{7,23,1151\}\), the extension \(K/\QQ\) is tamely ramified at \(p\), and the field discriminant exponent satisfies the minimal tame ramification formula
\[
v_p\!\bigl(\mathrm{Disc}(K)\bigr)=\sum_{\mathfrak p\mid p} f(\mathfrak p\mid p)\,\bigl(e(\mathfrak p\mid p)-1\bigr).
\]
In particular,
\[
v_7\!\bigl(\mathrm{Disc}(K)\bigr)=2,\qquad v_{23}\!\bigl(\mathrm{Disc}(K)\bigr)=2,\qquad v_{1151}\!\bigl(\mathrm{Disc}(K)\bigr)=1,
\]
with corresponding inertia group orders \(3,3,2\), respectively.
\end{corollary}
\begin{proof}
By Proposition~\ref{prop:pom-a4t-newman-octic-field-prime-decomp}, the ramification indices at \(p=7,23,1151\) are \(3,3,2\), respectively, all coprime to \(p\), hence tame.
Under tame ramification the different exponent is \(e-1\), so the discriminant exponent satisfies the stated sum formula; substituting the \((e,f)\) of each prime ideal yields the numerical values.
\end{proof}

\begin{proposition}[Two-prime certificate and \texorpdfstring{$S_8$}{S8}]\label{prop:pom-a4t-newman-octic-field-galois-s8}
Let \(L\) be the splitting field of \(\mathcal Z_4\), and let \(G:=\mathrm{Gal}(L/\QQ)\le S_8\) act on the eight roots of \(\mathcal Z_4\).
Then \(G\cong S_8\).
\end{proposition}
\begin{proof}
By Eisenstein's criterion, \(\mathcal Z_4\) is irreducible over \(\QQ[z]\), hence \(G\) acts transitively on the eight roots.
By Proposition~\ref{prop:pom-a4t-newman-octic-field-basic}, the discriminant \(\mathrm{Disc}(\mathcal Z_4)\) ramifies only at \(\{2,7,23,1151\}\), so at unramified primes one can read the cycle type from the factorization pattern via Frobenius elements.
\par
At \(p=13\), \(\mathcal Z_4\bmod 13\) is irreducible (single factor of degree eight), hence \(G\) contains an \(8\)-cycle.
At \(p=37\), \(\mathcal Z_4\bmod 37\) factors into one irreducible cubic and one irreducible quintic, hence \(G\) contains an element of cycle type \((3)(5)\), in particular a \(5\)-cycle.
\par
To prove primitivity, suppose for contradiction that \(G\) is imprimitive; then there exists a nontrivial block system with block size \(b\) a proper divisor of \(8\), so \(b\in\{2,4\}\) and the number of blocks is \(8/b\in\{4,2\}\).
But when \(G\) contains a \(5\)-cycle, that cycle can act on the block system only trivially or as a cycle of length \(5\); since \(5>b\), it cannot fix a single block, hence must induce a \(5\)-cycle on the blocks, forcing \(5\mid (8/b)\), contradicting \(8/b\in\{2,4\}\).
Thus \(G\) is primitive.
Since \(5\le 8-3\), Jordan's theorem implies \(A_8\subseteq G\).
On the other hand, an \(8\)-cycle is an odd permutation, so \(G\not\subseteq A_8\).
Together, \(G=S_8\).
\end{proof}

\begin{corollary}[Residue algebra factorization and nilpotent thickness]\label{cor:pom-a4t-newman-octic-field-residue-algebra}
Following the factorization in Proposition~\ref{prop:pom-a4t-newman-octic-field-prime-decomp}, by the Chinese remainder theorem and the structure of local discrete valuation rings, the residue algebras at the four ramified primes admit explicit direct product decompositions:
\begin{align*}
\mathcal O_K/2\mathcal O_K &\cong \FF_2[\varepsilon]/(\varepsilon^8),\\
\mathcal O_K/7\mathcal O_K &\cong \FF_7[\varepsilon]/(\varepsilon^3)\times \FF_{7^5},\\
\mathcal O_K/23\mathcal O_K &\cong \FF_{23}\times \FF_{23}[\varepsilon]/(\varepsilon^3)\times \FF_{23^2}\times \FF_{23^2},\\
\mathcal O_K/1151\mathcal O_K &\cong \FF_{1151}[\varepsilon]/(\varepsilon^2)\times (\FF_{1151^2})^{3}.
\end{align*}
Hence the Newman critical octic field has nilpotent thickness spectrum \((8;3;3;2)\) at ramification points, with corresponding residue degree spectrum \((1;\,5;\,1,2,2;\,2,2,2)\).
\end{corollary}

\begin{remark}[Normalized local parameters and valuation scaling]\label{rem:pom-a4t-newman-octic-field-local-parameters}
For the four ramified prime ideals in Proposition~\ref{prop:pom-a4t-newman-octic-field-prime-decomp}, let \(v_{\mathfrak p}\) denote the extension of the \(p\)-adic valuation on \(\QQ_p\) normalized so that \(v_{\mathfrak p}(p)=1\).
Then \(\theta\) is a local parameter for \(\mathfrak p_2\) with
\(
v_{\mathfrak p_2}(\theta)=1/8
\);
similarly,
\(
\theta-2,\theta-6,\theta+499
\)
are local parameters for \(\mathfrak p_7,\mathfrak p_{23},\mathfrak p_{1151}\), respectively, satisfying
\[
v_{\mathfrak p_7}(\theta-2)=1/3,\qquad
v_{\mathfrak p_{23}}(\theta-6)=1/3,\qquad
v_{\mathfrak p_{1151}}(\theta+499)=1/2.
\]
Hence at the four ramification points there exists a set of normalized ``coordinate–valuation'' locks: any local perturbation of \(\theta\) is scaled under the \(p\)-adic metric by a magnification rate directly controlled by the ramification indices \((8,3,3,2)\), which can be rewritten as a set of valuation constants.
\end{remark}
