\section{Problem 2: A Nonvanishing Test Vector for the Local Rankin--Selberg Integral}

\begin{proposition}
Let $F$ be a non-archimedean local field, let $\psi$ be a nontrivial additive character of conductor $\mathfrak o$, and let $\Pi$ (resp. $\pi$) be a generic irreducible admissible representation of $\mathrm{GL}_{n+1}(F)$ (resp. $\mathrm{GL}_n(F)$), realized in $\mathcal W(\Pi,\psi^{-1})$ (resp. $\mathcal W(\pi,\psi)$). Let $\mathfrak q$ be the conductor ideal of $\pi$, choose $Q\in F^\times$ generating $\mathfrak q^{-1}$, and set
\[
u_Q=I_{n+1}+Q E_{n,n+1}.
\]
Then there exist $W\in\mathcal W(\Pi,\psi^{-1})$ and $V\in\mathcal W(\pi,\psi)$ such that
\[
I(s;W,V):=\int_{N_n\backslash \mathrm{GL}_n(F)}
W(\operatorname{diag}(g,1)u_Q)\,V(g)\,|\det g|^{s-\frac12}\,d\bar g
\]
is finite and nonzero for every $s\in\mathbb C$.
\end{proposition}

\begin{lemma}[Mirabolic restriction, strong form]\label{lem:q2-mirabolic-strong}
Let $G_{n+1}=\mathrm{GL}_{n+1}(F)$, $P_{n+1}$ be its mirabolic subgroup, and let $\Pi$ be generic irreducible admissible. The restriction map
\[
\operatorname{Res}_{P_{n+1}}:\mathcal W(\Pi,\psi^{-1})\to C^\infty(P_{n+1})
\]
has image containing
\[
C_c^\infty\bigl(N_{n+1}\backslash P_{n+1},\psi^{-1}\bigr)
:=\{f\in C_c^\infty(P_{n+1})\,:\,f(np)=\psi^{-1}(n)f(p)\}.
\]
Equivalently, for every $f\in C_c^\infty(N_{n+1}\backslash P_{n+1},\psi^{-1})$, there exists $W\in\mathcal W(\Pi,\psi^{-1})$ such that $W|_{P_{n+1}}=f$.
\end{lemma}

\begin{proof}
This is the standard Bernstein--Zelevinsky/Kirillov statement for generic representations of $\mathrm{GL}_{n+1}(F)$: the lowest nonzero derivative piece of $\Pi|_{P_{n+1}}$ is the compactly induced $\psi^{-1}$-equivariant module on $N_{n+1}\backslash P_{n+1}$, embedded in the Whittaker model via restriction.
\end{proof}

\begin{remark}
Lemma~\ref{lem:q2-mirabolic-strong} is the precise external input from Bernstein--Zelevinsky derivative theory for $p$-adic $\mathrm{GL}_m$, together with the Kirillov realization of generic representations on the mirabolic subgroup.
\end{remark}

\begin{lemma}[Integrand descends to $N_n\backslash G_n$]\label{lem:q2-quotient-welldefined}
Fix $W\in\mathcal W(\Pi,\psi^{-1})$, $V\in\mathcal W(\pi,\psi)$, and $s\in\mathbb C$. Define
\[
\Phi_s(g):=W(\operatorname{diag}(g,1)u_Q)\,V(g)\,|\det g|^{s-\frac12},\qquad g\in G_n.
\]
Then $\Phi_s(n_0g)=\Phi_s(g)$ for all $n_0\in N_n$.
\end{lemma}

\begin{proof}
For $n_0\in N_n$, Whittaker equivariance gives
\[
W(\operatorname{diag}(n_0g,1)u_Q)=\psi^{-1}(n_0)W(\operatorname{diag}(g,1)u_Q),
\]
\[
V(n_0g)=\psi(n_0)V(g),
\]
and $|\det(n_0g)|=|\det g|$ since $\det n_0=1$. Multiplying yields $\Phi_s(n_0g)=\Phi_s(g)$.
\end{proof}

\begin{proof}[Proof of the proposition]
Write $G_r=\mathrm{GL}_r(F)$ and
\[
P_{n+1}=\left\{\begin{pmatrix} g & x\\ 0 & 1\end{pmatrix}:g\in G_n,\ x\in F^n\right\}.
\]
Choose Haar measures $dg$ on $G_n$ and $dn$ on $N_n$, and let $d\bar g$ be the induced quotient measure on $N_n\backslash G_n$.

Since $\pi$ is generic, choose $V\in\mathcal W(\pi,\psi)$ with $V(I_n)\neq 0$. By smoothness of $\pi$, there exists compact open $K_0\subset G_n$ fixing the underlying vector; set $K:=K_0\cap G_n(\mathfrak o)$, so
\[
V(k)=V(I_n)\neq 0\qquad (k\in K).
\]

Set
\[
H:=\{\operatorname{diag}(k,1):k\in K\}\subset P_{n+1}.
\]
Because $K\subset G_n(\mathfrak o)$ and $\psi$ has conductor $\mathfrak o$, the induced generic character is trivial on $N_n\cap K$, hence on $N_{n+1}\cap H$.

Define $f:P_{n+1}\to\mathbb C$ by
\[
f(p)=\begin{cases}
\psi^{-1}(n),&p\in nHu_Q\text{ for some }n\in N_{n+1},\\
0,&\text{otherwise.}
\end{cases}
\]
This is well-defined: if $n_1h_1u_Q=n_2h_2u_Q$, then $n_2^{-1}n_1=h_2h_1^{-1}\in N_{n+1}\cap H$, so $\psi^{-1}(n_1)=\psi^{-1}(n_2)$. Thus $f\in C_c^\infty(N_{n+1}\backslash P_{n+1},\psi^{-1})$.

By Lemma~\ref{lem:q2-mirabolic-strong}, choose $W\in\mathcal W(\Pi,\psi^{-1})$ with $W|_{P_{n+1}}=f$.
Then
\[
W(\operatorname{diag}(k,1)u_Q)=1\quad (k\in K).
\]
Also, if $W(\operatorname{diag}(g,1)u_Q)\neq 0$, then $\operatorname{diag}(g,1)u_Q\in N_{n+1}Hu_Q$, so for some $n\in N_{n+1},k\in K$,
\[
\operatorname{diag}(g,1)u_Q=n\operatorname{diag}(k,1)u_Q,
\]
thus
\[
\operatorname{diag}(g,1)=n\operatorname{diag}(k,1).
\]
Writing $n=\begin{psmallmatrix} n_0 & v\\ 0 & 1\end{psmallmatrix}$ with $n_0\in N_n$, comparison of the top-right block gives $v=0$, hence $g=n_0k\in N_nK$.

Therefore the descended integrand on $N_n\backslash G_n$ (well-defined by Lemma~\ref{lem:q2-quotient-welldefined}) is supported in
\[
(N_n\cap K)\backslash K\subset N_n\backslash G_n.
\]
Since $K$ is compact open and $N_n\cap K$ is compact, this quotient is compact, so its $d\bar g$-measure is finite; being nonempty open in the support of Haar quotient measure, its measure is positive.

For $g=n_0k$ with $n_0\in N_n$, $k\in K$, we have
\[
W(\operatorname{diag}(g,1)u_Q)=\psi^{-1}(n_0),
\qquad
V(g)=\psi(n_0)V(I_n),
\]
so
\[
W(\operatorname{diag}(g,1)u_Q)V(g)=V(I_n).
\]
Also $|\det g|=1$ on $N_nK$, hence
\[
I(s;W,V)=V(I_n)\cdot\operatorname{vol}\bigl((N_n\cap K)\backslash K\bigr),
\]
a nonzero constant independent of $s$. Therefore the integral is finite and nonzero for all $s\in\mathbb C$.
\end{proof}
