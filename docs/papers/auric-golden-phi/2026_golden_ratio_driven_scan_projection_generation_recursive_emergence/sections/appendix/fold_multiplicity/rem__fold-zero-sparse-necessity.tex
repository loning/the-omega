\begin{remark}[The necessity of shifting from “zero point geometry” to “amplitude distribution”]\label{rem:fold-zero-sparse-necessity}
Theorem \ref{thm:fold-zero-density-sparse} shows that when \(m\to\infty\) the relative density of the zero point frequency decays exponentially in the full frequency space \(\ZZ/(F_{m+2}\ZZ)\).
Therefore any limiting quantity centered on frequency averaging (such as the non-trivial spectral energy averaging in theorem \ref{thm:fold-collision-spectrum}) cannot be dominated by "zero-point magnitude";
To obtain true asymptotic information, one must understand that at non-zero frequencies
\(
|\widehat d_m(k)|=2^m\prod_{j=1}^m\bigl|\cos(\pi kF_{j+1}/F_{m+2})\bigr|
\)
product amplitude distribution. Equivalently,
\[
\frac{|\widehat d_m(k)|^2}{2^{2m}}
=\prod_{j=1}^m\cos^2\Bigl(\pi k\,F_{j+1}/F_{m+2}\Bigr)
=2^{-m}\prod_{j=1}^m\Bigl(1+\cos\Bigl(2\pi k\,F_{j+1}/F_{m+2}\Bigr)\Bigr),
\]
Therefore, the normalized energy profile can be regarded as a type of finite Riesz product driven by Fibonacci frequency; its extreme value and large deviation control are the entrance to further upgrade "zero-point geometry" to "oscillation statistics".
\end{remark}

\begin{theorem}[collision upper bound of zero-point reduction]
\label{thm:fold-collision-zero-reduction}
Note $\mathcal{Z}_m$ If \ref{def:fold-spectrum-zero-set} is defined, then
\[
\mathsf{Col}_m\le 1-\frac{|\mathcal{Z}_m|}{F_{m+2}}.
\]
In particular, when $|\mathcal{Z}_m|\ge 1$
\[
\mathsf{Col}_m\le 1-\frac{1}{F_{m+2}}.
\]
\end{theorem}

\textbf{Proof:}
By theorem \ref{thm:fold-collision-spectrum},
\(
\mathsf{Col}_m-\frac{1}{F_{m+2}}=\frac{1}{2^{2m}F_{m+2}}\sum_{k\ne 0}|\widehat d_m(k)|^2
\)。
And $|\widehat d_m(k)|\le\sum_r d_m(r)=2^m$, and the value of $k\in\mathcal{Z}_m$ is 0, so
\(
\sum_{k\ne 0}|\widehat d_m(k)|^2\le (F_{m+2}-1-|\mathcal{Z}_m|)\,2^{2m}
\)。
Substitute to get
\[
\mathsf{Col}_m
\le
\frac{1}{F_{m+2}}+\frac{F_{m+2}-1-|\mathcal{Z}_m|}{F_{m+2}}
=1-\frac{|\mathcal{Z}_m|}{F_{m+2}}.
\]
Certification completed.

\begin{remark}[Achievable solution for accurately calculating \texorpdfstring{$\card{\mathcal{Z}_m}$}{|Z_m|} using divisor structure]\label{rem:fold-zero-coset-algorithm}

\begin{enumerate}
\item Enumerate \(d\mid(m+2)\) and \(d<m+2\), let \(g=F_d\), and filter out items that do not satisfy \(g\mid F_{m+2}/2\);
\item Construct the congruence \(k\equiv F_{m+2}/(2g)\pmod{F_{m+2}/g}\) for the preserved \(g\), and use the lemma \ref{lem:fold-zero-coset-intersection} (more generally, the multiple congruence compatibility criterion can be used) to calculate the size of any finite intersection;
\item Use inclusion--exclusion or its equivalent union count implementation to get the exact value of \(\card{\mathcal{Z}_m}\).
\end{enumerate}
This process reduces "scanning \(k\)" to "scanning a small number of divisors of \(m+2\) + a limited number of \(\gcd\)/divisibility determinations", and can be directly connected to the numerical audit pipeline.
The reproducible script can be found at \texttt{scripts/exp\_fold\_zero\_coset\_union\_count.py}.

\subsection{Folding binary interval to Zeckendorf prefix: maximum fiber and saturation discrimination}\label{app:fold-binary-prefix}

The previous article mainly discusses the folding multiplicity induced by \(\Fold_m:\Omega_m=\{0,1\}^m\to X_m\) (Fibonacci weight caliber).
In order to describe "projection compression of \(\,2^m\) binary addresses in the Zeckendorf (\(\varphi\)-base) layer", a folded projection with the same scale as \(|\Omega_m|=2^m\) but with \emph{integer interval} as input is introduced here.

\begin{definition}[binary interval fold \(\mathrm{Fold}^{\mathrm{bin}}_m\)]\label{def:fold-bin}
Let \(Z:\ZZ_{\ge 0}\to\{0,1\}^{\NN}\) be Zeckendorf encoding: for each \(n\ge 0\), there is a unique legal bit string
\[
Z(n)=(z_1,z_2,\dots),\qquad z_k\in\{0,1\},\quad z_k z_{k+1}=0,
\]
make
\[
n=\sum_{k\ge 1} z_k F_{k+1}.
\]
Remember the truncation operator \(\pi_m(z_1,z_2,\dots):=(z_1,\dots,z_m)\in X_m\). definition
\[
\mathrm{Fold}^{\mathrm{bin}}_m:\{0,1,\dots,2^m-1\}\to X_m,\qquad
\mathrm{Fold}^{\mathrm{bin}}_m(n):=\pi_m(Z(n)).
\]
\end{definition}

\begin{definition}[\(K(m)\) with Zeckendorf value]\label{def:K-of-m}
Let \(K(m)\) be satisfied
\begin{equation}\label{eq:K-of-m}
F_{K(m)+1}\le 2^m-1 < F_{K(m)+2}
\end{equation}
unique integer. Define the Zeckendorf value for \(w=(w_1,\dots,w_m)\in X_m\)
\[
V_m(w):=\sum_{k=1}^m w_k F_{k+1}.
\]
\end{definition}

\begin{proposition}[fiber tail representation]\label{prop:fold-bin-fiber-tail}
Let \(m\ge 1\), \(w\in X_m\), and let \(K=K(m)\). Note the tail length \(L:=K-m\ge 0\).
Then \(\mathrm{Fold}^{\mathrm{bin}}_m{}^{-1}(w)\) has a one-to-one correspondence with the tail bit string \((c_{m+1},\dots,c_K)\in\{0,1\}^{L}\) that satisfies the following constraints:
\[
n
=V_m(w)+\sum_{k=m+1}^{K} c_k F_{k+1},
\]
in
\begin{enumerate}
\item \textbf{Adjacent constraints:}\quad \(c_k c_{k+1}=0\) (for \(m+1\le k\le K-1\));
\item \textbf{Boundary constraints:}\quad \(w_m\,c_{m+1}=0\);
\item \textbf{Threshold constraint:}\quad \(V_m(w)+\sum_{k=m+1}^{K} c_k F_{k+1}\le 2^m-1\).
\end{enumerate}
Therefore fiber multiplicity
\(
d^{\mathrm{bin}}_m(w):=\bigl|\mathrm{Fold}^{\mathrm{bin}}_m{}^{-1}(w)\bigr|
\)
Depends only on \((V_m(w),w_m)\).
\end{proposition}

Explicit fiber classification and stiffness histogram of \begin{corollary}[\texorpdfstring{$m=6$}{m=6}]\label{cor:fold-bin-m6-fiber-hist}

\begin{proposition}[\texorpdfstring{$\chi^2$}{chi2} Identity of divergence and collision gap (bin-fold)]\label{prop:fold-bin-chi2-col}
make

\]

\[

\qquad

\]
And \(\chi^2\) the divergence satisfies
\[
\chi^2\!\bigl(p^{\mathrm{bin}}_m\|u_m\bigr)
:=\sum_{w\in X_m}\frac{\bigl(p^{\mathrm{bin}}_m(w)-u_m(w)\bigr)^2}{u_m(w)}
=|X_m|\,g_{\mathrm{col}}^{2,\mathrm{bin}}(m)
=|X_m|\,\mathsf{Col}^{\mathrm{bin}}_m-1.
\]

\end{proposition}
\begin{proof}
Defined by \(p^{\mathrm{bin}}_m(w)=d^{\mathrm{bin}}_m(w)/2^m\) and
\(

=2^{-2m}\sum_{w}(d^{\mathrm{bin}}_m(w))^2
\)
Obtain the first identity.
\par

\[
\sum_w\bigl(p^{\mathrm{bin}}_m(w)-u_m(w)\bigr)^2
=\sum_w \bigl(p^{\mathrm{bin}}_m(w)\bigr)^2-\frac{1}{|X_m|}
=\mathsf{Col}^{\mathrm{bin}}_m-\frac{1}{|X_m|}
=g_{\mathrm{col}}^{2,\mathrm{bin}}(m).
\]
Finally use \(u_m(w)=1/|X_m|\) to get
\(

\)，
This gives the chain of equations shown. Certification completed.
\end{proof}

\begin{definition}[binary range folding fold-gauge group]\label{def:fold-bin-gauge-group}
Fixed \(m\ge 1\). Defining Fiber Automorphism Groups for Binary Interval Folding
\[
\mathcal{G}^{\mathrm{bin}}_m
:=
\mathrm{Aut}\!\bigl(\mathrm{Fold}^{\mathrm{bin}}_m\bigr)
:=
\left\{\sigma\in \mathrm{Sym}(\{0,\dots,2^m-1\}):\ \mathrm{Fold}^{\mathrm{bin}}_m\circ\sigma=\mathrm{Fold}^{\mathrm{bin}}_m\right\}.
\]
Equivalently, \(\sigma\) can be arbitrarily replaced within each fiber \(\mathrm{Fold}^{\mathrm{bin}}_m{}^{-1}(w)\) without changing the stable type label \(w\in X_m\).
\end{definition}

\begin{proposition}[fiber replacement decomposition]\label{prop:fold-bin-gauge-decomposition}
Order \(d^{\mathrm{bin}}_m(w):=\bigl|\mathrm{Fold}^{\mathrm{bin}}_m{}^{-1}(w)\bigr|\).
then there is a natural isomorphism
\[
\mathcal{G}^{\mathrm{bin}}_m\ \cong\ \prod_{w\in X_m} S_{d^{\mathrm{bin}}_m(w)}.
\]
\end{proposition}
\begin{proof}
The fiber \(\{\mathrm{Fold}^{\mathrm{bin}}_m{}^{-1}(w)\}_{w\in X_m}\) of \(\mathrm{Fold}^{\mathrm{bin}}_m\) gives a disjoint partition of \(\{0,\dots,2^m-1\}\).
For any \(\sigma\in\mathrm{Aut}(\mathrm{Fold}^{\mathrm{bin}}_m)\) and any \(n\in \mathrm{Fold}^{\mathrm{bin}}_m{}^{-1}(w)\), both
\(\mathrm{Fold}^{\mathrm{bin}}_m(\sigma(n))=\mathrm{Fold}^{\mathrm{bin}}_m(n)=w\), so \(\sigma\) must be replaced inside each fiber.
On the other hand, if for each \(w\in X_m\) we take any intra-fiber substitution \(\sigma_w\in S_{d^{\mathrm{bin}}_m(w)}\) and combine them into the substitution \(\sigma=\prod_w\sigma_w\) for all \(\{0,\dots,2^m-1\}\), then by construction we have \(\mathrm{Fold}^{\mathrm{bin}}_m(\sigma(n))=\mathrm{Fold}^{\mathrm{bin}}_m(n)\) for any \(n\), so \(\mathrm{Fold}^{\mathrm{bin}}_m\circ\sigma=\mathrm{Fold}^{\mathrm{bin}}_m\).
This leads to the direct integral analytical isomorphism shown.
\end{proof}

\begin{proposition}[Wedderburn decomposition of fiber group algebra]\label{prop:fold-bin-groupoid-algebra-wedderburn}
Let \(\CC[\mathcal{R}^{\mathrm{bin}}_m]\) be the complex group embryo algebra of \(\mathcal{R}^{\mathrm{bin}}_m\) (based on finite support functions on arrow sets and convolution as multiplication).
Then there is a natural algebraic isomorphism
\[
\CC[\mathcal{R}^{\mathrm{bin}}_m]\ \cong\ \bigoplus_{w\in X_m} M_{d^{\mathrm{bin}}_m(w)}(\CC).
\]
Therefore \(\CC[\mathcal{R}^{\mathrm{bin}}_m]\) is a finite-dimensional semi-single algebra and its Wedderburn block decomposition is completely determined by the degenerate spectral histogram \(\{d:\#\mathcal{S}_{m,d}\}\).
\end{proposition}
\begin{proof}
For each \(w\in X_m\), let \(F_w:=\mathrm{Fold}^{\mathrm{bin}}_m{}^{-1}(w)\).
Take base in \(\CC[F_w\times F_w]\)
\(
E_{ab}:=\mathbf{1}_{(a\to b)}
\)（\(a,b\in F_w\)）。
Calculated directly from the convolution definition
\(
E_{ab}*E_{cd}=\delta_{bc}\,E_{ad}
\)，
So \(\CC[F_w\times F_w]\cong M_{|F_w|}(\CC)=M_{d^{\mathrm{bin}}_m(w)}(\CC)\).
The arrow sets of different \(w\) do not intersect, and the convolution has no cross terms, so the whole algebra is a direct sum. Certification completed.
\end{proof}

\begin{theorem}[Exponential accuracy of Stirling first term: linear difference between \texorpdfstring{$g_m$}{gm} and \texorpdfstring{$\kappa_m$}{kappa}]\label{thm:fold-bin-gauge-volume-stirling}
Under the setting of proposition \ref{prop:fold-bin-gauge-volume},
\[
g_m=\kappa_m-1+O\!\left(\frac{|X_m|\,m}{2^m}\right)\qquad(m\to\infty).
\]
In particular,
\[
\kappa_m-g_m
=1+O\!\left(\frac{|X_m|\,m}{2^m}\right)
=1+O\!\left(m\left(\frac{\varphi}{2}\right)^m\right),
\]
Thus \(\kappa_m-g_m=1+o(1)\).
\end{theorem}
\begin{proof}
There is a Stirling-type remainder bound for the integer \(d\ge 1\)
\(
\log(d!)=d\log d-d+O(\log d)
\)。
Substitute
\(
\log|\mathcal{G}^{\mathrm{bin}}_m|
=\sum_{w\in X_m}\log(d^{\mathrm{bin}}_m(w)!)
\)
And divided by \(2^m\), we get
\[
g_m
=\frac{1}{2^m}\sum_{w\in X_m}d^{\mathrm{bin}}_m(w)\log d^{\mathrm{bin}}_m(w)
-\frac{1}{2^m}\sum_{w\in X_m}d^{\mathrm{bin}}_m(w)
 +O\!\left(\frac{1}{2^m}\sum_{w\in X_m}\log d^{\mathrm{bin}}_m(w)\right).
\]
The first item is \(\kappa_m\) and the second item is given by \(\sum_w d^{\mathrm{bin}}_m(w)=2^m\) \(-1\).
For the error term, use \(d^{\mathrm{bin}}_m(w)\le 2^m\Rightarrow \log d^{\mathrm{bin}}_m(w)\le m\log 2\), so
\(
\sum_{w\in X_m}\log d^{\mathrm{bin}}_m(w)\le |X_m|\,m\log 2
\)，
Thus getting \(O(|X_m|m/2^m)\).
Finally \(|X_m|=F_{m+2}=\Theta(\varphi^m)\), so \(|X_m|/2^m=\Theta((\varphi/2)^m)\), substitute it back to get the second row. Certification completed.
\end{proof}
