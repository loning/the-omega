\section{可审计的 profinite/Chebotarev 证书：有限 \texorpdfstring{$(m,n,N)$}{(m,n,N)}、扭曲谱隙与缺陷极限}\label{app:auditable-profinite-chebotarev}

本附录迁移并改写一条对 BridgeRH 直接有用的“可执行协议层”：把同余/角色扭曲的 Chebotarev 型等分布（prime-like 侧的分层主项）压缩为一份\textbf{有限数据证书}。其价值在于：它为 (B1.b)（轨道侧一致性）与 (B4)（规格层审计）提供一条与 Abel--Weil/Rouch{\'e} 正交的回归通道——\emph{按模数/同余 cylinder} 的分布律必须与最坏扭曲谱半径比的证书一致；任何失配都可被当作可否证信号。

\subsection{对象：有限层同余标签、扭曲通道与 cylinder}\label{subsec:profinite-chebotarev-objects}
取一个有限阿贝尔群 $G_m$（可视为 prime-register 在模数 $m$ 上的可见标签层），并记其角色群为 $\widehat{G_m}$，平凡角色为 $\chi_0$。
设核侧有限有向图 $\Gamma$ 的每条边 $e$ 附带：
\begin{itemize}
  \item \textbf{势/代价} $\kappa(e)\in\ZZ_{\ge 0}$（用于 $u$-加权，见附录 \ref{app:bridge-rh-obligations} 的定义 \ref{def:zetaK-two-variable}）；
  \item \textbf{同余标签} $g(e)\in G_m$（一个 $G_m$-cocycle；闭轨道标签为边标签之积）。
\end{itemize}
对每个角色 $\chi\in\widehat{G_m}$，定义扭曲邻接矩阵族
$$
\bigl(M_{\chi}(u)\bigr)_{ij}:=\sum_{e:i\to j}\chi(g(e))\,u^{\kappa(e)}.
$$
记平凡通道的 Perron 根为
$$
\lambda(u):=\rho(M_{\chi_0}(u)),
$$
并记最坏非平凡扭曲谱半径比为
$$
\eta_m(u):=\max_{\chi\in\widehat{G_m}\setminus\{\chi_0\}}\frac{\rho(M_{\chi}(u))}{\lambda(u)}\in(0,1].
$$
给定 cylinder 事件 $C\subseteq G_m$，令 $p_{n,g}(u)$ 表示长度为 $n$ 的 primitive 闭轨道中标签落在 $g\in G_m$ 的加权计数（权重为轨道势的乘法权重 $u^{\kappa(\gamma)}$），并令 $p_n(u)=\sum_{g\in G_m}p_{n,g}(u)$。
定义经验比例
$$
\Pi_{n,m}(C;u):=\frac{\sum_{g\in C}p_{n,g}(u)}{p_n(u)}\in[0,1].
$$

\subsection{有限 \texorpdfstring{$(m,n,N)$}{(m,n,N)} 证书：尾项 \texorpdfstring{$+$}{+} TV 项}\label{subsec:profinite-chebotarev-finite}

\begin{assumption}[有限样本的 TV 包络]\label{ass:tv-envelope}
存在可复算上界 $\epsilon_{m,N}\ge 0$ 使得对同余标签的经验直方图 $\hat{\pi}_{m,N}$ 与真实边缘 $\pi_m$ 有
$$
D_{\mathrm{TV}}(\hat{\pi}_{m,N},\pi_m)\le \epsilon_{m,N}.
$$
（$\epsilon_{m,N}$ 可由 Markov 浓缩、不等距采样差异度界或区间算术协议给出；本附录仅把它当作可审计输入。）
\end{assumption}

\begin{theorem}[可审计的 profinite/Chebotarev 证书：有限 \texorpdfstring{$(m,n,N)$}{(m,n,N)}]\label{thm:auditable-profinite-chebotarev-finite-bridge}
固定层 $m$、cylinder $C\subseteq G_m$ 与协议可达的 $u$。
若存在 $\varepsilon>0$ 使得在该层满足统一谱隙假设
$$
\rho(M_{\chi}(u))\le \lambda(u)^{1/2-\varepsilon}\qquad(\forall\,\chi\neq\chi_0),
$$
则存在常数 $K_1=K_1(m,u)$ 使得对所有 $n\ge 1$ 有确定性尾项界
$$
\left|\Pi_{n,m}(C;u)-\frac{|C|}{|G_m|}\right|
\le
K_1\,\frac{|G_m|}{\lambda(u)^{(1/2+\varepsilon)n}}.
$$
进一步地，若以经验直方图 $\hat{\pi}_{m,N}$ 实现同一 cylinder 读出，并且满足假设 \ref{ass:tv-envelope}，
则有完全显式的有限数据证书
$$
\boxed{\;
\left|\Pi_{n,m}(C;u)-\frac{|C|}{|G_m|}\right|
\le
K_1\,\frac{|G_m|}{\lambda(u)^{(1/2+\varepsilon)n}}
\ +\ 2\,\epsilon_{m,N}\; }.
$$
\end{theorem}

\begin{proof}
对同余标签的计数作角色分解（对有限阿贝尔群的 Fourier 反演）：
主项来自平凡角色 $\chi_0$，而偏差由非平凡角色的扭曲通道贡献控制。
在长度 $n$ 上，非平凡通道的贡献被 $\max_{\chi\neq\chi_0}\rho(M_\chi(u))^{n}$ 上界控制；在统一谱隙假设下即得到 $\lambda(u)^{(1/2-\varepsilon)n}$ 量级，从而归一化为比例后得到所示尾项（常数吸收到 $K_1$）。
第二项由 TV 距离控制有界可观测的期望差：对指示函数 $1_C$ 有
$
|\mathbb{E}_{\hat{\pi}}1_C-\mathbb{E}_{\pi}1_C|\le 2D_{\mathrm{TV}}(\hat{\pi},\pi)
$，
代入假设 \ref{ass:tv-envelope} 即得。证毕。
\end{proof}

\begin{remark}[谱隙证书的输出格式（Lyapunov--LMI）]\label{rem:chebotarev-lyapunov}
定理 \ref{thm:auditable-profinite-chebotarev-finite-bridge} 的统一谱隙假设可被组织为一组可复算 PSD 证书：
对每个非平凡角色 $\chi\neq\chi_0$，输出 $H_{\chi,u}\succ 0$ 使
$$
M_{\chi}(u)^\ast H_{\chi,u} M_{\chi}(u)\ \prec\ \lambda(u)^{1-2\varepsilon}H_{\chi,u},
$$
则由附录 \ref{app:xi-certificate-layer} 的命题 \ref{prop:lyapunov-spectral-radius-certificate} 得到
$
\rho(M_\chi(u))<\lambda(u)^{1/2-\varepsilon}
$。
因此，Chebotarev 证书可被压缩为“有限个矩阵不等式证书 + 有限样本 TV 包络”两部分，天然可并行生成与并行复算。
\end{remark}

\begin{remark}[扩散障碍：统一谱隙假设的逆极限门槛]\label{rem:chebotarev-diffusion-barrier}
定理 \ref{thm:auditable-profinite-chebotarev-finite-bridge} 的统一谱隙假设是一个\emph{跨模数层}的强规格。
在 Dirichlet 模轴的 near-coboundary 小角二次律下，最坏扭曲比值 \(\eta_m(u)\) 以 \(1-\Theta(m^{-2})\) 贴近 \(1\)，从而混合尺度为 \(\Theta(m^2)\)，并导出 fixed finite kernel 的 uniform-\(\sqrt{\lambda}\) no-go（附录 \ref{app:diffusion-barrier-spectral-tax}）。
因此在 inverse-limit 语义中，更自然的可审计口径往往是“模数 \(m\) 与长度 \(n\) 的双极限可见域”，或显式引入随 \(m\) 增长的核重整化族。
\end{remark}

\subsection{Plancherel 能量恒等式：同余偏差的 \texorpdfstring{$\ell^2$}{l2} 标尺}\label{subsec:chebotarev-plancherel}
除 TV 误差外，同余分层还存在一个完全确定、且与角色扭曲通道严格对齐的 \(\ell^2\) 能量标尺：它把“均方偏差”写成“非平凡角色能量和”。

\begin{proposition}[Chebotarev--Plancherel 能量恒等式]\label{prop:chebotarev-plancherel-energy}
固定层 \(m\)、长度 \(n\) 与协议可达的 \(u\)。定义点质量分布
$$
\pi_{n,m}(g;u):=\frac{p_{n,g}(u)}{p_n(u)}\qquad(g\in G_m),
$$
并令其中心化偏差为
$$
f_{n,m}(g;u):=\pi_{n,m}(g;u)-\frac{1}{|G_m|}.
$$
对任意角色 \(\chi\in\widehat{G_m}\) 定义 Fourier 系数（非归一化 DFT）
$$
\widehat f_{n,m}(\chi;u):=\sum_{g\in G_m} f_{n,m}(g;u)\,\overline{\chi(g)}.
$$
则成立 Parseval/Plancherel 恒等式
$$
\boxed{\;
\sum_{g\in G_m}\left|f_{n,m}(g;u)\right|^2
\ =\ 
\frac{1}{|G_m|}\sum_{\chi\in\widehat{G_m}}\left|\widehat f_{n,m}(\chi;u)\right|^2
\ =\ 
\frac{1}{|G_m|}\sum_{\chi\neq\chi_0}\left|\widehat \pi_{n,m}(\chi;u)\right|^2\; },
$$
其中 \(\widehat \pi_{n,m}(\chi;u):=\sum_{g}\pi_{n,m}(g;u)\,\overline{\chi(g)}\) 且 \(\widehat f_{n,m}(\chi_0;u)=0\)。
\end{proposition}
\begin{proof}
有限阿贝尔群上的角色基正交给出 Parseval 恒等式：
对任意函数 \(h:G_m\to\CC\)，有
$$
\sum_{g\in G_m}|h(g)|^2=\frac{1}{|G_m|}\sum_{\chi\in\widehat{G_m}}|\widehat h(\chi)|^2,
$$
其中 \(\widehat h(\chi)=\sum_{g}h(g)\overline{\chi(g)}\)。
将 \(h=f_{n,m}(\cdot;u)\) 代入得第一等号。
另一方面，由 \(\sum_g\pi_{n,m}(g;u)=1\) 得 \(\sum_g f_{n,m}(g;u)=0\)，从而 \(\widehat f_{n,m}(\chi_0;u)=0\)。
并且对 \(\chi\neq\chi_0\)，常数项的 Fourier 系数为 \(0\)，故
$$
\widehat f_{n,m}(\chi;u)=\widehat \pi_{n,m}(\chi;u).
$$
代入 Parseval 得第二等号。证毕。
\end{proof}

\begin{remark}[可审计用途：由 \texorpdfstring{$\ell^2$}{l2} 偏差反推出瓶颈扭曲模态]\label{rem:chebotarev-plancherel-use}
命题 \ref{prop:chebotarev-plancherel-energy} 提供一条与 TV 上界正交的“能量护栏”：
\begin{itemize}
  \item 若观测到 \(\sum_g|f_{n,m}(g;u)|^2\) 在某个尺度上无法继续降低，则至少存在一个非平凡角色 \(\chi\neq\chi_0\) 使得 \(|\widehat \pi_{n,m}(\chi;u)|\) 仍然显著，从而定位出瓶颈扭曲模态；
  \item 若同时输出 Lyapunov--LMI 的扭曲谱隙证书（注 \ref{rem:chebotarev-lyapunov}），则该能量应与谱隙指数衰减相容；任何不相容都可作为可否证信号。
\end{itemize}
因此，在 (B1.b) 的同余分层审计中，建议把“TV/区间误差上界”与“\(\ell^2\) 能量守恒式”并联作为回归指纹。
\end{remark}

\subsection{缺陷极限：仅凭折叠边缘无法突破的 TV/KL 下界}\label{subsec:defect-limit}

\begin{definition}[纤维缺陷标量]\label{def:fiber-defect-varkappa}
对任意边缘 $\pi\in\Delta(X_m)$，定义纤维缺陷
$$
\varkappa_m(\pi):=\mathbb{E}_{\pi}[\log d_m(X)].
$$
\end{definition}

\begin{proposition}[缺陷极限（可审计）：任意 lift 相对最大熵 lift 的 KL/TV 上界]\label{prop:defect-limited-audit-bridge}
固定 $m$，取任意 $\mu\in\Delta(\Omega_m)$，记 $\pi=P_m\mu$，$\mu^e:=Q_m\pi$ 为纤维均匀提升（见正文定理 \ref{thm:ledger-identity}）。
则有
$$
D_{\mathrm{KL}}(\mu\|\mu^e)\ \le\ \varkappa_m(\pi),
$$
从而由 Pinsker 不等式得到
$$
D_{\mathrm{TV}}(\mu,\mu^e)\ \le\ \sqrt{\frac{\varkappa_m(\pi)}{2}}.
$$
\end{proposition}
\begin{proof}
由信息账本恒等式（定理 \ref{thm:ledger-identity}）整理得
$$
D_{\mathrm{KL}}(\mu\|\mu^e)=H(\pi)+\mathbb{E}_{\pi}[\log d_m(X)]-H(\mu).
$$
由于 $\pi=P_m\mu$ 为确定性推前，熵满足 $H(\pi)\le H(\mu)$，故
$
D_{\mathrm{KL}}(\mu\|\mu^e)\le \mathbb{E}_{\pi}[\log d_m(X)]=\varkappa_m(\pi)
$。
再用 Pinsker 不等式即得 TV 上界。证毕。
\end{proof}

\begin{remark}[对 BridgeRH 的含义]\label{rem:defect-meaning}
命题 \ref{prop:defect-limited-audit-bridge} 给出一个与谱隙无关的“可审计极限”：当 $\varkappa_m(\pi)$ 较大时，仅凭折叠边缘 $\pi$ 不可能把任何 lift 级同余/cylinder 可观测的系统性误差压到 $o(\sqrt{\varkappa_m(\pi)})$。
因此，在 (B1.c)/(B4) 的协议设计中，若希望以更高模数层 $G_m$ 的 cylinder 统计作为桥接回归指纹，就必须同时提高可见寄存器的容量或引入更强的证书层（例如 Lyapunov--LMI 的扭曲谱隙证书），否则会被纤维缺陷的不可见自由度预算所限制。
\end{remark}
