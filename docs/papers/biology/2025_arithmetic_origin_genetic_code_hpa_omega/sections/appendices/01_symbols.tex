\section{Symbols and fixed objects}
\label{app:symbols}

\begin{itemize}
\item $\Sigma=\{A,C,G,U\}$: RNA alphabet.
\item $\mathsf{Codon}=\Sigma^3$: set of $64$ codons.
\item $\mathsf{Gen}:\mathsf{Codon}\twoheadrightarrow \mathsf{AA}\cup\{\mathrm{Stop}\}$: standard genetic code map.
\item $\mu:\Sigma\to\{00,01,10,11\}$: bijective two-bit nucleotide encoding; search space size $4!=24$.
\item $\mathcal{K}=\{\mathrm{AUG,UAA,UAG,UGA}\}$: control codon set (start and standard stops).
\item $S(\mu)=\sum_{c\in\mathcal{K}}\mathbf{1}\{w_\mu(c)\in X_6^{\mathrm{bdry}}\}$: control-boundary alignment score.
\item $\Omega_6=\{0,1\}^6$: six-bit microstate space.
\item $X_6\subset\{0,1\}^6$: admissible words with no adjacent ones; $|X_6|=21$.
\item $X_6^{\mathrm{bdry}}=\{100001,100101,101001\}$: boundary sector.
\item $\mathrm{Fold}_6:\{0,\dots,63\}\twoheadrightarrow X_6$: Zeckendorf-window folding map.
\item $V(w)=\sum_{k=1}^6 w_k F_{k+1}$: Zeckendorf value on $X_6$.
\item For codon $c$: $N_\mu(c)\in\{0,\dots,63\}$ is the binary index, $w_\mu(c)=\mathrm{Fold}_6(N_\mu(c))$, $V_\mu(c)=V(w_\mu(c))$, $\Delta_\mu(c)=N_\mu(c)-V_\mu(c)$.
\item $V_{\mathrm{low},\mu}(c):=V_\mu(c)$ and $V_{\mathrm{high},\mu}(c):=\Delta_\mu(c)$: low/high split with $N_\mu(c)=V_{\mathrm{low},\mu}(c)+V_{\mathrm{high},\mu}(c)$.
\item For a sequence split into codons $(c_i)$: $Z(i):=V_{\mu^\ast}(c_i)$ and $U(i):=\Delta_{\mu^\ast}(c_i)$.
\item $B(i):=\mathbf{1}\{w_{\mu^\ast}(c_i)\in X_6^{\mathrm{bdry}}\}$: boundary indicator at codon position $i$.
\end{itemize}
