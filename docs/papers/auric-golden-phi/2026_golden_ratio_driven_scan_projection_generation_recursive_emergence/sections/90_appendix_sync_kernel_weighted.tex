\subsubsection{加权核的压力与大偏差（可审计接口）}
令 $\phi$ 为同步图边上的可观测量（例如输出位 $e$ 的指示函数，或"到达含 $\overline{1}$ 状态"的指示函数）。定义加权矩阵
\[
(B_\theta)_{ij}:=\sum_{e:i\to j}\exp\bigl(\theta\,\phi(e)\bigr),
\]
其中求和遍历所有从 $i$ 到 $j$ 的边。记
\[
\lambda(\theta):=\rho(B_\theta),\qquad P(\theta):=\log\lambda(\theta).
\]
则 $P(\theta)$ 为有限型情形的压力函数，且 $P(0)=\log 3$。令
\[
I(\alpha):=\sup_{\theta\in\RR}\bigl(\theta\alpha-(P(\theta)-P(0))\bigr),
\]
则 $I(\alpha)$ 给出 $\phi$ 的长程平均的大偏差率函数（标准热力学形式口径，见 \cite{Ruelle1978Thermodynamic,ParryPollicott1990Zeta}）。

加权 $\zeta$ 仍由
\[
\zeta_{B,\theta}(z)=\frac{1}{\det(I-zB_\theta)}
\]
给出，并可定义加权 primitive 轨道谱
\[
a_n(\theta):=\mathrm{Tr}(B_\theta^n),\qquad
p_n(\theta)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}(d\theta),
\]
以及相应的加权 Mertens 常数族（对 $z\to 1/\lambda(\theta)$ 取 Abel 有限部）。
可审计数值由脚本 \texttt{scripts/exp\_sync\_kernel\_weighted\_pressure.py} 输出。

\subsubsection{加权 primitive--periodic 的 Witt 分解}\label{app:witt-weighted}
以下取输出位势 $\varphi(e)=e$，令
\[
u=e^\theta,\qquad B(u)=B_0+uB_1,\qquad \Delta(z,u):=\det(I-zB(u)).
\]
其中 $B_0$（resp.\ $B_1$）只计数输出位 $e=0$（resp.\ $e=1$）的边。
则
\[
\zeta(z,u)=\exp\!\left(\sum_{n\ge 1}\frac{\mathrm{Tr}(B(u)^n)}{n}z^n\right)=\Delta(z,u)^{-1}.
\]
令
\[
a_n(u):=\mathrm{Tr}(B(u)^n),\qquad
P(z,u):=\sum_{n\ge 1}p_n(u)z^n,
\]
其中
\(
p_n(u)=\sum_{\tau\ \mathrm{primitive},\,|\tau|=n}u^{S_\tau\varphi}
\)
为 primitive 轨道的加权谱多项式。Euler 乘积给出
\[
\zeta(z,u)=\prod_{\tau\ \mathrm{primitive}}\bigl(1-z^{|\tau|}u^{S_\tau\varphi}\bigr)^{-1}.
\]
取对数并展开幂重复得到 Witt 分解
\[
\log\zeta(z,u)=\sum_{m\ge 1}\frac{1}{m}P(z^m,u^m).
\]
由 Möbius 反演得到
\[
P(z,u)=\sum_{m\ge 1}\frac{\mu(m)}{m}\log\zeta(z^m,u^m),
\]
对系数比较即有
\[
p_n(u)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}(u^d)
\quad\Longleftrightarrow\quad
p_n(\theta)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}(d\theta).
\]

\subsubsection{Big-Witt/necklace 桥包与可检验同余}\label{app:witt-bridge-package}
上式不仅是“可计算公式”，而是 Big-Witt（necklace ring）上的标准坐标变换：$a_n(u)$ 为 ghost 分量，$p_n(u)$ 为 Witt 坐标，而 Euler 乘积是 plethystic exponential。等价地，对任意带参矩阵族 $T(u)$，有严格恒等式
\[
\boxed{\ \zeta_T(z,u)=\mathrm{Exp}_W\bigl(\mathrm{Mob}_W(\mathrm{TrSeq}(T))\bigr)\ }
\]
其中 $\mathrm{TrSeq}(T)=(a_n(u))_{n\ge 1}$、$\mathrm{Mob}_W(a)=p$ 给出 $p_n(u)=\frac1n\sum_{d\mid n}\mu(d)\,a_{n/d}(u^d)$，而 $\mathrm{Exp}_W$ 由
\(
\log\zeta=\sum_{m\ge 1}\frac{1}{m}P(z^m,u^m)
\)
定义。该“桥包”直接产出一批可检验算术约束：

\begin{proposition}[交换图的函子化：intertwiner 下的自然性]\label{prop:witt-bridge-functorial}
取系数环 \(R\)（例如 \(R=\ZZ[u]\) 或 \(R=\ZZ[u,u^{-1}]\)），并令 \(\mathbf{End}_R\) 的对象为 \((V,T)\)（有限秩自由 \(R\)-模与其 \(R\)-线性自同态），态射为 intertwiner \(f:(V,T)\to (W,S)\) 满足 \(f\circ T=S\circ f\)。
令 \(\mathbf{Seq}_R\) 为 \(R\)-值序列范畴（按分量逐项映射），并定义
\[
\mathsf{Ghost}:\mathbf{End}_R\to \mathbf{Seq}_R,\qquad
(V,T)\mapsto (a_n)_{n\ge 1},\ a_n:=\Tr(T^n).
\]
则 \(\mathsf{Ghost}\) 为函子，且对任意 intertwiner \(f\) 有 \(a_n(T)=a_n(S)\)（因此 \(\zeta_T=\zeta_S\)）。
再令 \(\mathsf{Mob}\) 为带 Adams 操作 \(\psi^d(u)=u^d\) 的 Witt--Möbius 变换
\[
\mathsf{Mob}(a)=p,\qquad
p_n(u):=\frac1n\sum_{d\mid n}\mu(d)\,a_{n/d}(u^d),
\]
并令 \(\mathsf{Euler}\) 为由 \(p\) 生成 Euler 乘积（等价于 plethystic exponential）的构造。
则对每个 \((V,T)\) 有严格恒等式
\[
\boxed{\;
\mathsf{Zeta}(V,T)
=\mathsf{Euler}\bigl(\mathsf{Mob}(\mathsf{Ghost}(V,T))\bigr)\; },
\qquad
\mathsf{Zeta}(V,T):=\exp\!\left(\sum_{n\ge 1}\frac{\Tr(T^n)}{n}z^n\right)=\det(I-zT)^{-1}.
\]
因此“trace（时间 ghost）\(\leftrightarrow\) primitive（空间原子）”的交换图不仅是类比，而是由 Big-Witt 坐标变换给出的严格自然变换。
\end{proposition}

\begin{proposition}[necklace 整除律]\label{prop:witt-necklace-div}
对任意 $n\ge 1$，有
\[
n\ \Big|\ \sum_{d\mid n}\mu(d)\,a_{n/d}(u^d)\qquad\text{于}\ \mathbb{Z}[u].
\]
\end{proposition}
\begin{proof}
由于 $p_n(u)\in\mathbb{Z}[u]$ 是 primitive 轨道的加权计数，多项式恒在整系数环内；由 $n\,p_n(u)=\sum_{d\mid n}\mu(d)a_{n/d}(u^d)$ 得整除结论。证毕。
\end{proof}

\begin{corollary}[$p$-幂 Dwork 型同余]\label{cor:witt-dwork-congruence}
取 $n=p^k$（素数 $p$），则
\[
a_{p^k}(u)\equiv a_{p^{k-1}}(u^p)\pmod{p^k}\qquad\text{于}\ \mathbb{Z}[u].
\]
\end{corollary}
\begin{proof}
由 $\mu(1)=1,\mu(p)=-1,\mu(p^2)=0$ 得
\(
p^k\,p_{p^k}(u)=a_{p^k}(u)-a_{p^{k-1}}(u^p)
\)。
利用命题 \ref{prop:witt-necklace-div} 即得结论。证毕。
\end{proof}

\begin{proposition}[系数环版本：necklace 整除律与 Dwork 型 \texorpdfstring{$p$}{p}-幂同余]\label{prop:witt-necklace-dwork-R}
令 \(R\) 为交换的 \(\ZZ\)-代数，并带有 Adams 操作族 \(\psi^d:R\to R\)（\(d\ge 1\)）满足
\(\psi^1=\mathrm{id}\)、\(\psi^{de}=\psi^d\circ\psi^e\)，且在参数多项式环情形 \(\psi^d\) 由逐变量幂替换给出（例如在 \(\ZZ[u_1,\dots,u_m]\) 上为 \(u_i\mapsto u_i^d\)）。
设 \(B(\mathbf u)\) 为一有限加权同步图/有限状态核所诱导的加权邻接矩阵族，其条目属于 \(R\)（因此每条边携带的权重为 \(R\) 中的一个元素），并定义
\[
a_n:=\Tr\bigl(B(\mathbf u)^n\bigr)\in R,\qquad n\ge 1.
\]
将 primitive 加权谱多项式定义为
\[
p_n:=\sum_{\tau\ \mathrm{primitive},\,|\tau|=n} w(\tau)\in R,
\]
其中 \(w(\tau)\in R\) 为该 primitive 轨道的权重乘积（对多参数即为对应单项式/权重的乘积）。
则对每个 \(n\ge 1\) 有严格恒等式（Witt/Möbius 反演）
\[
n\,p_n=\sum_{d\mid n}\mu(d)\,\psi^d\!\bigl(a_{n/d}\bigr)\in R.
\]
从而得到两条对任何此类“Euler 乘积核”成立的算术约束：
\begin{enumerate}
  \item \textbf{（necklace 整除）} \(\sum_{d\mid n}\mu(d)\,\psi^d(a_{n/d})\in nR\)；
  \item \textbf{（Dwork 型同余）} 对任意素数 \(p\) 与 \(k\ge 1\)，有
  \[
  a_{p^k}\equiv \psi^{p}\!\bigl(a_{p^{k-1}}\bigr)\pmod{p^k}\qquad\text{于}\ R.
  \]
\end{enumerate}
\end{proposition}
\begin{proof}
在有限图/SFT 核口径下，周期点（闭环）与 primitive 轨道之间满足标准关系
\[
a_n=\sum_{d\mid n} d\,\psi^{n/d}(p_d)\qquad(n\ge 1),
\]
其含义是：长度 \(n\) 的闭环按最小周期 \(d\) 分解后，每个 primitive 轨道在 \(n/d\) 次幂重复下贡献 \(d\) 个相位起点；而权重在幂重复下由 Adams 操作 \(\psi^{n/d}\) 推进（在多参数多项式环上即为逐变量 \(x\mapsto x^{n/d}\) 的幂替换）。
对上式作 Möbius 反演得到
\(
n\,p_n=\sum_{d\mid n}\mu(d)\psi^d(a_{n/d})
\)，并且 \(p_n\in R\) 由其作为 primitive 轨道加权和的定义保证。
当 \(n=p^k\) 时，因 \(\mu(p^j)=0\)（\(j\ge 2\)），上式塌缩为
\(
p^k p_{p^k}=a_{p^k}-\psi^{p}(a_{p^{k-1}})
\)，从而给出所示的 \(\bmod p^k\) 同余。证毕。
\end{proof}

\begin{corollary}[\texorpdfstring{$p$}{p}-进超同余：\texorpdfstring{$a_{p^k}(-1)$}{a_{p^k}(-1)} 被 \texorpdfstring{$p^k$}{p^k} 整除]\label{cor:witt-dwork-supercongruence-minus1}
取任意奇素数 \(p\)。则对任意 \(k\ge 1\) 有
\[
\boxed{\ a_{p^k}(-1)\equiv 0\pmod{p^k}\ }.
\]
\end{corollary}
\begin{proof}
由推论 \ref{cor:witt-dwork-congruence}，对 \(k\ge 1\) 有
\[
a_{p^k}(u)\equiv a_{p^{k-1}}(u^p)\pmod{p^k}\qquad\text{于}\ \ZZ[u].
\]
在 \(u=-1\) 处特化并用 \(p\) 为奇数（从而 \((-1)^p=-1\)）得
\[
a_{p^k}(-1)\equiv a_{p^{k-1}}(-1)\pmod{p^k}.
\]
而 \(p^{k-1}\) 为奇数，故由命题 \ref{prop:trace-palindrome} 的结论 \(a_n(-1)=0\)（奇数 \(n\)）知 \(a_{p^{k-1}}(-1)=0\)。
于是 \(a_{p^k}(-1)\equiv 0\pmod{p^k}\)。证毕。
\end{proof}

\begin{remark}[\texorpdfstring{$p$}{p}-幂方向的 Witt 解释：Frobenius 兼容的 ghost 分量]\label{rem:witt-ptypical-ghost}
把 \(a_n(u)\) 视作 Big-Witt（necklace）环上的 ghost 坐标，则 \(\{a_{p^k}(u)\}_{k\ge 0}\) 是其在 \(p\)-幂索引上的 ghost 子序列。
推论 \ref{cor:witt-dwork-congruence} 的 Dwork 型同余表明该子序列在 \(u\mapsto u^p\) 的 Adams/Frobenius 操作下满足强的 \(p\)-进兼容性：
\[
a_{p^k}(u)-a_{p^{k-1}}(u^p)\in p^k\,\ZZ[u].
\]
因此同步核的 primitive/trace 桥包不仅在复解析侧可检验（\(\zeta\)、Euler 乘积），在 \(p\)-进整性侧也携带一条自然的“Frobenius-lift”约束。
在后续若考虑 \(p\)-进方向的插值或对比不同核的“算术指纹”，该同余可作为最基本的可审计输入。
\end{remark}

\begin{proposition}[\texorpdfstring{$p$}{p}-幂 primitive 坐标的显式公式]\label{prop:witt-ppower-primitive-coordinate}
对任意素数 \(p\) 与 \(k\ge 1\)，Big-Witt/Möbius 反演在 \(n=p^k\) 处塌缩为
\[
\boxed{\;
p_{p^k}(u)=\frac{a_{p^k}(u)-a_{p^{k-1}}(u^p)}{p^k}\in\ZZ[u]\; }.
\]
因此推论 \ref{cor:witt-dwork-congruence} 等价于“\(p\)-幂方向的 primitive 坐标 \(p_{p^k}(u)\) 在整系数环内良定义”，并给出一条可审计的 \(p\)-进差分指纹序列 \(\{p_{p^k}(u)\}_{k\ge 0}\)。
\end{proposition}
\begin{proof}
由定义
\(
p_n(u)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}(u^d)
\)。
当 \(n=p^k\) 时，除数仅为 \(p^j\)，且 \(\mu(p^j)=0\)（\(j\ge 2\)），故仅剩 \(d=1\) 与 \(d=p\) 两项，得到
\[
p_{p^k}(u)=\frac{1}{p^k}\bigl(a_{p^k}(u)-a_{p^{k-1}}(u^p)\bigr).
\]
由命题 \ref{prop:witt-necklace-div} 知右端确为 \(\ZZ[u]\) 元素。证毕。
\end{proof}

\begin{lemma}[核版 Dieudonn\'e--Dwork 整性判据（形式幂级数）]\label{lem:witt-dieudonne-dwork}
设 \(R\) 为无 \(p\)-挠的 \(p\)-进完备环（例如 \(R=\ZZ_p[u]\)），并定义 Frobenius 提升算子
\[
\varphi:\ R[[T]]\to R[[T]],\qquad
\varphi\!\left(\sum_{n\ge 0}c_n(u)T^n\right):=\sum_{n\ge 0}c_n(u^p)\,T^{pn}.
\]
若 \(F(T)\in 1+T\,R\otimes_{\ZZ_p}\QQ_p[[T]]\) 满足
\[
\boxed{\;
\frac{\varphi(F(T))}{F(T)^p}\in 1+pT\,R[[T]]\;},
\]
则必有 \(F(T)\in 1+T\,R[[T]]\)（即 \(F\) 的全部系数 \(p\)-整）。
\end{lemma}
\begin{proof}
这是 \(p\)-典型 Witt/Artin--Hasse 理论中常用的整性判据（Dieudonn\'e--Dwork 形式）。
在本文所需形式下，可用如下思路证明：对 \(F\) 取 \(p\)-进对数 \(\log F\)（其在 \(T\)-进拓扑下收敛），将条件改写为
\(
\varphi(\log F)-p\log F\in pT\,R[[T]]
\)；
再按 \(T\) 的次数作归纳，可逐次推出 \(\log F\) 的系数均在 \(R\) 中，从而 \(F=\exp(\log F)\) 亦在 \(1+T\,R[[T]]\) 内。证毕。
\end{proof}

\begin{theorem}[核版 Dwork 引理：\texorpdfstring{$p$}{p}-典型指数因子的 \(p\)-整性]\label{thm:witt-dwork-zp-integrality}
固定素数 \(p\)。定义
\[
S_p(T;u):=\sum_{k\ge 0}\frac{a_{p^k}(u)}{p^k}\,T^{p^k}\in \QQ_p[u][[T]],
\qquad
Z_p(T;u):=\exp\!\bigl(S_p(T;u)\bigr).
\]
则有
\[
\boxed{\;
Z_p(T;u)\in 1+T\,\ZZ_p[u][[T]]\; }.
\]
\end{theorem}
\begin{proof}
取引理 \ref{lem:witt-dieudonne-dwork} 中的 \(R=\ZZ_p[u]\)。
对 \(S_p\) 施加 \(\varphi\) 得
\[
\varphi(S_p)(T;u)
=\sum_{k\ge 0}\frac{a_{p^k}(u^p)}{p^k}\,T^{p^{k+1}}
=\sum_{k\ge 1}\frac{a_{p^{k-1}}(u^p)}{p^{k-1}}\,T^{p^{k}}.
\]
另一方面
\[
pS_p(T;u)
=p\,a_1(u)\,T+\sum_{k\ge 1}\frac{a_{p^k}(u)}{p^{k-1}}\,T^{p^k}.
\]
因此
\[
\varphi(S_p)(T;u)-pS_p(T;u)
=-p\,a_1(u)\,T+\sum_{k\ge 1}\frac{a_{p^{k-1}}(u^p)-a_{p^k}(u)}{p^{k-1}}\,T^{p^k}.
\]
由推论 \ref{cor:witt-dwork-congruence}，对 \(k\ge 1\) 有
\(
a_{p^k}(u)-a_{p^{k-1}}(u^p)\in p^k\ZZ[u]
\)，
故每一项
\(
\frac{a_{p^{k-1}}(u^p)-a_{p^k}(u)}{p^{k-1}}
\)
均在 \(p\,\ZZ_p[u]\) 中。又 \(-p a_1(u)T\in pT\,\ZZ_p[u][[T]]\)，从而
\[
\varphi(S_p)-pS_p\in pT\,\ZZ_p[u][[T]].
\]
令 \(Z_p=\exp(S_p)\)，则
\[
\frac{\varphi(Z_p)}{Z_p^p}
=\exp\!\bigl(\varphi(S_p)-pS_p\bigr)\in 1+pT\,\ZZ_p[u][[T]].
\]
由引理 \ref{lem:witt-dieudonne-dwork} 得 \(Z_p\in 1+T\,\ZZ_p[u][[T]]\)。证毕。
\end{proof}

\begin{proposition}[unit-root 极限不变量：\texorpdfstring{$u^p=u$}{u^p=u} 时 \texorpdfstring{$a_{p^k}(u)$}{a_{p^k}(u)} 在 \texorpdfstring{$\ZZ_p$}{Z_p} 中收敛]\label{prop:witt-unit-root-limit}
令 \(p\) 为奇素数。若 \(u\) 满足 \(u^p=u\)（例如 \(u\in\mu_{p-1}\subset\ZZ_p^\times\) 的 Teichm\"uller 单位），则序列 \(\{a_{p^k}(u)\}_{k\ge 0}\) 在 \(\ZZ_p\) 中为 Cauchy 序列，并收敛到极限
\[
\boxed{\;
A_p(u):=\lim_{k\to\infty}a_{p^k}(u)\in \ZZ_p\; }.
\]
并且有显式 \(p\)-进误差界
\[
v_p\!\bigl(a_{p^k}(u)-a_{p^{k-1}}(u)\bigr)\ge k\qquad(k\ge 1).
\]
\end{proposition}
\begin{proof}
在条件 \(u^p=u\) 下，推论 \ref{cor:witt-dwork-congruence} 退化为
\(
a_{p^k}(u)\equiv a_{p^{k-1}}(u)\ (\mathrm{mod}\ p^k)
\)，
即 \(a_{p^k}(u)-a_{p^{k-1}}(u)\in p^k\ZZ_p\)，从而给出上式的 \(p\)-进赋值界。
因此对任意 \(r>s\)，有
\[
a_{p^r}(u)-a_{p^s}(u)=\sum_{k=s+1}^{r}\bigl(a_{p^k}(u)-a_{p^{k-1}}(u)\bigr)\in p^{s+1}\ZZ_p,
\]
故为 Cauchy 序列并收敛于 \(\ZZ_p\)。证毕。
\end{proof}

\begin{remark}[unit-root 极限与自对偶的相容性]\label{rem:witt-unit-root-self-dual}
若进一步使用命题 \ref{prop:trace-palindrome} 的回文关系 \(a_n(u)=u^n a_n(1/u)\)，则在 \(u^p=u\) 的 Teichm\"uller 情形有 \(u^{p^k}=u\)，从而
\[
a_{p^k}(u)=u\,a_{p^k}(1/u).
\]
令 \(k\to\infty\) 并用命题 \ref{prop:witt-unit-root-limit} 的收敛性，可得极限不变量满足
\[
\boxed{\ A_p(u)=u\,A_p(1/u)\ }.
\]
这给出 unit-root 指纹在自对偶对称下的一个可审计约束。
\end{remark}

\subsubsection{同步核的自对偶对称}\label{app:kernel-self-dual}
对同步核存在状态对合 $\iota$，其作用为
\[
\iota:\quad
000\leftrightarrow 101,\quad
001\leftrightarrow 100,\quad
002\leftrightarrow 010,\quad
0\overline{1}2\leftrightarrow 11\overline{1},\quad
1\overline{1}2\leftrightarrow 01\overline{1}.
\]
令 $\Pi$ 为该置换的矩阵，则
\[
\Pi^{-1}B_0\Pi=B_1,\qquad \Pi^{-1}B_1\Pi=B_0,
\]
从而
\[
\Pi^{-1}B(u)\Pi=uB(1/u).
\]
于是
\[
\Delta(z,u)=\det(I-zB(u))=\det(I-uzB(1/u))=\Delta(uz,1/u).
\]

\begin{proposition}[自对偶诱导的 \texorpdfstring{$(1+u)$}{(1+u)}--\texorpdfstring{$(1-u)$}{(1-u)} 两参数正常形]\label{prop:self-dual-normal-form-1pmu}
设 \(B(u)=B_0+uB_1\) 且自对偶置换矩阵 \(\Pi\) 满足 \(\Pi^{-1}B_0\Pi=B_1\)。
令
\[
V_+:=\ker(\Pi-I),\qquad V_-:=\ker(\Pi+I),
\qquad V=V_+\oplus V_-,
\]
并把 \(B_0\) 在分解 \(V_+\oplus V_-\) 下写成分块算子
\[
B_0=
\begin{pmatrix}
X & Y\\
Z & W
\end{pmatrix}.
\]
则对所有 \(u\in\CC\) 有
\[
\boxed{
B(u)=B_0+u\Pi^{-1}B_0\Pi
=
\begin{pmatrix}
(1+u)X & (1-u)Y\\
(1-u)Z & (1+u)W
\end{pmatrix}.
}
\]
在本同步核上，\(\Pi\) 为五个互换的直和，故 \(\dim V_+=\dim V_-=5\)。
\end{proposition}
\begin{proof}
在分解 \(V=V_+\oplus V_-\) 下，\(\Pi\) 作用为 \(\Pi|_{V_+}=+I,\ \Pi|_{V_-}=-I\)，故对任意分块算子 \(T=\bigl(\begin{smallmatrix}X&Y\\ Z&W\end{smallmatrix}\bigr)\) 有
\(
\Pi^{-1}T\Pi=\bigl(\begin{smallmatrix}X&-Y\\ -Z&W\end{smallmatrix}\bigr)
\)。

\end{proof}

\begin{proposition}[表示级自对偶：一维角色扭曲的函数方程]\label{prop:kernel-self-dual-character}
沿用 \S\ref{app:kernel-self-dual} 的记号。设 \(G\) 为有限阿贝尔群，\(\chi:G\to\CC^\times\) 为一维角色，且假设群标签只依赖输出位 \(e\in\{0,1\}\)：对 \(e=0\) 取标签为单位元、对 \(e=1\) 取某固定元 \(g_1\in G\)。
则相应的角色扭曲矩阵可写成
\[
B_\chi(u)=B_0+u\,\chi(g_1)\,B_1.
\]
在自对偶对称 \(\Pi^{-1}B_0\Pi=B_1,\ \Pi^{-1}B_1\Pi=B_0\) 下，有相似恒等式
\[
\boxed{\;
\Pi^{-1}B_\chi(u)\Pi
=u\,\chi(g_1)\,B_{\chi^{-1}}(1/u)\; }.
\]
因此行列式与 \(\zeta\)-函数满足表示级函数方程
\[
\boxed{\;
\Delta_\chi(z,u):=\det(I-zB_\chi(u))
=\Delta_{\chi^{-1}}\!\bigl(u\,\chi(g_1)\,z,\ 1/u\bigr)\; },
\]
\[
\boxed{\;
\zeta_\chi(z,u):=\Delta_\chi(z,u)^{-1}
=\zeta_{\chi^{-1}}\!\bigl(u\,\chi(g_1)\,z,\ 1/u\bigr)\; }.
\]
\end{proposition}
\begin{proof}
由 \(\Pi^{-1}B_0\Pi=B_1,\ \Pi^{-1}B_1\Pi=B_0\) 得
\[
\Pi^{-1}B_\chi(u)\Pi
=B_1+u\,\chi(g_1)\,B_0
=u\,\chi(g_1)\Bigl(B_0+\frac{1}{u}\chi(g_1)^{-1}B_1\Bigr)
=u\,\chi(g_1)\,B_{\chi^{-1}}(1/u).
\]
对两边取 \(\det(I-z\,\cdot)\) 并用 \(\det(I-z\alpha A)=\det(I-(\alpha z)A)\) 即得所示行列式函数方程；\(\zeta\) 版本由取倒数得到。证毕。
\end{proof}

\begin{proposition}[周期迹多项式的回文对称]\label{prop:trace-palindrome}
记
\[
a_n(u):=\Tr(B(u)^n)=\sum_{k=0}^{n}a_{n,k}u^k.
\]
则对所有 $n\ge 1$ 有
\[
a_n(u)=u^n a_n(1/u)
\quad\Longleftrightarrow\quad
a_{n,k}=a_{n,n-k}.
\]
\end{proposition}

\begin{proof}
由自对偶 $\Pi^{-1}B(u)\Pi=uB(1/u)$ 与迹不变性，
\[
a_n(u)=\Tr(B(u)^n)=\Tr\!\bigl((uB(1/u))^n\bigr)=u^n\Tr(B(1/u)^n)=u^n a_n(1/u),
\]
比较系数即得结论。证毕。
\end{proof}

\begin{corollary}[奇阶消失律]\label{cor:trace-odd-vanish}
若 $n$ 为奇数，则
\[
a_n(-1)=0,
\qquad
(u+1)\mid a_n(u)\ \ \text{于}\ \mathbb{Z}[u].
\]
\end{corollary}
\begin{proof}

\end{proof}

\begin{corollary}[周期层的中心矩恒等式：奇阶严格为零]\label{cor:trace-central-moments-zero}
沿用命题 \ref{prop:trace-palindrome} 的记号。对任意整数 \(m\ge 0\)，都有中心矩恒等式
$$
\boxed{\ \sum_{k=0}^{n}\left(k-\frac{n}{2}\right)^{2m+1}a_{n,k}=0\ }\qquad(\text{于 }\QQ).
$$
特别地，取 \(m=0\) 得
$$
\sum_{k=0}^{n}k\,a_{n,k}=\frac{n}{2}\sum_{k=0}^{n}a_{n,k}=\frac{n}{2}\,a_n(1).
$$
若进一步把系数正规化为 \(\pi^{\mathrm{per}}_{n,k}:=a_{n,k}/a_n(1)\)（当 \(a_n(1)\neq 0\)），则 \(\{\pi^{\mathrm{per}}_{n,k}\}_k\) 关于 \(n/2\) 精确对称，所有奇阶中心矩严格为零。
\end{corollary}
\begin{proof}
由命题 \ref{prop:trace-palindrome} 有 \(a_{n,k}=a_{n,n-k}\)。
对任意函数 \(f\) 满足 \(f(x)=-f(-x)\)（奇函数），把求和按 \(k\leftrightarrow n-k\) 配对得
$$
\sum_{k=0}^{n} f\!\left(k-\frac{n}{2}\right)a_{n,k}
=\frac12\sum_{k=0}^{n}\Bigl(
f\!\left(k-\frac{n}{2}\right)a_{n,k}
f\!\left((n-k)-\frac{n}{2}\right)a_{n,n-k}
\Bigr)=0,
$$
因为 \((n-k)-\frac{n}{2}=-(k-\frac{n}{2})\) 且 \(a_{n,n-k}=a_{n,k}\)。
取 \(f(x)=x^{2m+1}\) 即得所示恒等式；\(m=0\) 的线性式为其特例。证毕。
\end{proof}

\subsubsection{自对偶 \texorpdfstring{$\Delta(z,u)=\Delta(uz,1/u)$}{Delta(z,u)=Delta(uz,1/u)} 的 s->1-s 函数方程}\label{app:kernel-functional-equation}
本小节把二变量自对偶对称压缩为 Dirichlet 变量下的一条单变量函数方程，从而与经典 $\zeta(s)$ 叙事中的“完成化/函数方程/临界线”三件套对齐。

\begin{proposition}[完成化行列式的函数方程]\label{prop:kernel-xi-functional-equation}
设同步核满足自对偶恒等式 $\Delta(z,u)=\Delta(uz,1/u)$（见 \S\ref{app:kernel-self-dual}），并取
\[
\lambda:=\rho(B(1))>1.
\]
定义沿曲线
\[
z(s):=\lambda^{-s},\qquad u(s):=\lambda^{2s-1}
\]
的单变量函数
\[
\Xi(s):=\Delta(z(s),u(s))=\Delta(\lambda^{-s},\lambda^{2s-1}).
\]
则有严格函数方程
\[
\boxed{\ \Xi(s)=\Xi(1-s)\ }.
\]
等价地，定义 $\zeta$ 的同一截面
\[
\zeta_{\mathrm{sd}}(s):=\zeta(z(s),u(s))=\Delta(\lambda^{-s},\lambda^{2s-1})^{-1},
\]
则 $\zeta_{\mathrm{sd}}(s)=\zeta_{\mathrm{sd}}(1-s)$。
\end{proposition}
\begin{proof}
由自对偶恒等式，
\[
\Xi(s)=\Delta(\lambda^{-s},\lambda^{2s-1})
=\Delta(\lambda^{2s-1}\lambda^{-s},\lambda^{1-2s})
=\Delta(\lambda^{s-1},\lambda^{1-2s}).
\]
而对 $1-s$ 有
\(
z(1-s)=\lambda^{-(1-s)}=\lambda^{s-1}
\)
且
\(
u(1-s)=\lambda^{2(1-s)-1}=\lambda^{1-2s}
\)，
故 $\Xi(s)=\Delta(z(1-s),u(1-s))=\Xi(1-s)$。证毕。
\end{proof}

\begin{remark}[临界线与对称固定点]\label{rem:kernel-critical-line-u1}
由定义 $u(s)=\lambda^{2s-1}$ 可见：直线 $\Re(s)=\tfrac12$ 满足 $|u(s)|=1$，并且 $s=\tfrac12$ 为对称固定点（对应 $u=1$）。
因此在该完成化截面上，“临界线”自然等价为“自对偶对称面/单位圆权重”。
\end{remark}

\begin{definition}[核 Hardy 函数：临界线实值化]\label{def:kernel-hardy-function}

Z(t):=\widetilde{\Delta}\!\bigl(\lambda^{-1/2},\,2it\log\lambda\bigr),
\qquad t\in\RR,
$$
并称其为同步核完成化行列式的 Hardy 型实值化。
\end{definition}

\begin{proposition}[Hardy 实值性、偶性与临界线截面恒等式]\label{prop:kernel-hardy-real-even}
对一切实数 \(t\in\RR\)，有
$$
\boxed{\ Z(t)\in\RR,\qquad Z(t)=Z(-t)\ }.
$$
并且若令 \(s=\tfrac12+it\)，则临界线上的截面恒等式成立：
$$
\boxed{\ Z(t)=\Xi\!\left(\tfrac12+it\right)\ }.
$$
\end{proposition}
\begin{proof}
取 \(\theta:=2it\log\lambda\) 与 \(u=e^\theta=\lambda^{2it}\)，则 \(|u|=1\)。

\[
\widetilde{\Delta}(z,\theta)=\Delta(e^{-\theta/2}z,e^\theta)
\]
可得
\[
Z(t)=\widetilde{\Delta}(\lambda^{-1/2},\theta)
=\Delta\!\bigl(e^{-\theta/2}\lambda^{-1/2},\,u\bigr)
=\Delta\!\bigl(\lambda^{-1/2-it},\,\lambda^{2it}\bigr)
=\Delta\!\bigl(\lambda^{-s},\,\lambda^{2s-1}\bigr)
=\Xi(s),
\]
从而得到 \(Z(t)=\Xi(\tfrac12+it)\)。
\par
另一方面，由 \(\Delta(z,u)=\sum_{j=0}^{10}c_j(u)z^j\) 与完成化定义立得

=\sum_{j=0}^{10}\bigl(u^{-j/2}c_j(u)\bigr)\,z^j
=\sum_{j=0}^{10}d_j(u)\,z^j,
\]
其中 \(d_j(u)=u^{-j/2}c_j(u)\)。
由推论 \ref{cor:completion-real-line}，当 \(|u|=1\) 时 \(d_j(u)\in\RR\)。

\par
偶性由完成化偶性 \(\widetilde{\Delta}(z,-\theta)=\widetilde{\Delta}(z,\theta)\)（\S\ref{app:kernel-completion}）直接推出：把 \(\theta=2it\log\lambda\) 代入得 \(Z(-t)=Z(t)\)。证毕。
\end{proof}

\subsubsection{压力函数的解析性与显式导数}\label{app:pressure-analytic}
对 $u=e^\theta>0$，矩阵 $B(u)$ 仍为 primitive，故 Perron 根 $\lambda(u)$ 在 $u>0$ 上为单根并随 $u$ 实解析。记 $\lambda=\lambda(u)$，其满足六次方程
\[
\begin{aligned}
F(\lambda,u)

\begin{theorem}[prime-local CLT：能量层的高斯局部极限与显式前因子]\label{thm:prime-local-clt}

  \item 若 \(n\) 为偶数，则 \(\widehat p_n(s)\) 为 \(s\) 的偶多项式；
  \item 若 \(n\) 为奇数，则 \(\widehat p_n(s)\) 为 \(s\) 的奇多项式。
\end{enumerate}
进一步，对任意整数 \(d\ge 1\) 定义
\[
C_d(s):=r^d+r^{-d}\in\ZZ[s],
\qquad
C_d(s)=2T_d(s/2)
\]
（\(T_d\) 为第一类 Chebyshev 多项式）。则完成化坐标下的幂替换满足严格的“倍角律”：
\[
r^{-dn}\,p_n(r^{2d})=\widehat p_n\!\bigl(C_d(s)\bigr),
\]
即 Witt/Möbius 反演中的 \(u\mapsto u^d\) 在完成化后等价于 \(s\mapsto C_d(s)\) 的 Chebyshev--Adams 运算。
\end{corollary}

\begin{corollary}[奇长度的严格奇偶平衡：\texorpdfstring{$p_n(-1)=0$}{p_n(-1)=0}]\label{cor:primitive-odd-parity-balance}
若 \(n\ge 3\) 为奇数，则由推论 \ref{cor:primitive-odd-div} 的整除律 \(u(u+1)\mid p_n(u)\) 得
\[
\boxed{\ p_n(-1)=\sum_{k\ge 0}(-1)^k p_{n,k}=0\ }.
\]
因此奇长度 primitive 轨道在“能量奇偶”两侧的计数严格相等（以符号加权意义表述）。
\end{corollary}
\begin{proof}
由系数镜像对称 \(p_{n,k}=p_{n,n-k}\) 可得（见上文）
\(
p_n(u)=u^n p_n(1/u)
\)。
令 \(u=r^2\) 得
\(
p_n(r^2)=r^{2n}p_n(r^{-2})
\)，
从而 \(r^{-n}p_n(r^2)\) 在 \(r\mapsto r^{-1}\) 下不变。由不变量子环恒等式
\(
\ZZ[r,r^{-1}]^{\,r\mapsto r^{-1}}=\ZZ[r+r^{-1}]=\ZZ[s]
\)
即得 \(\widehat p_n(s)\in\ZZ[s]\)。
\par
奇偶性来自指数奇偶：\(p_n(r^2)\) 只含 \(r^{2k}\)，乘上 \(r^{-n}\) 后幂指数与 \(n\) 同奇偶；而 \(\ZZ[s]\) 中 \(s\mapsto -s\) 的奇偶分解与 \(r\mapsto -r\) 的幂奇偶一致，故得到 \(\widehat p_n\) 的奇偶性结论。
\par
最后令 \(s=r+r^{-1}\)，则
\(
C_d(s)=r^d+r^{-d}
\)
满足递推 \(C_{d+1}=sC_d-C_{d-1}\)，因而 \(C_d\in\ZZ[s]\)，并与 Chebyshev 多项式关系 \(C_d(s)=2T_d(s/2)\) 等价。代入 \(u=r^2\) 及 \(u^d=r^{2d}\) 得
\[
r^{-dn}p_n(r^{2d})=\widehat p_n(r^d+r^{-d})=\widehat p_n(C_d(s)),
\]
即所述倍角律。证毕。
\end{proof}

\begin{corollary}[完成化的 Chebyshev--Witt 变换：\texorpdfstring{$\widehat a\leftrightarrow \widehat p$}{hat a <-> hat p}]\label{cor:chebyshev-witt-completed}
在同一完成化坐标 \(u=r^2,s=r+r^{-1}\) 下，定义 completed ghost（迹）多项式
\[
\widehat a_n(s):=r^{-n}a_n(r^2)=r^{-n}\Tr(B(r^2)^n)\in\ZZ[s].
\]
则对任意 \(n\ge 1\) 有严格恒等式
\[
\boxed{\ 
\widehat p_n(s)=\frac1n\sum_{d\mid n}\mu(d)\,\widehat a_{n/d}\!\bigl(C_d(s)\bigr),
\qquad
\widehat a_n(s)=\sum_{d\mid n}d\,\widehat p_d\!\bigl(C_{n/d}(s)\bigr)
\ }.
\]
\end{corollary}
\begin{proof}
对 \(u=r^2\) 的 Witt/Möbius 反演
\(
p_n(u)=\frac1n\sum_{d\mid n}\mu(d)a_{n/d}(u^d)
\)
两侧同乘 \(r^{-n}\)，并用完成化倍角律
\(
r^{-dn}f(r^{2d})=\widehat f(C_d(s))
\)
（对 \(f=a_{n/d}\) 与 \(f=p_d\) 分别应用）即可得到第一式；第二式同理由
\(
a_n(u)=\sum_{d\mid n} d\,p_d(u^{n/d})
\)
完成化得到。
\end{proof}

\begin{corollary}[素数长度同余：\texorpdfstring{$\widehat a_p(s)\equiv C_p(s)\ (\mathrm{mod}\ p)$}{hat a_p congruent Cp mod p}]\label{cor:completed-prime-congruence}
沿用推论 \ref{cor:chebyshev-witt-completed} 的完成化记号。若 \(p\) 为素数，则有整系数同余
\[
\boxed{\ \widehat a_p(s)\equiv C_p(s)\pmod p\ }.
\]
\end{corollary}
\begin{proof}
对素数 \(p\)，由 \(\widehat p_p(s)=\frac1p\bigl(\widehat a_p(s)-\widehat a_1(C_p(s))\bigr)\)（推论 \ref{cor:chebyshev-witt-completed} 的第一式，因子仅为 \(d=1,p\)）可知
\[
\widehat a_p(s)-\widehat a_1(C_p(s))\in p\,\ZZ[s].
\]
而在本核上 \(a_1(u)=\Tr(B(u))=p_1(u)=1+u\)，故
\(
\widehat a_1(s)=r^{-1}(1+r^2)=r+r^{-1}=s
\)，从而 \(\widehat a_1(C_p(s))=C_p(s)\)。
代回即得同余。证毕。
\end{proof}
