\begin{conclusion}[例外幂和的两项主导展开与显式余项界]\label{con:xi-terminal-replica-softcore-exceptional-moments-exponential-hierarchy}
取整数 \(m\ge 3\)，并定义
\[
A_{m,0}:=2^{m},\qquad
A_{m,1}:=3\cdot 2^{m-2},\qquad
A_{m,2}:=5\cdot 2^{m-3}.
\]
则对一切 \(q\ge 2\) 存在严格分解
\[
\boxed{\;
S_m(q)=2^{-m}\Big(A_{m,0}^{q}+mA_{m,1}^{q}+\Omega_m(q)+R_m(q)\Big)\;}
\]
其中余项 \(R_m(q)\) 满足显式界
\[
0\le R_m(q)\le (2^{m}-m-2)\,A_{m,2}^{q}.
\]
从而得到精确的相对误差展开
\[
\frac{S_m(q)}{2^{m(q-1)}}=
1+m\left(\frac34\right)^{q}
\frac{\Omega_m(q)}{2^{mq}}
O_m\!\left(\left(\frac58\right)^{q}\right),
\qquad (q\to\infty),
\]
其中 \(O_m\) 的隐常数可取为 \(2^{m}-m-2\)（由上式余项界直接给出）。
\end{conclusion}
\begin{proof}
由结论 \ref{con:xi-terminal-replica-softcore-exceptional-power-sum-word-trace-sympower}，
\[
2^mS_m(q)=\Omega_m(q)+\sum_{\substack{w\in\{J,K\}^m\\ w\neq K^m}}\bigl(\Tr M(w)\bigr)^q,
\qquad M(w)=A_1\cdots A_m.
\]
将求和按字母 \(K\) 的出现次数 \(r=\#\{t:A_t=K\}\) 分层。
\par
\textbf{(i) \(r=0\).}
仅有字词 \(w=J^m\)，且 \(\Tr(J^m)=2^m=A_{m,0}\)。
\par
\textbf{(ii) \(r=1\).}
恰有 \(m\) 个字词，为 \(KJ^{m-1}\) 的循环移位；由结论 \ref{con:xi-terminal-replica-softcore-fixed-zero-count-trace-extremal}（取 \(r=1\)）知该层词迹恒为
\(\Tr M(w)=3\cdot 2^{m-2}=A_{m,1}\)。
\par
\textbf{(iii) \(2\le r\le m-1\).}
同样由结论 \ref{con:xi-terminal-replica-softcore-fixed-zero-count-trace-extremal} 得
\[
\Tr M(w)\le F_{r+3}\,2^{m-r-1}.
\]
若 \(r=2\)，则右端为 \(F_{5}\,2^{m-3}=5\cdot 2^{m-3}=A_{m,2}\)。
若 \(r\ge 3\)，由递推 \(F_{n+1}\le 2F_n\)（\(n\ge 1\)）得粗界 \(F_{r+3}\le 2^{r}\)，从而
\[
F_{r+3}\,2^{m-r-1}\le 2^{r}\,2^{m-r-1}=2^{m-1}<5\cdot 2^{m-3}=A_{m,2}.
\]
因此对所有 \(2\le r\le m-1\) 均有 \(\Tr M(w)\le A_{m,2}\)。
该部分字词总数为 \(2^m-2-m\)（从全部 \(2^m\) 个字词中剔除 \(K^m\)、\(J^m\) 与 \(r=1\) 的 \(m\) 个字词），因此令
\[
R_m(q):=\sum_{\substack{w\in\{J,K\}^m\\ w\neq K^m,\ r(w)\ge 2}}\bigl(\Tr M(w)\bigr)^q
\]
即得 \(0\le R_m(q)\le (2^m-m-2)A_{m,2}^q\)。
把三层贡献与 \(\Omega_m(q)\) 合并并除以 \(2^m\)，得到所示严格分解。
\par
最后，将分解式两侧除以 \(2^{m(q-1)}=2^{-m}A_{m,0}^q\) 得到
\[
\frac{S_m(q)}{2^{m(q-1)}}=
1+m\left(\frac{A_{m,1}}{A_{m,0}}\right)^q
\frac{\Omega_m(q)}{2^{mq}}
\frac{R_m(q)}{2^{mq}},
\]
其中 \(\frac{A_{m,1}}{A_{m,0}}=\frac34\) 且 \(\frac{R_m(q)}{2^{mq}}\le (2^m-m-2)\left(\frac{A_{m,2}}{A_{m,0}}\right)^q=(2^m-m-2)\left(\frac58\right)^q\)，从而得到相对误差展开。证毕。
\end{proof}

\begin{conclusion}[全迹矩的两项主导展开与显式余项界]\label{con:xi-terminal-replica-softcore-trace-moments-exponential-hierarchy}
保持结论 \ref{con:xi-terminal-replica-softcore-exceptional-moments-exponential-hierarchy} 的记号（取 \(m\ge 3\) 并沿用 \(A_{m,0},A_{m,1},A_{m,2}\) 与余项 \(R_m(q)\)）。
则对一切 \(q\ge 2\) 成立严格分解
\[
\boxed{\;
\Tr\!\bigl((T^{(q)})^m\bigr)
=2^{-m}\Big(A_{m,0}^{q}+mA_{m,1}^{q}+L_m^{q}+R_m(q)\Big)\;}
\]
并且余项界
\(
0\le R_m(q)\le (2^{m}-m-2)\,A_{m,2}^{q}
\)
仍然成立。
因此
\[
\frac{\Tr((T^{(q)})^m)}{2^{m(q-1)}}
=1+m\left(\frac34\right)^{q}
\left(\frac{L_m}{2^{m}}\right)^{q}
O_m\!\left(\left(\frac58\right)^{q}\right),
\qquad (q\to\infty),
\]
其中 \(O_m\) 的隐常数可取为 \(2^{m}-m-2\)。
\end{conclusion}
\begin{proof}[证明要点]
由结论 \ref{con:xi-terminal-replica-softcore-trace-exceptional-difference}，
\(
\Tr((T^{(q)})^m)=S_m(q)+2^{-m}(L_m^q-\Omega_m(q))
\)；
将结论 \ref{con:xi-terminal-replica-softcore-exceptional-moments-exponential-hierarchy} 的
\(
S_m(q)=2^{-m}(A_{m,0}^q+mA_{m,1}^q+\Omega_m(q)+R_m(q))
\)
代入即可抵消 \(\Omega_m(q)\)，得到所示分解与同一余项界；相对误差展开由两侧同除以 \(2^{m(q-1)}\) 并代入 \(A_{m,1}/A_{m,0}=3/4\)、\(A_{m,2}/A_{m,0}=5/8\) 得出。证毕。
\end{proof}
