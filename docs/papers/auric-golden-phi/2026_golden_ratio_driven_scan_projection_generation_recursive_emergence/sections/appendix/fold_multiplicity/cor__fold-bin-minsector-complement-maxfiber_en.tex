\begin{corollary}[Cross-model identity: dual-scale locking between minimal degeneracy sector complement size and Fold maximal fiber]\label{cor:fold-bin-minsector-complement-maxfiber-audited}
On the audited window \(m\in\{6,8,10,12\}\), let \(\mathcal{S}_{m,d}\subseteq X_m\) be the degeneracy spectral sectors of bin-fold with minimal degeneracy \(d_{\min}(m)\) as in Definition \ref{def:fold-bin-degeneracy-sectors}.
Denote the maximal fiber cardinality of Fold at level \(n\) by
\(
D_n:=\max_{x\in X_n} d_n(x)
\)
(see Theorem \ref{thm:pom-max-fiber}).
Then the exact identity holds
\[
\bigl|X_m\setminus \mathcal{S}_{m,d_{\min}(m)}\bigr| = D_{2m-2}.
\]
\end{corollary}
\begin{proof}
By Theorem \ref{thm:fold-bin-min-degeneracy-fib-audited}, on the above \(m\) we have \(\bigl|\mathcal{S}_{m,d_{\min}(m)}\bigr|=|X_{m-2}|\).
Moreover, the stable type counting law gives \(|X_n|=F_{n+2}\), thus
\[
\bigl|X_m\setminus \mathcal{S}_{m,d_{\min}(m)}\bigr|
=|X_m|-|X_{m-2}|
=F_{m+2}-F_m
=F_{m+1}.
\]
On the other hand, \(2m-2=2(m-1)\) is even; by Theorem \ref{thm:pom-max-fiber} we obtain
\(
D_{2m-2}=F_{(m-1)+2}=F_{m+1}
\).
The two expressions are equal, completing the proof.
\end{proof}

\begin{corollary}[Non-infinite-extendability of minimal degeneracy Fibonacci law]\label{cor:fold-bin-min-degeneracy-fib-nonpersistent}
Adopting the notation of Definition \ref{def:fold-bin-degeneracy-sectors},
if for infinitely many sufficiently large even \(m\) the identity
\[
d_{\min}(m)=F_{m/2}
\]
holds simultaneously, then this contradicts the total preimage counting identity \(|\Omega_m|=2^m\).
Therefore, the audited equation \(d_{\min}(m)=F_{m/2}\) of Theorem \ref{thm:fold-bin-min-degeneracy-fib-audited} cannot be unconditionally extended as a law "holding for all \(m\ge 6\)" on the even window.
\end{corollary}
\begin{proof}
By definition, the fiber sizes of bin-fold satisfy
\[
2^m
=|\Omega_m|
=\sum_{w\in X_m}\bigl|(\Fold_m^{\mathrm{bin}})^{-1}(w)\bigr|
\ge |X_m|\,d_{\min}(m).
\]
Moreover, the stable type counting law gives \(|X_m|=F_{m+2}\), so if \(d_{\min}(m)=F_{m/2}\) then
\[
2^m\ \ge\ F_{m+2}\,F_{m/2}.
\]
By the exponential asymptotics \(F_n\asymp \varphi^n\) of Fibonacci numbers, we have \(F_{m+2}F_{m/2}\asymp \varphi^{3m/2}\).
Note that \(\varphi^3=2+\sqrt5>4\), thus \(\varphi^{3/2}>2\), so there exists \(m_0\) such that for all even \(m\ge m_0\) we have \(F_{m+2}F_{m/2}>2^m\), contradicting the above inequality. This completes the proof.
\end{proof}

\begin{corollary}[Minimal sector density limit under subcovering isomorphism extrapolation]\label{cor:fold-bin-minsector-density-phi-minus2}
Adopting the notation of Definition \ref{def:fold-bin-degeneracy-sectors},
if the "subcovering isomorphism" identity
\(
\bigl|\mathcal{S}_{m,d_{\min}(m)}\bigr|=|X_{m-2}|
\)
holds for a sequence of even \(m\) tending to infinity, then the limit exists
\[
\lim_{\substack{m\to\infty\\ m\ \mathrm{even}}}
\frac{\bigl|\mathcal{S}_{m,d_{\min}(m)}\bigr|}{|X_m|}
\;=\;
\varphi^{-2}.
\]
\end{corollary}
\begin{proof}
On the above \(m\),
\(
\frac{|\mathcal{S}_{m,d_{\min}(m)}|}{|X_m|}=\frac{|X_{m-2}|}{|X_m|}=\frac{F_m}{F_{m+2}}
\).
From \(F_{m+2}\sim \varphi^2 F_m\), the limit is \(\varphi^{-2}\). This completes the proof.
\end{proof}

\begin{corollary}[Degeneracy spectral multivaluedness forces non-free action: free group action and linear quotient obstruction]\label{cor:foldbin6-degeneracy-multivalued-nonfree}
At \(m=6\), the bin-fold fiber degeneracy histogram is \(2:8,3:4,4:9\) (Remark \ref{rem:fold-bin-gauge-volume-m6}), so the fiber sizes take multiple values within the same window.
Therefore, the equivalence relation
\(
\omega\sim\eta\iff \Fold^{\mathrm{bin}}_6(\omega)=\Fold^{\mathrm{bin}}_6(\eta)
\)
cannot be realized by the orbit partition of any free action of a finite group on \(\Omega_6\); nor can it arise from any linear coset partition of an abelian group (Corollary \ref{cor:terminal-foldbin6-no-linear-kernel}).
Thus any scheme interpreting this many-to-one compression as "symmetry/gauge redundancy" must necessarily require variable stabilizers (non-free action) or more general groupoid structures.
\end{corollary}
\begin{proof}
If a finite group \(H\) acts freely on \(\Omega_6\), then any orbit has size identically equal to \(|H|\), so all orbits are equicardinal; this contradicts the degeneracy multivaluedness, hence impossible.
For linear coset partitions, the fibers are also equicardinal; its non-realizability on window-\(6\) is given by Corollary \ref{cor:terminal-foldbin6-no-linear-kernel}. This completes the proof.
\end{proof}
