\section{Optional physics-consensus inputs (interface / matching)}
\label{app:physics_consensus_inputs}

\AuditTag This appendix records optional inputs that align the present protocol language with widely used, data-facing physics conventions.
They are \emph{not} used as premises in the theorem-level folding core.
They may be used as explicitly scoped premises for \emph{extended} interface/audit modules that are marked as such (e.g.\ Appendix~\ref{app:bh_wormholes_pointer} and Appendix~\ref{app:bh_planck_capacity_calibration}).
Their role is to provide a transparent, audit-friendly interface/matching route for readers who prefer to treat the Standard Model and low-energy gravity as empirical effective-theory constraints.

\begin{assumption}[P1 (Low-energy EFT consensus: SM + GR as effective theories)]
\label{ass:consensus_sm_gr_eft}
\MatchTag On experimentally accessible energy scales, observed dynamics is described by a local Lorentz-invariant effective field theory (EFT).
The non-gravitational sector is captured by the Standard Model gauge theory with its established chiral matter content and quantum-number conventions, while gravity is captured by Einstein gravity at leading order in a low-energy derivative expansion (with higher-curvature terms treated as EFT corrections).
\end{assumption}

\begin{assumption}[P2 (Empirical three-factor gauge structure at the interface)]
\label{ass:consensus_three_factor_gauge}
\MatchTag On the physical identification layer, the observed non-gravitational interactions organize into three independent gauge sectors.
Equivalently, in the standard continuum dictionary one may model the connected gauge redundancy as a compact three-factor product with one abelian factor and two non-abelian simple factors,
\[
G_{\mathrm{gauge}}\ \cong\ U(1)\times G_2\times G_3,
\]
where $G_2$ and $G_3$ are compact, connected, simple, and non-isomorphic.
In the unbroken electroweak regime, the gauge-boson multiplicities in these three sectors are $(1,3,8)$, so the corresponding Lie algebras may be identified as
\[
\mathfrak{u}(1)\oplus \mathfrak{su}(2)\oplus \mathfrak{su}(3)
\quad (\text{up to finite quotients at the global-group level}).
\]
This assumption provides an empirical candidate-family route for the three-factor interface dictionary used in Proposition~\ref{prop:channel_to_gauge}.
\end{assumption}

\begin{assumption}[P3 (Matching-scale existence and RG propagation as a matching-layer dictionary)]
\label{ass:consensus_matching_scale_rg}
\MatchTag There exists a matching scale $\mu_\ast$ (together with declared scheme conventions) at which protocol-level outputs can be mapped to EFT boundary data (masses, couplings, thresholds) in the standard dictionary.
Below $\mu_\ast$, the propagation of these boundary data to laboratory scales is governed by standard EFT renormalization-group flow, while residual scheme/threshold dependence is treated as a matching-layer effect rather than as a theorem-level consequence of the folding core.
\end{assumption}

\begin{remark}[Relation to optional microscopic implementations]
\AuditTag Assumptions~\ref{ass:consensus_sm_gr_eft}--\ref{ass:consensus_matching_scale_rg} are \emph{not} claimed to be equivalent to any specific microscopic architecture input.
They are a data-facing interface route: they declare the effective structure one aims to reproduce.
A separate optional route supplies a concrete microscopic organization that can realize this structure (Appendix~\ref{app:internal_fiber_g2_optional}), but no bidirectional equivalence is asserted.
\end{remark}

\subsection{Boundary/Planck information pack (strong-field interface inputs)}
\label{subsec:pphys_bh_planck_pack}

\paragraph{Scope.}
\AuditTag The inputs in this subsection are standard external targets from semiclassical black-hole thermodynamics and information bounds.
They are recorded only to support boundary-capacity calibration statements and strong-field pointer modules in the present paper.
They are not used anywhere in the theorem-level folding core.

\begin{assumption}[PBH1 (Planck area and information units)]
\label{ass:bh_planck_units_info}
\MatchTag Let $\ell_P^2:=G\hbar/c^3$ denote the Planck area scale.
For an entropy $S$ measured in physical units, define the corresponding information content in bits by
\[
I:=\frac{S}{k_B\ln 2}.
\]
\end{assumption}

\begin{assumption}[PBH2 (Bekenstein--Hawking area law; semiclassical)]
\label{ass:bh_area_law}
\MatchTag For a stationary black hole with horizon area $A$, the leading semiclassical entropy is
\[
S_{\mathrm{BH}}=\frac{k_B A}{4\ell_P^2}.
\]
See, e.g., \cite{Bekenstein1973,BardeenCarterHawking1973,Hawking1975,Wald1984,Bousso2002}.
\end{assumption}

\begin{assumption}[PBH3 (Covariant entropy bound and saturation; external)]
\label{ass:bh_covariant_entropy_bound}
\MatchTag Assume the covariant entropy bound in the form
\[
S\le \frac{k_B A}{4\ell_P^2}
\]
for an appropriate light-sheet/screen area $A$, and assume that horizons saturate this bound at leading order.
See, e.g., \cite{Bousso2002}.
\end{assumption}

\begin{assumption}[PBH4 (Bekenstein bound; external)]
\label{ass:bh_bekenstein_bound}
\MatchTag For a system of energy $E$ contained in radius $R$, the entropy satisfies
\[
S\le 2\pi k_B \frac{E R}{\hbar c}.
\]
See, e.g., \cite{Bekenstein1981}.
\end{assumption}

\begin{assumption}[PBH5 (Schwarzschild package and thermodynamic scalings; external)]
\label{ass:bh_schwarzschild_package}
\MatchTag For a Schwarzschild black hole of mass $M$, let $R_s:=2GM/c^2$ denote the Schwarzschild radius and $A:=4\pi R_s^2$ the horizon area.
When used as a thermodynamic reference, the standard semiclassical scalings
$S_{\mathrm{BH}}\propto M^2$ and $T_H\propto 1/M$
are treated as external inputs.
See, e.g., \cite{Hawking1975,Wald1984}.
\end{assumption}

\subsection{Decay/radiation dictionary pack (lifetime/width and thermal proxies)}
\label{subsec:pphys_decay_radiation_pack}

\paragraph{Scope.}
\AuditTag The inputs in this subsection are optional matching-layer dictionaries that relate lifetime/width language and (when desired) thermal-proxy language.
They are used only by extended interface modules that explicitly cite them (e.g.\ Appendix~\ref{app:leakage_kernel} and Appendix~\ref{app:protected_low_leakage_phase}).

\begin{assumption}[PDR1 (Lifetime--width dictionary; external matching convention)]
\label{ass:pdr_lifetime_width}
\MatchTag For a process that is well-modeled by an exponential survival law in physical time,
\[
P(t)=\e^{-\gamma t},
\]
define the lifetime $\tau:=1/\gamma$.
If one additionally uses the standard energy-width convention $\Gamma_E:=\hbar \gamma$, then $\tau=\hbar/\Gamma_E$.
\end{assumption}

\begin{assumption}[PDR2 (Thermal-proxy convention for emission; optional)]
\label{ass:pdr_thermal_proxy}
\MatchTag When an emission channel is approximately thermal in the standard semiclassical sense, one may summarize its intensity by an effective temperature parameter $T_{\mathrm{eff}}$.
This is a reporting convention: it does not assert that the underlying microdynamics is in equilibrium.
\end{assumption}

\subsection{Low-leakage to low-temperature pack (optional matching inputs)}
\label{subsec:pphys_low_leakage_pack}

\paragraph{Scope.}
\AuditTag This subsection records optional bridge assumptions used to translate the tick-trap notion ``low leakage'' into temperature language.
They are not implied by any information-capacity bound alone; they are matching-layer conventions or model choices.

\begin{assumption}[PPL1 (Low leakage implies low effective temperature; matching dictionary)]
\label{ass:ppl_low_leakage_low_temp}
\MatchTag In a declared comparison class, assume that the effective emission temperature proxy is monotone in a leakage/intensity proxy, so that smaller leakage rate corresponds to lower effective temperature at fixed comparison conventions.
\end{assumption}

\begin{assumption}[PPL2 (Gap-suppression model family; optional)]
\label{ass:ppl_gap_suppression_model}
\MatchTag In an optional model family for protected phases, assume a gap-suppressed leakage law of the schematic form
\[
\Gamma(T)\approx \Gamma_0\,\exp\!\left(-\frac{\Delta}{k_B T}\right),
\]
for some positive parameters $(\Gamma_0,\Delta)$ declared by the module that uses this model.
\end{assumption}
