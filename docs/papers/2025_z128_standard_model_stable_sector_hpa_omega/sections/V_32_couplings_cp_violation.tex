
This section records two interface points: (i) coupling constants as geometric normalization costs, and (ii) CP violation as a CP-odd phase-space volume with discrete multiplicity.
We treat the closed expressions below as CAP-closed interface normalizations: they follow from explicitly declared finite candidate families and canonical geometric data, and their mismatch to scheme-/scale-dependent experimental conventions is recorded as a matching-layer factor.
The same normalization philosophy is pursued, in a broader constant-geometry context, in the companion manuscript \cite{Ma2025PCG}.

\subsection{Three-channel impedance and the fine-structure constant}
\label{subsec:alpha_impedance}

The inverse fine-structure constant admits a closed three-stratum geometric impedance expression:
\begin{equation}\label{eq:alpha_geo}
\alpha_{\mathrm{em}}^{-1}{}_{\mathrm{geo}}=4\pi^3+\pi^2+\pi\approx 137.0363037759.
\end{equation}
We make the interface closure declarations explicit.

\paragraph{Tick-first bridge: from periodic phase to geometric normalization cost.}
In the tick-first dictionary (Section~\ref{sec:tick_calculus}), the only primitive input is the sequential update count and the only primitive closure rule is CAP.
Any internal phase variable is therefore a \emph{periodic} datum attached to tick evolution, with the canonical low-complexity period $2\pi$ (circle normalization).
When independent protocol constraints must be satisfied in series, their combined effect is multiplicative at the level of weights and additive after the logarithmic readout map (Proposition~\ref{prop:serial_impedance}).
The role of compact phase spaces and their canonical volumes in the definitions below is to supply protocol-invariant, finite-description normalization costs for such periodic data, compatible with ray-space $\ZZ_2$ identifications intrinsic to finite readout.

\begin{proposition}[Serial composition and logarithmic impedance (interface)]\label{prop:serial_impedance}
If a readout protocol must satisfy independent constraints in a fixed hierarchy (``strata'') in series, assign each stratum $j$ a multiplicative weight $w_j\in(0,1]$ and define its log-cost $V_j:=-\log w_j$ \cite{Shannon1948,CoverThomas2006}.
If the total weight is $w_{\mathrm{tot}}=\prod_j w_j$, then the corresponding impedance is the additive cost
\[
\alpha_{\mathrm{geo}}^{-1}:=-\log w_{\mathrm{tot}}=\sum_j V_j.
\]
\end{proposition}
\begin{proof}
This is the identity $-\log\!\left(\prod_j w_j\right)=\sum_j (-\log w_j)$.
\end{proof}

\begin{definition}[Geometric cost as canonical phase volume (CAP closure form)]\label{def:impedance_volume}
For each stratum $j$, let $\mathcal{M}_j$ be a compact phase space drawn from a declared finite candidate family.
Define the cost as the canonical volume
\[
V_j:=\mathrm{Vol}(\mathcal{M}_j),
\]
where $\mathrm{Vol}$ is computed using the standard bi-invariant metric on compact Lie groups together with induced quotient metrics for projective $\ZZ_2$ identifications (Lemma~\ref{lem:group_volumes}).
The choice of $\mathcal{M}_j$ is treated as a bounded-complexity CAP closure within the stated family (Axiom~\ref{ax:cap} and Definition~\ref{def:bounded_complexity_closure}).
\end{definition}

\begin{remark}[Metric normalization and matching-layer rescaling]\label{rem:metric_normalization_volumes}
Throughout this section we use the standard unit-radius normalizations for compact groups (Lemma~\ref{lem:group_volumes}), so that $SU(2)\cong S^3$ has $\mathrm{Vol}(SU(2))=2\pi^2$.
Any alternative overall metric rescaling on a stratum rescales $\mathrm{Vol}(\mathcal{M}_j)$ by a fixed factor and therefore shifts the corresponding impedance contribution $V_j$ by a matching-layer normalization change rather than introducing additional internal degrees of freedom.
\end{remark}

\begin{definition}[Electromagnetic three-stratum phase spaces (CAP closure form)]\label{def:em_phase_spaces}
For the electromagnetic normalization channel, select the three strata within an explicit finite candidate family built from the primitive set
\[
\mathfrak{P}:=\{U(1),SU(2),SO(3),\mathbb{R}P^1\},
\]
equipped with canonical volumes (Lemma~\ref{lem:group_volumes}) and with projective quotients encoding the ray-space $\ZZ_2$ identification intrinsic to readout.
At the $m=6$ anchor, the CAP-minimal choice yields
\[
\mathcal{M}_{\mathrm{bulk}}\cong U(1)\times SU(2),\qquad
\mathcal{M}_{\mathrm{boundary}}\cong SO(3),\qquad
\mathcal{M}_{\mathrm{line}}\cong \mathbb{R}P^1
\]
(Proposition~\ref{prop:alpha_phase_factorization_rigidity}).
\end{definition}

\begin{remark}[Geometric meaning of the three electromagnetic strata]\label{rem:em_strata_meaning}
The three strata in Definition~\ref{def:em_phase_spaces} are chosen to align with three protocol-level geometric objects that are present once one insists on finite-resolution readout and transport.
The factor $U(1)$ models phase, $SU(2)$ captures the spinorial double cover of rotations, $SO(3)$ encodes the orientation of a local rigid frame, and $\mathbb{R}P^1\cong U(1)/\{\pm 1\}$ models the projective ray identification of a one-dimensional phase degree of freedom \cite{Nakahara2003,HallLieGroups2015}.
The purpose of the definition is not to claim an ontic manifold, but to fix a minimal closed normalization dictionary in which ray-space $\ZZ_2$ quotients are treated explicitly rather than absorbed into conventions.
\end{remark}

\begin{lemma}[Canonical volumes of \texorpdfstring{$U(1)$}{U(1)}, \texorpdfstring{$SU(2)$}{SU(2)}, \texorpdfstring{$SO(3)$}{SO(3)}, and \texorpdfstring{$\mathbb{R}P^1$}{RP1}]\label{lem:group_volumes}
With the standard unit-radius normalizations,
\[
\mathrm{Vol}(U(1))=2\pi,\qquad
\mathrm{Vol}(SU(2))=\mathrm{Vol}(S^3)=2\pi^2.
\]
Moreover, the $\ZZ_2$ quotients satisfy
\[
\mathrm{Vol}(SO(3))=\mathrm{Vol}(SU(2)/\{\pm \ind\})=\pi^2,\qquad
\mathrm{Vol}(\mathbb{R}P^1)=\mathrm{Vol}(U(1)/\{\pm 1\})=\pi.
\]
Consequently, $\mathrm{Vol}(U(1)\times SU(2))=4\pi^3$.
\end{lemma}
\begin{proof}
The first identity is the circumference of the unit circle.
For $SU(2)\cong S^3$, integrating the standard volume form on the unit $3$-sphere gives $\mathrm{Vol}(S^3)=2\pi^2$ \cite{Nakahara2003,HallLieGroups2015}.
The quotient maps $SU(2)\to SO(3)=SU(2)/\{\pm \ind\}$ and $U(1)\to \mathbb{R}P^1=U(1)/\{\pm 1\}$ are two-sheeted Riemannian coverings under the induced quotient metrics, so the quotient volumes are halved.
The product identity follows by multiplicativity of product volumes.
\end{proof}

\begin{proposition}[Primitive factorization rigidity for the electromagnetic strata]\label{prop:alpha_phase_factorization_rigidity}
Fix the primitive candidate set
\[
\mathfrak{P}:=\{U(1),SU(2),SO(3),\mathbb{R}P^1\},
\]
equipped with the canonical volumes of Lemma~\ref{lem:group_volumes}.
Restrict the bulk stratum to be a product of two primitives and restrict the boundary and line strata to be single primitives.
Impose the target scalings
\[
\mathrm{Vol}(\mathcal{M}_{\mathrm{bulk}})=4\pi^3,\qquad
\mathrm{Vol}(\mathcal{M}_{\mathrm{boundary}})=\pi^2,\qquad
\mathrm{Vol}(\mathcal{M}_{\mathrm{line}})=\pi.
\]
Then, up to ordering, the strata are forced as
\[
\mathcal{M}_{\mathrm{bulk}}\cong U(1)\times SU(2),\qquad
\mathcal{M}_{\mathrm{boundary}}\cong SO(3),\qquad
\mathcal{M}_{\mathrm{line}}\cong \mathbb{R}P^1.
\]
\end{proposition}
\begin{proof}
By Lemma~\ref{lem:group_volumes}, the one-factor values in $\mathfrak{P}$ are $2\pi$, $2\pi^2$, $\pi^2$, and $\pi$.
Any two-factor product has the form $2^m\pi^k$ with $m\in\{0,1,2\}$ and $k\in\{2,3,4\}$.
The constraint $\mathrm{Vol}(\mathcal{M}_{\mathrm{bulk}})=4\pi^3$ forces exponent pattern $(1,2)$ and coefficient $4$, hence the factors must be $U(1)$ and $SU(2)$.
For one-factor realizations, $\pi^2$ occurs only for $SO(3)$ and $\pi$ occurs only for $\mathbb{R}P^1$ in $\mathfrak{P}$, forcing the boundary and line strata.
\end{proof}

\begin{remark}[Serial aggregation is not interchangeable]\label{rem:alpha_aggregation_isolation}
Under the serial semantics of Proposition~\ref{prop:serial_impedance}, the impedance is additive after the logarithmic readout map.
For context, if one replaces serial aggregation by other standard aggregation rules acting on the three stratum costs $V_{\mathrm{bulk}}=4\pi^3$, $V_{\mathrm{boundary}}=\pi^2$, and $V_{\mathrm{line}}=\pi$, the resulting values are far from the CODATA reference:
the Euclidean aggregation $\sqrt{V_{\mathrm{bulk}}^2+V_{\mathrm{boundary}}^2+V_{\mathrm{line}}^2}$ gives $124.4568430929$,
the max aggregation $\max\{V_{\mathrm{bulk}},V_{\mathrm{boundary}},V_{\mathrm{line}}\}$ gives $124.0251067212$,
and the parallel-impedance aggregation $(V_{\mathrm{bulk}}^{-1}+V_{\mathrm{boundary}}^{-1}+V_{\mathrm{line}}^{-1})^{-1}$ gives $2.3381204994$.
\end{remark}
\begin{theorem}[Three-stratum geometric impedance for \texorpdfstring{$\alpha_{\mathrm{em}}$}{alpha_em}]\label{thm:alpha_three_channel}
Under Proposition~\ref{prop:serial_impedance} and the CAP-closed phase-volume dictionary in Definitions~\ref{def:impedance_volume}--\ref{def:em_phase_spaces}, one has
\[
\alpha_{\mathrm{em}}^{-1}{}_{\mathrm{geo}}
=\mathrm{Vol}(U(1)\times SU(2))+\mathrm{Vol}(SO(3))+\mathrm{Vol}(\mathbb{R}P^1)
=4\pi^3+\pi^2+\pi,
\]
hence \eqref{eq:alpha_geo}.
\end{theorem}
\begin{proof}
By Proposition~\ref{prop:serial_impedance} and Definition~\ref{def:impedance_volume}, the impedance equals the serial sum of the three stratum volumes.
By Definition~\ref{def:em_phase_spaces} and Lemma~\ref{lem:group_volumes}, $\mathrm{Vol}(U(1)\times SU(2))=4\pi^3$, $\mathrm{Vol}(SO(3))=\pi^2$, and $\mathrm{Vol}(\mathbb{R}P^1)=\pi$.
Summing gives \eqref{eq:alpha_geo}.
\end{proof}
This matches the philosophy of the present paper: coupling strength is not an arbitrary input, but a normalization cost induced by finite-resolution readout and compensation.
\paragraph{Interface reading.}
If each defect channel induces a compensation ``impedance'' (a normalization cost for maintaining protocol-consistent phase transport), then the electromagnetic coupling is the effective cost seen after channel composition.
In this sense, \eqref{eq:alpha_geo} is a numerical instance of the same template as Proposition~\ref{prop:gauge_compensation}, specialized to a dimensionless normalization observable.

\paragraph{CODATA comparison and explicit mismatch size.}
Let $\alpha^{-1}_{\mathrm{CODATA}}$ denote the CODATA recommended inverse fine-structure constant \cite{CODATA2022RMP}.
The mismatch in inverse impedance is:
\begin{equation}\label{eq:alpha_codata_mismatch}
\Delta\alpha^{-1}:=\alpha_{\mathrm{em}}^{-1}{}_{\mathrm{geo}}-\alpha_{\mathrm{CODATA}}^{-1}\approx 3.046\times 10^{-4},
\qquad
\frac{\Delta\alpha^{-1}}{\alpha_{\mathrm{CODATA}}^{-1}}\approx 2.22\times 10^{-6}.
\end{equation}
Since the CODATA uncertainty on $\alpha^{-1}$ is at the $10^{-8}$ level \cite{CODATA2022RMP}, the ppm-scale deviation in \eqref{eq:alpha_codata_mismatch} should be read as a protocol-level matching factor between a closed normalization target and a renormalized experimental convention, not as an ``agreement within error bars'' claim.
Equivalently, in the multiplicative weight variable $w=\exp(-\alpha^{-1})$, the interface mismatch is a single factor $s_\alpha=\exp(\Delta\alpha^{-1})\approx 1.0003046$.
Using the one-loop dictionary in Remark~\ref{rem:qed_running_scale_shift}, the additive mismatch $\Delta\alpha^{-1}$ can be equivalently represented as a matching-layer scale ratio $\mu_\ast/\mu_0=\exp(-(2\pi/b)\Delta\alpha^{-1})$ at fixed one-loop coefficient $b$.

\paragraph{Rigidity at bounded coefficient complexity.}
At bounded coefficient complexity (small nonnegative integers in the ansatz $a\pi^3+b\pi^2+c\pi$), the triple $(a,b,c)=(4,1,1)$ is rigidly singled out as the unique best fit in a finite search domain in the audit log-mismatch norm.

under deterministic tie-break rules (coefficient sum, then lexicographic $(a,b,c)$) is $(a,b,c)=(4,1,1)$.
\end{proposition}
\begin{proof}
This is a finite exhaustive enumeration over $286$ triples in the simplex $a+b+c\le 10$ with deterministic tie-break rules.
The top candidates and runner-up gap are recorded in Table~\ref{tab:alpha_coeff_rigidity} and are reproducible by \texttt{scripts/exp\_alpha\_coeff\_rigidity.py}.

\subsection{Electroweak normalization and the Weinberg angle}
\label{subsec:weinberg_angle}

At the $Z$ scale, we record a closed-theory electroweak matching normalization:
\begin{equation}\label{eq:weinberg_3_13}
\sin^2\theta_W(\mu_Z)=\frac{3}{13},
\end{equation}
and the gauge-sector normalizations can be written as
\begin{equation}\label{eq:ew_inverse_alphas}
\alpha_w^{-1}(\mu_Z)=3\pi^2,\qquad \alpha_Y^{-1}(\mu_Z)=10\pi^2,\qquad \alpha^{-1}(\mu_Z)=13\pi^2,
\end{equation}
under a canonical volume assignment consistent with the Standard Model relation between gauge couplings and the electromagnetic coupling \cite{Glashow1961,Weinberg1967,Salam1968,PeskinSchroeder1995,PDGRPP2024}.
\begin{definition}[Electroweak inverse couplings as weighted volumes (CAP closure form)]\label{def:ew_volumes}
At the matching scale $\mu=\mu_Z$, define the inverse couplings by weighted canonical volumes
\[
\alpha_w^{-1}(\mu_Z)=\dim(\mathfrak{su}(2))\,\mathrm{Vol}(SO(3)),\qquad
\alpha_Y^{-1}(\mu_Z)=\Bigl(\sum_{f\in\mathrm{SM}}Y_f^2\Bigr)\mathrm{Vol}(SO(3)),
\]
with hypercharge normalized by $Q=T_3+Y$.
The normalization is treated as a CAP-closed interface dictionary: the weights are discrete invariants already fixed by the closed labeling at the anchor (Lemma~\ref{lem:gauge_dim_order} and Lemma~\ref{lem:hypercharge_sum_10}), and the remaining numerical comparison to $Z$-scale $\overline{\mathrm{MS}}$ conventions is recorded as a matching-layer mismatch (Table~\ref{tab:ew_alpha_pi2_rigidity} and Table~\ref{tab:ew_sin2_rational_rigidity}).
\end{definition}

\begin{remark}[Why the weight $\sum Y^2$ is canonical at the interface]\label{rem:why_sum_y2}
In standard gauge theory, the normalization of an abelian coupling is tied to the choice of charge generator, and the leading vacuum-polarization/running coefficients depend on sums of squared charges (with multiplicities) \cite{PeskinSchroeder1995,PDGRPP2024}.
Thus the appearance of $\sum Y^2$ as a discrete weight in Definition~\ref{def:ew_volumes} is compatible with the conventional role of hypercharge-squared sums in $U(1)$ normalization and renormalization, even though this paper treats the relation only as a matching-layer dictionary rather than as a derived $\beta$-function statement \cite{MartinRobertson2019TadpoleFreeMSbar}.
\end{remark}

\begin{theorem}[Electroweak normalization and the Weinberg angle]\label{thm:weinberg_angle}
Under Definition~\ref{def:ew_volumes}, one has \eqref{eq:ew_inverse_alphas} and, consequently, \eqref{eq:weinberg_3_13}.
\end{theorem}
\begin{proof}
One has $\dim(\mathfrak{su}(2))=3$ (Lemma~\ref{lem:gauge_dim_order}) and $\sum_{f\in\mathrm{SM}}Y_f^2=10$ for three generations (Lemma~\ref{lem:hypercharge_sum_10}).
With the canonical quotient metric, $\mathrm{Vol}(SO(3))=\pi^2$ \cite{Nakahara2003}.
Thus $\alpha_w^{-1}(\mu_Z)=3\pi^2$ and $\alpha_Y^{-1}(\mu_Z)=10\pi^2$, giving $\alpha^{-1}(\mu_Z)=\alpha_w^{-1}+\alpha_Y^{-1}=13\pi^2$.
Finally, using the standard electroweak identity $\sin^2\theta_W=\alpha/\alpha_w=\alpha_w^{-1}/\alpha^{-1}$ \cite{PeskinSchroeder1995,PDGRPP2024} yields $\sin^2\theta_W(\mu_Z)=3/13$.
\end{proof}
\begin{remark}
We use $\alpha_w$ as the $SU(2)$ coupling (often denoted $\alpha_2$ in standard notation), and $\alpha_Y$ as the hypercharge coupling.
\end{remark}
\paragraph{Interface reading.}
The integer decomposition $13=3+10$ in \eqref{eq:ew_inverse_alphas} aligns with the $SU(2)$ gauge-sector dimension $3$ (Lemma~\ref{lem:gauge_dim_order}) and the three-generation hypercharge-squared sum $10$ (Lemma~\ref{lem:hypercharge_sum_10}), providing a shared integer backbone across the labeling interface and the electroweak normalization interface.
Here it is recorded as a consistency check that the three-channel template can support the SM electroweak mixing pattern.

The closed model gives

\alpha^{-1}(\mu_Z)=13\pi^2\approx 128.3048572142,
\qquad
\sin^2\theta_W(\mu_Z)=\frac{3}{13}\approx 0.2307692308.
\end{equation}
PDG quotes $\alpha^{-1}_{\mathrm{PDG}}(\mu_Z)\approx 127.955$ and $\sin^2\theta_{W,\mathrm{PDG}}(\mu_Z)\approx 0.23122$ in the $\overline{\mathrm{MS}}$ convention \cite{PDGRPP2024}.
Thus,
\begin{equation}\label{eq:ew_deviation_alphaZ}
13\pi^2-127.955\approx 3.50\times 10^{-1}\qquad (2.73\times 10^{-3}\ \text{relative}),
\end{equation}
and
\begin{equation}\label{eq:ew_deviation_sin2}
\frac{3}{13}-0.23122\approx -4.51\times 10^{-4}\qquad (-1.95\times 10^{-3}\ \text{relative}).

\begin{remark}[Unification-scale benchmarks vs.\ $Z$-scale conventions]\label{rem:gutsin2_benchmark}
The electroweak mixing angle is scale dependent.
In minimal $SU(5)$-type grand-unified models, one has the familiar tree-level unification-scale relation $\sin^2\theta_W=3/8$ \cite{GeorgiGlashow1974SU5,Langacker1981GUT}.
Running and threshold corrections then relate unification-scale parameters to $Z$-scale $\overline{\mathrm{MS}}$ conventions \cite{PDGRPP2024,PeskinSchroeder1995}.
Accordingly, our $Z$-scale targets in \eqref{eq:ew_closed_values} are recorded as matching-layer normalizations at $\mu_Z$, not as unification-scale relations.
\end{remark}

\paragraph{Bounded-complexity rigidity.}
Within the finite search domain $1\le n\le 50$ in the ansatz $\alpha^{-1}(\mu_Z)=n\pi^2$, the coefficient $n=13$ is the unique minimizer of the audit mismatch $e(n):=|\log((n\pi^2)/\alpha^{-1}_{\mathrm{PDG}}(\mu_Z))|$ (Table~\ref{tab:ew_alpha_pi2_rigidity}).
Within the rational search domain $1\le q\le 50$ for reduced rationals $p/q$, the fraction $3/13$ is the unique minimizer of $e(p/q):=|\log((p/q)/\sin^2\theta_{W,\mathrm{PDG}}(\mu_Z))|$ under the deterministic tie-break $(q,p)$ (Table~\ref{tab:ew_sin2_rational_rigidity}).
Combining the two fixes yields $\alpha_w^{-1}=3\pi^2$ and $\alpha_Y^{-1}=10\pi^2$ by \eqref{eq:ew_inverse_alphas}.
\begin{proposition}[Integer and rational rigidity at the electroweak scale]\label{prop:ew_rigidity}
Let $\alpha^{-1}_{\mathrm{PDG}}(\mu_Z)\approx 127.955$ and $\sin^2\theta_{W,\mathrm{PDG}}(\mu_Z)\approx 0.23122$ be the PDG reference values \cite{PDGRPP2024}.
Within the finite domains $1\le n\le 50$ in the ansatz $\alpha^{-1}(\mu_Z)=n\pi^2$ and $1\le q\le 50$ in reduced rationals $p/q$ for $\sin^2\theta_W(\mu_Z)$, the unique choices are

\alpha^{-1}(\mu_Z)=13\pi^2,\qquad \sin^2\theta_W(\mu_Z)=\frac{3}{13},
\]
and consequently $\alpha_w^{-1}(\mu_Z)=3\pi^2$ and $\alpha_Y^{-1}(\mu_Z)=10\pi^2$.
\end{proposition}
\begin{proof}

\end{proof}

\begin{remark}[One-loop scale shifts and additive mismatch]\label{rem:qed_running_scale_shift}
At one loop, gauge couplings run logarithmically: in a standard normalization one has an affine dependence
\[
\alpha^{-1}(\mu)=\alpha^{-1}(\mu_0)-\frac{b}{2\pi}\log\!\left(\frac{\mu}{\mu_0}\right)+\cdots,
\]
where the coefficient $b$ is fixed by the charged field content and convention \cite{PeskinSchroeder1995,WeinbergQFT1995}.
Therefore an additive mismatch $\Delta\alpha^{-1}$ can be equivalently represented as a multiplicative scale ratio
\[
\frac{\mu_\ast}{\mu_0}=\exp\!\left(-\frac{2\pi}{b}\,\Delta\alpha^{-1}\right)
\]
at fixed one-loop coefficient.
This is the sense in which we interpret ppm- to per-mille-level deviations as matching-layer scale shifts rather than as ``agreement within error bars'' statements.
\end{remark}

\begin{remark}[RG flow expressed in the resolution coordinate]\label{rem:rg_in_r_coordinate}
The paper uses the resolution coordinate $r(\mu)=\log(\mu/m_e)/\log\varphi$.
Since $\log\mu=\log m_e + r\log\varphi$, derivatives convert as
\[
\frac{\dd}{\dd r}=(\log\varphi)\,\frac{\dd}{\dd\log\mu}.
\]
Thus any standard renormalization-group equation $\dd g/\dd\log\mu=\beta(g)$ becomes
\[
\frac{\dd g}{\dd r}=(\log\varphi)\,\beta(g).
\]
In particular, the one-loop affine running of $\alpha^{-1}$ in Remark~\ref{rem:qed_running_scale_shift} corresponds to a constant slope $\dd \alpha^{-1}/\dd r=-(b\log\varphi)/(2\pi)$.
We use this only as a translation dictionary between standard scheme-dependent running and the protocol resolution language; the paper does not derive $\beta$-functions from the folding layer.
\end{remark}

\paragraph{A Fibonacci-scale interface remark.}
Using the one-loop scale-shift dictionary in Remark~\ref{rem:qed_running_scale_shift}, the mismatch $13\pi^2-\alpha^{-1}_{\mathrm{PDG}}(\mu_Z)$ can be phrased as a logarithmic scale ratio $\mu_\ast/\mu_Z$.

\subsection{A CP-odd phase space and the Jarlskog invariant}
\label{subsec:cp_jarlskog}

For quark mixing, a basis-independent measure of CP violation is the Jarlskog invariant $J$ \cite{Jarlskog1985}.
We record the resulting canonical closed value as a CAP-closed normalization dictionary.
\begin{definition}[CP-odd phase space (CAP closure form)]\label{def:cp_phase_space}
Within the primitive candidate set $\mathfrak{P}=\{U(1),SU(2),SO(3),\mathbb{R}P^1\}$ with canonical volumes (Lemma~\ref{lem:group_volumes}), define the CP-odd readout sector by
\begin{equation}\label{eq:cp_phase_space}
\mathcal{M}_{\mathrm{CP}}\cong SO(3)\times SO(3)\times SO(3)\times \mathbb{R}P^1.
\end{equation}
This choice is CAP-closed within the stated family by the volume-factorization rigidity in Proposition~\ref{prop:cp_phase_factorization_rigidity}.
\end{definition}

\begin{remark}[Structure of \texorpdfstring{$SO(3)^3\times \mathbb{R}P^1$}{SO(3)^3 x RP1}]\label{rem:cp_phase_space_structure}
In the PDG parameterization of a three-generation mixing matrix, the real mixing degrees of freedom can be organized as three independent rotations, while the physically relevant complex phase is defined only modulo rephasings and discrete identifications.
The factor $SO(3)^3$ models the three rotation degrees of freedom, and the $\mathbb{R}P^1$ factor models a projective phase degree of freedom after the $\ZZ_2$ ray identification, consistent with expressing CP violation through rephasing-invariant quantities such as the Jarlskog invariant $J$ \cite{Jarlskog1985,PDGRPP2024,Nakahara2003}.
\end{remark}
\begin{lemma}[Canonical volume of \texorpdfstring{$\mathcal{M}_{\mathrm{CP}}$}{M_CP}]\label{lem:cp_volume_pi7}
Under the canonical quotient metrics, $\mathrm{Vol}(\mathcal{M}_{\mathrm{CP}})=\pi^7$.
\end{lemma}
\begin{proof}
By Lemma~\ref{lem:group_volumes}, $\mathrm{Vol}(SO(3))=\pi^2$ and $\mathrm{Vol}(\mathbb{R}P^1)=\pi$.
Therefore $\mathrm{Vol}(\mathcal{M}_{\mathrm{CP}})=(\pi^2)^3\cdot \pi=\pi^7$.
\end{proof}
\begin{equation}\label{eq:j_geo}
d_{\mathrm{CP}}:=\dim(\mathfrak{su}(3))+\dim(\mathfrak{su}(2))=8+3=11,
\qquad
J_{\mathrm{geo}}:=\frac{1}{d_{\mathrm{CP}}\,\mathrm{Vol}(\mathcal{M}_{\mathrm{CP}})}=\frac{1}{11\pi^7}.
\end{equation}

\begin{remark}[Why the multiplicity excludes the abelian factor]\label{rem:dcp_excludes_u1}
The nontrivial CP-odd rephasing-invariant residue encoded by $J$ is tied to genuinely non-abelian mixing structure in a three-generation setting \cite{Jarlskog1985,PDGRPP2024}.
Accordingly, in the low-complexity multiplicity count we weight only the non-abelian gauge-sector dimensions $8$ and $3$; the abelian $U(1)$ factor is treated as part of the phase normalization already accounted for in the ray/projective quotient structure.
\end{remark}
\begin{proposition}[Primitive factorization rigidity for the CP-odd phase space]\label{prop:cp_phase_factorization_rigidity}
Fix the primitive candidate set $\mathfrak{P}=\{U(1),SU(2),SO(3),\mathbb{R}P^1\}$ with canonical volumes as in Lemma~\ref{lem:group_volumes}.
Among products of primitives with $\mathrm{Vol}(\mathcal{M})=\pi^7$, the factorization is unique (up to ordering):
\[
\mathcal{M}\cong SO(3)\times SO(3)\times SO(3)\times \mathbb{R}P^1.
\]
\end{proposition}
\begin{proof}
From Lemma~\ref{lem:group_volumes}, each primitive volume is one of $2\pi$, $2\pi^2$, $\pi^2$, $\pi$.
Any product therefore has the form $2^m\pi^k$ with $m\in\ZZ_{\ge 0}$.
The constraint $\mathrm{Vol}(\mathcal{M})=\pi^7$ forces $m=0$, hence no factor can be $U(1)$ or $SU(2)$, and all factors must be chosen from $\{SO(3),\mathbb{R}P^1\}$.
To achieve exponent $7$ with $SO(3)$ contributing $\pi^2$ and $\mathbb{R}P^1$ contributing $\pi$, the only possibility is three $SO(3)$ factors and one $\mathbb{R}P^1$ factor.
\end{proof}
The interpretation compatible with the present protocol picture is that attempting to ``pull back'' $CP$ (conjugation plus reflection) into a fixed readout protocol leaves an unavoidable residual geometric phase (holonomy), whose magnitude is controlled by a CP-odd phase-space volume and a discrete gauge-sector multiplicity.
\paragraph{Connection viewpoint.}
In the Standard Model, the CKM matrix is the unitary mixing matrix for quarks, with CP violation originating from an irreducible complex phase (three generations) \cite{Cabibbo1963,KobayashiMaskawa1973,PDGRPP2024}.
The protocol picture naturally expresses such phases as holonomy data of a connection on a protocol parameter manifold.
The CP-odd invariant is then a rephasing-invariant residue of holonomy, consistent with the volume/multiplicity expression \eqref{eq:j_geo}.

\paragraph{PDG comparison and rigidity signal.}
PDG quotes $J_{\mathrm{PDG}}=(3.00\pm 0.15)\times 10^{-5}$ \cite{PDGRPP2024}.
The closed reference value \eqref{eq:j_geo} gives
\begin{equation}\label{eq:j_geo_numeric}
J_{\mathrm{geo}}=\frac{1}{11\pi^7}\approx 3.009942547\times 10^{-5},
\end{equation}
with relative deviation $\approx 3.31\times 10^{-3}$ at the PDG central value.
Numerically, $J_{\mathrm{geo}}-J_{\mathrm{PDG,central}}\approx 9.94\times 10^{-8}$, corresponding to $\approx 6.6\times 10^{-2}\sigma$ under the quoted PDG uncertainty.
Moreover, within the bounded search domain $1\le a\le 50$ and $1\le n\le 20$ in the ansatz $J=1/(a\pi^n)$, the pair $(a,n)=(11,7)$ is uniquely selected by finite rigidity enumeration (Table~\ref{tab:jarlskog_pi_rigidity}).
\begin{proposition}[Bounded-complexity rigidity for CKM CP violation]\label{prop:jarlskog_rigidity_stmt}
Let $J_{\mathrm{PDG}}=3.00\times 10^{-5}$ be the PDG central value \cite{PDGRPP2024}.
Within the finite domain $1\le a\le 50$ and $1\le n\le 20$ in the ansatz $J=1/(a\pi^n)$, the unique minimizer of the audit mismatch
\[
e(a,n):=\left|\log\!\left(\frac{1/(a\pi^n)}{J_{\mathrm{PDG}}}\right)\right|
\]
under deterministic tie-break rules (minimize $a+n$, then lexicographic $(a,n)$) is $(a,n)=(11,7)$, yielding $J_{\mathrm{geo}}=1/(11\pi^7)$.
\end{proposition}
\begin{proof}
Finite exhaustive enumeration with deterministic tie-break rules; reproduced by \texttt{scripts/exp\_jarlskog\_pi\_rigidity.py} and recorded in Table~\ref{tab:jarlskog_pi_rigidity}.
\end{proof}

\subsection{CKM mixing magnitudes as bounded-complexity depths}
\label{subsec:ckm_mixing_depths}

The protocol interface naturally expresses small dimensionless amplitudes in the golden resolution coordinate
\begin{equation}\label{eq:r_mix_def}
r_{\mathrm{mix}}(x):=-\frac{\log x}{\log\varphi}\qquad(x>0),
\end{equation}
analogous to the mass-resolution coordinate used in Section~\ref{sec:mass_spectrum_closure}.
We test a minimal bounded-complexity closure for the three small CKM magnitudes $|V_{us}|$, $|V_{cb}|$, and $|V_{ub}|$ (PDG conventions) \cite{PDGRPP2024}.

\paragraph{Forcing rationale (normalization and depth).}
In the protocol viewpoint, mixing magnitudes are overlap amplitudes between normalized readout modes.
We restrict to a bounded candidate family fixed by two protocol-native mechanisms:
(i) normalization/equipartition in an effective $d$-dimensional mixing subspace.
For a Haar-uniform unitary matrix $U\in U(d)$ one has $\mathbb{E}|U_{ij}|^2=1/d$, so the minimal-description overlap scale is $|U_{ij}|\sim 1/\sqrt{d}$ \cite{Mezzadri2007,CollinsSnady2006}.
(ii) golden-branch depth costs produce exponentially small weights with base $\varphi$; taking square roots at the amplitude level yields half-depth factors $\varphi^{-k/2}$ (cf.\ the zeta/Abel pole-barrier scaling in Section~\ref{subsec:e_channel}).
Accordingly, we treat CKM hierarchies as a bounded-complexity closure problem in discrete depth/normalization variables rather than as free continuous parameters.

\paragraph{Candidate family.}
Let $V_{\max}:=\max_{w\in X_6}V(w)=20$ at $(m,n)=(6,3)$.
We consider the bounded family
\begin{equation}\label{eq:ckm_candidate_family}
|V_{us}|=\frac{1}{\sqrt{d}},\quad 1\le d\le V_{\max},
\qquad
|V_{cb}|=\varphi^{-k_{23}/2},\quad |V_{ub}|=\varphi^{-k_{13}/2},
\qquad k_{23},k_{13}\in\NN,
\end{equation}
and select coefficients by a bounded-complexity minimax search.
Here $d$ is a normalization-dimension parameter (unrelated to the fixed window length $m=6$ used elsewhere in the paper).

\begin{remark}[Why $d\le V_{\max}$ and why $k\le 2B$]\label{rem:ckm_bounds_rationale}
The bound $d\le V_{\max}$ ties the normalization parameter to the intrinsic finite invariant range at $(m,n)=(6,3)$: by Proposition~\ref{prop:V_bijection}, $V(w)$ takes exactly the values $\{0,\dots,V_{\max}\}$, so $V_{\max}=20$ is a protocol-intrinsic integer cutoff available without importing external scales.
For the depth exponents, the amplitudes are written in half-depth form $\varphi^{-k/2}$ (cf.\ \eqref{eq:r_mix_def}); therefore restricting $k$ to a box of size $O(B)$ makes the exponent complexity comparable to the integer search radius $B$ used for other closures.
The choice $k\le 2B$ is a minimal symmetry between integer and half-integer depth budgets: it allows exponents up to depth $B$ in the amplitude-squared scale, while keeping the candidate domain finite and auditable.
\end{remark}

\begin{proposition}[Bounded-complexity rigidity for CKM magnitudes]\label{prop:ckm_mixing_rigidity}
Fix PDG reference magnitudes $|V_{us}|,|V_{cb}|,|V_{ub}|$ \cite{PDGRPP2024}.
For each bound $B\in\{1,\dots,20\}$, minimize the objective
\[
\max\Bigl\{\bigl|\log(|V_{us}|_{\mathrm{pred}}/|V_{us}|_{\mathrm{PDG}})\bigr|,
\ \bigl|\log(|V_{cb}|_{\mathrm{pred}}/|V_{cb}|_{\mathrm{PDG}})\bigr|,
\ \bigl|\log(|V_{ub}|_{\mathrm{pred}}/|V_{ub}|_{\mathrm{PDG}})\bigr|\Bigr\},
\]
over the finite domain $1\le d\le \min(B,V_{\max})$ and $1\le k_{23},k_{13}\le 2B$, using lexicographic tie-break rules (sum error, then coefficient sum, then $(d,k_{23},k_{13})$).
At $B=20$ the unique minimizer is
\[
(d,k_{23},k_{13})=(20,13,23),
\]
and, within the tested range $B=1,\dots,20$, the pair $(k_{23},k_{13})$ stabilizes at $B=12$ and remains constant for all $B$ with $12\le B\le 20$ (Table~\ref{tab:ckm_mixing_rigidity}).
\end{proposition}
\begin{proof}
Finite exhaustive enumeration with deterministic tie-break rules; reproduced by the accompanying script.
\end{proof}

\begin{remark}[Candidate-domain size]\label{rem:ckm_domain_size}
At bound $B$, the search domain has size
\[
|\Theta(B)|=\min(B,V_{\max})\cdot (2B)^2.
\]
At $B=20$ and $V_{\max}=20$, this is $|\Theta|=20\cdot 40^2=32000$ candidates.
\end{remark}

\subsection{CKM matrix closure from three magnitudes and a Jarlskog anchor}
\label{subsec:ckm_matrix_closure}

To turn the magnitude closure of Table~\ref{tab:ckm_mixing} into a fully specified $3\times 3$ mixing matrix, we use the PDG standard CKM parameterization by three angles and one CP phase \cite{PDGRPP2024}.
In this convention,
\[
|V_{ub}|=s_{13},\qquad |V_{us}|=s_{12}c_{13},\qquad |V_{cb}|=s_{23}c_{13},
\]
and the Jarlskog invariant is
\[
J=s_{12}s_{23}s_{13}c_{12}c_{23}c_{13}^2\sin\delta.
\]
We therefore (i) extract $(s_{12},s_{23},s_{13})$ from the magnitude triplet, (ii) solve for $\delta$ using a chosen $J$ normalization, and (iii) reconstruct the full CKM matrix.

\paragraph{Reference reconstruction.}
To avoid mixing incompatible PDG conventions for different entries, we define a single ``PDG reference reconstruction'' by taking the PDG central values of $|V_{us}|$, $|V_{cb}|$, $|V_{ub}|$, and $J$ used elsewhere in this paper, extracting $(s_{12},s_{23},s_{13},\delta)$ via the above identities, and then generating all $|V_{ij}|$ from the PDG parameterization.
The closed prediction uses the bounded-complexity minimizer for $(|V_{us}|,|V_{cb}|,|V_{ub}|)$ and the rigid $J_{\mathrm{geo}}$ value \eqref{eq:j_geo}.
