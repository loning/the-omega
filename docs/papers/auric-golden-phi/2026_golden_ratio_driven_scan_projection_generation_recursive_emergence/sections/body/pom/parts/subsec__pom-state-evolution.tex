\subsection{Ontological state space and local evolution}\label{subsec:pom-state-evolution}
\begin{definition}[Resolution State Space]\label{def:pom-state-space}
For resolution $m$, define the ontology space
\[
\mathcal{H}_m:=\mathcal{X}_m\times \mathcal{M}_m,
\]
Among them, $\mathcal{X}_m$ â€‹â€‹is the appearance coordinate (such as the truncation of $X_m$ or $X_\infty$), and $\mathcal{M}_m$ is the hidden fiber (which can include the local activation block, synchronized core state and margin of $\mathcal{S}_\varphi,\mathcal{T},\mathcal{P}$, etc.).
\end{definition}

\begin{definition}[Locality and Limited Propagation]\label{def:pom-locality}
Given one-step evolution operator of ontology
\[
U:\mathcal{H}_m\to\mathcal{H}_m
\]
It is also said that \(U\) â€‹â€‹satisfies \textbf{locality and limited propagation}, if each step only relies on a limited window and only updates local blocks of limited radius. This definition is consistent with the local kernel/online normalization scope of Section \ref{sec:A-kernel-compare-main}.
\end{definition}

\subsection{Three types of projections: canonical projection, atomic projection and order projection}\label{subsec:pom-projections}
\\begin{definition}[Canonical projection $P_Z$]\\label{def:pom-proj-norm}`r`nAt the window level, $P_Z$ is defined as the normalized projection $\\Fold_m:\\Omega_m\\to X_m$ (Definition \\ref{def:fold-word}), implemented as a value-preserving local rewriting system that is terminating and confluent (Theorem \\ref{thm:fold-suite}). At the level of operator algebra, $P_Z$ conditional expectations induced by folding
\[
E_m:\mathcal{A}_m\to\mathcal{B}_m
\]
Realizes (proposition \ref{prop:fold-conditional-expectation}), so $P_Z$ â€‹â€‹is an idempotent projection and is independent of the reduction order.
\end{definition}

\begin{definition}[Atomic projection $P_{\mathrm{prim}}$]\label{def:pom-proj-prim}
Given the kernel/evolution operator $T$, let the time trace sequence $a_n:=\Tr(T^n)$. Define primitive orbit spectrum
\[
p_n:=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}.
\]
Note $P_{\mathrm{prim}}(a):=(p_n)_{n\ge 1}$, then invert by Euler product and MÃ¶bius/Witt, we have
\[
Z=E\circ P_{\mathrm{prim}}\circ T,
\qquad
Z=E\circ M\circ T.
\]
Therefore $P_{\mathrm{prim}}$ â€‹â€‹is a canonical readout gate that projects a "time trace" into a prime-like atomic layer.
\end{definition}

\begin{definition}[Sequence projection $P_{\le}$]\label{def:pom-proj-order}
Order projection $P_{\le}$ â€‹â€‹refers to the readout certificate that maps ontological states to ordered constraints (such as symbols, comparison decisions, or interval constraints). Its goal is to encode global order relationships such as "$0\le r<b$" into observable outputs. Unlike $P_Z$, $P_{\le}$ is not necessarily ring homomorphic, nor is it guaranteed to be locally parallelizable; its readout often requires cross-window aggregation and is therefore a major source of non-parallelizable growth.
\end{definition}

\begin{definition}[Temperature sequence projection $\mathrm{PROJ}_u$ â€‹â€‹(soft sequence gate)]\label{def:pom-proj-temp}
Let $\Gamma$ â€‹â€‹be some finite graph kernel for order/constraint readout (such as comparison, interval constraint, or residue hierarchical kernel), and let each edge $e$ have a nonnegative additive cost potential
\[
E(e)\in\ZZ_{\ge 0},
\qquad
E(\gamma)=\sum_{e\in\gamma}E(e).
\]
For $u\in(0,1]$, define a weighted adjacency matrix
\[
M_E(u)_{ij}:=\sum_{e:i\to j}u^{E(e)}.
\]
Let $\mu_u$ â€‹â€‹be the equilibrium/Parry measure of this weighted SFT (equivalently the one-step Markov measure given by the Perron--Frobenius eigenvectors of $M_E(u)$). For any certificate readout function $R$ (such as comparison symbols, quotient interval labels, or congruence class labels), define \textbf{temperature-ordered projection} as its push-forward distribution under $\mu_u$:
\[
\mathrm{PROJ}_u(R):=R_\ast\mu_u.
\]
When $u$ â€‹â€‹is close to $1$, $\mathrm{PROJ}_u$ embodies "soft constraints/average readout"; when $u$ decreases ($\beta=-\log u$ increases), the readout gradually concentrates on the path set with lower cost (more satisfying constraints/closer to the optimal).
\end{definition}

\begin{remark}[Sequence projection as zero temperature limit]\label{rem:pom-order-as-zero-temp}
Under the local constant potential (side potential) setting, the zero-temperature limit $u\to 0$ â€‹â€‹(equivalent to $\beta\to+\infty$) pushes the equilibrium state $\mu_u$ to \emph{the set of ground states}: favoring the period/sub-shift component that minimizes the average cost $\bar E=\lim_{n\to\infty}E(\gamma_n)/|\gamma_n|$ (see the general conclusion of \cite{ChazottesGambaudoUgalde2011ZeroTempLocConst}). Therefore, at the mechanism level, sequence projection $P_{\le}$ can be understood as "the hardening limit of soft sequence gate $\mathrm{PROJ}_u$ at zero temperature": switching from weighted averaging to selection/feasibility screening (typically $\arg\min$ structure), which is also the source of $P_{\le}$'s irreducible introduction of non-parallel readout depth.

On a finite graph, the zero-temperature ground state can be accurately characterized by the minimum mean cycle weight: let $\phi_\theta=\langle\theta,\Phi\rangle$ â€‹â€‹or generally $\phi=E$, then the minimum average cost is equal to the min-mean-cycle value on the graph, and can be found in polynomial time using the Karp algorithm (\cite{Karp1978MinCycleMean}). This translates the key constraints of ordinal projection into an auditable composite certificate interface.
\end{remark}

\subsubsection{Convex geometry induced by potential: reachable mean convex body and dual pressure}\label{subsec:pom-potential-convex-geometry}
\paragraph{Potential vector as intrinsic coordinate}
For finite graph kernel (SFT) \(\Gamma\), assign a multi-dimensional additive observation quantity (potential vector) to each edge
$$
\varphi:E(\Gamma)\to \ZZ^d.
$$
For any \(\sigma\)-invariant measure \(\mu\), its average observation vector is defined as
$$
m(\mu):=\int \varphi\,d\mu\ \in\ \RR^d.
$$
Definition \textbf{reachable mean set} (rotation/mean set)
$$
\mathcal{R}_\varphi:=\{m(\mu):\ \mu\ \text{is}@@TK15@ @\Gamma\ \text{on}\ \sigma\text{-invariant measure}\}\subset\RR^d.
$$
Since the invariant measure set is compact and convex under the weak-\(\ast\) â€‹â€‹topology, and \(\mu\mapsto m(\mu)\) is a continuous linear mapping, \(\mathcal{R}_\varphi\) is also a compact convex set.

\paragraph{Pressure surface and rate function: convex duality reads out "geometry"}
For parameter \(\theta\in\RR^d\), let the edge weight be \(\exp\langle\theta,\varphi(e)\rangle\), and get the weighted matrix family \(M_\theta\) and pressure
$$
P(\theta):=\log\rho(M_\theta).
$$
In the original case, \(\nabla P(\theta)\) â€‹â€‹gives the typical mean density at equilibrium, while the Hessian \(\nabla^2P(\theta)\) gives the covariance density (the text \ref{subsec:logM-analyticity} has fixed this "differentiable fingerprint" as an implementation interface).
More importantly: for a fixed benchmark \(\theta_0\), the LDP rate function of the empirical mean can be written as
$$
I_{\theta_0}(x)=\sup_{t\in\RR^d}\Bigl(\langle t,x\rangle-\bigl(P(\theta_0+t)-P(\theta_0)\bigr)\Bigr),
$$
Its effective domain falls exactly on \(\mathcal{R}_\varphi\) â€‹â€‹(outside the domain, take \(+\infty\)).
Therefore, "geometry" need not be embedded a priori in Euclidean space: it is generated endogenously by the juxtaposition of potential vectors with kernel constraints, and is read out quantitatively by the Legendre--Fenchel duality of \((P\leftrightarrow I)\).

\paragraph{6D hypercube and 3D slices (minimum form)}
If \(6\) â€‹â€‹binary channels (value \(0/1\) at each step) are selected as potential components \(\varphi=(\varphi_1,\dots,\varphi_6)\), then any orbit of length \(n\) gives a counting vector
$$
N_n:=\sum_{t=0}^{n-1}\varphi(e_t)\in\{0,1,\dots,n\}^6,
$$
After normalization it falls on \([0,1]^6\). A truly reachable set is one of the convex subvolumes clipped by the allowed paths of the kernel (i.e., the instantiation of \(\mathcal{R}_\varphi\) under this choice).
The so-called "3D section of a 6D cube" is the section obtained by fixing \(3\) â€‹â€‹coordinates (or fixing \(3\) linear combinations, such as \(\chi=\varphi_e-\varphi_2\)); and performing gradient/Hessian reading on the slice in \(\theta\) space corresponds to the first-order and second-order local measurements of the convex body on the slice.

\begin{remark}[Category boundaries and non-parallel growth]\label{rem:pom-factor-boundary}
Any sliding block factor code has a constant online delay, so all readout depths that grow with scale must come from non-factor code projections (typically $P_{\le}$) or leave the mix in the evolving depth $D_U$. This â€œcategory boundaryâ€ elevates the growth term from an algorithmic trick to a structural constraint.
\end{remark}

\begin{definition}[Character/Congruent Projection $P_\chi$]\label{def:pom-proj-chi}
Role projection $P_\chi$ â€‹â€‹refers to making congruence/role layering of primitive orbits according to additive observable quantities (such as $\chi(\gamma),N_-(\gamma),N_e(\gamma)$), and reading out the residue class label. It can be realized by the twisted matrix family $M(q)$ and the Dirichlet--Mertens constant tensor (see \ref{subsec:arity-dirichlet-mertens-tensor}), so it is a concretization gate of $P_{\le}$ in the "order/congruence/role readout" direction, and can also be regarded as a role-based enhancement of $P_{\mathrm{prim}}$.
\end{definition}

\begin{remark}[Dominant axis of sequence constraint: collision axis]\label{rem:pom-collision-axis}
In the real input 40-state kernel, the worst-case distortion spectrum radius of \(((3,3,5))\) â€‹â€‹is reached by a pure collision cocycle (note \ref{prop:arity-335-collision-dominant}), indicating that the dominant limiting factor of order projection is not the "value axis" but the collision/destruction axis. Therefore, if space is exchanged for time, it is more effective to introduce separable collision axis beams first than to simply expand the value channel.
\end{remark}

\begin{remark}["Adding axes does not necessarily speed up": more subdivided congruent readout may be slower]\label{rem:pom-more-axis-slower}
In three-dimensional congruential stratification, two sets of auditable worst-case distortion ratios have been given:
\[
\rho_{3,3,3}/\lambda\approx 0.9366863402,
\qquad
\rho_{5,5,5}/\lambda\approx 0.9628140340
\]
(Appendix for numerical output of \ref{app:real-input-40-arity-3d}; see scripts \texttt{scripts/exp\_sync\_kernel\_real\_input\_40\_arity\_3d.py}).
Note that \(\rho_{5,5,5}/\lambda\) â€‹â€‹is closer to \(1\), so its distortion spectral gap \(-\log(\rho/\lambda)\) is smaller and the corresponding mixing scale is larger.

This gives a structural consequence: when the worst distortion is dominated by \emph{pure collisional cocycle}, increasing the mode of the congruence axis (finer residue/role readout) introduces more non-trivial characters, but \textbf{the slowest mode is not therefore eliminated by "value axis subdivision"}; instead, it may approach the resonance further and thus slow down the mixing.
Therefore, the effective way to "exchange space for time" in this phase area is not to simply increase the character dimension, but to change the core geometry to decouple/weaken the collision axis (such as introducing a detachable collision axis bundle or changing the projection gate arrangement).
\end{remark}

\subsection{Projection word syntax and cost model}\label{subsec:pom-words}
\begin{definition}[Projective words and semigroups]\label{def:pom-projection-word}
Let the projected alphabet $\{Z,\le\}$ â€‹â€‹represent one reading of $P_Z$ and one reading of $P_{\le}$ respectively. For any instruction/microcode $I$, define the projection word
\[
W(I)=(k_Z(I),k_{\le}(I)),
\]
Where $k_Z,k_{\le}$ â€‹â€‹is the number of calls to canonical/order projection. make
\[
\mathcal{W}=\langle Z,\le\rangle^*/\sim,\qquad Z^2\sim Z
\]
is a projective word semigroup (the idempotence law comes from $P_Z\circ P_Z=P_Z$); other equivalence relations can be attached according to the causal cone/local commutativity of the specific implementation.
\end{definition}

\begin{remark}[Extended projection word of atomic projection]\label{rem:pom-word-prim}
If the atomic projection is included in the readout, the alphabet can be extended to $\{Z,\mathrm{prim},\le\}$, and noted
\[
W^{\mathrm{full}}(I)=(k_Z(I),k_{\mathrm{prim}}(I),k_{\le}(I)).
\]
The default for arithmetic instructions is $k_{\mathrm{prim}}=0$, and atomic projection mainly serves the readout and mechanism audit of prime-like spectra.
\end{remark}

\begin{remark}[Projection word token for character projection]\label{rem:pom-word-chi}
When sequential reading is implemented by $P_\chi$, the projected alphabet can be rewritten as $\{Z,\chi\}$ and written
\(
W_\chi(I)=(k_Z(I),k_\chi(I))
\)ã€‚
At this time $k_\chi$ â€‹â€‹only counts the number of calls of role/congruential projection, as a concrete version of $k_{\le}$.
\end{remark}

\begin{definition}[cross-resolution reduced count]\label{def:pom-rescale-count}
In addition to the readout gate $P_Z,P_{\le}$ â€‹â€‹in the projection word, if the instruction/microcode $I$ explicitly calls the resolution reduction projection gate
\(
\mathrm{RESCALE}\downarrow:\Omega_{m+1}\to X_m
\)
(See note \ref{rem:pom-rescale-gauge-anomaly}), then the number of calls is recorded as $k_{\downarrow}(I)$; if cross-resolution readout is not involved, it is taken as $k_{\downarrow}(I)=0$.
\end{definition}

\begin{definition}[Fiber selection and core selection]\label{def:pom-fiber-kernel-choice}
For any instruction/microcode $I$, record the prior fiber family activated during its implementation as
$$
\mathcal{F}(I)\subseteq \{\mathcal{S}_\varphi,\mathcal{T},\mathcal{P},\widehat{\ZZ}\},
$$
Corresponding to the scale fiber, phase fiber, prime register fiber and profinite congruence axis respectively (definition \ref{def:pom-scale-fiber}, \ref{def:pom-phase-fiber}, \ref{def:pom-prime-fiber}, \ref{def:pom-profinite}).
And remember that the normalization core used for standard readout is $K_Z(I)$ â€‹â€‹(that is, the specific core that implements $P_Z$), and the core used for sequence/role readout is $K_{\le}(I)$ or $K_{\chi}(I)$ (the specific core that implements $P_{\le}$ or $P_\chi$).
\end{definition}

\begin{remark}[Tight group packaging of phase fiber and register axis: general solenoid expansion]\label{rem:pom-solenoid-extension}
Definition \ref{def:pom-fiber-kernel-choice} juxtaposes the phase fiber \(\mathcal{T}\) â€‹â€‹and the congruence axis \(\widehat{\ZZ}\) as a priori fiber family that can be activated by \(\mathrm{LIFT}\).
At the level of compact commutative groups, these two axes can be uniformly encapsulated by a canonical compact abelian group expansion: defining a one-dimensional universal solenoid
\begin{equation}\label{eq:pom-universal-solenoid-quotient}
\boxed{\;
\Sol\ :=\ (\RR\times \widehat{\ZZ})/\ZZ,\qquad
n\cdot(t,a):=(t+n,\ a-n)\; }.
\end{equation}
Then there exists a natural short exact sequence
\begin{equation}\label{eq:pom-universal-solenoid-ses}
\boxed{\;
0\ \longrightarrow\ \widehat{\ZZ}\ \xrightarrow{\ \iota_{\widehat{\ZZ}}\ }\ \Sol\ \xrightarrow{\ \pi_{\TT}\ }\ \TT\ \longrightarrow\ 0\;}
\end{equation}
in
\(
\iota_{\widehat{\ZZ}}(a)=[(0,a)]
\)
for embedding, while
\(
\pi_{\TT}([(t,a)])=e^{2\pi \mathrm{i}t}
\)
is a surjection (its core is isomorphic to \(\widehat{\ZZ}\)).
\par
The structural meaning of this expansion is that the phase circle \(\TT\) can be read out as the quotient group of \(\Sol\) (corresponding to the projected image of the phase fiber \(\mathcal{T}\)), and the congruence axis \(\widehat{\ZZ}\) as the kernel is "built in" as an invisible degree of freedom; therefore any forgetting/irreversibility that occurs under phase projection can be seen in \(\Sol\) The layer is strictly promoted to a reversible group action and the difference is externalized as the residual coordinate in the kernel direction.
This elevates the narrative constraint of â€œinformation external conservation/reversibilityâ€ into an auditable group expansion structure interface.
\end{remark}

\begin{definition}[Command signature: Projection word $\times$ â€‹â€‹Nuclear fingerprint $\times$ Fiber selection]\label{def:pom-instruction-signature}
Let $\mathfrak{F}(K)$ â€‹â€‹be the spectral fingerprint of the nucleus (defining \ref{def:kernel-spectral-fingerprint}). For any instruction/microcode $I$, define its (audit) instruction signature as
$$
\Sigma(I):=\Bigl(W(I),\ k_{\downarrow}(I),\ \mathfrak{F}\bigl(K_Z(I)\bigr),\ \mathfrak{F}\bigl(K_{\le}(I)\bigr),\ \mathcal{F}(I)\Bigr),
$$
Among them, $W(I)$ â€‹â€‹is the projection word (defined as \ref{def:pom-projection-word}), $k_{\downarrow}(I)$ is the number of cross-resolution reduction calls (defined as \ref{def:pom-rescale-count}), and $\mathcal{F}(I)$ is the fiber selection (defined as \ref{def:pom-fiber-kernel-choice}). When sequential readout is implemented by $P_\chi$, $\mathfrak{F}(K_{\le}(I))$ can be replaced by $\mathfrak{F}(K_{\chi}(I))$.
\end{definition}

\begin{remark}[Why $W$ â€‹â€‹is not enough: nuclear geometry enters the fingerprint]\label{rem:pom-signature-why}
The projection word $W(I)$ â€‹â€‹only records "how many times the projection gate is called" and does not record "what is the geometry of the projection gate". The nuclear fingerprint $\mathfrak{F}(K)$ (such as $\zeta_K$, primitive hierarchical spectrum $\{B_{K,n}(u)\}$, finite part constant and twisted spectral gap) determines the difference in the mechanism layer of the same projection gate (Appendix \ref{app:sync-kernel-A-compare}). So a stronger unified classification object would be $\Sigma(I)$ instead of just $W(I)$.
\end{remark}

\begin{proposition}[The signature is the same $\Rightarrow$ â€‹â€‹The fingerprint is the same as the cost model]\label{prop:pom-signature-audit}
Suppose two instructions/microcode $I,J$ â€‹â€‹satisfy $\Sigma(I)=\Sigma(J)$. but:
\begin{enumerate}
  \item \textbf{The projection audit invariants are consistent:} The two give the same auditable fingerprints such as $\zeta$, primitive atomic spectrum (including its weighted layering), pressure curve endpoints and distortion spectral gaps under the same nuclear fingerprint scope (completely determined by $\mathfrak{F}(K_Z),\mathfrak{F}(K_{\le})$).
  \item \textbf{Cost model is of the same type:} Under the scope of defining \ref{def:pom-cost}, the two have the same projection word coefficient $(k_Z,k_{\le})$ â€‹â€‹and the same number of cross-resolution order reductions $k_{\downarrow}$, and their constant terms (such as $\delta_Z$) and the mixed scale (such as $\tau_{\mathrm{mix}}$) are consistent by the same kernel fingerprint, so the structural decomposition of the cost model is consistent:
  $$
  \begin{aligned}
  T_I(n;m)&=k_Z\,\delta_Z+k_{\le}\,T_{\le}(n)+k_{\downarrow}\,T_{\downarrow}(m)+D_U(n),\\
  T_J(n;m)&=k_Z\,\delta_Z+k_{\le}\,T_{\le}(n)+k_{\downarrow}\,T_{\downarrow}(m)+D_U'(n),
  \end{aligned}
  $$
  The difference may only appear in the "linearization depth/evolution depth" item $D_U$, which is determined by the fiber selection $\mathcal{F}$ and the specific evolution arrangement (definition \ref{def:pom-space-time}).
\end{enumerate}
\end{proposition}

\begin{definition}[Cost Model]\label{def:pom-cost}
Let $\delta_Z$ be the baseline readout depth of a canonical projection (e.g. fixed delay of online normalization, see theorem \ref{thm:online-delay-fold}), let $T_{\le}(n)$ be the aggregation depth of a sequential projection, let $T_{\downarrow}(m)$ be the typical writeback support scale of a cross-resolution reduction gate $\mathrm{RESCALE}\downarrow$ (note \ref{rem:pom-rescale-gauge-anomaly}), let $D_U(n)$ is the depth required to complete linearization in ontology evolution. Then the definition
\[
T_I(n;m)=k_Z(I)\,\delta_Z+k_{\le}(I)\,T_{\le}(n)+k_{\downarrow}(I)\,T_{\downarrow}(m)+D_U(n).
\]
This model separates projection readout from evolution depth and is used to compare the differences in strategies of different implementations in placing "mixing" in time or space.
\end{definition}

\begin{remark}[Resolution downgrading is not "throwing the tail": \texorpdfstring{$\mathrm{RESCALE}\downarrow$}{RESCALE down} is lower than the specification limit]\label{rem:pom-rescale-gauge-anomaly}
In addition to \(P_Z,P_{\le}\) â€‹â€‹in the projection word, the resolution transformation itself can also be regarded as a projection gate that cannot be omitted: on the microstate space \(\Omega_{m+1}\), naive truncation
\(
r_{m+1\to m}:\Omega_{m+1}\to\Omega_m
\)
Generally not compatible with folding, while folding is aware of restriction
\(
\widetilde r_{m+1\to m}(\omega)=\pi_{m+1\to m}(\Fold_{m+1}(\omega))\in X_m
\)
is the reduced-resolution readout that is consistent across scales (section \ref{subsec:fold-omega-restriction}).
So in the ISA/microcode abstraction, we can put
$$
\mathrm{RESCALE}\downarrow:\ \Omega_{m+1}\to X_m,\qquad \omega\mapsto \widetilde r_{m+1\to m}(\omega)
$$
Treated as a primitive projection gate to account for costs.
\par
More importantly: the gate's \emph{space} projection support scale is linear in the worst case. Define the specification difference \(G_m(\omega)\) â€‹â€‹as
\(
|r_{m+1\to m}(\omega)\oplus \widetilde r_{m+1\to m}(\omega)|_0
\)
(definition \ref{def:fold-gauge-anomaly}), then the theorem \ref{thm:fold-gauge-anomaly-max} gives the worst lower bound \(G_m^{\max}=m\).
And under the uniform microstate baseline, its \emph{typical} support scale is also linear and has a closed-form constant density:
\(
\lim_{m\to\infty}\bar G_m/m=4/9
\)
(Theorem \ref{thm:fold-gauge-anomaly-density}), so on average about \((4/9)m\) â€‹â€‹outer bits need to be touched.
If this "supported rewrite amount" is recorded as \(T_{\downarrow}(m)\) â€‹â€‹as an independent non-parallel overhead item, then under typical circumstances
\[
T_{\downarrow}(m)\ \asymp\ \frac{4}{9}m.
\]
This shows that "online latency is constant" (causal cone length) does not rule out the worst case scenario of "reducing one-order resolution requires touching the entire surface register": the two describe orthogonal complexity components (latency vs. \  â€‹â€‹overall support scale).
\end{remark}

\begin{remark}[Atomic projection readout cost]\label{rem:pom-cost-prim}
If you need to read the primitive atomic layer spectrum, you can add it to the cost model
\[
k_{\mathrm{prim}}(I)\,T_{\mathrm{prim}}(n),
\]
Where $T_{\mathrm{prim}}(n)$ â€‹â€‹represents the readout of the orbital layer spectrum or the MÃ¶bius inversion depth; for arithmetic instructions, take $k_{\mathrm{prim}}=0$.
\end{remark}

\begin{remark}[The upper bound of time under the character projection door]\label{rem:pom-cost-chi}
When the sequence readout is implemented by $P_\chi$, $\tau_{\mathrm{mix}}$ can be used as the effective scale of $T_{\le}$ (see note \ref{rem:pom-chi-mixing}), thus obtaining
\[
T_I(n;m)\ \approx\ k_Z(I)\,\delta_Z\ +\ k_\chi(I)\,\tau_{\mathrm{mix}}\ +\ k_{\downarrow}(I)\,T_{\downarrow}(m)\ +\ D_U(n).
\]
This gives a direct quantified interface to the "projection word $\Rightarrow$ â€‹â€‹non-parallel overhead".
\end{remark}

\begin{remark}[Tri-orthogonal fingerprint of non-parallel overhead (delayed/supported/hybrid)]\label{rem:pom-nonparallel-three-meters}
Definition \ref{def:pom-cost} â€‹â€‹gives the total cost.
If the scope of "a priori intra-fiber linearization (scale/phase/prime registers, etc.) is an ontology structure and can be expanded in parallel", then $D_U(n)$ â€‹â€‹can be regarded as a parallel item and does not include the \emph{non-parallelizable} overhead.
Therefore, the part that is relevant to scale and cannot be eliminated through parallelization can be characterized by the following three mutually exclusive fingerprints:
\begin{itemize}
  \item \textbf{Delay fingerprint \(\delta\):}Specifies the minimum online delay (causal cone length) of projection gate \(P_Z\). For Zeckendorf normalization we have
  \(\delta_{\min}(\Fold_\infty)=3\) â€‹â€‹(proposition \ref{prop:pom-delay-min}), and increasing the space cannot push it to \(2\) (note \ref{rem:pom-space-time-orth}).
  \item \textbf{Support fingerprint \(G_m\):} The number of surface digits involved in cross-resolution reduction/writeback correction. Its worst case \(G_m^{\max}=m\) (theorem \ref{thm:fold-gauge-anomaly-max}), the typical density limit under a uniform baseline is
  \(\lim_{m\to\infty}\bar G_m/m=4/9\) â€‹â€‹(Theorem \ref{thm:fold-gauge-anomaly-density}), thus \(T_{\downarrow}(m)\asymp (4/9)m\) (note \ref{rem:pom-rescale-gauge-anomaly}).
  \item \textbf{Hybrid Fingerprint \(\tau_{\mathrm{mix}}\):} The convergence scale of the ordinal/congruential/character readout gate (\(P_{\le}\) or its characterized version \(P_\chi\)). The necessary depth to reach the error threshold \(\varepsilon\) is satisfied
  \[
  n\ \gtrsim\ \tau_{\mathrm{mix}}(u)\,\log(1/\varepsilon),
  \qquad
  \tau_{\mathrm{mix}}(u)=\frac{1}{\log(\lambda_1(u)/\Lambda(u))}
  \]
  (Corollary \ref{cor:pom-order-mixing}; note \ref{rem:pom-order-mixing-scale} and note \ref{rem:pom-chi-mixing}).
\end{itemize}
Under this scope, the non-parallel difference of "Four Rules and Shang Yu" is only reflected in whether the sequence gate appears in the projection word: if \(k_{\le}=0\), the non-parallel delayed main term is only a constant \(\delta_Z\);
If the semantics include canonical solution selection (sign, minimum nonnegative remainder, quotient remainder constraints \(0\le r<b\), etc.), then \(k_{\le}\ge 1\) is required and a growing term scaled by \(\tau_{\mathrm{mix}}\) is introduced.
If the instruction explicitly calls \(\mathrm{RESCALE}\downarrow\), the fingerprint \(G_m\) is supported to give an independent and linearly scaled writeback budget.
\par
In addition, the \emph{space} non-injectivity of the canonical projection gate \(P_Z\) can still be spectrally quantified by the Perron root \(\{r_q\}_{q\ge 2}\) (or the pressure curve) of the collision moment-kernel (theorem \ref{thm:pom-collision-kernel-ak}): This spectral fingerprint determines the tail position and the prime-register Budget certificate, but it describes the "same image collision intensity" and should not be confused with the above three types of fingerprints that cannot be read in parallel.
A direct and recomputable example is: after raising the input legality constraint to a 40-state real input kernel, the space side collision constant of "addition as a projection" changes significantly (Appendix \S\ref{app:add-collision-spectrum}).
\end{remark}

\subsection{Projected words with parameters \texorpdfstring{$2$}{2}-Category and auditable rewriting}\label{subsec:pom-2cat-rewrite}

\begin{proposition}[Additive composite law of error budget]\label{prop:pom-eps-sound-compose}
If \(w\Longrightarrow_{\varepsilon}w'\) â€‹â€‹and \(w'\Longrightarrow_{\delta}w''\), then \(w\Longrightarrow_{\varepsilon+\delta}w''\).

\[
\varepsilon_{\mathrm{tot}}\ \le\ \sum_{i=1}^{r}\varepsilon_i.
\]
\end{proposition}
\begin{proof}
Get the corresponding bad event \(E_1,E_2\).
There is \(Y_w=Y_{w'}=Y_{w''}\) on \(E_1^c\cap E_2^c\), so the bad event is \(E_1\cup E_2\), and its probability does not exceed \(\varepsilon+\delta\). Certification completed.
\end{proof}

\begin{theorem}[Generally $k$â€™s collision core and Perron master score]\label{thm:pom-collision-kernel-ak}

\[

\]

\end{theorem}
\begin{proof}[Proof idea (constructible, audit chain closed)]
The key input fact is: $\Fold_\infty$ â€‹â€‹can be realized by a fixed-delay deterministic finite state online transducer (theorem \ref{thm:online-delay-fold}), so the finite resolution $\Fold_m$ can also be regarded as the combined readout of "finite state synchronization graph + terminal function".

Let the state set of the transducer be $Q$, and the one-step transfer is given by the deterministic rule of "read one bit $\Rightarrow$ and output one bit + state update". Construct its $k$ resynchronization product automaton: the state is $Q^k$, read in $(d^{(1)},\dots,d^{(k)})\in\{0,1\}^k$ at the same time in one step, and get $k$ output bits $(e^{(1)},\dots,e^{(k)})\in\{0,1\}^k$ and the next state. Then impose \textbf{equality constraint} $e^{(1)}=\cdots=e^{(k)}$ (and include the finite output of the terminal function into the equality constraint: this can be achieved by incorporating the "terminal window" into the state). This results in a finite directed graph whose multiplicity of edges is equal to the number of input $k$-tuples that satisfy the equality constraint.

Therefore, the number of acceptable paths with length $m$ â€‹â€‹is exactly the count of "$k$ input words are transduced to the same output word", that is, $S_k(m)$. Taking the adjacency matrix of the graph as $A_k$, and using the start and end vectors to encode the initial state set and the limited modification of the terminal function, we get $S_k(m)=u_k^{\mathsf T}A_k^{m}v_k$ (the boundary terms of small $m$ can be absorbed into the "sufficiently large" threshold).

Finally, the limit is obtained by applying the Perron--Frobenius theorem to the intrinsically strongly connected components of $A_k$: the dominant spectral radius $r_k$ controls the exponential growth rate of the number of paths.
\end{proof}

\begin{proposition}[Transfer weights of symmetry quotient moment-kernel: computable in polynomial time]\label{prop:pom-ak-symmetric-transition-poly}
Following the structure of theorem \ref{thm:pom-collision-kernel-ak}, assume that the online synchronization core is a deterministic transducer: state set $Q$, input alphabet $\{0,1\}$, and there is a deterministic function
$$
\delta:Q\times\{0,1\}\to Q,\qquad \varepsilon:Q\times\{0,1\}\to\{0,1\},
$$
Among them, $\delta(q,d)$ â€‹â€‹is the next state and $\varepsilon(q,d)$ is the output bit. Let $k\ge 2$, and use the histogram state
$$
\nu\in\NN^{Q},\qquad \sum_{q\in Q}\nu(q)=k
$$
Indicates the number of occurrences of $k$ â€‹â€‹copies in each synchronization state.

Then for any output bit $e\in\{0,1\}$ â€‹â€‹and any two histograms $\nu,\nu'$, the weight transferred from $\nu$ to $\nu'$ in one step with the constraint of â€œbit-by-bit output is consistent and equal to $e$â€ can be written as
$$
w_e(\nu,\nu')
=
\sum_{\{a_q\}}
\ \prod_{q\in Q}\binom{\nu(q)}{a_q},
$$
where the summation is performed over all integer families $\{a_q\}_{q\in Q}$, satisfying:
\begin{itemize}
  \item \textbf{(Output Consistency Constraint)} For each $q\in Q$, there must be
  $$
  \varepsilon(q,0)\neq e\ \Rightarrow\ a_q=0,\qquad
  \varepsilon(q,1)\neq e\ \Rightarrow\ a_q=\nu(q),
  $$
  That is, $a_q$ â€‹â€‹only takes the value on the allowed input bits ($a_q$ is interpreted as "in the track of state $q$, select the number of input $0$");
  \item \textbf{(Next histogram constraint)} $\nu'$ â€‹â€‹is uniquely determined by $\{a_q\}$ and the requirements are met
  $$
  \nu'(q')
  =
  \sum_{q\in Q}\Bigl(
  a_q\,\mathbf{1}_{\delta(q,0)=q'}+(\nu(q)-a_q)\,\mathbf{1}_{\delta(q,1)=q'}
  \Bigr),
  \qquad \forall q'\in Q.
  $$
\end{itemize}
Therefore, the one-step adjacency weight of the symmetric quotient kernel is
$$
\widetilde A_k(\nu,\nu')=w_0(\nu,\nu')+w_1(\nu,\nu').
$$
And when $|Q|$ â€‹â€‹is fixed (@TK1@@ In this paper), given any $\nu$, all $w_e(\nu,\nu')$ (for all $\nu'$) can be calculated in $\mathrm{poly}(k)$ time by doing a state-by-state polynomial convolution/dynamic programming on $q\in Q$, so that the construction of \(\widetilde A_k\) is an executable polynomial-level process.
\end{proposition}
\begin{proof}[Point of proof (count = convolution with binomial coefficients)]
Fixed $\nu$ â€‹â€‹and output bit $e$. For each synchronization state $q$, the $\nu(q)$ tracks in $q$ are selected from the input bits $d\in\{0,1\}$ that satisfy $\varepsilon(q,d)=e$; when both inputs are allowed, the number of ways to select $a_q$ is to take $0$ and the rest to be $1$. $\binom{\nu(q)}{a_q}$;$a_q$ is forced to $0$ or $\nu(q)$ when only one input is allowed.

After all $\{a_q\}$ â€‹â€‹are given, the next state of each track is determined by $\delta(q,0)$ or $\delta(q,1)$, thus obtaining the unique next histogram $\nu'$, whose coordinates satisfy the linear constraints in the question. Summing over all feasible $\{a_q\}$ gives $w_e(\nu,\nu')$.

At the algorithm level, this is a set of counting convolutions that "select $\Rightarrow$ â€‹â€‹independently by state and aggregate by target state": treat the selection of each $q$ as a polynomial on the variable family $\{z_{q'}\}_{q'\in Q}$
$$
P_{q,e}(z):=\mathbf{1}_{\varepsilon(q,0)=e}\,z_{\delta(q,0)}+\mathbf{1}_{\varepsilon(q,1)=e}\,z_{\delta(q,1)},
$$
Then there is a generating function for fixed $\nu$
$$
\prod_{q\in Q} P_{q,e}(z)^{\nu(q)}
=\sum_{\nu'} w_e(\nu,\nu')\,z^{\nu'},
$$
Among them $z^{\nu'}:=\prod_{q'\in Q} z_{q'}^{\nu'(q')}$. When $|Q|$ is fixed, coefficient extraction can be implemented by state-by-state convolution DP, with a certain fixed degree polynomial of complexity $k$. Certification completed.
\end{proof}

\begin{corollary}[Coefficient extraction form: transfer polynomial generating function]\label{cor:pom-ak-coefficient-extraction}
Follow the notation of the proposition \ref{prop:pom-ak-symmetric-transition-poly}.
For each synchronization state \(q\in Q\) â€‹â€‹and output bit \(e\in\{0,1\}\), define the transition polynomial
\[
P_{q,e}(z):=\sum_{d\in\{0,1\}:\ \varepsilon(q,d)=e} z_{\delta(q,d)}.
\]
For the histogram state \(\nu\in\NN^Q\) â€‹â€‹(\(\sum_q\nu(q)=k\)), define
\[
F_{\nu,e}(z):=\prod_{q\in Q}P_{q,e}(z)^{\nu(q)}.
\]
Then there is a strict coefficient extraction identity
\[
F_{\nu,e}(z)=\sum_{\nu'} w_e(\nu,\nu')\,z^{\nu'},
\qquad
w_e(\nu,\nu')=[z^{\nu'}]\,F_{\nu,e}(z),
\]
Where \([z^{\nu'}]\) â€‹â€‹represents the coefficient of the monomial \(z^{\nu'}=\prod_{q'\in Q} z_{q'}^{\nu'(q')}\).
Therefore, the one-step weight of the symmetry quotient moment-kernel satisfies
\[

\]
\end{corollary}
