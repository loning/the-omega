\subsubsection{arity-charge 的大偏差率函数（\texorpdfstring{$\chi$}{chi} 切片）}\label{subsec:chi-ldp}
沿 $\chi$-切片 $\theta(t)=(t,0,-t)$，压力函数 $P_\chi(t)=\log\rho(M_{\theta(t)})$ 给出 $\chi$ 的指数型倾斜族；相应的大偏差率函数可由 Legendre 对偶直接读出。

\begin{theorem}[Gärtner--Ellis 率函数：\texorpdfstring{$\chi$}{chi} 的 LDP]\label{thm:chi-gartner-ellis}
设 $M_{\theta(t)}$ 在 $t$ 的某邻域内原始且 $P_\chi(t)=\log\rho(M_{\theta(t)})$ 可微（在本文的有限维正矩阵口径下在局部自动成立）。
令
\[
I_\chi(\alpha):=\sup_{t\in\RR}\bigl(t\alpha-P_\chi(t)\bigr).
\]
则对长度为 $n$ 的轨道上平均电荷 $\bar\chi_n:=\chi(\gamma)/n$（在对应的 Gibbs/倾斜平衡态下），其偏离事件满足大偏差原理，且好率函数为 $I_\chi$。
此外，典型值 $\bar\alpha=\mathbb{E}[\chi]=P_\chi'(0)$ 处有二次近似
\[
I_\chi(\bar\alpha+\delta)=\frac{\delta^2}{2P_\chi''(0)}+o(\delta^2),
\]
其中 $P_\chi''(0)=\mathrm{Var}_\infty(\chi)$ 为协方差密度。
\end{theorem}

\begin{remark}[与 Abel 有限部常数曲线的同一切片对比]
$P_\chi(t)$ 控制 leading 指数尺度（典型密度、涨落与大偏差），而 $\log\mathfrak{M}_\chi(t)$ 控制 prime-orbit Mertens 主项中的常数级漂移：
\[
\sum_{n\le N}\frac{p_{n,\chi}(t)}{\lambda_\chi(t)^n}=\log N+\gamma+\log\mathfrak{M}_\chi(t)+o(1),
\qquad
\lambda_\chi(t)=e^{P_\chi(t)}.
\]
因此在同一切片上同时记录 $t\mapsto(P_\chi(t),\log\mathfrak{M}_\chi(t))$，可把“指数尺度的代价”（由 $P_\chi$ 给出）与“常数级漂移的再分配”（由 $\log\mathfrak{M}_\chi$ 给出）作为两种互补指纹并列比较。
\end{remark}

\paragraph{把 \texorpdfstring{$\log\mathfrak{M}$}{logM} 完全对齐到显式 \texorpdfstring{$\zeta_M$}{zeta\_M}（闭式常数 + 数值）}
对 40 状态真实输入核，附录 \ref{app:real-input-40-abel} 给出
\[
\det(I-zM)=(1-z)^2(1+z)^3(1-3z+z^2)(1+z-z^2),
\]
因此主极点为 $z_\star=\varphi^{-2}$（对应 $\lambda=\varphi^2$）。并且主极点留数常数具有代数闭式
\[
C=\lim_{z\to z_\star}(1-\lambda z)\,\zeta_M(z)=\frac{47+21\sqrt5}{100}\approx 0.9395742753.
\]
将该 $C$ 代入上文的 Möbius--Abel 闭式即可得到 Abel/Hadamard 有限部常数与其导出的 Mertens 常数（数值见附录 \ref{app:real-input-40-finite-part}）：
\[
\log\mathfrak{M}\approx -0.254312661901716,\qquad
\mathfrak{M}\approx 0.7754493106833835,\qquad
\mathsf{M}=\gamma+\log\mathfrak{M}\approx 0.3229030029998169.
\]

\paragraph{可计算流程（与“离散化—正则化—有限部”同构）}
给定一个有限图移位的邻接矩阵 $M$（或加权矩阵族 $M_\theta$），并固定截断阶 $K\ge 2$，可按下述步骤计算有限部常数：
\begin{enumerate}
  \item \textbf{离散化：}由图/转导器得到矩阵 $M$（或 $M_\theta$），并定义 $\zeta(z)=1/\det(I-zM)$（或 $\zeta_\theta(z)=1/\det(I-zM_\theta)$）。
  \item \textbf{主尺度：}计算 Perron--Frobenius 主特征值 $\lambda=\rho(M)$（或 $\lambda(\theta)=\rho(M_\theta)$），并令 $z_\star=1/\lambda$（或 $z_\star(\theta)=1/\lambda(\theta)$）。
  \item \textbf{留数常数：}计算主极点留数常数
  \[
  C:=\lim_{z\to z_\star}(1-\lambda z)\zeta(z)
  \qquad
  (\text{或 }C(\theta):=\lim_{z\to z_\star(\theta)}(1-\lambda(\theta)z)\zeta_\theta(z)).
  \]
  在“简单极点”情形，$C$ 亦可由行列式导数或特征值乘积表达为闭式（见附录 \ref{app:real-input-40-finite-part}）。
  \item \textbf{正则化（Abel 采样）：}对 $k=2,\dots,K$，在收敛圆盘内采样 $\log\zeta(z_\star^k)$（或 $\log\zeta_\theta(z_\star(\theta)^k)$）。
  \item \textbf{有限部：}用 Möbius 尾的截断和近似
  \[
  \log\mathfrak{M}\approx \log C+\sum_{k=2}^{K}\frac{\mu(k)}{k}\log\zeta(z_\star^k)
  \quad
  (\text{或 }\log\mathfrak{M}(\theta)\approx \log C(\theta)+\sum_{k=2}^{K}\frac{\mu(k)}{k}\log\zeta_\theta(z_\star(\theta)^k)).
  \]
  \item \textbf{Mertens 常数输出：}定义
  \[
  \mathsf{M}:=\gamma+\log\mathfrak{M},
  \qquad
  \mathsf{M}(\theta):=\gamma+\log\mathfrak{M}(\theta).
  \]
\end{enumerate}
由于 $|z_\star|<1$，故 $z_\star^k$ 指数衰减，且 $\log\zeta(z)=O(z)$（$z\to 0$）使得 $\log\zeta(z_\star^k)$ 在 $k$ 上快速趋零，从而该流程在数值上具有很强的稳定性与快速收敛性。

\paragraph{orbit--Mertens 乘积（把 $\mathsf{M}$ 变成可对比标量）}
定义长度截断的 prime-orbit Mertens 乘积
\[
\mathcal{P}(N):=\prod_{\substack{\tau\ \mathrm{primitive}\\|\tau|\le N}}\left(1-\lambda^{-|\tau|}\right)^{-1}.
\]
取对数并用 $-\log(1-x)=x+O(x^2)$ 得
\[
\log\mathcal{P}(N)=\sum_{n\le N}\frac{p_n}{\lambda^n}+O(1)
=\log N+\mathsf{M}+o(1),
\]
因此
\[
\mathcal{P}(N)\sim e^{\mathsf{M}}\,N=e^{\gamma}\mathfrak{M}\,N.
\]
若用“轨道范数” $N(\tau):=\lambda^{|\tau|}$ 改写，并令 $x=\lambda^N$（等价于 $N=\log x/\log\lambda$），则得到等价的 $\log x$ 型主项
\[
\prod_{N(\tau)\le x}\left(1-\frac{1}{N(\tau)}\right)^{-1}
\sim \frac{e^{\gamma}\mathfrak{M}}{\log\lambda}\,\log x.
\]

\paragraph{加权接口与 \texorpdfstring{$\log\mathfrak{M}(\theta)$}{logM(theta)} 相图}
对边势向量 $\Phi$（例如附录 \ref{app:real-input-40-logm-theta} 中的 $\varphi_e,\varphi_-,\varphi_2$），定义加权矩阵族 $M_\theta$ 与
\(
\zeta_\theta(z)=1/\det(I-zM_\theta)
\)。
令 $\lambda(\theta)=\rho(M_\theta)$、$z_\star(\theta)=\lambda(\theta)^{-1}$。与上式同口径，定义
\[
P_\theta(z)=\sum_{k\ge 1}\frac{\mu(k)}{k}\log\zeta_\theta(z^k).
\]
\begin{definition}[Abel/Hadamard 有限部常数]\label{def:abel-finite-part}
对有限图移位的加权矩阵族 $M_\theta$，定义 Abel/Hadamard 有限部常数为
\[
\log\mathfrak{M}(\theta):=\mathrm{fp}_{r=1}\Bigl(P_\theta(r/\lambda(\theta))+\log(1-r)\Bigr),
\]
其中 $P_\theta(z)=\sum_{k\ge 1}\frac{\mu(k)}{k}\log\zeta_\theta(z^k)$ 为 Möbius--$\log\zeta$ 变换，$\lambda(\theta)=\rho(M_\theta)$ 为 Perron--Frobenius 主特征值，$\mathrm{fp}_{r=1}$ 表示在 $r=1$ 处的有限部（Laurent 常数项）。
\end{definition}
\begin{proposition}[Abel 有限部的 Möbius--$\zeta$ 闭式]\label{prop:abel-finite-part-mobius-zeta}
在定义 \ref{def:abel-finite-part} 的设定下，Abel 有限部常数具有可计算闭式
\[
\log\mathfrak{M}(\theta)
=\log C(\theta)+\sum_{k\ge 2}\frac{\mu(k)}{k}\log\zeta_\theta\bigl(z_\star(\theta)^k\bigr),
\qquad
C(\theta):=\lim_{z\to z_\star(\theta)}(1-\lambda(\theta)z)\,\zeta_\theta(z),
\]
其中 $C(\theta)$ 为主极点留数常数，$z_\star(\theta)=\lambda(\theta)^{-1}$ 为主极点位置。
\end{proposition}
等价地（与附录 \ref{app:real-input-40-logm-theta} 同口径），亦可写成一行 Möbius--Abel 形式
\[
\log\mathfrak{M}(\theta)=\log C(\theta)-\sum_{m\ge 2}\frac{P_\theta(\lambda(\theta)^{-m})}{m}.
\]
据此定义加权 Mertens 常数
\(
\mathsf{M}(\theta):=\gamma+\log\mathfrak{M}(\theta)
\)，从而把“离散化—正则化—有限部”落实为一条可扫参数的相图接口；数值相图入口脚本见
\texttt{scripts/exp\_sync\_kernel\_real\_input\_40\_logM\_theta.py}。

\paragraph{三维隐含属性的典型密度与协方差密度（\texorpdfstring{$\theta=0$}{theta=0}）}
在三势 $\theta=(\theta_e,\theta_-,\theta_2)$ 下，$P(\theta)=\log\rho(M_\theta)$ 的梯度与 Hessian 给出三条观测量的典型密度与协方差密度（详见附录 \ref{app:real-input-40-logm-theta} 的复现段落）：
\[
\nabla P(0)=\left(\frac{23-7\sqrt5}{20},\frac{5-\sqrt5}{20},\frac{3-\sqrt5}{10}\right)\approx(0.367376,\ 0.138197,\ 0.076393),
\]
\[
\nabla^2P(0)\approx
\begin{pmatrix}
0.048741 & 0.032574 & 0.015964\\
0.032574 & 0.275035 & 0.115735\\
0.015964 & 0.115735 & 0.067331
\end{pmatrix}.
\]
\begin{remark}[三通道二阶耦合的“中介结构”]
由 Hessian 的非对角项可见二阶耦合具有明显层级：
\[
H_{-2}>H_{e-}>H_{e2}.
\]
这意味着“负携带--碰撞”耦合最强，而“输出--碰撞”最弱；输出与负携带的相关性可被理解为在碰撞通道上被放大并重分配的结果。该层级关系是数值可审计的（见附录 \ref{app:real-input-40-logm-theta} 的复现脚本输出）。
\end{remark}

\paragraph{arity-charge 的同一接口并入}
定义 arity-charge
\[
\chi:=\varphi_e-\varphi_2\in\{-1,0,1\}.
\]
更一般地，若把 $\chi$ 作为一条额外势并入加权，则它只引入线性重参数化（无需扩展状态空间）：
\[
P(\theta_e,\theta_-,\theta_2,\theta_\chi)=P(\theta_e+\theta_\chi,\theta_-,\theta_2-\theta_\chi).
\]
在三参数压力 $P(\theta_e,\theta_-,\theta_2)=\log\rho(M_\theta)$ 下，对 $\chi$ 的线性加权对应一维切线
\[
P_\chi(t):=P(t,0,-t).
\]
因此 $\chi$ 的典型密度与方差密度由
\(
P_\chi'(0),P_\chi''(0)
\)
读出，并可与同一族 $\log\mathfrak{M}(\theta)$ 在该方向的切片一起作为“素数影子”的分叉维度；具体数值与可复现计算见附录 \ref{app:real-input-40-logm-theta} 与
\texttt{scripts/exp\_sync\_kernel\_real\_input\_40\_logM\_chi.py}。

\paragraph{\texorpdfstring{$\chi$}{chi} 的一维切片相图（可计算流程）}
固定
\(
\theta(t)=(t,0,-t)
\)
并令
\[
P_\chi(t):=P(\theta(t))=\log\rho(M_{\theta(t)}),
\qquad
\log\mathfrak{M}_\chi(t):=\log\mathfrak{M}(\theta(t)).
\]
对每个 $t$，按上文“可计算流程”的步骤计算 $\lambda(t)=e^{P_\chi(t)}$、$z_\star(t)=\lambda(t)^{-1}$、$C(t)$，并用截断 Möbius 尾近似
\[
\log\mathfrak{M}_\chi(t)\approx \log C(t)+\sum_{k=2}^{K}\frac{\mu(k)}{k}\log\zeta_{\theta(t)}(z_\star(t)^k).
\]
该一维切片在 $t=0$ 的统计量由压力导数给出：
\[
\mathbb{E}[\chi]=P_\chi'(0),\qquad \mathrm{Var}_\infty(\chi)=P_\chi''(0),
\]
并且在真实输入核上（附录 \ref{app:real-input-40-logm-theta}）有闭式与数值
\[
\mathbb{E}[\chi]=\frac{17-5\sqrt5}{20}\approx 0.2909830056,
\qquad
\mathrm{Var}_\infty(\chi)\approx 0.0841438530.
\]
若记三维 Hessian（协方差密度）为 $H=\nabla^2P(0)$，则
\[
\mathrm{Var}_\infty(\chi)=H_{ee}+H_{22}-2H_{e2},
\qquad
\mathrm{Cov}_\infty(\chi,\varphi_-)=H_{e-}-H_{-2}\approx -0.0831619.
\]
