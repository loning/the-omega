\section{Six-Dimensional Cut-and-Project Ontology and Observer Parameterization}

\subsection{Ontological Object of Cut-and-Project Scheme}

\begin{definition}[Cut-and-project data]\label{def:cap-data}
Given the lattice $L=\ZZ^6\subset\RR^6$, choose a direct sum decomposition
\[
\RR^6=E_{\parallel}\oplus E_{\perp},\qquad \dim E_{\parallel}=\dim E_{\perp}=3,
\]
and let $\pi_{\parallel},\pi_{\perp}$ be the linear projections to $E_{\parallel},E_{\perp}$ respectively. Take a compact window (acceptance domain)
\[
W\subset E_{\perp},
\]
and an internal phase (window shift)
\[
u\in E_{\perp}.
\]
Define the point set
\[
\Lambda(u):=\Bigl\{\pi_{\parallel}(n):\ n\in L,\ \pi_{\perp}(n)+u\in W\Bigr\}\subset E_{\parallel}.
\]
\end{definition}

\begin{remark}
If $\pi_{\parallel}(L)$ is dense in $E_{\parallel}$ and $\partial W$ satisfies moderate regularity (for instance, boundary has zero measure), then $\Lambda(u)$ is a typical three-dimensional quasiperiodic Delone set, and different $u$ correspond to different local realizations.
\end{remark}

\subsection{Realization of the ``Golden'' Structure (Model Class)}

\begin{assumption}[Golden field parameterization]\label{assump:golden-field}
Take the number field $K=\QQ(\varph)$ where $\varph$ is the golden ratio, requiring the matrix coefficients of $\pi_{\parallel},\pi_{\perp}$ to belong to $K$, and satisfy under Galois conjugation $\sigma:\varph\mapsto \varph'=-1/\varph$ the relation
\[
\pi_{\perp}=\sigma(\pi_{\parallel})
\]
(or equivalent conjugacy consistency). This assumption ensures: physical and internal subspaces are algebraically conjugate, the golden ratio serves as a controlling parameter in projection slope and window geometry.
\end{assumption}

\begin{remark}
This assumption belongs to the ``model class'' rather than a required axiom; subsequent closed-layer conclusions about Sturmian/Zeckendorf/information rate require only ``translation/rotation dynamics on compact abelian groups + binary readout + stabilization kernel,'' not specific six-dimensional coordinate expressions.
\end{remark}

\subsection{Rigorous Definition of Observer}

\begin{definition}[Observer as realization parameter]\label{def:observer}
An observer is defined as a four-tuple
\[
\mathcal{O}:=(E_{\parallel},E_{\perp},W,u),
\]
where $(E_{\parallel},E_{\perp},W)$ fixed describes the same ``ontology scheme,'' while $u\in E_{\perp}$ describes different realizations of the same ontology (different cross-section phase/window shift). The physical point set determined by $\mathcal{O}$ is $\Lambda(u)$.
\end{definition}

\subsection{Observer--Spacetime Duality: Readout Operators and Image (Interface)}

We lift the ``observer'' to a readout protocol parameter. Let
$$
O=(\mathcal{S},u,\alpha,\mathcal{P},m,\mathrm{Fold}),
$$
where $\mathcal{S}$ denotes geometric choice (for instance, $E_{\parallel},E_{\perp},W$, etc.), $u$ is the internal phase, $\alpha$ and $\mathcal{P}$ provide torus sweep and binary partition, $m$ is window length, and $\mathrm{Fold}=\mathrm{Fold}_m$ is the stabilization kernel.

\begin{assumption}[Main text fixed bandwidth]\label{assump:m6-fixed-main-chain}
Unless otherwise stated, the main text fixes $m=6$ and $\mathrm{Fold}=\mathrm{Fold}_6$.
\end{assumption}

Let $\Pi_O$ denote the geometry--symbol factor mapping induced by $(\mathcal{S},u,\alpha,\mathcal{P})$, and $q_m$ the finite window readout of length $m$. Define the readout operator
$$
\mathcal{R}_{O,m}=\mathrm{Fold}_m\circ q_m\circ \Pi_O.
$$
For orbits $s_{t+1}=T(s_t)$, define the visible sequence $x_t=\mathcal{R}_{O,m}(s_t)\in X_m^{\mathrm Z}$. Thus ``spacetime image'' is objectified as $\mathrm{Im}(\mathcal{R}_{O,m})$ and its induced structures (transfer kernel, reachability graph, and cylinder set conditions).

\begin{definition}[Local patch and patch frequency]\label{def:patch-frequency}
For $R>0$, let $B_R\subset E_{\parallel}$ be a ball of radius $R$. Define the $R$-patch centered at $x\in\Lambda(u)$ as
\[
P_R(x;u):=(\Lambda(u)-x)\cap B_R.
\]
For a given patch type $P$, if the limit
\[
\mathrm{freq}(P;u):=\lim_{R\to\infty}\frac{\card{\{x\in\Lambda(u)\cap B_R:\ P_R(x;u)\cong P\}}}{\card{\Lambda(u)\cap B_R}}
\]
exists, we call it the patch frequency.
\end{definition}
