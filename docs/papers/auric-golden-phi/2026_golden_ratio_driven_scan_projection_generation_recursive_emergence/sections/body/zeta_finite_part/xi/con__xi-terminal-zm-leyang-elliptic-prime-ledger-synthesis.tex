\begin{theorem}[导数分歧与 Lee--Yang 分歧的唯一公共素数]\label{thm:xi-terminal-zm-leyang-elliptic-unique-common-prime}
在
\[
E:\ Y^2=X^3-X+\tfrac14,\qquad y:=X^2+Y-\tfrac12
\]
的设定下，记倍点 Latt\`es 映射分子分母为
\[
N(X):=X^4+2X^2-2X+1,\qquad D(X):=4X^3-4X+1.
\]
则成立：
\begin{enumerate}
  \item \(\operatorname{Res}(N,D)=37^2\)，故 \(\Phi(X)=N(X)/D(X)\) 仅在 \(p=37\) 发生稳定约化降阶（由次数 \(4\) 降至次数 \(2\)）；
  \item 三条 Lee--Yang 控制三次
  \[
  C(y)=16y^3-8y^2-4y+1,\quad
  P_{\mathrm{LY}}(y)=256y^3+411y^2+165y+32,\quad
  Q_{\mathrm{LY}}(y)=256y^3+195y^2+93y-48
  \]
  的判别式满足
  \[
  \Disc(C)=2^8\cdot 37,\quad
  \Disc(P_{\mathrm{LY}})=-3^9\cdot 31^2\cdot 37,\quad
  \Disc(Q_{\mathrm{LY}})=-3^3\cdot 37\cdot 2677^2,
  \]
  因而其公共素因子集合恰为 \(\{37\}\)；
  \item 素数 \(37\) 是唯一同时触发“倍点动力系统次数降阶”与“Lee--Yang 分支控制多项式不可避免重根碰撞”的素数，并与曲线的坏约化素一致。对任意 \(p\neq 37\) 有
  \[
  \#E(\FF_p)=p+1-a_p,
  \]
  其中 \(a_p\) 与 level \(37\)、weight \(2\) 新形式的 Hecke 本征值一致。
\end{enumerate}
\end{theorem}
\begin{proof}
第一条由结论 \ref{con:xi-terminal-zm-elliptic-lattes-badprime-37-degree-drop} 的结果式与模 \(37\) 显式分解直接给出。第二条由结论 \ref{con:xi-terminal-zm-leyang-badprime-37-common-double-root} 的判别式分解直接给出，三者公共素因子交集即 \(\{37\}\)。第三条将前两条合并，并与注记 \ref{rem:xi-terminal-elliptic-37a1-modular-anchor} 的模性锚定对齐即可。证毕。
\end{proof}

\begin{theorem}[Lee--Yang 观测量在倍点动力下的显式四分支代数递归]\label{thm:xi-terminal-zm-leyang-elliptic-four-branch-recursion}
令
\[
y_0:=y(P),\qquad y_1:=y([2]P).
\]
则存在整系数代数关系
\[
\mathcal H(y_0,y_1)=0,\qquad \mathcal H\in\ZZ[y_0,y_1],
\]
其在 \((y_0,y_1)\) 的双次数为 \((16,4)\)，并可写为
\[
\mathcal H(y_0,y_1)=A_4(y_0)y_1^4+A_3(y_0)y_1^3+A_2(y_0)y_1^2+A_1(y_0)y_1+A_0(y_0).
\]
其中显式系数见结论 \ref{con:xi-terminal-zm-elliptic-weight-doubling-modular-equation} 的闭式证书。进一步有判别式分解
\[
\Disc_{y_1}\!\bigl(\mathcal H(y_0,\cdot)\bigr)
=-y_0(y_0-1)P_{\mathrm{LY}}(y_0)\,Q_{12}(y_0)^2Q_{26}(y_0)^2,
\]
其中 \(Q_{12},Q_{26}\in\ZZ[y_0]\) 分别为次数 \(12,26\) 多项式。于是四分支递归的平方自由分歧集合为
\[
y_0(y_0-1)P_{\mathrm{LY}}(y_0)=0.
\]
\end{theorem}
\begin{proof}
由结论 \ref{con:xi-terminal-zm-elliptic-weight-doubling-modular-equation}，\((y_0,y_1)\) 满足双次数 \((16,4)\) 的四次模方程 \(\Phi_2(y_0,y_1)=0\)，取 \(\mathcal H:=\Phi_2\) 即得。判别式分解由结论 \ref{con:xi-terminal-zm-elliptic-weight-doubling-disc-squareclass} 直接给出；去平方因子后得到所示分歧集合。证毕。
\end{proof}

\begin{corollary}[倍点时间链的边际平稳性]\label{cor:xi-terminal-zm-leyang-elliptic-doubling-stationarity}

\begin{proof}
\([2]:E(\CC)\to E(\CC)\) 为满射连续群同态，保持 Haar 测度。故 \([2^n]\) 亦保持 Haar 测度，进而 \(y_n\) 的分布恒为 \(y_*(\mathrm{Haar})\)。证毕。
\end{proof}

\begin{theorem}[Lee--Yang 非解析集合的扰动稳定性与对偶根数据重构]\label{thm:xi-terminal-zm-leyang-langlands-dual-root-stability}
设 \(G\) 为连通复约化群，\(\lambda\) 为正则最高权，\(W\) 为 Weyl 群，\(P_\lambda:=\operatorname{conv}(W\!\cdot\!\lambda)\)。
令
\[
E^{\mathrm{LY}}_\lambda(\mu):=\log\rho\!\bigl(V_\lambda(\exp\mu)\bigr),\qquad \mu\in\mathfrak a^+,
\]
并取文中恒等式 \(E^{\mathrm{LY}}_\lambda=h_{P_\lambda}\)。
则存在 \(\varepsilon_\lambda>0\)，使得对任意凸紧集 \(K\) 的支撑函数 \(h_K\)，若
\[
\|h_K-h_{P_\lambda}\|_\infty<\varepsilon_\lambda,
\]
则 \(h_K\) 与 \(h_{P_\lambda}\) 具有同一法扇组合型，故其不可微超平面排列完全一致。于是可由 \(h_K\) 在同源意义下唯一重构 Langlands 对偶根数据。
\end{theorem}
\begin{proof}
支撑函数与 Hausdorff 距离满足
\(
d_H(K,P_\lambda)=\|h_K-h_{P_\lambda}\|_\infty
\)。
当 \(\lambda\) 正则时，\(P_\lambda\) 的法扇即 Weyl 扇；该组合型由有限组严格线性不等式刻画，因此在支撑函数空间对应开邻域。取其半径为 \(\varepsilon_\lambda\) 的一致范数球，则其中任意 \(h_K\) 保持同一法扇。不可微集合是法扇的余维一骨架，故随之不变。Weyl 扇确定对偶根系并进一步确定对偶根数据（至同源），结论成立。证毕。
\end{proof}

\begin{theorem}[圆维约束下 Gödel 素数账本的 Lambert--\(W\) 精细下界]\label{thm:xi-terminal-zm-cdim-prime-ledger-lambertw-lower-bound}
设有限支撑指数向量 \(r:\mathbb P\to\NN\) 的 Gödel 账本编码为
\[
N(r):=\prod_{p}p^{r(p)}.
\]
令 \(\cdim(G):=\dim(\widehat G^\circ)\)，并记 \(r_0:=\cdim(G)\in\NN\)。
若系统需记录 \(N\) 个可区分状态且保持 \(r_0\) 个独立相位通道可审计，则所需素数轴数 \(k\) 满足
\[
k\ge \frac{r_0\log N}{\mathrm W(r_0\log N)}\,(1-o(1)),
\]
其中 \(\mathrm W\) 为 Lambert--\(W\) 函数主支。等价地，
\[
k\ge
\frac{r_0\log N}
{\log(r_0\log N)-\log\log(r_0\log N)}\,(1+o(1)).
\]
\end{theorem}
\begin{proof}
记 primorial
\[
P_k^\#:=\prod_{i=1}^k p_i,\qquad \log P_k^\#=\vartheta(p_k),
\]
其中 \(\vartheta(x)=\sum_{p\le x}\log p\)。由素数定理，
\[
\vartheta(p_k)\sim p_k\sim k\log k.
\]
可审计容量要求给出 \(\log P_k^\#\gtrsim r_0\log N\)。设
\(
A:=r_0\log N
\)，则必须有
\[
k\log k\gtrsim A.
\]
方程 \(k\log k=A\) 的解为
\[
k=\frac{A}{\mathrm W(A)}.
\]
故得到
\[
k\ge \frac{A}{\mathrm W(A)}(1-o(1))
=\frac{r_0\log N}{\mathrm W(r_0\log N)}(1-o(1)).
\]
再由
\(
\mathrm W(A)=\log A-\log\log A+o(1)
\)
即得第二式。证毕。
\end{proof}

\begin{theorem}[临界值平方性与偶符号亏损]\label{thm:xi-terminal-zm-jg-critical-square-even-sign-defect}
设 \(K\) 为特征不为 \(2\) 的域，\(P(z)=a_n z^n+\cdots+a_0\in K[z]\) 可分离且 \(a_n a_0\neq 0\)。令 \(F:=K(r)\)，并定义
\[
Q_r(w):=\operatorname{Res}_z\!\bigl(P(z),\,r z^2-w z+r^{-1}\bigr)\in F[w].
\]
记 \(w_1,\dots,w_n\) 为 \(Q_r\) 在某一分裂域中的根（按重数），并置 \(\Delta_i:=w_i^2-4\)。则：
\begin{enumerate}
  \item
  \[
  Q_r(2)=r^n P(1/r)^2,\qquad Q_r(-2)=r^n P(-1/r)^2.
  \]
  \item
  \[
  \prod_{i=1}^n (w_i^2-4)=
  \left(\frac{r^n P(1/r)P(-1/r)}{a_n a_0}\right)^2\in F.
  \]
  特别地，若 \(P(1/r)P(-1/r)\neq 0\)，则该乘积在 \(F^\times/(F^\times)^2\) 中为平凡类。
  \item 若 \(P(1/r)P(-1/r)\neq 0\)，设 \(M/F\) 为 \(Q_r\) 的分裂域，并令
  \[
  N:=M\bigl(\sqrt{\Delta_1},\dots,\sqrt{\Delta_n}\bigr),
  \]
  则
  \[
  [N:M]\mid 2^{n-1},
  \]
  且在 Kummer 理论同构下有
  \[
  \operatorname{Gal}(N/M)\subseteq
  \left\{
  (\varepsilon_1,\dots,\varepsilon_n)\in (\ZZ/2\ZZ)^n:\ \varepsilon_1+\cdots+\varepsilon_n=0
  \right\}.
  \]
\end{enumerate}
\end{theorem}
\begin{proof}
设 \(z_1,\dots,z_n\) 为 \(P\) 的根。由结式根表示，
\[
Q_r(w)=a_n^2\prod_{j=1}^n \bigl(r z_j^2-w z_j+r^{-1}\bigr).
\]
令 \(w=2\)：
\[
r z_j^2-2z_j+r^{-1}=r^{-1}(r z_j-1)^2,
\]
从而
\[
Q_r(2)=a_n^2 r^{-n}\Bigl(\prod_{j=1}^n (r z_j-1)\Bigr)^2.
\]
又
\[
\prod_{j=1}^n (r z_j-1)
=(-1)^n r^n \frac{P(1/r)}{a_n},
\]
故 \(Q_r(2)=r^n P(1/r)^2\)。同理得 \(Q_r(-2)=r^n P(-1/r)^2\)。

再由
\[
Q_r(w)=(a_n a_0)\prod_{i=1}^n (w-w_i)
\]
可得
\[
Q_r(2)Q_r(-2)
=(a_n a_0)^2\prod_{i=1}^n (2-w_i)(-2-w_i)
=(a_n a_0)^2\prod_{i=1}^n (w_i^2-4).
\]
代入前述两式即得第二条。

在第三条条件下，\(\Delta_i\in M^\times\)。设
\[
V:=\operatorname{span}_{\FF_2}\{[\Delta_1],\dots,[\Delta_n]\}\subset M^\times/(M^\times)^2.
\]
Kummer 理论给出
\(
\operatorname{Gal}(N/M)\cong V^\vee
\)
与
\(
[N:M]=2^{\dim_{\FF_2}V}
\)。
第二条说明
\(
\sum_i [\Delta_i]=0
\)，故 \(\dim_{\FF_2}V\le n-1\)，于是 \([N:M]\mid 2^{n-1}\)；
并且对应符号翻转向量满足偶校验约束。证毕。
\end{proof}

\begin{corollary}[二次纤维上界的改进与 \(D_n\) 型收缩]\label{cor:xi-terminal-zm-jg-fiber-bound-dn-contraction}
在定理 \ref{thm:xi-terminal-zm-jg-critical-square-even-sign-defect} 第三条设定下，令 \(L/F\) 为 \(P\) 的分裂域。则
\[
L\subseteq N,\qquad [L:M]\mid 2^{n-1}.
\]
因此 \(\operatorname{Gal}(L/M)\) 的符号部分落入偶校验子群；相应超正交上界由 \(W(B_n)\) 强制收缩至 \(W(D_n)\)。
\end{corollary}
\begin{proof}
由
\[
z=\frac{w_i\pm\sqrt{w_i^2-4}}{2r}
\]
可知 \(P\) 的各根均落在 \(N\) 中，故 \(L\subseteq N\)。再由
\(
[L:M]\mid [N:M]
\)
与定理 \ref{thm:xi-terminal-zm-jg-critical-square-even-sign-defect} 第三条即得。证毕。
\end{proof}

\begin{theorem}[代数尺度的有限例外相干性]\label{thm:xi-terminal-zm-jg-algebraic-scale-finite-exception-coherence}
设 \(K\) 为特征不为 \(2\) 的域，\(P\in K[z]\) 可分离且 \(a_n a_0\neq 0\)，\(r\in K^\times\)。
记 \(L/K\) 与 \(M/K\) 分别为 \(P\) 与 \(Q_r\) 的分裂域。若对任意根对 \((z_i,z_j)\) 均有
\[
r^2\neq (z_i z_j)^{-1},
\]
则
\[
M=L,\qquad \operatorname{Gal}(Q_r/K)\cong \operatorname{Gal}(P/K).
\]
并且对固定 \(P\)，使上述条件失效的 \(r\in K^\times\) 仅落在有限集合
\[
\mathcal E_P(K):=
\left\{
r\in K^\times:\ \exists\,i,j,\ r^2=(z_i z_j)^{-1}
\right\}
\]
中。
\end{theorem}
\begin{proof}
由 \(w_i=r z_i+r^{-1}z_i^{-1}\in L\) 得 \(M\subseteq L\)。

反向包含：固定 \(i\)，令 \(w_i:=r z_i+r^{-1}z_i^{-1}\in M\)，并设
\[
g_i(z):=r z^2-w_i z+r^{-1}\in M[z].
\]
其两根为 \(z_i\) 与 \(r^{-2}z_i^{-1}\)。条件保证 \(r^{-2}z_i^{-1}\) 不是 \(P\) 的根，故 \(\gcd(P,g_i)\) 在 \(M[z]\) 中为一次多项式，且唯一根为 \(z_i\)。因此 \(z_i\in M\)。对全部 \(i\) 成立，得 \(L\subseteq M\)，从而 \(L=M\)。

有限性由集合 \(\{(z_i z_j)^{-1}\}_{i,j}\) 有限立即推出：每个给定值在 \(K\) 中至多产生两个平方根，故 \(\mathcal E_P(K)\) 有限。证毕。
\end{proof}

\begin{theorem}[Lee--Yang 单尺度解析反演与吸引分支唯一性]\label{thm:xi-terminal-zm-leyang-single-scale-analytic-inversion}
设 \(P\in\CC[z]\) 可分离且全部根满足 \(|z_i|=1\)，取实尺度 \(r>1\)。定义
\[
E_r:=\{\,w=r z+r^{-1}z^{-1}:|z|=1\,\},
\qquad
Q_r(w):=\operatorname{Res}_z\!\bigl(P(z),\,r z^2-w z+r^{-1}\bigr).
\]
则对 \(Q_r\) 的任一根 \(w\)，二次方程
\[
r z^2-w z+r^{-1}=0
\]
有两解
\[
z_\pm=\frac{w\pm\sqrt{w^2-4}}{2r},
\]
并满足
\[
|z_+|=1,\qquad |z_-|=r^{-2}
\]
（按适当重标记）。因此存在唯一分支选择
\[
s_r:\{Q_r\text{ 的根}\}\to \{P\text{ 的根}\},
\]
其中 \(s_r(w)\) 定义为唯一满足 \(|z|=1\) 的解。该选择恢复 \(P\) 的根多重集，进而由 Vieta 对称多项式唯一恢复 \(P\) 的系数。
\end{theorem}
\begin{proof}
若 \(w\) 是 \(Q_r\) 的根，则 \(w=r z_i+r^{-1}z_i^{-1}\) 对某根 \(z_i\) 成立，且对应二次方程的两根正是 \(z_i\) 与 \(r^{-2}z_i^{-1}\)。
由 \(|z_i|=1\) 得另一根模长为 \(r^{-2}\)。因 \(r>1\)，二者模长严格分离，故单位圆分支唯一。对全部根实施该选择得到 \(P\) 的根多重集，再由 Vieta 公式恢复系数。证毕。
\end{proof}

\begin{theorem}[Gödel 编码下的整数平方审计证书]\label{thm:xi-terminal-zm-godel-integer-transport-square-audit}
设 \(P\in\ZZ[z]\)，取互素整数 \(a,b\ge 1\)，令 \(r=a/b\)。定义清分母后的输运结式
\[
Q_{a,b}(w):=\operatorname{Res}_z\!\bigl(P(z),\,a^2 z^2-a b\,w z+b^2\bigr)\in\ZZ[w].
\]
则
\[
Q_{a,b}(2),\ Q_{a,b}(-2)\in\ZZ
\]
均为完全平方；并且
\[
\operatorname{Res}_w\!\bigl(Q_{a,b}(w),\,w^2-4\bigr)=Q_{a,b}(2)\,Q_{a,b}(-2)
\]
同样为完全平方。
\end{theorem}
\begin{proof}
代入 \(w=2\) 与 \(w=-2\) 分别得
\[
a^2 z^2-2ab z+b^2=(a z-b)^2,\qquad
a^2 z^2+2ab z+b^2=(a z+b)^2.
\]
故
\[
Q_{a,b}(2)=\operatorname{Res}_z\!\bigl(P(z),(a z-b)^2\bigr)
=\operatorname{Res}_z\!\bigl(P(z),a z-b\bigr)^2,
\]
\[
Q_{a,b}(-2)=\operatorname{Res}_z\!\bigl(P(z),(a z+b)^2\bigr)
=\operatorname{Res}_z\!\bigl(P(z),a z+b\bigr)^2.
\]
两者皆为整数平方。再由结式对因子分解的乘性
\[
\operatorname{Res}_w\!\bigl(Q_{a,b}(w),w^2-4\bigr)
=\operatorname{Res}_w\!\bigl(Q_{a,b}(w),w-2\bigr)\,
\operatorname{Res}_w\!\bigl(Q_{a,b}(w),w+2\bigr)
=Q_{a,b}(2)Q_{a,b}(-2),
\]
即得结论。证毕。
\end{proof}
