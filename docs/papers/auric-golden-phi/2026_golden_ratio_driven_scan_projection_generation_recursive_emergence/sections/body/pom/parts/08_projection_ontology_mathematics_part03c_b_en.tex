\begin{proposition}[Odd-lag neutrality (auditable)]\label{prop:pom-moment-odd-lag-neutrality}
Within the verified range \(q=2,\dots,23\) (Tables \ref{tab:fold_collision_moment_recursions}, \ref{tab:fold_collision_moment_recursions_9_17}, and \ref{tab:fold_collision_moment_recursions_18_23}), the minimal integer-coefficient recurrence for \(S_q(m)\) can be written as
\[
S_q(m)=\sum_{i=1}^{d_q} c_i\,S_q(m-i)\qquad(m\ge m_0),
\]
where the order \(d_q\) is always odd and satisfies the \textbf{odd-lag coefficient zero-sum}:
\[
\sum_{\substack{1\le i\le d_q\\ i\ \mathrm{odd}}} c_i=0.
\]
Equivalently, under the convention
\[
P_q(\lambda)=\lambda^{d_q}-\sum_{i=1}^{d_q}c_i\,\lambda^{d_q-i},
\]
we have
\[
P_q(1)+P_q(-1)=0.
\]
That is, the \textbf{sum of even-power coefficients of \(P_q\) is \(0\)} (since when \(d_q\) is odd, even-power terms correspond exactly to odd-lag terms in the recurrence).
\end{proposition}

\begin{proof}
Since \(d_q\) is odd, \(1^{d_q}+(-1)^{d_q}=0\), hence
\[
P_q(1)+P_q(-1)
= -\sum_{i=1}^{d_q} c_i\bigl(1^{d_q-i}+(-1)^{d_q-i}\bigr)
= -2\sum_{\substack{1\le i\le d_q\\ i\ \mathrm{odd}}}c_i,
\]
thus \(\sum_{i\ \mathrm{odd}}c_i=0\iff P_q(1)+P_q(-1)=0\).
In the above three tables, \((c_1,\dots,c_{d_q})\) are explicitly given term-by-term; for each \(q\), directly summing the odd-index terms yields the zero-sum identity, proving the proposition.
\end{proof}
