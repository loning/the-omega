\section{Emergence of Arithmetic: Constructing $(\oplus,\otimes)$ on Stable Address Space}\label{sec:emergent-arithmetic}

This section in situ constructs natural number objects and addition, multiplication operations on stable address space under POMmin, and proves the resulting structure is isomorphic to the classical semiring. We take stable address space as the finite-support part of Zeckendorf-valid digits, namely $Z$ in Definition \ref{def:Z-fiber}; notation-wise it can also be viewed as the finite-support sub-object of $X_\infty$.\par

\subsection{Semiring structure of natural numbers via quotient}
Define the finite-support digit configuration space
$$
D:=\Bigl\{a=(a_k)_{k\ge 1}: a_k\in\NN,\ \#\{k:a_k\neq 0\}<\infty\Bigr\}.
$$
Let the Fibonacci sequence $(F_k)_{k\ge 1}$ satisfy $F_1=F_2=1$, $F_{k+2}=F_{k+1}+F_k$, and define the weighted value
$$
\mathrm{Val}(a):=\sum_{k\ge 1} a_k F_{k+1}.
$$
On $D$ introduce an equivalence relation $\equiv$ generated by base and recursion relations, such that basis vectors satisfy
$$
[e_2]=2[e_1],\qquad [e_{k+2}]=[e_{k+1}]+[e_k]\ \ (k\ge 1),
$$
where $e_k$ denotes the unit vector in position $k$.\par

\begin{theorem}[Emergence of natural numbers: quotient semiring isomorphism]\label{thm:emergent-N}
The map $\mathrm{Val}:D/\!\equiv\ \to \NN$ is a commutative semiring isomorphism; and the Zeckendorf-valid digit space $Z$ provides a unique canonical representative for each equivalence class: for any $a\in D$, there exists a unique $c\in Z$ such that $a\equiv c$ and $\mathrm{Val}(a)=\mathrm{Val}(c)$.\par
\noindent\textnormal{\textbf{Depends on:} Definition \ref{def:Z-fiber}, Zeckendorf unique representation theorem \cite{Zeckendorf1972}, and recursion relations of equivalence $\equiv$.}\par
\end{theorem}

\begin{proof}
By the Zeckendorf representation theorem \cite{Zeckendorf1972}, every $N\in\NN$ has a unique valid digit $Z(N)\in Z$ such that $\mathrm{Val}(Z(N))=N$, so $\mathrm{Val}$ is surjective. Further, in the quotient semiring $D/\!\equiv$, the recursion relations give $[e_k]=F_{k+1}[e_1]$, so any $[a]$ is uniquely characterized by $\mathrm{Val}(a)$, yielding isomorphism; existence and uniqueness of canonical representatives follow from Zeckendorf uniqueness.\par
\end{proof}

\subsection{Fold and normal form: emergence of addition}
For $c,d\in Z$ define bit-by-bit accumulation: let
$$
(c+d)_k:=c_k+d_k\in\{0,1,2\}.
$$
Since accumulation typically breaks valid syntax, we need to fold-normalize back to project to $Z$.\par

\begin{definition}[Stable addition]\label{def:stable-add}
Define
$$
c\oplus d := \Fold_\infty(c+d).
$$
Here $\Fold_\infty$ is taken to be the extended fold operator on configuration space $D$ as given in Definition \ref{def:fold-extension-domain} (its implementation and properties appear in Appendix \ref{app:fold-rewriting}). Equivalently, we can also define via the value map: $c\oplus d := \mathrm{Val}^{-1}(\mathrm{Val}(c)+\mathrm{Val}(d))$.\par
\end{definition}

\begin{proposition}[Algorithmic proposition for fold normalization]\label{prop:fold-normalize-alg}
For any $m\ge 1$ and any $\omega\in\Omega_m$, there exists a deterministic algorithm that in finite steps outputs the unique normal form $x=\Fold_m(\omega)\in X_m$, and can be accompanied by an auditable certificate chain consisting of single-step local rewrites for the verifier to check step-by-step (Appendix \ref{app:fold-rewriting}). Thus, fold normalization recovers the \emph{stable canonical representative}, not fiber-internal microstate differences.\par
\end{proposition}

\begin{corollary}[Fold formula]\label{cor:fold-add}
For any $c,d\in Z$, we have $c\oplus d=\Fold_\infty(c+d)$, and $\mathrm{Val}(c\oplus d)=\mathrm{Val}(c)+\mathrm{Val}(d)$.\par
\end{corollary}

\begin{theorem}[Definitional isomorphism of addition]\label{thm:add-isom}
$(Z,\oplus)$ is a commutative monoid, and
$$
\mathrm{Val}:(Z,\oplus)\to(\NN,+)
$$
is an isomorphism.\par
\end{theorem}

\begin{proof}
By definition $\mathrm{Val}(c\oplus d)=\mathrm{Val}(c)+\mathrm{Val}(d)$, and by the bijectivity of $\mathrm{Val}$, commutativity, associativity, and unit structure of $(\NN,+)$ transfer to $(Z,\oplus)$.\par
\end{proof}

\subsection{Emergence of multiplication and semiring isomorphism}
\begin{definition}[Stable multiplication]\label{def:stable-mul}
For $c,d\in Z$, define
$$
c\otimes d := \mathrm{Val}^{-1}\bigl(\mathrm{Val}(c)\cdot \mathrm{Val}(d)\bigr)\in Z.
$$
\end{definition}

\begin{theorem}[Definitional isomorphism of multiplication]\label{thm:mul-isom}
$(Z,\oplus,\otimes)$ is a commutative semiring, and
$$
\mathrm{Val}:(Z,\oplus,\otimes)\to (\NN,+,\times)
$$
is a semiring isomorphism.\par
\end{theorem}

\begin{proof}
By definition $\mathrm{Val}$ preserves both addition and multiplication, and $\mathrm{Val}$ is bijective; therefore commutativity, associativity, and distributivity transfer from $(\NN,+,\times)$.\par
\end{proof}

\subsection{Endogenization of multiplication and addition: minimal primitive set}
\begin{proposition}[Minimization of definition chain and cancellativity (definitional economy)]\label{prop:definitional-economy-arith}
In the construction of this section, the primitive set of arithmetic structure can be compressed to "fold normalization $\Fold_\infty$ (and its auditable implementation) $+$ zero $0_Z$ $+$ successor $S(c)=\Fold_\infty(c+e_1)$". On this primitive set:\par
\noindent\textbf{(i)} Stable addition $\oplus$ can be endogenously derived by successor iteration (see Remark \ref{rem:succ-smaller-primitives});\par
\noindent\textbf{(ii)} Stable multiplication $\otimes$ need not be a primitive: it can be completely eliminated by a derived operation $\odot$ defined by "addition iteration" (Theorem \ref{thm:mul-from-add}).\par
Thus, the definition of multiplication via $\mathrm{Val}^{-1}$ in Definition \ref{def:stable-mul} is merely shorthand for convenience: the content of the semiring isomorphism conclusion is that \emph{internally definable} arithmetic operations are equivalent to classical $(\NN,+,\times)$, not that classical operations are pre-installed as primitives.\par
\end{proposition}

\begin{theorem}[Multiplication can be endogenously derived from addition and iteration]\label{thm:mul-from-add}
Define
$$
c\odot d := \underbrace{c\oplus c\oplus\cdots\oplus c}_{\mathrm{Val}(d)\text{ times}},
\qquad
\text{if }\mathrm{Val}(d)=0\text{ take }0_Z,
$$
where $0_Z$ is the all-zero sequence. Then for any $c,d\in Z$, we have
$$
c\odot d = c\otimes d.
$$
\end{theorem}

\begin{proof}
Since $\mathrm{Val}$ preserves $\oplus$ and is bijective, we get
$$
\mathrm{Val}(c\odot d)=\mathrm{Val}(c)\mathrm{Val}(d)=\mathrm{Val}(c\otimes d),
$$
and by bijectivity we conclude equality.\par
\end{proof}

\begin{remark}[Successor operator and smaller primitive set]\label{rem:succ-smaller-primitives}
Similarly, we can endogenously derive $\oplus$ as successor iteration via the successor operator $S(c)=\Fold_\infty(c+e_1)$, thus obtaining an even smaller primitive set (arithmetic generated by "successor $+$ fold").\par
\end{remark}
