\paragraph{\texorpdfstring{$q$}{q}-Order Collision Moments Hardcore Homomorphization: From Fiber Independent Sets to Disjoint Target Graph}
Theorem \ref{thm:pom-fiber-indset-factorization} canonically aligns each fiber \(\Fold_m^{-1}(x)\) with the independent set family \(\mathsf{Ind}(\Gamma_x)\) of the candidate site-induced graph \(\Gamma_x\), so that
\(
d_m(x)=\card{\Fold_m^{-1}(x)}=\mathsf{I}(\Gamma_x)
:=\card{\mathsf{Ind}(\Gamma_x)}.
\)
Below we give a strict identity that lifts "exponentiation" to "homomorphism counting": the action of \(q\) is no longer manifested as numerical power operation, but rewritten as the local compatibility geometry of a universal target graph.

\begin{proposition}[Strict Identity Between Independent Set Powers and Disjoint Target Graph Homomorphism Counting]\label{prop:pom-indset-power-hom-disj}
Let \(\Gamma=(V,E)\) be a finite simple graph, and write its total number of independent sets as \(\mathsf{I}(\Gamma)\).
For any integer \(q\ge 1\), let \([q]:=\{1,2,\dots,q\}\).
Then the strict identity holds
\[
\boxed{\;
\mathsf{I}(\Gamma)^q
\;=\;
\#\Bigl\{\Phi:V\to 2^{[q]}\ \Bigm|\ \forall\,uv\in E,\ \Phi(u)\cap\Phi(v)=\varnothing\Bigr\}\; }.
\]
Define the disjoint target graph \(\mathsf{Disj}_q\) as: its vertex set is \(2^{[q]}\), with adjacency relation
\(
A\sim B \iff A\cap B=\varnothing
\)
(\(A,B\in 2^{[q]}\)).
Then the right-hand side is exactly the homomorphism count \(\mathrm{hom}(\Gamma,\mathsf{Disj}_q)\).
\end{proposition}
\begin{proof}
Given a \(q\)-tuple of independent sets \((S_1,\dots,S_q)\in\mathsf{Ind}(\Gamma)^q\), define
\[
\Phi(v):=\{i\in[q]:\ v\in S_i\}\qquad(v\in V).
\]
If \(uv\in E\), then for each \(i\), we cannot have both \(u,v\in S_i\), so \(\Phi(u)\cap\Phi(v)=\varnothing\).
Conversely, given any mapping \(\Phi:V\to 2^{[q]}\) satisfying the disjointness constraint, define
\[
S_i:=\{v\in V:\ i\in\Phi(v)\}\qquad(1\le i\le q).
\]
If \(uv\in E\) and \(u,v\in S_i\), then \(i\in\Phi(u)\cap\Phi(v)\), contradicting the disjointness constraint; hence each \(S_i\) is an independent set.
These two constructions are mutual inverses, yielding the stated counting identity.
Finally, graph homomorphisms \(\Gamma\to\mathsf{Disj}_q\) are exactly mappings \(V\to 2^{[q]}\) satisfying "edge \(\mapsto\) edge," i.e., forcing the image sets to be disjoint for each edge, so the right-hand side equals \(\mathrm{hom}(\Gamma,\mathsf{Disj}_q)\).
\end{proof}

\begin{corollary}[Homomorphization Rewriting of Collision Moments: Universal Target Graph \texorpdfstring{$\mathsf{Disj}_q$}{Disj-q}]\label{cor:pom-collision-moment-hom-disj}
For any integer \(q\ge 1\) and any resolution \(m\ge 1\), the strict identity holds
\[
\boxed{\;
S_q(m)=\sum_{x\in X_m} d_m(x)^q
\;=\;
\sum_{x\in X_m}\mathrm{hom}(\Gamma_x,\mathsf{Disj}_q)\; }.
\]
\end{corollary}
\begin{proof}
By Theorem \ref{thm:pom-fiber-indset-factorization}, \(d_m(x)=\mathsf{I}(\Gamma_x)\).
Applying Proposition \ref{prop:pom-indset-power-hom-disj} to \(\Gamma=\Gamma_x\) and summing over \(x\in X_m\) completes the proof.
\end{proof}

\begin{remark}[Categorification Entry: Organizing "Local Fusion" Skeleton via Quotient/Orbit Structure of Universal Target Graph]\label{rem:pom-disj-target-graph-fusion}
Corollary \ref{cor:pom-collision-moment-hom-disj} lifts the "\(q\)-th order collision statistics" of \(\mathrm{MOM}_q\) from the power sum \(\sum_x d_m(x)^q\) to a homomorphism counting problem to the fixed target graph \(\mathsf{Disj}_q\).
Thus, the action of \(q\) can be interpreted as changing the local compatibility geometry of the target graph (vertices as subsets, edges as disjointness constraints), rather than mere "exponentiation."
Under this caliber, if further taking quotient/orbit decomposition of \(\mathsf{Disj}_q\) (e.g., stratifying by subset size), the local channel decomposition of high-order collision kernels can be viewed as a composable "fusion matrix" skeleton, naturally aligning with the channelization language of wreath lift/Artin factorization (see \S\ref{subsubsec:pom-wreath-artin-factorization}).
\end{remark}

\paragraph{Quasi-Multiplicativity: Interface Cost of \texorpdfstring{$\log S_q$}{log Sq} is Only \texorpdfstring{$O(q)$}{O(q)}}
\begin{proposition}[Quasi-Multiplicativity and Approximate Subadditivity of Collision Moments]\label{prop:pom-sq-quasi-multiplicative}
For any integer \(q\ge 1\), there exists a constant \(K_q\le q\log 2\) (depending only on \(q\), independent of \(m,n\)) such that for all \(m,n\ge 1\),
\[
\boxed{\;
\bigl|\log S_q(m+n)-\log S_q(m)-\log S_q(n)\bigr|\le K_q\; }.
\]
Equivalently, there exist explicit constants \(c_q:=2^{-q}\), \(C_q:=2^{q}\) such that
\[
\boxed{\;
c_q\,S_q(m)S_q(n)\ \le\ S_q(m+n)\ \le\ C_q\,S_q(m)S_q(n)\; }.
\]
\end{proposition}
\begin{proof}
Fold normalization is generated by the local rewrite kernel \((011\leftrightarrow 100)\) (Theorem \ref{thm:fold-suite}), so when concatenating length \(m\) and length \(n\) windows to length \(m+n\), the cross-boundary influence comes only from \(O(1)\) bits in the interface neighborhood: at most one "hardcore compatibility" constraint is introduced at the interface.
\par
In the candidate site-induced graph caliber, this means: for any concatenation type \(x\in X_{m+n}\), its candidate graph \(\Gamma_x\) differs from the "boundary-free juxtaposition of left and right segment candidate graphs" only by a single-edge-level local modification at the interface (equivalently, at most one hardcore edge added or removed).
For any finite graph \(G\) and its single-edge perturbation \(G'\), the independent set counts satisfy
\(
\tfrac12\,\mathsf{I}(G)\le \mathsf{I}(G')\le 2\,\mathsf{I}(G)
\):
for example, if \(G'\) is obtained from \(G\) by adding an edge \(uv\), then "deleting \(u\)" gives an injection from the independent set family of \(G\) taking both \(\{u,v\}\) to the independent set family of \(G'\), hence \(\mathsf{I}(G)\le 2\mathsf{I}(G')\); the reverse inequality follows by symmetry.
\par
By Theorem \ref{thm:pom-fiber-indset-factorization}, for any resolution \(k\ge 1\) and any \(x\in X_k\), we have \(d_k(x)=\mathsf{I}(\Gamma_x)\).
Thus the single-edge perturbation at the interface causes the fiber multiplicity to change by at most a factor of \(2\);
raising this change to the \(q\)-th power and summing over types yields at most a multiplicative perturbation of factor \(2^q\) in the collision moments under concatenation,
so we can take \(K_q\le q\log 2\) and \((c_q,C_q)=(2^{-q},2^q)\) such that the stated inequalities hold.
\end{proof}

\paragraph{Discrete Log-Convexity of \texorpdfstring{$r_q$}{rq} and "Free Energy Order"}
\begin{proposition}[Discrete Log-Convexity and Slope Monotonicity of Growth Rate]\label{prop:pom-rq-logconvex-free-energy-order}
Fix an integer \(q\ge 2\), and assume the limit growth rate
\[
r_q:=\lim_{m\to\infty}S_q(m)^{1/m}
\]
exists and is positive (for example, existence can be deduced from the quasi-multiplicativity of Proposition \ref{prop:pom-sq-quasi-multiplicative}).
Then:
\begin{enumerate}
  \item \textbf{(Discrete log-convexity)} For all integers \(q\ge 2\),
  \[
  \boxed{\ r_q^{2}\ \le\ r_{q-1}\,r_{q+1}\ }.
  \]
  \item \textbf{(Slope monotonicity)} Let \(\tau_q:=\log r_q\), then the difference slope \(\tau_q-\tau_{q-1}\) is monotone non-decreasing in \(q\).
  In particular, \(q\mapsto \frac{1}{q}\bigl(\tau_q-\tau_0\bigr)=\frac{1}{q}\log(r_q/r_0)\) is non-decreasing.
  \item \textbf{(Strictness)} If at some scale the fiber multiplicity is not constant (equivalently, \(\{d_m(x):x\in X_m\}\) contains at least two distinct values), then the above inequality holds strictly for each finite \(q\).
\end{enumerate}
\end{proposition}
\begin{proof}
\textbf{(1)} For each fixed \(m\), by H\"older's inequality (or convexity of \(q\mapsto \log S_q(m)\)), we have
\(
S_q(m)^2\le S_{q-1}(m)\,S_{q+1}(m)
\).
Taking the \(m\)-th root on both sides and letting \(m\to\infty\) yields \(r_q^2\le r_{q-1}r_{q+1}\).
\par
\textbf{(2)} By (1), the discrete convexity of \(\tau_q=\log r_q\) gives
\(
\tau_{q+1}-\tau_q\ge \tau_q-\tau_{q-1}
\).
For convex sequences, difference slope monotonicity follows immediately; and
\(
\frac{1}{q}(\tau_q-\tau_0)=\frac{1}{q}\sum_{j=1}^{q}(\tau_j-\tau_{j-1})
\)
is the average of the first \(q\) difference slopes, hence non-decreasing in \(q\).
\par
\textbf{(3)} If the multiplicity is not constant, then for fixed \(m\), the log-sum-exp convexity of the sequence \(q\mapsto S_q(m)\) is strict (for \(q>0\)), hence \(S_q(m)^2< S_{q-1}(m)S_{q+1}(m)\);
taking the limit yields strictness.
\end{proof}
\begin{corollary}[Ground State Normalized Convex Order: Free Energy Defect Slope Monotonicity with \texorpdfstring{$r_0=\varphi$}{r0=phi} as Baseline]\label{cor:pom-rq-groundstate-normalized-slope}
Let
\(
r_0:=\lim_{m\to\infty}|X_m|^{1/m}=\varphi
\),
and for integer \(q\ge 1\) define
\[
a_q:=\frac{\log r_q-\log r_0}{q}.
\]
Then \((a_q)_{q\ge 1}\) is monotone non-decreasing; strictly increasing in the non-trivial folding case.
Moreover,
\[
\lim_{q\to\infty}a_q=\frac12\log\varphi.
\]
Equivalently, for any integers \(1\le q_1<q_2\), the interpolation inequality chain holds
\[
\boxed{\;
r_{q_2}\ \ge\ r_0^{\,1-q_2/q_1}\,r_{q_1}^{\,q_2/q_1}\; }.
\]
\end{corollary}
\begin{proof}
By Proposition \ref{prop:pom-rq-logconvex-free-energy-order}, \(\tau_q:=\log r_q\) is a discrete convex sequence, so \(q\mapsto \frac{1}{q}(\tau_q-\tau_0)\) is monotone non-decreasing; this is precisely the monotonicity of \((a_q)\).
If the fiber multiplicity is not constant, Proposition \ref{prop:pom-rq-logconvex-free-energy-order}(3) gives strictness.
For the limit, by the endpoint theorem \ref{prop:pom-rq-qinfty-endpoint} (also see Corollary \ref{cor:pom-double-limit-commute-principal-exponent}), we have \(\lim_{q\to\infty}\frac{1}{q}\log r_q=\frac12\log\varphi\), while \(\frac{1}{q}\log r_0\to 0\), hence \(\lim_{q\to\infty}a_q=\frac12\log\varphi\).
Finally, convexity gives the chord inequality
\(
\tau_{q_2}\ge (1-q_2/q_1)\tau_0+(q_2/q_1)\tau_{q_1}
\),
exponentiating yields the stated interpolation chain.
\end{proof}
