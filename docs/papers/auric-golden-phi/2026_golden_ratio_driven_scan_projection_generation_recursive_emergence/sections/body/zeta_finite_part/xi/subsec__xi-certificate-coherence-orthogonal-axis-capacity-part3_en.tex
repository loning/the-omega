\begin{corollary}[fixresolutionbudget下的不可见阈值（比特/ledger口径）]\label{cor:xi-offcritical-visibility-threshold-bit-budget}

\begin{equation}\label{eq:xi-offcritical-bit-budget}
\boxed{\;

\end{equation}

\end{corollary}

\begin{proof}

最后一句由Corollary \ref{cor:xi-null-second-order-radial-channel} 的“二阶radial通道被纯phase制度抹除”给出。\qed
\end{proof}

式 \eqref{eq:xi-offcritical-bit-budget} 的首项 \(2\log|\gamma|\) 体现了height增长对endpointresolution的不可约需求；

\begin{theorem}[有限采样盲区与带状禁入certificate：单半径 Jensen lower bound给出verifiable排除域]\label{thm:xi-jensen-single-radius-band-exclusion}
let \(T\ge 1\)、\(\Delta\in(0,\tfrac12]\)，并Definition
\[
  R(T,\Delta)^2:=\frac{T^2+(\Delta-1)^2}{T^2+(\Delta+1)^2}\in(0,1).
\]
then对for any满足 \(|\gamma|\le T\)、\(\delta=\tfrac12-\beta\ge\Delta\) 的离critical linezero \(\rho=\beta+\mathrm{i}\gamma\)，其disk像 \(w_\rho\) 满足
\[
  |w_\rho|\le R(T,\Delta).
\]
进一步，对for any审计半径 \(\varrho\in(R(T,\Delta),1)\) 有单zero Jensen lower bound
\[
  D_\zeta(\varrho)\ \ge\ \log\frac{\varrho}{|w_\rho|}\ \ge\ \log\frac{\varrho}{R(T,\Delta)}
\]
（Proposition \ref{prop:app-jensen-single-zero-lower-bound}）。
therefore，if在某一选定半径 \(\varrho>R(T,\Delta)\) 处kernel验到
\[
  D_\zeta(\varrho)\ <\ \log\frac{\varrho}{R(T,\Delta)},
\]
then可作为一份离线verifiable的\emph{带状禁入certificate}：在该protocol口径下，不there exists满足 \(|\gamma|\le T\) 且 \(\tfrac12-\beta\ge\Delta\) 的离critical linezero。
\end{theorem}

\begin{proof}
由Theorem \ref{thm:xi-offcritical-quadratic-radial-compression} 的闭式，
\[
|w_\rho|^2
:=1-\frac{4\delta}{\gamma^2+(1+\delta)^2}
:=\frac{\gamma^2+(1-\delta)^2}{\gamma^2+(1+\delta)^2}.
\]
注意到当 \(\gamma^2\) 增大时，上式右端单调增；当 \(\delta\) 增大时，上式右端单调减（在 \(\delta\in(0,1)\) 的带内）。
therefore在约束 \(|\gamma|\le T\)、\(\delta\ge\Delta\) 下，
\[
|w_\rho|^2
\le \frac{T^2+(1-\Delta)^2}{T^2+(1+\Delta)^2}
:=R(T,\Delta)^2,
\]
thus \(|w_\rho|\le R(T,\Delta)\)。
当 \(\varrho>|w_\rho|\) 时，Proposition \ref{prop:app-jensen-single-zero-lower-bound} 给出
\(
D_\zeta(\varrho)\ge \log(\varrho/|w_\rho|)
\)，再用 \(|w_\rho|\le R(T,\Delta)\) 得到所述lower bound与certificate形式。\qed
\end{proof}

\[

=\exp\!\bigl(-2(1-r)\Phi_{-1}(B)+o(1-r)\bigr)
=1-2(1-r)\Phi_{-1}(B)+o(1-r),
\]
thus
\(
1-|B(-r)|^2=2(1-r)\Phi_{-1}(B)+o(1-r)
\)。
注意 \(1-r^2=2(1-r)+o(1-r)\)，遂得 \(\mathcal{J}_{-1}(B)=\Phi_{-1}(B)\)。
\par
另一方面，有限 Blaschke 乘product在闭diskneighborhood解析且 \(|B(-1)|=1\)，hence
\(
B(-r)=B(-1)+(1-r)B'(-1)+o(1-r)
\)；
将其代入并展开得
\(
1-|B(-r)|^2=-2(1-r)\Re\!\bigl(B'(-1)/B(-1)\bigr)+o(1-r)
\)，
再除以 \(1-r^2\) 并let \(r\uparrow 1\) 即得
\(
\mathcal{J}_{-1}(B)=\Re\!\bigl(-B'(-1)/B(-1)\bigr)
\)。
与第一步合并即得结论。\qed
\end{proof}

\paragraph{heightcompression与endpoint解耦：两条certificate通道的可见scale分裂}

\begin{definition}[两endpoint吸收coefficient]\label{def:xi-endpoint-two-absorption-coefficients}
let \(B\) 为有限 Blaschke 乘product，zero集合为 \(\{a_k\}_{k=1}^{N}\subset\mathbb{D}\)（按重数计）。
除Proposition \ref{prop:xi-endpoint-absorption-coefficient} 的endpoint \(-1\) 吸收coefficient \(\Phi_{-1}(B)\) 外，Definitionendpoint \(+1\) 的吸收coefficient
\[
\Phi_{+1}(B):=\sum_{k=1}^{N}P_{a_k}(+1)=\sum_{k=1}^{N}\frac{1-|a_k|^2}{|1-a_k|^2}\ \in[0,\infty).
\]
\end{definition}

\begin{theorem}[Adams scale门的endpoint吸收奇偶律与线性放大]\label{thm:xi-endpoint-absorption-adams-rescaling}
let \(B\) 为有限 Blaschke 乘product，并let Adams scale门 \(\pi_m(w):=w^m\)（\(m\in\NN\)）。
记幂拉回 \(B^{(m)}:=B\circ\pi_m\)，即 \(B^{(m)}(w)=B(w^m)\)。
then对for any \(m\ge 1\) 有
\[
\boxed{\;
\Phi_{-1}\bigl(B^{(m)}\bigr)=
\begin{cases}
m\,\Phi_{-1}(B),& m\ \text{为奇数},\\[2pt]
m\,\Phi_{+1}(B),& m\ \text{为偶数},
\end{cases}
\qquad
\Phi_{+1}\bigl(B^{(m)}\bigr)=m\,\Phi_{+1}(B).\;}
\]
\end{theorem}

\begin{proof}
先由Proposition \ref{prop:xi-endpoint-absorption-coefficient} 推出endpoint \(+1\) 的radial吸收律：let \(\widetilde B(w):=B(-w)\)，then \(\widetilde B\) 仍为有限 Blaschke 乘product且 \(\widetilde B(-r)=B(r)\)。
and
\[
\Phi_{-1}(\widetilde B)
:=\sum_{k=1}^{N}\frac{1-|(-a_k)|^2}{|1+(-a_k)|^2}
=\sum_{k=1}^{N}\frac{1-|a_k|^2}{|1-a_k|^2}
=\Phi_{+1}(B),
\]
hence由Proposition \ref{prop:xi-endpoint-absorption-coefficient} 得
\[
\log|B(r)|
=\log|\widetilde B(-r)|
=-(1-r)\,\Phi_{-1}(\widetilde B)+o(1-r)
=-(1-r)\,\Phi_{+1}(B)+o(1-r)
\qquad(r\uparrow 1).
\]
\par
对 \(\Phi_{-1}(B^{(m)})\)，take \(w=-r\uparrow-1\)，then
\(
B^{(m)}(-r)=B((-r)^m)
\)。
if \(m\) 为奇数，then \((-r)^m=-r^m\uparrow-1\)，由Proposition \ref{prop:xi-endpoint-absorption-coefficient}（对 \(r^m\uparrow 1\) 应用）得
\[
\log|B^{(m)}(-r)|=\log|B(-r^m)|=-(1-r^m)\,\Phi_{-1}(B)+o(1-r^m).
\]
if \(m\) 为偶数，then \((-r)^m=r^m\uparrow+1\)，由上一步的 \(+1\) radial吸收律得
\[
\log|B^{(m)}(-r)|=\log|B(r^m)|=-(1-r^m)\,\Phi_{+1}(B)+o(1-r^m).
\]
又因 \(1-r^m=m(1-r)+o(1-r)\)（\(r\uparrow 1\)），and \(B^{(m)}\) 仍为有限 Blaschke 乘product可适用Proposition \ref{prop:xi-endpoint-absorption-coefficient}，thus
\(
\log|B^{(m)}(-r)|=-(1-r)\,\Phi_{-1}(B^{(m)})+o(1-r)
\)；
比较一阶coefficient即得所述奇偶律。
\par
对 \(\Phi_{+1}(B^{(m)})\)，take \(w=r\uparrow+1\)，then \(B^{(m)}(r)=B(r^m)\) 且 \(r^m\uparrow 1\)。
由上一步的 \(+1\) radial吸收律对 \(B\) 与 \(B^{(m)}\) 分别应用，得
\[
\log|B^{(m)}(r)|
=\log|B(r^m)|
=-(1-r^m)\,\Phi_{+1}(B)+o(1-r^m)
=-(1-r)\,m\,\Phi_{+1}(B)+o(1-r)
\]
以及
\(
\log|B^{(m)}(r)|=-(1-r)\,\Phi_{+1}(B^{(m)})+o(1-r)
\)。
比较一阶coefficient得 \(\Phi_{+1}(B^{(m)})=m\Phi_{+1}(B)\)。\qed
\end{proof}

\begin{remark}[奇偶性与endpoint通道的退化/合并]\label{rem:xi-endpoint-absorption-adams-parity}
幂门 \(\pi_m(w)=w^m\) 在endpoint上的像满足 \(\pi_m(+1)=+1\) 且 \(\pi_m(-1)=(-1)^m\)：
奇 \(m\) 时endpoint \(-1\) 保持不变，thus \(\Phi_{-1}\) 通道被线性放大而不改变基点；
偶 \(m\) 时endpoint \(-1\) 被送至 \(+1\)，hence \(\Phi_{-1}(B^{(m)})\) 读take的是 \(B\) 在 \(+1\) 的吸收coefficient \(\Phi_{+1}(B)\)。
特别地，偶 \(m\) 时endpoint双通道在 \(B^{(m)}\) 上合并为同一指标：
\[
\Phi_{-1}(B^{(m)})=\Phi_{+1}(B^{(m)})=m\,\Phi_{+1}(B),
\]
thusendpoint审计在该scale上退化为一维。
\end{remark}

\begin{remark}[auditable的终端判别形式：endpoint一维指数与多scale审计链]\label{rem:xi-endpoint-absorption-terminal-index}
在 Hardy/Smirnov 因子分解与“偏移仅由 Blaschke 数据承载”的口径下，离slice偏移可被compression为endpoint一维指数 \(\Phi_{-1}(B)\)（并配套endpoint \(+1\) 指数 \(\Phi_{+1}(B)\)，Definition \ref{def:xi-endpoint-two-absorption-coefficients}）：

\paragraph{horizon重整化：endpoint通量的严格缩放律}
endpoint吸收coefficient \(\Phi_{-1}(B)\) 是盘内点状defect的单点compression；与之配套的scale语义由“dilation半群”给出。

下述Theorem表明：endpoint Poisson weight（亦即通量的单原子贡献）在该scale链下严格按因子 \(m\) 放大。

\begin{theorem}[endpoint Poisson weight的scale共变性]\label{thm:xi-endpoint-poisson-weight-scaling}
let \(m>1\) 且 \(\psi_m\) 如 \eqref{eq:scale-psi-m}。
对for any \(a\in\mathbb{D}\) 的endpoint Poisson weight
\(
P_a(-1):=\frac{1-|a|^2}{|1+a|^2}
\)
有严格identity
\[
\boxed{\;P_{\psi_m(a)}(-1)=m\,P_a(-1)\;}
\]
thereforeif \(B\) 为有限 Blaschke 乘product、其zero集合为 \(\{a_k\}_{k=1}^{N}\subset\mathbb{D}\)（按重数计），并let
\[
a_k^{(n)}:=\psi_{m^n}(a_k),
\]
再let \(B^{(n)}\) 为以 \(\{a_k^{(n)}\}\) 为zero集合的 Blaschke 乘product（重数保持），thenendpoint毛发通量满足
\[
\boxed{\;\Phi_{-1}(B^{(n)})=m^n\,\Phi_{-1}(B)\;}
\]

\end{theorem}

\begin{proof}

\Re\!\left(\frac{1-z}{1+z}\right)=\frac{1-|z|^2}{|1+z|^2}=P_z(-1),

最后一式由逐项求和与 \(\psi_{m^n}=\psi_m^{\circ n}\) 的半群性质推出。\qed
\end{proof}

\end{proposition}

\begin{corollary}[height窗下的二次compression：Toeplitz 截断阶与 \texorpdfstring{\(\varphi\)}{phi}-标定律]\label{cor:xi-toeplitz-height-phi-calibration}
沿用Proposition \ref{prop:xi-extreme-resonance-toeplitz-threshold} 的记号，并let
\(
m_\star(\rho):=\lceil\log_{\varphi}Q_\rho\rceil
\)
为由horizondepth \(1-|w_\rho|^2\) 所确定的 \(\varphi\)-resolution阈值（Theorem \ref{thm:xi-bulk-horizon-phi-area-law}）。
then Toeplitz--PSD 的有限级kernel验if要对该pole产生有限维失败witness，所需矩阶满足
\[
N\ \gtrsim\ \varphi^{m_\star(\rho)}\,\log\!\bigl(1+\delta\,\varphi^{m_\star(\rho)}\bigr)
\qquad(\delta:=\tfrac12-\beta>0).
\]
thus对fix矩阶 \(N\) 的截断kernel验而言，可被强制检出的pole只能满足
\(
m_\star(\rho)\ \lesssim\ \log_{\varphi}N
\)
（至多差一个logarithmic因子）。
\par
更具体地，给定偏移lower bound \(\delta\ge\delta_0>0\) 与height窗 \(|\gamma|\le T\)，记该窗内的最坏品质因数
\[
Q_{\max}(T,\delta_0):=\frac{T^2+(1+\delta_0)^2}{4\delta_0},
\qquad
m_{\max}(T,\delta_0):=\left\lceil\log_{\varphi}Q_{\max}(T,\delta_0)\right\rceil.
\]
then任何希望在该height窗内\emph{统一}排除潜在离slicepole的有限级 Toeplitz--PSD certificate都必须满足
\[
N\ \gtrsim\ Q_{\max}(T,\delta_0)\,\log\!\bigl(1+\delta_0 Q_{\max}(T,\delta_0)\bigr)
\asymp \frac{T^2}{\delta_0}\,\log(1+T^2),
\]
and相应的 \(\varphi\)-resolution阈值满足
\[
m_{\max}(T,\delta_0)=2\log_{\varphi}T+O(1)\qquad(T\to\infty).
\]
\end{corollary}
\begin{proof}
由Definition \(m_\star(\rho)=\lceil\log_{\varphi}Q_\rho\rceil\) 知 \(Q_\rho=\Theta(\varphi^{m_\star(\rho)})\)，将其代入Proposition \ref{prop:xi-extreme-resonance-toeplitz-threshold} 即得第一式。
反向估计由同一式直接给出：if \(m_\star(\rho)\) 远大于 \(\log_\varphi N\)，then右端主项严格超过 \(N\)，有限级kernel验无法强制产生失败witness。
\par
最后，对 \(\delta\in(0,1)\) 有
\(
Q_\rho=(\gamma^2+1)/(4\delta)+\tfrac12+\delta/4
\)
随 \(|\gamma|\) 单调增加且随 \(\delta\) 单调减少，hence在约束 \(|\gamma|\le T\)、\(\delta\ge\delta_0\) 下有
\(
Q_\rho\le Q_{\max}(T,\delta_0)
\)
and最坏情形在 \((|\gamma|,\delta)=(T,\delta_0)\) 达到。
将 \(Q_{\max}\) 代入Proposition \ref{prop:xi-extreme-resonance-toeplitz-threshold} 得统一lower bound；而 \(m_{\max}=\lceil\log_\varphi Q_{\max}\rceil\) 给出最后一式。\qed
\end{proof}

最后需要强调的是：“零漏泄”应被理解为verifiable制度而非单点偶然。
在disk化完成化 \(F_\zeta\) 上，RH equivalent todefect恒零 \(D_{\zeta}(\varrho)\equiv 0\)（Corollary \ref{cor:app-rh-iff-jensen-defect-zero}），亦equivalent to对for any趋近列 \(\varrho_n\uparrow 1\) 的可数约束 \(D_{\zeta}(\varrho_n)=0\)（Theorem \ref{thm:app-jensen-countable-criterion}）。
在该“零漏泄制度”下，Jensen--Blaschke 机制排除一切盘内点状zero；结合完成化自对偶参数 \(u=u(s)\) 的fixslice \(|u|=1\iff \Re(s)=\tfrac12\)，得到：点状zero只能承载于fixslice，thus只能承载于 \(\Re(s)=\tfrac12\)。
与之配套，拉马努金型对称在this paper框架中的作用并非“交换轴”，而是通过局部spectrum界强制phase化闭包，thus逐通道消去幅度/radial漏泄（\S\ref{subsubsec:ramanujan-phase-lift-symmetry}；Theorem \ref{thm:ramanujan-phase-lift-principle}）。
\par
therefore在this paper体系中，unitary slice 是连续可见参数的极小截面：偏离该截面的radial信息在endpointlayer以二次compression律进入（Theorem \ref{thm:xi-offcritical-quadratic-radial-compression}），其可见化需要不可约的endpointresolution/位budget（Theorem \ref{thm:xi-offcritical-endpoint-resolution-lower-bound}），并可在有限height带上由单半径 Jensen defectinequalitycertificate化（Theorem \ref{thm:xi-jensen-single-radius-band-exclusion}）。
共动局部化虽可在线性scale显影偏移，但其代价是引入必须显式给定的可寻址前缀（注 \ref{rem:xi-comoving-horizon-localization-address-null}）；在fix可见interface下if拒绝externalized该前缀，then离slice偏移仍被迫通过endpoint二次radial通道进入并触发相应capacitythreshold。
the abovebudget轴在protocol语义中相互orthogonal且不可替代（Theorem \ref{thm:xi-null-complete-trichotomy-offline}）：提升某一轴的budget不能抵消另一轴的threshold违背；在拒绝externalized模式轴时，“离slice断言”在typelayer不可形成verifiable地址并稳定退化为 \(\mathsf{NULL}\)。
