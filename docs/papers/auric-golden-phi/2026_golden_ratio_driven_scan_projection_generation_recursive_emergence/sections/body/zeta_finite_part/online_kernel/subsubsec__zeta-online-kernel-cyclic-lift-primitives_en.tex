\subsubsection{Primitive orbits and Dirichlet combs for cyclic lifts: divisibility screening, closed forms, and square-root scaling}\label{subsubsec:zeta-online-kernel-cyclic-lift-primitives}
This paragraph aligns the cyclic phase lift \(M^{\langle q\rangle}=M\otimes P_q\) from Appendix \ref{app:real-input-40-finite-rh} (Proposition \ref{prop:finite-rh-phase-lift}) with the
\(s\)-plane pole dictionary (Proposition \ref{prop:s-plane-spectrum-poles}) from the main text
and the primitive M\"obius inversion template (Remark \ref{prop:finite-dimensional-explicit-formula}),
thereby obtaining a set of closed-form corollaries that can be cited directly: on the critical-line pole side the cyclic lift realizes a "\(q\)-comb" refinement,
while on the length/trace side it introduces a strict divisibility gate, and from this one derives the closed form for primitive orbit counts and their error terms.

\begin{corollary}[Closed form for primitive orbit counts under cyclic lift]\label{cor:zeta-cyclic-lift-primitive-orbits}
Let \(M\) be a finite-dimensional matrix, take an integer \(q\ge 2\), let \(P_q\) denote the \(q\)-cyclic permutation matrix, and set \(M^{\langle q\rangle}:=M\otimes P_q\) (Proposition \ref{prop:finite-rh-phase-lift}).
Denote the trace sequences by \(a_n:=\Tr(M^n)\), \(a_n^{\langle q\rangle}:=\Tr((M^{\langle q\rangle})^n)\), and the corresponding primitive orbit counts by
\[
p_n:=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d},
\qquad
p_n^{\langle q\rangle}:=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a^{\langle q\rangle}_{n/d}
\]
(Witt/M\"obius inversion; see Remark \ref{prop:finite-dimensional-explicit-formula}).
Then:
\begin{enumerate}
  \item If \(q\nmid n\), then \(p_n^{\langle q\rangle}=0\).
  \item If \(n=qm\), then the following closed form holds:
  \[
  \boxed{\;
  p_{qm}^{\langle q\rangle}
  =\frac{1}{m}\sum_{d\mid m}\mu(d)\,\Tr\!\bigl(M^{qm/d}\bigr)\; }.
  \]
\end{enumerate}
\end{corollary}
\begin{proof}
By the divisibility screening of Corollary \ref{cor:finite-rh-phase-lift-trace-filter},
\(
a_n^{\langle q\rangle}=q a_n
\)
if and only if \(q\mid n\), and \(a_n^{\langle q\rangle}=0\) otherwise.
Substituting into the M\"obius inversion yields the first assertion; when \(n=qm\), only those terms with \(n/d\) still divisible by \(q\) (i.e., \(d\mid m\)) contribute, yielding the second closed form. \qed
\end{proof}

\begin{corollary}[The \(q\)-comb refinement of Dirichlet poles and RH invariance]\label{cor:zeta-cyclic-lift-dirichlet-comb}
Let \(M\) be a finite-dimensional matrix with \(\lambda=\rho(M)>1\), and let
\(
Z_M(s)=\det(I-\lambda^{-s}M)^{-1}
\)
be the Dirichlet-ized \(\zeta\)-interface (Proposition \ref{prop:s-plane-spectrum-poles}).
For the cyclic lift \(M^{\langle q\rangle}=M\otimes P_q\), the Dirichlet-ized function
\(
Z_{M^{\langle q\rangle}}(s)=\det(I-\lambda^{-s}M^{\langle q\rangle})^{-1}
\)
satisfies:
\begin{enumerate}
  \item \textbf{(Real-part invariance)} The set of real parts of all poles of \(Z_{M^{\langle q\rangle}}\) coincides exactly with that of \(Z_M\).
  \item \textbf{(Imaginary-part density amplification)} If \(\mu\neq 0\) is a nonzero eigenvalue of \(M\), then the fundamental imaginary spacing of the corresponding pole sequence shrinks from \(2\pi/\log\lambda\) to
  \[
  \boxed{\;
  \frac{2\pi}{q\log\lambda}\; },
  \]
  equivalently, the imaginary-part pole density is amplified by exactly a factor of \(q\) (the pole-comb formula of Proposition \ref{prop:finite-rh-phase-lift}).
  \item \textbf{(Kernel-RH preservation)} The kernel RH / Ramanujan radius bound
  \(
  \max_{\mu\neq\lambda}|\mu|\le \sqrt{\lambda}
  \)
  is strictly invariant under cyclic lift (Proposition \ref{prop:finite-rh-phase-lift}); its \(s\)-plane formulation---namely that the open half-strip \(\tfrac12<\Re(s)<1\) contains no poles---is preserved (Proposition \ref{prop:s-plane-spectrum-poles}).
\end{enumerate}
\end{corollary}
\begin{proof}
Proposition \ref{prop:finite-rh-phase-lift} gives the spectral phase subdivision
\(
\mathrm{spec}(M^{\langle q\rangle})=\{\mu\,\omega_q^k\}
\)
with moduli unchanged; applying the dictionary of Proposition \ref{prop:s-plane-spectrum-poles},
\(
s=(\log\mu+2\pi i\mathbb{Z})/\log\lambda
\),
immediately yields (1) and (2).
Assertion (3) involves only moduli, and hence follows directly from \(|\omega_q^k|=1\). \qed
\end{proof}

\begin{proposition}[``Prime shadow'' leading term and square-root error for cyclic lifts]\label{prop:zeta-cyclic-lift-prime-shadow-asymptotic}
Let \(M\) be a primitive nonneg integer matrix with \(\lambda=\rho(M)>1\), and denote
\(
\Lambda:=\max_{\mu\neq\lambda}|\mu|
\).
Let \(p_n^{\langle q\rangle}\) be the primitive orbit counts from Corollary \ref{cor:zeta-cyclic-lift-primitive-orbits}.
Then for any integer \(m\ge 1\) one has the asymptotic formula
\[
\boxed{\;
p_{qm}^{\langle q\rangle}
=\frac{\lambda^{qm}}{m}
+O\!\left(\frac{\max\{\Lambda^{qm},\,\lambda^{qm/2}\}}{m}\right),\qquad
p_n^{\langle q\rangle}=0\ (q\nmid n)\; }.
\]
In particular, if the kernel RH holds (\(\Lambda\le\sqrt{\lambda}\)), then
\[
\boxed{\;
p_{qm}^{\langle q\rangle}
=\frac{\lambda^{qm}}{m}
+O\!\left(\frac{\lambda^{qm/2}}{m}\right).\;}
\]
Moreover, when there exists a boundary eigenvalue \(|\mu|=\sqrt{\lambda}\), this square-root scale is in general not improvable (see the parity-gating proposition for RH-sharp oscillations, Proposition \ref{prop:finite-rh-parity-general}, in the appendix).
\end{proposition}
\begin{proof}
Substitute the closed form from Corollary \ref{cor:zeta-cyclic-lift-primitive-orbits} into
\(
\Tr(M^n)=\lambda^n+O(\Lambda^n)
\):
the leading term comes from the \(d=1\) Perron contribution \(\lambda^{qm}/m\).
Part of the error arises from \(O(\Lambda^{qm/d})\), whose largest magnitude is \(O(\Lambda^{qm})\);
another part comes from Perron sub-leading terms at \(d\ge 2\): \(\lambda^{qm/d}\le \lambda^{qm/2}\).
Combining and dividing by \(m\) gives the stated upper bound; under the RH condition \(\Lambda^{qm}\le \lambda^{qm/2}\), the simplified form follows immediately. \qed
\end{proof}

\begin{corollary}[Sign-flip modes and the critical-line half-lattice comb]\label{cor:zeta-signflip-half-lattice}
Let \(M\) be a finite-dimensional matrix with \(\lambda=\rho(M)>1\), and set \(Z_M(s)=\det(I-\lambda^{-s}M)^{-1}\).
Then the following two assertions are equivalent:
\begin{enumerate}
  \item \(M\) has an eigenvalue \(\mu=-\sqrt{\lambda}\);
  \item \(Z_M\) has a half-lattice pole array on the critical line \(\Re(s)=\tfrac12\):
  \[
  \boxed{\;
  s=\frac12+\frac{(2k+1)\pi i}{\log\lambda}\qquad(k\in\ZZ)\; }.
  \]
\end{enumerate}
More generally, if there exists a boundary eigenvalue \(\mu=\sqrt{\lambda}e^{i\theta}\), then the pole sequence on the critical line is
\[
\boxed{\;
s=\frac12+\frac{(\theta+2\pi k)i}{\log\lambda}\qquad(k\in\ZZ)\; }.
\]
\end{corollary}
\begin{proof}
By Proposition \ref{prop:s-plane-spectrum-poles}, \(\lambda^{-s}\mu=1\) if and only if
\(
s=(\log\mu+2\pi i k)/\log\lambda
\).
When \(\mu=\sqrt{\lambda}e^{i\theta}\), one may take \(\log\mu=\tfrac12\log\lambda+i(\theta+2\pi\ell)\), and substitution yields the general formula;
setting \(\theta=\pi\) gives the half-lattice case.
The converse reasoning proceeds identically: a critical-line pole yields \(|\mu|=\sqrt{\lambda}\) with argument \(\theta\) (modulo \(2\pi\)). \qed
\end{proof}

\paragraph{Finite probes and resolution: addressability and audit limits for RH-sharp combs}
Let \(M\) be RH-sharp: there exists a boundary eigenvalue \(\mu=\sqrt{\lambda}e^{i\theta}\) (\(\lambda=\rho(M)>1\)).
By the comb formula of Corollary \ref{cor:zeta-cyclic-lift-dirichlet-comb}, for any \(q\ge 2\) the cyclic lift
\(
M^{\langle q\rangle}=M\otimes P_q
\)
generates a \(q\)-subdivision imaginary comb on the critical line \(\Re(s)=\tfrac12\).
Taking the imaginary part modulo the fundamental period \(2\pi/\log\lambda\), define
\[
\mathcal T_q(\theta):=
\left\{\left.\frac{\theta+2\pi j}{q\log\lambda}\ \right|\ j=0,1,\dots,q-1\right\}
\ \subset\ \RR/\Bigl(\frac{2\pi}{\log\lambda}\Bigr)\ZZ .
\]

\begin{theorem}[Finite-probe evasion: any finite probe can be evaded by cyclic-lift closure]\label{thm:zeta-cyclic-lift-finite-probe-evasion}
Given any finite set
\(
S\subset \RR/\bigl(\tfrac{2\pi}{\log\lambda}\bigr)\ZZ
\).
Except in the degenerate aligned case (i.e., \(\theta\equiv 0\ (\mathrm{mod}\ 2\pi)\) and \(0\in S\)), there exist infinitely many integers \(q\ge 2\) such that
\[
S\cap \mathcal T_q(\theta)=\varnothing.
\]
Consequently, any ``finite-probe audit'' that verifies a \(\zeta\)-type object only at finitely many critical-line sampling points cannot falsify RH-sharp in the sense of cyclic-lift closure: in the presence of a boundary mode, one can always choose certain lift orders \(q\) such that all sampling points avoid the comb poles.
\end{theorem}
\begin{proof}
Via the isomorphism
\(
x:=t\log\lambda/(2\pi)\ (\mathrm{mod}\ 1)
\)
identify the quotient group \(\RR/(\tfrac{2\pi}{\log\lambda})\ZZ\) with the circle \(\mathbb T:=\RR/\ZZ\), and set \(\alpha:=\theta/(2\pi)\in[0,1)\).
Then \(\mathcal T_q(\theta)\) corresponds to the phase-point set
\[
\mathcal P_q(\alpha):=\left\{\left.\frac{\alpha+j}{q}\ \right|\ j=0,1,\dots,q-1\right\}\subset\mathbb T,
\]
and for any \(x\in\mathbb T\) one has the equivalence
\(
x\in\mathcal P_q(\alpha)\Longleftrightarrow qx-\alpha\in\ZZ
\).
Let \(X\subset\mathbb T\) denote the image of \(S\). The desired conclusion is equivalent to showing there exist infinitely many \(q\ge 2\) with \(X\cap\mathcal P_q(\alpha)=\varnothing\).
\par
\emph{(1) \(\alpha\) irrational:} If \(\alpha\notin\QQ\), then for any rational \(x\) we have \(qx\in\QQ\) and thus \(qx-\alpha\notin\ZZ\) (no solution); for any irrational \(x\), if \(q_1x-\alpha,q_2x-\alpha\in\ZZ\) then \((q_1-q_2)x\in\ZZ\), forcing \(x\in\QQ\), a contradiction, so each irrational \(x\) excludes at most one \(q\).
Hence the finite set \(X\) excludes only finitely many \(q\), leaving infinitely many \(q\) available.
\par
\emph{(2) \(\alpha\) rational:} Suppose \(\alpha=a/b\in\QQ\cap[0,1)\) in reduced form.
If \(b=1\) (i.e., \(\alpha=0\)), then \(x\in\mathcal P_q(0)\) is equivalent to \(qx\in\ZZ\).
When \(0\notin X\), all points in \(X\) that could be hit are rational with denominator \(\ge 2\), so each point excludes only a set of multiples of some positive integer; choosing infinitely many large primes \(q\) avoids all of them.
When \(0\in X\), we have \(0\in\mathcal P_q(0)\) for all \(q\), so evasion is impossible (corresponding to the degenerate aligned case in the theorem statement).
\par
Now assume \(b\ge 2\), and take any prime divisor \(p\mid b\).
If \(x=m/n\in X\cap\QQ\) is in reduced form and there exists \(q\) with \(qx-\alpha\in\ZZ\), then necessarily \(b\mid n\);
write \(n=bd\) (\(d\ge 1\)). From
\(
q\cdot \frac{m}{bd}-\frac{a}{b}\in\ZZ
\)
we deduce
\(
qm-ad\in bd\,\ZZ
\),
and thus (since \(\gcd(m,bd)=1\)) \(q\equiv ad\,m^{-1}\ (\mathrm{mod}\ bd)\).
In particular, \(q=dq'\) with \(q'\equiv a m^{-1}\ (\mathrm{mod}\ b)\), hence \(\gcd(q',b)=1\).
Therefore, for any \(p\mid b\) we have
\[
v_p(q)=v_p(d)\qquad(\text{when }x=m/(bd)\ \text{ is hit}).
\]
Since \(X\) is finite, the set of values \(v_p(d)\) corresponding to all hittable rational points is finite.
Take \(N\) larger than the maximum of this finite set and set \(q:=p^{N}\);
then \(v_p(q)=N\) cannot equal any hittable \(v_p(d)\), so \(X\cap\mathcal P_q(\alpha)=\varnothing\).
Letting \(N\to\infty\) yields infinitely many such \(q\). \qed
\end{proof}

\begin{proposition}[Covering radius of the addressable grid: generic phase yields \texorpdfstring{$\Theta(Q^{-1})$}{Theta(1/Q)}]\label{prop:zeta-cyclic-lift-addressable-grid-covering-radius}
Retaining the notation \(\alpha\in[0,1)\) and \(\mathcal P_q(\alpha)\) from above, define the union of grids up to order \(Q\):
\[
\mathcal P_{\le Q}(\alpha):=\bigcup_{q=2}^{Q}\mathcal P_q(\alpha)\subset\mathbb T .
\]
Let \(d_{\mathbb T}\) denote the circle distance. Then for any \(Q\ge 2\) one has
\[
\boxed{\;
\frac{\min\{\alpha,\,1-\alpha\}}{Q}
\ \le\
\sup_{x\in\mathbb T} d_{\mathbb T}\bigl(x,\mathcal P_{\le Q}(\alpha)\bigr)
\ \le\
\frac{1}{2Q}\; }.
\]
In particular, when \(\alpha\in(0,1)\) is fixed, the above formula gives a worst-case covering radius of order \(\Theta(Q^{-1})\).
\end{proposition}
\begin{proof}
The upper bound follows from the single-layer comb at \(q=Q\): \(\mathcal P_Q(\alpha)\) is an equispaced grid with step \(1/Q\), so any \(x\) has distance \(\le 1/(2Q)\) to \(\mathcal P_Q(\alpha)\), and hence to the union \(\mathcal P_{\le Q}(\alpha)\).
\par
For the lower bound, take \(x=0\): for each \(q\), the nearest point in \(\mathcal P_q(\alpha)\) to \(0\) is \(\alpha/q\) or \(1-(1-\alpha)/q\), so
\(
d_{\mathbb T}(0,\mathcal P_q(\alpha))=\min\{\alpha,1-\alpha\}/q
\);
minimizing over \(q\le Q\) gives \(\min\{\alpha,1-\alpha\}/Q\).
Hence the worst-case covering radius (supremum over \(x\)) is at least this value. \qed
\end{proof}

\begin{proposition}[Counting lower bound: worst-case covering radius at least \texorpdfstring{$\gg Q^{-2}$}{gg Q^-2}]\label{prop:zeta-cyclic-lift-covering-radius-counting-lb}
Retaining the notation \(\mathcal P_{\le Q}(\alpha)\), let
\(
N_Q:=\bigl|\mathcal P_{\le Q}(\alpha)\bigr|
\)
denote its cardinality (note that duplicates may occur, so this is only an upper-bound count).
Then for any \(Q\ge 2\) there exists \(x\in\mathbb T\) such that
\[
\boxed{\;
d_{\mathbb T}\bigl(x,\mathcal P_{\le Q}(\alpha)\bigr)\ \ge\ \frac{1}{Q(Q+1)}\; }.
\]
\end{proposition}
\begin{proof}
Clearly the cardinality satisfies
\[
N_Q\le \sum_{q=2}^{Q} q=\frac{Q(Q+1)}{2}-1.
\]
If one covers all points of \(\mathcal P_{\le Q}(\alpha)\) by arcs of radius \(r\), the total covering length is at most \(2r\,N_Q\).
Taking \(r<\frac{1}{2N_Q}\), the total length is \(<1\), so there must exist an uncovered point \(x\) satisfying
\(
d_{\mathbb T}(x,\mathcal P_{\le Q}(\alpha))\ge r
\).
Substituting \(N_Q\le Q(Q+1)/2\) and taking \(r=\frac{1}{Q(Q+1)}\) yields the result. \qed
\end{proof}

\begin{corollary}[Oracle resolution law: worst-case resolution scale imposed by lift-order bound \(Q\)]\label{cor:zeta-cyclic-lift-resolution-law}
Under the RH-sharp setting, define the worst-case resolution (modulo the fundamental period):
\[
\varepsilon_Q:=\sup_{t\in \RR/(\tfrac{2\pi}{\log\lambda})\ZZ}\ \inf_{2\le q\le Q}\ \inf_{t_q\in\mathcal T_q(\theta)} d(t,t_q),
\]
where \(d\) denotes the circle distance (aligned with \(d_{\mathbb T}\) above).
Then the normalized identity holds:
\[
\frac{\varepsilon_Q\log\lambda}{2\pi}
=\sup_{x\in\mathbb T} d_{\mathbb T}\bigl(x,\mathcal P_{\le Q}(\alpha)\bigr),
\qquad \alpha=\frac{\theta}{2\pi},
\]
and hence by Proposition \ref{prop:zeta-cyclic-lift-addressable-grid-covering-radius} we obtain the two-sided bound
\[
\boxed{\;
\frac{\min\{\alpha,\,1-\alpha\}}{Q}
\ \le\
\frac{\varepsilon_Q\log\lambda}{2\pi}
\ \le\
\frac{1}{2Q}\; }.
\]
Thus, for a generic phase \(\alpha\in(0,1)\) fixed, in order to reduce the worst-case uncertainty to scale \(\varepsilon\), any mechanism relying on ``cyclic lift up to order \(Q\)'' must satisfy in terms of resources
\[
Q\ \gtrsim\ \frac{1}{\varepsilon\log\lambda},
\]
and this \(Q^{-1}\) scaling is not improvable in order of magnitude.
For the aligned case \(\alpha=0\) and the \(Q^{-2}\) improvement, see Remark \ref{rem:zeta-cyclic-lift-aligned-farey-q2}.
\end{corollary}

\begin{remark}[Aligned case \texorpdfstring{$\alpha=0$}{alpha=0} and the \texorpdfstring{$Q^{-2}$}{Q^-2} improvement]\label{rem:zeta-cyclic-lift-aligned-farey-q2}
When \(\alpha=0\) (equivalent to \(\theta\equiv 0\ (\mathrm{mod}\ 2\pi)\)), \(\mathcal P_q(0)=\{j/q\}\) is the standard rational grid, and its union \(\mathcal P_{\le Q}(0)\) is of the same order as the Farey grid with denominators \(\le Q\); the worst-case covering radius improves to \(\Theta(Q^{-2})\) (the classical Dirichlet/Farey approximation order).
In this case, the resource scaling improves from the generic-phase \(Q\asymp \varepsilon^{-1}\) to \(Q\asymp \varepsilon^{-1/2}\) (in the normalized variable \(x\)).
\end{remark}
