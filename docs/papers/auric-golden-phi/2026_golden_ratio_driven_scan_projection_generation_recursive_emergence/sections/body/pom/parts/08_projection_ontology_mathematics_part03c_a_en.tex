\subsubsection{Fourth-Order Collision Moment and \texorpdfstring{$\mathrm{R\acute{e}nyi}$}{Renyi}--4 Projection Entropy Rate}\label{subsec:pom-s4}
\begin{definition}[Fourth-order folding collision moment]\label{def:pom-s4}
Define the fourth-order collision sum
$$
S_4(m):=\sum_{x\in X_m} d_m(x)^4.
$$
Equivalently, drawing four windows \(\omega_1,\omega_2,\omega_3,\omega_4\) independently and uniformly from \(\Omega_m\), one has
$$
\mathbb{P}\bigl[\Fold_m(\omega_1)=\Fold_m(\omega_2)=\Fold_m(\omega_3)=\Fold_m(\omega_4)\bigr]=\frac{S_4(m)}{16^m}.
$$
\end{definition}

\begin{proposition}[Five-state nonnegative integer realization of the fourth-order collision kernel]\label{prop:pom-s4-recurrence}
There exists a five-state nonnegative integer matrix
$$
A_4:=
\begin{pmatrix}
0&1&0&0&0\\
0&0&7&0&1\\
0&1&2&0&0\\
1&0&1&0&0\\
0&0&0&1&0
\end{pmatrix},
$$
together with readout vectors
$$
u_4:=(9,404,1613,25,114)^{\mathsf T},\qquad
v_4:=(1,0,0,0,0)^{\mathsf T},
$$
such that for all \(m\ge 2\),
$$
S_4(m)=2\,u_4^{\mathsf T}A_4^{\,m-2}v_4.
$$
Consequently \(S_4(m)\) satisfies an exact fifth-order recurrence: for all \(m\ge 13\),
$$
S_4(m)=2S_4(m-1)+7S_4(m-2)+2S_4(m-4)-2S_4(m-5).
$$
\end{proposition}
\begin{proof}
From Table \ref{tab:fold_collision_moment_recursions} one can verify in an auditable fashion that \(S_4(m)\) satisfies an integer recurrence of order \(5\) (script \texttt{scripts/exp\_fold\_collision\_moment\_recursions.py}). On the other hand, the above \(A_4\) and \((u_4,v_4)\) provide a fully explicit five-state nonnegative integer realization; its consistency with the recurrence and with the enumerated values for small \(m\) can be verified term by term via the script \texttt{scripts/exp\_fold\_collision\_kernel\_A4\_verify.py}. \qed
\end{proof}

\begin{corollary}[Fourth-order collision exponent and \texorpdfstring{$h_4$}{h4}]\label{cor:pom-s4-asymptotic}
Let \(r_4:=\rho(A_4)\) be the largest real root of the polynomial
$$
x^5-2x^4-7x^3-2x+2=0
$$
(\(r_4\approx 3.846059041178\)). Then
$$
S_4(m)=\Theta(r_4^m),
\qquad
h_4:=\lim_{m\to\infty}\frac{1}{m}\Bigl(-\log\sum_{x\in X_m}\pi_m(x)^4\Bigr)
=\log\!\left(\frac{16}{r_4}\right).
$$
\end{corollary}
\begin{proof}
By Proposition \ref{prop:pom-s4-recurrence}, the denominator of the generating function is \(\det(I-zA_4)\), whose zero of smallest modulus is \(z_\star=1/\rho(A_4)\), whence \(S_4(m)=\Theta(r_4^m)\). Moreover,
$$
\sum_{x\in X_m}\pi_m(x)^4=\sum_x\frac{d_m(x)^4}{16^m}=\frac{S_4(m)}{16^m},
$$
and substituting the asymptotic yields \(h_4=\log(16/r_4)\). \qed
\end{proof}


\begin{remark}[Near-RH breaking of the fourth-order collision kernel: spectral decomposition numerical certificate (\texorpdfstring{$A_4$}{A4})]\label{rem:pom-collision-rh-break-a4-spectrum}
Write \(r_4=\rho(A_4)\approx 3.846059041178\).

\mu_2\approx -1.963592213449,\qquad
\mu_3\approx 0.503913309522,\qquad
\mu_{4,5}\approx -0.193190068626\pm 0.698726683936\,\mathrm{i}.
\]
Therefore the sub-spectral radius is
\[
\Lambda_4:=\max_{\mu\in\sigma(A_4)\setminus\{r_4\}}|\mu|
=|\mu_2|
\approx 1.963592213449,
\]
while
\(
\sqrt{r_4}\approx 1.961137180612
\), so that the kernel-version RH square-root bound (the analogous condition in Definition \ref{def:kernel-rh}) is strictly violated on the fourth-order collision kernel:

\Lambda_4>\sqrt{r_4},\qquad
R_4:=\frac{\Lambda_4}{\sqrt{r_4}}\approx 1.001251841463.

In terms of the ``pole real-part exponent'' one may write

\beta_4:=\frac{\log \Lambda_4}{\log r_4}\approx 0.500928740205=\frac12+9.2874\times 10^{-4},

and one may introduce the near-critical offset
\[
\delta_4:=\log\!\left(\frac{\Lambda_4^2}{r_4}\right)=2\log R_4\approx 0.002502117125.
\]
 This excess is dominated by a stable negative real mode (\(\mu_2<0\)), hence corresponding to an ineliminable anti-phase oscillatory slow mode.
\end{remark}

\begin{remark}[One-parameter RH critical point of the fourth-order collision kernel: edge-weight threshold \texorpdfstring{$t_\star$}{t*} (resultant elimination, auditable)]\label{rem:pom-collision-rh-break-a4-threshold}
Treating the unique multiple edge weight \(7\) in \(A_4\) as a continuous parameter \(t>0\), define the one-parameter family

\begin{pmatrix}

0&1&2&0&0\\

y^8-2y^6+2y^5-2y^4+2y^3-2=0.

Setting \(z:=-y>0\) yields the equivalent equation

\end{remark}

\begin{remark}[Near-RH breaking of the fourth-order collision kernel (\texorpdfstring{$A_4$}{A4})]\label{rem:pom-collision-rh-break-a4}

\begin{proposition}[Geometrization of \texorpdfstring{$h_q$}{hq}: R\'enyi dimension spectrum of the projection output measure]\label{prop:pom-renyi-dimension-spectrum}
Let \(w_m(x)=d_m(x)/2^m\) be the folding pushforward of the uniform microstate. Identify the resolution \(m\) with a geometric scale via
\[
\varepsilon_m:=\varphi^{-m}.
\]
Define the approximate \(q\)-th order R\'enyi dimension of the output measure at scale \(\varepsilon_m\) as
\[
D_q(m):=\frac{1}{q-1}\cdot \frac{\log\sum_{x\in X_m}w_m(x)^q}{\log \varepsilon_m}
\qquad(q\ge 2).
\]
If the limit \(h_q=\lim_{m\to\infty}\frac{1}{m}H_q(m)\) exists (which holds, for instance, when \(S_q(m)=\Theta(r_q^m)\)), then the limit
\[
D_q:=\lim_{m\to\infty}D_q(m)
\]
exists and satisfies the closed form
\[
\boxed{\;
D_q=\frac{h_q}{(q-1)\log\varphi}\; }.
\]
\end{proposition}
\begin{proof}
By definition \(H_q(m)=-\log\sum_x w_m(x)^q\) and \(\log\varepsilon_m=-m\log\varphi\), so
\[
D_q(m)=\frac{1}{q-1}\cdot \frac{-H_q(m)}{-m\log\varphi}
=\frac{1}{(q-1)\log\varphi}\cdot \frac{1}{m}H_q(m).
\]
Taking \(m\to\infty\) yields the result. \qed
\end{proof}

\begin{proposition}[Universal squeeze: higher-order collision principal exponent and projection entropy rate locked by \texorpdfstring{$|X_m|$}{|Xm|} and the maximum fiber]\label{prop:pom-rq-universal-bounds}
For integer \(q\ge 2\), let
$$
S_q(m):=\sum_{x\in X_m} d_m(x)^q,\qquad d_m(x):=\bigl|\Fold_m^{-1}(x)\bigr|.
$$
Let \(r_q\) denote the exponential growth rate of \(S_q(m)\) (e.g., the Perron root given by Theorem \ref{thm:pom-collision-kernel-ak}; equivalently \(r_q=\lim_{m\to\infty}S_q(m)^{1/m}\)), and define the unnormalized \(q\)-th order projection entropy rate
$$
h_q:=\log\!\left(\frac{2^q}{r_q}\right)=q\log 2-\log r_q.
$$
Then the following two-sided bounds hold:
$$
\frac{2^q}{\varphi^{q-1}}\ \le\ r_q\ \le\ 2\,\varphi^{(q-1)/2},
$$
equivalently,
$$
(q-1)\log\!\left(\frac{2}{\sqrt{\varphi}}\right)\ \le\ h_q\ \le\ (q-1)\log\varphi.
$$
\end{proposition}
\begin{proof}
For any fixed \(m\) and \(q\ge 2\), by the power-mean/H\"older inequality,
$$
S_q(m)=\sum_{x\in X_m}d_m(x)^q\ \ge\ \frac{\bigl(\sum_x d_m(x)\bigr)^q}{|X_m|^{\,q-1}}
=\frac{2^{qm}}{F_{m+2}^{\,q-1}},
$$
where \(\sum_x d_m(x)=|\Omega_m|=2^m\) and \(|X_m|=F_{m+2}\).
On the other hand, letting \(D_m:=\max_{x\in X_m} d_m(x)\),
$$
S_q(m)=\sum_x d_m(x)^q\ \le\ D_m^{\,q-1}\sum_x d_m(x)=2^m D_m^{\,q-1}.
$$
Taking logarithms of both bounds, dividing by \(m\), and letting \(m\to\infty\), using the Fibonacci asymptotic \(F_{m+2}\sim \varphi^{m}/\sqrt5\) and the maximum fiber asymptotic \(D_m=\Theta(\varphi^{m/2})\) (Theorem \ref{thm:pom-max-fiber}), gives
$$
\log r_q=\lim_{m\to\infty}\frac{1}{m}\log S_q(m)\in\Bigl[q\log 2-(q-1)\log\varphi,\ \log 2+\frac{q-1}{2}\log\varphi\Bigr],
$$
so the bounds on \(r_q\) hold. Substituting into \(h_q=q\log 2-\log r_q\) yields the equivalent bounds on \(h_q\). \qed
\end{proof}

\begin{proposition}[Extreme-order endpoint: \texorpdfstring{$r_q^{1/q}\to\sqrt{\varphi}$}{rq^(1/q)->sqrt(phi)} and the normalized entropy endpoint constant]\label{prop:pom-rq-qinfty-endpoint}
Let \(r_q\) denote the exponential growth rate of \(S_q(m)\) (e.g., given by the Perron root of the moment-kernel), and write \(\varphi\) for the golden ratio.
Then as \(q\to\infty\),
\[
\boxed{\;
\lim_{q\to\infty} r_q^{1/q}=\sqrt{\varphi}\; }.
\]
Equivalently,
\[
\lim_{q\to\infty}\frac{1}{q}\,h_q
=\log\!\left(\frac{2}{\sqrt{\varphi}}\right),
\qquad
\lim_{q\to\infty}\frac{1}{q-1}\,h_q
=\log\!\left(\frac{2}{\sqrt{\varphi}}\right),
\]
so the R\'enyi dimension spectrum endpoint satisfies
\[
\lim_{q\to\infty}D_q
=\frac{\log\!\left(2/\sqrt{\varphi}\right)}{\log\varphi}
\approx 0.94042009
\]
(where \(D_q\) is as in Proposition \ref{prop:pom-renyi-dimension-spectrum}).
\par
Moreover, for any integer \(q\ge 2\) there is an explicit error bound
\[
\boxed{\;
0\ \le\ \log\!\left(\frac{2}{\sqrt{\varphi}}\right)-\frac{h_q}{q}\ \le\ \frac{\log\varphi}{q}\; }.
\]
\end{proposition}
\begin{proof}
For any \(m,q\ge 2\), from
\(
D_m:=\max_x d_m(x)
\)
one obtains the two-sided squeeze
\[
D_m^q\ \le\ S_q(m)\ \le\ |X_m|\,D_m^q.
\]
Taking the \(1/(mq)\)-th power of both sides gives
\[
D_m^{1/m}\ \le\ S_q(m)^{1/(mq)}\ \le\ D_m^{1/m}\,|X_m|^{1/(mq)}.
\]
Letting \(m\to\infty\), the left-hand side is given by Theorem \ref{thm:pom-max-fiber} as
\(
\lim_{m\to\infty}D_m^{1/m}=\varphi^{1/2}
\);
the factor on the right-hand side satisfies
\(
\lim_{m\to\infty}|X_m|^{1/(mq)}=\varphi^{1/q}
\)
(since \(|X_m|=F_{m+2}\sim \varphi^m/\sqrt5\)).
Therefore for each fixed \(q\ge 2\),
\[
\sqrt{\varphi}\ \le\ r_q^{1/q}\ \le\ \sqrt{\varphi}\,\varphi^{1/q}.
\]
Letting \(q\to\infty\) gives \(\lim_{q\to\infty}r_q^{1/q}=\sqrt{\varphi}\).
The endpoint limits then follow from \(h_q=q\log 2-\log r_q\) and the closed form in Proposition \ref{prop:pom-renyi-dimension-spectrum}.
\par
From \(\sqrt{\varphi}\le r_q^{1/q}\le \sqrt{\varphi}\,\varphi^{1/q}\) one obtains
\[
\frac{1}{2}\log\varphi\ \le\ \frac{1}{q}\log r_q\ \le\ \frac{1}{2}\log\varphi+\frac{1}{q}\log\varphi.
\]
Substituting into \(\frac{h_q}{q}=\log 2-\frac{1}{q}\log r_q\) yields
\[
0\ \le\ \log\!\left(\frac{2}{\sqrt{\varphi}}\right)-\frac{h_q}{q}\ \le\ \frac{\log\varphi}{q}.
\]
\qed
\end{proof}

\begin{proposition}[Convexity of the pressure and concavity of the fingerprint]\label{prop:pom-pressure-convexity}
For any \(m\ge 2\), the function \(q\mapsto \log S_q(m)\) is convex on \(q\ge 0\).
Therefore, when \(S_q(m)=\Theta(r_q^m)\) (as guaranteed, for instance, by the Perron principal scale of Proposition \ref{prop:pom-moment-kernel-exists}), the pressure
\[
P(q):=\log r_q
\]
satisfies discrete convexity at integer points: for any integers \(q_0<q_1<q_2\),
\[
\boxed{\;
P(q_1)\ \le\ \frac{q_2-q_1}{q_2-q_0}\,P(q_0)\ +\ \frac{q_1-q_0}{q_2-q_0}\,P(q_2)\; }.
\]
Equivalently, the unnormalized R\'enyi fingerprint \(h_q=q\log 2-P(q)\) is concave at integer points:
\[
\boxed{\;
h_{q_1}\ \ge\ \frac{q_2-q_1}{q_2-q_0}\,h_{q_0}\ +\ \frac{q_1-q_0}{q_2-q_0}\,h_{q_2}\; }.
\]
\end{proposition}
\begin{proof}
Fix \(m\). Take any \(q_0,q_2\ge 0\) and \(\theta\in(0,1)\), and let \(q_1=\theta q_0+(1-\theta)q_2\).
By H\"older's inequality (or convexity of log-sum-exp),
\[
S_{q_1}(m)=\sum_{x\in X_m}d_m(x)^{q_1}
\le
\left(\sum_{x\in X_m}d_m(x)^{q_0}\right)^{\theta}
\left(\sum_{x\in X_m}d_m(x)^{q_2}\right)^{1-\theta}
=S_{q_0}(m)^{\theta}S_{q_2}(m)^{1-\theta}.
\]
Taking the logarithm gives the convexity of \(q\mapsto \log S_q(m)\).
\par
Let \(q_0<q_1<q_2\) be integers and take
\(
\theta=\frac{q_2-q_1}{q_2-q_0}
\)
so that \(q_1=\theta q_0+(1-\theta)q_2\).
Taking \(\frac{1}{m}\log\) of the above and letting \(m\to\infty\) gives
\[
\log r_{q_1}\le \theta\log r_{q_0}+(1-\theta)\log r_{q_2},
\]
which is the discrete convexity of \(P\). Finally, \(h_q=q\log 2-P(q)\) is a linear function minus a convex function, hence concave at integer points. \qed
\end{proof}
