\paragraph{Dyadic Chebyshev--Dwork closure, deterministic strong Szeg\H{o} decomposition, and major-arc Gram spectral criterion: from \texorpdfstring{$p$}{p}-adic fingerprints to bottleneck single-spectrum certificates}
The following conclusions unify three superficially heterogeneous ``auditable computation'' mechanisms as referenceable interfaces: congruence closure on the dyadic tower with logarithmic depth, object-level determinant identities for discrete Green/Laplacian with golden-point arithmetization, and inverse characterization of major-arc concentration via finite-dimensional Gram spectral-radius criterion.
\begin{enumerate}
  \item \textbf{\texorpdfstring{$p=2$}{p=2} Frobenius closes over the invariant ring \texorpdfstring{$\ZZ[t]$}{Z[t]}.}
  The completed Dwork congruence strictly closes in the invariant coordinate \(t=u+u^{-1}\) as the Chebyshev--Dwork congruence chain
  \(
  A_{2^k}(t)\equiv A_{2^{k-1}}(t^2-2)\ (\mathrm{mod}\ 2^k)
  \), thereby precisely translating the \(2\)-adic arithmetic compatibility of \(u\mapsto u^2\) into double-angle dynamics on \(\ZZ[t]\) (Theorem~\ref{thm:chebyshev-dwork-congruence-chain}).

  \item \textbf{Matrix-free audit of logarithmic depth: fixed-precision \texorpdfstring{$2^k$}{2^k} computation can be done on univariate dynamics.}
  On the dyadic tower, congruence verification for \(B(u)^{2^k}\) can be replaced by iteratively substituting \(t\mapsto t^2-2\) in \((\ZZ/2^k\ZZ)[t]\) and reducing modulo; the recursion step count is \(k\), logarithmic in the length \(n=2^k\) (Corollary~\ref{cor:chebyshev-dwork-matrixfree}).

  \item \textbf{Stable derivative residual spectrum: \texorpdfstring{$p$}{p}-adic scaling law generates an infinite family of \texorpdfstring{$\mathbb{F}_p$}{F_p} fingerprints.}
  Defining \(\theta\)-derivative moments \(A^{(r)}_n\) via \(u=e^\theta\), the Dwork congruence implies the order-by-order scaling congruence
  \(A^{(r)}_{p^k}\equiv p^rA^{(r)}_{p^{k-1}}\ (\mathrm{mod}\ p^k)\)
  (Proposition~\ref{prop:witt-theta-derivative-filter}), yielding stable leading residues
  \(\alpha_{p,r}\in\mathbb{F}_p\) and \(p=2\) stable binary fingerprints \(\varepsilon_r\in\{0,1\}\) (Corollaries~\ref{cor:witt-theta-stable-residue},~\ref{cor:witt-theta-binary-fingerprint}).

  \item \textbf{Strict identity version of strong Szeg\H{o} decomposition: area term, constant term, and exponentially small correction all explicitly closed.}
  For the mixed-boundary Laplacian \(L_k\) and \(t>0\), there exists an object-level identity
  \[
  \det(L_k+tI)=\frac{\cosh((2k+1)\eta(t))}{\cosh(\eta(t))}
  =q(t)^{-k}\frac{1+q(t)^{2k+1}}{1+q(t)},
  \]
  yielding a strict decomposition of \(\log\det\) into ``area term + constant term + exponentially small remainder'' with explicit upper-bound certificates (Corollary~\ref{cor:pom-Lk-area-law-qform}; see also Proposition~\ref{prop:pom-Kk-strong-szego-bulk-surface}).

  \item \textbf{Uniqueness of golden coupling and mass-gap locking: \texorpdfstring{$t=1$}{t=1} uniquely locks the free-energy density to \texorpdfstring{$2\log\varphi$}{2 log phi}.}
  The equation \(\eta(t)=\log\varphi\) admits at most one solution on \(t>0\), and is strictly equivalent to \(q(t)=\varphi^{-2}\) and
  \(\lim_{k\to\infty}\frac1k\log\det(L_k+tI)=2\log\varphi\),
  yielding the same unique positive solution \(t_\star=1\) (Corollary~\ref{cor:pom-Lk-golden-coupling-unique}).

  \item \textbf{Full arithmetization of the golden point: two-point Green correlation closed form with unified denominator \texorpdfstring{$F_{2k+1}$}{F_{2k+1}}.}
  At \(t=1\), the determinant and inverse matrix elements of \(A_k=L_k+I\) are completely arithmetized:
  \(\det(A_k)=F_{2k+1}\) and
  \((A_k^{-1})_{ij}=\frac{F_{2i}F_{2(k-j)+1}}{F_{2k+1}}\)
  (Corollary~\ref{cor:pom-Lk-t1-fibonacci-det-green}; see also Remark~\ref{rem:pom-Lk-fibonacci-area-law-exact} for Fibonacci-identity realizations of area term/constant term/exponential correction).

  \item \textbf{Exact exponential convergence of boundary Riccati flow and Fibonacci iteration at golden point.}
  The reflection coefficient \(q_k(t)=(L_k+tI)^{-1}_{11}\) satisfies a Schur--Riccati recursion with fully explicit error form
  \(0<q_k(t)-q(t)=\frac{(1-q(t)^2)q(t)^{2k}}{1+q(t)^{2k+1}}\);
  at \(t=1\) it degenerates to the Fibonacci ratio
  \(q_k(1)=F_{2k-1}/F_{2k+1}\to\varphi^{-2}\)
  (Proposition~\ref{prop:pom-Lk-boundary-riccati-recursion}; see also Corollary~\ref{cor:pom-Lk-boundary-fixed-point-riccati}).

  \item \textbf{Boundary thermodynamic response and Parry-measure homologous closure: spectral-side derivative equals measure-side expectation.}
  The surface free energy \(g(t)=-\log(1+q(t))\) satisfies
  \(g'(t)=\frac{q(t)}{1+q(t)}\cdot\frac{1}{\sqrt{t(4+t)}}\)
  (Corollary~\ref{cor:pom-Lk-surface-free-energy});
  at \(t=1\) we have
  \(\frac{q(1)}{1+q(1)}=\pi(1)\) and \(g'(1)=\pi(1)/\sqrt5\), where \(\pi(1)\) is the mass of the Parry maximal-entropy measure for the golden-mean shift on symbol \(1\) (Corollary~\ref{cor:pom-Lk-t1-parry-surface-derivative}).

  \item \textbf{Complete invariant for negative-spectrum determinant: product law of Poisson weights and pseudo-hyperbolic Vandermonde.}
  The Pick--Poisson Gram determinant for point defects has the closed factorization
  \(
  \det\mathsf P=(\prod_j P_{a_j}(-1))(\prod_{j<k}\rho(a_j,a_k)^2)
  \), strictly aligned with the negative-eigenvalue product limit of Toeplitz certificates:
  \(\prod_{j=1}^{\kappa}(-\lambda_j^-(\mathbf{T}_N))\to\det\mathsf P\)
  (Proposition~\ref{prop:xi-pick-poisson-det-factorization}).

  \item \textbf{\texorpdfstring{$M^4$}{M^4} inverse principle and unified bottleneck single-criterion reduction: improvement reducible to numerical decay of rational-frequency Gram spectral radius.}
  Affine uniform residual near-saturation holds if and only if Fourier mass concentrates on some rational major arc (Theorem~\ref{thm:gm-affine-inverse-major-arc}), forcing high-density near-affine autocorrelation structure (Proposition~\ref{prop:gm-major-arc-rigidity-affine-autocorr}).
  Simultaneously, the residual spectral norm is in constant-factor equivalence to the maximal eigenvalue of the explicit finite-dimensional large-sieve Gram matrix \(K_M\) (Theorem~\ref{thm:gm-residual-opnorm-gram-equivalence}), so all power-saving improvement space unifies to the single computable criterion
  \(
  \|K_M\|_{\mathrm{op}}\ll M^{-\vartheta}
  \)
  holding (Corollary~\ref{cor:gm-KM-computable-vtheta}; see also the dichotomy theorem~\ref{thm:gm-affine-improvement-dichotomy}).
\end{enumerate}
