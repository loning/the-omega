\begin{conclusion}[KL exact difference of arithmetic volume vs geometric volume: entropic gap identity for boundary dimension]\label{con:xi-kl-arith-geo-gap-identity}
At resolution \(m\), let fiber multiplicity \(d_m(x):=\card{\Fold_m^{-1}(x)}\) for \(\Fold_m:\Omega_m\to X_m\), satisfying conservation
\(
\sum_{x\in X_m} d_m(x)=\card{\Omega_m}=2^m
\).
On \(X_m\) define two distributions
\[
u_m(x):=\frac{1}{|X_m|},
\qquad
w_m(x):=\frac{d_m(x)}{2^m}.
\]
Then the relative entropy satisfies a strict identity
\[
\boxed{\;
D_{\mathrm{KL}}(u_m\mid w_m)
=\log\frac{2^m}{|X_m|}
-\frac{1}{|X_m|}\sum_{x\in X_m}\log d_m(x)\;}
\]
so that the difference between
\(
\log(2^m/|X_m|)
\)
and
\(
|X_m|^{-1}\sum_x \log d_m(x)
\)
equals exactly the relative entropy from \(u_m\) to \(w_m\).
Moreover
\[
D_{\mathrm{KL}}(u_m\mid w_m)=0
\iff
u_m\equiv w_m
\iff
d_m(x)\equiv \frac{2^m}{|X_m|}\ \ (\forall x\in X_m).
\]
\end{conclusion}

\begin{conclusion}[Three-level sandwich for dimension-volume-entropy: three averages of \texorpdfstring{$\log d$}{log d} form an ordered chain]\label{con:xi-logd-three-average-chain-kl}
Using notation from Conclusion \ref{con:xi-kl-arith-geo-gap-identity}, define three scale exponents
\[
\gamma_m^{\mathrm{geo}}
:=\frac{1}{m|X_m|}\sum_{x\in X_m}\log d_m(x),
\qquad
\gamma_m^{\mathrm{arith}}
:=\frac{1}{m}\log\frac{2^m}{|X_m|},
\qquad
\gamma_m^{\mathrm{Sh}}
:=\frac{1}{m2^m}\sum_{x\in X_m}d_m(x)\log d_m(x).
\]
Then the ordered chain holds
\[
\gamma_m^{\mathrm{geo}}\ \le\ \gamma_m^{\mathrm{arith}}\ \le\ \gamma_m^{\mathrm{Sh}},
\]
and the endpoint differences have exact KL expressions
\[
\gamma_m^{\mathrm{arith}}-\gamma_m^{\mathrm{geo}}=\frac{1}{m}D_{\mathrm{KL}}(u_m\mid w_m),
\qquad
\gamma_m^{\mathrm{Sh}}-\gamma_m^{\mathrm{arith}}=\frac{1}{m}D_{\mathrm{KL}}(w_m\mid u_m).
\]
Therefore both inequalities saturate simultaneously if and only if fibers are completely equimultiple (i.e., \(u_m\equiv w_m\)).
\end{conclusion}

\begin{conclusion}[Jeffreys divergence as numerical certificate of fractal asymmetry: exact formula for gap between two visible means]\label{con:xi-jeffreys-logd-gap-certificate}
Under notation from Conclusion \ref{con:xi-logd-three-average-chain-kl}, define Jeffreys divergence
\[
J(u_m,w_m):=D_{\mathrm{KL}}(u_m\mid w_m)+D_{\mathrm{KL}}(w_m\mid u_m).
\]
Then there is a strict identity
\[
\gamma_m^{\mathrm{Sh}}-\gamma_m^{\mathrm{geo}}=\frac{1}{m}J(u_m,w_m).
\]
Thus the difference in average \(\log d_m\) seen under the two calibers "boundary uniform sampling" vs "microscopic uniform sampling (pushforward weight \(d_m\))" equals exactly the scaled Jeffreys divergence;
this quantity vanishes if and only if \(u_m\equiv w_m\).
\end{conclusion}

\[
m\log 2
=\log|X_m|
+\sum_{x\in X_m}u_m(x)\log d_m(x)\ +\ D_{\mathrm{KL}}(u_m\mid w_m),

\[
m\log 2

\end{conclusion}

\begin{conclusion}[Equimultiplicity rigidity criterion: scaled relative entropy vanishes if and only if geometric mean exponent reaches arithmetic exponent]\label{con:xi-equimultiplicity-kl-vanishing}
Assume the limit
\[
h_\partial:=\lim_{m\to\infty}\frac{1}{m}\log|X_m|

\begin{enumerate}
  \item \(\displaystyle \lim_{m\to\infty}\frac{1}{m}D_{\mathrm{KL}}(u_m\mid w_m)=0\);
  \item \(\displaystyle \lim_{m\to\infty}\gamma_m^{\mathrm{geo}}=\log 2-h_\partial\);
  \item \(\displaystyle \lim_{m\to\infty}\bigl(\gamma_m^{\mathrm{arith}}-\gamma_m^{\mathrm{geo}}\bigr)=0\).
\end{enumerate}
Thus whether scaled relative entropy vanishes can serve as an asymptotic criterion for "boundary thickness uniformity": if it vanishes, the geometric mean exponent reaches the upper bound forced by the boundary volume term; if not, there exists an incompressible entropic gap.
\end{conclusion}

\begin{conclusion}[Entropic criterion for boundary phase transition: linear growth rates of two KLs give critical exponents]\label{con:xi-boundary-phase-transition-kl-rates}

\[
\lim_{m\to\infty}\bigl(\gamma_m^{\mathrm{Sh}}-\gamma_m^{\mathrm{geo}}\bigr)=c_{u\to w}+c_{w\to u}.

Therefore one can view \(c_{u\to w}+c_{w\to u}\) as a univariate certificate of "boundary phase transition strength": it simultaneously measures the separation magnitude of endpoint averages and equals the linear growth rate of Jeffreys divergence.
