\section{Discriminant factorization and the branch locus}\label{sec:disc}

\subsection{Complete description of the branch locus}

View $\Pi(\lambda,y)$ in \eqref{eq:spectral-poly} as a degree-$4$ polynomial in $\lambda$.
Spectral branching---a qualitative change in the root configuration---occurs precisely when two or more roots in $\lambda$ collide, which is controlled by the discriminant.
Computing the discriminant of the spectral polynomial to determine the critical boundary is a standard and well-established technique in the analysis of partition-function zeros; see Sokal~\cite{Sokal2001} for chromatic polynomials and Chang--Shrock~\cite{ChangShrock2001} for transfer-matrix models.

\begin{theorem}\label{thm:disc-factor}
The discriminant of $\Pi(\lambda,y)$ with respect to $\lambda$ admits the complete factorization
\begin{equation}\label{eq:disc-factor}
\Disc_{\lambda}\Pi(\lambda,y)=-y(y-1)\bigl(256y^3+411y^2+165y+32\bigr).
\end{equation}
Hence the parameter values where the root configuration of $\Pi(\cdot,y)$ is non-simple are exactly
\[
\Branch=\{y=0\}\cup\{y=1\}\cup\{256y^3+411y^2+165y+32=0\}.
\]
\end{theorem}

\begin{proof}
The discriminant of a monic degree-$4$ polynomial equals, up to sign, the resultant $\Res_{\lambda}(\Pi,\partial_\lambda\Pi)$, which is the determinant of a $7\times 7$ Sylvester matrix.
The explicit computation and full factorization are given in Appendix~\ref{app:disc}.
\end{proof}

\begin{remark}[Elliptic origin of the factorization]\label{rmk:elliptic-origin}
The factorization~\eqref{eq:disc-factor} admits a divisor-theoretic explanation developed in Appendix~\ref{app:elliptic}.
The spectral curve $\Pi(\lambda,y)=0$ is isomorphic to the elliptic curve $E:\xi^2+\xi=\lambda^3-\lambda$ of conductor~$37$ (Theorem~\ref{thm:ellipticization}), with $y=\lambda^2+\xi$ defining a degree-$4$ morphism $\varphi:E\to\mathbb{P}^1_y$.
The discriminant $\Disc_\lambda(\Pi)$ equals the norm $\mathrm{N}_{\QQ(E)/\QQ(y)}(\partial_\lambda\Pi)$ of the different of the function field extension (Corollary~\ref{cor:norm-identity}), and the branch cubic $c(y)$ is the resultant image of the ramification locus of~$\varphi$:
\[
\operatorname{Res}_\lambda\!\bigl(16\lambda^3-9\lambda^2+1,\;4\lambda y-(4\lambda^3-3\lambda^2-2\lambda+1)\bigr)=-64\,c(y).
\]
The factors $y$ and $y-1$ correspond to the rational ramification points $(1,-1)$ and $(-1,0)$ on~$E$, while the cubic factor $c(y)$ encodes the three non-rational ramification points.
This decomposition into rational and non-rational ramification is intrinsic to the elliptic morphism $\varphi$ and explains the irreducibility of the cubic factor over~$\QQ$.
\end{remark}
\begin{remark}[Auditable \(S_4\) criterion]\label{rmk:galois-s4-main}
By Theorem~\ref{thm:disc-factor}, 
\[
\Disc_\lambda\Pi(\lambda,y)=-y(y-1)c(y),
\]
so for generic \(y\) the discriminant is not a square in \(\QQ(y)\), hence \(\mathrm{Gal}(\Pi/\QQ(y))\not\subseteq A_4\).

A standard second audit step is to specialize at \(y=2\):
\[
\Pi(\lambda,2)=\lambda^4-\lambda^3-5\lambda^2+\lambda+6.
\]
Its reduction modulo \(5\) is irreducible over \(\mathbb F_5\), giving a Frobenius \(4\)-cycle; its reduction modulo \(3\) factors as \(\lambda(\lambda^3+2\lambda^2+\lambda+1)\) with irreducible cubic, giving a Frobenius \(3\)-cycle.
Any transitive subgroup of \(S_4\) containing both a \(4\)-cycle and a \(3\)-cycle is \(S_4\), so the generic Galois group is \(S_4\) (and this is recorded in Proposition~\ref{prop:galois-s4}).
\end{remark}

\subsection{The nontrivial branch cubic and its unique real root}

Let
\begin{equation}\label{eq:cubic}
c(y):=256y^3+411y^2+165y+32.
\end{equation}

\begin{proposition}\label{prop:unique-real-root}
The cubic $c(y)$ has exactly one real root, denoted $\yLY$, and it satisfies
\[
\yLY\in(-1.13446,-1.13445).
\]
\end{proposition}

\begin{proof}
Using the cubic discriminant formula for $a y^3+b y^2+c y+d$, we obtain
\[
\Disc(c)=b^2c^2-4ac^3-4b^3d-27a^2d^2+18abcd= -3^9\cdot 31^2\cdot 37<0
\]
for $(a,b,c,d)=(256,411,165,32)$, so $c$ has one real root and two complex conjugate roots.
Direct evaluation gives
\[
c\!\left(-\frac{113446}{100000}\right)=-\frac{1766421977216}{10^{15}}<0,\qquad
c\!\left(-\frac{113445}{100000}\right)=\frac{442426892000}{10^{15}}>0,
\]
so $\yLY\in(-1.13446,-1.13445)$ is a certified rational enclosure.
\end{proof}

Excluding the endpoint degeneracies $y\in\{0,1\}$, the real axis therefore contains a unique nontrivial branch point $\yLY<0$.

\subsection{The negative-axis non-bifurcating equimodular point}

\begin{definition}[Non-bifurcating negative-axis equimodular point]\label{def:equimodular-negative}
Let $y\in(-1,0)\setminus\Branch$ and write $\lambda_1(y),\dots,\lambda_4(y)$ for the roots of $\Pi(\lambda,y)=0$.
We call $y$ \emph{equimodular} if exactly two distinct branches satisfy
\[
|\lambda_i(y)|=|\lambda_j(y)|=\max_{1\le k\le4}|\lambda_k(y)|,\qquad i\neq j.
\]
It is \emph{non-bifurcating} if the tied pair is not a collision root and if the remaining branch satisfies a strictly smaller modulus.
\end{definition}

\begin{equation}\label{eq:p-eq}
p_{\mathrm{eq}}(y):=256 y^9+1219 y^8+2542 y^7+3090 y^6+2446 y^5+1315 y^4+478 y^3+112 y^2+16 y+1.
\end{equation}

\begin{theorem}\label{thm:yeq}
There is a unique real root \(\yEQ\) of \(p_{\mathrm{eq}}(y)\), and
\[
\yEQ\in\left(-\frac{160092}{10^6},-\frac{160091}{10^6}\right)\subset(-1,0).
\]
At \(y=\yEQ\), \(\Pi(\lambda,\yEQ)\) has a spectral quadruple of the form
\[
\{-r,\, r e^{\pm i\theta},\,\gamma\}
\]
with \(r>0\), \(\theta\in(0,\pi)\), \(\gamma\in\RR\); hence \(|-r|=|r e^{\pm i\theta}|\), so \(y\) is non-bifurcating equimodular in the sense of Definition~\ref{def:equimodular-negative}.
\begin{enumerate}
\item\label{thm:yeq-a} \( \yEQ\notin\{0,1\}\cup\{c(y)=0\}\), and therefore \(\yEQ\notin\Branch\).
\item\label{thm:yeq-b} \(\yEQ\neq\yLY\). In particular, \(y=\yEQ\) is not a collision point.
\end{enumerate}
\end{theorem}

\begin{proof}
Write
\[
p(y):=p_{\mathrm{eq}}(y)=256 y^9+1219 y^8+2542 y^7+3090 y^6+2446 y^5+1315 y^4+478 y^3+112 y^2+16 y+1.
\]
A Sturm computation for \(p'(y)\) shows that \(p'\) has no real zero.
Since \(p'(0)=16>0\), one has \(p'(y)>0\) for all \(y\in\RR\).
Hence \(p\) is strictly increasing on \(\RR\).
Because \(p(-1)=-1<0\) and \(p(0)=1>0\), it follows that \(p\) has a unique real root, necessarily in \((-1,0)\).
The rational sign check
\[
p\!\left(-\frac{160092}{10^6}\right)<0,\qquad
p\!\left(-\frac{160091}{10^6}\right)>0
\]
therefore gives the stated enclosure for \(\yEQ\).

To prove the spectral pattern at \(y=\yEQ\), consider the Vieta system attached to
\[
\Pi(\lambda,y)=(\lambda+r)(\lambda^2-2ru\,\lambda+r^2)(\lambda-\gamma),
\]
where \(r>0\), \(u=\cos\theta\), and \(\gamma\in\RR\).
Comparing coefficients with \(\Pi(\lambda,y)=\lambda^4-\lambda^3-(2y+1)\lambda^2+\lambda+y(y+1)\) gives
\[
\begin{aligned}
\gamma+r(2u-1)-1&=0,\\
r^2(1-2u)+r\gamma(2u-1)+(2y+1)&=0,\\
-r^3+r^2\gamma(1-2u)+1&=0,\\
-r^3\gamma-y(y+1)&=0.
\end{aligned}
\]
A lexicographic Gr\"obner basis elimination of this ideal with variable order
\(\gamma>u>r>y\) contains the four polynomials
\[
464400\,\gamma-G(y),\qquad 464400\,u-U(y),\qquad 154800\,r-R(y),\qquad y(y-1)p(y),
\]
where
\begin{align*}
R(y)&=134486528y^{10}+483593422y^9+607285371y^8+164180370y^7\\
&\quad-388277527y^6-529931405y^5-332584911y^4-117184519y^3\\
&\quad-20570342y^2-1306587y+154800,\\[0.5ex]
U(y)&=482308864y^{10}+1706564761y^9+2094645598y^8+502601835y^7\\
&\quad-1402581401y^6-1833768815y^5-1113100493y^4-374387672y^3\\
&\quad-59074271y^2-3208406y+464400,\\[0.5ex]
G(y)&=268412672y^{10}+995577653y^9+1307666279y^8+418843905y^7\\
&\quad-789459598y^6-1147638370y^5-737672314y^4-263726431y^3\\
&\quad-47434933y^2-3640063y.
\end{align*}
Hence, for any real root \(y_0\) of \(p(y)\) with \(y_0\notin\{0,1\}\), the triple
\[
r_0=\frac{R(y_0)}{154800},\qquad
u_0=\frac{U(y_0)}{464400},\qquad
\gamma_0=\frac{G(y_0)}{464400}
\]
satisfies the Vieta system exactly.

Evaluating these rational functions on the certified interval for \(\yEQ\) yields
\[
\begin{aligned}
0.9567017&<r_0<0.9567033,\\
0.94237518&<u_0<0.94237534,\\
0.15355676&<\gamma_0<0.15355705.
\end{aligned}
\]
Thus \(r_0>0\), \(u_0\in(0,1)\), and \(0<\gamma_0<r_0\).
Let \(\theta_0:=\arccos(u_0)\in(0,\pi/2)\).
Then
\[
\lambda^2-2r_0u_0\,\lambda+r_0^2
=\bigl(\lambda-r_0e^{i\theta_0}\bigr)\bigl(\lambda-r_0e^{-i\theta_0}\bigr),
\]
so
\[
\Pi(\lambda,\yEQ)
=(\lambda+r_0)\bigl(\lambda-r_0e^{i\theta_0}\bigr)\bigl(\lambda-r_0e^{-i\theta_0}\bigr)(\lambda-\gamma_0).
\]
This is the claimed spectral quadruple
\(\{-r_0,r_0e^{\pm i\theta_0},\gamma_0\}\), and \(|\gamma_0|<r_0\) shows that \(\yEQ\) is non-bifurcating equimodular.

Finally, Proposition~\ref{prop:unique-real-root} shows that the unique real root \(\yLY\) of \(c(y)\) lies in \((-1.13446,-1.13445)\), whereas \(\yEQ\in(-1,0)\).
Therefore \(\yEQ\notin\{0,1\}\cup\{c(y)=0\}\), proving \eqref{thm:yeq-a}, and also \(\yEQ\neq\yLY\), proving \eqref{thm:yeq-b}.
\end{proof}

\begin{remark}[Arithmetic of the branch cubic]\label{rmk:branch-cubic-arithmetic}
The discriminant $\Disc(c)=-3^9\cdot 31^2\cdot 37$ has squarefree part $-3\cdot 37=-111$.
The cubic number field $K=\QQ(\yLY)$ has field discriminant $\Disc(K/\QQ)=-3^3\cdot 37=-999$, so the ramification set of~$K$ is $\{3,37\}$---the same primes that control the conductor of the elliptic curve~$E$ (conductor~$37$) and its reduction type.
The unique quadratic subfield of the Galois closure is $\QQ(\sqrt{-111})$, which is the third quadratic subfield of the biquadratic extension $\QQ(\sqrt{-3},\sqrt{37})$; here $\QQ(\sqrt{-3})$ marks the self-inversive channel of $\Pi$ and $\QQ(\sqrt{37})$ is the quadratic resolvent of the $2$-division polynomial of~$E$.
The detailed arithmetic is developed in Appendix~\ref{app:elliptic}, Propositions~\ref{prop:cubic-isomorphism}--\ref{prop:cubic-disc}.
\end{remark}

\subsection{Algebraic parametrization of nontrivial branch collisions}

\begin{theorem}[Nontrivial branch parametrization]\label{thm:branch-param}
Let $y_0\in\RR\setminus\{0,1\}$ and suppose $c(y_0)=0$. Then $\Pi(\lambda,y_0)=0$ has a double root $\lambda_\star$ and
\begin{equation}\label{eq:branch-lam-cubic}
16\lambda_\star^3-9\lambda_\star^2+1=0,\qquad
y_0=\frac{4\lambda_\star^3-3\lambda_\star^2-2\lambda_\star+1}{4\lambda_\star}.
\end{equation}
Conversely, for any root $\lambda_\star$ of the first equation, the $y_0$ in the second equation satisfies $c(y_0)=0$ and gives a nontrivial branch point of $\Pi$.
\begin{proof}
The implication ``$\Pi$ has double root at a nontrivial branch point'' to \eqref{eq:branch-lam-cubic} is the elimination statement in Corollary~\ref{cor:double-root-linear}; combining it with \(\Pi(\lambda_\star,y_0)=0\) yields the cubic in \(\lambda_\star\) and then the linear relation for \(y_0\).
Conversely, eliminating \(\lambda\) between \(16\lambda^3-9\lambda^2+1\) and the affine relation
\[
4\lambda y-(4\lambda^3-3\lambda^2-2\lambda+1)=0
\]
gives
\[
\Res_\lambda\!\bigl(16\lambda^3-9\lambda^2+1,\;4\lambda y-(4\lambda^3-3\lambda^2-2\lambda+1)\bigr)=-64\,c(y),
\]
so the branch-point condition is satisfied.
\end{proof}
\end{theorem}

\begin{theorem}[Puiseux amplitude at a nontrivial collision]\label{thm:branch-puiseux-sign}
Let \(y_0\in\RR\setminus\{0,1\}\) be a nontrivial branch point and \(\lambda_\star\) be its repeated root.
If \(i_1\neq i_2\) index the two colliding local branches of \(\lambda(y)\), then
\[
\lambda_{i_j}(y)=\lambda_\star+(-1)^j\kappa_\star \sqrt{y-y_0}+O(y-y_0),\qquad j=1,2,
\]
with
\begin{equation}\label{eq:kappa-formula-main}
\kappa_\star^2=\frac{2(3\lambda_\star^2-1)}{3(\lambda_\star^2-1)}.
\end{equation}
In particular, $\kappa_\star^2$ is a root of \(324u^3-540u^2+333u-74=0\) (see Section~\ref{app:elliptic}, Theorem~\ref{thm:puiseux-amplitude}), hence \(\kappa_\star^2>0\).
\end{theorem}

\begin{proof}
Since \(y_0\notin\{0,1\}\) and \(c(y_0)=0\), the discriminant factorization
\(\Disc_\lambda\Pi(\lambda,y)=-y(y-1)c(y)\) shows that \(y_0\) is a simple branch value:
the factors \(y\), \(y-1\), and \(c(y)\) are pairwise coprime, and the cubic \(c\) has no repeated root because \(\Disc(c)\neq 0\) by Proposition~\ref{prop:unique-real-root}.
Therefore Appendix~\ref{app:elliptic}, Theorem~\ref{thm:puiseux-amplitude}, applies directly and yields
\[
\lambda_{i_j}(y)=\lambda_\star+(-1)^j\kappa_\star\sqrt{y-y_0}+O(y-y_0),
\qquad
\kappa_\star^2=\frac{2(3\lambda_\star^2-1)}{3(\lambda_\star^2-1)}.
\]
The same theorem shows that \(\kappa_\star^2\) is a root of
\[
324u^3-540u^2+333u-74=0.
\]
For the unique real nontrivial branch point \(y_0=\yLY\), this cubic has one real root, and the sign check
\[
324\cdot 0^3-540\cdot 0^2+333\cdot 0-74<0,\qquad
324\cdot 1^3-540\cdot 1^2+333\cdot 1-74>0
\]
places that root in \((0,1)\), hence \(\kappa_\star^2>0\).
\end{proof}

\subsection{Spectral-root collision at \texorpdfstring{$\yLY$}{y\_LY}}

\begin{proposition}\label{prop:yLY-collision}
At $y=\yLY$, exactly two spectral roots of $\Pi(\lambda,y)$ collide.
For $y$ slightly above $\yLY$, the colliding pair consists of two distinct real roots; for $y$ slightly below $\yLY$, the pair becomes a complex conjugate pair.
We do not assign a dominance status in this theorem.
\end{proposition}

\begin{proof}
At $y=\yLY$, the discriminant $\Disc_\lambda\Pi$ vanishes by Theorem~\ref{thm:disc-factor}, so at least two roots collide.
Since $c(\yLY)=0$ while $\yLY\neq 0,1$, the branch point arises from the cubic factor.
By Corollary~\ref{cor:double-root-linear} (in Appendix~\ref{app:elliptic}), for each branch value of $c(y)=0$ there is a unique linear relation in $\lambda$ determining the repeated root, hence at $y=\yLY$ exactly two roots collide.

For the real-to-complex transition: the polynomial $\Pi(\lambda,y)$ has real coefficients for real~$y$, so non-real roots occur in conjugate pairs.
Near a simple collision of two real roots, the local Puiseux expansion (Theorem~\ref{thm:puiseux-amplitude} in Appendix~\ref{app:elliptic}) shows that
$\lambda(y)\approx \lambda_\star\pm \kappa_\star\sqrt{y-\yLY}$,
with $\kappa_\star^2\in\RR$.
If $\kappa_\star^2>0$, then for $y>\yLY$ the two branches are distinct reals, and for $y<\yLY$ they form a complex conjugate pair.
From Theorem~\ref{thm:puiseux-amplitude}, $\kappa_\star^2=u$ is a root of
\[
324u^3-540u^2+333u-74=0.
\]
This cubic has negative discriminant, so it has exactly one real root. Since
\[
324\cdot 0^3-540\cdot 0^2+333\cdot 0-74<0,\qquad
324\cdot 1^3-540\cdot 1^2+333\cdot 1-74>0,
\]
the real root satisfies $u\in(0,1)$, so $\kappa_\star^2>0$.

\end{proof}
