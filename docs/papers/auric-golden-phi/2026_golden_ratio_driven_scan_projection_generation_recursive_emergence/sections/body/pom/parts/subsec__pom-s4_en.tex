\subsubsection{Fourth-Order Collision Moment and \texorpdfstring{$\mathrm{R\acute{e}nyi}$}{Renyi}--4 Projection Entropy Rate}\label{subsec:pom-s4}
\begin{definition}[Fourth-order folding collision moment]\label{def:pom-s4}
Define the fourth-order collision sum
$$
S_4(m):=\sum_{x\in X_m} d_m(x)^4.
$$
Equivalently, drawing four windows \(\omega_1,\omega_2,\omega_3,\omega_4\) independently and uniformly from \(\Omega_m\), one has
$$
\mathbb{P}\bigl[\Fold_m(\omega_1)=\Fold_m(\omega_2)=\Fold_m(\omega_3)=\Fold_m(\omega_4)\bigr]=\frac{S_4(m)}{16^m}.
$$
\end{definition}

\begin{proposition}[Five-state nonnegative integer realization of the fourth-order collision kernel]\label{prop:pom-s4-recurrence}
There exists a five-state nonnegative integer matrix
$$
A_4:=
\begin{pmatrix}
0&1&0&0&0\\
0&0&7&0&1\\
0&1&2&0&0\\
1&0&1&0&0\\
0&0&0&1&0
\end{pmatrix},
$$
together with readout vectors
$$
u_4:=(9,404,1613,25,114)^{\mathsf T},\qquad
v_4:=(1,0,0,0,0)^{\mathsf T},
$$
such that for all \(m\ge 2\),
$$
S_4(m)=2\,u_4^{\mathsf T}A_4^{\,m-2}v_4.
$$
Consequently, \(S_4(m)\) satisfies an exact fifth-order recurrence: for all \(m\ge 13\),
$$
S_4(m)=2S_4(m-1)+7S_4(m-2)+2S_4(m-4)-2S_4(m-5).
$$
\end{proposition}
\begin{proof}
By Table \ref{tab:fold_collision_moment_recursions}, one can verify in an auditable manner that \(S_4(m)\) satisfies an integer recurrence of order \(5\) (script \texttt{scripts/exp\_fold\_collision\_moment\_recursions.py}). On the other hand, the above \(A_4\) and \((u_4,v_4)\) provide a completely explicit five-state nonnegative integer realization; its consistency with the recurrence and the enumerated values for small \(m\) can be verified entry by entry via the script \texttt{scripts/exp\_fold\_collision\_kernel\_A4\_verify.py}. \qed
\end{proof}

\begin{remark}[Zero-coefficient lock: strict missing term in the fourth-order minimal recurrence]\label{rem:pom-s4-zero-coefficient-lock}
The fifth-order recurrence given in Proposition \ref{prop:pom-s4-recurrence} exhibits a strict internal gap in its lag structure: the coefficient of \(S_4(m-3)\) is identically \(0\).
Equivalently, its characteristic polynomial (i.e., \(\det(\lambda I-A_4)\))
\[
\chi_{A_4}(\lambda)=\lambda^5-2\lambda^4-7\lambda^3-2\lambda+2
\]
strictly lacks the \(\lambda^2\) term.
Among the minimal integer-coefficient recurrences verified in the range \(q=2,\dots,23\) (Tables \ref{tab:fold_collision_moment_recursions}, \ref{tab:fold_collision_moment_recursions_9_17}, \ref{tab:fold_collision_moment_recursions_18_23}), such structural ``mid-order zero coefficient'' appears only at \(q=4\).
Moreover, this gap is consistent with the spectral self-duality freezing identity of the fourth-order kernel family: for a one-parameter family \(A_4(t)\) with characteristic polynomial \(P_t\), its even part is completely independent of \(t\), as shown in Proposition \ref{prop:pom-a4t-spectral-selfduality-invariants}.
Therefore, beyond the near-RH critical-line margin and crossing order, \(q=4\) carries an even finer auditable fingerprint: it is the smallest integer matrix order where the minimal recurrence exhibits a strict coefficient gap at the integer-coefficient level.
\end{remark}

\begin{corollary}[Fourth-order collision exponent and \texorpdfstring{$h_4$}{h4}]\label{cor:pom-s4-asymptotic}
Let \(r_4:=\rho(A_4)\) be the largest real root of the polynomial
$$
x^5-2x^4-7x^3-2x+2=0
$$
(\(r_4\approx 3.846059041178\)). Then
$$
S_4(m)=\Theta(r_4^m),
\qquad
h_4:=\lim_{m\to\infty}\frac{1}{m}\Bigl(-\log\sum_{x\in X_m}\pi_m(x)^4\Bigr)
:=\log\!\left(\frac{16}{r_4}\right).
$$
\end{corollary}
\begin{proof}
By Proposition \ref{prop:pom-s4-recurrence}, the generating function denominator is \(\det(I-zA_4)\), whose smallest-modulus zero is \(z_\star=1/\rho(A_4)\), whence \(S_4(m)=\Theta(r_4^m)\). Moreover,
$$
\sum_{x\in X_m}\pi_m(x)^4=\sum_x\frac{d_m(x)^4}{16^m}=\frac{S_4(m)}{16^m},
$$
and substituting the asymptotic yields \(h_4=\log(16/r_4)\). \qed
\end{proof}

\begin{proposition}[Negative real sub-spectrum induces Chebyshev-type deviation leading term: two-term expansion of trace and primitive orbits]\label{prop:pom-a4-trace-primitive-two-term}
Following the notation in Remark \ref{rem:pom-collision-rh-break-a4-spectrum}, let \(r_4=\rho(A_4)\), and let \(\mu_-<0\) be the unique negative real eigenvalue satisfying \(|\mu_-|=\Lambda_4\) (numerically, \(\mu_-=\mu_2\)).
Define
\[
\rho_4:=\max\!\left\{\sqrt{r_4},\ \max_{\mu\in\sigma(A_4)\setminus\{r_4,\mu_-\}}|\mu|\right\}.
\]
Then \(\rho_4<\Lambda_4\), and for all \(n\ge 1\) the two-term expansion holds
\[
\Tr(A_4^n)=r_4^n+\mu_-^n+O(\rho_4^n)
=r_4^n+(-\Lambda_4)^n+O(\rho_4^n).
\]
Correspondingly, the primitive orbit count \(p_n(A_4)\) (Definition \ref{def:pom-proj-prim}) satisfies
\[
p_n(A_4)=\frac{r_4^n}{n}+\frac{\mu_-^n}{n}+O\!\Bigl(\frac{\rho_4^n}{n}\Bigr).
\]
Therefore the even/odd length discrepancy in \(p_n(A_4)\) is exponentially dominated by \(\Lambda_4^n/n\) and exhibits strict alternating sign.
\end{proposition}
\begin{proof}
Since \(\mathrm{Disc}_x(p_7)\neq 0\) (Proposition \ref{prop:pom-a4-galois-s5}), \(A_4\)'s characteristic polynomial has no repeated roots, hence \(A_4\) is diagonalizable over \(\CC\), and
\[
\Tr(A_4^n)=\sum_{\mu\in\sigma(A_4)}\mu^n.
\]
Decompose the spectrum as \(\{r_4,\mu_-\}\cup(\sigma(A_4)\setminus\{r_4,\mu_-\})\), and control the remaining terms by \(\rho_4\) to obtain the trace expansion.
Note that \(\rho_4<\Lambda_4\) follows from \(\Lambda_4>\sqrt{r_4}\) (Remark \ref{rem:pom-collision-rh-break-a4-spectrum}).
\par
The Möbius inversion for primitive orbit counts gives
\(
p_n(A_4)=\frac{1}{n}\sum_{d\mid n}\mu(d)\Tr(A_4^{n/d})
\).
Substituting the trace expansion: the leading term comes from \(d=1\);
while terms with \(d\ge 2\) satisfy \(|\Tr(A_4^{n/d})|\ll r_4^{n/d}\le r_4^{n/2}=(\sqrt{r_4})^n\le \rho_4^n\),
so their total contribution is \(O(\rho_4^n/n)\).
Meanwhile \(\mu_-^n=(-\Lambda_4)^n\) provides the alternating-sign sub-leading term. \qed
\end{proof}

\begin{remark}[Near-RH violation of the fourth-order collision kernel (\texorpdfstring{$A_4$}{A4})]\label{rem:pom-collision-rh-break-a4}

\medskip\noindent\textbf{Audit conclusion (derivative formula and algebraic certificate for the leading-pole residue constant \(C\): \(A_2,A_3,A_4\))}
Let \(A\) be a finite-state collision kernel (SFT adjacency matrix), with characteristic polynomial normalized to have leading coefficient \(1\):
\[

and suppose the Perron root \(r=\rho(A)\) is a simple root. Write

\[
\boxed{\;
C(A)=\frac{r^{d-1}}{\chi_A'(r)}
\;=\;
\prod_{\lambda_j\in\sigma(A)\setminus\{r\}}\left(1-\frac{\lambda_j}{r}\right)^{-1}\; }.
\]
Therefore, once \(\chi_A\) and \(r\) have been independently audited (e.g., given by the recurrence denominator/characteristic polynomial and the Perron root certificate), \(C(A)\) can be exactly reconstructed without computing a residue limit or explicitly finding the full spectrum; moreover, any nonnegative integer realization preserving the same \(\chi_A\) (see Remark \ref{rem:pom-a3-alternative-realization}) must yield the same \(C(A)\).

\paragraph{Symbolic dynamics and operator algebra invariants: Bowen--Franks group, flow equivalence, and Cuntz--Krieger \texorpdfstring{$K$}{K}-theory}
Besides \(\zeta\), primitive spectra, and finite-part constants, integer adjacency matrices carry a suite of integer lattice invariants highly sensitive to flow equivalence and stable isomorphism classification (see Appendix \ref{app:op-algebra} and corresponding discussions in the real-input kernel appendix).
For any integer matrix \(A\in M_n(\ZZ)\), define the Bowen--Franks group
\[
\mathrm{BF}(A):=\ZZ^n/(I-A)\ZZ^n,
\]
(see \cite{BowenFranks1977HomologyZeroDim})
and for the Cuntz--Krieger algebra \(\mathcal O_A\) defined by \(A\), take the standard isomorphisms
\[
K_0(\mathcal O_A)\cong \mathrm{coker}(I-A^\mathsf T),\qquad
K_1(\mathcal O_A)\cong \ker(I-A^\mathsf T)
\]
(see \cite{CuntzKrieger1980}).

\begin{proposition}[Smith normal form and Bowen--Franks group of low-order collision kernels]\label{prop:pom-collision-bf-snf-q234}
Let \(A_2,A_3,A_4(t)\) be the low-order collision kernels from Proposition \ref{prop:pom-s2-recurrence}, Proposition \ref{prop:pom-s3-recurrence}, and Remark \ref{rem:pom-collision-rh-break-a4-threshold} respectively (where \(t\in\ZZ_{\ge 0}\)).
Then
\[
\det(I-A_2)=-1,\qquad \det(I-A_3)=-3,\qquad \det(I-A_4(t))=-(t+1),
\]
and the corresponding Smith normal forms satisfy
\[
\mathrm{SNF}(I-A_2)=\mathrm{diag}(1,1,1),\qquad
\mathrm{SNF}(I-A_3)=\mathrm{diag}(1,1,3),\qquad
\mathrm{SNF}(I-A_4(t))=\mathrm{diag}(1,1,1,1,t+1).
\]
Therefore
\[
\mathrm{BF}(A_2)=0,\qquad
\mathrm{BF}(A_3)\cong \ZZ/3\ZZ,\qquad
\mathrm{BF}(A_4(t))\cong \ZZ/(t+1)\ZZ.
\]
\end{proposition}
\begin{proof}
For \(A_2,A_3\), directly compute \(I-A_q\) and perform Smith reduction to read off the invariant factors; for \(A_4(t)\), perform the same reduction for integer \(t\) and verify its diagonal form's stable pattern as a function of \(t\) (consistent with \(\det(I-A_4(t))=-(t+1)\)).
Reproducible script: \texttt{scripts/exp\_collision\_kernel\_A234\_operator\_algebra\_flow\_invariants.py}. \qed
\end{proof}

\begin{corollary}[Flow equivalence types of low-order collision kernels]\label{cor:pom-collision-flow-equivalence-full-shift}
For the three kernel classes in Proposition \ref{prop:pom-collision-bf-snf-q234}, all satisfy \(\det(I-A_q)<0\), and the corresponding edge shifts are irreducible shifts of finite type.
Therefore by Franks's flow equivalence classification theorem (with \(\mathrm{BF}(A)\) and \(\mathrm{sgn}\det(I-A)\) as complete invariants; see \cite{Franks1984FlowEquivalenceSFT}) one has:
\[
(X_{A_2},\sigma)\ \sim_{\mathrm{flow}}\ \text{full }2\text{-shift},\qquad
(X_{A_3},\sigma)\ \sim_{\mathrm{flow}}\ \text{full }4\text{-shift},
\]
\[
(X_{A_4(t)},\sigma)\ \sim_{\mathrm{flow}}\ \text{full }(t+2)\text{-shift}\qquad(t\in\ZZ_{\ge 0}).
\]
\end{corollary}

\begin{remark}[Flow equivalence invariant collapse and fine spectral information invisibility]\label{rem:pom-collision-flow-collapse}
Corollary \ref{cor:pom-collision-flow-equivalence-full-shift} shows: despite the low-order collision kernels exhibiting nontrivial fine structure at the spectral decomposition and arithmetic certificate level (e.g., anti-phase slow oscillations triggered by negative real modes, stable existence of non-real spectrum in the fourth-order kernel family, and near-RH critical-line behavior), the Franks invariant recognizes only a full shift flow equivalence type on a single vertex.
Therefore, to capture the fine spectral/arithmetic information of collision kernels, one must appeal to dynamical invariants finer than flow equivalence or higher operator-algebra hierarchy structures.
\end{remark}

\begin{proposition}[Cuntz--Krieger algebras of low-order collision kernels: complete identification with Cuntz algebra families]\label{prop:pom-collision-ck-classification-q234}
For the three kernel classes in Proposition \ref{prop:pom-collision-bf-snf-q234}, their Cuntz--Krieger algebra \(K\)-groups satisfy
\[
K_0(\mathcal O_{A_2})=0,\ K_1(\mathcal O_{A_2})=0;\qquad
K_0(\mathcal O_{A_3})\cong \ZZ/3\ZZ,\ K_1(\mathcal O_{A_3})=0;\qquad
K_0(\mathcal O_{A_4(t)})\cong \ZZ/(t+1)\ZZ,\ K_1(\mathcal O_{A_4(t)})=0.
\]
Moreover, the unit classes satisfy: \([1_{\mathcal O_{A_3}}]=0\in K_0(\mathcal O_{A_3})\), while \([1_{\mathcal O_{A_4(t)}}]\) is a generator in \(K_0(\mathcal O_{A_4(t)})\).
Hence in the Kirchberg--Phillips classification sense (with \(K\)-theory invariants plus unit class; see \cite{Phillips2000ClassificationPurelyInfinite}) one has isomorphic identifications
\[
\mathcal O_{A_2}\cong \mathcal O_2,\qquad
\mathcal O_{A_3}\cong M_3(\mathcal O_4),\qquad
\mathcal O_{A_4(t)}\cong \mathcal O_{t+2}\ \ (t\in\ZZ_{\ge 0}),
\]
in particular \(\mathcal O_{A_4}\cong \mathcal O_9\).
\end{proposition}
\begin{proof}
Since \(\det(I-A_q)\neq 0\), \(I-A_q^\mathsf T\) is invertible over \(\QQ\), hence \(K_1(\mathcal O_{A_q})\cong\ker(I-A_q^\mathsf T)=0\).
On the other hand, Proposition \ref{prop:pom-collision-bf-snf-q234}'s Smith form gives
\(
K_0(\mathcal O_{A_q})\cong\mathrm{coker}(I-A_q^\mathsf T)\cong\mathrm{coker}(I-A_q)
\)'s cyclic decomposition.
The unit class corresponds to the image of \((1,\dots,1)^{\mathsf T}\) in the cokernel: for \(A_3\), take \(x=(1,0,-3)^{\mathsf T}\) satisfying
\(
(I-A_3^\mathsf T)x=(1,1,1)^{\mathsf T}
\), hence \([1_{\mathcal O_{A_3}}]=0\).
For \(A_4(t)\), let \(M:=I-A_4(t)^\mathsf T\), \(v:=(1,1,1,1,1)^{\mathsf T}\), then
\[
\operatorname{adj}(M)\,v=\bigl(-(3t+4),\ -1,\ 4(t+1),\ -(2t+3),\ -(t+2)\bigr)^{\mathsf T},
\]
whose coordinate gcd is \(1\). If there exists \(x\in\ZZ^{5}\) with \(Mx=kv\), then by \(\operatorname{adj}(M)Mx=\det(M)x\) we get \((t+1)\mid k\).
On the other hand, by the identity \(M\,\operatorname{adj}(M)=\det(M)I\) we know \(x=\operatorname{adj}(M)v\) gives \(Mx=\det(M)v\), whence \(v\)'s order in \(\mathrm{coker}(M)\cong \ZZ/(t+1)\ZZ\) is exactly \(t+1\), i.e., \([1_{\mathcal O_{A_4(t)}}]\) is a generator.
Finally, \(\mathcal O_n\)'s invariants are \((K_0,K_1,[1])=(\ZZ/(n-1)\ZZ,0,1)\), and \(M_k(\mathcal O_n)\)'s unit class is \(k\cdot [1]\); according to this one obtains the stated isomorphic identifications. \qed
\end{proof}
