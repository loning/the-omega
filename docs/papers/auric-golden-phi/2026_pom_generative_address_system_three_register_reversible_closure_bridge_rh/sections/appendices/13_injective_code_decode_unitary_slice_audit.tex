\section{注入/编码的完全形式化（code/decode）与 unitary slice 审计接口的数值证书协议}\label{app:inject-code-audit}

本附录给出两类“可审计注入”的完全形式化：
\begin{itemize}
  \item (i) 用于 prime-register 残差承载的事件编码（$\mathrm{code}/\mathrm{decode}$，可判定互逆）；
  \item (ii) 用于 unitary slice（$\abs{u}=1$）零点审计的数值证书协议（误差界、舍入模型与 $\NULL$ 判定的可重复性）。
\end{itemize}

\subsection{预备：基础可判定编码器（整数、配对、有限序列）}

\paragraph{整数到自然数的双射编码}
定义编码 $\eta_{\ZZ}:\ZZ\to\NN$ 与解码 $\eta_{\ZZ}^{-1}:\NN\to\ZZ$：
$$
\eta_{\ZZ}(z)=
\begin{cases}
2z, & z\ge 0,\\
-2z-1, & z<0,
\end{cases}
\qquad
\eta_{\ZZ}^{-1}(n)=
\begin{cases}
n/2, & n\ \text{为偶数},\\
-(n+1)/2, & n\ \text{为奇数}.
\end{cases}
$$
则 $\eta_{\ZZ}$ 为双射，且 $\eta_{\ZZ}^{-1}$ 为其严格逆；两者均为可判定（由奇偶与整数运算决定）。

\paragraph{Cantor 配对函数与可判定互逆}
定义 Cantor 配对函数 $\pi:\NN^2\to\NN$：
$$
\pi(a,b)=\frac{(a+b)(a+b+1)}{2}+b .
$$
给定 $n\in\NN$，其逆分解可用以下可判定算法实现：令
$$
w=\left\lfloor\frac{\sqrt{8n+1}-1}{2}\right\rfloor,\qquad t=\frac{w(w+1)}{2},
$$
再令
$$
b=n-t,\qquad a=w-b .
$$
则得到 $(a,b)=\pi^{-1}(n)$，并满足 $\pi(a,b)=n$。因此 $\pi$ 为双射，且 $\pi^{-1}$ 可判定（整数平方根与取整运算可实现）。
该配对可迭代用于高维元组编码：
$$
\pi_3(a,b,c)=\pi\bigl(a,\pi(b,c)\bigr),\quad
\pi_k(a_1,\dots,a_k)=\pi\bigl(a_1,\pi_{k-1}(a_2,\dots,a_k)\bigr).
$$

\paragraph{自描述有限序列编码（\texorpdfstring{$\mathrm{Seq}/\mathrm{Unseq}$}{Seq/Unseq}）}
除 Gödel（素数幂）编码外，我们还需要一种\textbf{不依赖外部长度参数}的序列编码，以便把“事件—轨迹—证书材料”直接压缩为单一自然数并可判定反解（正文第 \ref{sec:conclusion-cluster} 节）。下文也记
$
\langle a,b\rangle:=\pi(a,b)
$。
定义
$$
\mathrm{Seq}():=0,\qquad
\mathrm{Seq}(x_1,\dots,x_T):=\langle x_1,\mathrm{Seq}(x_2,\dots,x_T)\rangle+1,
$$
并定义
$$
\mathrm{Unseq}(0):=(),\qquad
\mathrm{Unseq}(n):=(x_1,\dots,x_T)
$$
当且仅当 $\pi^{-1}(n-1)=(x_1,r)$ 且 $\mathrm{Unseq}(r)=(x_2,\dots,x_T)$。

\begin{proposition}[自描述序列的可判定互逆]\label{prop:seq-unseq-invertible}
$\mathrm{Seq}:\NN^{<\omega}\to\NN$ 与 $\mathrm{Unseq}:\NN\to\NN^{<\omega}$ 为可判定互逆双射：
$$
\mathrm{Unseq}(\mathrm{Seq}(s))=s,\qquad
\mathrm{Seq}(\mathrm{Unseq}(n))=n.
$$
\end{proposition}

\begin{proof}
对序列长度归纳。$T=0$ 显然。$T>0$ 时由 $\pi,\pi^{-1}$ 的互逆性与递归定义得到唯一性与互逆性；可判定性来自每一步仅调用一次 $\pi^{-1}$ 与有限次整数运算。证毕。
\end{proof}

\paragraph{Gödel（素数幂）编码与可判定互逆}
记素数序列 $p_1<p_2<\cdots$。对有限序列 $\mathbf n=(n_1,\dots,n_T)\in\NN^T$ 定义 Gödel 编码
$$
\mathrm{G}(\mathbf n)=\prod_{t=1}^{T} p_t^{n_t+1}\in\NN_{\ge 1}.
$$
由算术基本定理，$\mathrm{G}$ 为单射。解码可由 $p$-进赋值 $v_p(\cdot)$ 给出：若 $N=\mathrm{G}(\mathbf n)$，则
$$
n_t=v_{p_t}(N)-1,\qquad t=1,\dots,T,
$$
并且 $T$ 可由最大满足 $v_{p_T}(N)>0$ 的 $T$ 唯一确定。
该解码在可计算意义上是可判定的：对任意固定输入 $N$，可用试除或一般分解算法计算各 $v_{p_t}(N)$（复杂度不作承诺，但可判定性成立）。

\paragraph{注}
本文 prime-register 残差向量采用“时间轴 $\leftrightarrow$ 素数轴”的构造：第 $t$ 步事件写入 $p_t$ 轴，形式上等价于对事件序列作 Gödel 编码。其“注入且可回放”的语义要求与结构性动机见附录 \ref{app:prime-uplift}，并与正文命题 \ref{prop:lossless-uplift} 的无损提升口径一致。

\subsection{原子事件的编码：\texorpdfstring{$\mathrm{code}/\mathrm{decode}$}{code/decode} 的可判定互逆性}

\paragraph{原子事件数据结构（抽象最小化）}
在分辨率 $m$ 下，折叠正规化策略 $S$ 产生一条有限局部原子重写轨迹
$$
a_0\to a_1\to\cdots\to a_T=\mathrm{NF}(a_0),
$$
并把第 $t$ 步原子事件记为 $e_t(\omega)$。我们要求存在一个“可审计的注入编码”
$$
\mathrm{code}:\mathcal E_m\hookrightarrow \NN_{\ge 1},
$$
且编码中包含“规则类型与位置等足以在回放时唯一确定逆操作的信息”。

为使该要求完全形式化，固定一个有限事件字母表 $\mathcal E_m$，其元素写为
$$
e=(\rho,k,\tau)\in \mathcal R\times \NN\times \NN,
$$
其中：
\begin{itemize}
  \item $\rho\in\mathcal R$ 为规则类型标签（例如 $R_{\mathrm{adj}},R_1,R_2,R_k$ 等有限集合；亦可把“方向/分支/消歧位”并入 $\rho$）；
  \item $k\in\NN$ 为窗口位置（或中心坐标）；
  \item $\tau\in\NN$ 为必要的消歧/侧条件标记（当仅凭 $(\rho,k)$ 不能唯一确定逆操作时引入；若不需要，可固定 $\tau=0$）。
\end{itemize}

要求“可回放性”在语义上等价于：对每个事件 $e$，存在一个部分双射 $E_e$（定义域由规则侧条件决定），使得
$$
a_{t}=E_{e_t}(a_{t-1}),\qquad a_{t-1}=E_{e_t}^{-1}(a_{t}),
$$
并且在回放阶段 $E_{e_t}^{-1}$ 可由 $e_t$ 唯一确定。

\paragraph{明确的 Cantor-配对式事件编码}
固定规则集合的一个可判定枚举
$$
\mathrm{idx}:\mathcal R\to\{0,1,\dots,\card{\mathcal R}-1\},
\quad
\mathrm{idx}^{-1}:\{0,1,\dots,\card{\mathcal R}-1\}\to\mathcal R .
$$
定义事件编码
$$
\mathrm{code}(e)=1+\pi_3\bigl(\mathrm{idx}(\rho),k,\tau\bigr)\in\NN_{\ge 1}.
$$
定义解码函数 $\mathrm{decode}:\mathrm{Im}(\mathrm{code})\to \mathcal E_m$ 为：
$$
\mathrm{decode}(n)=
\bigl(\mathrm{idx}^{-1}(r),k,\tau\bigr),
\quad \text{其中 } (r,k,\tau)=\pi_3^{-1}(n-1).
$$
\begin{proposition}[可判定互逆性]\label{prop:code-decode-invertible}
对任意 $e\in\mathcal E_m$ 有 $\mathrm{decode}(\mathrm{code}(e))=e$。对任意 $n\in\mathrm{Im}(\mathrm{code})$ 有 $\mathrm{code}(\mathrm{decode}(n))=n$。并且 $\mathrm{code},\mathrm{decode}$ 均为可判定函数（由 $\pi,\pi^{-1}$ 与有限枚举的可判定性给出）。
\end{proposition}

\begin{proof}
$\pi_3$ 为双射（由 Cantor 配对的可判定互逆迭代而得），$+1$ 仅用于把编码像落在 $\NN_{\ge 1}$。故互逆性直接成立；可判定性由显式逆算法给出。
\end{proof}

\paragraph{Gödel-素数幂式事件编码（可选替代）}
亦可把事件三元组先映射为一个自然数
$$
g(e)=\pi_3\bigl(\mathrm{idx}(\rho),k,\tau\bigr)\in\NN,
$$
再用素数幂单点编码
$$
\mathrm{code}_{\mathrm{G}}(e)=2^{g(e)+1}\in\NN_{\ge 1}.
$$
则解码由对 $2$-进赋值 $v_2$ 完成：
$$
g(e)=v_2(\mathrm{code}_{\mathrm{G}}(e))-1,
$$
再用 $\pi_3^{-1}$ 还原 $(\rho,k,\tau)$。该方案在“只需单素数轴”场景中尤为直接，但在 prime-register 的“时间轴写入不同素数”方案下通常不必采用。

\subsection{prime-register 残差向量的编码与回放：从 \texorpdfstring{$r^S(\omega)$}{r^S(omega)} 到可构造逆过程}

\paragraph{残差向量定义与“时间轴到素数轴”写入}
在固定确定性策略 $S$ 后，对输入 $\omega\in\Omega_m$ 构造 prime-register 残差向量 $r^S(\omega)\in P$：
$$
r^S(\omega)(p_t)=\mathrm{code}\bigl(e_t(\omega)\bigr)\quad (1\le t\le T),
\qquad r^S(\omega)(p)=0\ \text{对其余素数 }p.
$$
并定义无损折叠提升
$$
\widehat{\Fold}_m^{\,S}(\omega)=\bigl(\Fold_m(\omega),\,r^S(\omega)\bigr)\in X_m\times P.
$$

此处 $P$ 为有限支撑指数向量空间（素数寄存器纤维）并带值映射 $N(r)=\prod_p p^{r(p)}$，由算术基本定理与 $\NN_{\ge 1}$ 同构。该“素数幂坐标承载残差”的动机与结构性说明见注 7.30（正文）与附录 \ref{app:prime-uplift}。

\paragraph{形式化回放算法 \texorpdfstring{$\mathrm{UNLIFT}_m^{\,S}$}{UNLIFT\_m^S}（部分函数口径）}
\textbf{输入：}$(x,r)\in X_m\times P$。\\
\textbf{输出：}若可回放则输出唯一 $\omega\in\Omega_m$，否则输出 $\NULL$。

\begin{enumerate}
  \item \textbf{事件序列提取：}令 $T$ 为满足 $r(p_T)\neq 0$ 的最大 $T$（若 $r=0$ 则 $T=0$）。若存在 $1\le t\le T$ 使 $r(p_t)=0$，则输出 $\NULL$。否则对每个 $1\le t\le T$：
  $$
  n_t:=r(p_t)\in\NN_{\ge 1},\qquad e_t:=\mathrm{decode}(n_t)\in\mathcal E_m.
  $$
  若对某个 $t$，$n_t\notin\mathrm{Im}(\mathrm{code})$（等价地 $\mathrm{decode}(n_t)$ 不定义），则输出 $\NULL$。

  \item \textbf{从正规形启动：}以 $x$ 的嵌入配置 $b_T:=\iota_m(x)$ 作为回放起点（$\iota_m$ 为稳定代表到配置空间的标准嵌入）。

  \item \textbf{逆序回放：}对 $t=T,T-1,\dots,1$，定义
  $$
  b_{t-1}:=E_{e_t}^{-1}(b_t),
  $$
  其中 $E_{e_t}^{-1}$ 为由事件 $e_t$ 唯一确定的逆原子重写。若某一步逆操作侧条件不满足（导致 $E_{e_t}^{-1}$ 不定义），则输出 $\NULL$。

  \item \textbf{读出原始微态：}由 $b_0$ 反推出唯一 $\omega\in\Omega_m$（即 $\omega=\iota_m^{-1}(b_0)$），并输出之。
\end{enumerate}

\begin{proposition}[回放正确性与可判定互逆]\label{prop:unlift-correctness}
对任意 $\omega\in\Omega_m$，有
$$
\mathrm{UNLIFT}_m^{\,S}\!\left(\widehat{\Fold}_m^{\,S}(\omega)\right)=\omega.
$$
因此 $\widehat{\Fold}_m^{\,S}$ 为单射，且 $\mathrm{UNLIFT}_m^{\,S}$ 给出其可构造逆过程；这与正文命题 \ref{prop:lossless-uplift} 以及命题 \ref{prop:uplift-injective}（附录 \ref{app:prime-uplift}）的“无损提升/可逆回放”口径一致。
\end{proposition}

\begin{proof}
(1) 由 $r$ 的定义，支撑 $\{p_t\}_{t\le T}$ 与值 $r(p_t)=\mathrm{code}(e_t)$ 唯一确定事件序列；\\
(2) 由编码约定，每步可唯一决定逆原子重写；\\
(3) 从 $\iota_m(x)$ 出发逆序回放唯一恢复 $\iota_m(\omega)$，从而恢复 $\omega$。
\end{proof}

\subsection{unitary slice 审计接口：数值证书协议、舍入模型与 \texorpdfstring{$\NULL$}{NULL} 判定的可重复性}

\paragraph{类型化域：unitary slice 的“可见参数轴”与 \texorpdfstring{$\NULL$}{NULL} 语义接口}
在 $\Xi_{\Delta,L}$ 的审计口径中，定义
$$
u_L(s)=L^{2s-1}.
$$
对任意 $L>1$，由命题 \ref{prop:critical-line-unitary-slice} 有等价关系
$$
\begin{aligned}
\Re(s)=\frac12
&\Longleftrightarrow\ \abs{u_L(s)}=1
\Longleftrightarrow\ u_L(s)\in \mathrm{Phase},\\
\mathrm{Phase}
&=\{u\in\CC:\abs{u}=1\}
=\{e^{i\theta}:\theta\in\RR\}.
\end{aligned}
$$

在三生成元片段 $(\mathrm{PROJ}_u,E_m,\mathrm{LIFT}_G)$ 的正规化闭包中，可复核的连续参数轴被限制为 unitary slice 的相位 $\theta$，而非一般分解 $u=re^{i\theta}$ 的径向参数 $r$。因此，“可寻址的零点证书”只能以 $\theta$-口径呈现（亦见附录 \ref{app:unitary-slice}）。

POM 的空值语义规定：无地址则读出不定义，记为 $\NULL$。在 unitary slice 类型安全陈述中，“偏离 $\Re(s)=\frac12$ 的点读出为 $\NULL$”被编译为体系内 soundness 结论；本附录在协议层将其落实为类型化地址域与证书生成约束（见下文“$\NULL$ 判定规则与可重复性定理（协议层）”与“与 PW 类型安全的接口对齐（语义层）”两段）。

\paragraph{数值表示与舍入模型（可审计、可复现）}
为使数值审计具有平台无关的可重复性，采用“外向舍入的区间算术”作为证书后端。固定精度参数 $p\in\NN$，定义可表示实数集合（dyadic 网格）
$$
\mathbb D_p=\left\{k\cdot 2^{-p}:k\in\ZZ\right\}.
$$
定义外向舍入算子（下舍入与上舍入）
$$
\mathrm{rd}_p^{\downarrow}(x)=\max\{d\in\mathbb D_p:d\le x\},\qquad
\mathrm{rd}_p^{\uparrow}(x)=\min\{d\in\mathbb D_p:x\le d\}.
$$
区间表示为 $I=[a,b]$（$a,b\in\mathbb D_p, a\le b$），并规定对基本运算 $\circ\in\{+,-,\times,/\}$ 的区间扩展采用外向舍入（以保证包含性），例如
$$
[a,b]\boxplus[c,d]=\bigl[\mathrm{rd}_p^{\downarrow}(a+c),\ \mathrm{rd}_p^{\uparrow}(b+d)\bigr],
$$
其余运算类似地取四端点组合并外向舍入，保证
$$
x\in[a,b],\ y\in[c,d]\ \Longrightarrow\ x\circ y\in [a,b]\ \boxdot\ [c,d],
$$
其中 $\boxdot$ 表示对应区间算子。

对需要的解析函数（如 $\exp,\log,\sin,\cos$），假定存在可验证实现，给出区间输出 $\mathrm{F}_p^{\mathrm{int}}(I)$ 满足
$$
x\in I\ \Longrightarrow\ f(x)\in \mathrm{F}_p^{\mathrm{int}}(I).
$$
该假设可由“正确舍入的高精度库 + 余项界”实现；其在本文中仅作为证书后端协议的接口合同，不参与前端数学推理。

\paragraph{可重复性约束}
协议要求所有输入参数（如 $L$、$\theta$、网格索引、迭代深度上界）以规范化的有理/ dyadic 形式给出；所有区间端点以 $\mathbb D_p$ 的标准表示输出；所有区间运算的舍入方向固定。由此保证同一输入在同一协议参数 $(p,\text{算法版本})$ 下输出完全一致（字节级一致），从而 $\NULL$ 判定可复现。

\paragraph{审计接口对象与证书类型}
考虑由二变量对象 $\Delta(z,u)$ 诱导的一变量截面
$$
\Xi_{\Delta,L}(s)=\Delta(L^{-s},L^{2s-1}),
$$
并在自对偶与完成化条件下，临界线 $s=\frac12+it$ 上具有 Hardy 型实值化性质。

在可审计矩阵族模型下，若
$$
\Delta(z,u)=\det(I-zB(u)),
$$
则在 unitary slice 上可重写为谱穿越判据：令 $\theta=2t\log L$，则
$$
\Xi_{\Delta,L}\!\left(\frac12+it\right)=\det\!\bigl(I-U_L(\theta)\bigr),
\qquad
U_L(\theta)=L^{-1/2}e^{-i\theta/2}B(e^{i\theta}),
$$
并且
$$
\Xi_{\Delta,L}\!\left(\frac12+it\right)=0\quad\Longleftrightarrow\quad 1\in\mathrm{Spec}\bigl(U_L(\theta)\bigr).
$$

据此定义两类证书。
\begin{enumerate}
  \item \textbf{单位切片证书（Unitary-Slice Certificate）}\\
  以类型化输入强制在切片上工作：把地址空间定义为
  $$
  \Addr_{\mathrm{US}}=\bigl\{(L,\theta,p):L\in\QQ_{>1},\ \theta\in\QQ,\ p\in\NN\bigr\},
  $$
  并约定审计点由
  $$
  s(L,\theta)=\frac12+i\frac{\theta}{2\log L}
  $$
  给出。该表示在语义层面把 $\Re(s)=\frac12$ 变为构造性事实，从而避免“数值判等（$\abs{u}=1$）”的不稳定性；也与“可见参数轴仅为 $\theta$”的闭包约束一致。

  \item \textbf{零点证书（Zero-Certificate）}\\
  给定 $(L,\theta,p)$，用区间算术计算实数区间
  $$
  J(\theta)=\mathrm{Eval}^{\mathrm{int}}_{p}\!\left(\det(I-U_L(\theta))\right)\subset\RR
  $$
  或等价地
  $$
  J(\theta)=\mathrm{Eval}^{\mathrm{int}}_{p}\!\left(\Xi_{\Delta,L}\!\left(\frac12+i\frac{\theta}{2\log L}\right)\right).
  $$
  若 $0\notin J(\theta)$，则输出 \textbf{NonZero} 证书（并给出符号）；可记为
  $
  \mathsf{NonZero}(L,\theta,p;J(\theta))
  $。\\
  若存在 $\theta_-<\theta_+$ 使得区间 $J(\theta_-)$、$J(\theta_+)$ 均不含 $0$ 且符号相反，则由连续性可构造 \textbf{RootInterval} 证书，并可用区间二分法输出包含零点的相位区间 $[\theta_-,\theta_+]$；可记为
  $
  \mathsf{RootInterval}(L,[\theta_-,\theta_+],p;J(\theta_-),J(\theta_+))
  $。该证书对外仅需披露：端点 dyadic 值、端点评估区间及其符号、迭代深度与精度 $p$。上述两类证书与正文第 \ref{subsec:cc-us-cert} 节的 NonZero/RootInterval 口径一致。
\end{enumerate}

\paragraph{\texorpdfstring{$\NULL$}{NULL} 判定规则与可重复性定理（协议层）}
定义审计接口输出为部分函数
$$
\mathsf{Audit}_{\mathrm{US}}:\Addr\rightharpoonup \mathsf{Cert}\ \cup\ \{\NULL\}.
$$

\paragraph{规则（强类型版，推荐）}
\begin{itemize}
  \item 若输入不属于 $\Addr_{\mathrm{US}}$（即无法写成 $(L,\theta,p)$ 的 unitary slice 地址形态），则定义输出为 $\NULL$。
  \item 若输入属于 $\Addr_{\mathrm{US}}$ 但在精度 $p$ 下无法生成所需的“端点不含零且符号确定”的区间证书（例如区间始终包含 $0$ 且达到预设最大精度/迭代阈值），则输出 $\NULL$。
  \item 若证书生成成功，则输出对应证书对象。
\end{itemize}
该 $\NULL$ 规则与“闭包内只允许 $\theta$-口径零点审计”一致，并与类型安全陈述的“偏移点不可寻址 $\Leftrightarrow$ 读出为 $\NULL$”相协调。

\paragraph{\texorpdfstring{$\Read$}{Read}：证书存在性读出（与正文一致）}
在协议层，可将 $\mathsf{Audit}_{\mathrm{US}}$ 的输出投影为一个三值读出（仅表示“断言已被证书化/证书缺失”，而不与数值混同）：
$$
\Read_{\mathrm{US}}(L,\theta,p)=
\begin{cases}
1, & \mathsf{Audit}_{\mathrm{US}}(L,\theta,p)\ \text{输出 NonZero 证书},\\
0, & \mathsf{Audit}_{\mathrm{US}}(L,\theta,p)\ \text{输出 RootInterval 证书},\\
\NULL, & \mathsf{Audit}_{\mathrm{US}}(L,\theta,p)=\NULL.
\end{cases}
$$
该定义与正文定义 \ref{def:cc-read-null} 的“$\NULL$ 作为证书缺失”的语义一致。

\begin{theorem}[协议层可重复性]\label{thm:audit-us-reproducibility}
在固定算法版本、固定精度参数 $p$ 与固定外向舍入规则下，$\mathsf{Audit}_{\mathrm{US}}$ 对任一规范化输入地址给出确定输出；因此由其诱导的 $\Read_{\mathrm{US}}$ 亦为确定性的三值读出。特别地，输出为 $\NULL$ 的判定在同一输入与同一参数下可重复（运行间无差异）。
\end{theorem}

\begin{proof}
地址规范化使输入序列唯一；区间算术在固定舍入方向下为确定性计算；证书生成过程（如二分路径）在固定停止条件下为确定性迭代。因而输出为完全函数式决定。
\end{proof}

\paragraph{与 PW 类型安全的接口对齐（语义层）}
在“三生成元正规化器 + 异常签名接口 + unitary slice 零点审计”的闭包内，类型安全结论给出：若 $\Re(s_0)\neq \frac12$，则 $s_0$ 在闭包下不可寻址，等价地读出为 $\NULL$。
本附录的协议把该“不可寻址性”落实为类型化地址域与证书生成的可计算约束：任何偏离 unitary slice 的输入不进入可审计地址域，因而统一返回 $\NULL$。这使 $\NULL$ 判定既具有逻辑含义（无地址）亦具有工程可复现性（固定协议即固定输出）。

\subsection{小结：与正文结构的对接点}
\begin{enumerate}
  \item \textbf{事件编码层：}通过 Cantor 配对与有限枚举，给出满足“可审计注入编码”接口合同的显式注入 $\mathrm{code}:\mathcal E_m\hookrightarrow\NN_{\ge 1}$，并给出可判定互逆的 $\mathrm{decode}$（命题 \ref{prop:code-decode-invertible}）。此外给出自描述序列编码 $\mathrm{Seq}/\mathrm{Unseq}$（命题 \ref{prop:seq-unseq-invertible}），用于把有限轨迹/证书材料压缩为单一自然数并可判定反解。

\end{enumerate}
