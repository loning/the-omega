\subsection{Alignment with the Witt--Chebyshev--Adams Bridge Package: Power Semigroup \texorpdfstring{$w\mapsto w^m$}{w->w(m)} and Logarithmic Principal Scale}\label{subsec:unit-disk-witt-chebyshev}
This paper has repeatedly used the ``doubling-angle/power map \(\leftrightarrow\) Chebyshev--Adams'' bridge on the unitary slice (see \S\ref{subsec:xi-from-self-dual} and Remark~\ref{rem:xi-joukowski-toy-rh-alignment} for the \(y\mapsto y+y^{-1}\) compression).
Under the disk parameter \(w\) of this appendix, this bridge can be written in a completely parameter-free form.

\paragraph{Boundary doubling angle and Chebyshev polynomials}
When \(w=e^{\mathrm{i}\theta}\in\TT\), let
\[
\mathfrak{r}:=e^{\mathrm{i}\theta/2},\qquad S:=\mathfrak{r}+\mathfrak{r}^{-1}=2\cos(\theta/2)\in[-2,2].
\]
Then for any integer \(m\ge 1\),
\[
w^m=e^{\mathrm{i}m\theta},\qquad
\mathfrak{r}^m+\mathfrak{r}^{-m}=2\cos(m\theta/2)=:C_m(S),
\]
where \(C_m\in\ZZ[S]\) is the Chebyshev--Adams polynomial representation (cf.\ \cite{MasonHandscomb2003Chebyshev}).
Therefore the power semigroup \(w\mapsto w^m\) in the \(S\)-coordinate corresponds precisely to the algebraic doubling-angle operation \(S\mapsto C_m(S)\).

\subsubsection{Fully Rational Commutative Diagram of Phase Coordinates and Completion Trace Coordinates}\label{subsubsec:phase-completion-rational-exchange}
This paragraph provides a ``fully rational'' fact that is especially crucial for unitary-slice interfaces: if the half-angle variable \(\mathfrak{r}\) is parametrized by the Cayley compactification coordinate \(x\in\overline{\RR}\), then both the completion trace coordinate \(S=\mathfrak{r}+\mathfrak{r}^{-1}\) and the Chebyshev--Adams dynamics \(S\mapsto C_d(S)\) can be realized as purely rational/integer-coefficient operations.

\paragraph{Cayley rational parametrization of the completion trace coordinate}
Take \(\mathfrak{r}\in\TT\) and parametrize it via the Cayley form \(\mathfrak{r}=\iota(x)=\frac{1+\mathrm{i}x}{1-\mathrm{i}x}\) (equation~\eqref{eq:unit-circle-cayley-embed}).
Then the completion trace coordinate has the fully rational closed form in the \(x\)-coordinate:
\begin{equation}\label{eq:phase-completion-trace-rational}
\boxed{\;
S(x):=\iota(x)+\iota(x)^{-1}
=\frac{2(1-x^2)}{1+x^2}\in[-2,2]\; }.
\end{equation}
Therefore, any unitary-slice audit interface that depends only on \(S\in[-2,2]\) as a continuous parameter can be equivalently rewritten with \(x\in\overline{\RR}\) as the continuous parameter, and the rewriting introduces neither radicals nor trigonometric functions.

\paragraph{Rational lift of Chebyshev--Adams}
The action of the power semigroup on the half-angle variable is \(\mathfrak{r}\mapsto \mathfrak{r}^{d}\) (\(d\ge 1\)), which compresses in the trace coordinate to \(S\mapsto C_d(S)\), where \(C_d\in\ZZ[S]\) and
\[
C_d(S)=\mathfrak{r}^{d}+\mathfrak{r}^{-d}\qquad\text{when}\qquad S=\mathfrak{r}+\mathfrak{r}^{-1}.
\]
More strongly, there exists a rational function \(\tau_d\in\QQ(x)\) in the Cayley coordinate \(x\) such that \(\mathfrak{r}^d=\iota(\tau_d(x))\), so that the following commutative diagram holds strictly:
\[
\begin{CD}
x @>\iota>> \mathfrak{r} @>>> S=\mathfrak{r}+\mathfrak{r}^{-1}\\
@V\tau_d VV @VV(\cdot)^d V @VV C_d V\\
\tau_d(x) @>\iota>> \mathfrak{r}^d @>>> C_d(S)
\end{CD}
\]
where \(C_d(S(x))=S(\tau_d(x))\).

\begin{proposition}[Fully rational realization of the Chebyshev--Adams action]\label{prop:chebyshev-adams-fully-rational}
For any integer \(d\ge 1\), there exist \(A_d,B_d\in\ZZ[x]\) such that
\begin{equation}\label{eq:adams-Ab-polynomials}
\boxed{\;
(1+\mathrm{i}x)^d=A_d(x)+\mathrm{i}B_d(x),\qquad
(1-\mathrm{i}x)^d=A_d(x)-\mathrm{i}B_d(x)\; }.
\end{equation}
Define
\begin{equation}\label{eq:adams-tau-d}
\boxed{\;\tau_d(x):=\frac{B_d(x)}{A_d(x)}\in\QQ(x)\; }.
\end{equation}
Then
\begin{equation}\label{eq:adams-cayley-lift}
\boxed{\;
\iota(x)^d=\iota\!\bigl(\tau_d(x)\bigr)\; }.
\end{equation}
Consequently, the trace coordinate satisfies the strict consistency relation
\begin{equation}\label{eq:adams-trace-commute}
\boxed{\;
C_d\!\bigl(S(x)\bigr)=S\!\bigl(\tau_d(x)\bigr)
=\frac{2\bigl(1-\tau_d(x)^2\bigr)}{1+\tau_d(x)^2}\; }.
\end{equation}
Moreover, \((A_d,B_d)\) can be computed by a purely multiplicative-additive recurrence in \(\ZZ[x]\):
\[
A_0=1,\ B_0=0,\qquad
A_{d+1}=A_d-xB_d,\quad
B_{d+1}=B_d+xA_d.
\]
Therefore, any unitary-slice audit or certificate construction that depends on \(S\mapsto C_d(S)\) can be realized using only multiply-add operations on integer-coefficient polynomials, without invoking trigonometric functions or algebraic radicals.
\end{proposition}
\begin{proof}
By the binomial expansion, the real and imaginary parts of \((1+\mathrm{i}x)^d\) are integer-coefficient polynomials, so \(A_d,B_d\in\ZZ[x]\) satisfying \eqref{eq:adams-Ab-polynomials} exist.
From \(\iota(x)=\frac{1+\mathrm{i}x}{1-\mathrm{i}x}\) one obtains
\[
\iota(x)^d=\frac{(1+\mathrm{i}x)^d}{(1-\mathrm{i}x)^d}
=\frac{A_d(x)+\mathrm{i}B_d(x)}{A_d(x)-\mathrm{i}B_d(x)}
=\frac{1+\mathrm{i}\tau_d(x)}{1-\mathrm{i}\tau_d(x)}
=\iota\!\bigl(\tau_d(x)\bigr),
\]
yielding \eqref{eq:adams-cayley-lift}.
Substituting \(S(t)=\iota(t)+\iota(t)^{-1}=\frac{2(1-t^2)}{1+t^2}\) at \(\tau_d(x)\) gives \eqref{eq:adams-trace-commute}.
The recurrence follows from expanding
\(
(1+\mathrm{i}x)^{d+1}=(1+\mathrm{i}x)(A_d+\mathrm{i}B_d)
\)
and comparing real and imaginary parts. \qed
\end{proof}

\paragraph{Half-angle lift as the minimal invisible fiber: branch information from \(w\) to \(\mathfrak{r}\)}
The map
\[
q:\TT\to\TT,\qquad q(\mathfrak{r})=\mathfrak{r}^{2}=w
\]
is a double covering with kernel \(\{\pm 1\}\cong \ZZ/2\ZZ\).
Therefore the minimal discrete degree of freedom invisible at the \(w\)-level (i.e., the unitary twist variable \(u\) level in the main text) is precisely the deck automorphism (half-angle branch) \(\mathfrak{r}\mapsto-\mathfrak{r}\) of this covering.
In the completion coordinate \(S=\mathfrak{r}+\mathfrak{r}^{-1}\), this branch manifests as
\(
S\mapsto -S
\), and is therefore explicitly activated in the even/odd jet splitting of \(S\).

\begin{proposition}[Half-angle residue at the endpoint \texorpdfstring{$w=-1$}{w=-1} and \texorpdfstring{$\ZZ/4$}{Z/4}-type values]\label{prop:half-angle-z4-residue}
At the endpoint \(w=-1\) (equivalently \(\theta=\pi\)), the completion coordinate satisfies \(S=0\).
Moreover, for any \(m\ge 1\), the value of the Chebyshev--Adams polynomial at the endpoint is
\begin{equation}\label{eq:chebyshev-endpoint-values}
\boxed{\;
C_m(0)=\mathfrak{r}^{m}+\mathfrak{r}^{-m}=2\cos(m\pi/2)\in\{0,\pm 2\}\; }.
\end{equation}
In particular, \(C_m(0)\) depends only on \(m\bmod 4\), so the action of the power semigroup at the endpoint carries a \(\ZZ/4\)-type discrete residue.
\end{proposition}
\begin{proof}
The half-angle of the endpoint \(w=-1=e^{i\pi}\) satisfies \(\mathfrak{r}^{2}=-1\), so \(\mathfrak{r}=\pm i=e^{\pm i\pi/2}\), and therefore \(S=\mathfrak{r}+\mathfrak{r}^{-1}=0\).
Since \(C_m(S)=\mathfrak{r}^m+\mathfrak{r}^{-m}=2\cos(m\theta/2)\), at \(\theta=\pi\) one obtains
\(
C_m(0)=2\cos(m\pi/2)
\), whose value depends only on \(m\bmod 4\). \qed
\end{proof}

\begin{proposition}[Endpoint orthogonal decomposition: \texorpdfstring{$\ZZ/4$}{Z/4} discrete payload and \texorpdfstring{$\RR_{+}$}{R+} precision payload]\label{prop:endpoint-orthogonal-decomposition-z4-Rplus}
In a neighborhood of the endpoint \(w=-1\) (completion coordinate \(S=0\)), the action of the scaling semigroup \(m\in\NN_{\ge 1}\) decomposes into two mutually decoupled payloads:
\begin{enumerate}
  \item \textbf{Discrete endpoint payload:} The endpoint state \(C_m(0)\in\{0,\pm 2\}\) depends only on \(m\bmod 4\) (Proposition~\ref{prop:half-angle-z4-residue}).
  \item \textbf{Continuous precision payload:} Take the Cayley phase axis \(u_{\mathrm C}(t)=\frac{1+\mathrm{i}t}{1-\mathrm{i}t}\) and let \(\mathfrak{w}(u)=\frac{1+u}{1-u}\) be the half-plane linearization coordinate.
  Then for \(t\neq 0\) one has \(\mathfrak{w}(u_{\mathrm C}(t))=\mathrm{i}/t\), and under the scaling \(t\mapsto mt\) the following holds:
  \[
  \mathfrak{w}\!\bigl(u_{\mathrm C}(mt)\bigr)=\frac{1}{m}\,\mathfrak{w}\!\bigl(u_{\mathrm C}(t)\bigr),
  \qquad
  y\!\bigl(u_{\mathrm C}(mt)\bigr)=y\!\bigl(u_{\mathrm C}(t)\bigr)+\log m,
  \]
  where \(y(u):=-\log|\mathfrak{w}(u)|\) is the endpoint precision potential \eqref{eq:endpoint-busemann-potential}.
\end{enumerate}
Therefore, the non-compactness at the endpoint accumulates along the precision potential \(y\)-axis as a \(\log m\) translation ledger, while the ``half-angle branch memory'' at the endpoint is entirely carried by the finite discrete residue \(m\bmod 4\).
\end{proposition}
\begin{proof}
The first assertion is Proposition~\ref{prop:half-angle-z4-residue}.
The second follows from \eqref{eq:scale-linearization-halfplane-coordinate}--\eqref{eq:scale-linearization-w-vs-x}: when \(u=u_{\mathrm C}(t)\) one has \(x(u)=t\) and \(\mathfrak{w}(u)=\mathrm{i}/x(u)=\mathrm{i}/t\).
The scaling \(t\mapsto mt\) in the \(u\)-variable corresponds to \(u\mapsto\psi_m(u)\), and by Proposition~\ref{prop:scale-linearization-identity-multipliers} one has \(\mathfrak{w}(\psi_m(u))=\mathfrak{w}(u)/m\), so that \(\mathfrak{w}(u_{\mathrm C}(mt))=\mathfrak{w}(u_{\mathrm C}(t))/m\).
Finally, by the definition \(y=-\log|\mathfrak{w}|\) (equation~\eqref{eq:endpoint-busemann-potential}) one obtains the addition law; this can also be viewed directly as the restriction of \eqref{eq:endpoint-busemann-addition-law} to the unitary Cayley axis. \qed
\end{proof}

\begin{proposition}[Endpoint derivative law: odd scaling gives linear amplification, even scaling degenerates to second order]\label{prop:chebyshev-endpoint-derivative-splitting}
Let \(m\ge 1\) and \(C_m(S)=2T_m(S/2)\) (\(T_m\) is the Chebyshev polynomial of the first kind).
Then at \(S=0\) one has
\begin{equation}\label{eq:chebyshev-endpoint-derivative}
\boxed{\;
C_m'(0)=m\sin(m\pi/2)\; }.
\end{equation}
Consequently:
\begin{enumerate}
  \item If \(m\) is even, then \(C_m'(0)=0\) and \(C_m\) is an even polynomial, so the endpoint jet starts at least at second order;
  \item If \(m\) is odd, then \(C_m(0)=0\) and \(|C_m'(0)|=m\), so the endpoint jet is linearly dominant.
\end{enumerate}
\end{proposition}
\begin{proof}
By the identity \(C_m(S)=2T_m(S/2)\) and the derivative formula \(T_m'(x)=mU_{m-1}(x)\) (\(U_{m-1}\) is the Chebyshev polynomial of the second kind), one obtains
\[
C_m'(0)=2\cdot T_m'(0)\cdot \frac12=T_m'(0)=mU_{m-1}(0).
\]
Since \(U_{m-1}(\cos\phi)=\frac{\sin(m\phi)}{\sin\phi}\), taking \(\cos\phi=0\Rightarrow \phi=\pi/2\) gives \(U_{m-1}(0)=\sin(m\pi/2)\), yielding \eqref{eq:chebyshev-endpoint-derivative}.
The even/odd splitting follows from \(\sin(m\pi/2)\in\{0,\pm 1\}\); moreover, from \(C_m(-S)=(-1)^m C_m(S)\) (corresponding to the half-angle branch \(\mathfrak{r}\mapsto-\mathfrak{r}\)) one sees that \(C_m\) is an even polynomial when \(m\) is even and an odd polynomial when \(m\) is odd. \qed
\end{proof}

\begin{corollary}[Two types of regularization for the endpoint jet: odd channels control first order, even channels enter higher order]\label{cor:chebyshev-endpoint-jet-regularization}
Let \(g\) be analytic on some open neighborhood containing the interval \([-2,2]\).
Then as \(S\to 0\) one has the expansion
\[
g(C_m(S))=
\begin{cases}
g(C_m(0)) + O(S^2), & m\ \text{even},\\[4pt]
g(0) + g'(0)\,C_m'(0)\,S+O(S^2), & m\ \text{odd}.
\end{cases}
\]
Therefore, in the endpoint neighborhood, odd \(m\) channels retain a linear principal scale, while even \(m\) channels produce no first-order term and can only affect higher-order jets and finite-part constants.
\end{corollary}
\begin{proof}
When \(m\) is even, \(C_m'(0)=0\) and \(C_m(0)\in\{\pm 2\}\), so \(C_m(S)=C_m(0)+O(S^2)\); Taylor-expanding \(g\) at \(C_m(0)\) gives \(g(C_m(S))=g(C_m(0))+O(S^2)\).
When \(m\) is odd, \(C_m(0)=0\) and \(C_m'(0)\neq 0\), so \(C_m(S)=C_m'(0)S+O(S^3)\); Taylor-expanding \(g\) at \(0\) yields the second line. \qed
\end{proof}

\paragraph{Even/odd sectors of primitives at the endpoint: principal singularities are monopolized by even lengths}
Let \(p_n(u)\in\ZZ[u]\) be the primitive polynomial obtained from Witt/M\"obius inversion (e.g., the \(\mathrm{Mob}\) output of \S\ref{subsec:operator-zeta-interface}; see also Appendix~\ref{app:kernel-completion} and the subsequent primitive interface).
In many kernel families, the primitive series satisfies a finite-length mirror symmetry \(p_n(u)=u^n p_n(1/u)\) (see Corollary~\ref{cor:primitive-finite-sym}), and for odd lengths a stronger divisibility law holds:

\begin{corollary}[Endpoint vanishing of odd-length primitives]\label{cor:primitive-odd-vanish-u-minus1}
If \(n\ge 3\) is odd, then \(u(u+1)\mid p_n(u)\) (Corollary~\ref{cor:primitive-odd-div}).
In particular, \(p_n(-1)=0\).
Therefore, decomposing the primitive generating function \(P(z,u)=\sum_{n\ge 1}p_n(u)z^n\) as
\[
P_{\mathrm{odd}}(z,u):=\sum_{n\ \mathrm{odd}}p_n(u)z^n,\qquad
P_{\mathrm{even}}(z,u):=\sum_{n\ \mathrm{even}}p_n(u)z^n,
\]
one sees that, apart from finitely many low-length terms (at most \(n=1\)), \(P_{\mathrm{odd}}(z,u)\) vanishes to at least first order at \(u=-1\);
consequently, any non-removable principal singularity at the endpoint \(u=-1\) can only be controlled by the even-length sector \(P_{\mathrm{even}}(z,u)\).
\end{corollary}

\begin{proposition}[Completion subring and the algebraic characterization of the ``unique angle'']\label{prop:completion-subring-unique-angle}
Let \(\mathfrak{r}\) be the half-angle variable and write \(S=\mathfrak{r}+\mathfrak{r}^{-1}\).
Then the invariant subring under the inversion action \(\mathfrak{r}\mapsto \mathfrak{r}^{-1}\) satisfies
\[
\boxed{\;\ZZ[\mathfrak{r},\mathfrak{r}^{-1}]^{\mathfrak{r}\mapsto \mathfrak{r}^{-1}}
=\ZZ[\mathfrak{r}+\mathfrak{r}^{-1}]
=\ZZ[S]\; }.
\]
Therefore, ``descending from the half-angle level to a unique angle'' is algebraically equivalent to taking the inversion-invariant subring; any degree of freedom beyond \(\ZZ[S]\) is equivalent to the invisible information component carried by the half-angle branch (and endpoint jets).
\end{proposition}
\begin{proof}
For an invariant element \(f(\mathfrak{r})=f(\mathfrak{r}^{-1})\), pairing the Laurent coefficients at \(k\) and \(-k\) gives
\(
f(\mathfrak{r})=\sum_{k\ge 0}a_k(\mathfrak{r}^k+\mathfrak{r}^{-k})
\).
Since \(\mathfrak{r}^k+\mathfrak{r}^{-k}=C_k(\mathfrak{r}+\mathfrak{r}^{-1})=C_k(S)\in\ZZ[S]\), the invariant subring is contained in \(\ZZ[S]\); the reverse inclusion follows directly from the definition. \qed
\end{proof}

\begin{proposition}[Endpoint decomposition paradigm: odd/even channels of the Witt scaling chain and odd/even sectors of primitives]\label{prop:endpoint-decomposition-paradigm}
In the completion coordinates \((w,\mathfrak{r},S)\) (where \(w=\mathfrak{r}^2,\ S=\mathfrak{r}+\mathfrak{r}^{-1}\)), the completion action corresponding to the power semigroup \(w\mapsto w^m\) is \(S\mapsto C_m(S)\).
At the endpoint \(w=-1\) (\(S=0\)), the local structure of the ``infinite scaling chain'' satisfies:
\begin{enumerate}
  \item \textbf{(Scaling channel splitting)} The endpoint jet splits strictly according to the parity of \(m\): odd \(m\) channels control the linear principal scale, while even \(m\) channels enter only higher-order jets and finite parts (Proposition~\ref{prop:chebyshev-endpoint-derivative-splitting} and Corollary~\ref{cor:chebyshev-endpoint-jet-regularization}).
  \item \textbf{(Primitive sector splitting)} In the primitive polynomial sequence, the odd-length sector \(n\ge 3\) vanishes strictly at \(u=-1\) (Corollary~\ref{cor:primitive-odd-vanish-u-minus1}), so endpoint principal singularities are entirely controlled by the even-length sector.
  \item \textbf{(Completion subring and invisible components)} All visible algebraic structure descends strictly to the inversion-invariant subring \(\ZZ[S]\) (Proposition~\ref{prop:completion-subring-unique-angle}); the information annihilated by the projection appears in the form of the half-angle branch (\(\ZZ/2\)), endpoint discrete residues (\(\ZZ/4\), Proposition~\ref{prop:half-angle-z4-residue}), and higher-order jet precision.
\end{enumerate}
\end{proposition}

\paragraph{Witt power-sum operator in the disk and logarithmic principal scale}
For any function \(f\) analytic in \(\mathbb{D}\), define
\begin{equation}\label{eq:app-witt-pow-operator}
\boxed{\;
(W_{\mathrm{pow}}f)(w):=\sum_{m\ge 1}\frac{1}{m}f(w^m),\qquad |w|<1\; }.
\end{equation}
Since \(|w|<1\Rightarrow |w^m|\to 0\), the series converges absolutely for \(|w|<1\) and defines an analytic function.
As the minimal test case, taking \(f(w)=w\) gives
\[
(W_{\mathrm{pow}}f)(w)=\sum_{m\ge 1}\frac{w^m}{m}=-\log(1-w),
\]
so the principal singularity of the ``power-chain superposition'' appears at the boundary logarithmic scale.
This logarithmic principal scale is structural: under the constraint of preserving both the \(1/m\) weighting and the power-substitution interface of \(W_{\mathrm{pow}}\), its boundary logarithmic behavior can only be stripped off as an auditable symbolic term in a regularization/renormalization caliber (e.g., the divergence coefficient controlled by the pole of \(\zeta(1+\sigma)\) under Dirichlet regularization; see Proposition~\ref{prop:endpoint-dirichlet-symbol-weights}).
This phenomenon is structurally isomorphic to the Witt/M\"obius inversion bridge package of this paper (Proposition~\ref{prop:zetaK-mobius-primitive}): at the formal level, the expansion of \(\log Z\) naturally produces the power-chain operator \(\sum_{m\ge 1}\frac{1}{m}(\cdot)\circ[m]\); under the disk-analytic caliber, this power-chain operator has a direct convergence domain and boundary singularity.
