\subsection{Quantum-mechanics theorem library (core)}
\label{app:qm_theorem_library_core}

This appendix records a compact set of standard quantum-mechanics structure theorems in a form compatible with the tick\,+\,CAP audit discipline.
We emphasize the separation between (i) mathematical statements about Hilbert spaces, operator algebras, and channels, and (ii) interface identifications that interpret these objects as physical symmetries, dynamics, and readout semantics.

\subsubsection{Wigner's theorem (symmetry as probability preservation)}
\paragraph{Setup.}
Let $\mathcal{H}$ be a complex Hilbert space, and let $\mathbb{P}(\mathcal{H})$ denote the projective space of rays $[\psi]$ for $\psi\ne 0$.
Define the transition probability between rays by
\[
P([\psi],[\phi]) := \frac{|\langle \psi|\phi\rangle|^2}{\|\psi\|^2\,\|\phi\|^2}.
\]

\begin{theorem}[Wigner (structural form)]
\label{thm:z128_wigner}
Let $T:\mathbb{P}(\mathcal{H})\to \mathbb{P}(\mathcal{H})$ be a bijection that preserves transition probabilities:
$P(T([\psi]),T([\phi]))=P([\psi],[\phi])$ for all nonzero $\psi,\phi\in\mathcal{H}$.
Then there exists either a unitary or an antiunitary operator $U$ on $\mathcal{H}$ such that
$T([\psi])=[U\psi]$ for all $\psi\ne 0$.
\end{theorem}
\begin{proof}[Reference]
Standard; see, e.g., \cite{Wigner1931}.
\end{proof}

\paragraph{Interface reading.}
\InterfaceTag If one declares that a ``symmetry'' is any transformation of pure states that preserves all transition probabilities, then such symmetries are forced to be unitary/antiunitary.
This is a rigidity statement about probability preservation, not a dynamical axiom.

\subsubsection{Stone's theorem (continuous time from one-parameter unitaries)}
\begin{theorem}[Stone (structural form)]
\label{thm:z128_stone}
Let $U:\RR\to \mathcal{U}(\mathcal{H})$ be a strongly continuous one-parameter unitary group: $U(t+s)=U(t)U(s)$ and $U(0)=I$.
Then there exists a unique self-adjoint operator $H$ such that
\[
U(t)=\e^{-\iu tH}
\qquad\text{for all }t\in\RR.
\]
\end{theorem}
\begin{proof}[Reference]
Standard; see, e.g., \cite{Stone1932}.
\end{proof}

\paragraph{Interface reading (two layers).}
\InterfaceTag In this paper, the tick primitive supplies a one-way sequencing variable and supports semigroup viewpoints (Appendix~\ref{app:arrow_of_time_semigroup_notes}).
Stone's theorem concerns the \emph{reversible} (group) layer: if one commits to an underlying reversible time-translation symmetry on a Hilbert space representation, it is generated by a self-adjoint operator.
The irreversible readout/coarse-graining layer is modeled by channels (Appendix~\ref{app:quantum_channels_cptp_stinespring}).

\subsubsection{Robertson--Schr\"odinger uncertainty (operator-level bound)}
\begin{proposition}[Robertson--Schr\"odinger uncertainty (finite-dimensional statement)]
\label{prop:z128_robertson_schrodinger}
Let $\rho$ be a density operator on a finite-dimensional Hilbert space and let $A,B$ be Hermitian operators.
Define centered operators $\Delta A:=A-\Tr(\rho A)I$ and $\Delta B:=B-\Tr(\rho B)I$ and variances
$\Var_\rho(A):=\Tr(\rho(\Delta A)^2)$, $\Var_\rho(B):=\Tr(\rho(\Delta B)^2)$.
Then
\[
\Var_\rho(A)\,\Var_\rho(B)\;\ge\;
\frac{1}{4}\,|\Tr(\rho[A,B])|^2
\;+\;
\frac{1}{4}\,|\Tr(\rho\{\Delta A,\Delta B\})|^2.
\]
\end{proposition}
\begin{proof}[Reference]
Standard; see, e.g., \cite{Robertson1929,Schrodinger1930}.
\end{proof}

\paragraph{Interface reading.}
\InterfaceTag The inequality provides a protocol-independent lower bound on joint sharpness when observables do not commute.
In the tick\,+\,CAP discipline, this bound is used as a consistency constraint for proposed readout dictionaries when mapping finite-resolution readout to continuum observables.

\subsubsection{Schmidt decomposition (minimal entanglement normal form)}
\begin{theorem}[Schmidt decomposition (finite-dimensional)]
\label{thm:z128_schmidt}
Let $\mathcal{H}_A,\mathcal{H}_B$ be finite-dimensional and let $|\psi\rangle\in\mathcal{H}_A\otimes\mathcal{H}_B$.
Then there exist orthonormal families $\{|i\rangle_A\}$ and $\{|i\rangle_B\}$ and nonnegative coefficients $\{\lambda_i\}$ with $\sum_i \lambda_i^2=1$ such that
\[
|\psi\rangle=\sum_i \lambda_i\,|i\rangle_A\otimes |i\rangle_B.
\]
The nonzero $\lambda_i$ are uniquely determined by $|\psi\rangle$ up to permutation.
\end{theorem}
\begin{proof}[Reference]
Standard linear algebra; see, e.g., \cite{NielsenChuang2000}.
\end{proof}

\paragraph{Interface reading.}
\InterfaceTag Schmidt rank and the spectrum of reduced states (Appendix~\ref{app:composite_systems_tensor_products}) provide the minimal interface invariants for entanglement across a bipartition.
