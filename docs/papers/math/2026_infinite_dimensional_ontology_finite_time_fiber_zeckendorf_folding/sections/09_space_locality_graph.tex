\section{空间：局部依赖图与关系化距离}

\subsection{局部依赖图公理}

\begin{definition}[局部依赖图]\label{def:local_dependency_graph}
空间由局部依赖图 $(V,\mathrm{Nbr})$ 给出。节点 $v\in V$ 表示基本单元（体素或其抽象等价类），邻域算子 $\mathrm{Nbr}(v)\subset V$ 给出允许依赖的有限集合。系统状态更新满足
$$
\mathrm{State}_{t+1}(v)=\mathsf T\bigl(\mathrm{State}_t(v),\mathrm{State}_t(\mathrm{Nbr}(v))\bigr),
$$
从而“空间结构”由依赖关系定义而非先验坐标。
\end{definition}

\subsection{空间子群：依赖图自同构与配置的可逆重标记}

本节把“空间运算”群化为对节点集合的可逆重标记（置换）。这些变换本身不改变局部规则 $\mathsf T$ 的形式，但会重标记“哪里是哪里”，从而提供空间对称与空间运算的最小可逆实现。

\begin{definition}[依赖图自同构群与空间作用]\label{def:space-automorphism-group}
定义局部依赖图的自同构群
$$
\mathrm{Aut}(V,\mathrm{Nbr})=\left\{\sigma\in \mathrm{Sym}(V):\forall v\in V,\ \sigma(\mathrm{Nbr}(v))=\mathrm{Nbr}(\sigma(v))\right\}.
$$
对任意配置 $s:V\to E_m$（或 $s:V\to \widetilde{\Omega}_m$），定义 $\sigma$ 对配置的作用为
$$
(\sigma\cdot s)(v)=s\bigl(\sigma^{-1}(v)\bigr).
$$
该作用给出一个表示同态
$$
\rho:\mathrm{Aut}(V,\mathrm{Nbr})\to \mathrm{Sym}(E_m^V),
\qquad
\rho(\sigma)(s)=\sigma\cdot s,
$$
并令空间子群为其像
$$
S_{m,V}:=\rho\bigl(\mathrm{Aut}(V,\mathrm{Nbr})\bigr)\le \mathrm{Sym}(E_m^V)
$$
（或在 $\widetilde{\Omega}_m^V$ 上的对应表示）。
\end{definition}

\begin{remark}[局部生成元：相邻换位]
在实际计算中可不必使用全体 $\mathrm{Aut}(V,\mathrm{Nbr})$。常见做法是选取一组局部置换生成元（例如图边上的相邻换位），生成一个更小但可计算的空间运算群。该选择与第 15 节把对象运动定义为“支撑集合在邻接结构上的迁移”相匹配。
\end{remark}

\subsection{两种距离：拓扑距离与信息距离}

在上述框架下定义两类距离：拓扑距离
$$
d_{\mathrm{top}}(u,v)=\text{图最短路长度},
$$
信息距离
$$
d_{\mathrm{inf}}(u,v)=\text{约束传播从 }u\text{ 推导 }v\text{ 所需最少步数}.
$$

\subsection{时间—空间复合与正规形（半直积型）}

为把“任意时间与空间运算”变为可计算的代数对象，取一个可逆的全局一步更新算子 $\tau\in\mathrm{Sym}(E_m^V)$ 作为离散时间生成元（参见定义 \ref{def:time-subgroup} 的思想：由单步更新生成循环时间群），并与空间子群 $S_{m,V}$ 生成时空运算群。

\begin{assumption}[空间齐性与正规化规则]\label{ass:space_homogeneity}
\InterfaceTag \textbf{建模约定：} 本假设用于把“时空运算群”的代数化简写成可计算正规形，并服务于后续算法实现；它不作为第 4--7 节组合定理的前提。
假设全局更新算子 $\tau$ 规范化空间子群 $S_{m,V}$：对任意 $\sigma\in S_{m,V}$，
$$
\tau\,\sigma\,\tau^{-1}\in S_{m,V}.
$$
从而共轭给出一个群自同构
$$
\alpha_\tau:S_{m,V}\to S_{m,V},\qquad \alpha_\tau(\sigma)=\tau\,\sigma\,\tau^{-1}.
$$
进一步地，由共轭作用可得到一个同态（作用）
$$
\langle\tau\rangle\to \mathrm{Aut}(S_{m,V}),\qquad \tau^k\mapsto \alpha_\tau^k.
$$
在常见的“同一局部规则在各点平移不变”设定下，上式可取 $\alpha_\tau=\mathrm{id}$（即 $\tau$ 与 $S_{m,V}$ 交换）。
\end{assumption}

\begin{proposition}[时空运算的正规形]\label{prop:ts-normal-form}
在假设 \ref{ass:space_homogeneity} 下，设
$$
G_{m,V}^{\mathrm{ts}}=\langle \tau,\, S_{m,V}\rangle \le \mathrm{Sym}(E_m^V).
$$
则任意 $g\in G_{m,V}^{\mathrm{ts}}$ 都可写为
$$
g=\tau^k s,\qquad k\in\ZZ,\ s\in S_{m,V},
$$
并且该表示在给定 $\alpha_\tau$ 的化简规则下是可算法化的：通过反复使用
$$
\tau\, s=\alpha_\tau(s)\,\tau,\qquad s\in S_{m,V},
$$
可将任意交错词移动为“时间块在左、空间块在右”的形式。若 $\tau$ 为有限阶 $n$，则可进一步将 $k$ 约化到 $0,1,\dots,n-1$；若 $\tau$ 为无限阶，则 $k\in\ZZ$ 的取值对应离散时间步数。
\end{proposition}

\begin{proof}
\MathTag 依赖：假设 \ref{ass:space_homogeneity}；以及群的生成与重写规则的标准归纳化简。
由 $G_{m,V}^{\mathrm{ts}}=\langle \tau, S_{m,V}\rangle$ 的定义，任意 $g\in G_{m,V}^{\mathrm{ts}}$ 可写为关于生成元集合 $\{\tau\}\cup S_{m,V}$ 的有限长度交错乘积（词）：
$$
g = a_1 a_2 \cdots a_L,\qquad a_i\in \{\tau,\tau^{-1}\}\cup S_{m,V}.
$$
在假设 \ref{ass:space_homogeneity} 下，$\alpha_\tau(s)=\tau s\tau^{-1}\in S_{m,V}$，从而对任意 $s\in S_{m,V}$ 有重写规则
$$
\tau s = \alpha_\tau(s)\tau,\qquad
s\tau^{-1}=\tau^{-1}\alpha_\tau(s).
$$
对给定的词反复应用上述规则，把所有 $\tau,\tau^{-1}$ 逐步交换到左侧，则可将 $g$ 化简为
$$
g=\tau^k s,\qquad k\in\ZZ,\ s\in S_{m,V}.
$$
若 $\tau$ 的阶为有限 $n$，则 $\tau^n=\mathrm{id}$，因此可将 $k$ 在 $\ZZ/n\ZZ$ 中约化到 $0,1,\dots,n-1$。证毕。
\end{proof}

\begin{remark}[生成元、关系与可计算化简]\label{rem:generators-relations}
为得到“任意时间与空间运算”的具体运算系统，可固定一组空间生成元 $\Sigma\subset S_{m,V}$（例如若干条边上的相邻换位）并令
$$
S_{m,V}=\langle \Sigma\rangle.
$$
则 $G_{m,V}^{\mathrm{ts}}$ 由生成元集合 $\{\tau\}\cup\Sigma$ 给出；其关键化简规则可写为
$$
\tau\,\sigma=\alpha_\tau(\sigma)\,\tau,\qquad \sigma\in \Sigma,
$$
以及 $\Sigma$ 内部的群关系（例如换位的阶为 $2$、以及图上相邻换位满足的 braid 关系等）。在此表示下，任意交错运算串都可通过局部替换规则化简至命题 \ref{prop:ts-normal-form} 的正规形。
\end{remark}

\subsection{与时间成本的耦合：传播步数作为时间贡献}

信息距离直接给出约束传播最少步数的下界，因此可作为时间成本 $T(\mathcal G)$ 的局部贡献代理量：若目标 $\mathcal G$ 的满足需要从一组源点集合向外传播一致性约束，则任何可行轨道的收敛时间均受相应 $d_{\mathrm{inf}}$ 的下界控制。该耦合在第 \ref{tab:audit-deps} 的审计链中被用作“空间—时间成本”接口，并在第 15 节实验中通过统计相关性进行检验。
