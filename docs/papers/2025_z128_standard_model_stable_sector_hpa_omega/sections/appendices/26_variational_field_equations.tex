\subsection{Field equations from variation (Einstein--Yang--Mills + information sector)}
\label{app:variational_field_equations}

This appendix records the standard variational consequences of the CAP-closed continuum action skeleton in Appendix~\ref{app:cap_continuum_action_closure}.
The role here is not novelty in calculus, but audit alignment: the dynamical equations used in later interface interpretations are explicitly the Euler--Lagrange equations of the declared action class.

\subsubsection{Reference action}
\label{subsec:ref_action_for_variation}

We take as reference representative (Appendix~\ref{app:cap_continuum_action_closure})
\begin{equation}
\label{eq:ref_action_for_variation}
S_{\mathrm{eff}}
=\int \dd^{4}x\,\sqrt{-g}\,
\left[
\frac{R-2\Lambda}{16\pi G}
-\lambda_F\, g^{\mu\nu}(\nabla_\mu\chi)(\nabla_\nu\chi)
-V(\chi^{2})
\;-\;\sum_{a}\frac{1}{4g_a^2}\Tr(F^{(a)}_{\mu\nu}F^{(a)\mu\nu})
\;+\;\mathcal{L}_{m}
\right].
\end{equation}

\subsubsection{Metric variation: Einstein equation with gauge and information stress}
\label{subsec:einstein_equation_variation}

\begin{theorem}[Einstein equation with total stress tensor]
\label{thm:einstein_equation_total_stress}
Varying \eqref{eq:ref_action_for_variation} with respect to the metric yields
\begin{equation}
\label{eq:einstein_total_stress}
G_{\mu\nu}+\Lambda g_{\mu\nu}
=8\pi G\left(T^{(m)}_{\mu\nu}+T^{(\chi)}_{\mu\nu}+T^{(\mathrm{YM})}_{\mu\nu}\right),
\end{equation}
where $T^{(m)}_{\mu\nu}$ is the matter stress tensor, $T^{(\chi)}_{\mu\nu}$ is the information/amplitude stress tensor,
and $T^{(\mathrm{YM})}_{\mu\nu}$ is the gauge-field stress tensor.
\end{theorem}

\begin{proof}[Proof sketch]
This is the standard metric variation of a diffeomorphism-invariant local action.
The Einstein--Hilbert term yields $G_{\mu\nu}$; the constant term yields $\Lambda g_{\mu\nu}$.
The remaining contributions are, by definition, the stress tensors obtained from varying the corresponding Lagrangian densities with respect to $g^{\mu\nu}$.
\end{proof}

\begin{proposition}[Information/amplitude stress tensor]
\label{prop:info_stress_tensor}
The contribution of the $\chi$-sector in \eqref{eq:ref_action_for_variation} is
\begin{equation}
\label{eq:info_stress_tensor}
T^{(\chi)}_{\mu\nu}
=2\lambda_F\left(\nabla_\mu\chi\,\nabla_\nu\chi-\frac12 g_{\mu\nu}(\nabla\chi)^2\right)
-g_{\mu\nu}V(\chi^2),
\qquad
(\nabla\chi)^2:=g^{\alpha\beta}\nabla_\alpha\chi\,\nabla_\beta\chi.
\end{equation}
\end{proposition}

\begin{proposition}[Yang--Mills stress tensor (per factor)]
\label{prop:ym_stress_tensor}
For each gauge factor $a$, the curvature term contributes
\begin{equation}
\label{eq:ym_stress_tensor}
T^{(\mathrm{YM},a)}_{\mu\nu}
=\frac{1}{g_a^2}\Tr\!\left(F^{(a)}_{\mu\alpha}F^{(a)}_{\nu}{}^{\alpha}-\frac14 g_{\mu\nu}F^{(a)}_{\alpha\beta}F^{(a)\alpha\beta}\right),
\end{equation}
and $T^{(\mathrm{YM})}_{\mu\nu}=\sum_a T^{(\mathrm{YM},a)}_{\mu\nu}$.
\end{proposition}

\subsubsection{Gauge variation: Yang--Mills equations}
\label{subsec:ym_equations_variation}

\begin{proposition}[Yang--Mills equation (schematic)]
\label{prop:ym_equation}
Varying \eqref{eq:ref_action_for_variation} with respect to the gauge connection of factor $a$ yields
\begin{equation}
\label{eq:ym_equation}
\nabla_\mu\left(\frac{1}{g_a^2}F^{(a)\mu\nu}\right)=J^{(a)\nu},
\end{equation}
where $J^{(a)\nu}$ is the matter current induced by $\mathcal{L}_m$ (and any explicit $\chi$ couplings, if present).
\end{proposition}

\paragraph{Remark.}
\InterfaceTag Equation \eqref{eq:ym_equation} is the continuum dictionary corresponding to the finite connection/holonomy skeleton (Section~\ref{sec:protocol_connections_holonomy}), where ``curvature'' is read as loop holonomy and sources correspond to persistent mismatch/defect sectors.

\subsubsection{Amplitude variation: the \texorpdfstring{$\chi$}{chi} equation}
\label{subsec:chi_equation_variation}

\begin{proposition}[Amplitude equation]
\label{prop:chi_eom}
Varying \eqref{eq:ref_action_for_variation} with respect to $\chi$ yields
\begin{equation}
\label{eq:chi_eom}
2\lambda_F\,\square\chi-\frac{\dd V}{\dd\chi}=0,
\qquad \square:=\nabla^\mu\nabla_\mu.
\end{equation}
\end{proposition}

\subsubsection{Conservation and frequency-redshift semantics}
\label{subsec:conservation_redshift}

\paragraph{Covariant conservation.}
\MathTag By diffeomorphism invariance one has $\nabla^\mu G_{\mu\nu}=0$ (Bianchi identity), hence \eqref{eq:einstein_total_stress} implies covariant conservation of the total stress:
\begin{equation}
\label{eq:total_stress_conservation}
\nabla^\mu\left(T^{(m)}_{\mu\nu}+T^{(\chi)}_{\mu\nu}+T^{(\mathrm{YM})}_{\mu\nu}\right)=0,
\end{equation}
with exchange terms between sectors determined by explicit couplings in $\mathcal{L}_m$.

\paragraph{Frequency-first interpretation (lapse/redshift).}
\InterfaceTag In a static or adiabatic regime, the lapse $N$ (Appendix~\ref{app:time_mass_delay}) can be identified with $\sqrt{-g_{00}}$ in a suitable gauge.
Then a primary observable is the \emph{frequency ratio} between two locations:
\[
\frac{\omega_{\mathrm{obs}}(x)}{\omega_{\mathrm{obs}}(y)}=\frac{N(y)}{N(x)},
\]
which is the continuum encoding of the tick-first idea that mass/energy and redshift are frequency dictionaries.

\subsubsection{Weak-field limit: Poisson equation and a \texorpdfstring{$1/r$}{1/r} potential}
\label{subsec:weak_field_poisson}

\paragraph{Newtonian template.}
\InterfaceTag In the weak-field, slow-motion regime one writes
$g_{00}\approx -(1+2\phi)$ with Newtonian potential $\phi$.
Keeping only leading order and taking the $00$ component of \eqref{eq:einstein_total_stress} yields a Poisson-type equation
\[
\Delta\phi = 4\pi G\,\rho_{\mathrm{eff}},
\]
where $\rho_{\mathrm{eff}}$ is the effective energy density extracted from the total stress tensor.
In vacuum outside localized sources, $\Delta\phi=0$ and the spherically symmetric exterior solution is $\phi(r)=-GM/r$.

\MatchTag The standard GR lapse and Shapiro-delay formulas used elsewhere in this paper as matching templates are recorded in Appendix~\ref{app:time_mass_delay_reference}.
In the frequency-first reading, these are equivalently statements about how clock rates and signal frequencies vary in the weak-field potential.
