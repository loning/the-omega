\paragraph{Time Image and Discretization under Finite-State Realization}

\begin{proposition}[Time Image under Finite-State Realization: Discrete Quantization vs.\ Densification Dichotomy]\label{prop:xi-finite-state-time-image-discrete-or-dense}
Let the visible groupoid \(\mathcal G\) possess a finitely generated directed realization (finite states, finite labeled edge set \(E\)), and let the additive time cocycle \(\widetilde{\ell}:\mathcal G\to\mathbb R\) satisfy \(\widetilde{\ell}(gh)=\widetilde{\ell}(g)+\widetilde{\ell}(h)\).
If the value set on generating edges \(\{\widetilde{\ell}(e):e\in E\}\) is finite, then
\[
\widetilde{\ell}(\mathcal G)\subseteq
H:=\Bigl\langle \widetilde{\ell}(e):e\in E\Bigr\rangle_{\mathbb Z}\ \le\ (\mathbb R,+).
\]
Moreover, \(H\) is finitely generated as a \(\mathbb Z\)-module and thus satisfies the dichotomy:
\[
H=h\mathbb Z\ \text{(discrete)}\qquad\text{or}\qquad \overline{H}=\mathbb R\ \text{(dense)}.
\]
In the discrete case, \(h\) can be taken as the smallest positive element of \(H\cap(0,\infty)\) (equivalently, \(\inf(H\cap(0,\infty))\)), and \(\widetilde{\ell}(\mathcal G)\subset h\mathbb Z\).
Furthermore, \(\widetilde{\ell}(\mathcal G)\) is at most countable, so any truly continuous time image (uncountable) can only be obtained in the limit as the number of states tends to infinity.
\end{proposition}

\begin{proof}
Since \(\mathcal G\) is generated by a finite edge set, \(\widetilde{\ell}(g)\) for any \(g\in\mathcal G\) is an integer linear combination of the generating edge values, hence \(\widetilde{\ell}(\mathcal G)\subseteq H\).
Any finitely generated \(\mathbb Z\)-subgroup \(H\le\mathbb R\) either contains a smallest positive element (and hence must be the cyclic group \(h\mathbb Z\) generated by that element) or is dense in \(\mathbb R\); this is the standard classification of additive subgroups.
Finally, the finitely generated group \(H\cong \mathbb Z^k\) is countable, so \(\widetilde{\ell}(\mathcal G)\) is countable.
\end{proof}

\paragraph{Euler Product, Holonomy, and Cohomological Reduction of RH}

\begin{theorem}[Time Euler Decomposition Theorem: Zero Mixed Curvature Forces a Prime Element Product Decomposition of the Partition Function]\label{thm:xi-time-euler-product-under-zero-mixed-curvature}
Let \((\mathcal M_\times,\circ_\times)\subset \mathrm{Mor}(\mathbf{PW})\) be the multiplicative composition monoid, and suppose there exists a time cocycle \(\widetilde{\ell}\) strictly additive with respect to \(\circ_\times\).
Let \(\mathcal P\subset\mathcal M_\times\) be the set of \(\circ_\times\)-irreducible elements (primes).
If the following conditions hold:
\begin{enumerate}
  \item \(\mathcal M_\times\) is commutative, left-cancellative, and is the free commutative monoid on the primes (equivalently, every element has a unique prime decomposition, counted with multiplicity);
  \item the mixed curvature of additive-multiplicative alternating squares is identically zero (Theorem \ref{thm:xi-mixed-curvature-zero-criterion}), so that time has no central compensation defect between the additive and multiplicative layers;
\end{enumerate}
then for any \(\lambda>0\) in the domain of absolute convergence, the multiplicative partition function
\[
Z_\times(\lambda):=\sum_{w\in\mathcal M_\times}e^{-\lambda \widetilde{\ell}(w)}
\]
admits the Euler product decomposition
\[
Z_\times(\lambda)=\prod_{p\in\mathcal P}\bigl(1-e^{-\lambda \widetilde{\ell}(p)}\bigr)^{-1}.
\]
Conversely, if \(Z_\times\) admits the above Euler product on some open interval \(\lambda\in(\lambda_0,\infty)\) and \(\mathcal P\) generates \(\mathcal M_\times\), then the mixed curvature must be zero (equivalently, additive and multiplicative updates commute in the time sense without requiring additional central compensation).
\end{theorem}

\begin{proof}
Under unique prime decomposition and additivity,
\(
\widetilde{\ell}\bigl(\prod_p p^{n_p}\bigr)=\sum_p n_p\widetilde{\ell}(p)
\).
For \(\lambda\) in the domain of absolute convergence, rewriting the sum according to prime decomposition and exchanging summation and product yields
\[
Z_\times(\lambda)
=\prod_{p\in\mathcal P}\sum_{n\ge 0}e^{-\lambda n\widetilde{\ell}(p)}
=\prod_{p\in\mathcal P}(1-e^{-\lambda\widetilde{\ell}(p)})^{-1}.
\]
In the converse direction, the Euler product holding on an open interval forces the coefficients at all orders to satisfy the multiplicative-type generating structure independent of primes, compatible with the path independence of "alternating squares having no defect"; if the mixed curvature were nonzero, the additive/multiplicative alternation would introduce central compensation terms not absorbable by the boundary, destroying this multiplicative decomposition, so \(K\equiv 0\) is necessary.
\end{proof}

\begin{theorem}[Prime Factorization Rigidity of Time Rays: Extremal Rays of the Positive Cone Are in Bijection with Morita-Irreducible Product Factors of the Protocol]\label{thm:xi-time-ray-prime-factorization-rigidity}
Let \(\mathcal G\) be an ergodic strongly connected visible groupoid, and let the time cocycle class \([\widetilde\ell]\in H^1(\mathcal G;\mathbb R)\) lie in the interior of the positive cone (equivalent to the existence of an all-edge strictly positive gauge; see the finite graph version of Proposition \ref{prop:xi-positive-time-cone-criterion}).
If \(\mathcal G\) admits a nontrivial product decomposition in the Morita sense
\[
\mathcal G\ \simeq\ \mathcal G^{(1)}\times \mathcal G^{(2)},
\]
and the two factors are independent in the audit sense (certificate composition spans tensor products, cost/trace/primitive fingerprints factor across factors),
then the time cohomology class must split as the sum of two non-collinear positive cocycles: there exist \([\widetilde\ell^{(i)}]\in H^1(\mathcal G^{(i)};\mathbb R)\) (each an interior point of the positive cone) such that
\[
[\widetilde\ell]\ =\ [\widetilde\ell^{(1)}]\oplus[\widetilde\ell^{(2)}],
\]
and \([\widetilde\ell^{(i)}]\) are respectively intrinsically determined by \(\mathcal G^{(i)}\) (up to gauge coboundary).
\par
Conversely, if there exists a split in the positive cone
\[
[\widetilde\ell]=[\alpha]+[\beta],
\qquad
[\alpha],[\beta]\ \text{are both interior points of the positive cone with disentangled supports (commutable to homotopy at the certificate level)},
\]
then \(\mathcal G\) must decompose into two nontrivial factors in the Morita sense, with the corresponding time cocycle classes (up to scaling/coboundary) being precisely \([\alpha]\) and \([\beta]\).
Therefore, the "irreducibility" of a protocol can be characterized by the extremal rays of the time positive cone: there exists a natural correspondence between the extremal rays of the positive cone and Morita-irreducible factors.
\end{theorem}
\begin{proof}
Morita product decomposition corresponds to tensor decomposition at the groupoid algebra level; under this decomposition, any additive cocycle (time) must be the sum of cocycles from the two factors, and positivity is preserved on the factors, yielding the first part of the conclusion.
In the converse direction, "disentangled supports/commutable to homotopy" ensures that certificate composition and fingerprint output tensorially split on the two families of generators; therefore one can separately close by factor generators to obtain two sub-groupoids, and the universal property gives the Morita decomposition.
The extremal ray characterization follows from: if \([\widetilde\ell]\) can be written in the positive cone as the sum of two non-collinear interior points, then the protocol decomposes in the above sense; conversely, Morita-irreducibility precludes such splitting. The proof is complete.
\end{proof}

\begin{theorem}[Cohomological Reduction of RH via Time Unification: Deviation from the Critical Line Is Equivalent to Nontrivial Time Holonomy]\label{thm:xi-rh-cohomological-reduction-holonomy}

\)
corresponding to \(\Xi(s)=\Xi(1-s)\).

\begin{enumerate}
  \item Strictly positive time: the time sum of any nontrivial closed loop is nonzero;
  \item The holonomy group \(\mathfrak H\) of the scale--time connection is trivial (Theorem \ref{thm:xi-scale-time-connection-holonomy-criterion});
  \item The critical slice pressure criterion holds (Theorem \ref{thm:xi-critical-slice-pressure-zero-drift-lyapunov}), so that \(\theta=0\) is the unique "double-zero" slice;
\end{enumerate}
then for any \(\theta\neq 0\), necessarily \(\rho(\mathcal L_{\theta})\neq 1\), and hence \(\det(I-e^{-s}\mathcal L_{\theta})\neq 0\) cannot occur on a slice with \(\theta\neq 0\).
Therefore all nontrivial zeros must satisfy \(\theta(s)=0\), i.e., \(\Re(s)=\tfrac12\).
Under the above representation hypotheses, the Riemann Hypothesis can be equivalently restated as the critical slice uniqueness forced by "vanishing scale--time holonomy."
\end{theorem}

\begin{proof}
By Theorem \ref{thm:xi-critical-slice-pressure-zero-drift-lyapunov}, under the auditable micro-reversibility and time-reversal framework, \(\theta=0\) is the unique slice simultaneously satisfying zero drift and zero Lyapunov exponent.
Trivial holonomy eliminates the freedom to introduce additional central compensation in cross-scale assembly, so that criticality can no longer be maintained along \(\theta\neq 0\) directions.
Therefore, for \(\theta\neq 0\), necessarily \(\rho(\mathcal L_{\theta})\neq 1\), contradicting the determinant zero occurring under non-critical spectral conditions; the conclusion follows.
\end{proof}

\begin{proof}

\end{proof}

\paragraph{Positive Time Gauge, Smith Reduction, and PBW Commutativity}

\begin{proposition}[Cohomological Positive Cone Criterion for Positive Time: Existence of an All-Edge-Positive Gauge]\label{prop:xi-positive-time-cone-criterion}
Let \(\Gamma=(V,E)\) be a finite strongly connected directed graph, \(t:E\to\mathbb R\) a time weight, and consider the gauge transformation \(t\mapsto t+\partial b\) (\(b:V\to\mathbb R\)).
Then there exists \(b\) such that \(t+\partial b\) is strictly positive on all edges if and only if for every directed cycle \(\gamma\),
\[
\sum_{e\in\gamma} t(e)>0.
\]
Equivalently, \([t]\in H^1(\Gamma;\mathbb R)\) lies in the interior of the positive cone defined by all directed cycles
\[
\mathcal C_+:=\Bigl\{[c]\in H^1:\ \langle[c],[\gamma]\rangle>0\ \text{for all nontrivial cycles }\gamma\Bigr\}.
\]
\end{proposition}

\begin{proof}
For any cycle \(\gamma\), \(\sum_{e\in\gamma}\partial b(e)=0\), so the cycle sum is a cohomological invariant.
If there exists \(b\) making all edges positive, then any cycle sum must be positive, establishing necessity.
The converse is the feasibility criterion for difference constraints: the inequality \(t(e)+b(r(e))-b(s(e))>0\) is equivalent to a strict difference constraint system, which is feasible if and only if no cycle has total weight \(\le 0\); in the present setting, this means all cycle sums are strictly positive.
\end{proof}

\begin{theorem}[Tropicalization Completeness of Time: Every "Composable--Comparable" Protocol Cost Necessarily Factorizes as a Non-Archimedean Valuation]\label{thm:xi-time-tropicalization-completeness}

\begin{enumerate}

\end{theorem}
\begin{proof}
The three axioms are precisely the semiring homomorphism conditions "multiplication goes to \(+\), addition goes to \(\min\), units/zeros preserved"; hence \(T:\mathfrak S\to\mathbb T\) is a semiring homomorphism.
Uniqueness comes from: the operations of \(\mathbb T\) are fixed by \(+\) and \(\min\); any map satisfying the three axioms, once its values on generators are given, has its extension on \(\mathfrak S\) completely determined by composition additivity and the minimum law of idempotent sums.

In the converse direction, gauge coboundary perturbation does not change cycle periods, hence does not change the cohomology principal class; with the cohomology principal class fixed, the choice of \(T\) corresponds only to the choice of gauge normalization and equivalent realization at the representative level. The proof is complete.
\end{proof}

t-t'=\partial b\ \text{for some }b:V\to h\mathbb Z
\quad\Longleftrightarrow\quad
\sum_{e\in\gamma}t(e)=\sum_{e\in\gamma}t'(e)\ \text{for all directed cycles }\gamma.
\]
Moreover, the verification "for all cycles" can be reduced to checking on any finite fundamental cycle family generating \(H_1(\Gamma;\mathbb Z)\): take any spanning tree \(T\), and the fundamental cycles \(\gamma_e\) formed by each chord edge \(e\in E\setminus T\) constitute a basis of the integer cycle space; checking equality of time sums on finitely many \(\{\gamma_e\}\) suffices to deduce cohomological equivalence.
\end{proposition}

\begin{proof}
Necessity follows directly from \(\sum\partial b=0\) on cycles.
The reverse direction is the standard characterization of \(1\)-cohomology on finite graphs: if two cocycles have the same pairing on a cycle basis, their difference is zero in \(H^1\), hence is a coboundary.
\end{proof}
