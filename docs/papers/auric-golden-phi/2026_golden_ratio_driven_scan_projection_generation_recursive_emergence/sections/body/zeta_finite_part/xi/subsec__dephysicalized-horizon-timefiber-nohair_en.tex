\begin{proposition}[logarithmic半径derivative律：defect的derivative即盘内zero计数]\label{prop:dephys-jensen-defect-log-derivative}

\end{proposition}

\begin{corollary}[exclusion半径的zero计数驱动律]\label{cor:dephys-repulsion-radius-zero-count-driving}

\end{corollary}

\begin{proof}

\end{proof}

其中 \(t(\varrho,\theta)\) 具有显式分解（式 \eqref{eq:app-t-rho-theta-explicit}）

t(\varrho,\theta)
=\frac{2\varrho\sin\theta}{1+\varrho^2+2\varrho\cos\theta}
\;+\;
\mathrm{i}\,\frac{1-\varrho^2}{1+\varrho^2+2\varrho\cos\theta}.

\end{theorem}

\begin{proof}

\begin{corollary}[Poisson--Busemann identity：time fiber坐标即endpoint Poisson weight]\label{cor:dephys-poisson-busemann-timefiber}
对for any \(w\in\mathbb{D}\)，let \(t=\mathcal{C}^{-1}(w)=x+\mathrm{i}y\in\mathbb{H}\)。
then成立严格identity
\[
\boxed{\;y=\mathcal{B}_{-1}(w)=\frac{1-|w|^2}{|1+w|^2}\;}
\]
其中 \(\mathcal{B}_{-1}\) 为以endpoint \(-1\) 为基点的指数化 Busemann 坐标（Proposition \ref{prop:app-busemann-poisson-minus1}）。
therefore，在endpoint紧化口径下，“time fiber”并非额外引入的连续坐标轴，而是endpoint \(-1\) 的谐measure密度（Poisson kernel）本身。
\end{corollary}

\begin{proof}
let \(t=x+\mathrm{i}y\in\mathbb{H}\) 且 \(w=\mathcal{C}(t)=\frac{1+\mathrm{i}t}{1-\mathrm{i}t}\)。
then有
\[
1+w
=1+\frac{1+\mathrm{i}t}{1-\mathrm{i}t}
=\frac{2}{1-\mathrm{i}t},
\qquad
|1+w|^2=\frac{4}{|1-\mathrm{i}t|^2}=\frac{4}{x^2+(1+y)^2}.
\]
and由
\(
|w|^2=\frac{|1+\mathrm{i}t|^2}{|1-\mathrm{i}t|^2}
\)
得
\[
1-|w|^2
=\frac{|1-\mathrm{i}t|^2-|1+\mathrm{i}t|^2}{|1-\mathrm{i}t|^2}
=\frac{4y}{x^2+(1+y)^2}.
\]
两式相除即得 \(\frac{1-|w|^2}{|1+w|^2}=y\)，thus \(\mathcal{B}_{-1}(w)=y\)。
特别地，出现在 \(1-|w|^2\) 与 \(|1+w|^2\) 中的constant \(4\) 仅为 Cayley 规一化下共形因子 \(2\) 的平方，在该不变量比值中严格消去。\qed
\end{proof}

\begin{remark}[Cayley 标尺与constant \(4\) 的可规一化性：不变量只依赖比值与logarithmic差]\label{rem:dephys-cayley-scale-gauge-four}
the above推导中的“\(4\)”属于坐标规一化constant，而非新的动力学不变量。
为说明这一点，考虑带标尺参数 \(\lambda>0\) 的 Cayley 族
\[
\mathcal{C}_{\lambda}(t):=\frac{\lambda+\mathrm{i}t}{\lambda-\mathrm{i}t}\qquad(t\in\mathbb{H}),
\]
它保持同一endpoint基准（\(t=\infty\) 仍映到 \(-1\)）但改变线性标尺。
直接计算得到
\[
|1+\mathcal{C}_{\lambda}(x+\mathrm{i}y)|^2
=\frac{4\lambda^2}{x^2+(\lambda+y)^2},
\qquad
1-|\mathcal{C}_{\lambda}(x+\mathrm{i}y)|^2
=\frac{4\lambda y}{x^2+(\lambda+y)^2}.
\]
therefore指数化 Busemann/Poisson 比值仍严格消去共形平方constant，并给出
\[
\mathcal{B}_{-1}\!\bigl(\mathcal{C}_{\lambda}(x+\mathrm{i}y)\bigr)=\frac{y}{\lambda}.
\]
takelogarithmic得到 \(b_{-1}:=\log\mathcal{B}_{-1}=\log y-\log\lambda\)，表明标尺transform只改变 \(b_{-1}\) 的加性zero；
therefore任何可观测几何量只能出现在对重标定不变的组合中（如 \(\mathcal{B}_{-1}\) 的比值、或 \(b_{-1}\) 的logarithmic差），而不可能依赖孤立的“\(4\)”constant。
在ledger语义中亦可把 \(b_{-1}\) 视作一类“熵势/logarithmic势”，此时规一化constant只决定加性zero而不改变差分观测；其形式角色与几何熵公式中的fix规一化constant（Example如黑洞熵中的 \(1/4\)）同型。
\end{remark}

\begin{remark}[共形紧化中的 \(2^2\) 因子与“面product律”形式类比]\label{rem:dephys-conformal-factor-2square-area-law}
结论 \ref{cor:dephys-poisson-busemann-timefiber} 中的共形平方constant并非一维特Example。
在更高维的紧化/projection坐标中，同样出现共形因子 \(\Omega(x)=\frac{2}{1+|x|^2}\) 及其平方（或更一般幂次），thus球面度量与面product元的拉回密度带有“\(4\)”型constant。

\qquad

\]
这进一步表明：this paperendpointhorizoninterface中出现的“\(4\)”应被理解为紧化共形因子的通用平方constant，而非特定算术object所独有的不变量。
\end{remark}

\begin{proposition}[曲率规一化定律：disk双曲度量中的coefficient \(4\) equivalent to常曲率 \(-1\) 的unique标尺]\label{prop:dephys-hyperbolic-curvature-normalization-four}
upper half-plane模型take双曲度量
\[
\dd s_{\mathbb{H}}^{2}:=\frac{\dd x^{2}+\dd y^{2}}{y^{2}}
\]
then其高斯曲率为 \(-1\)。
任take把 \(\mathbb{H}\) 共形等价到unit circle盘 \(\mathbb{D}\) 的 Möbius transform \(w=w(t)\)，disk上的双曲度量必可写为
\[
\boxed{\;
\dd s_{\mathbb{D}}^{2}=\frac{4\,|\dd w|^{2}}{(1-|w|^{2})^{2}}\;}
\]
and该coefficient \(4\) 由“曲率规一化”为 \(-1\) unique锁定：
if改写为
\(
\dd s_{\mathbb{D}}^{2}=\frac{\alpha\,|\dd w|^{2}}{(1-|w|^{2})^{2}}
\),
then曲率变为
\[
\boxed{\;K_{\alpha}=-\frac{4}{\alpha}.\;}
\]
\end{proposition}

\begin{proof}
对共形度量 \(\dd s^{2}=\rho(w)^{2}(\dd x^{2}+\dd y^{2})=\rho(w)^{2}|\dd w|^{2}\)，高斯曲率满足
\(
K=-\rho^{-2}\Delta(\log\rho)
\)
（\(\Delta\) 为欧氏拉普拉斯）。
take \(\rho_{\alpha}(w)=\frac{\sqrt{\alpha}}{1-|w|^{2}}\)，then
\(
\log\rho_{\alpha}=\frac12\log\alpha-\log(1-|w|^{2})
\)，constant项拉普拉斯为零，且直接计算得
\(
\Delta\bigl(-\log(1-|w|^{2})\bigr)=\frac{4}{(1-|w|^{2})^{2}}
\)。
hence
\[
K_{\alpha}
=-\frac{(1-|w|^{2})^{2}}{\alpha}\cdot \frac{4}{(1-|w|^{2})^{2}}
=-\frac{4}{\alpha}.
\]
therefore \(K_{\alpha}=-1\) if and only if \(\alpha=4\)。
而for any Möbius 等价 \(w=w(t)\) 仅改变坐标，不改变“曲率 \(-1\) 的unique共形完备度量”这一规范化结论，hencedisk模型必落在 \(\alpha=4\) 的标尺上。\qed
\end{proof}

\begin{remark}[horizonmeasure规一化：Poisson kernel的单位质量与曲率 \(-1\) 标尺的相容性]\label{rem:dephys-poisson-kernel-unit-mass-curvature-gauge}
在unit circle盘上，Poisson kernel

该单位质量约定与this paperfix的 Carath\'eodory/Herglotz 表示口径相容：boundarymeasure以 \(\frac{1}{2\pi}\dd\theta\) 归一化为概率measure。
\par
从几何角度看，Proposition \ref{prop:dephys-hyperbolic-curvature-normalization-four} 表明“coefficient \(4\)”对应曲率 \(-1\) 的unique标尺；一旦整体重标定该coefficient，就equivalent to改变曲率（thus改变几何单位），并将在以logarithmic势/面product作ledger的量（Example如 \(b_{-1}=\log\mathcal{B}_{-1}\) 或defect熵 \(S_{\mathrm{def}}\)）中体现为单位换算constant。

\end{remark}

\paragraph{endpoint吸引半群与“时间”Definition：unique Koenigs depth}

\begin{theorem}[endpoint线性化的rigidity：Koenigs depth的unique性]\label{thm:dephys-endpoint-koenigs-rigidity}

\begin{equation}\label{eq:dephys-koenigs-functional-equation}
h\bigl(\phi_m(u)\bigr)=\frac{1}{m}\,h(u)\qquad(\forall\,m>0).
\end{equation}
then在该neighborhood内there existsconstant \(c\neq 0\) 使 \(h(u)=c\,\mathfrak{w}(u)\)，其中 \(\mathfrak{w}(u)=\frac{1+u}{1-u}\) 为endpoint半平面坐标（式 \eqref{eq:scale-linearization-halfplane-coordinate}）；and对all \(m>0\) 与all \(u\in\mathbb{D}\) 有 \(\phi_m=\psi_m\)，其中 \(\psi_m\) 由式 \eqref{eq:scale-psi-m} 给出。

\[

\]

\[
Y\!\bigl(\psi_m(u)\bigr)=Y(u)+\log m
\]
（即式 \eqref{eq:endpoint-busemann-addition-law}）。
\end{theorem}

\begin{proof}
由 \eqref{eq:dephys-koenigs-functional-equation} 在 \(u=-1\) 处求导并用 \(h'(-1)\neq 0\) 得 \(\phi_m'(-1)=1/m\)。
另一方面，endpoint坐标 \(\mathfrak{w}\) 将 \(\mathbb{D}\) conjugate到右半平面 \(\mathbb{H}_{\mathrm{R}}\)（式 \eqref{eq:scale-linearization-halfplane-coordinate}），并把 \(\phi_m\) conjugate为 \(\mathbb{H}_{\mathrm{R}}\) 的自isomorphism且保持 \(0\) 与 \(\infty\) 不动；hencethere exists \(a_m>0\) 使
\(
\mathfrak{w}(\phi_m(u))=a_m\,\mathfrak{w}(u)
\)，并由链式法then得 \(a_m=\phi_m'(-1)=1/m\)。
于是 \(\mathfrak{w}(\phi_m(u))=\mathfrak{w}(u)/m\)，与Proposition \ref{prop:scale-linearization-identity-multipliers} 的 \(\mathfrak{w}(\psi_m(u))=\mathfrak{w}(u)/m\) 一致；因 \(\mathfrak{w}\) 单射，得 \(\phi_m=\psi_m\)。
再let \(g(u):=\frac{h(u)}{\mathfrak{w}(u)}\)（在 \(u=-1\) neighborhood内解析），由 \eqref{eq:dephys-koenigs-functional-equation} 与 \(\mathfrak{w}(\psi_m(u))=\mathfrak{w}(u)/m\) 得 \(g(\psi_m(u))=g(u)\)；
takefor any \(m>1\)，iteration \(\psi_m^{\circ n}(u)\to-1\)（endpoint吸引），由连续性得 \(g(u)=g(-1)\) 在neighborhood内恒定，thus \(h=c\,\mathfrak{w}\)。
最后的平移律即为式 \eqref{eq:endpoint-busemann-addition-law}。\qed
\end{proof}

    其zero事件在半径方向的“折角/跳变”由logarithmic半径derivative律 \( \frac{\dd}{\dd(\log\varrho)}D_\zeta(\varrho)=N_{F_\zeta}(\varrho) \) 精确承载（Proposition \ref{prop:dephys-jensen-defect-log-derivative}），并由exclusion半径驱动律 \( \frac{\dd}{\dd(\log\varrho)}\log r_\ast(\varrho)=1-N_{F_\zeta}(\varrho) \) 给出可定位的停滞witness（Corollary \ref{cor:dephys-repulsion-radius-zero-count-driving}）。
    \item \textbf{正性—Toeplitz--PSD layer级链：}
    lethorizon Carath\'eodory logarithmicderivativefunction \(\mathscr{C}_\zeta\) 如附录 \eqref{eq:app-horizon-carath-def} Definition。

    \(
    \mathbf{T}_N=(c_{j-k})_{0\le j,k\le N}\succeq 0
    \)

  由此，两条链在数学上等价，但失败witness互补：前者以defectlower bound/exclusion半径停滞为witness，后者以有限维负eigenvalue为witness。

其结果是：在该protocol口径下，任何“信息泄漏/implementation漂移/endpoint异常”if要与可见kernel验结论共存，必须表现为至少一类certificate的失败；

\end{theorem}

\begin{proof}

\end{proof}

\begin{remark}[endpoint阻碍的二元orthogonal分解：\texorpdfstring{$\zeta$}{zeta} 型logarithmic主scale与 \texorpdfstring{$L(\chi_4)$}{L(chi4)} 型反变本征通道]\label{rem:endpoint-obstruction-dichotomy}
在endpoint \(w\to-1\) 的符号分解口径下，endpoint载荷在偶/反变两类观测上呈现为一对严格orthogonal的分量，其来源可归因于projection系统自isomorphism在endpointneighborhood所诱导的有限维不变子表示。
其一为 \(\zeta\)-型logarithmic主scale：endpoint三通道权kernel的 Dirichlet 正then化coefficient具有闭式 \(\zeta(1+\sigma)\) 符号表示，且其divergence结构由 \(\zeta\) 在 \(1\) 处的pole完全控制（Proposition \ref{prop:endpoint-dirichlet-symbol-weights}）。
其对应的重整化有限部在 \(\sigma\downarrow 0\) 下具有显式闭式（Theorem \ref{thm:endpoint-even-finite-part-renormalization}）。
其二为 \(\chi_4\)-型反变本征通道：在最小反变fiber坐标 \(\delta\) 上，endpoint反变观测抽出一维不变子表示，其符号为 \(L(1+\sigma,\chi_4)\)，并与the above \(\zeta\)-型偶通道权kernel严格解耦（Theorem \ref{thm:endpoint-zeta-Lchi4-symbol-pair}）；and该通道在 \(\sigma\downarrow 0\) 处解析，limit输出为constant \(L(1,\chi_4)=\pi/4\)（Corollary \ref{cor:endpoint-odd-analytic-at-zero}）。
此外，在 primitive 奇长扇区中，endpoint分支差分certificate满足选择律消失，thusendpoint主rigidity退化为纯 \(\zeta\)-型logarithmicledger（注 \ref{rem:endpoint-logscale-parity-decouple}）。
\end{remark}
与有界slice上的语义保守性（Theorem \ref{thm:pom-sem-conservative}）合并后，the above共变/下降律把“隐藏分量”约束为auditable的不变量输出域，thus将判等与implementation自审计收束为有限可复kernelfingerprint。

\begin{theorem}[implementation无关的defectcertificate链：由 Jensen defect到 RH]\label{thm:dephys-impl-independent-defect-certificate}
fix一列半径 \(\varrho_n\uparrow 1\)。
在有界slice \(\mathbf{PW}^{\ast}_{\le}\) 上，letthere exists一列projection词 \(w_n\) 及一列可离线复kernel的有限certificateobject \(C_n\)，使得certificate输出一个upper bound \(\widehat D_n\in\RR_{\ge 0}\) 与误差budget \(\varepsilon_n\ge 0\) 并满足
\begin{equation}\label{eq:dephys-defect-certificate-soundness}
D_{\zeta}(\varrho_n)\le \widehat D_n+\varepsilon_n\qquad(\forall n).
\end{equation}
if \(\widehat D_n+\varepsilon_n\to 0\)，then RH 成立。
此外，ifeach \((\widehat D_n,\varepsilon_n)\) 仅依赖于 \(\mathsf{Sem}(w_n)\) 的有限输出（Definition \ref{def:pom-sem-functor}），then该defectcertificate链在slice内与implementation选择无关：对for any \(w_n'\in\mathbf{PW}^{\ast}_{\le}\) if \(\mathsf{Sem}(w_n')=\mathsf{Sem}(w_n)\)，thenthere existsnaturalisomorphism \(w_n'\simeq w_n\)（Theorem \ref{thm:pom-sem-conservative}），and \((\widehat D_n,\varepsilon_n)\) 在该语义意义下保持不变。
\end{theorem}

\begin{proof}
由 \eqref{eq:dephys-defect-certificate-soundness} 与exclusion半径Definition \(r_{\ast}(\varrho)=\varrho e^{-D_{\zeta}(\varrho)}\) 得
\[
r_{\ast}(\varrho_n)\ge \varrho_n\exp\!\bigl(-(\widehat D_n+\varepsilon_n)\bigr)\xrightarrow[n\to\infty]{}1,
\]
hence由Theorem \ref{thm:dephys-defect-repulsion-radius} 推出 RH。
implementation中立性部分：if \((\widehat D_n,\varepsilon_n)\) 仅依赖 \(\mathsf{Sem}(w_n)\)，then \(\mathsf{Sem}(w_n')=\mathsf{Sem}(w_n)\) implies两者输出一致；并由Theorem \ref{thm:pom-sem-conservative} 得 \(w_n'\simeq w_n\)。
在“地址先于take值”的约定下（\S\ref{sec:recursive-addressing}），certificateobject的there exists性即其可寻址性；thus该upper bound链不携带implementation侧的隐式自由度，
亦不可能通过递归地址化引入新的可见事件（Proposition \ref{prop:recursive-addressing-refinement} 与Corollary \ref{cor:recursive-addressing-visible-sigma-nonexpansion}）。\qed
\end{proof}

由此得到一个体系内闭合的可证伪重述：任何需要radial通道解释的偏离，if不经由 residue/逆limit轴或the above离散certificate的非局部externalized，就必在可见审计中触发 \(\mathsf{NULL}\) 并被protocol拒绝；因而，zero审计的可见certificate只能沿“defect—endpoint—phase”链路出现，而不会被额外连续坐标register所稀释。相反，一切可逆化的新增自由度只能以离散毛发进入certificate域，而不可能表现为与phase解耦的额外连续坐标register。

在此基础上，\S\ref{subsec:xi-certificate-coherence-orthogonal-axis-capacity} 进一步把“可见certificate”的一致性提升为certificateobject的逆limit同一性，并将离critical line信息的endpointprojection严格化为二次radial通道与由precision势诱导的capacitylower bound，thus给出“构造离线zero”的可证伪interface表述。
