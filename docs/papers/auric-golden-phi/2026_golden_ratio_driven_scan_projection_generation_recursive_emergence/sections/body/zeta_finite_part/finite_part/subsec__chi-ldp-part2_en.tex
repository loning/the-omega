\paragraph{Intrinsic phase and oscillation period of the subleading mode (auditable)}
Proposition \ref{prop:chi-slice-time-correlation} provides the exponential envelope scale of the correlation $\rho_{\mathrm{corr}}(t)=\Lambda(t)/\lambda_1(t)$. In the finite-dimensional case, this exponential decay is dominated by one or a pair of subleading eigenvalues: if $\lambda_2(t)$ is not a positive real number, then the correlation function, in addition to the decay envelope, also carries an oscillation factor controlled by $\arg(\lambda_2(t))$.

\begin{remark}[Subleading phase and intrinsic oscillation period]
\label{prop:chi-slice-subleading-phase}
Under the setup of Proposition \ref{prop:chi-slice-time-correlation}, let
\[
\lambda_1(t):=\rho\bigl(M_{\theta(t)}\bigr)>0,
\qquad
\lambda_2(t)\in\mathrm{spec}(M_{\theta(t)})\setminus\{\lambda_1(t)\}
\]
be the subleading eigenvalue satisfying $|\lambda_2(t)|=\Lambda(t)$ (if a conjugate pair appears, choose either one). Define
\[
\rho_2(t):=\frac{|\lambda_2(t)|}{\lambda_1(t)}\in[0,1),
\qquad
\omega(t):=\arg\bigl(\lambda_2(t)\bigr)\in(-\pi,\pi],
\]
and when $\omega(t)\ne 0$, define the intrinsic oscillation period
\[
T_{\mathrm{osc}}(t):=\frac{2\pi}{|\omega(t)|}.
\]
Then for any centered finite-state observable $f$, the spectral representation of its correlation function contains a subleading term
\[
\left(\frac{\lambda_2(t)}{\lambda_1(t)}\right)^n
=\rho_2(t)^n\,e^{i n\omega(t)},
\]
so that when $\omega(t)\ne 0$, the correlation decay is superimposed with an intrinsic phase frequency $\omega(t)$ on the exponential envelope $\rho_2(t)^n$; when $\omega(t)=\pi$ (i.e., $\lambda_2(t)<0$), a two-step alternating oscillation emerges.
\end{remark}

\paragraph{Trivial-layer collapse \texorpdfstring{$+$}{+} mechanism purity constant (genuine mechanism invariant)}
For this kernel,
\[
\Delta(z)=\det(I-zM)=(1-z)^2(1+z)^3(1-3z+z^2)(1+z-z^2)
\]
yields a natural ``trivial layer/mechanism layer'' separation: \((1-z)^2(1+z)^3\) corresponds to boundary fixed points and sign-trivial modes, while the two quadratic factors carry the genuine mechanism blocks.
Accordingly, one can define a ``mechanism purity constant'' after stripping the trivial drift, for cross-kernel/cross-twist comparable alignment.

\begin{definition}[Mechanism purity constant: stripping trivial factors]\label{def:real-input-40-mechanism-purity-constant}
Let \(z_\star=\lambda^{-1}=\varphi^{-2}\), and write
\[
\Delta_{\mathrm{triv}}(z):=(1-z)^2(1+z)^3,\qquad
\Delta_{\mathrm{mech}}(z):=(1-3z+z^2)(1+z-z^2).
\]
Define the \textbf{genuine mechanism contribution (mechanism purity constant)} as
\[
\boxed{\;
\log\mathfrak{M}_{\mathrm{mech}}
:=
-\sum_{m\ge 2}\frac{\mu(m)}{m}\log \Delta_{\mathrm{mech}}(z_\star^{\,m})
\; }.
\]
And define the \textbf{trivial-layer contribution} as
\[
\boxed{\;
\log\mathfrak{M}_{\mathrm{triv}}
:=\log\mathfrak{M}-\log\mathfrak{M}_{\mathrm{mech}}.\;}
\]
\end{definition}

\begin{proposition}[Explicit trivial-layer collapse and two-kernel mechanism constant]\label{prop:real-input-40-triv-mech-split}
Under the notation of Definition \ref{def:real-input-40-mechanism-purity-constant}, the following strict decomposition holds:
\[
\boxed{\ \log\mathfrak{M}=\log\mathfrak{M}_{\mathrm{triv}}+\log\mathfrak{M}_{\mathrm{mech}}\ }.
\]
The trivial-layer contribution can be written entirely in closed form as
\[
\boxed{\;
\log\mathfrak{M}_{\mathrm{triv}}
=\log C-z_\star+3z_\star^2+2\log(1-z_\star)+3\log(1+z_\star)\; }.
\]
Equivalently, \(\mathfrak{M}_{\mathrm{mech}}:=\exp(\log\mathfrak{M}_{\mathrm{mech}})\) agrees with the quadratic carry constant in the appendix (Theorem \ref{thm:real-input-40-carry-constant}), and the above decomposition is precisely the ``two-kernel compression'' identity (Proposition \ref{prop:real-input-40-mertens-two-series}).
\end{proposition}

\paragraph{Computable procedure (isomorphic to ``discretization--regularization--finite part'')}
Given an adjacency matrix $M$ of a finite graph shift (or a weighted matrix family $M_\theta$), and fixing a truncation order $K\ge 2$, the finite-part constant can be computed by the following steps:
\begin{enumerate}
  \item \textbf{Discretization:} Obtain the matrix $M$ (or $M_\theta$) from the graph/transducer, and define $\zeta(z)=1/\det(I-zM)$ (or $\zeta_\theta(z)=1/\det(I-zM_\theta)$).
  \item \textbf{Leading scale:} Compute the Perron--Frobenius leading eigenvalue $\lambda=\rho(M)$ (or $\lambda(\theta)=\rho(M_\theta)$), and set $z_\star=1/\lambda$ (or $z_\star(\theta)=1/\lambda(\theta)$).
  \item \textbf{Residue constant:} Compute the leading-pole residue constant
  \[
  C:=\lim_{z\to z_\star}(1-\lambda z)\zeta(z)
  \qquad
  (\text{or }C(\theta):=\lim_{z\to z_\star(\theta)}(1-\lambda(\theta)z)\zeta_\theta(z)).
  \]
  In the ``simple pole'' case, $C$ can also be expressed in closed form via the determinant derivative or eigenvalue product (see Appendix \ref{app:real-input-40-finite-part}).
  \item \textbf{Regularization (Abel sampling):} For $k=2,\dots,K$, sample $\log\zeta(z_\star^k)$ (or $\log\zeta_\theta(z_\star(\theta)^k)$) within the convergence disk.
  \item \textbf{Finite part:} Approximate using the truncated M\"{o}bius tail sum
  \[
  \log\mathfrak{M}\approx \log C+\sum_{k=2}^{K}\frac{\mu(k)}{k}\log\zeta(z_\star^k)
  \quad
  (\text{or }\log\mathfrak{M}(\theta)\approx \log C(\theta)+\sum_{k=2}^{K}\frac{\mu(k)}{k}\log\zeta_\theta(z_\star(\theta)^k)).
  \]
  \item \textbf{Mertens constant output:} Define
  \[
  \mathsf{M}:=\gamma+\log\mathfrak{M},
  \qquad
  \mathsf{M}(\theta):=\gamma+\log\mathfrak{M}(\theta).
  \]
\end{enumerate}
Since $|z_\star|<1$, $z_\star^k$ decays exponentially, and $\log\zeta(z)=O(z)$ ($z\to 0$) ensures that $\log\zeta(z_\star^k)$ tends to zero rapidly in $k$, so this procedure enjoys strong numerical stability and rapid convergence.

\paragraph{Orbit--Mertens product (turning $\mathsf{M}$ into a comparable scalar)}
Define the length-truncated prime-orbit Mertens product
\[
\mathcal{P}(N):=\prod_{\substack{\tau\ \mathrm{primitive}\\|\tau|\le N}}\left(1-\lambda^{-|\tau|}\right)^{-1}.
\]
Taking logarithms and using $-\log(1-x)=x+O(x^2)$ yields
\[
\log\mathcal{P}(N)=\sum_{n\le N}\frac{p_n}{\lambda^n}+O(1)
=\log N+\mathsf{M}+o(1),
\]
and therefore
\[
\mathcal{P}(N)\sim e^{\mathsf{M}}\,N=e^{\gamma}\mathfrak{M}\,N.
\]
If one rewrites using the ``orbit norm'' $N(\tau):=\lambda^{|\tau|}$ and sets $x=\lambda^N$ (equivalently $N=\log x/\log\lambda$), one obtains the equivalent $\log x$-type leading term
\[
\prod_{N(\tau)\le x}\left(1-\frac{1}{N(\tau)}\right)^{-1}
\sim \frac{e^{\gamma}\mathfrak{M}}{\log\lambda}\,\log x.
\]

\paragraph{Weighted interface and \texorpdfstring{$\log\mathfrak{M}(\theta)$}{logM(theta)} phase diagram}
For the edge potential vector $\Phi$ (e.g., $\varphi_e,\varphi_-,\varphi_2$ in Appendix \ref{app:real-input-40-logm-theta}), define the weighted matrix family $M_\theta$ and
\(
\zeta_\theta(z)=1/\det(I-zM_\theta)
\).
Let $\lambda(\theta)=\rho(M_\theta)$, $z_\star(\theta)=\lambda(\theta)^{-1}$. In the same calibration as above, define
\[
P_\theta(z)=\sum_{k\ge 1}\frac{\mu(k)}{k}\log\zeta_\theta(z^k).
\]
\begin{definition}[Abel/Hadamard finite-part constant]\label{def:abel-finite-part}
For the weighted matrix family $M_\theta$ of a finite graph shift, define the Abel/Hadamard finite-part constant as
\[
\log\mathfrak{M}(\theta):=\mathrm{fp}_{r=1}\Bigl(P_\theta(r/\lambda(\theta))+\log(1-r)\Bigr),
\]
where $P_\theta(z)=\sum_{k\ge 1}\frac{\mu(k)}{k}\log\zeta_\theta(z^k)$ is the M\"{o}bius--$\log\zeta$ transform, $\lambda(\theta)=\rho(M_\theta)$ is the Perron--Frobenius leading eigenvalue, and $\mathrm{fp}_{r=1}$ denotes the finite part (Laurent constant term) at $r=1$.
\end{definition}
\begin{proposition}[M\"{o}bius--$\zeta$ closed form of the Abel finite part]\label{prop:abel-finite-part-mobius-zeta}
Under the setup of Definition \ref{def:abel-finite-part}, the Abel finite-part constant admits the computable closed form
\[
\log\mathfrak{M}(\theta)
=\log C(\theta)+\sum_{k\ge 2}\frac{\mu(k)}{k}\log\zeta_\theta\bigl(z_\star(\theta)^k\bigr),
\qquad
C(\theta):=\lim_{z\to z_\star(\theta)}(1-\lambda(\theta)z)\,\zeta_\theta(z),
\]
where $C(\theta)$ is the leading-pole residue constant and $z_\star(\theta)=\lambda(\theta)^{-1}$ is the leading pole location.
\end{proposition}
Equivalently (in the same calibration as Appendix \ref{app:real-input-40-logm-theta}), this can also be written in a single-line M\"{o}bius--Abel form
\[
\log\mathfrak{M}(\theta)=\log C(\theta)-\sum_{m\ge 2}\frac{P_\theta(\lambda(\theta)^{-m})}{m}.
\]
Accordingly, define the weighted Mertens constant
\(
\mathsf{M}(\theta):=\gamma+\log\mathfrak{M}(\theta)
