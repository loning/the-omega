\begin{remark}[Hankel minimal realization functor: lifting sequences to minimal linear representations]\label{rem:pom-minh-functor}

\end{remark}

\subsubsection{Moment-Kernel Construction for Arbitrary-Order Collision Moments and Multifractal Spectrum}\label{subsec:pom-moment-general}

\paragraph{From \texorpdfstring{$q$}{q}-fold collisions to finite-state moment-kernels}
The key novelty at orders two and three is that the collision moment is not merely an ``enumerable sequence'' but is generated by a \textbf{finite-state fiber-product kernel}, thereby falling entirely within the unified Perron--Frobenius/\(\zeta\) interface.
This fact holds for arbitrary \(q\ge 2\): for fixed \(q\), the constraint that ``\(q\) windows fold to the same stable type'' is a finite-memory condition and can therefore be compiled into a finite graph (SFT/sofic) whose adjacency matrix performs the counting.

\begin{definition}[\texorpdfstring{$q$}{q}-fold collision relation and moment sequence]\label{def:pom-moment-q}
For an integer \(q\ge 2\), define the \(q\)-fold collision relation
$$
R_m^{(q)}:=
\{(\omega^{(1)},\dots,\omega^{(q)})\in\Omega_m^q:\ \Fold_m(\omega^{(1)})=\cdots=\Fold_m(\omega^{(q)})\},
$$
and define the \(q\)-th order collision moment
$$
S_q(m):=|R_m^{(q)}|=\sum_{x\in X_m} d_m(x)^q.
$$
\end{definition}

\begin{proposition}[Existence and executable construction of the moment-kernel at arbitrary order]\label{prop:pom-moment-kernel-exists}
For each fixed \(q\ge 2\), there exists a finite-state nonnegative integer kernel (adjacency matrix) \(A_q\) together with integer readout vectors \(u_q,v_q\) such that
$$
S_q(m)=u_q^{\mathsf T}A_q^{\,m-m_0}v_q\qquad(m\ge m_0),
$$
so that \(\{S_q(m)\}_{m\ge m_0}\) satisfies a \textbf{finite-order integer-coefficient linear recursion}.

Moreover, \(A_q\) can be explicitly constructed via the following ``compilation pipeline'' (and is therefore derivable, rewritable, and computable):
\begin{enumerate}
  \item Choose the deterministic online kernel realizing \(\Fold_\infty\) (the synchronizing kernel with delay \(\delta_{\min}=3\), whose explicit finite-state representation is given in the appendix);
  \item Take \(q\) copies and form the synchronous direct product, extending states to \(q\)-tuples;
  \item At each step retain only those transitions for which all \(q\) copies produce \textbf{the same output bit}, and project onto the reachable essential strongly connected component;
  \item The resulting finite graph is a \(q\)-fold collision fiber-product kernel; its adjacency matrix is the desired \(A_q\), and its path count yields \(S_q(m)\).
\end{enumerate}
\end{proposition}

\begin{proof}
View \(\Fold_\infty\) as a deterministic finite transducer: the input alphabet consists of window letters (or input letters induced by two legal input tracks), the output consists of stable address bits; its internal state space is finite and has constant delay (the online kernel of Proposition \ref{prop:pom-delay-min} is a concrete realization).
For fixed \(q\), the constraint ``\(q\) copies have identical output bit-by-bit'' is a local synchronization constraint: it only needs to compare whether the current output symbols agree at each step, and hence remains finite-state recognizable.
One can therefore form the synchronous direct product of the transducer and apply the ``output-consistency'' filter, obtaining a finite directed graph \(G^{(q)}\).
The number of legal paths of length \(m\) equals the count of \(R_m^{(q)}\) (i.e., \(S_q(m)\)), which can thus be written in the matrix-power readout form \(u_q^{\mathsf T}A_q^{m-m_0}v_q\) for some nonnegative integer adjacency matrix.
The linear recursion follows immediately from the Cayley--Hamilton theorem, and its coefficients are integers (since \(A_q\) is an integer matrix). This completes the proof.
\end{proof}

\begin{proposition}[Finite-state integrability for fixed \texorpdfstring{$q$}{q}: minimal integer-coefficient recursion and Hankel rank]\label{prop:pom-moment-minrecurrence-hankel}
For each integer \(q\ge 2\), the sequence \(m\mapsto S_q(m)\) satisfies from some \(m_0\) onward a homogeneous integer-coefficient linear recursion
\[
S_q(m)=\sum_{j=1}^{d_q} c_{q,j}\,S_q(m-j)\qquad(m\ge m_0),
\]
where \((c_{q,j})\subset\ZZ\) and the order \(d_q\) can be taken to be minimal.
This minimal order \(d_q\) equals the Hankel rank of the sequence (equivalently, equals the dimension of the minimal linear representation \(a_n=u^{\mathsf T}A^n v\); Theorem \ref{thm:hankel-rank-minimal}), and the minimal recursion is uniquely determined under the ``monic'' convention.
\end{proposition}
\begin{proof}
By Proposition \ref{prop:pom-moment-kernel-exists}, for fixed \(q\) there exists an integer-matrix realization \(S_q(m)=u_q^{\mathsf T}A_q^{\,m-m_0}v_q\);
by the Cayley--Hamilton theorem, \(\{S_q(m)\}_{m\ge m_0}\) necessarily satisfies some integer-coefficient constant-coefficient recursion, whence the existence of \(d_q\) is established.
The equivalence of minimal order with Hankel rank/minimal realization dimension is given by Theorem \ref{thm:hankel-rank-minimal}; the uniqueness of the minimal recursion is a direct consequence of that equivalence characterization (minimal realizations are unique up to isomorphism, hence their annihilator polynomials are unique). This completes the proof.
\end{proof}

\begin{remark}[Order collapse and \texorpdfstring{$q$}{q}-axis ``non-complexity'']\label{rem:pom-moment-order-collapse-nonmonotone}
Within the verified range, the minimal order \(d_q\) does not grow monotonically with \(q\): for example \(d_9=7\) (Table \ref{tab:fold_collision_moment_recursions_9_17}), and moreover \(d_{13}=11<d_{12}=13\), \(d_{15}=11<d_{14}=13\) (same table).
This non-monotonic collapse is consistent with the modal constraint framework of Theorem \ref{thm:pom-moment-resonance}: raising the moment order does not necessarily introduce an equivalent number of new exponential modes; hidden symmetries and algebraic cancellations can force modal mergers, thereby reducing the Hankel-visible dimension.
Thus in the collision statistics of Fold, the Rényi order \(q\) and the minimal state complexity \(d_q\) are conceptually separable: higher-order moments may expose structural fingerprints of dimension reduction, rather than monotonically raising the ``state count.''
\end{remark}

\begin{algorithm}[Recursion polynomial as primitive spectral interface of moment-kernel (no matrix extraction)]\label{alg:pom-recurrence-polynomial-spectral-extraction}
For fixed \(q\ge 2\), given a finite sample \(\{S_q(m)\}\) (of length satisfying \(>\!2d_q\) redundancy), one can recover spectral fingerprints without constructing any realization matrix:
\begin{enumerate}
  \item Use the Hankel null-space/Prony--Padé equivalent interface to recover the minimal order \(d_q\) and the recursion coefficients \((c_{q,1},\dots,c_{q,d_q})\);
  \item Construct the recursion characteristic polynomial
  \[
  P_q(\lambda):=\lambda^{d_q}-\sum_{j=1}^{d_q}c_{q,j}\lambda^{d_q-j};
  \]
  \item Let \(r_q\) be the Perron root of \(P_q\), and let
  \(
  \Lambda_q:=\max\{|\mu|:\ \mu\in\mathrm{Roots}(P_q),\ \mu\ne r_q\}
  \),
  then \((r_q,\Lambda_q)\) yields the kernel-version RH fingerprint (\(\beta_q,\delta_q\), etc.; see Remark \ref{rem:pom-collision-kernel-rh-phase});
  \item For any prime \(p\), the factorization of \(\,P_q(\lambda)\bmod p\) yields a mod \(p\) fingerprint orthogonal to the real spectrum (e.g., the multiplicity staircase law for \((\lambda+1)\); see Remarks \ref{rem:pom-charpoly-mod2-staircase-q2-10} and \ref{rem:pom-charpoly-mod2-shadow}).
\end{enumerate}
Thus the recursion polynomial \(P_q\) can be viewed as a more primitive spectral carrier object: the matrix \(A_q\) is merely one realization thereof, not a necessary structure.
\end{algorithm}
\begin{remark}[Computable symmetric compression: \texorpdfstring{$O(q^2)$}{O(q^2)} orbit representation of the star fiber-product]\label{rem:pom-star-sympress}
The general compilation pipeline of Proposition \ref{prop:pom-moment-kernel-exists} encounters exponential blowup of the synchronous direct-product state count as \(q\) increases.
However, in a class of ``star-coupled'' situations (where the collision constraint can be organized as ``a central track and each leaf track separately satisfy the same pair constraint,'' and this pair constraint admits a \emph{central input symbol} decomposition---for example, the existence of a \(3\)-state pair kernel decomposition \(A_2=B_0+B_1\)),
the \(q\)-fold collision star kernel naturally possesses leaf permutation symmetry:
the uncompressed state space can be taken as \(\{1,2,3\}^{q-1}\), while the Perron principal eigenvalue is determined by the symmetric subspace, allowing compression to occupation-count states
\((n_1,n_2,n_3)\) (satisfying \(n_1+n_2+n_3=q-1\)) and reducing the dimension to
\(\binom{q+1}{2}\).
The script \texttt{scripts/exp\_fold\_collision\_star\_kernel\_sympress\_demo.py} provides an executable transition-counting interface for this ``orbit compression \(\Rightarrow\) \(\widetilde A_q\) \(\Rightarrow\) Perron root estimate'' pipeline (with input being the decomposition matrices \(B_0,B_1\)).
\end{remark}

\begin{remark}[Two-stage compression law: symmetric power quotient kernel \texorpdfstring{$+$}{+} Hankel minimization]\label{rem:pom-sympower-then-minh}
For fixed \(q\), the moment-kernel compilation reduces the \(q\)-fold collision constraint to a finite kernel \(A_q\) (Proposition \ref{prop:pom-moment-kernel-exists}), whose most naive realization corresponds to an exponential state space from the \(q\)-fold synchronous direct product.
Remark \ref{rem:pom-star-sympress} (and the more general symmetric power/orbit compression viewpoint, Remark \ref{rem:pom-moment-symmetric-power-substitution-operator}) shows that when the counting target has full leaf permutation symmetry, one can first take the quotient by the symmetric group action, compressing the exponential state space to polynomial-dimensional ``symmetric power quotient kernel'' \(\widetilde A_q\).
\par
Subsequently, on the sequence side one applies the Hankel minimal realization functor \(\mathsf{MinH}\) (Remark \ref{rem:pom-minh-functor}) to \(S_q(m)\), obtaining the truly visible minimal dimension \(d_q\) (Table \ref{tab:fold_collision_moment_recursions_9_17}).
This yields an executable two-stage compression interface:
\[
\boxed{\;
\text{Exponential direct-product kernel}\ \Longrightarrow\ \text{Symmetric power quotient kernel }\widetilde A_q\ \Longrightarrow\ \mathsf{MinH}(S_q)\ \text{minimal dimension }d_q\; }.
\]
When \(\Delta(q)=(2\lfloor q/2\rfloor+1)-d_q>0\) (Table \ref{tab:fold_collision_resonance_gap_delta_q_9_17}),
this gap can be viewed as the irreducible discrepancy between the ``nominal modal dimension of the symmetric power quotient kernel'' and the ``truly visible dimension of the sequence semantics'':
it corresponds to a set of resonance linear constraints forced by the Hankel rank drop (see Table \ref{tab:fold_collision_resonance_nullmodes_q9_17}), thereby compressing the ``multiplicative \(q\)-fold interaction'' at the auditable level into an integer gap certificate and a set of explicit constraint vectors.
\end{remark}

\begin{proposition}[Factorized symmetric power upper bound: multiplicative recursive control of minimal realization dimension]\label{prop:pom-factorized-sympower-upper-bound}
Following the notation of Definition \ref{def:pom-moment-q}, let \(S_q(m)=\sum_{x\in X_m}d_m(x)^q\).
For an integer \(q\ge 2\), let \(d_q\) denote the minimal integer-coefficient recursion order of the sequence \(\{S_q(m)\}_{m\ge 2}\) (equivalently, the Hankel minimal realization dimension; see Theorem \ref{thm:hankel-rank-minimal} and Corollary \ref{cor:pom-moment-dim-collapse}).
Take any integers \(q_1,q_2\ge 2\) and write \(q=q_1q_2\).
Then there is a deterministic upper bound
\[
\boxed{\;
d_{q}\ \le\ \binom{q_2+d_{q_1}-1}{d_{q_1}-1}\; }.
\]
Therefore, for any prime factorization \(q=\prod_{i=1}^{k}p_i\), recursive application along any bracketing order yields a family of explicit upper bounds.
Under the ``prime factor chain + stepwise minimization/compression'' compilation strategy (Remark \ref{rem:pom-mom-prime-factor-compiler}), this upper bound builds the nominal dimension of the subsequent symmetric power quotient upon the already-compressed intermediate realization;
when intermediate compression genuinely reduces dimension (e.g., when \(d_{q_1}\) is significantly smaller than the nominal symmetric power dimension of a single-pass compilation), one obtains complexity control strictly superior to the ``single-pass symmetric power upper bound.''
\end{proposition}
\begin{proof}
Take some minimal linear realization \((u,A,v)\) for \(q_1\) (Proposition \ref{prop:pom-hankel-realization-protocol}), of dimension \(d_{q_1}\) and satisfying
\(
S_{q_1}(m)=u^{\mathsf T}A^{m-m_0}v
\) (\(m\ge m_0\)).
View this realization as a finite-dimensional ``back-end kernel'' (weighted automaton/linear representation) and apply the \(q_2\)-fold moment compilation to it: take \(q_2\) synchronous copies and at each step apply the ``output-consistency'' local filter.
In the counting semantics of fold moments, this compilation is equivalent to taking the \(q_2\)-fold fiber product of the \(q_1\)-fold fiber product; by associativity of fiber products,
\(
\mathrm{MOM}_{q_2}\circ \mathrm{MOM}_{q_1}\cong \mathrm{MOM}_{q_1q_2}
\)
is strictly faithful in counting semantics, so the above construction indeed realizes \(S_{q_1q_2}(m)\).
\par
The \(q_2\)-fold synchronous direct-product kernel has copy permutation symmetry.
Taking the quotient by this symmetry (symmetric power quotient kernel) reduces the state space to occupation vectors (histograms); its cardinality is at most
\(\binom{q_2+d_{q_1}-1}{d_{q_1}-1}\), i.e., the symmetric power dimension upper bound at homogeneous degree \(q_2\).
By the definition of minimality, \(d_{q_1q_2}\) does not exceed the state dimension of any realization, whence the conclusion follows. This completes the proof.
\end{proof}

\paragraph{Symmetric power lifting and structural instability of kernel-version RH: first-variation amplification and excitation filtering thresholds}
Proposition \ref{prop:pom-factorized-sympower-upper-bound} provides a natural route of ``factorwise compilation \(\Rightarrow\) symmetric power quotient compression'': it is benign in dimension (polynomial states) but not conservative in spectral bounds.
The following result shows that, except in the spectral-rank-one degenerate case, the symmetric power lifting \(\mathrm{Sym}^b\) will amplify any non-Perron mode to eventually cross the square-root bound; thus if high-order moment-kernel construction relies solely on ``occupation-number symmetric power quotient,'' the kernel-version RH cannot be maintained under the multiplicative lifting \(b\to\infty\).

\begin{theorem}[Instability of the square-root spectral bound under symmetric power: eventual violation of kernel-version RH except for spectral rank one]\label{thm:pom-sympower-sqrt-bound-instability}
Let \(A\) be a primitive nonnegative matrix with Perron root \(\rho:=\rho(A)>1\), and write the non-Perron spectral radius
\[
\eta:=\max\bigl\{|\mu|:\ \mu\in\mathrm{spec}(A),\ \mu\ne \rho\bigr\}.
\]
For an integer \(b\ge 2\), write \(\mathrm{Sym}^b(A)\) for the induced operator on the \(b\)-th symmetric power representation (whose Perron root is \(\rho^b\)).
If \(\eta>0\), then there exists an integer \(b_0\ge 2\) such that for all \(b\ge b_0\),
\[
\max\bigl\{|\lambda|:\ \lambda\in\mathrm{spec}(\mathrm{Sym}^b(A)),\ \lambda\ne\rho^b\bigr\}\ >\ \sqrt{\rho^b},
\]
i.e., the symmetric power eventually violates the square-root spectral bound of kernel-version RH (the homologous condition of Definition \ref{def:kernel-rh}).
\end{theorem}
\begin{proof}
Take the Perron eigenvector \(u\) of \(A\) and an eigenvector \(v\) for an eigenvalue \(\mu\ne\rho\) satisfying \(|\mu|=\eta\).
Then on the tensor power \(A^{\otimes b}\), the vector \(u^{\otimes(b-1)}\otimes v\) is an eigenvector with eigenvalue \(\rho^{b-1}\mu\).
Symmetrizing it yields \(u^{\odot(b-1)}\odot v\), which is nonzero and belongs to the symmetric power subrepresentation, whence
\(\rho^{b-1}\mu\in\mathrm{spec}(\mathrm{Sym}^b(A))\).
Thus
\[
\max_{\lambda\ne\rho^b}|\lambda|\ \ge\ \rho^{b-1}\eta.
\]
On the other hand, \(\sqrt{\rho^b}=\rho^{b/2}\).
If \(\eta>0\), then \(\rho^{b-1}\eta>\rho^{b/2}\) is equivalent to \(\eta\,\rho^{b/2-1}>1\), which holds for sufficiently large \(b\), whence the conclusion follows. This completes the proof.
\end{proof}

\begin{corollary}[Sole stable class: kernel-version RH holds on all symmetric powers if and only if spectral rank-one degeneration]\label{cor:pom-sympower-sqrt-bound-stable-iff-spectral-rankone}
Under the setup of Theorem \ref{thm:pom-sympower-sqrt-bound-instability}, the following statements are equivalent:
\begin{enumerate}
  \item For all \(b\ge 2\), \(\mathrm{Sym}^b(A)\) satisfies the square-root spectral bound;
  \item \(\eta=0\);
  \item \(\mathrm{spec}(A)\subseteq\{\rho,0\}\) (the nonzero spectrum contains only the Perron root, spectral rank-one degeneration).
\end{enumerate}
\end{corollary}
\begin{proof}
\((2)\Leftrightarrow(3)\) follows by definition.
For \((2)\Rightarrow(1)\), when \(\eta=0\), the non-Perron spectrum of \(\mathrm{Sym}^b(A)\) is entirely zero, so the square-root spectral bound holds trivially.
For \((1)\Rightarrow(2)\), taking the contrapositive is given by Theorem \ref{thm:pom-sympower-sqrt-bound-instability}. This completes the proof.
\end{proof}

\begin{theorem}[Functorial kernel-version RH impossibility: multiplicatively closed Perron-faithful subchains have finite-order rupture except for spectral rank one]\label{thm:pom-functorial-kernel-rh-impossibility}
Let \(A\in\RR^{d\times d}\) be a primitive nonnegative matrix with Perron root \(\rho>1\), and take a Perron eigenvector \(u\succ0\).
For each integer \(b\ge 1\), let \(V_b\subseteq \mathrm{Sym}^b(\RR^d)\) be a subspace satisfying:
\begin{enumerate}
  \item \textbf{(Perron line fidelity)} \(u^{\odot b}\in V_b\);
  \item \textbf{(Multiplicative closure)} For all \(b\ge 2\) and all \(v\in V_1\), \(u^{\odot(b-1)}\odot v\in V_b\);
  \item \textbf{(Invariance)} \(V_b\) is invariant under \(\mathrm{Sym}^b(A)\). Write the restriction operator \(B_b:=\mathrm{Sym}^b(A)|_{V_b}\), and write
  \[
  \eta(B_b):=\max\bigl\{|\lambda|:\ \lambda\in\mathrm{spec}(B_b),\ \lambda\ne \rho(B_b)\bigr\}.
  \]
\end{enumerate}
Then if \(\eta(B_1)>0\) (i.e., \(B_1\) is not spectral-rank-one degenerate), there exists a finite \(b_0\) such that \(B_{b_0}\) violates the square-root spectral bound of kernel-version RH:
\[
\eta(B_{b_0})\ >\ \rho(B_{b_0})^{1/2}=\rho^{b_0/2}.
\]
Equivalently: if for all \(b\ge 1\) we have \(\eta(B_b)\le \rho(B_b)^{1/2}\), then necessarily \(\eta(B_1)=0\), whence \(B_1\) spectrally degenerates to Perron line \(\oplus\) zero spectrum.
\end{theorem}
\begin{proof}
From \(u^{\odot b}\in V_b\) and the invariance of \(V_b\), we know \(u^{\odot b}\) is an eigenvector of \(B_b\) with eigenvalue \(\rho^b\), hence \(\rho(B_b)=\rho^b\).
\par
If \(\eta(B_1)>0\), take some eigenvalue \(\mu\ne\rho\) of \(B_1\) satisfying \(|\mu|=\eta(B_1)\), with corresponding eigenvector \(v\in V_1\).
By multiplicative closure, \(u^{\odot(b-1)}\odot v\in V_b\).
Moreover,
\[
\mathrm{Sym}^b(A)\bigl(u^{\odot(b-1)}\odot v\bigr)
=(Au)^{\odot(b-1)}\odot(Av)
=(\rho u)^{\odot(b-1)}\odot(\mu v)
=\rho^{b-1}\mu\,(u^{\odot(b-1)}\odot v),
\]
whence \(\rho^{b-1}\mu\in\mathrm{spec}(B_b)\).
Thus
\[
\eta(B_b)\ \ge\ \rho^{b-1}|\mu|=\rho^{b-1}\eta(B_1).
\]
Comparing with \(\rho(B_b)^{1/2}=\rho^{b/2}\), we see
\(
\eta(B_b)>\rho^{b/2}
\)
if and only if \(\eta(B_1)\rho^{b/2-1}>1\), which holds for sufficiently large \(b\).
Taking any \(b_0\) satisfying this inequality yields the conclusion. This completes the proof.
\end{proof}

\begin{definition}[First-variation sector and faithful compilation]\label{def:pom-first-variation-sector}
Let \(A\) be a primitive nonnegative matrix with Perron root \(\rho\), and take a Perron eigenvector \(u\).
For an integer \(b\ge 2\), define the \textbf{first-variation sector} in the symmetric power:
\[
\mathcal{E}^{(1)}_{b}(A):=
\mathrm{span}\Bigl\{u^{\odot(b-1)}\odot v:\ v\in \ker(A-\mu I),\ \mu\in\mathrm{spec}(A)\setminus\{\rho\}\Bigr\}.
\]
Here \(\odot\) denotes the symmetric tensor product (equivalently: the product after taking the symmetrization projection on the tensor product).
If there exists a surjective intertwiner \(\Pi:\mathrm{Sym}^b(A)\twoheadrightarrow B\) such that \(\Pi\circ \mathrm{Sym}^b(A)=B\circ\Pi\),
then this compilation is called \textbf{first-variation faithful} at order \(b\) if \(\Pi\) is not identically zero on \(\mathcal{E}^{(1)}_{b}(A)\).
\end{definition}

\begin{theorem}[First-variation fidelity \texorpdfstring{$\Rightarrow$}{=>} high-order kernel-version RH impossibility: fatal amplification under MOM multiplicative lifting]\label{thm:pom-first-variation-fidelity-rh-breaking}
Fix an integer \(a\ge 1\).
Let \(A_a\) be a primitive nonnegative matrix, write \(\rho_a:=\rho(A_a)>1\) and \(\eta_a:=\max_{\mu\ne\rho_a}|\mu|>0\).
For each integer \(b\ge 2\), suppose there exists a surjective intertwiner
\[
\Pi_{a,b}:\mathrm{Sym}^b(A_a)\twoheadrightarrow A_{ab}
\]
satisfying \(\Pi_{a,b}\circ \mathrm{Sym}^b(A_a)=A_{ab}\circ \Pi_{a,b}\), and suppose this compilation is first-variation faithful at order \(b\) (Definition \ref{def:pom-first-variation-sector}).
Then there exists an integer \(b_0\ge 2\) such that for all \(b\ge b_0\) (satisfying fidelity), we have
\[
\max\bigl\{|\lambda|:\ \lambda\in\mathrm{spec}(A_{ab}),\ \lambda\ne\rho(A_{ab})\bigr\}\ >\ \sqrt{\rho(A_{ab})},
\]
i.e., the \(ab\)-body kernel necessarily violates the square-root spectral bound of kernel-version RH.
\end{theorem}
\begin{proof}
Take the Perron eigenvector \(u_a\) of \(A_a\). By first-variation fidelity, there exists an eigenvector \(v\) for \(\mu\in\mathrm{spec}(A_a)\setminus\{\rho_a\}\) such that
\(\Pi_{a,b}(u_a^{\odot(b-1)}\odot v)\ne 0\).
By the intertwiner relation, this image vector is an eigenvector of \(A_{ab}\) with eigenvalue \(\rho_a^{b-1}\mu\).
Thus
\[
\max_{\lambda\ne\rho(A_{ab})}|\lambda|
\ \ge\
|\rho_a^{b-1}\mu|
\ \ge\
\rho_a^{b-1}\eta_a.
\]
On the other hand, in the moment-counting framework of this paper, we always have \(S_{ab}(m)\le S_a(m)^b\), whence the exponential rate satisfies
\(\rho(A_{ab})\le \rho(A_a)^b=\rho_a^b\).
Thus
\(
\sqrt{\rho(A_{ab})}\le \rho_a^{b/2}
\).
Altogether,
\[
\frac{\max_{\lambda\ne\rho(A_{ab})}|\lambda|}{\sqrt{\rho(A_{ab})}}
\ \ge\
\eta_a\,\rho_a^{b/2-1}\ \xrightarrow[b\to\infty]{}\ +\infty,
\]
hence for sufficiently large \(b\) this is necessarily greater than \(1\), yielding the stated crossing. This completes the proof.
\end{proof}

\begin{theorem}[Linear excitation filtering lower bound: to maintain kernel-version RH on symmetric power subquotients, one must eliminate a linear proportion of low-excitation sectors]\label{thm:pom-excitation-filter-linear-bound}
Fix a primitive nonnegative \(A\), write \(\rho=\rho(A)>1\), \(\eta=\max_{\mu\ne\rho}|\mu|\in(0,\rho)\).
For each \(b\ge 2\), let \(B_b\) be some nonzero subquotient of \(\mathrm{Sym}^b(A)\), and write \(\rho(B_b)\le \rho^b\).
Define the ``minimal surviving excitation count'' \(k_b\) to be the smallest integer satisfying the property: the spectrum of \(B_b\) contains an eigenvalue \(\lambda\) such that
\[
|\lambda|\ \ge\ \rho^{\,b-k_b}\eta^{\,k_b}.
\]
If \(B_b\) satisfies kernel-version RH (i.e., its non-Perron spectral radius \(\le \sqrt{\rho(B_b)}\)), then necessarily
\[
\boxed{\;
k_b\ \ge\ \frac{b}{2}\cdot \frac{\log\rho}{\log(\rho/\eta)}\; }.
\]
In particular, \(k_b=\Omega(b)\).
Moreover, in the RH-sharp case \(\eta=\sqrt{\rho}\), the above formula forces \(k_b\ge b\), i.e., one must eliminate all non-Perron excitations (leaving only the Perron line).
\end{theorem}
\begin{proof}
By the definition of \(k_b\), there exists a surviving mode satisfying \(|\lambda|\ge \rho^{b-k_b}\eta^{k_b}\).
Kernel-version RH requires \(|\lambda|\le \sqrt{\rho(B_b)}\le \rho^{b/2}\), hence
\[
\rho^{b-k_b}\eta^{k_b}\ \le\ \rho^{b/2}.
\]
Simplifying yields \((\eta/\rho)^{k_b}\le \rho^{-b/2}\).
Taking logarithms and noting \(\log(\eta/\rho)=-\log(\rho/\eta)<0\), we obtain the stated lower bound.
The RH-sharp case substitutes \(\log(\rho/\eta)=\tfrac12\log\rho\) to get \(k_b\ge b\). This completes the proof.
\end{proof}

\begin{theorem}[Occupation-number symmetric power quotient alone \texorpdfstring{$\Rightarrow$}{=>} high-order RH necessarily ruptures; infinite kernel RH \texorpdfstring{$\Rightarrow$}{=>} linear excitation filtering selector family]\label{thm:pom-occupancy-symquotient-rh-obstruction}
Fix \(a\ge 1\) and suppose \(A_a\) is primitive, \(\rho(A_a)>1\) and \(\eta(A_a)>0\).
\begin{enumerate}
  \item \textbf{(Occupation-number symmetric power quotient alone \(\Rightarrow\) high-order RH necessarily ruptures)}
  If the realization of \(A_{ab}\) is given solely by the ``occupation-number symmetric power quotient'' (i.e., only taking quotient by permutation symmetry, without introducing additional \(b\)-dependent projection/selector subfamilies to eliminate low-excitation sectors),
  then there exists \(b_0\) such that for all \(b\ge b_0\), \(A_{ab}\) necessarily violates the square-root spectral bound of kernel-version RH.
  \item \textbf{(Infinite kernel RH \(\Rightarrow\) linear excitation filtering selector family)}
  If kernel-version RH is required to hold simultaneously for all \(b\ge 2\) with fixed \(a\), then unless \(\eta(A_a)=0\) (spectral rank-one degeneration), one must introduce a family of selectors/projections \(P_{a,b}\) growing with \(b\), such that in \(\mathrm{Sym}^b(A_a)\) at least all sectors with excitation count \(<k_b\) are eliminated, where \(k_b\) satisfies the linear lower bound of Theorem \ref{thm:pom-excitation-filter-linear-bound}.
  In protocol semantics, this is equivalent to appending an external ``selector axis'' specifying \(b\)-dependent filtering thresholds and retained components (i.e., an oracle-ized compilation layer).
\end{enumerate}
\end{theorem}
\begin{proof}
Point 1: In an implementation using only the occupation-number symmetric power quotient, \(A_{ab}\) is given by the standard quotient representation (occupation-number/histogram state space) of \(\mathrm{Sym}^b(A_a)\), hence there exists a natural quotient map as a surjective intertwiner.
This quotient map is nonzero on the first-variation vector \(u_a^{\odot(b-1)}\odot v\) (for \(v\) any non-Perron eigenvector), so first-variation fidelity holds for infinitely many (in fact, for all) \(b\). By Theorem \ref{thm:pom-first-variation-fidelity-rh-breaking}, high-order crossing occurs.
\par
Point 2: If kernel-version RH holds simultaneously for all \(b\), then any visible/surviving non-Perron mode must satisfy the linear threshold of Theorem \ref{thm:pom-excitation-filter-linear-bound}; hence one must introduce filtering projections growing with \(b\) to eliminate all low-excitation sectors.
When \(b\) is unbounded, the threshold \(k_b\) grows without bound, so this projection family carries external selector information growing with scale; in the audit protocol language of this paper, it can be viewed as an oracle-ized compilation layer. This completes the proof.
\end{proof}

\paragraph{From \texorpdfstring{$r_q$}{r\_q} to pressure and the ``fiber multiplicity spectrum'': strict convex-conjugate upper bound}
The verified \(q\)-th order collision kernels satisfy \(S_q(m)=\Theta(r_q^m)\), where \(r_q=\rho(A_q)\) is the Perron root (Corollaries \ref{cor:pom-s2-asymptotic}, \ref{cor:pom-s3-asymptotic}, \ref{cor:pom-s4-asymptotic} and Table \ref{tab:fold_collision_moment_recursions}).
Therefore the moment fingerprint not only yields the single-point constant \(h_q=\log(2^q/r_q)\) but also induces a ``pressure curve'' amenable to Legendre transformation, thereby promoting ``spatial loss'' to a genuine spectral structure.

\begin{definition}[Moment pressure and normalized log-moment generating function]\label{def:pom-moment-pressure}
For \(q\ge 2\) define
$$
\tau(q):=\limsup_{m\to\infty}\frac{1}{m}\log S_q(m),
$$
and define the normalized pressure (i.e., the \(q\)-th order log-moment generating function of the output distribution \(w_m(x)=d_m(x)/2^m\))
$$
P(q):=\limsup_{m\to\infty}\frac{1}{m}\log\sum_{x\in X_m}w_m(x)^q
=\tau(q)-q\log 2.
$$
When there exists a Perron root \(r_q\) such that \(S_q(m)=\Theta(r_q^m)\), one has \(\tau(q)=\log r_q\), and \(P(q)=\log(r_q/2^q)=-h_q\).
\end{definition}
