\begin{proposition}[Dirichlet-to-Neumann equivalence: \texorpdfstring{$|D|$}{|D|} is the generator of the horizon coarse-graining semigroup]\label{con:discussion-dtn-generator}
Let
\[
P_t(x):=\frac{1}{\pi}\frac{t}{x^2+t^2}\qquad(t>0)
\]
be the Poisson kernel on \(\RR\), and let the coarse-graining semigroup be \(T_t f:=P_t*f\).
For \(f\in\mathcal{S}(\RR)\), define the upper half-plane harmonic extension
\(
u_f(x,y):=(P_y*f)(x)
\).
Then the Dirichlet-to-Neumann identity holds:
\[
\boxed{\;
\partial_y u_f(x,0)=-(|D|f)(x),\qquad \widehat{|D|f}(\xi)=|\xi|\,\widehat f(\xi)\;}
\]
and therefore
\[
\boxed{\;T_t=\exp(-t|D|)\qquad(t>0).\;}
\]
Moreover, within the class of \emph{translation-invariant, Markov, and first-order homogeneous} strongly continuous convolution semigroups, the generator symbol must be \(c|\xi|\) (\(c\ge 0\)), so after time-scale normalization, \( |D| \) (equivalently DtN) is the unique generator; therefore the dissipation law and irreversible ``recycling'' induced by Poisson coarse-graining are not accidental features of the implementation but are forced by the unique dynamics of the horizon geometry.
\end{proposition}
\begin{proof}[Proof sketch]
From \(\widehat{P_y}(\xi)=e^{-y|\xi|}\), we get \(\widehat u_f(\xi,y)=e^{-y|\xi|}\widehat f(\xi)\); differentiating in \(y\) and setting \(y\downarrow 0\) yields \(\partial_y u_f(\cdot,0)=-|D|f\);
the semigroup property then gives \(T_t=\exp(-t|D|)\).
For uniqueness: a translation-invariant Markov semigroup has Fourier-side multiplier \(e^{-t\psi(\xi)}\) (L\'evy--Khintchine, \(\psi\) continuous negative definite with \(\psi(0)=0\)); first-order homogeneity gives \(\psi(\lambda\xi)=\lambda\psi(\xi)\), hence \(\psi(\xi)=c|\xi|\). This completes the proof.
\end{proof}

\begin{proposition}[Fourier--Laplace fingerprint of a two-dimensional defect measure: invertible tomography formula for general \texorpdfstring{$\nu$}{nu}]\label{con:discussion-defect-measure-fourier-laplace}
Let \(\nu\) be a finite positive measure on \(\RR\times(0,\infty)\), and define the visible boundary density
\[
H_\nu(x):=\int_{\RR\times(0,\infty)}\frac{4\delta}{(x-\gamma)^2+(1+\delta)^2}\,\dd\nu(\gamma,\delta),
\qquad
H_{\nu,t}:=P_t*H_\nu.
\]
Then its Fourier transform satisfies the strict decomposition
\[
\boxed{\;
\widehat{H_{\nu,t}}(\xi)
=4\pi\,e^{-t|\xi|}
\int_{\RR\times(0,\infty)}
\frac{\delta}{1+\delta}\,
e^{-(1+\delta)|\xi|}\,e^{-\mathrm{i}\gamma\xi}\,\dd\nu(\gamma,\delta).\;}
\]
Letting \(s:=|\xi|\) and defining the two-parameter fingerprint
\[
F(\xi,s):=e^{(t+1)s}\widehat{H_{\nu,t}}(\xi),
\]
we have
\[
\boxed{\;
F(\xi,s)
=4\pi\int_{\RR\times(0,\infty)}\frac{\delta}{1+\delta}\,e^{-s\delta}\,e^{-\mathrm{i}\gamma\xi}\,\dd\nu(\gamma,\delta).\;}
\]
Therefore \(F\) is the tensorized combination of the Fourier transform in \(\gamma\) and the Laplace transform in \(\delta\); in particular, the map \(\nu\mapsto F\) is injective on the class of finite measures, so \(\nu\) can be uniquely recovered from \(F\) (evaluated on \(\RR\times(0,\infty)\)).
\end{proposition}
\begin{proof}[Proof sketch]
For fixed \((\gamma,\delta)\), the kernel \(\frac{4\delta}{(x-\gamma)^2+(1+\delta)^2}\) has Fourier transform \(4\pi\frac{\delta}{1+\delta}e^{-(1+\delta)|\xi|}e^{-\mathrm{i}\gamma\xi}\) (from \(\int_{\RR}\frac{e^{-\mathrm{i}\xi t}}{t^2+a^2}\dd t=\frac{\pi}{a}e^{-a|\xi|}\)).
Applying the Poisson semigroup multiplier \(\widehat{P_t}(\xi)=e^{-t|\xi|}\) gives the first formula; multiplying by \(e^{(t+1)|\xi|}\) gives the second.
Injectivity follows from the uniqueness of the Fourier transform and Laplace transform on finite measures. This completes the proof.
\end{proof}

\begin{proposition}[Strong converse for the index gap: sublinear leakage limit forces index saturation]\label{con:discussion-index-gap-strong-converse}

\Gamma:=\log\frac{\mathrm{Ind}(E_1)\mathrm{Ind}(E_2)}{\mathrm{Ind}(E_0)}\ \ge\ 0.
\]

\[
I\bigl(\mathcal{B}_1^{\otimes n}:\mathcal{B}_2^{\otimes n}\mid\mathcal{B}_0^{\otimes n}\bigr)=o(n),
\]
then necessarily \(\Gamma=0\), i.e., index saturation holds.
\end{proposition}
\begin{proof}[Proof sketch]
By Appendix Corollary \ref{cor:op-algebra-cmi-lower-bound-by-index-gap},
\(
I(\mathcal{B}_1:\mathcal{B}_2\mid\mathcal{B}_0)\ge \Gamma
\).
For tensor powers, the index and logarithmic index gap tensorize multiplicatively/additively, yielding the linear lower bound
\(
I(\mathcal{B}_1^{\otimes n}:\mathcal{B}_2^{\otimes n}\mid\mathcal{B}_0^{\otimes n})\ge n\Gamma
\).
This contradicts the \(o(n)\) assumption, hence \(\Gamma=0\). This completes the proof.
\end{proof}

\begin{proposition}[Randomness cost lower bound: the external randomness needed for delinking is at least the index gap]\label{con:discussion-randomness-cost-lower-bound}

\boxed{\;H(R)\ \ge\ \Gamma-\varepsilon.\;}
\]
Therefore, the index gap provides not only a lower bound on the non-cancellable linking residual, but also the minimal randomness cost that must be paid to suppress the residual to \(\varepsilon\).

I(\mathcal{B}_1:\mathcal{B}_2\mid\mathcal{B}_0)
\le I(\mathcal{B}_1:\mathcal{B}_2,R\mid\mathcal{B}_0)

\),

  \mathrm{View}^{\Pi}(w)\ \equiv\ \mathrm{View}^{\Pi}(w').
  \]
  \item \textbf{(Information-theoretic zero-knowledge/simulability)} \(\Pi\) satisfies horizon simulability at order \(N\): there exists a simulator \(\mathsf{S}_N\) such that \(\mathrm{View}^{\Pi}(w)\equiv \mathsf{S}_N(H_N(w))\) for all \(w\).
\end{enumerate}
In this sense, ``all information the verifier can learn'' is strictly equal to the information carried by \(H_N\), independent of the remaining degrees of freedom of the bulk witness; this is of the same type as the regime statement ``visible information = horizon hair prefix'' in the no-hair caliber of this paper.
\end{proposition}

\begin{proof}
\((2)\Rightarrow(1)\) follows immediately from \(H_N(w)=H_N(w')\Rightarrow \mathsf{S}_N(H_N(w))\equiv \mathsf{S}_N(H_N(w'))\).
\par
\((1)\Rightarrow(2)\) Let \(c\in\CC^{N+1}\).
If there exists \(w_c\in\mathcal{W}\) with \(H_N(w_c)=c\), then define \(\mathsf{S}_N(c)\) as the distribution of \(\mathrm{View}^\Pi(w_c)\);
by (1), this definition is independent of the choice of \(w_c\), so \(\mathsf{S}_N\) is well-defined.
If no witness exists with \(H_N(w)=c\), then assign any distribution as \(\mathsf{S}_N(c)\) (this value is never actually invoked).
Then for all \(w\), \(\mathrm{View}^\Pi(w)\equiv \mathsf{S}_N(H_N(w))\). This completes the proof.
\end{proof}

\begin{proposition}[Toeplitz--PSD threshold law with a single-atom negative defect]\label{prop:discussion-toeplitz-negative-atom-threshold}
Let \(\mu_+\) be a positive measure on \(\TT\), and assume it is absolutely continuous with respect to the Haar measure \(\dd\theta/(2\pi)\) with density satisfying
\[
m_0\ \le\ \frac{\dd\mu_+}{\dd\theta/(2\pi)}\ \le\ M_0
\qquad\text{a.e.\ on }\TT,
\]
where \(0<m_0\le M_0<\infty\).
Take \(\xi_0\in\TT\) and \(\varepsilon>0\), and let the signed effective measure be
\[
\mu_{\mathrm{eff}}:=\mu_+-\varepsilon\,\delta_{\xi_0}.
\]
Denoting its moments as
\(
c_n:=\int_{\TT}\xi^{-n}\,\dd\mu_{\mathrm{eff}}(\xi)
\)
and the corresponding Toeplitz principal submatrix as
\[
\mathbf{T}_N:=\bigl(c_{j-k}\bigr)_{0\le j,k\le N}.
\]
Then there exists an explicit PSD failure threshold:
\begin{enumerate}
  \item If \(\varepsilon\le \dfrac{m_0}{N+1}\), then \(\mathbf{T}_N\succeq 0\) (the single-atom negative defect can be absorbed by the background at order \(N\)).
  \item If \(\varepsilon> \dfrac{M_0}{N+1}\), then \(\mathbf{T}_N\not\succeq 0\) (the single-atom negative defect is forced to be exposed at order \(N\)).
\end{enumerate}
Therefore, in this minimal defect model, the minimal order required for \(\mathbf{T}_N\) to necessarily fail satisfies the dimensional scaling
\[
N_{\min}\ \asymp\ \varepsilon^{-1}.
\]
\end{proposition}

\begin{proof}
For any vector \(x=(x_0,\dots,x_N)^{\mathsf T}\in\CC^{N+1}\), let the trigonometric polynomial be
\(
p(\xi):=\sum_{k=0}^{N}x_k\xi^k
\).
Then by the standard identity
\[
x^\ast \mathbf{T}_N x
=\int_{\TT}|p(\xi)|^2\,\dd\mu_{\mathrm{eff}}(\xi)
=\int_{\TT}|p(\xi)|^2\,\dd\mu_+(\xi)-\varepsilon\,|p(\xi_0)|^2.
\]
By the density lower bound and Parseval's identity,
\[
\int_{\TT}|p(\xi)|^2\,\dd\mu_+(\xi)
\ge m_0\int_{\TT}|p(\xi)|^2\frac{\dd\theta}{2\pi}
=m_0\sum_{k=0}^{N}|x_k|^2
=m_0\|x\|_2^2.
\]
On the other hand, by Cauchy--Schwarz,
\(
|p(\xi_0)|
\le \sum_{k=0}^{N}|x_k|
\le \sqrt{N+1}\,\|x\|_2
\), hence
\[
x^\ast \mathbf{T}_N x
\ge \bigl(m_0-\varepsilon(N+1)\bigr)\|x\|_2^2.
\]
Therefore when \(\varepsilon\le m_0/(N+1)\), the right side is nonneg for all \(x\), so \(\mathbf{T}_N\succeq 0\).
\par
To prove the second claim, take \(x=u\) where \(u_k:=\xi_0^{-k}\) (\(0\le k\le N\)), so \(\|u\|_2^2=N+1\) and \(p(\xi_0)=\sum_{k=0}^{N}u_k\xi_0^k=N+1\).
By the density upper bound,
\[
u^\ast \mathbf{T}_N u
=\int_{\TT}|p(\xi)|^2\,\dd\mu_+(\xi)-\varepsilon|p(\xi_0)|^2
\le M_0\|u\|_2^2-\varepsilon(N+1)^2
=(N+1)\bigl(M_0-\varepsilon(N+1)\bigr).
\]
When \(\varepsilon> M_0/(N+1)\), this is negative, so there exists a direction making the quadratic form negative, hence \(\mathbf{T}_N\not\succeq 0\). This completes the proof.
\end{proof}

\begin{remark}[Hair \texorpdfstring{$=$}{=} verifiable invariant; microstate \texorpdfstring{$=$}{=} zero-knowledge witness; \texorpdfstring{$\varepsilon$}{eps} as unified interface parameter]\label{rem:discussion-hair-microstate-eps-interface}
Under the order-\(N\) horizon observable constraint, \(H_N(w)\) provides the ``hair layer'' (verifiable invariant prefix accessible to the verifier), while the fibers of \(H_N\) provide the ``microstate layer'' (remaining degrees of freedom of the bulk witness); Proposition \ref{prop:discussion-nohair-iff-zk} shows: any regime depending only on the hair layer automatically satisfies simulability in the information-theoretic sense, so that microstates constitute zero-knowledge witnesses under this caliber.
\par
Proposition \ref{prop:discussion-toeplitz-negative-atom-threshold} further provides a verifiable infeasibility bound: when all potential off-slice defects satisfy \(\varepsilon\ll (N+1)^{-1}\), the single-atom negative defect can be absorbed by the background within the \(\mathbf{T}_N\) hierarchy, and any verification based solely on order-\(N\) horizon certificates cannot force the defect to be distinguished.
Therefore \(\varepsilon\) (or equivalently \(h(\rho)\) under Riemann zero semantics) becomes the unified interface parameter connecting all three: it simultaneously characterizes the off-slice depth, controls the Toeplitz hierarchy failure order, and provides the minimal information budget scale required for ``falsifiability under simulability semantics'' (see Corollary \ref{cor:discussion-zk-area-law-gamma2-delta}).
\end{remark}

\begin{corollary}[Distinguishability area law under endpoint quadratic compression: \texorpdfstring{$h(\rho)^{-1}$}{h(rho)^(-1)} as the minimal order budget]\label{cor:discussion-zk-area-law-gamma2-delta}
In the fixed Cayley chart, let the disk image of an off-critical-line zero \(\rho=\beta+\mathrm{i}\gamma\) be \(w_\rho=\mathcal{C}(\gamma+\mathrm{i}\delta)\in\mathbb{D}\) (\(\delta:=\tfrac12-\beta>0\)), and define its horizon depth as
\[
h(\rho):=1-|w_\rho|^2=\frac{4\delta}{\gamma^2+(1+\delta)^2}
\qquad\text{(Proposition \ref{prop:app-offcritical-radius-compression})}.
\]
Then when \(|\gamma|\to\infty\) and \(\delta\) is bounded,
\[
h(\rho)\ \asymp\ \frac{\delta}{\gamma^2},
\qquad
h(\rho)^{-1}\ \asymp\ \frac{\gamma^2}{\delta}.
\]
Under the abstraction of coarse-graining ``off-slice pointwise information'' on the horizon as a minimal effective defect mass \(\varepsilon\asymp h(\rho)\),
Proposition \ref{prop:discussion-toeplitz-negative-atom-threshold} provides the auditable necessary budget:
if only the order-\(N\) horizon multipole moment prefix \(H_N\) is accessible, then for an off-slice offset \(\delta\) at height \(|\gamma|\), the minimal detectable order in the Toeplitz--PSD hierarchy sense satisfies
\[
N_{\min}(\rho)\ \asymp\ h(\rho)^{-1}\ \asymp\ \frac{\gamma^2}{\delta}.
\]
A more refined detection threshold within the framework (including residual audit and endpoint focusing logarithmic correction) is given by Proposition \ref{prop:xi-extreme-resonance-toeplitz-threshold}:
\(
N\gtrsim Q_\rho\log(1+\delta Q_\rho)
\)
where \(Q_\rho=h(\rho)^{-1}\).
\end{corollary}

\begin{remark}[Zero-knowledge compilation of the Toeplitz/Schur structure: Verblunsky prefix as a pure-horizon witness]\label{rem:discussion-toeplitz-zk-compile}
Within the horizon certificate language of this paper, the feasibility of \(\mathbf{T}_N\succeq 0\) is not merely a linear matrix inequality, but naturally carries a low-dimensional parametrized witness: when \(\mathbf{T}_N\succ 0\), there exists a unique Verblunsky/Schur parameter prefix \((\alpha_0,\dots,\alpha_{N-1})\in\mathbb{D}^{N}\) such that the Toeplitz determinant and failure step can be localized by a single scalar threshold crossing (Theorem \ref{thm:xi-toeplitz-det-verblunsky}), and this prefix can be explicitly generated from finite-order local jet data (Theorem \ref{thm:xi-verblunsky-from-local-jet}).
\par
Therefore, for the public statement ``there exists some bulk witness \(w\) such that \(H_N(w)=c^{(N)}\) and \(\mathbf{T}_N(c^{(N)})\succeq 0\),'' one can take a \emph{pure-horizon object} (e.g., the \((\alpha_k)\) prefix and its recursive consistency) as the witness carrier, without exposing the bulk microscopic structure that produced it.
From an algorithmic structure perspective, the one-dimensional convolution geometry of Toeplitz/Schur makes ``recursive consistency + local feasibility constraints'' compilable into finitely many convolution-type polynomial identities and a few norm inequalities, achievable at quasilinear verification cost in the Toeplitz multiplication (FFT) sense; on this basis, if standard commitment primitives and classical zero-knowledge compilation frameworks are introduced, one obtains a statistical zero-knowledge argument whose verifier's acquired information is strictly limited to the public \(c^{(N)}\) (i.e., the horizon hair), thereby perfectly aligning with the ``observable completeness \(\Leftrightarrow\) simulability'' semantics of Proposition \ref{prop:discussion-nohair-iff-zk}.
\end{remark}

\begin{conjecture}[Finite-rank negative defect decomposition template (directly refutable)]\label{conj:discussion-finite-rank-negative-defect}
If RH is false, then within this paper's disk-completion horizon interface, the horizon positivity failure of \(\zeta\) (Carath\'eodory failure, equivalently Toeplitz--PSD hierarchy failure at some level) can at any fixed resolution slice be approximated as a finite-rank negative defect decomposition:
\[
\mu_{\mathrm{eff}}=\mu_+ - \sum_{j=1}^r \varepsilon_j\,\delta_{\xi_j},
\qquad
\mu_+\ge 0,\quad r<\infty,\quad \varepsilon_j>0,
\]
where each \(\varepsilon_j\) is of the same order as the horizon depth of some off-slice zero cluster (under the fixed chart, \(\varepsilon_j\asymp h(\rho_j)\)).
Moreover, for the deepest among the defect clusters \(\rho_\star\), the minimal moment order required for Toeplitz--PSD failure satisfies the universal lower bound

N_{\min}\ \gtrsim\ h(\rho_\star)^{-1}\ \asymp\ \frac{\gamma_\star^2}{\delta_\star}.
\]
The refutation caliber is explicit: once an RH counterexample can be constructed where the horizon positivity failure cannot be approximated by any finite-rank negative defect model (e.g., an infinite-rank negative spectral measure is essentially required), the template fails; conversely, if the template holds, then the area law lower bound derived from endpoint quadratic compression becomes a structural consequence of any general RH violation.
\end{conjecture}

\begin{definition}[Horizon falsification complexity invariant]\label{def:discussion-horizon-falsification-complexity}
For any bulk property \(\mathcal{P}\) (whose truth value can be expressed in the horizon certificate language of this paper), define its \textbf{horizon falsification complexity} as
\[
\mathfrak{C}(\mathcal{P})
:=\inf\Bigl\{N:\ \exists\ \text{an auditable certificate chain depending only on }H_N\text{ sufficient to refute }\mathcal{P}\Bigr\}.
\]
In particular, when \(\mathcal{P}\) is taken to be ``the horizon Toeplitz--PSD full hierarchy holds'' (i.e., the horizon formulation of RH; Appendix Theorem \ref{thm:app-horizon-toeplitz-lmi}), \(\mathfrak{C}(\mathcal{P})\) measures the minimal falsification order: it is mathematically equivalent to the localization problem for the minimal failure order (Theorem \ref{thm:xi-toeplitz-det-verblunsky}), and under the simulability semantics of Proposition \ref{prop:discussion-nohair-iff-zk} simultaneously serves as the counting proxy for ``the information budget required for falsifiability.''
\end{definition}
