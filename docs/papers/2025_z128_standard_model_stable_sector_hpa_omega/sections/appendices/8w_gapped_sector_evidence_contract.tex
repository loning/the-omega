\subsection{Gapped-sector evidence contract (protocol spectral gap $\Rightarrow$ scattering prerequisites)}
\label{app:gapped_sector_evidence_contract}

\AuditTag This appendix connects the scattering condition bundle (Appendix~\ref{app:scattering_haag_ruelle_theorems}) to auditable finite diagnostics already produced by the paper's reproducible pipeline.
It does \emph{not} prove a 4D mass gap; instead, it defines a checkable evidence contract for when the manuscript is allowed to treat a sector as effectively gapped/stable for theorem-facing scattering statements.

\subsubsection{Protocol spectral-gap witness (finite diagnostic)}
\paragraph{Diagnostic input.}
Appendix~\ref{app:generated_tables} contains the spectral-gap/mixing diagnostic for the protocol RG operator $F_n$ along the balanced chain; see Table~\ref{tab:kernel_rg_operator_spectral_gap}.

\paragraph{Witness definition.}
\begin{definition}[Protocol spectral-gap witness]
\label{def:protocol_spectral_gap_witness}
\MathTag Fix a window of resolutions $n\in\{n_{\min},\dots,n_{\max}\}$.
We say the protocol RG operator exhibits a \emph{spectral-gap witness} at threshold $\gamma_\star>0$ on that window if
\[
1-|\lambda_2(F_n)|\ \ge\ \gamma_\star
\qquad\text{for all }n\in\{n_{\min},\dots,n_{\max}\},
\]
with $|\lambda_2(F_n)|$ taken from the deterministic diagnostic in Table~\ref{tab:kernel_rg_operator_spectral_gap}.
\end{definition}

\subsubsection{Contract: from witness to scattering prerequisites}
\begin{assumption}[Gapped-sector evidence contract (protocol \texorpdfstring{$\Rightarrow$}{=>} scattering)]
\label{ass:gapped_sector_evidence_contract}
Assume:
\begin{itemize}
\item a protocol spectral-gap witness holds on the declared window (Definition~\ref{def:protocol_spectral_gap_witness}),
\item microcausality is discharged in the declared protocol subclass (e.g.\ Appendix~\ref{app:protocol_subclass_tensor_net}),
\item a covariant vacuum representation with a spectrum condition is declared (Appendix~\ref{app:covariance_spectrum_from_protocol_dynamics}).
\end{itemize}
Under this contract, the manuscript is allowed to treat the corresponding sector as a \emph{gapped/stable scattering sector} for the purposes of applying the Haag--Ruelle/LSZ theorem layer, with all remaining caveats carried by the canonical scattering failure points (S1--S3).
\end{assumption}

\paragraph{Scope boundary (explicit).}
\AuditTag Assumption~\ref{ass:gapped_sector_evidence_contract} is an auditable \emph{interface contract}, not a proof of a QFT mass gap.
It exists to make the scattering prerequisites testable against the paper's finite diagnostics, and to ensure that any use of scattering is explicitly conditional on a declared window and checkable evidence.
