\subsubsection{Negative square index and Pontryagin realization: indexed characterization of extreme resonances, Toeplitz inertia stabilization, and horizon renormalization duality}\label{subsubsec:xi-negative-square-index-pontryagin-realization}
In the scalar caliber of \S\ref{subsec:xi-certificate-coherence-orthogonal-axis-capacity}, we have strictly compressed the RH horizon criterion to
Carathéodory positivity of \(\mathscr{C}_{\zeta}\) (equivalent to Toeplitz-PSD hierarchy), and further transcribed it to the Schur/CMV conservative realization interface via Cayley-Schur transformation (Theorems \ref{thm:xi-horizon-carath-schur-equivalence}--\ref{thm:xi-horizon-conservative-realization}).
This section, without introducing any extra-system semantics, elevates "extreme resonance (pole in stability domain)" from \emph{geometric location} to an \emph{indexed spectral realization problem}:
true resonance is no longer characterized by "where is the pole", but by "what is the minimum number of negative-energy degrees of freedom needed to realize this transfer function"; this number is the Pontryagin index/negative square index.
In the RH phase this index is zero, thus returning to purely Hilbert passive (unitary) realization.

\paragraph{Generalized Schur/Carathéodory class and negative square index}
Following the notation of Cayley-Schur transformation \eqref{eq:xi-horizon-cayley-schur}, let \(S\) be a meromorphic function on \(\mathbb{D}\) (scalar or matrix-valued cases both applicable), and define its de Branges-Rovnyak type kernel
\begin{equation}\label{eq:xi-generalized-schur-kernel}
\boxed{\;
K_{S}(w,z):=\frac{I-S(w)S(z)^\ast}{1-w\overline{z}}\;}
\qquad(w,z\in\mathbb{D}).
\end{equation}
Here \(I\) is the identity operator of compatible dimension; in scalar case take \(I=1\).

\begin{definition}[Negative square index of generalized Schur class]\label{def:xi-generalized-schur-negative-squares}
Say \(S\) belongs to the \textbf{generalized Schur class} \(\mathcal{S}_{\kappa}\) if for any finite point set \(\{w_j\}_{j=1}^{n}\subset\mathbb{D}\), the number of negative eigenvalues (counting multiplicity) of the Gram (block) matrix
\(
\bigl(K_S(w_i,w_j)\bigr)_{1\le i,j\le n}
\)
does not exceed \(\kappa\); the minimal such \(\kappa\) is called the \textbf{negative square index} of \(S\), denoted
\(\mathrm{sq}_-(S)=\kappa\).
When \(\kappa=0\), \eqref{eq:xi-generalized-schur-kernel} is a positive definite kernel and \(S\) reduces to the usual Schur class.
\end{definition}

\begin{definition}[Generalized Carathéodory class]\label{def:xi-generalized-carath-class}
Say meromorphic function \(C\) belongs to the \textbf{generalized Carathéodory class} \(\mathcal{C}_{\kappa}\) if its Cayley-Schur transform
\(
S=\frac{C-1}{C+1}
\)
belongs to \(\mathcal{S}_{\kappa}\).
We also call \(\kappa\) the negative square index of \(C\), denoted \(\mathrm{sq}_-(C):=\mathrm{sq}_-(S)\).
\end{definition}

\begin{remark}[Alignment with Toeplitz-PSD/Carathéodory positivity]\label{rem:xi-negative-squares-vs-psd}
When \(\kappa=0\), \(\mathcal{C}_0\) is the usual Carathéodory class: \(C\) analytic in \(\mathbb{D}\) with \(\Re C\ge 0\);
in the context of \(\mathscr{C}_{\zeta}\), this is equivalent to RH (Corollary \ref{cor:app-horizon-carath-rh}), and strictly equivalent to Toeplitz-PSD hierarchy (Theorem \ref{thm:app-horizon-toeplitz-lmi}).
Therefore \(\mathrm{sq}_-(\mathscr{C}_{\zeta})\) can be viewed as a realization-independent integer index of "RH deviation".
\end{remark}

\begin{theorem}[Disk interior zeros \texorpdfstring{$\Longleftrightarrow$}{<->} horizon poles: RH and index annihilation]\label{thm:xi-disk-zero-pole-index-reduction}
In the appendix disk completion interface, let
\(
F_{\zeta}(w):=\Xi_{\zeta}(s(w))
\)
be the disk readout function (Theorem \ref{thm:app-rh-iff-disk-zero-free}), and let horizon logarithmic derivative
\(
\mathscr{C}_{\zeta}(w):=-\Xi_{\zeta}'(s(w))/\Xi_{\zeta}(s(w))
\).
Then for any \(w_0\in\mathbb{D}\),
\[
\boxed{\;
F_{\zeta}(w_0)=0\ \Longleftrightarrow\ \mathscr{C}_{\zeta}\ \text{has pole at}\ w_0\;}
\]
and the pole order equals the zero multiplicity of \(F_{\zeta}\) at \(w_0\).
Therefore RH is equivalent to \(F_{\zeta}\) having no zeros in \(\mathbb{D}\) (Theorem \ref{thm:app-rh-iff-disk-zero-free}), also equivalent to \(\mathscr{C}_{\zeta}\) being analytic in \(\mathbb{D}\) and belonging to the usual Carathéodory class (Corollary \ref{cor:app-horizon-carath-rh}), thus equivalent to
\[
\boxed{\;\mathrm{sq}_-(\mathscr{C}_{\zeta})=0.\;}
\]
\end{theorem}

\begin{proof}
By definition, \(F_{\zeta}(w_0)=0\) if and only if \(\Xi_{\zeta}(s(w_0))=0\);
and logarithmic derivative \(-\Xi_{\zeta}'/\Xi_{\zeta}\) has a pole at zeros, with order equal to zero multiplicity, giving the equivalence.
The remaining conclusions follow directly from disk RH equivalence and Carathéodory positivity criterion.
\end{proof}

\paragraph{True resonance counting: disk interior pole count \texorpdfstring{$=$}{=} negative square index}
Let \(C\) be meromorphic in \(\mathbb{D}\) with finite pole set (counting algebraic multiplicity) interior to the disk.
Define
\begin{equation}\label{eq:xi-polecount-kappa}
\boxed{\;
\kappa_{\mathrm{pole}}(C):=\sum_{a\in\mathbb{D}}\operatorname{ord}_{a}(C)\in\NN\cup\{0\}\;}
\end{equation}
representing total pole order sum in disk (location-independent version of true resonance count).

\begin{theorem}[Pole count \texorpdfstring{$\Longleftrightarrow$}{<->} negative square index]\label{thm:xi-poles-negative-squares-equivalence}
Let \(C\) be meromorphic in \(\mathbb{D}\), with no poles on boundary \(\partial\mathbb{D}\), and satisfying \(\kappa_{\mathrm{pole}}(C)=r<\infty\).
Then
\[
C\in\mathcal{C}_{r}\setminus\mathcal{C}_{r-1},
\qquad
S=\frac{C-1}{C+1}\in\mathcal{S}_{r}\setminus\mathcal{S}_{r-1},
\qquad
\mathrm{sq}_-(C)=\mathrm{sq}_-(S)=r.
\]
In particular, \(r=0\) if and only if \(C\) belongs to usual Carathéodory class (analytic in \(\mathbb{D}\) with \(\Re C\ge 0\)).
Taking \(C=\mathscr{C}_{\zeta}\) yields: "existence of off-slice zeros" can be strictly transcribed as \(\mathrm{sq}_-(\mathscr{C}_{\zeta})>0\).
\end{theorem}

\begin{proof}
This is a standard conclusion of generalized Schur/Carathéodory theory. The negative square index of meromorphic \(C\) equals its total disk interior pole multiplicity,
and Cayley-Schur transformation preserves negative square index (see function class and representation theory in \cite{KreinLanger1977ExtensionPiKappaI,AlpayDijksmaRovnyakDeSnoo1997SchurPontryagin}).
When \(r=0\) it reduces to usual Carathéodory/Schur positive definite kernel case (Theorem \ref{thm:xi-horizon-carath-schur-equivalence}).
\end{proof}

\paragraph{Pontryagin realization: negative square index equals minimal negative-energy degrees of freedom}
\begin{theorem}[Minimal Pontryagin conservative realization]\label{thm:xi-minimal-pontryagin-conservative-realization}
If \(S\in\mathcal{S}_{\kappa}\), then there exists a Pontryagin space \(\Pi_{\kappa}\) with index \(\kappa\) and a conservative (\(J\)-unitary) linear system colligation on it such that \(S\) is its characteristic function (transfer function).
Moreover, the Pontryagin index of this realization is \emph{minimal}, exactly equal to \(\mathrm{sq}_-(S)=\kappa\).
When \(\kappa=0\), \(\Pi_{\kappa}\) reduces to Hilbert space, thus yielding purely passive unitary scattering realization; for \(S_{\zeta}\) this agrees with the Hilbert caliber of Theorem \ref{thm:xi-horizon-conservative-realization}.
\end{theorem}

\begin{proof}
By Definition \ref{def:xi-generalized-schur-negative-squares}, kernel \(K_S\) has \(\kappa\) negative squares; the corresponding reproducing kernel Pontryagin space \(\mathcal{H}(S)\) exists with index \(\kappa\).
Generalized de Branges-Rovnyak model gives a (\(J\)-unitary) realization on \(\mathcal{H}(S)\) with characteristic function \(S\), and index minimality follows from minimality of negative square number (see \cite{AlpayDijksmaRovnyakDeSnoo1997SchurPontryagin}).
\end{proof}

\paragraph{Toeplitz-inertia stabilization and auditable readout of negative square index}
Let
\begin{equation}\label{eq:xi-generalized-carath-series}
\boxed{\;
C(w)=c_0+2\sum_{n\ge 1}c_n w^n,\qquad c_{-n}:=c_n^\ast\;}
\end{equation}
and let Toeplitz truncation
\(
\mathbf{T}_N:=(c_{j-k})_{0\le j,k\le N}
\).

\begin{definition}[Generalized Carathéodory index (Toeplitz inertia definition)]\label{def:xi-generalized-carath-index-toeplitz}
For the above \(C\) define its (possibly infinite) Carathéodory index
\begin{equation}\label{eq:xi-carath-index-toeplitz}
\boxed{\;
\kappa_{\mathrm{Car}}(C):=\sup_{N\ge 0}\nu_-\!\bigl(\mathbf{T}_N(C)\bigr)\ \in\ \NN\cup\{0,\infty\}.\;}
\end{equation}
When \(\kappa_{\mathrm{Car}}(C)=0\), \(\mathbf{T}_N(C)\succeq 0\) holds for all \(N\), thus returning to usual Carathéodory/Toeplitz-PSD case.
In this paper's finite-type caliber (finite disk interior poles and no boundary poles), Theorem \ref{thm:xi-toeplitz-inertia-stabilizes-to-kappa} shows this index is a finite integer and can be directly read from stable negative inertia of high-order truncations.
\end{definition}

\begin{theorem}[Toeplitz negative inertia stable readout]\label{thm:xi-toeplitz-inertia-stabilizes-to-kappa}
Let \(C\in\mathcal{C}_{\kappa}\) with only finitely many poles in \(\mathbb{D}\) (and no poles on \(\partial\mathbb{D}\)).
Then there exists \(N_0\) such that for all \(N\ge N_0\)
\begin{equation}\label{eq:xi-toeplitz-inertia-stable}
\boxed{\;
\nu_-\!\bigl(\mathbf{T}_N\bigr)=\kappa. \;}
\end{equation}
Therefore the negative square index \(\kappa=\mathrm{sq}_-(C)\) can be \emph{stably} read from the number of negative eigenvalues of sufficiently high-order Toeplitz truncations;
when \(\kappa=0\) this reduces to Toeplitz-PSD hierarchy \(\{\mathbf{T}_N\succeq 0\}_{N\ge 0}\) holding entirely.
\par
In particular, under the above finite-type caliber we necessarily have \(\kappa_{\mathrm{Car}}(C)=\kappa\) (Definition \ref{def:xi-generalized-carath-index-toeplitz}).
\end{theorem}

\begin{proof}
The negative square index of generalized Carathéodory class can equivalently be characterized as the negative inertia upper bound of corresponding Toeplitz forms,
and under finite-type caliber "finite disk interior poles, no boundary poles", this negative inertia stabilizes at sufficiently high-order truncations and equals the index itself (see generalized moment problem/Toeplitz form theory in \cite{KreinLanger1977ExtensionPiKappaI,AlpayDijksmaRovnyakDeSnoo1997SchurPontryagin}).
Combining with this paper's finite-window true resonance index \(\kappa_N:=\nu_-(\mathbf{T}_N)\) (Theorem \ref{thm:xi-finite-window-true-resonance-index}) yields \eqref{eq:xi-toeplitz-inertia-stable}.
\end{proof}

\begin{corollary}[Inertia-stable resonance counting principle: finite-dimensional readout of disk interior zero count]\label{cor:xi-inertia-stable-counts-disk-zeros}
Let \(F_{\zeta}\) have only finitely many zeros \(\{w_j\}_{j=1}^{r}\subset\mathbb{D}\) in \(\mathbb{D}\) (counting multiplicity), with no zeros on \(\partial\mathbb{D}\).
Then \(\mathscr{C}_{\zeta}\) has exactly \(r\) poles (counting multiplicity) in \(\mathbb{D}\), and there exists \(N_0\) such that for all \(N\ge N_0\)
\[
\boxed{\;
\nu_-\!\bigl(\mathbf{T}_N(\mathscr{C}_{\zeta})\bigr)=r. \;}
\]
Therefore disk interior zero count (equivalent to off-critical-line zero count, Theorem \ref{thm:app-rh-iff-disk-zero-free}) can be directly read from the stable value of Toeplitz truncation negative inertia at sufficiently high order; RH is equivalent to this stable value being zero.
\end{corollary}

\begin{proof}
By Theorem \ref{thm:xi-disk-zero-pole-index-reduction}, zero/pole multiplicities correspond one-to-one, so disk interior pole total is \(r\).
Then from Theorem \ref{thm:xi-poles-negative-squares-equivalence} and Theorem \ref{thm:xi-toeplitz-inertia-stabilizes-to-kappa} the conclusion follows.
\end{proof}

\begin{corollary}[Discrete hair--continuous hair dichotomy: negative inertia bounded if and only if defect is purely point-wise finite (finite-type caliber)]\label{cor:xi-discrete-vs-continuous-hair-by-negative-inertia}
Let \(C\) be generalized Carathéodory data induced by this paper's horizon interface, and let \(\mathbf{T}_N(C)\) be its Toeplitz principal submatrix.
Under finite-type caliber "finite disk interior poles, no boundary poles", the following are equivalent:
\begin{enumerate}
  \item There exists constant \(K<\infty\) such that
  \(\sup_{N}\nu_-\!\bigl(\mathbf{T}_N(C)\bigr)\le K\);
  \item Corresponding defect inner function factor contains only finite Blaschke factor with trivial singular inner factor (i.e., defect is finite point-wise with no continuous hair; see \S\ref{subsubsec:xi-hardy-inner-clark-carleson} inner factor decomposition and Proposition \ref{prop:xi-analytic-boundary-density-iff-no-singular-inner});
  \item Visible defect density \(H\) is real analytic on real axis, and its Cauchy/Herglotz transform is rational (equivalently: defect measure is finite atomic and uniquely recoverable by pole-residue inversion; see Theorem \ref{thm:xi-offcritical-poisson-herglotz-inversion}).
\end{enumerate}
Therefore, once defect exhibits infinite point-wise degrees of freedom (infinite Blaschke) or nontrivial singular inner factor (continuous hair), defect degrees of freedom cannot be closed within any fixed finite dimension: in Toeplitz caliber this manifests as \(\kappa_{\mathrm{Car}}(C)=\infty\) (Definition \ref{def:xi-generalized-carath-index-toeplitz}), i.e., \(\nu_-\!\bigl(\mathbf{T}_N(C)\bigr)\to\infty\).
\end{corollary}
\begin{proof}[Proof outline]
(1) \(\Longleftrightarrow\) (finite point-wise defect):
\(\sup_N\nu_-(\mathbf{T}_N(C))<\infty\) is equivalent to \(\kappa_{\mathrm{Car}}(C)<\infty\), which under finite-type caliber by Theorem \ref{thm:xi-poles-negative-squares-equivalence} is equivalent to finite disk interior pole total; in this paper's readout dictionary this corresponds to defect carried by finite Blaschke factor (Theorem \ref{thm:xi-disk-zero-pole-index-reduction}).
\par
(No singular inner factor \(\Longleftrightarrow\) real analytic density) given by Proposition \ref{prop:xi-analytic-boundary-density-iff-no-singular-inner};
and "finite atomic \(\Longleftrightarrow\) rational Herglotz" and pole-residue inversion given by Theorem \ref{thm:xi-offcritical-poisson-herglotz-inversion}.
Merging these equivalences on the same defect interface yields equivalence of (i)-(iii).
\par
The last paragraph is direct statement of contrapositive: if infinite Blaschke or singular inner factor exists, defect no longer belongs to finite type (no longer completely characterized by finite disk interior poles/finite atomic measure), thus \(\kappa_{\mathrm{Car}}(C)=\infty\) and forces \(\nu_-(\mathbf{T}_N(C))\) unbounded.
\end{proof}

\begin{theorem}[Toeplitz-Schur-Cohn double certificate consistency (finite-order inertia alignment)]\label{thm:xi-toeplitz-schur-cohn-inertia-alignment}
Let \(C\) at \(w=0\) give finite jet \((c_0,\dots,c_N)\), and let \(\mathbf{T}_N(C)\) be corresponding Toeplitz truncation.
Let \(S=\frac{C-1}{C+1}\) and take its \(N\)-step Schur recursion generated finite-type Schur data (equivalently take finite segment reflection coefficients/Verblunsky parameters), thus construct a channel polynomial \(p_N\) consistent with this finite data (e.g., corresponding Schur-Cohn test polynomial).
Then there exists invertible triangular matrix \(L_N\) such that
\[
\boxed{\;
\mathbf{T}_N(C)\ =\ L_N^\ast\,H(p_N)\,L_N\;}
\]
where \(H(p_N)\) is Schur-Cohn Hermitian form matrix (Remark \ref{rem:xi-schur-cohn-sdp-interface}).
Therefore finite-order negative inertia is consistent:
\[
\boxed{\;
\nu_-\!\bigl(\mathbf{T}_N(C)\bigr)=\nu_-\!\bigl(H(p_N)\bigr).\;}
\]
If further coefficients of \(p_N\) lie in invariant subring \(\QQ[u+u^{-1}]\), then \(H(p_N)\) can equivalently be viewed as univariate polynomial matrix semidefinite constraint on interval \(x\in[-2,2]\) (Remark \ref{rem:xi-schur-cohn-sdp-interface}), thus compressing "full-angle verification" to auditable certificate of interval PSD.
\end{theorem}

\begin{proof}
Schur recursion equivalently encodes finite jet data as finite segment reflection coefficients; compatible finite-type realization can be given by Schur-Cohn/CMV parameterization (see \cite{DymYoung1990SchurCohnMatrixPolynomials,Simon2005OPUC1}).
In this finite-type realization, \(\mathbf{T}_N(C)\) and \(H(p_N)\) are both different coordinate expressions of the same finite-dimensional indefinite type (or energy form), differing by an invertible triangular basis change, hence there exists stated congruence decomposition and inertia consistency follows.
\end{proof}

\paragraph{Toeplitz-scan Hankel isomorphic decomposition: negative subspace completely spanned by tomographic spectral fingerprint}
In \S\ref{subsubsec:xi-horizon-spectral-measure-tomography}, exponential spectral fingerprint
\(
u_n:=\widehat H(n\Delta)
\)
of comoving scan profile \(H\)
induces Hankel block \(\mathsf H^{(M)}:=(u_{p+q})_{0\le p,q\le M}\) (Theorem \ref{thm:xi-comoving-scan-hankel-rank-defect}).
Merging with Toeplitz inertia stable readout (Theorem \ref{thm:xi-toeplitz-inertia-stabilizes-to-kappa}) allows \emph{explicit isomorphism} of Toeplitz negative subspace in finite dimension to this tomographic fingerprint space.

\begin{corollary}[Toeplitz-scan Hankel congruence decomposition: negative subspace completely determined by tomographic spectral fingerprint]\label{cor:xi-toeplitz-scan-hankel-congruence-decomposition}
Under finite defect caliber, take \(N\ge N_0\) such that \(\nu_-(\mathbf{T}_N(C))=\kappa\) (Theorem \ref{thm:xi-toeplitz-inertia-stabilizes-to-kappa}),
and take \(M=\kappa-1\) such that \(\mathrm{rank}(\mathsf H^{(M)})=\kappa\) (Theorem \ref{thm:xi-comoving-scan-hankel-rank-defect}).
Denote \(\mathsf H_{\kappa-1}:=\mathsf H^{(\kappa-1)}\).
Then there exist invertible matrix \(R_N\) and some \(P_N\succeq 0\) such that for all sufficiently large \(N\)
\[
\boxed{\;
R_N^\ast\,\mathbf{T}_N(C)\,R_N
=
\begin{pmatrix}
P_N & 0\\[2pt]
0 & -\,\mathsf H_{\kappa-1}^{\ast}\mathsf H_{\kappa-1}
\end{pmatrix}. \;}
\]
Thus
\[
\boxed{\;
\nu_-\!\bigl(\mathbf{T}_N(C)\bigr)=\mathrm{rank}(\mathsf H_{\kappa-1})=\kappa. \;}
\]
Therefore, all negative directions of Toeplitz hierarchy do not diffusely destabilize, but can be completely characterized within stable intervals by Hankel block generated from scan spectral fingerprint \(\{u_n\}\) (through its Gram \(\mathsf H_{\kappa-1}^{\ast}\mathsf H_{\kappa-1}\)).
\end{corollary}

\begin{proof}
By Sylvester inertia theorem, there exists invertible matrix \(U_N\) such that
\(
U_N^\ast\mathbf{T}_N(C)U_N=\mathrm{diag}(P_N,-I_{\kappa})
\)
where \(P_N\succeq 0\) (corresponding to positive subspace restriction).
On the other hand, from \(\mathrm{rank}(\mathsf H_{\kappa-1})=\kappa\) we get \(\mathsf H_{\kappa-1}\) invertible, hence
\[
\begin{pmatrix} I & 0\\ 0 & \mathsf H_{\kappa-1}\end{pmatrix}^{\!\ast}
\begin{pmatrix} P_N & 0\\ 0 & -I_{\kappa}\end{pmatrix}
\begin{pmatrix} I & 0\\ 0 & \mathsf H_{\kappa-1}\end{pmatrix}
=
\begin{pmatrix} P_N & 0\\ 0 & -\mathsf H_{\kappa-1}^{\ast}\mathsf H_{\kappa-1}\end{pmatrix}.
\]
Taking \(R_N:=U_N\mathrm{diag}(I,\mathsf H_{\kappa-1})\) yields stated congruence decomposition.
Last formula follows from \(\nu_-(P_N)=0\) and \(\nu_-( -\mathsf H_{\kappa-1}^{\ast}\mathsf H_{\kappa-1})=\mathrm{rank}(\mathsf H_{\kappa-1})\) immediately.
\end{proof}

\paragraph{Horizon renormalization duality: index invariant while endpoint flux linearly amplifies}
Appendix \eqref{eq:scale-psi-m} introduces Möbius semigroup \(\{\psi_m\}_{m>0}\subset\mathrm{Aut}(\mathbb{D})\) preserving fixed points \(\{\pm 1\}\), whose endpoint Poisson/Busemann weights satisfy strict scaling law \(P_{\psi_m(a)}(-1)=mP_a(-1)\) (Theorem \ref{thm:xi-endpoint-poisson-weight-scaling}).
This semigroup provides a complementary duality of "index invariant—endpoint flux amplification" for true resonance index.

\begin{theorem}[Horocycle depth and quadratic suppression: quantitative constraint of endpoint geometry on detection]\label{thm:xi-horocycle-depth-quadratic-suppression}
If there exists off-critical-line zero \(\rho=\beta+\mathrm{i}\gamma\) (\(\beta<\tfrac12\)) and let \(\delta:=\tfrac12-\beta>0\), its disk image \(w_\rho=\mathcal{C}(\gamma+\mathrm{i}\delta)\in\mathbb{D}\) satisfies
\[
\boxed{\;
\delta=\mathcal{B}_{-1}(w_\rho)=\frac{1-|w_\rho|^2}{|1+w_\rho|^2},\qquad
1-|w_\rho|^2=\delta\,|1+w_\rho|^2.\;}
\]
Therefore off-line offset enters quadratically in radial term at endpoint neighborhood, horocycle depth \(\mathcal{B}_{-1}\) provides intrinsic readout coordinate of "true resonance geometric strength";
this quadratic suppression also explains the structural lower bound of Toeplitz/Schur-Cohn certificate detection order in high-height intervals (determined by endpoint focusing).
\end{theorem}

\begin{proof}
See Theorem \ref{thm:xi-offcritical-quadratic-radial-compression} and Corollary \ref{cor:app-offcritical-horocycle-busemann}.
\end{proof}

\paragraph{Quantitative duality of Toeplitz certificate length and boundary layer depth}
In this paper's disk horizon interface, Toeplitz-PSD failure at finite dimension provides refutator (\(\mathbf{T}_N\nsucceq 0\)), while endpoint geometry compresses off-slice offset to thin boundary layer coordinate (Theorem \ref{thm:xi-horocycle-depth-quadratic-suppression}).
This paragraph merges them into a citable \emph{quantitative duality}: Toeplitz certificate order \(N\) acts as resolution depth of boundary layer; the closer defect points approach unit circle, any strategy attempting to verify/exclude this defect via finite-order Toeplitz-PSD certificate must pay infeasibility lower bound of \(N\) diverging.

\begin{theorem}[Quantitative duality of Toeplitz certificate length and boundary layer depth]\label{thm:xi-toeplitz-length-vs-boundary-depth}
Under coefficient caliber of Appendix \ref{subsubsec:app-horizon-weyl-carath-toeplitz}, suppose \(\mathscr{C}_{\zeta}\) has disk interior pole \(a\in\mathbb{D}\) (equivalent to disk completion \(F_{\zeta}\) having disk interior zero; Theorem \ref{thm:xi-disk-zero-pole-index-reduction}).
Denote its boundary layer depth
\[
h:=1-|a|\in(0,1).
\]
Under "remainder coefficients auditably bounded" assumption (same caliber as Appendix Theorem \ref{thm:app-horizon-toeplitz-detection-threshold}), Toeplitz-PSD finite-order verification has infeasibility threshold determined by boundary layer depth.
Let
\[
Q:=\frac{1}{1-|a|^2},\qquad
\delta:=\mathcal{B}_{-1}(a)=\frac{1-|a|^2}{|1+a|^2},
\]
then there exists constant \(C>0\) such that if wishing to \emph{exclude} this disk interior pole at regime level via finite-order Toeplitz-PSD certificate (requiring sufficiently high order to still force failure witness to appear), certificate length must satisfy
\[
\boxed{\;
N\ \ge\ C^{-1}\,Q\,\log\!\bigl(1+\delta Q\bigr)\;}
\]
thus as depth \(h\downarrow 0\) (pole approaches unit circle) required order is forced to diverge.
In particular, at coarse scale there is necessary lower bound
\[
\boxed{\;
N\ \gtrsim\ \frac{1}{h}\;}
\]
(endpoint thin layer mechanism further introduces logarithmic correction).
\end{theorem}

\begin{proof}
This is direct corollary of Appendix Theorem \ref{thm:app-horizon-toeplitz-detection-threshold}: for off-slice zero \(\rho\)'s disk image \(w_\rho=a\) define quality factor \(Q=(1-|a|^2)^{-1}\asymp h^{-1}\), substituting into \eqref{eq:app-horizon-toeplitz-detect-threshold} yields stated threshold; at coarse scale from \(Q\asymp h^{-1}\) immediately get \(N\gtrsim h^{-1}\).
\end{proof}

\begin{proposition}[Geometric lower bound of negative spectral magnitude: minimal negative eigenvalue controls visible depth]\label{prop:xi-toeplitz-min-eig-controls-depth}
Continue setting of Theorem \ref{thm:xi-toeplitz-length-vs-boundary-depth}.
Denote minimal eigenvalue of Toeplitz truncation as \(\lambda_{\min}(\mathbf{T}_N)=-\eta_N<0\).
Then under boundary layer dominated extreme resonance caliber (i.e., disk interior pole near unit circle, and remainder has uniform bound within audit window), there exists disk interior pole \(a\in\mathbb{D}\) such that its boundary layer depth satisfies
\[
\boxed{\;
1-|a|\ \ge\ c\,\frac{\eta_N}{N}\;}
\]
where \(c>0\) is absolute constant independent of remainder audit bound.
Therefore "negativity amplitude" of finite-order Toeplitz certificate is not only binary refutator, but simultaneously provides auditable lower bound scale of defect depth.
\end{proposition}

\begin{proof}
Denote \(\|x\|_2=1\) as unit eigenvector corresponding to \(\lambda_{\min}\), then
\(
\eta_N=-x^\ast\mathbf{T}_N x
\).
Expanding quadratic form by Toeplitz diagonal and estimating cross-correlation terms by Cauchy-Schwarz yields
\[
\eta_N\ \le\ 2\sum_{n=1}^{N}|c_n|+|c_0|
\ \le\ (2N+1)\max_{1\le n\le N}|c_n|+|c_0|.
\]
On the other hand, under pole principal part caliber of Appendix Proposition \ref{prop:app-horizon-pole-imprint}, \(c_n\) in boundary layer regime is dominated by finitely many geometric terms, with amplitude of same order as \(1-|a|\) (from \(|1+a|^2\le 4\) and endpoint compression law \(1-|a|^2=\delta|1+a|^2\) can control principal part amplitude to \(O(1-|a|)\)).
Merging above two formulas and rearranging yields existence of some pole \(a\) satisfying \(1-|a|\gtrsim \eta_N/N\).
(More refined constant tracking and logarithmic correction can follow coefficient imprint estimate of Appendix Theorem \ref{thm:app-horizon-toeplitz-detection-threshold}.)
\end{proof}

\begin{theorem}[Index invariance \(\times\) endpoint flux amplification]\label{thm:xi-index-flux-two-laws}
Take \(m>1\) and let \(\widetilde C:=C\circ\psi_m^{-1}\), \(\widetilde S:=\frac{\widetilde C-1}{\widetilde C+1}\).
Then:
\begin{enumerate}
  \item \textbf{(Index invariance)} \(\mathrm{sq}_-(\widetilde C)=\mathrm{sq}_-(C)\) (equivalent to \(\widetilde S\in\mathcal{S}_{\kappa}\iff S\in\mathcal{S}_{\kappa}\)).
  \item \textbf{(Endpoint flux amplification)} If point-wise data carried by finite Blaschke factor \(B\) and expressed via endpoint absorption coefficient \(\Phi_{-1}(B)\) its endpoint Poisson-Busemann mass (Proposition \ref{prop:xi-endpoint-absorption-coefficient}), then
  \[
  \boxed{\;
  \Phi_{-1}\!\bigl(B\circ\psi_m^{-1}\bigr)=m\,\Phi_{-1}(B)\; }.
  \]
\end{enumerate}
\end{theorem}

\begin{proof}
First item is direct corollary of negative square number invariance under \(\mathrm{Aut}(\mathbb{D})\): for any \(\phi\in\mathrm{Aut}(\mathbb{D})\), kernels \(K_{S\circ\phi}\) and \(K_S\)'s Gram matrices differ by invertible diagonal conjugate factor at any finite point set, thus negative inertia (negative square number) remains invariant (see \cite{AlpayDijksmaRovnyakDeSnoo1997SchurPontryagin}).
Second item follows from definition \(\Phi_{-1}(B)=\sum_k P_{a_k}(-1)\) and Theorem \ref{thm:xi-endpoint-poisson-weight-scaling} applied term-wise.
\end{proof}

\begin{theorem}[Computability of endpoint flux via reflection coefficients: exponential precision approximation of \texorpdfstring{$\Phi_{-1}$}{Phi-(-1)} by finite CMV certificate]\label{thm:xi-endpoint-flux-finite-cmv-computable}
Let point-wise defect be carried by finite Blaschke factor \(B\) (same finite defect caliber as Proposition \ref{prop:xi-endpoint-absorption-coefficient}), and let \((\alpha_n)_{n\ge 0}\) be its Schur/Verblunsky coefficient sequence (generated by Schur recursion; see Definition \ref{def:xi-horizon-schur-parameters} and OPUC standard theory \cite{Simon2005OPUC1}).
For each \(N\ge 1\), let \(B^{(N)}\) denote \(N\)-step Schur truncation determined by finite segment data \((\alpha_0,\dots,\alpha_{N-1})\) (equivalently: rational inner function determined by corresponding finite CMV certificate \(U^{(N)}\)).
Define finite certificate endpoint flux estimator
\[
\boxed{\;
\mathcal{F}_N:=\Re\!\left(-\frac{(B^{(N)})'(-1)}{B^{(N)}(-1)}\right)\; }.
\]
Then:
\begin{enumerate}
  \item \textbf{(Finite degree exactness)} If \(\deg(B)=\kappa<\infty\), then when \(N\ge \kappa\) we have \(B^{(N)}\equiv B\), thus \(\mathcal{F}_N=\Phi_{-1}(B)\) (from Theorem \ref{thm:xi-endpoint-julia-indicator-equals-absorption}'s \(\Phi_{-1}(B)=\Re(-B'(-1)/B(-1))\)).
  \item \textbf{(Exponential error bound)} Therefore there exist constants \(C>0\), \(q\in(0,1)\) such that for all \(N\ge 1\)
  \[
  \bigl|\Phi_{-1}(B)-\mathcal{F}_N\bigr|\ \le\ C\,q^{N}.
  \]
  \item \textbf{(Computability)} \(\mathcal{F}_N\) depends only on finite certificate parameters \((\alpha_0,\dots,\alpha_{N-1})\), and can be implemented within purely multiplicative-additive rational operations through finite Schur recursion and rational evaluation/differentiation at \(w=-1\) (then take real part).
\end{enumerate}
\end{theorem}

\begin{proof}
Finite Blaschke product Schur recursion terminates in finite steps: if \(\deg(B)=\kappa\), its Schur parameters satisfy \(|\alpha_n|<1\) (\(n<\kappa\)) with terminal term falling on unit circle, thus \(B\) can be completely recovered from finite segment parameters (\cite{Simon2005OPUC1}).
Therefore for \(N\ge\kappa\) we have \(B^{(N)}\equiv B\), from Theorem \ref{thm:xi-endpoint-julia-indicator-equals-absorption} obtain \(\mathcal{F}_N=\Phi_{-1}(B)\), thus exponential error bound holds for any \(q\in(0,1)\) with sufficiently large \(C\).
Computability follows directly from rationality of Schur recursion and rational function form of \(B^{(N)}\).
\end{proof}

\begin{corollary}[Rigid sufficient condition of cross-scale closure: bounded endpoint flux \(\Rightarrow\) \texorpdfstring{$\kappa=0$}{kappa=0}]\label{cor:xi-bounded-flux-forces-kappa0}
Let point-wise data be carried by finite Blaschke factor \(B\), and there exists some \(m>1\) such that sequence
\(\{\Phi_{-1}(B\circ\psi_{m^n}^{-1})\}_{n\ge 0}\)
is uniformly bounded.
Then \(\Phi_{-1}(B)=0\), thus \(B\equiv 1\) (Proposition \ref{prop:xi-endpoint-absorption-coefficient}), point-wise true resonance vanishes;
combining Theorem \ref{thm:xi-poles-negative-squares-equivalence} yields \(\mathrm{sq}_-(C)=0\).
\end{corollary}

\paragraph{Multichannel strictification: completely positive visible realization \texorpdfstring{$\Longleftrightarrow$}{<->} visible channel index annihilation}
\begin{theorem}[Completely positive visible realization \texorpdfstring{$\Rightarrow$}{=>} visible channel negative square index zero]\label{thm:xi-cp-visible-implies-kappa0}
Fix resolution \(m\) and let \(\mathcal{A}_m=\CC[G^{\mathrm{BC}}_m]\).
If there exists \(\mathcal{A}_m\)-valued operator Carathéodory function \(F_m:\mathbb{D}\to\mathcal{A}_m\) satisfying completely positive block Toeplitz-PSD hierarchy
\(\mathbf{T}^{\mathcal{A}_m}_N(F_m)\succeq 0\) (\(\forall N\)) (Theorem \ref{thm:xi-operator-herglotz-toeplitz-completeness}),
then for each visible channel (irreducible representation) \(\pi\in\widehat{G^{\mathrm{BC}}_m}\), its component \(F_{m,\pi}=\pi(F_m)\) is (matrix-valued) usual Carathéodory, thus its Cayley-Schur transform
\[
S_{m,\pi}(w):=\bigl(F_{m,\pi}(w)-I\bigr)\bigl(F_{m,\pi}(w)+I\bigr)^{-1}
\]
belongs to usual Schur class (negative square index zero):
\[
\boxed{\;\mathrm{sq}_-\!\bigl(S_{m,\pi}\bigr)=0\qquad(\forall \pi\in\widehat{G^{\mathrm{BC}}_m}).\;}
\]
Therefore after strictification on visible channels, there exists uniform Hilbert (\(\kappa=0\)) unitary scattering realization; all visible singularities can only be carried via boundary spectral measure.
\end{theorem}

\begin{proof}
By Theorem \ref{thm:xi-fourier-pontryagin-block-toeplitz}, \(\mathbf{T}^{\mathcal{A}_m}_N(F_m)\succeq 0\) is equivalent to matrix Toeplitz truncation \(\mathbf{T}_N(F_{m,\pi})\succeq 0\) for each \(\pi\);
by matrix-valued Carathéodory-Toeplitz completeness (block-wise version of Theorem \ref{thm:xi-operator-herglotz-toeplitz-completeness}), this is equivalent to \(\Re F_{m,\pi}(w)\succeq 0\).
Further Cayley-Schur transformation sends usual Carathéodory precisely to usual Schur (\(\kappa=0\)) class, hence \(\mathrm{sq}_-(S_{m,\pi})=0\).
\end{proof}

\begin{corollary}[True resonance or strictification failure \texorpdfstring{$\Rightarrow$}{=>} finite-dimensional refutator]\label{cor:xi-multichannel-witness-kappa-or-anom}
Under above setting, if some visible channel exhibits \(\mathrm{sq}_-(S_{m,\pi})>0\) (true resonance index nonzero),
then there necessarily exists finite \(N\) such that \(\mathbf{T}_N(F_{m,\pi})\) exhibits negative eigenvalue, thus block Toeplitz-PSD fails at finite order and provides offline verifiable refutator (Corollary \ref{cor:xi-cp-breakdown-finite-witness}).
If strictification does not hold on visible channels (nonzero anomaly), then no visible quotient exists that can unify multichannels into single \(\mathcal{A}_m\)-valued completely positive Herglotz representation; the anomaly class itself is independent finite-dimensional refutation interface (Remark \ref{rem:xi-anom-as-cp-visible-obstruction}).
\end{corollary}

\begin{theorem}[Inverse spectral interface of finite-window threshold resonance: CMV/Schur parameters \texorpdfstring{$\Longleftrightarrow$}{<->} boundary root certificate]\label{thm:xi-finite-window-threshold-resonance-inverse-spectral}
In finite-window model, threshold resonance (unit circle root condition) can be equivalently rewritten as CMV-like unitary spectral realization with explicit Schur/Verblunsky parameter constraints:
there exists five-diagonal CMV-like companion matrix \(U_L^{\mathrm{CMV}}\) such that
\(\widetilde\Phi_L(y)=\det(yI-U_L^{\mathrm{CMV}})\), and
\[
\mathrm{Roots}(\widetilde\Phi_L)\subseteq\TT
\Longleftrightarrow
U_L^{\mathrm{CMV}}\ \text{is unitary}
\Longleftrightarrow
|\alpha_k(L)|<1\ \ (k< D-1),\ \ |\alpha_{D-1}(L)|=1
\]
(Proposition \ref{prop:xi-hilbert-polya-cmv}).
Same condition can also be given via Schur-Cohn Hermitian form matrix (semi)definiteness LMI certificate, and when coefficients lie in \(\QQ[u+u^{-1}]\) further compresses to interval PSD feasibility (Remark \ref{rem:xi-schur-cohn-sdp-interface}).
Therefore finite-window threshold resonance provides a suite of discretized, verifiable prototypes of Toeplitz-PSD/Carathéodory hierarchy: any "true resonance" deviation from threshold carrying will manifest as Schur parameter violation or corresponding PSD/inertia certificate failure, thus producing finite-dimensional refutator (Theorem \ref{thm:xi-horizon-reflection-finite-witness} and Theorem \ref{thm:xi-toeplitz-inertia-stabilizes-to-kappa}).
\end{theorem}
