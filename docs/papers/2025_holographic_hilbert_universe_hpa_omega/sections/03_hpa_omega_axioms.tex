
\section{Axioms and layer discipline in \texorpdfstring{HPA--$\Omega$}{HPA--Omega} (with a Hilbert-folding extension)}
\label{sec:axioms}

\subsection{Closed layer vs.\ interpretation layer}
\label{sec:layers}

We adopt a strict two-layer writing discipline:
\begin{itemize}
\item \textbf{Closed layer}: explicit axioms, regularization conventions, and theorems provable from them.
\item \textbf{Interpretation layer}: physical narratives (black holes, wormholes, agency, etc.) that do not enter the proof chain.
\end{itemize}
All formal results in this paper are stated in the closed layer, with assumptions clearly declared.

\subsection{Scan--projection readout and induced probability (O5)}
\label{sec:axiom_O5}

\begin{axiom}[O5: scan--projection readout and induced measure]
\label{ax:O5}
At finite resolution $\varepsilon>0$, the observer accesses a family of readout effects $\{E_k^{(\varepsilon)}\}_k$ (a POVM) on an effective Hilbert space $\mathcal{H}_{\mathrm{eff}}$, satisfying $\sum_k E_k^{(\varepsilon)}=\ind$.
Given an effective state $\omega_{\mathrm{eff}}$, the outcome probabilities are induced by
\[
\mathbb{P}_k^{(\varepsilon)}=\omega_{\mathrm{eff}}\!\left(E_k^{(\varepsilon)}\right).
\]
Operational (discrete) time is represented by scan ticks $t\in\ZZ_{\ge 0}$ indexing the internal scan.
\end{axiom}

This axiom encodes the principle that probability is not an external semantic postulate but an interface object induced by finite-resolution readout.

\subsection{Scan algebra and Weyl pair noncommutativity (O6)}
\label{sec:axiom_O6}

\begin{axiom}[O6: scan algebra with a Weyl pair]
\label{ax:O6}
On $\mathcal{H}_{\mathrm{eff}}$ there exist unitary operators $U_{\mathrm{scan}}$ and $V$ satisfying the Weyl relation
\[
U_{\mathrm{scan}}V=\e^{2\pi\iu\alpha}VU_{\mathrm{scan}},
\qquad \alpha\in(0,1)\setminus\QQ.
\]
The noncommutativity is intrinsic and provides a structural source of complementarity and uncertainty for scan shift and phase pointer readout.
\end{axiom}

\subsection{Canonical regularization path: Abel first, then finite part (R1)}
\label{sec:axiom_R1}

\begin{axiom}[R1: Abel first, then finite part]
\label{ax:R1}
Infinite-horizon scan traces are defined by Abel regularization on $|r|<1$ and are compared/renormalized only along the canonical path $r\uparrow 1$.
When a universal divergence of the form $c_{-1}/(1-r)$ is present, the admissible protocol value is the finite-part constant term extracted along this path.
\end{axiom}

\subsection{Hilbert folding (H1): a constructive spacetime extension}
\label{sec:axiom_H1}

\begin{axiom}[H1: Hilbert Folding Axiom]
\label{ax:H1}
For each resolution level $n\in\NN$ and each fixed dimension $d\ge 2$, there exists a discrete Hilbert address map
\[
H_n:\{0,1,\dots,2^{dn}-1\}\to \Sigma_n=\{0,1,\dots,2^n-1\}^d
\]
that is a bijection and satisfies one-step locality \eqref{eq:hilbert_locality}.
The observer-accessible boundary degrees of freedom at resolution $n$ are organized on the screen lattice $\Sigma_n$ with nearest-neighbor locality, and scan ticks address them by $x_t^{(n)}:=H_n(t\bmod 2^{dn})$.
\end{axiom}

H1 makes ``space'' an explicit finite-resolution object, derived from scan time by an auditable address map.
Implementation overhead (routing) then becomes a physical resource that can be measured and compared across regions.

\paragraph{Remark (mathematical existence).}
The existence of discrete Hilbert orders with one-step locality is a standard constructive fact in the space-filling curve literature \cite{Hilbert1891,Sagan1994}.
The axiom content of H1 is the physical identification of observer-accessible boundary degrees of freedom with such an addressable lattice at each resolution.
