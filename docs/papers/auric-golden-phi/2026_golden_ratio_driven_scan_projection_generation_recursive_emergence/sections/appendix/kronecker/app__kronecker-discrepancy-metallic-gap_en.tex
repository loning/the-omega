\subsection{Gap for the Metallic Mean Family (Constant-Type Continued Fractions)}\label{app:kronecker-discrepancy-metallic-gap}

This subsection records only the monotonicity conclusion for certificate coefficients used in subsequent sections.

\begin{definition}[Constant type and the metallic mean family]\label{def:const-type}
For an integer $A\ge 1$, set
\[
\alpha_A:=[0;A,A,A,\dots],
\qquad
\lambda_A:=\frac{A+\sqrt{A^2+4}}{2}>1.
\]
\end{definition}

\begin{theorem}[Strict gap for the metallic mean family: the golden branch minimizes the logarithmic coefficient]\label{thm:metallic-gap}
Let $A\ge 1$ and write \(\kappa(A):=A/\log\lambda_A\).
Then the function $\kappa(A)$ is strictly increasing on $[1,\infty)$. In particular, for all integers $A\ge 2$,
\[
\kappa(A)\ >\ \kappa(1)=\frac{1}{\log\varphi}.
\]
\end{theorem}
