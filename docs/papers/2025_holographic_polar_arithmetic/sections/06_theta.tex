\section{The scan operator \texorpdfstring{$\Theta$}{Theta}: time/readout must arise from a genuine unitary}
\label{sec:theta}

\subsection{Why a scalar phase is insufficient}
If $\Theta$ is merely a complex number $\e^{2\pi\iu\alpha}$, then for any state $|\psi\rangle$ we have $\Theta^k|\psi\rangle=\e^{2\pi\iu k\alpha}|\psi\rangle$, which differs only by a global phase and produces no observable structure. Therefore a scan that drives readout must be represented by a nontrivial unitary operator on a nontrivial Hilbert space.

\subsection{A minimal model: the Koopman operator for an irrational rotation}
Let $X:=\RR/\ZZ\cong S^1$ with Lebesgue measure $\mu$, and set $\hilbert:=L^2(X,\mu)$. Fix an irrational $\alpha\in(0,1)\setminus\QQ$. Define
$$
T_\alpha(x):=x+\alpha\pmod{1},
$$
and the Koopman operator $\Theta$ by
$$
(\Theta f)(x):=f(T_\alpha(x))=f(x+\alpha).
$$

\begin{proposition}[Unitarity]
\label{prop:theta_unitary}
The operator $\Theta$ is unitary on $\hilbert$.
\end{proposition}

\begin{proof}
For $f,g\in L^2(X)$,
$$
\langle \Theta f,\Theta g\rangle=\int_X \overline{f(x+\alpha)}\,g(x+\alpha)\,d\mu(x)=\int_X \overline{f(u)}\,g(u)\,d\mu(u)=\langle f,g\rangle,
$$
using invariance of Lebesgue measure under rotations.
\end{proof}

\subsection{A Weyl pair and intrinsic noncommutativity}
\label{subsec:weyl_pair}
On $\hilbert=L^2(X,\mu)$ define a second unitary $V$ by multiplication with the basic character,
$$
(Vf)(x):=\e^{2\pi\iu x} f(x).
$$
Then $U:=\Theta$ and $V$ satisfy the Weyl commutation relation
$$
UV=\e^{2\pi\iu\alpha}VU.
$$
Indeed,
$$
(UVf)(x)=(Vf)(x+\alpha)=\e^{2\pi\iu(x+\alpha)}f(x+\alpha)=\e^{2\pi\iu\alpha}(VUf)(x).
$$
The $C^*$-algebra generated by $U$ and $V$ is the irrational rotation algebra (noncommutative $2$-torus) for irrational $\alpha$; see e.g.~\cite{Rieffel1981,Connes1994}. In particular, scan shift and phase multiplication cannot be simultaneously diagonalized.

\begin{corollary}[No common eigenvectors]
\label{cor:no_common_eigenvectors}
If $\alpha\notin\QQ$, there is no nonzero $f\in\hilbert$ such that $Uf=\lambda f$ and $Vf=\mu f$ for some $\lambda,\mu\in S^1$.
\end{corollary}

\begin{proof}
If $Uf=\lambda f$ and $Vf=\mu f$, then $UVf=\lambda\mu f$ and $VUf=\mu\lambda f$. The Weyl relation gives $UVf=\e^{2\pi\iu\alpha}VUf$, hence $(1-\e^{2\pi\iu\alpha})\lambda\mu f=0$. Since $\alpha\notin\QQ$ implies $\e^{2\pi\iu\alpha}\ne 1$, it follows that $f=0$.
\end{proof}

\begin{theorem}[A variance-type uncertainty relation for a Weyl pair]
\label{thm:weyl_uncertainty}
Let $U$ and $V$ be unitary operators satisfying
$$
UV = \e^{\iu\Phi} VU,\qquad U^\dagger V = \e^{-\iu\Phi} VU^\dagger,
\qquad 0\le \Phi\le \pi.
$$
For a normalized state $|\psi\rangle$, define the unitary variances
$$
\Delta U^2 := 1-|\langle \psi|U|\psi\rangle|^2,\qquad
\Delta V^2 := 1-|\langle \psi|V|\psi\rangle|^2,
$$
and set $A:=\tan(\Phi/2)$. Then
$$
(1+2A)\,\Delta U^2\,\Delta V^2 + A^2(\Delta U^2+\Delta V^2)\ \ge\ A^2.
$$
\end{theorem}

\begin{proof}
This is Theorem~1 of~\cite{MassarSpindel2008}.
\end{proof}

\begin{remark}[Quantitative irreducibility for irrational rotations]
In the rotation model one has $UV=\e^{2\pi\iu\alpha}VU$. Choosing the representative $\Phi\in(0,\pi]$ with $\e^{\iu\Phi}=\e^{2\pi\iu\alpha}$ (equivalently, $\Phi=2\pi\min\{\alpha,1-\alpha\}$) and noting that $\Delta U^2$ depends only on $|\langle U\rangle|$, Theorem~\ref{thm:weyl_uncertainty} applies. In particular, for $\alpha\notin\QQ$ one has $\Phi\ne 0$, hence $A>0$, which forces a strictly nontrivial tradeoff between localization in the $U$- and $V$-modes.
\end{remark}

\subsection{Separating multiplicative phase from scan time}
Define scan time as the iteration index $k\in\ZZ_{\ge 0}$ and the orbit
$$
x_k:=T_\alpha^k(x_0)=x_0+k\alpha\pmod{1},\qquad f_k:=\Theta^k f_0.
$$
Define a scan phase increment $\delta:=2\pi\alpha$ and a scan phase $\theta_t(k):=k\delta$. The separation principle is:

\begin{itemize}
    \item $\theta_\times(n)$ is determined by the multiplicative structure of $n$ and satisfies $\theta_\times(mn)\equiv\theta_\times(m)+\theta_\times(n)$.
    \item $\theta_t(k)$ is determined by scan time $k$ and is independent of $n$.
\end{itemize}

This avoids the structural conflict that arises when one attempts to use a single index simultaneously as multiplicative coordinate and time/scan coordinate.

\subsection{The golden branch as a minimal grammar choice}
Let $\varphi:=\frac{1+\sqrt{5}}{2}$ and choose
$$
\alpha:=\varphi^{-1}=\varphi-1.
$$
This choice is not treated as mystical. It is singled out by two concrete extremal features.

\begin{itemize}
    \item \textbf{Anti-resonance (Diophantine extremality).} The golden ratio attains the optimal Hurwitz constant: among all irrational $\alpha$, it maximizes the Diophantine ``gap'' to rational resonances (equivalently, it is maximally badly approximable); see e.g.~\cite{Cassels1957,KuipersNiederreiter1974}.
    \item \textbf{Binary-minimal canonical encoding.} In Ostrowski numeration (Section~\ref{sec:zeckendorf}) the allowed digit alphabet at place $k$ is $\{0,1,\dots,a_{k+1}\}$. Requiring this alphabet to be binary at every place forces $a_i=1$ for all $i$, hence $\alpha=[0;1,1,1,\dots]=\varphi^{-1}$.
\end{itemize}

\begin{proposition}[Binary-minimality of the golden slope]
\label{prop:golden_binary_minimal}
Let $\alpha=[0;a_1,a_2,\dots]\in(0,1)\setminus\QQ$. The Ostrowski digit constraints \eqref{eq:ostrowski_digits} have binary digit alphabets at every place (equivalently $a_i=1$ for all $i$) if and only if $\alpha=\varphi^{-1}$. In that case the constraints degenerate to the Zeckendorf constraints (Section~\ref{sec:zeckendorf}).
\end{proposition}

\begin{proof}
In \eqref{eq:ostrowski_digits} the first digit satisfies $0\le b_1<a_1$ and, for $k\ge 1$, one has $0\le b_{k+1}\le a_{k+1}$. Thus the digit alphabet is binary at every place if and only if $a_1=1$ and $a_{k+1}=1$ for all $k\ge 1$, which is equivalent to $\alpha=[0;1,1,1,\dots]=\varphi^{-1}$. The degeneration to Zeckendorf is exactly the golden case described in Section~\ref{sec:zeckendorf}.
\end{proof}
