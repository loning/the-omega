
\section{Audit status classification}
\label{app:audit_table}

This appendix records the audit status of the main components of the paper.

\subsection{Closed chain (definitions and theorems)}

\begin{itemize}
\item \textbf{Standard facts.}
\begin{itemize}
\item Weyl/Kronecker equidistribution for irrational rotations (Theorems~\ref{thm:weyl_1d}--\ref{thm:kronecker_md}) \cite{Weyl1916,KuipersNiederreiter1974}.
\item Discrepancy bounds such as Koksma--Hlawka (Theorem~\ref{thm:koksma_hlawka}) \cite{KuipersNiederreiter1974}.
\end{itemize}
\item \textbf{New definitions.}
\begin{itemize}
\item $\ScanAlg$ (Definitions~\ref{def:scan_alg_object}, \ref{def:scan_alg_morphism}).
\item $\PerDatum$ and period data (Definitions~\ref{def:period_datum}, \ref{def:perdatum_object}, \ref{def:perdatum_morphism}).
\item Holographic scanning functor $\HSP$ (Definition~\ref{def:hsp_functor}).
\item Protocol-stable period data (Definition~\ref{def:wish}).
\end{itemize}
\item \textbf{Closed conclusions.}
\begin{itemize}
\item Scan averages realize period values (Corollaries~\ref{cor:scan_period_1d}, \ref{cor:scan_period_md}).
\item Auditable error decomposition and truncation bounds (Proposition~\ref{prop:error_decomposition}, Proposition~\ref{prop:zeta_trunc_bounds}).
\end{itemize}
\end{itemize}

\subsection{Programmatic components (not used as premises)}

\begin{itemize}
\item \textbf{Selection principle.} Conjecture~\ref{conj:selection_principle} proposes a falsifiable stability--complexity functional; it is not used in any proof.
\item \textbf{Motivic/Langlands compatibility.} Section~\ref{sec:langlands_compatibility} formulates a commutative-diagram objective factoring functorial upgrades through periods and motives; it is not assumed.
\end{itemize}
