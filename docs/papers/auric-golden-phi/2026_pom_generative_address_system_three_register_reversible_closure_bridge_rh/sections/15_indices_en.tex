\section{Index of Key Terms}

\subsection{Core semantic concepts}
\textbf{Address} ($\Addr$): Identifier for object locations; read is triggered by address.\par
\textbf{Read} ($\Read$): Partial function $\Read:\Addr\rightharpoonup \mathrm{ValDom}$ yielding values or $\NULL$ (undefined).\par
\textbf{$\NULL$}: Symbol for undefined read, strictly separated from numeric 0.\par
\textbf{Auditability}: Correctness verifiable by deterministic rules in finite steps.\par
\textbf{Certificate}: Finite data permitting soundness verification of an assertion.\par

\subsection{Syntactic and algebraic structures}
\textbf{POMmin}: Minimal axiom system for ontological mathematics.\par
\textbf{Fold normalization} ($\Fold_m$): Multiway map imposing stable golden-ratio syntax $(X_m)$.\par
\textbf{Zeckendorf valid digits} ($Z$): Finitely-supported sequences forbidding adjacent 1's.\par
\textbf{Three registers}: Scale (golden ratio), phase (unit circle), prime (exponent vectors).\par

\subsection{Information and ledger}
\textbf{Information ledger identity}: Entropy decomposition into visible $H(\pi)$, fiber volume, and KL residual.\par
\textbf{Fiber count} ($d_m(x)$): Size of preimage under fold, measuring irreversibility.\par
\textbf{Record-axis injectification}: Externalize residuals to achieve lossless lifting.\par
\textbf{Precision potential} ($P(x)$): Budget for radial compactification, $\log(1+x^2)$.\par

\subsection{Verification and type systems}
\textbf{Unitary slice}: Parameter domain where $|u_L(s)|=1$ coincides with critical line $\Re(s)=\tfrac12$.\par
\textbf{Type-safety (soundness)}: Addressable zeros forced to critical line under unitary slice protocol.\par
\textbf{Projection word} ($\mathrm{PW}$): Morphism-carrying proof object with decidable fragment $\mathsf{PW}_{\mathrm{dec}}$.\par
\textbf{BridgeRH}: Set of four-part falsifiable conditions ($B1$--$B4$) supporting classical RH reduction.\par

\subsection{Obstacles and boundaries}
\textbf{Fourth register}: Additional continuous register needed for radial modes beyond unitary slice.\par
\textbf{Divisible group obstacle}: $\RR_{>0}$ not embeddable into free abelian prime-register.\par
\textbf{Fixed-precision boundedness}: Addressable radial domain finite when precision index fixed.\par
\textbf{Collision-induced} $\NULL$: Undefined read when collision-free record-axis fails.\par
