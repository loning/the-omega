\section{Modular geodesic flow and Gauss-map renormalization (notes)}
\label{app:modular_flow_gauss_map}

\AuditTag This appendix records standard ``mother space'' facts connecting continued fractions to modular dynamics.
It provides a canonical source for the continued-fraction/Ostrowski structures that appear throughout the golden-branch layer (Section~\ref{subsec:zeckendorf_coding} and Appendix~\ref{app:discrepancy_ostrowski}).
These results are not used as premises in theorem-level folding proofs; they serve as an audit-facing justification for why the continued-fraction module is not an ad hoc digitization.

\subsection{Modular surface and the Gauss map}
\label{subsec:modular_surface_gauss}

Let $\mathbb{H}=\{\tau\in\CC:\ImPart(\tau)>0\}$ be the upper half-plane.
The modular group is generated by
\[
T:\tau\mapsto \tau+1,\qquad S:\tau\mapsto -\frac{1}{\tau},
\]
and the modular surface is the quotient orbifold
\[
M:=\mathrm{PSL}_2(\ZZ)\backslash\mathbb{H}.
\]
The geodesic flow on $M$ admits a classical symbolic coding whose base map is the Gauss map
\[
G(\xi)=\left\{\frac{1}{\xi}\right\},\qquad \xi\in(0,1),
\]
and whose symbols are the continued-fraction digits of $\xi=[0;a_1,a_2,\ldots]$.

\begin{theorem}[Series suspension model (classical)]
\label{thm:series_suspension_gauss_z128}
\MathTag There exists a Poincar\'e cross-section for the geodesic flow on $M$ whose first-return map is conjugate to the Gauss map $G$ on $(0,1)$.
Moreover, the geodesic flow is measurably isomorphic to the suspension flow over $G$ with roof function
\[
r(\xi)=-2\log \xi.
\]
In particular, the continued-fraction digits of $\xi$ arise as the symbolic itinerary of successive returns.
\end{theorem}
\begin{remark}
See \cite{Series1985,KatokHasselblatt1995} for standard constructions and proofs.
\end{remark}

\subsection{Invariant Gauss measure and digit law}
\label{subsec:gauss_measure_digit_law}

\begin{theorem}[Gauss invariant measure and digit distribution (classical)]
\label{thm:gauss_digit_law_z128}
\MathTag The Gauss map $G(\xi)=\{1/\xi\}$ preserves the probability measure
\[
\dd\mu(\xi)=\frac{1}{\log 2}\,\frac{\dd \xi}{1+\xi},\qquad \xi\in(0,1),
\]
and is ergodic with respect to $\mu$.
Writing $\xi=[0;a_1,a_2,\ldots]$, the first digit satisfies
\[
\mu(a_1=k)=\log_2\!\left(1+\frac{1}{k(k+2)}\right)\qquad (k\ge 1).
\]
\end{theorem}
\begin{remark}
See \cite{IosifescuKraaikamp2002} for proofs and further quantitative results.
\end{remark}

\subsection{Gauss--Kuzmin convergence (finite-time relaxation)}
\label{subsec:gauss_kuzmin_z128}

\begin{theorem}[Gauss--Kuzmin exponential convergence (classical)]
\label{thm:gauss_kuzmin_convergence_z128}
\MathTag Let $\nu$ be a probability measure on $(0,1)$ that is absolutely continuous with respect to Lebesgue measure with a density of bounded variation.
Then there exist constants $C>0$ and $0<\rho<1$ such that for all $n\ge 0$,
\[
\sup_{x\in(0,1)}\left|\nu\!\left(G^n(\xi)\le x\right)-\mu\!\left((0,x]\right)\right|
\le C\,\rho^n,
\]
where $\mu$ is the Gauss invariant measure from Theorem~\ref{thm:gauss_digit_law_z128}.
The optimal $\rho$ is known as the Gauss--Kuzmin--Wirsing constant (numerically $\rho\approx 0.30366$).
\end{theorem}
\begin{remark}
See \cite{IosifescuKraaikamp2002} for proofs and numerical constants.
\end{remark}

\subsection{Relation to Ostrowski numeration used in the paper}
\label{subsec:gauss_to_ostrowski_relation}

\AuditTag The modular-geodesic origin above provides a canonical meaning to the continued-fraction digits of a boundary irrational parameter.
In the present paper, the scan slope $\alpha$ is such a boundary irrational (Section~\ref{subsec:weyl_scan}), and the Ostrowski/Zeckendorf module used for discrepancy certificates (Appendix~\ref{app:discrepancy_ostrowski}) is the corresponding canonical integer coordinate system built from the continued fraction of $\alpha$.
This appendix does not identify the scan orbit with geodesic flow; it records that, once the modular stage is adopted, the digit module is not an arbitrary choice.
