\section{附录}

\subsection{附录 A：算子代数表述（投影作为概念，读出作为 \texorpdfstring{$0/1$}{0/1}）}
\label{app:op-algebra}

令 $\mathcal{H}$ 为 Hilbert 空间，$U$ 为幺正算子表示扫描步（群作用生成元的表示），$P$ 为投影算子表示窗口读出。则时间平移后的读出算子为
\[
P_t=U^{-t}PU^t.
\]
长度 $m$ 的地址 $a=a_0\cdots a_{m-1}\in\{0,1\}^m$ 可对应投影乘积
\[
P_a=\prod_{j=0}^{m-1} Q_{j}(a_j),\quad
Q_{j}(1)=P_j,\quad Q_{j}(0)=I-P_j,
\]
在可交换情形下 $P_a$ 即柱集指示投影；对给定态 $\rho$（密度算子），其读出概率为 $\Tr(\rho P_a)$。
在该表述中，“概念由地址确定”即为投影 $P_a$ 的定义依赖于地址；而“概念值为 $0/1$”对应在纯态 $\lvert\psi\rangle$ 的精确测量极限下，结果为投影谱 $\{0,1\}$。

\subsection{附录 B：稳定类型集合 \texorpdfstring{$X_m$}{X\_m} 的计数}

记 $a_m=|X_m|$。按末位分类：
\begin{itemize}
  \item 若末位为 $0$，则前 $m-1$ 位任意稳定词，共 $a_{m-1}$ 个；
  \item 若末位为 $1$，则倒数第二位必为 $0$，前 $m-2$ 位任意稳定词，共 $a_{m-2}$ 个。
\end{itemize}
故 $a_m=a_{m-1}+a_{m-2}$，且 $a_1=2,a_2=3$，从而 $a_m=F_{m+2}$。

\subsection{附录 C：折叠算子 \texorpdfstring{$\Fold_m$}{Fold\_m} 的可计算性与基本性质}

Zeckendorf 规范形可由贪心算法计算：对给定 $N\in\NN$，反复减去不超过当前余量的最大 Fibonacci 权重，并跳过相邻权重以强制无相邻 $1$。算法终止且输出唯一规范形 $Z(N)=(c_1,c_2,\dots)$（\cite{Zeckendorf1972}）。

对任意 $\omega=\omega_1\cdots\omega_m\in\Omega_m$，按定义 \ref{def:fold-word} 先计算 Fibonacci 加权值
\[
N(\omega)=\sum_{k=1}^m \omega_k F_{k+1},
\]
再取 $Z(N(\omega))$ 的前 $m$ 位得到 $\Fold_m(\omega)=\pi_m(Z(N(\omega)))\in X_m$，因此 $\Fold_m$ 显然可计算。

命题 \ref{prop:fold-basic} 的三个性质可直接由 Zeckendorf 唯一性推出：
\begin{itemize}
  \item \textbf{规范性：} $\pi_m(Z(\cdot))$ 输出的前 $m$ 位满足无相邻 $1$，故落在 $X_m$；
  \item \textbf{幂等性：}若 $\omega\in X_m$，则其加权值 $N(\omega)$ 已是合法 Zeckendorf 表示，唯一性迫使 $Z(N(\omega))$ 的前 $m$ 位等于 $\omega$；
  \item \textbf{满射性：}对任意 $x\in X_m$，取 $\omega=x$ 即有 $\Fold_m(\omega)=x$。
\end{itemize}

\begin{lemma}[Zeckendorf 范围上的双射]\label{lem:zeckendorf-range-bijection}
对任意 $m\ge 1$，若 $0\le N\le F_{m+2}-1$，则 $Z(N)$ 的非零位索引不超过 $m$，从而
\[
\pi_m(Z(N))=Z(N)\ \text{（作为长度 }m\text{ 的补零截断）}.
\]
因此映射 $N\mapsto \pi_m(Z(N))$ 在 $\{0,1,\dots,F_{m+2}-1\}$ 上给出到 $X_m$ 的双射，其逆映射为 $w\mapsto V_m(w)$。
\end{lemma}

\textbf{证明：}
若 $Z(N)$ 在第 $m+1$ 位有 $1$，则 $N\ge F_{m+2}$（因为该位权重即为 $F_{m+2}$ 且其余项非负），与 $N\le F_{m+2}-1$ 矛盾。故 $c_{m+1}=c_{m+2}=\cdots=0$，从而截断不丢信息并给出双射结论。

\subsection{附录 D：折叠多重度的渐近与“分辨率不足”}

平均折叠多重度
\[
d_m=\frac{|\Omega_m|}{|X_m|}=\frac{2^m}{F_{m+2}}
\]
满足
\[
d_m\sim C\left(\frac{2}{\varphi}\right)^m,\quad C=\frac{\sqrt5}{\varphi^2}.
\]
因此对固定窗口长度 $m$，观测将大量微态压缩为同一稳定类型；当 $m$ 增大时，稳定类型虽增多，但微态增长更快，从而“可区分细节”在有限分辨率下持续被折叠。这给出“规律不可读出可归因于分辨率不足”的数学版本：不可读出对应不同微态在折叠后不可区分，而稳定规律对应 $X_m$ 上的递归计数与不变分布仍可复核。

\subsection{附录 E：归一化运算的算法化实现（值层运算 + Zeckendorf 规范形）}

第 7 节中运算以 $\mathrm{Val}^{-1}$ 定义并由 Zeckendorf 唯一性保证良定义。其可计算实现可取最短闭环：
\begin{itemize}
  \item \textbf{加法：}给定 $c,d\in X_\infty$，先计算整数 $N=\mathrm{Val}(c)+\mathrm{Val}(d)$，再以附录 C 的 Zeckendorf 贪心算法求 $Z(N)$，输出 $c\oplus d=Z(N)$。
  \item \textbf{乘法：}同理取 $N=\mathrm{Val}(c)\mathrm{Val}(d)$，再取 Zeckendorf 规范形。
\end{itemize}
该实现不依赖任何模拟或外加结构，且终止性与唯一性均由 Zeckendorf 定理给出（\cite{Zeckendorf1972}）。

\subsection{附录 E$'$：折叠与归一化的复杂度界（按分辨率 $m$ 与按比特两种口径）}

本附录将正文中“可计算、可终止”的表述升级为显式复杂度界。为避免模型争议，我们同时给出两种等价口径：
\begin{itemize}
  \item \textbf{按分辨率口径：}把 $m$ 视为输入长度，允许在常数时间内执行对 $\{0,1,\dots,F_{m+3}\}$ 范围整数的基本算术（这是把大整数成本吸收进常数的 RAM 口径；适合强调“随分辨率线性步数”）。
  \item \textbf{按比特口径：}把整数按其二进制长度计成本。由于 $F_{m+3}\asymp \varphi^{m}$，有 $\log_2 F_{m+3}=\Theta(m)$，因此“分辨率口径”的线性步数对应于“比特口径”的多项式成本。
\end{itemize}

\begin{lemma}[权重与数值大小]\label{lem:bitlength-fib}
对任意 $m\ge 1$，Fibonacci 权重满足
\[
\sum_{k=1}^m F_{k+1}=F_{m+3}-1,
\]
因此对任意 $\omega\in\Omega_m$ 都有 $0\le N(\omega)\le F_{m+3}-1$。并且二进制长度满足
\[
\log_2(F_{m+3})=\Theta(m).
\]
\end{lemma}

\begin{proposition}[\texorpdfstring{$\Fold_m$}{Fold\_m} 的复杂度]\label{prop:fold-complexity}
给定分辨率 $m$ 的微态 $\omega\in\Omega_m$，折叠 $\Fold_m(\omega)$（定义 \ref{def:fold-word}）可在以下复杂度内计算：
\begin{itemize}
  \item \textbf{分辨率口径：}时间 $O(m)$，空间 $O(m)$（可选；若在线生成 Fibonacci 权重并流式输出规范形，可做到额外空间 $O(m)$ 存储输出、其余 $O(1)$）。
  \item \textbf{比特口径：}时间 $O(m^2)$ 比特运算，空间 $O(m)$ 个 $O(m)$ 比特整数。
\end{itemize}
\end{proposition}

\begin{proof}
先计算 $N(\omega)=\sum_{k=1}^m \omega_k F_{k+1}$：这是 $m$ 项加权求和，分辨率口径为 $O(m)$ 步。由引理 \ref{lem:bitlength-fib}，中间量二进制长度为 $\Theta(m)$，故比特口径下每次加法为 $O(m)$，总为 $O(m^2)$。

随后对 $N(\omega)$ 取 Zeckendorf 贪心规范形：每次选取不超过余量的最大 Fibonacci 权重并跳过相邻位。最高位索引不超过 $m+2$（引理 \ref{lem:bitlength-fib}），因此迭代次数 $O(m)$；分辨率口径为 $O(m)$ 步。比特口径下每次“比较/减法”成本为 $O(m)$，故总为 $O(m^2)$。最后截断取前 $m$ 位为 $O(m)$。
\end{proof}

\begin{proposition}[稳定加法与稳定乘法的复杂度]\label{prop:stable-arith-complexity}
设 $c,d\in X_\infty$ 仅有限个 $1$，并令
\[
m:=\max\{k:\ c_k=1\text{ 或 } d_k=1\}
\]
为输入的有效分辨率。则稳定加法 $c\oplus d$ 与稳定乘法 $c\otimes d$（定义 \ref{def:stable-add} 与 \ref{def:stable-mul}）可按附录 E 的“值层运算 + Zeckendorf 规范形”实现，并满足：
\begin{itemize}
  \item \textbf{分辨率口径：}加法时间 $O(m)$；乘法时间 $O(m)+T_\times$，其中 $T_\times$ 为值层整数乘法成本（在单位成本 RAM 下可视为常数或视实现而定）。
  \item \textbf{比特口径：}令 $b=\Theta(m)$ 为 $\mathrm{Val}(c),\mathrm{Val}(d)$ 的比特长度，并令 $M(b)$ 为 $b$ 比特整数乘法成本（例如学校乘法 $M(b)=O(b^2)$）。则
  \[
  \text{加法时间 }O(m^2),\qquad
  \text{乘法时间 }O(M(m)+m^2).
  \]
\end{itemize}
\end{proposition}

\begin{proof}
加法：先做值层加法 $N=\mathrm{Val}(c)+\mathrm{Val}(d)$，再做 Zeckendorf 规范化并输出。规范化成本由命题 \ref{prop:fold-complexity} 的分析给出（将输入从 $\omega\in\Omega_m$ 换为有限支撑配置并不改变最高权重索引阶）。比特口径下值层加法为 $O(b)=O(m)$，主项来自规范化的 $O(m^2)$。

乘法：值层乘法成本为 $M(b)=M(\Theta(m))$，再加 Zeckendorf 规范化 $O(m^2)$，得到所述界。
\end{proof}

\subsection{附录 G：多尺度一致性（restriction 逆系统与逆极限）}\label{app:multiscale-inverse-limit}

本附录补齐第 \ref{subsec:folding-multiscale} 小节的证明，严格化“分辨率 $m$ 不改变统一对象；不同 $m$ 的结果一致仅精细程度不同”的数学含义。

\begin{proof}[引理 \ref{lem:pi-functorial-golden} 的证明]
对任意 $(x_1,\dots,x_{m_3})\in X_{m_3}$，两边都是截断到前 $m_1$ 位，故逐点相等。对 $\pi_{m_2}=\pi_{m_2}(\cdot)$ 的一致性同理：先截断到前 $m_2$ 位，再截断到前 $m_1$ 位，等于直接截断到前 $m_1$ 位。
\end{proof}

\begin{proof}[定理 \ref{thm:inverse-limit-golden} 的证明]
定义映射
\[
\Phi:X_\infty\to \varprojlim X_m,\qquad
\Phi(c):=\bigl(\pi_m(c)\bigr)_{m\ge 1}.
\]
由引理 \ref{lem:pi-functorial-golden}，$(\pi_m(c))_{m\ge 1}$ 满足逆极限的一致性条件，故 $\Phi$ 良定义。

反向，给定一致族 $(x^{(m)})_{m\ge 1}\in\varprojlim X_m$，定义无穷序列 $c=(c_1,c_2,\dots)$ 为：令 $c_k:=x^{(m)}_k$（取任意 $m\ge k$）。一致性条件保证该定义与所取 $m$ 无关。并且每个 $x^{(m)}\in X_m$ 满足无相邻 $1$，从而 $c\in X_\infty$。

最后验证互逆：对 $c\in X_\infty$，由定义立得反向构造返回 $c$；对一致族 $(x^{(m)})$，其反向构造的 $c$ 满足 $\pi_m(c)=x^{(m)}$，故返回原一致族。因而 $\Phi$ 为双射，且构造显然自然。证毕。
\end{proof}

\subsection{附录 H：Kronecker 序列的一维星差异度上界（部分商控制）}\label{app:kronecker-discrepancy}

本附录为命题 \ref{prop:discrepancy-certificate} 中关于 Kronecker 序列差异度的定量断言提供一个可直接引用的标准形式。

\begin{theorem}[部分商控制的差异度上界]\label{thm:kronecker-discrepancy}
设 $\alpha\in\RR\setminus\QQ$，连分数展开为 $\alpha=[a_0;a_1,a_2,\dots]$，收敛分数为 $p_n/q_n$。对任意 $N\ge 1$，取 $n$ 使 $q_n\le N<q_{n+1}$。则对任意初相位 $x_0$ 的 Kronecker 序列 $x_t=\{x_0+t\alpha\}$，其星差异度满足估计
\[
D_N^*(\alpha)\ \le\ \frac{1}{N}\left(1+\sum_{j=1}^{n+1}a_j\right).
\]
\end{theorem}

\begin{remark}
该上界可由连分数逼近与三间隙（three-gap）结构导出；系统化表述可参见 \cite{KuipersNiederreiter1974}，以及对一维 Kronecker 序列的近期综述性证明路线 \cite{Goda2024QuasiUniformKronecker}。
\end{remark}

\begin{corollary}[常数型（badly approximable）给出的 \(O((\log N)/N)\)]\label{cor:kronecker-discrepancy-const-type}
若 $\alpha$ 的部分商有界：存在 $A\ge 1$ 使得 $a_j\le A$ 对所有 $j\ge 1$ 成立，则
\[
D_N^*(\alpha)=O\!\left(\frac{\log N}{N}\right).
\]
更具体地，定理 \ref{thm:kronecker-discrepancy} 给出
\[
D_N^*(\alpha)\ \le\ \frac{1+(n+1)A}{N},
\quad
q_n\le N<q_{n+1},
\]
并且 $q_n$ 至少指数增长，从而 $n=O(\log N)$。
\end{corollary}

\begin{corollary}[黄金分支的显式上界]\label{cor:kronecker-discrepancy-golden}
取 $\alpha=\varphi^{-1}$，则其连分数部分商满足 $a_j\equiv 1$。因此对任意 $N\ge 2$，
\[
D_N^*(\varphi^{-1})\ \le\ \frac{3+\left\lceil \frac{\log(\sqrt5\,N)}{\log \varphi}\right\rceil}{N}.
\]
\end{corollary}
