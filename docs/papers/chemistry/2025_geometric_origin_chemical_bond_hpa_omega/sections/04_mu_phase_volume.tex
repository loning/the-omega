\section{The geometric origin of \texorpdfstring{$\mu=m_p/m_e$}{mu}: internal phase volume and scale separation}
\label{sec:mu_geo}

Chemistry requires not only bound electronic states but also a robust separation between fast electronic and slow nuclear motion. This separation is controlled by the dimensionless proton--electron mass ratio $\mu=m_p/m_e$. In the HPA--$\Omega$ constant-geometry program, $\mu$ is again a \emph{target} dimensionless invariant of internal protocol geometry; the minimal closed model yields the phase-volume prediction $\mu_{\mathrm{geo}}$. We record the minimal closed-theory chain yielding
\begin{equation}
\mu_{\mathrm{geo}}=6\pi^5,
\end{equation}
following \cite{Ma2025PCG}.

\subsection{Mass ratios as phase-volume costs (closed layer)}
\begin{definition}[Mass-ratio invariants]\label{def:mass_ratio_chem}
For particle classes $i,j$, define the dimensionless ratio
\begin{equation}
\frac{m_i}{m_j}=\frac{I_i}{I_j},
\end{equation}
where $I_i$ is a protocol-geometric invariant induced by the internal defect/holonomy data of class $i$.
\end{definition}

\begin{axiom}[Mass invariants as phase-volume costs]\label{ax:mass_volume_chem}
For each particle class $i$, the invariant $I_i$ is given by a canonical phase volume of an internal compact manifold $\mathcal{M}_i$ (or a finite disjoint union), normalized so that the electron satisfies $I_e=1$:
\begin{equation}
I_i=\mathrm{Vol}(\mathcal{M}_i),
\qquad
\mathrm{Vol}\!\left(\bigsqcup_{a=1}^A \mathcal{M}_{i,a}\right)=\sum_{a=1}^A \mathrm{Vol}(\mathcal{M}_{i,a}).
\end{equation}
\end{axiom}

\subsection{Proton internal phase space (three-color sector sum)}
\begin{axiom}[Proton internal phase space]\label{ax:proton_phase_space_chem}
Model the proton as a sum over three identical color sectors ($N_c=3$) \cite{PeskinSchroeder1995}. Each color sector is assigned the compact phase space
\begin{equation}
\mathcal{M}_{q}\cong SO(3)\times SO(3)\times U(1),
\end{equation}
so that the proton internal space $\mathcal{M}_p$ is the disjoint union of three copies of $\mathcal{M}_q$.
\end{axiom}
\begin{remark}
At the level of canonical volumes, this choice is equivalent to $SO(3)\times SU(2)\times \mathbb{R}P^1$ because $\mathrm{Vol}(SO(3))\mathrm{Vol}(U(1))=\mathrm{Vol}(SU(2))\mathrm{Vol}(\mathbb{R}P^1)$. The convention above treats rotational sectors projectively while keeping the phase circle unquotiented \cite{Ma2025PCG}.
\end{remark}

\subsection{Theorem-level value and interface matching}
\begin{theorem}[Proton--electron mass ratio]\label{thm:mu_geo_chem}
Under Axioms~\ref{ax:mass_volume_chem}--\ref{ax:proton_phase_space_chem}, the geometric prediction for the proton--electron mass ratio is
\begin{equation}
\mu_{\mathrm{geo}}:=\frac{m_p}{m_e}=6\pi^5\approx 1836.1181087117.
\end{equation}
\end{theorem}
\begin{proof}
By Definition~\ref{def:mass_ratio_chem} and Axiom~\ref{ax:mass_volume_chem}, $m_p/m_e=I_p/I_e=I_p$ since $I_e=1$. By Axiom~\ref{ax:proton_phase_space_chem},
\begin{equation}
I_p=3\,\mathrm{Vol}(SO(3)\times SO(3)\times U(1)).
\end{equation}
Using $\mathrm{Vol}(SO(3))=\pi^2$ and $\mathrm{Vol}(U(1))=2\pi$ yields
\begin{equation}
I_p=3(\pi^2)(\pi^2)(2\pi)=6\pi^5.
\end{equation}
\end{proof}

Let $\mu_{\mathrm{exp}}$ denote the CODATA recommended value \cite{CODATA2022RMP}. Define the multiplicative interface matching input
\begin{equation}
s_\mu:=\frac{\mu_{\mathrm{exp}}}{\mu_{\mathrm{geo}}}.
\end{equation}
As emphasized in \cite{Ma2025PCG}, $s_\mu$ is not a per-system tuning knob but a single interface encoding nonperturbative QCD binding and radiative structure. In chemistry-facing applications, the key output is not the exact value of $\mu$ itself but the existence of a large hierarchy $\mu\gg 1$ and the induced expansion parameter $\mu^{-1/2}$.

\paragraph{Quantitative size.}
Using the CODATA 2022 central value quoted in \cite{Ma2025PCG,CODATA2022RMP}, the offset is
\[
\Delta\mu:=\mu_{\mathrm{geo}}-\mu_{\mathrm{exp}}\approx -3.456\times 10^{-2},
\qquad
\frac{\Delta\mu}{\mu_{\mathrm{exp}}}\approx -1.88\times 10^{-5},
\qquad
s_\mu\approx 1.0000188.
\]
