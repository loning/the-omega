\section{Appendix B: Proof Sketches of Key Propositions}

\subsection{B.1 Weyl Pair Has No Common Eigenvector}

Let $\alpha\notin\QQ$ and $U_{\mathrm{scan}}V=e^{2\pi i\alpha}VU_{\mathrm{scan}}$. If there exists nonzero $\psi$ that is an eigenvector of both, then
\[
U_{\mathrm{scan}}V\psi=uv\psi,\qquad
VU_{\mathrm{scan}}\psi=vu\psi.
\]
By the Weyl relation, $uv=e^{2\pi i\alpha}uv$, implying $e^{2\pi i\alpha}=1$, contradiction. Hence no common eigenvector exists.

\subsection{B.2 $|X_m|=F_{m+2}$}

For binary words of length $m$ with no adjacent $1$'s, classify by first bit: if first bit is $0$, the remainder is any element of $X_{m-1}$; if first bit is $1$, the second bit must be $0$, and the remainder is any element of $X_{m-2}$. Thus
\[
|X_m|=|X_{m-1}|+|X_{m-2}|,
\]
with initial values $|X_1|=2$, $|X_2|=3$, yielding $|X_m|=F_{m+2}$.

\subsection{B.3 Folding Average Degeneracy and Exponential Scaling}

From the surjection $\Fold_m:\{0,1\}^m\to X_m$, we have
\[
d_m=\frac{1}{|X_m|}\sum_{w\in X_m}\big|\Fold_m^{-1}(w)\big|=\frac{2^m}{|X_m|}=\frac{2^m}{F_{m+2}}.
\]
By Fibonacci asymptotics $F_{m+2}\asymp \varphi^{m}$, we get $d_m\asymp (2/\varphi)^m$.
