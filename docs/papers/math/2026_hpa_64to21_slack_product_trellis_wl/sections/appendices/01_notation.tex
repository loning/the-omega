
\section{记号表}

\begin{itemize}
  \item $n$：比特长度。
  \item $\Xn=\{0,1\}^n$：微观态集合。
  \item $\Zn=\{z\in\{0,1\}^n:\ z\text{ 不含 }11\}$：宏观稳定态集合。
  \item $\Head:\Xn\to \Zn$：折叠/段首算子（定义 \ref{def:Head}）。
  \item $\Tail:\Xn\to \Xn$：时间推进/段尾算子（定义 \ref{def:Tail}）。
  \item $\tau(x)$：微观态 $x$ 的时间深度（定义 \ref{def:tau}）。
  \item $s(z)$：宏观态 $z$ 的 slack 向量（定义 \ref{def:slack_vector}）。
  \item $K(z)=\abs{\Head^{-1}(z)}$：纤维大小（定义 \ref{def:complexity_triple}）。
  \item $\mathsf{H}(z)=\log K(z)$：时间信息量（搜索复杂度的一种刻画）。
  \item $D(z)=\max_j s_j$：最坏时间深度。
  \item $G_n$：trellis 图（定义 \ref{def:trellis}）。
  \item $Q_n$：宏观加权转移图（定义 \ref{def:Qn}）。
\end{itemize}
