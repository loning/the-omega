\section{Explicit 1D Fibonacci DQCA model and numerical protocol}
\label{app:fib_dqca}

This appendix specifies a concrete 1D Fibonacci (cut-and-project) Dirac quantum cellular automaton (DQCA) used as a checkable subsystem model for the quasicrystal program in Part~III. It also fixes the diagnostics (dispersion extraction, group velocities, and norm errors) used for the numerical evidence reported in Part~VI.\par
Throughout, $\phi=\tfrac{1+\sqrt5}{2}$ and $\alpha=\phi^{-2}$.

\subsection{Fibonacci chain as a cut-and-project set}
Fix an intercept (phason shift) $\rho\in[0,1)$ and define the internal coordinate
\begin{equation}
    u_j \equiv \{j\alpha+\rho\}\in[0,1),\qquad j\in\mathbb{Z},
\end{equation}
where $\{\cdot\}$ denotes the fractional part. Define the tile-type word $\sigma(j)\in\{L,S\}$ by
\begin{equation}
    \sigma(j)=
    \begin{cases}
    L,& u_j\in[0,1-\alpha),\\
    S,& u_j\in[1-\alpha,1).
    \end{cases}
\label{eq:fib_sigma_appendix}
\end{equation}
This is the standard Sturmian coding of an irrational rotation and yields the Fibonacci word for $\alpha=\phi^{-2}$ \cite{MorseHedlund1940,Lothaire2002}. Lemma~\ref{lemma:sturmian_frequencies} records the asymptotic symbol frequencies.
\paragraph{Remark (relation to scan--projection readout).}
Eq.~(\ref{eq:fib_sigma_appendix}) is the spatial analogue of the intrinsic scan--projection readout of Part~II (Axiom~O5): the index $j$ plays the role of the tick, $\rho$ is an initial phase/phason shift, and the binary output $\sigma(j)\in\{L,S\}$ is obtained by window projection of the rotation orbit.
\begin{lemma}[Symbol frequencies]
\label{lemma:sturmian_frequencies}
For irrational $\alpha$, the sequence $\sigma(j)$ defined by Eq.~(\ref{eq:fib_sigma_appendix}) has well-defined asymptotic symbol frequencies:
\begin{equation}
    \lim_{N\to\infty}\frac{1}{N}\#\{0\le j<N:\sigma(j)=S\}=\alpha,\qquad
    \lim_{N\to\infty}\frac{1}{N}\#\{0\le j<N:\sigma(j)=L\}=1-\alpha.
\end{equation}
\end{lemma}
\begin{proof}
The sequence $u_j=\{j\alpha+\rho\}$ is equidistributed in $[0,1)$ for irrational $\alpha$ (Weyl equidistribution). Therefore the asymptotic fraction of indices with $u_j$ in an interval equals the interval length. Applying this to the partition $[0,1)=[0,1-\alpha)\cup[1-\alpha,1)$ yields the stated frequencies.
\end{proof}
\paragraph{Corollary (mean coin/mass parameter).}
If the coin angles are scaled as $\theta_{L/S}=m_{L/S}\varepsilon$ with fixed $m_{L/S}$, then the spatial average of the local mass/mixing parameter satisfies
\begin{equation}
    \lim_{N\to\infty}\frac{1}{N}\sum_{j=0}^{N-1} m_{\sigma(j)} = (1-\alpha)m_L+\alpha m_S,
\end{equation}
with $\alpha=\phi^{-2}$ in the golden case.
The geometric embedding is obtained by assigning two edge lengths
\begin{equation}
    \ell_L\equiv \phi\,\varepsilon,\qquad \ell_S\equiv \varepsilon,
\end{equation}
and defining vertex positions recursively by
\begin{equation}
    r_0=0,\qquad r_{j+1}=r_j+\ell_{\sigma(j)}.
\end{equation}
The embedded set $\{r_j\}\subset\mathbb{R}$ is a 1D cut-and-project quasicrystal (Fibonacci chain).

\subsection{Local gates and the one-step unitary}
For continuum-limit diagnostics we work in the single-particle sector
\begin{equation}
    \mathcal{H}_{1p}=\ell^2(\mathbb{Z})\otimes\mathbb{C}^2,
\end{equation}
with internal basis states $\{|R\rangle,|L\rangle\}$.
Let the on-site (Long/Short) coin rotations be
\begin{equation}
    C_L\equiv e^{-i\theta_L\sigma_y},\qquad C_S\equiv e^{-i\theta_S\sigma_y}.
\end{equation}
Define the position-dependent coin operator
\begin{equation}
    C\equiv \sum_{j\in\mathbb{Z}} |j\rangle\langle j|\otimes C_{\sigma(j)}.
\end{equation}
Define the conditional nearest-neighbor shift
\begin{equation}
    S\equiv \sum_{j\in\mathbb{Z}}\Big(|j+1\rangle\langle j|\otimes|R\rangle\langle R| + |j-1\rangle\langle j|\otimes|L\rangle\langle L|\Big).
\end{equation}
The one-step Floquet operator is the coined quantum walk
\begin{equation}
    U \equiv S\,C.
\end{equation}
Unitarity is immediate from unitarity of $S$ (a permutation on the basis) and of $C$ (block-diagonal with unitary blocks).

\subsection{Phason shift and phason strain}
The intercept $\rho$ shifts the acceptance window in internal space and generates a family of locally indistinguishable Fibonacci chains (different global phason choices). A convenient strain model lets $\rho$ vary slowly with the site index $j$:
\begin{equation}
    \rho\to \rho(j),\qquad u_j\equiv \{j\alpha+\rho(j)\}.
\end{equation}
Define the discrete phason strain by the incremental change
\begin{equation}
    \eta_j \equiv \rho(j+1)-\rho(j),
\end{equation}
and its norms (used in the numerics) by
\begin{equation}
    \|\eta\|_{\infty}\equiv \sup_j |\eta_j|,\qquad
    \|\eta\|_{2;N}\equiv \left(\frac{1}{N}\sum_{j=0}^{N-1}\eta_j^2\right)^{1/2}.
\end{equation}
The unstrained Fibonacci chain corresponds to $\eta_j\equiv 0$. Small $\|\eta\|$ generates a controlled density of local phason flips (local changes of the L/S pattern).

\subsection{Periodic approximants and dispersion extraction}
To define a momentum-space diagnostic on an aperiodic texture we use periodic Fibonacci approximants. Let $F_0=0$, $F_1=1$, $F_{n+2}=F_{n+1}+F_n$ be Fibonacci numbers and set
\begin{equation}
    \alpha_n \equiv \frac{F_{n-1}}{F_n}\to \alpha.
\end{equation}
Define a length-$N$ periodic texture by taking $N\equiv F_n$ and replacing $\alpha$ with $\alpha_n$ in Eq.~(\ref{eq:fib_sigma_appendix}), so $\sigma_n(j+N)=\sigma_n(j)$. The finite-volume Floquet operator $U_N$ acts on $\mathbb{C}^{2N}$ and is block-circulant up to the coin texture.\par
For each quasi-momentum $k\in[-\pi,\pi)$, implement the Bloch boundary condition
\begin{equation}
    \psi(j+N)=e^{ikN}\psi(j),
\end{equation}
and compute the $2\times 2$ quasienergy branches $E^{(n)}_{\pm}(k)$ from the eigenphases of the resulting $2N\times 2N$ matrix representation of $U_N(k)$. The group velocity is defined by
\begin{equation}
    v^{(n)}_{\pm}(k)\equiv \frac{\partial E^{(n)}_{\pm}(k)}{\partial k},
\end{equation}
computed numerically by finite differences on a $k$ grid.

\subsection{Error norms}
Let $E_D(k)=\sqrt{(c_{\mathrm{eff}}k)^2+m_{\mathrm{eff}}^2}$ be the target Dirac dispersion in the long-wavelength regime, with $(c_{\mathrm{eff}},m_{\mathrm{eff}})$ fit from small-$k$ data. We define the dispersion error at cutoff $k_{\max}$ by
\begin{equation}
    \delta_E^{(n)}(k_{\max})\equiv \max_{|k|\le k_{\max}} \big|E^{(n)}_{+}(k)-E_D(k)\big|.
\end{equation}
For wavepacket propagation, given an initial normalized state $|\psi_0\rangle\in\mathcal{H}_{1p}$ and integer time $T$, define
\begin{equation}
    \epsilon_{\mathrm{prop}}(T)\equiv \|U^T|\psi_0\rangle-|\psi_D(t)\rangle\|,\qquad t\equiv T\varepsilon,
\end{equation}
where $|\psi_D(t)\rangle$ is the corresponding Dirac evolution under a chosen continuum Hamiltonian $H_D$ (with the same fitted parameters and the same initial envelope). In the numerical implementation we report both $\epsilon_{\mathrm{prop}}(T)$ and its scaling with $(\varepsilon,k_{\max},\|\eta\|)$.

\section{Projection filtering and a doubler gap bound (1D Fibonacci case)}
\label{app:doubling_filter}
This appendix spells out the acceptance-window Fourier transform and a simple mass-gap bound for the projection-filter mechanism discussed in Section~\ref{sec:doubling_filter}, specializing to the standard 1D Fibonacci cut-and-project construction from $\mathbb{Z}^2$.\par

\subsection{Interval window and explicit Fourier transform}
For the Fibonacci chain, the internal space is one-dimensional and the acceptance window $W\subset\mathbb{R}$ can be taken as an interval of length $R>0$ (up to an affine change of coordinates). Let $W=[-R/2,R/2]$ and define the normalized Fourier transform
\begin{equation}
    \widehat{W}(q)\equiv \frac{1}{R}\int_{-R/2}^{R/2} e^{-iqy}\,dy
    = e^{-iq\cdot 0}\,\frac{\sin(qR/2)}{qR/2}
    \equiv \mathrm{sinc}\!\left(\frac{qR}{2}\right),
\end{equation}
with $\mathrm{sinc}(x)\equiv \sin(x)/x$ and $\widehat{W}(0)=1$.
The elementary bound $|\sin x|\le 1$ implies
\begin{equation}
    |\widehat{W}(q)| \le \min\left\{1,\ \frac{2}{|q|R}\right\}.
\label{eq:window_sinc_bound}
\end{equation}

\subsection{A concrete lower bound for would-be doubler momenta}
In the cut-and-project embedding picture one decomposes embedding momentum $k\in(\mathbb{R}/(2\pi/\varepsilon)\mathbb{Z})^2$ into components $(k_\parallel,k_\perp)$ along the physical and internal directions. For a ``mirror'' shift by $\pi/\varepsilon$ in one embedding coordinate, Diophantine irrationality of the slope implies that the induced internal component is not arbitrarily small; in the noble (golden) case one can take a bound of the form
\begin{equation}
    |q_{\mathrm{doubler}}|\equiv |k_\perp|\ge \frac{c_0}{\varepsilon}
\end{equation}
for some constant $c_0>0$ that depends on the embedding convention and on the chosen family of mirror shifts. For golden slopes, such lower bounds are a manifestation of bad Diophantine approximability (Appendix~\ref{app:homogenization}).\par

\subsection{Mass gap from the window bound}
With the projection-filter mass term $\mathcal{M}(q)=\mu(1-\widehat{W}(q))$ from Eq.~(\ref{eq:proj_filter_mass}), the sinc bound (\ref{eq:window_sinc_bound}) implies that for $|q|\ge c_0/\varepsilon$ one has $|\widehat{W}(q)|\le 2\varepsilon/(c_0R)$. Therefore
\begin{equation}
    |\mathcal{M}(q_{\mathrm{doubler}})| \ge \mu\left(1-\frac{2\varepsilon}{c_0R}\right).
\end{equation}
Choosing the UV normalization $\mu=\mu_0/\varepsilon$ yields
\begin{equation}
    |\mathcal{M}(q_{\mathrm{doubler}})| \ge \frac{\mu_0}{\varepsilon}\left(1-\frac{2\varepsilon}{c_0R}\right),
\end{equation}
which is of order $1/\varepsilon$ for small $\varepsilon$, as claimed.\par

\subsection{Numerical cross-check on periodic approximants}
For the periodic approximants used in Appendix~\ref{app:fib_dqca}, a would-be doubler would appear as a near-zero quasienergy branch at Brillouin-edge momenta. The fixed-parameter script \texttt{scripts/generate\_fib\_dqca\_figures.py} regenerates the dispersion figures and can be extended to directly evaluate the smallest quasienergy magnitude at $k=\pi$ as a diagnostic for the doubler sector.

\section{Dirac limit and homogenization framework}
\label{app:homogenization}

We provide a proof of Theorem~\ref{prop:dirac_limit} for the translation-invariant 1D coined walk (the baseline Dirac QCA). We then record definitions and averaging statements used when discussing cut-and-project (quasicrystal) textures and phason fields.

\subsection{Proof of Theorem~\ref{prop:dirac_limit} (translation-invariant 1D case)}
Let $\mathcal{H}_{1p}=\ell^2(\mathbb{Z})\otimes\mathbb{C}^2$ and fix $\varepsilon>0$. Define the shift $S$ on basis states by
\begin{equation}
    S\big(|j\rangle\otimes|R\rangle\big)=|j+1\rangle\otimes|R\rangle,\qquad
    S\big(|j\rangle\otimes|L\rangle\big)=|j-1\rangle\otimes|L\rangle,
\end{equation}
and the (translation-invariant) coin by $C_\varepsilon\equiv e^{-i\varepsilon m\sigma_y}$ with fixed $m\in\mathbb{R}$. The one-step unitary is $U_\varepsilon=S\,C_\varepsilon$.

Write the lattice quasi-momentum variable as $k\in[-\pi,\pi)$ and set the physical momentum $p\equiv k/\varepsilon$. In momentum representation the shift is diagonal,
\begin{equation}
    S(k)=e^{-ik\sigma_z}=e^{-i\varepsilon p\,\sigma_z},
\end{equation}
so the Bloch-reduced step is
\begin{equation}
    U_\varepsilon(k)=S(k)\,C_\varepsilon
    =e^{-i\varepsilon p\,\sigma_z}\,e^{-i\varepsilon m\,\sigma_y}.
\end{equation}
For $|p|\le p_{\max}$ and $\varepsilon$ sufficiently small, $U_\varepsilon(k)$ lies in a neighborhood of the identity on which the principal matrix logarithm is analytic. Define the effective Hamiltonian by
\begin{equation}
    H_{\mathrm{eff}}^{(\varepsilon)}(p)\ \equiv\ \frac{i}{\varepsilon}\log U_\varepsilon(\varepsilon p).
\end{equation}
Using the Baker--Campbell--Hausdorff expansion and $\|[\sigma_z,\sigma_y]\|=2\|\sigma_x\|=2$, one obtains the uniform bound
\begin{equation}
    \big\|H_{\mathrm{eff}}^{(\varepsilon)}(p) - (p\,\sigma_z+m\,\sigma_y)\big\|
    \le C_0\,\varepsilon\,(p_{\max}^2+m^2),
\end{equation}
for some absolute constant $C_0$ (the bound is uniform over $|p|\le p_{\max}$).

Let $t\ge 0$ be fixed and set $T\equiv\lfloor t/\varepsilon\rfloor$. For each $p$ the evolution after $T$ steps satisfies
\begin{equation}
    U_\varepsilon(\varepsilon p)^T = \exp\!\left(-it\,H_{\mathrm{eff}}^{(\varepsilon)}(p)\right)\,+\,\mathcal{O}(\varepsilon),
\end{equation}
where the $\mathcal{O}(\varepsilon)$ term accounts for the rounding error between $T\varepsilon$ and $t$. Since the matrix exponential is Lipschitz on bounded sets, for $|p|\le p_{\max}$ one has
\begin{equation}
    \left\|\exp\!\left(-it\,H_{\mathrm{eff}}^{(\varepsilon)}(p)\right)-\exp\!\left(-it\,(p\,\sigma_z+m\,\sigma_y)\right)\right\|
    \le t\,\big\|H_{\mathrm{eff}}^{(\varepsilon)}(p) - (p\,\sigma_z+m\,\sigma_y)\big\|\,e^{t\,\|H_{\mathrm{eff}}^{(\varepsilon)}(p)\|}.
\end{equation}
Combining the uniform bounds above yields
\begin{equation}
    \sup_{|p|\le p_{\max}}\left\|U_\varepsilon(\varepsilon p)^T-\exp\!\left(-it\,(p\,\sigma_z+m\,\sigma_y)\right)\right\|
    \le C(t,p_{\max})\,\varepsilon.
\end{equation}
For a band-limited state $|\psi_0\rangle$ with support $|p|\le p_{\max}$, Parseval's identity transfers this uniform bound to the full norm on $\mathcal{H}_{1p}$, proving Theorem~\ref{prop:dirac_limit} with $H_D=-i\sigma_z\partial_x+m\sigma_y$.

\subsection{Setup and Definitions}
Let $\Lambda \subset \mathbb{R}^d$ be a cut-and-project set defined by the projection $P: \mathbb{R}^n \to \mathbb{R}^d$ ($n > d$) of the strip $S = \mathbb{R}^d \times W$, where $W \subset \mathbb{R}^{n-d}$ is a compact window.
Sites are $x \in \Lambda$. The unitary evolution is $\hat{U} = S \cdot C$, with shift $S = \sum_x \sum_j |x+e_j\rangle\langle x| \otimes P_j$ and coin $C = \bigoplus_x C_x$.

\begin{definition}[Phason coordinate and phason strain]
For a cut-and-project point $x=P_{\parallel}(\vec{n})$ with $\vec{n}\in\mathbb{Z}^n$, its internal (perpendicular) coordinate is $x_{\perp}\equiv P_{\perp}(\vec{n})\in W$. Since the window $W$ is compact, $x_{\perp}$ is bounded uniformly by $\mathrm{diam}(W)$; this bound is a property of the internal \emph{coordinate}, not of any strain field.

A \emph{phason displacement field} is a (possibly slowly varying) internal shift $u(x)\in\mathbb{R}^{n-d}$ that locally replaces the window by $W+u(x)$ and hence parametrizes deviations from a perfect linear cut. The corresponding \emph{phason strain} is the spatial variation of this displacement, measured for instance by $\|\nabla u\|$ in continuum descriptions or by discrete differences (e.g.\ the 1D strain $\eta_j$ in Appendix~\ref{app:fib_dqca}). A perfect cut corresponds to $u\equiv\mathrm{const}$ and hence vanishing strain.
\end{definition}

\begin{definition}[Finite-window structure factor (diffraction diagnostic)]
Let $\Lambda\subset\mathbb{R}^d$ be a discrete set and let $B_R\subset\mathbb{R}^d$ be a ball of radius $R$. The (normalized) finite-window structure factor is
\begin{equation}
    S_R(k)\ \equiv\ \frac{1}{|\Lambda\cap B_R|}\left|\sum_{x\in\Lambda\cap B_R} e^{-i\,k\cdot x}\right|^2,\qquad k\in\mathbb{R}^d.
\end{equation}
For regular cut-and-project (model) sets, $S_R(k)$ converges (in the appropriate weak sense) to a pure-point diffraction measure whose Bragg spectrum is determined by the dual lattice and the window function; see, e.g., \cite{Hof1995}. In this manuscript, ``controlling the structure factor'' refers to controlling this diffraction content (e.g.\ location and weights of dominant peaks) in the regime relevant for long-wavelength propagators.
\end{definition}

\begin{lemma}[Ergodic Averaging]
\label{lemma:ergodic}
Let $f: \mathbb{R}^{n-d} \to \mathbb{C}$ be a Riemann-integrable function on the internal space. For an irrational cut (where the projection of $\mathbb{Z}^n$ is dense in $W$), the lattice sum converges to the integral:
\begin{equation}
    \lim_{R \to \infty} \frac{1}{|\Lambda \cap B_R|} \sum_{x \in \Lambda \cap B_R} f(x_\perp)
    = \frac{1}{\mathrm{Vol}(W)} \int_W f(y)\,dy.
\end{equation}
This follows from Weyl-type equidistribution for cut-and-project sets; see, e.g., \cite{Hof1995}.
\end{lemma}
\noindent\textit{Remark.} Lemma~\ref{lemma:ergodic} is the cut-and-project analogue of the orbit-trace identity recorded in Appendix~\ref{app:orbit_calculus}.

\subsection{Homogenization remarks (quasicrystal extension)}
For coined walks on regular lattices, rigorous long-wavelength Dirac limits (including variable-mass and gauge-coupled extensions) are available in the QW/QCA literature; see, e.g., \cite{Arrighi2014,Dariano2014,Bisio2015}. Extending such results to cut-and-project substrates requires controlling the quasicrystal structure factor together with the spatial variation of internal-coordinate (phason) fields. In this manuscript we use the ergodic averaging statement (Lemma~\ref{lemma:ergodic}) and explicit 1D diagnostics (Appendix~\ref{app:fib_dqca}) as the checkable backbone of the quasicrystal program; a full higher-dimensional homogenization proof is left for future work.

\subsection{Diophantine property (golden case)}
Quantitative control of near-commensurabilities in quasiperiodic constructions is often organized by Diophantine (badly-approximable) bounds. For the golden ratio $\phi=\tfrac{1+\sqrt5}{2}$, continued-fraction theory gives the sharp estimate
\begin{equation}
    \left| \phi - \frac{p}{q} \right|\ \ge\ \frac{1}{\sqrt{5}\,q^2}\qquad (p,q\in\mathbb{Z},\ q\ge 1),
\end{equation}
with equality attained along the Fibonacci convergents. This arithmetic rigidity underlies the Fibonacci return-time structure discussed in Appendix~\ref{app:three_gap} and the Zeckendorf/Ostrowski digit constraints used in Part~II and Appendix~\ref{app:ostrowski}.

\subsection{Three-gap structure and Fibonacci return times}
\label{app:three_gap}
The same bad-approximability property that controls the Diophantine bounds above also governs the arithmetic structure of intrinsic phase-orbit readouts (Part~II), modeled by an irrational rotation $x_{t+1}=x_t+\alpha\ (\mathrm{mod}\ 1)$ with $\alpha=\phi^{-2}$. A classical result (the Three-Gap/Three-Distance Theorem) states that for any irrational $\alpha$, the ordered set of points $\{n\alpha\}_{n=0}^{N-1}$ partitions the circle into at most three distinct gap lengths \cite{Slater1967}.
Moreover, when $N$ equals a denominator $q_k$ of a continued-fraction convergent of $\alpha$, the partition exhibits a particularly rigid two-gap structure, and the closest returns occur at such $q_k$.

For the golden case $\alpha=\phi^{-2}$, the continued fraction is eventually constant, so the convergent denominators are Fibonacci numbers (up to an index shift). Consequently, the strongest near-returns and the most uniform finite sampling of the phase orbit occur on Fibonacci time scales. This provides a concrete number-theoretic underpinning for the Fibonacci ``resonant eras'' discussed in Part~VI.

\subsection{Torus rotations from finite-dimensional spectra: aperiodicity, equidistribution, recurrence}
\label{app:torus_rotation}
Several claims in Part~VI can be reduced to standard facts about translations on tori. We record them in a form adapted to finite-dimensional unitary spectra.

\begin{proposition}[Projective evolution as a torus rotation]
\label{prop:projective_torus}
Let $U$ be a unitary on a finite-dimensional Hilbert space with an eigenbasis $\{|n\rangle\}_{n=0}^{d-1}$ and eigenvalues $U|n\rangle = e^{-i2\pi \alpha_n}|n\rangle$, where $\alpha_n\in\mathbb{R}/\mathbb{Z}$. For a state $|\psi\rangle=\sum_{n} c_n|n\rangle$ with support $S\equiv\{n:\,c_n\neq 0\}$, define the relative-phase vector
\begin{equation}
    \omega\equiv (\alpha_n-\alpha_{n_0})_{n\in S\setminus\{n_0\}}\ \in\ (\mathbb{R}/\mathbb{Z})^{|S|-1},
\end{equation}
for any fixed $n_0\in S$. Then the projective orbit $[U^t|\psi\rangle]$ is completely determined by the torus rotation $t\mapsto t\,\omega\ (\mathrm{mod}\ 1)$.
\end{proposition}
\begin{proof}
Write $U^t|\psi\rangle=\sum_{n\in S} c_n e^{-i2\pi t\alpha_n}|n\rangle$. In projective space, the global phase $e^{-i2\pi t\alpha_{n_0}}$ is irrelevant, so the ray is determined by the relative phases $e^{-i2\pi t(\alpha_n-\alpha_{n_0})}$ for $n\in S\setminus\{n_0\}$, i.e.\ by $t\,\omega$.
\end{proof}

\begin{proposition}[No exact repetition criterion]
\label{prop:aperiodic_criterion}
With the notation above, the projective orbit is periodic (i.e.\ there exists $T\ge 1$ such that $[U^T|\psi\rangle]=[|\psi\rangle]$) if and only if $T\,\omega=0$ in $(\mathbb{R}/\mathbb{Z})^{|S|-1}$, equivalently $T(\alpha_n-\alpha_{n_0})\in\mathbb{Z}$ for all $n\in S$.
In particular, if at least one difference $\alpha_n-\alpha_{n_0}$ with $c_n\neq 0$ is irrational, then the orbit has no exact period.
\end{proposition}
\begin{proof}
The condition $[U^T|\psi\rangle]=[|\psi\rangle]$ means $U^T|\psi\rangle=e^{i\theta}|\psi\rangle$ for some $\theta$. Comparing coefficients in the eigenbasis gives $e^{-i2\pi T\alpha_n}=e^{i\theta}$ for all $n\in S$, i.e.\ $e^{-i2\pi T(\alpha_n-\alpha_{n_0})}=1$ for all $n\in S$, which is equivalent to the stated integer condition.
\end{proof}

\begin{theorem}[Equidistribution on the orbit closure]
\label{thm:equidistribution_closure}
Let $\omega\in(\mathbb{R}/\mathbb{Z})^m$ and consider the translation $T_\omega:x\mapsto x+\omega$ on the torus $\mathbb{T}^m$. Let $H\equiv \overline{\{t\omega:\,t\in\mathbb{Z}\}}\subset\mathbb{T}^m$ be the orbit closure of $0$ (a subtorus). Then for any continuous $f$ on $H$ and any $x\in H$,
\begin{equation}
    \lim_{N\to\infty}\frac{1}{N}\sum_{t=0}^{N-1} f(x+t\omega) \;=\; \int_H f\,d\mu_H,
\end{equation}
where $\mu_H$ is the normalized Haar measure on $H$ \cite{KuipersNiederreiter1974,Weyl1916}.
\end{theorem}
\begin{proof}
By Fourier analysis on the compact abelian group $H$, it suffices to test characters $\chi$ of $H$. For a character $\chi$, the average is
\begin{equation}
    \frac{1}{N}\sum_{t=0}^{N-1}\chi(x+t\omega)=\chi(x)\,\frac{1}{N}\sum_{t=0}^{N-1}\chi(\omega)^t.
\end{equation}
If $\chi(\omega)\neq 1$ the geometric-series average tends to $0$ as $N\to\infty$; if $\chi(\omega)=1$, then $\chi$ is constant on the dense subgroup generated by $\omega$ and hence is the trivial character on $H$. Therefore the only surviving Fourier mode is the constant one, implying convergence to the Haar integral.
\end{proof}

\begin{proposition}[Torus translations are not mixing]
\label{prop:torus_not_mixing}
Let $T_\omega$ be a translation on a torus $\mathbb{T}^m$ by a nonzero $\omega$. Then $T_\omega$ is not mixing with respect to Haar measure. In particular, there exist bounded observables $f,g$ with zero mean such that the correlation $\langle f\circ T_\omega^t, g\rangle$ does not converge to $0$ as $t\to\infty$.
\end{proposition}
\begin{proof}
Take a nontrivial character $\chi_k(x)\equiv e^{2\pi i k\cdot x}$ with $k\in\mathbb{Z}^m\setminus\{0\}$. Let $f=\chi_k$ and $g=\overline{\chi_k}$. Both have zero Haar mean. Then
\begin{equation}
    \langle f\circ T_\omega^t, g\rangle
    = \int_{\mathbb{T}^m} e^{2\pi i k\cdot (x+t\omega)}\,e^{-2\pi i k\cdot x}\,dx
    = e^{2\pi i t\,k\cdot \omega},
\end{equation}
which has modulus $1$ for all $t$ and therefore cannot converge to $0$.
\end{proof}

\begin{proposition}[Quantitative recurrence bound (Dirichlet approximation)]
\label{prop:dirichlet_recurrence}
Let $\omega\in\mathbb{R}^m$ and let $\|\cdot\|_{\mathbb{T}^m}$ denote distance to $0$ in $\mathbb{T}^m$. Then for any integer $Q\ge 1$ there exists $1\le t\le Q^m$ such that
\begin{equation}
    \|t\,\omega\|_{\mathbb{T}^m} \le \frac{1}{Q}.
\end{equation}
Consequently, for any $\varepsilon>0$ there exists $t=\mathcal{O}(\varepsilon^{-m})$ such that $t\omega$ returns within $\varepsilon$ of $0$.
\end{proposition}
\begin{proof}
This is the standard simultaneous Dirichlet approximation theorem: among the $Q^m+1$ points $\{t\omega\}_{t=0}^{Q^m}$ in $[0,1)^m$, two must lie in the same cube of side length $1/Q$, so their difference $t\omega$ is within $1/Q$ of an integer vector.
\end{proof}

\paragraph{Application to the golden-spectrum unitary.}
For the diagonal golden-spectrum ansatz of Part~VI, the projective dynamics is a torus translation of the form above with a frequency vector determined by the eigenphase differences $\alpha_n-\alpha_{n_0}$. Proposition~\ref{prop:aperiodic_criterion} gives a simple sufficient condition for absence of exact periods, while Theorem~\ref{thm:equidistribution_closure} and Proposition~\ref{prop:dirichlet_recurrence} justify equidistribution on the orbit closure and the existence of arbitrarily close returns (with polynomial recurrence-time bounds in the effective torus dimension). In the golden case, continued-fraction optimality further organizes the \emph{best} near-returns on Fibonacci time scales, as summarized in Appendix~\ref{app:three_gap}.

\begin{corollary}[Recurrence obstructs strictly monotone continuous ``complexity'' along a fixed finite spectrum]
\label{cor:recurrence_no_monotone}
Let $U$ be a finite-dimensional unitary and let $|\psi\rangle$ be a state whose projective orbit has arbitrarily close returns (as in Proposition~\ref{prop:dirichlet_recurrence} for the associated torus rotation). Then no continuous functional $C:\mathbb{P}(\mathcal{H})\to\mathbb{R}$ can be strictly increasing along the entire orbit $\{[U^t|\psi\rangle]\}_{t\ge 0}$.
\end{corollary}
\begin{proof}
Arbitrarily close returns mean that there exists a sequence $t_k\to\infty$ with $[U^{t_k}|\psi\rangle]\to[|\psi\rangle]$ in $\mathbb{P}(\mathcal{H})$. By continuity, $C([U^{t_k}|\psi\rangle])\to C([|\psi\rangle])$. A strictly increasing sequence cannot have a subsequence that converges back to its initial value, so $C$ cannot be strictly increasing along all $t$.
\end{proof}

\subsection{Golden-mean constraint, capacity, and Zeckendorf coding}
\label{app:golden_mean_capacity}
Finite-information locality suggests that operational discretizations of intrinsic readouts should obey simple admissibility constraints. A canonical example is the \emph{golden-mean constraint} on binary strings (no adjacent excitations): for $\mathbf{b}=(b_1,\dots,b_L)\in\{0,1\}^L$ require $b_i b_{i+1}=0$. The number of admissible strings satisfies a Fibonacci recursion and grows as
\begin{equation}
    N(L)=F_{L+2}\sim \frac{\phi^{L+2}}{\sqrt{5}},
\end{equation}
so the topological entropy is $h_{\mathrm{top}}=\lim_{L\to\infty}\frac{1}{L}\ln N(L)=\ln\phi$, and the corresponding capacity is
\begin{equation}
    C=\log_2\phi \approx 0.6942\ \text{bits/site}.
\end{equation}
Zeckendorf's theorem gives the matching uniqueness statement in arithmetic form: every integer admits a unique representation as a sum of non-consecutive Fibonacci numbers, which is equivalent to the golden-mean admissibility condition on digits \cite{Zeckendorf1972,AlloucheShallit2003}. In this sense, the Zeckendorf/Ostrowski representation is the canonical discretization compatible with Fibonacci combinatorics and finite-resolution constraints in the golden construction.

\section{Ostrowski numeration, Sturmian coding, and Zeckendorf time}
\label{app:ostrowski}

This appendix provides a concise proof of the Zeckendorf specialization used in Proposition~\ref{prop:ostrowski_zeckendorf} and summarizes the general Ostrowski framework associated with an irrational rotation.

\subsection{Ostrowski numeration associated with an irrational slope}
Let $\alpha\in(0,1)\setminus\mathbb{Q}$ have continued fraction expansion $\alpha=[0;a_1,a_2,\dots]$ with convergents $p_n/q_n$. Define $q_{-1}\equiv 0$, $q_0\equiv 1$ and
\begin{equation}
    q_{n+1}=a_{n+1}q_n+q_{n-1}\qquad (n\ge 0).
\end{equation}
The \emph{Ostrowski representation} associated with $\alpha$ states that every integer $t\in\mathbb{Z}_{\ge 0}$ admits a unique expansion
\begin{equation}
    t=\sum_{n=0}^{m} b_n q_n,
\end{equation}
with digits satisfying $0\le b_0<a_1$, $0\le b_n\le a_{n+1}$ for $n\ge 1$, and the admissibility constraint that if $b_n=a_{n+1}$ then $b_{n-1}=0$ (see, e.g., \cite{AlloucheShallit2003}).

\subsection{Golden case and the Zeckendorf rule}
For the golden-slope choice $\alpha=\phi^{-2}$ one has $\alpha=[0;2,1,1,1,\dots]$, so $a_1=2$ and $a_n=1$ for $n\ge 2$. The recurrence above then yields
\begin{equation}
    q_n = F_{n+2}\qquad (n\ge 0),
\end{equation}
with Fibonacci numbers defined by $F_0=0$, $F_1=1$, $F_{n+2}=F_{n+1}+F_n$. The Ostrowski digit constraints therefore reduce to $b_n\in\{0,1\}$ with no adjacent ones, exactly the Zeckendorf admissibility rule used in Proposition~\ref{prop:ostrowski_zeckendorf}.

\subsection{Proof of Zeckendorf's theorem (existence and uniqueness)}
\label{app:zeckendorf_proof}
\begin{theorem}[Zeckendorf]
Every $t\in\mathbb{Z}_{\ge 0}$ admits a unique representation
\begin{equation}
    t=\sum_{k=0}^{m} b_k F_{k+2},\qquad b_k\in\{0,1\},\qquad b_k b_{k+1}=0.
\end{equation}
\end{theorem}
\begin{proof}
\textbf{Existence.} For $t=0$ the empty sum works. For $t>0$, let $n\ge 2$ be maximal with $F_n\le t$. Set $t_1\equiv t-F_n$. Since $F_{n+1}=F_n+F_{n-1}$, maximality implies $t< F_{n+1}$ and hence $t_1<t-F_n< F_{n-1}$. Therefore any further Fibonacci summand in a representation of $t_1$ has index at most $n-2$, so it cannot be consecutive with $F_n$. Iterating the same greedy step on the remainder terminates and produces a representation with no consecutive summands.

\textbf{Uniqueness.} We use the bound that the maximal sum representable using Fibonacci numbers $\{F_2,\dots,F_{m+1}\}$ without consecutive indices equals $F_{m+2}-1$. This follows by induction: for $m=0$ the maximum is $0=F_2-1$, and for $m\ge 1$ the maximum either excludes $F_{m+1}$ (giving at most $F_{m+1}-1$ by induction) or includes $F_{m+1}$ and must exclude $F_m$, giving at most $F_{m+1}+(F_m-1)=F_{m+2}-1$.

Now suppose two admissible representations of the same $t$ exist, with digit strings $(b_k)$ and $(b'_k)$ not identical. Let $m$ be the largest index with $b_m\ne b'_m$. Without loss of generality $b_m=1$ and $b'_m=0$. Then the difference between the two sums is at least $F_{m+2}$ minus a sum of admissible terms from $\{F_2,\dots,F_{m+1}\}$, whose magnitude is at most $F_{m+2}-1$ by the bound above. Hence the difference cannot vanish, a contradiction. Therefore the representation is unique.
\end{proof}

\section{Orbit calculus and finite-part regularization}
\label{app:orbit_calculus}
This appendix records the orbit-calculus conventions used when relating scan-generated discrete data to continuum functionals, in accordance with Axiom~O8. The minimal template is an irrational circle rotation; the same logic extends to torus translations and to cut-and-project dynamical systems via Haar averages on orbit closures (Appendix~\ref{app:torus_rotation}).

\subsection{Orbit difference and orbit trace (irrational rotation template)}
Let $X=\mathbb{R}/\mathbb{Z}$ and fix $\alpha\in(0,1)\setminus\mathbb{Q}$. Define the rotation
$$
T_\alpha(x):=x+\alpha\pmod{1}.
$$
For a sufficiently regular $f:X\to\mathbb{C}$ define the orbit difference operator
$$
(\mathcal{D}_\alpha f)(x):=\frac{f(x+\alpha)-f(x)}{\alpha}.
$$
This is an intrinsic discrete derivative surrogate along the scan step.\par
Define the orbit average (a discrete integral surrogate) by
$$
A_N(f;x_0):=\frac{1}{N}\sum_{t=0}^{N-1} f(T_\alpha^t(x_0)).
$$

\begin{theorem}[Orbit trace for an irrational rotation]
If $f$ is continuous and $\alpha\notin\mathbb{Q}$, the limit
$$
\mathrm{Tr}_\alpha(f):=\lim_{N\to\infty}A_N(f;x_0)
$$
exists, is independent of $x_0$, and equals the space average with respect to Lebesgue measure $\mu$ on $X$:
$$
\mathrm{Tr}_\alpha(f)=\int_X f(x)\,d\mu(x).
$$
\end{theorem}
\begin{proof}[Reference]
This is a standard consequence of unique ergodicity/equidistribution of irrational rotations; see, e.g., \cite{Weyl1916,KuipersNiederreiter1974}.
\end{proof}

\subsection{Abel finite part for divergent orbit sums}
Let $u_t:=f(T_\alpha^t(x_0))$ be the orbit sample sequence (not assumed absolutely summable). Define its Abel generating function
$$
S_f(r):=\sum_{t=0}^{\infty} u_t r^t,\qquad 0<r<1.
$$
If $f$ is bounded then $S_f(r)$ converges absolutely for $0<r<1$. If as $r\uparrow 1$ the function admits an asymptotic expansion of the form
$$
S_f(r)=\sum_{j=1}^{J} \frac{c_j}{(1-r)^j} + C + o(1),
$$
we define the orbit finite part by
$$
\mathrm{FP}_\alpha(f):=C.
$$

\begin{lemma}[Uniqueness of the finite part]
If $S_f(r)$ admits two expansions of the above form with constants $C$ and $C'$, then $C=C'$.
\end{lemma}
\begin{proof}
Subtracting the two expansions gives
$$
0=\sum_{j=1}^{J''}\frac{d_j}{(1-r)^j}+(C-C')+o(1),
$$
as $r\uparrow 1$, where $J''=\max\{J,J'\}$ and $d_j$ are coefficient differences (missing coefficients treated as $0$). Multiplying by $(1-r)^{J''}$ and letting $r\uparrow 1$ yields $d_{J''}=0$. Repeating inductively gives $d_j=0$ for all $j\ge 1$, hence $C-C'+o(1)=0$ and therefore $C=C'$.
\end{proof}

\noindent
In this manuscript, whenever a constant term depends on a regularization choice, the Abel finite-part convention above (together with its natural extensions beyond the circle template) is taken as the normalization prescribed by Axiom~O8.

\section{Variational Derivation of Informational Gravity}
\label{app:action_proof}

We rigorously derive the field equations from the Principle of Least Action.

\subsection{Assumptions and Definitions}
\begin{enumerate}
    \item \textbf{Metric Signature}: We use the "mostly plus" convention $(-+++)$.
    \item \textbf{Connection}: The connection $\Gamma^\lambda_{\mu\nu}$ is the Levi-Civita connection (torsion-free).
    \item \textbf{Scalar Identification}: Let $\varrho$ denote a coarse-grained information/probability density and define the Fisher-amplitude scalar $\chi\equiv \sqrt{\varrho}$ (equivalently $\varphi\equiv\ln\chi$).
    \item \textbf{Matter Coupling}: Standard Model fields $\psi_m$ couple to the metric via the standard Lagrangian $\mathcal{L}_m$.
\end{enumerate}

\paragraph{Compilation-cost identification.}
Under the cost--density identification of Proposition~\ref{prop:cost_density_powerlaw},
\begin{equation}
    \varrho(x)=\varrho_0\left(\frac{\kappa(x)}{\kappa_0}\right)^p,
\end{equation}
we have
\begin{equation}
    \chi=\sqrt{\varrho}=\sqrt{\varrho_0}\left(\frac{\kappa}{\kappa_0}\right)^{p/2},
\end{equation}
so the Fisher-amplitude kinetic term becomes a routing-gradient term proportional to $\varrho\,(\nabla\ln(\kappa/\kappa_0))^2$.

\subsection{The Omega Action}
The total action is the sum of the Einstein--Hilbert term, the Fisher-amplitude term (proportional to Fisher information of $\varrho$), and the Matter term:
\begin{equation}
    S = \int d^4x \sqrt{-g} \left[ \frac{R}{16\pi G} - \lambda_F\,\chi^2\, g^{\mu\nu} \nabla_\mu \varphi \nabla_\nu \varphi - V(\varrho) + \mathcal{L}_m(g_{\mu\nu}, \psi_m) \right].
\end{equation}
\noindent
In the minimal sector used in this manuscript we equivalently write the Fisher term in terms of $\chi$:
\begin{equation}
    S = \int d^4x \sqrt{-g} \left[ \frac{R}{16\pi G} - \lambda_F g^{\mu\nu} \nabla_\mu \chi \nabla_\nu \chi - V(\chi^2) + \mathcal{L}_m(g_{\mu\nu}, \psi_m) \right],
\label{eq:omega_action_appendix_chi}
\end{equation}
which is equivalent to the $\varphi$-form by the identity $\nabla_\mu\chi=\chi\,\nabla_\mu\varphi$ (and is consistent with Eq.~(\ref{eq:omega_action_chi}) in Part~V).

\subsection{Variation with respect to the Metric}
We vary $S$ with respect to $g^{\mu\nu}$.
\begin{enumerate}
    \item \textbf{Gravity}: $\delta S_{EH} = \int \sqrt{-g} \frac{1}{16\pi G} (G_{\mu\nu}) \delta g^{\mu\nu}$.
    \item \textbf{Information}: Using $\delta \sqrt{-g} = -\frac{1}{2}\sqrt{-g} g_{\mu\nu} \delta g^{\mu\nu}$, the variation of the Fisher-amplitude term is:
    \begin{equation}
        \delta S_{info} = -\lambda_F \int \sqrt{-g} \left[ \nabla_\mu \chi \nabla_\nu \chi \,\delta g^{\mu\nu} - \frac{1}{2} g_{\rho\sigma} (\nabla \chi)^2 \delta g^{\rho\sigma} \right].
    \end{equation}
    This yields the \textbf{Information Stress-Energy Tensor}:
    \begin{equation}
        T_{\mu\nu}^{info} \equiv -\frac{2}{\sqrt{-g}} \frac{\delta (S_{info}+S_V)}{\delta g^{\mu\nu}}
        = 2\lambda_F \left( \nabla_\mu \chi \nabla_\nu \chi - \frac{1}{2} g_{\mu\nu} (\nabla \chi)^2 \right) - g_{\mu\nu} V(\chi^2),
    \end{equation}
    where $S_V = -\int d^4x \sqrt{-g}\, V(\varrho)$.
    \item \textbf{Matter}: By definition, $T_{\mu\nu}^{(m)} \equiv -\frac{2}{\sqrt{-g}} \frac{\delta S_m}{\delta g^{\mu\nu}}$.
\end{enumerate}
Collecting terms, we obtain the field equations:
\begin{equation}
    G_{\mu\nu} = 8\pi G \left( T_{\mu\nu}^{(m)} + T_{\mu\nu}^{info} \right).
\end{equation}

\subsection{Variation with respect to the Scalar Field}
Varying with respect to $\chi$ yields the equation of motion for the information field.
\begin{equation}
    \delta S_{info} = -2\lambda_F \int d^4x \sqrt{-g} \nabla^\mu \chi \nabla_\mu (\delta \chi)
    = 2\lambda_F \int d^4x \sqrt{-g} (\Box \chi)\, \delta \chi.
\end{equation}
Including the potential term, $\delta S_V = -\int d^4x \sqrt{-g}\, \frac{dV}{d\chi}\, \delta\chi$, the Euler--Lagrange equation is:
\begin{equation}
    2\lambda_F \Box \chi - \frac{dV}{d\chi} = 0,
\end{equation}
with $\frac{dV}{d\chi} = 2\chi\,\frac{dV}{d\varrho}$ when $V=V(\varrho)$ and $\varrho=\chi^2$.

\subsection{Conservation Laws}
Diffeomorphism invariance ensures $\nabla^\mu G_{\mu\nu} = 0$. Thus, the total stress tensor must be conserved:
\begin{equation}
    \nabla^\mu (T_{\mu\nu}^{(m)} + T_{\mu\nu}^{info}) = 0.
\end{equation}
Checking the divergence of the information tensor explicitly:
\begin{equation}
    \nabla^\mu T_{\mu\nu}^{info} = 2\lambda_F [ (\Box \chi) \nabla_\nu \chi + \nabla^\mu \chi \nabla_\mu \nabla_\nu \chi - \frac{1}{2} \nabla_\nu (\nabla_\alpha \chi \nabla^\alpha \chi) ] - \nabla_\nu V(\chi^2).
\end{equation}
Using $\nabla_\nu (\nabla \chi)^2 = 2 \nabla^\mu \chi \nabla_\nu \nabla_\mu \chi$, the last two terms cancel. We are left with:
\begin{equation}
    \nabla^\mu T_{\mu\nu}^{info} = \left( 2\lambda_F \Box \chi - \frac{dV}{d\chi} \right) \nabla_\nu \chi.
\end{equation}
Thus, provided the scalar field satisfies its equation of motion, the information stress tensor is separately conserved.

\subsection{The Information Current}
We define a natural covariant ``information current'' by
\begin{equation}
    J^\mu \equiv \chi\, \nabla^\mu \chi = \frac{1}{2}\nabla^\mu \varrho.
\end{equation}
In the minimal case $V\equiv 0$, the scalar equation reduces to $\Box \chi = 0$.

\subsection{FLRW reduction with lapse (derivation of the background equations)}
\label{app:frw_lapse_derivation}
In Part~V we use an FLRW metric with homogeneous lapse $N(t)$,
\begin{equation}
    ds^2=-N^2(t)\,dt^2+a^2(t)\left[\frac{dr^2}{1-kr^2}+r^2 d\Omega^2\right],
\end{equation}
and define proper time by $d\tau=N(t)\,dt$. The proper-time Hubble parameter is
\begin{equation}
    H_\tau\equiv \frac{1}{a}\frac{da}{d\tau}=\frac{\dot a}{a\,N}.
\end{equation}
For a perfect-fluid stress tensor $T^{\mu}{}_{\nu}=\mathrm{diag}(-\rho,p,p,p)$ and cosmological term $\Lambda(t)$, the Einstein equations yield the standard Friedmann system in $\tau$:
\begin{align}
    H_\tau^2 &= \frac{8\pi G}{3}\rho + \frac{\Lambda(t)}{3} - \frac{k}{a^2},\\
    \frac{1}{a}\frac{d^2a}{d\tau^2} &= -\frac{4\pi G}{3}\left(\rho+3p\right) + \frac{\Lambda(t)}{3}.
\end{align}
Rewriting the second equation in the $t$-coordinate uses
\begin{equation}
    \frac{d}{d\tau}=\frac{1}{N}\frac{d}{dt},\qquad
    \frac{d^2a}{d\tau^2}=\frac{1}{N^2}\ddot a-\frac{\dot N}{N^3}\dot a,
\end{equation}
so that
\begin{equation}
    \frac{\ddot a}{a}-\frac{\dot N}{N}\frac{\dot a}{a}
    =
    -\frac{4\pi G}{3}N^2(\rho+3p)+\frac{\Lambda(t)}{3}N^2,
\end{equation}
equivalently the form quoted in Part~V.

Finally, covariant conservation $\nabla_\mu T^{\mu\nu}=0$ implies in proper time
\begin{equation}
    \frac{d\rho}{d\tau}+3H_\tau(\rho+p)=0,
\end{equation}
which is equivalent to $\dot\rho+3(\dot a/a)(\rho+p)=0$ in the $t$-coordinate because both terms acquire the same factor $1/N$.

\section{Reciprocal-lattice templates and \texorpdfstring{low-$\ell$}{low-l} alignment}
\label{app:moire_cmb}
Part~VI uses a reciprocal-lattice (``moir\'e'') template to motivate preferred-axis signatures in low-$\ell$ CMB multipoles. Here we record the elementary spherical-harmonic relation for sums of plane waves on the sky.

\begin{proposition}[Plane-wave template and spherical-harmonic coefficients]
\label{prop:plane_wave_harmonics}
Let $\hat{n}\in S^2$ denote a sky direction and consider a (real) scalar template of the form
\begin{equation}
    V(\hat n)=\sum_{k\in\mathcal{K}} V_k\,e^{i\,\vec{k}\cdot \hat n},\qquad V_{-k}=\overline{V_k},
\end{equation}
where $\mathcal{K}$ is a finite (or sufficiently summable) set of wavevectors. Expand $V$ in spherical harmonics, $V(\hat n)=\sum_{\ell,m} a_{\ell m}Y_{\ell m}(\hat n)$. Then
\begin{equation}
    a_{\ell m} = 4\pi\, i^{\ell}\sum_{k\in\mathcal{K}} V_k\, j_{\ell}(|k|)\,\overline{Y_{\ell m}(\hat{k})},
\end{equation}
where $j_{\ell}$ is the spherical Bessel function and $\hat{k}\equiv \vec{k}/|\vec{k}|$.
\end{proposition}
\begin{proof}
Use the standard plane-wave expansion on the unit sphere,
\begin{equation}
    e^{i\,\vec{k}\cdot \hat n}=4\pi\sum_{\ell,m} i^{\ell}\,j_{\ell}(|k|)\,Y_{\ell m}(\hat n)\,\overline{Y_{\ell m}(\hat{k})},
\end{equation}
and collect coefficients of $Y_{\ell m}(\hat n)$. The reality condition $V_{-k}=\overline{V_k}$ ensures $V(\hat n)$ is real.
\end{proof}

\begin{corollary}[Dominant axis implies aligned low-$\ell$ multipoles]
\label{cor:axis_alignment}
If the set of dominant wavevectors $\mathcal{K}$ is concentrated near a single axis $\hat{k}_\ast$ (for example, $\mathcal{K}$ is dominated by $\pm \vec{k}_\ast$), then the low-$\ell$ coefficients $a_{\ell m}$ are dominated by the corresponding spherical harmonics evaluated at $\hat{k}_\ast$. In particular, in the exactly axisymmetric case $V(\hat n)=f(\hat{k}_\ast\cdot \hat n)$, one can choose coordinates with $z$-axis along $\hat{k}_\ast$ so that $a_{\ell m}=0$ for all $m\neq 0$, implying that the quadrupole and octopole share the same preferred axis.
\end{corollary}
\begin{proof}
In the axisymmetric case, $V$ depends only on the polar angle $\theta$ relative to $\hat{k}_\ast$, so it is invariant under rotations about $\hat{k}_\ast$. Spherical harmonics with $m\neq 0$ transform with nontrivial phase under such rotations, hence cannot appear. Therefore only $m=0$ modes remain in that coordinate system, and all low-$\ell$ multipoles are aligned with the same axis.
\end{proof}

\section{Symmetry constraints on modified dispersion}
\label{app:mdr_symmetry}
The modified-dispersion discussion in Parts~VI and VIII uses only basic symmetry constraints. We record the minimal expansion statement used to justify the absence of linear ($n=1$) time-of-flight terms in parity-symmetric lattice/QCA settings.

\begin{proposition}[Parity-symmetric analytic dispersion has no $n=1$ MDR term]
\label{prop:no_linear_mdr}
Let $E(p)$ be a (positive) dispersion relation for a massless mode in one dimension, with $p$ the physical momentum, and assume that the squared dispersion is analytic and parity symmetric near $p=0$:
\begin{equation}
    E^2(p)=E^2(-p),\qquad
    E^2(p)=c^2 p^2 + \beta_2 p^4 + \beta_3 p^6 + \cdots \quad (|p|\ll p_{\mathrm{UV}}).
\end{equation}
Then the group velocity satisfies
\begin{equation}
    v(p)\equiv \frac{dE}{dp}=c\left[1+\mathcal{O}(p^2)\right],
\end{equation}
so there is no linear-in-energy ($n=1$) time-of-flight correction in an MDR parameterization; the leading correction is quadratic in energy (equivalently quartic in momentum).
\end{proposition}
\begin{proof}
From the assumed expansion,
\begin{equation}
    E(p)=c|p|\left[1+\frac{\beta_2}{2c^2}p^2+\mathcal{O}(p^4)\right]
\end{equation}
for $p$ in a one-sided neighborhood (e.g.\ $p>0$). Differentiating gives $v(p)=c+\mathcal{O}(p^2)$, so the first correction is quadratic in $p$ (hence quadratic in $E$ at leading order).
\end{proof}

\begin{proposition}[Spherical designs suppress low-$\ell$ anisotropies]
\label{prop:design_suppresses_lowell}
Let $\{\hat u_i\}_{i=1}^M\subset S^2$ be a spherical $t$-design and let $Y_{\ell m}$ denote the standard spherical harmonics. Then for every $1\le \ell\le t$ and every $m\in\{-\ell,\dots,\ell\}$ one has
\begin{equation}
    \frac{1}{M}\sum_{i=1}^M Y_{\ell m}(\hat u_i)=0.
\end{equation}
Equivalently, the discrete empirical measure $\frac{1}{M}\sum_i \delta_{\hat u_i}$ has vanishing multipole moments up to degree $t$.
\end{proposition}
\begin{proof}
For fixed $(\ell,m)$, $Y_{\ell m}$ is the restriction to $S^2$ of a homogeneous harmonic polynomial of total degree $\ell$ on $\mathbb{R}^3$, hence is (the restriction of) a polynomial of degree $\ell$. Since $\{\hat u_i\}$ is a $t$-design and $\ell\le t$, its discrete average equals the normalized spherical average:
\begin{equation}
    \frac{1}{M}\sum_{i=1}^M Y_{\ell m}(\hat u_i)=\frac{1}{4\pi}\int_{S^2} Y_{\ell m}(\hat u)\,d\Omega(\hat u).
\end{equation}
The integral vanishes for all $\ell\ge 1$ by orthogonality of spherical harmonics (equivalently, $Y_{\ell m}$ has zero mean on the sphere), proving the claim.
\end{proof}

\begin{proposition}[Design-averaged plane waves are isotropic to order $t$]
\label{prop:design_plane_wave_isotropy}
Let $\{\hat u_i\}_{i=1}^M\subset S^2$ be a spherical $t$-design and define, for $q\in\mathbb{R}^3$,
\begin{equation}
    F(q)\ \equiv\ \frac{1}{M}\sum_{i=1}^M e^{i\,q\cdot \hat u_i}.
\end{equation}
Then the Taylor polynomial of $F$ at $q=0$ of total degree $t$ coincides with that of the isotropic spherical average
\begin{equation}
    \overline{F}(q)\ \equiv\ \frac{1}{4\pi}\int_{S^2} e^{i\,q\cdot \hat u}\,d\Omega(\hat u)
    \;=\; j_0(|q|)\;=\;\frac{\sin |q|}{|q|},
\end{equation}
so that, as $|q|\to 0$,
\begin{equation}
    F(q)=\overline{F}(q)+\mathcal{O}(|q|^{t+1}).
\end{equation}
In particular, all anisotropic contributions in the small-$|q|$ expansion are pushed to order at least $t{+}1$.
\end{proposition}
\begin{proof}
Expand the exponential as a Taylor series,
\begin{equation}
    e^{i\,q\cdot \hat u}=\sum_{n=0}^{t}\frac{i^n}{n!}(q\cdot \hat u)^n\;+\;R_{t+1}(q,\hat u),
\end{equation}
where the remainder satisfies $|R_{t+1}(q,\hat u)|\le |q|^{t+1}/(t{+}1)!$ for all unit $\hat u$ (by the standard Taylor remainder bound).
For each $n\le t$, $(q\cdot \hat u)^n$ is a polynomial in the components of $\hat u$ of total degree $n$, hence its discrete average over a $t$-design equals its spherical average:
\begin{equation}
    \frac{1}{M}\sum_{i=1}^M (q\cdot \hat u_i)^n
    =\frac{1}{4\pi}\int_{S^2} (q\cdot \hat u)^n\,d\Omega(\hat u).
\end{equation}
Therefore the truncated Taylor sums of $F$ and $\overline{F}$ match term-by-term up to degree $t$, and the difference is controlled by the averaged remainder, giving $F(q)-\overline{F}(q)=\mathcal{O}(|q|^{t+1})$.
Finally, the identity $\overline{F}(q)=\sin|q|/|q|$ follows by rotational invariance: align $q$ with the $z$-axis and integrate $e^{i|q|\cos\theta}$ over the sphere.
\end{proof}

\section{Discrete scale invariance and \texorpdfstring{$1/f$}{1/f} spectra}
\label{app:one_over_f}
The ``pink'' ($1/f$) noise ansatz used in Part~VI can be derived from a minimal scale-invariance mechanism: a log-uniform distribution of characteristic times (continuous scale invariance) or a geometric ladder of times (discrete scale invariance).

\begin{proposition}[$1/f$ spectrum from a log-uniform mixture of Lorentzians]
\label{prop:one_over_f}
Let $x(t)$ be a stationary process obtained as a superposition of independent Ornstein--Uhlenbeck components with relaxation times $\tau\in[\tau_{\min},\tau_{\max}]$ whose density is log-uniform,
\begin{equation}
    \rho(\tau)=\frac{C}{\tau}\,\mathbf{1}_{[\tau_{\min},\tau_{\max}]}(\tau),
\end{equation}
and whose single-component power spectral density is Lorentzian, $S_\tau(f)=\frac{A\,\tau}{1+(2\pi f\tau)^2}$. Then the total PSD satisfies
\begin{equation}
    S(f)=\int_{\tau_{\min}}^{\tau_{\max}} \rho(\tau)\,S_\tau(f)\,d\tau \;=\; \frac{A C}{2\pi}\,\frac{1}{f}\int_{2\pi f\tau_{\min}}^{2\pi f\tau_{\max}} \frac{du}{1+u^2}.
\end{equation}
In the intermediate band $1/\tau_{\max}\ll f\ll 1/\tau_{\min}$, the integral is approximately constant, hence $S(f)\propto 1/f$.
\end{proposition}
\begin{proof}
Substitute $u\equiv 2\pi f\tau$ to obtain the stated formula. In the regime $2\pi f\tau_{\min}\ll 1\ll 2\pi f\tau_{\max}$ one has $\int_{u_{\min}}^{u_{\max}}\frac{du}{1+u^2}\approx \frac{\pi}{2}$, giving $S(f)\approx \frac{AC}{4}\,\frac{1}{f}$.
\end{proof}

\paragraph{Discrete scale invariance.}
If the characteristic times form a geometric ladder $\tau_k=\tau_0\,\phi^k$ (golden scaling) with weights approximating a log-uniform density, the resulting spectrum exhibits the same $1/f$ envelope with log-periodic modulations (a periodic function of $\log_\phi f$), consistent with the ``log-periodic peaks'' template used in Part~VI.

\begin{proposition}[Common-mode field implies positive cross-correlation]
\label{prop:common_mode_crosscorr}
Let two colocated detectors have outputs
\begin{equation}
    y_1(t)=x(t)+n_1(t),\qquad y_2(t)=x(t)+n_2(t),
\end{equation}
where $x(t)$ is a stationary ``common'' signal and $n_1,n_2$ are stationary detector noises that are uncorrelated with $x$ and with each other. Then the cross power spectral density satisfies
\begin{equation}
    S_{12}(f)\equiv \langle \widetilde{y}_1(f)\,\overline{\widetilde{y}_2(f)}\rangle = S_x(f),
\end{equation}
so the cross-spectrum inherits the sign/shape of the common field and is strictly positive for a positive PSD $S_x(f)$.
\end{proposition}
\begin{proof}
Expand $\widetilde{y}_1\overline{\widetilde{y}_2}=(\widetilde{x}+\widetilde{n}_1)(\overline{\widetilde{x}}+\overline{\widetilde{n}_2})$ and take expectation. The mixed terms vanish by the stated uncorrelatedness assumptions, leaving $S_{12}(f)=\langle \widetilde{x}(f)\overline{\widetilde{x}(f)}\rangle=S_x(f)$.
\end{proof}

\section{Echo time delay from a near-horizon reflector}
\label{app:echo_delay}
The echo-delay estimate in Part~VI follows from the standard tortoise-coordinate logarithm near a horizon. We record the elementary derivation in the Schwarzschild case.

\begin{proposition}[Logarithmic echo delay]
\label{prop:echo_delay}
Consider a Schwarzschild black hole of mass $M$ (in units $G=c=1$) with tortoise coordinate
\begin{equation}
    r_\ast(r)=r+2M\ln\!\left(\frac{r}{2M}-1\right).
\end{equation}
Suppose a partially reflecting structure is located at $r_0=2M+\varepsilon$ with $0<\varepsilon\ll 2M$, and consider wavepackets that reflect between $r_0$ and an exterior potential barrier at $r\simeq r_b$ (e.g.\ near the photon sphere). Then the echo time delay between successive reflections is
\begin{equation}
    \Delta t_{\mathrm{echo}}
    \simeq 2\bigl[r_\ast(r_b)-r_\ast(r_0)\bigr]
    = 4M\ln\!\left(\frac{2M}{\varepsilon}\right) + \mathcal{O}(M),
\end{equation}
where the $\mathcal{O}(M)$ term depends on the choice of $r_b$ but is subdominant for $\varepsilon\ll 2M$.
\end{proposition}
\begin{proof}
In Schwarzschild coordinates, radial null propagation satisfies $t\simeq \pm r_\ast+\mathrm{const}$. A round trip from $r_0$ to $r_b$ and back therefore takes $2[r_\ast(r_b)-r_\ast(r_0)]$. For $\varepsilon\ll 2M$, one has
\begin{equation}
    r_\ast(2M+\varepsilon)=2M+\varepsilon+2M\ln\!\left(\frac{\varepsilon}{2M}\right),
\end{equation}
so $-r_\ast(r_0)$ contains the dominant logarithm $2M\ln(2M/\varepsilon)$, giving the stated leading term.
\end{proof}
\noindent
\textit{Conversion to SI.} Restoring $G$ and $c$ gives $\Delta t_{\mathrm{echo}}\simeq \frac{4GM}{c^3}\ln\!\left(\frac{2GM}{c^2\varepsilon}\right)$.

\subsection{Min-cut entropy bound}
For tensor-network states/codes, the elementary and fully rigorous statement is an \emph{upper bound} on boundary entanglement entropy in terms of bond cuts.

\begin{theorem}[Min-cut entropy upper bound]
\label{thm:mincut_entropy}
Let $|\Psi\rangle$ be a (pure) boundary state obtained from contracting a tensor network with uniform bond dimension $\chi$ on internal edges. For any boundary region $A$, let $\gamma_A$ be a cut that separates $A$ from its complement $A^c$ and crosses $|\gamma_A|$ internal edges. Then the von Neumann entropy satisfies
\begin{equation}
    S(\rho_A)\ \le\ |\gamma_A|\,\ln \chi,
\end{equation}
and in particular $S(\rho_A)\le |\gamma_{\min}(A)|\ln\chi$ for a minimal cut $\gamma_{\min}(A)$.
\end{theorem}
\begin{proof}
Cutting $|\gamma_A|$ bonds of dimension $\chi$ yields a bipartition in which the Schmidt rank across $(A|A^c)$ is at most $\chi^{|\gamma_A|}$. Therefore $S(\rho_A)\le \ln(\mathrm{rank}\,\rho_A)\le |\gamma_A|\ln\chi$.
\end{proof}

\begin{proposition}[Page-curve mechanism from competing cuts (tensor-network island picture)]
\label{prop:page_curve_mincut}
Consider a tensor-network boundary state $|\Psi\rangle$ with bond dimension $\chi$ and a boundary partition into $R$ (radiation) and $B$ (the remaining boundary). Suppose the network contains a distinguished set of ``horizon'' edges of cardinality $|\partial I|$ such that there exist two natural classes of cuts homologous to $R$:
\begin{itemize}
    \item \textbf{No-island cut}: cuts that separate $R$ from $B$ without crossing the horizon bottleneck;
    \item \textbf{Island cut}: cuts that additionally enclose a bulk region $I$ and hence cross the horizon bottleneck, incurring an additive cost $|\partial I|$.
\end{itemize}
Then the min-cut entropy bound implies
\begin{equation}
    S(\rho_R)\ \le\ \min\!\Big\{ |\gamma_{\mathrm{no}}(R)|,\ |\gamma_{\mathrm{is}}(R\cup I)|+|\partial I| \Big\}\,\ln\chi,
\end{equation}
where $\gamma_{\mathrm{no}}(R)$ and $\gamma_{\mathrm{is}}(R\cup I)$ denote the minimal cuts within the respective classes. If, along an evaporation/transfer process, the minimizing class changes from ``no-island'' at early times to ``island'' at late times, this reproduces a Page-curve-shaped entropy profile by a discrete analogue of a quantum extremal surface transition.
\end{proposition}
\begin{proof}
The min-cut bound (Theorem~\ref{thm:mincut_entropy}) holds for any cut, hence for the best cut in each class. Taking the minimum over the two admissible classes yields the stated inequality. The ``Page curve'' statement follows from the observation that a change in the minimizing cut class causes $S(\rho_R)$ to transition from being controlled by the no-island cut size to being controlled by the island cut size plus the horizon cost.
\end{proof}

\subsection{Entanglement wedge reconstruction (perfect/approximate-perfect networks)}
To connect the holographic map $\Phi$ to reconstruction, we use the standard tensor-network QECC results (HaPPY / OAQEC); see, e.g., \cite{Pastawski2015,Harlow2017}.

\begin{theorem}[Entanglement wedge reconstruction for a perfect tensor network]
\label{thm:ewr_perfect}
Assume the encoding $\Phi$ is implemented by a tensor network built from perfect tensors and defines an isometry from a bulk code space into the boundary Hilbert space. Let $A$ be a boundary region and let $\mathcal{W}_E(A)$ be the bulk region enclosed by $A$ and a minimal cut $\gamma_{\min}(A)$. Then for any bulk operator $\phi_x$ supported on $x\in\mathcal{W}_E(A)$ there exists a boundary operator $\mathcal{O}_A$ supported only on $A$ such that, on the code subspace,
\begin{equation}
    \Phi\,\phi_x\,\Phi^\dagger\ =\ \mathcal{O}_A\otimes I_{A^c}.
\end{equation}
\end{theorem}
\begin{proof}
This is the standard ``operator pushing''/``tensor pushing'' argument of the HaPPY code. Because a perfect tensor is an isometry from any subset of at most half its legs to the complementary legs, one can iteratively move (conjugate) the action of $\phi_x$ through the network along a directed flow that ends on the boundary region $A$, provided $x$ lies in the wedge defined by the minimal cut homologous to $A$. The resulting boundary operator is supported on $A$ and reproduces the bulk action on the code subspace.
\end{proof}

\begin{corollary}[Approximate reconstruction for $\epsilon$-perfect tensors]
\label{cor:ewr_eps}
If the network tensors are $\epsilon$-perfect in operator norm (each intended isometry condition holds up to error $\epsilon$) and the pushing path from $x$ to $A$ crosses at most $d$ tensors, then there exists a boundary operator $\mathcal{O}_A$ supported on $A$ such that
\begin{equation}
    \left\|\left(\Phi\,\phi_x\,\Phi^\dagger\right)-\left(\mathcal{O}_A\otimes I_{A^c}\right)\right\|\ \le\ C\,d\,\epsilon,
\end{equation}
for a constant $C$ that depends only on the local arity (degree) of the tensors.
\end{corollary}
\begin{proof}
Each approximate-isometry step in the pushing construction incurs an operator-norm error bounded by $\mathcal{O}(\epsilon)$. Summing the errors over at most $d$ steps yields the stated bound.
\end{proof}

\paragraph{Relation to Axiom~O4.}
Theorem~\ref{thm:ewr_perfect} provides an explicit sufficient condition under which an encoding map $\Phi$ realizes exact entanglement-wedge reconstruction. Corollary~\ref{cor:ewr_eps} gives a minimal quantitative error bound in an $\epsilon$-perfect setting. These results are one concrete route to implementing the reconstruction requirement in Axiom~O4.

\subsection{Rigidity bound for local temporal drift of coarse-grained couplings}
\label{app:rigidity_drift}
The drift estimates in Part~VI use only a minimal ``averaging'' mechanism: if an effective coupling is a coarse-grained average over many microscopic contributions, then any bounded-rate microscopic update produces a parametrically suppressed macroscopic drift.

\begin{proposition}[Averaging bound]
\label{prop:averaging_bound}
Let $g(t)$ be an effective (dimensionless) coupling defined as an average over $N_{\mathrm{dof}}$ bounded microscopic contributions (not to be confused with the lapse $N(t)$ used elsewhere in this manuscript),
\begin{equation}
    g(t)=\frac{1}{N_{\mathrm{dof}}}\sum_{i=1}^{N_{\mathrm{dof}}} g_i(t),
\end{equation}
and assume that during a time interval of duration $\Delta t$ at most $M$ of the $g_i$ change, each by at most $|\Delta g_i|\le \delta$. Then
\begin{equation}
    |\Delta g|\ \le\ \frac{M}{N_{\mathrm{dof}}}\,\delta.
\end{equation}
If $g(t)$ is strictly positive and bounded below by $g_{\min}>0$, then the fractional drift obeys
\begin{equation}
    \left|\frac{\Delta g}{g}\right|\ \le\ \frac{M}{N_{\mathrm{dof}}}\,\frac{\delta}{g_{\min}}.
\end{equation}
\end{proposition}
\begin{proof}
By the triangle inequality,
\begin{equation}
    |\Delta g|
    =\left|\frac{1}{N_{\mathrm{dof}}}\sum_{i=1}^{N_{\mathrm{dof}}}\Delta g_i\right|
    \le \frac{1}{N_{\mathrm{dof}}}\sum_{i=1}^{N_{\mathrm{dof}}}|\Delta g_i|
    \le \frac{M}{N_{\mathrm{dof}}}\,\delta.
\end{equation}
Dividing by $g\ge g_{\min}$ gives the fractional bound.
\end{proof}

\paragraph{Holographic scaling of $N_{\mathrm{dof}}$.}
In holographic near-saturation regimes, one may identify $N_{\mathrm{dof}}$ with the number of independent boundary degrees of freedom in a causal patch, $N_{\mathrm{dof}}\sim A/(4l_P^2\ln 2)$, so that any bounded-rate microscopic update yields a very small macroscopic drift. The estimate is purely kinematic; the remaining dynamical question is to determine the effective update rate $M$ and the microscopic step size $\delta$ for a concrete Omega network.

\section{Formal Existence Proofs}

\subsection{Finite-information localization and approximate factorization}
We formalize the finite-information postulate in a way compatible with operator-algebraic locality. In continuum AQFT, local von Neumann algebras are typically type~III and do not admit an exact tensor factorization $\mathcal{H}\simeq\mathcal{H}_R\otimes\mathcal{H}_{R^c}$; this is tied to ultraviolet entanglement across sharp boundaries (see, e.g., \cite{Haag1996}). The finite-information axiom is therefore interpreted as a statement about a \emph{regulated/operational} description: at finite resolution (or on a finite-dimensional code subspace), the relevant effective algebra is type~I and admits an approximate subsystem structure.\par

\begin{theorem}[Finite-information effective dimension]
\label{thm:finite_information_dimension}
Let $V$ be a finite causal region with boundary area $A(\partial V)$.
Assume that the set of physically distinguishable states accessible to observers localized in $V$ obeys a Bekenstein--Hawking entropy bound
\begin{equation}
    S_{\max}(V)\ \le\ \frac{A(\partial V)}{4 l_P^2}.
\end{equation}
Then any family of pairwise orthogonal operational states $\{|\psi_i\rangle\}_{i=1}^N$ localized in $V$ has cardinality bounded by
\begin{equation}
    N\ \le\ \exp\!\left(\frac{A(\partial V)}{4 l_P^2}\right).
\end{equation}
Equivalently, the operational state space in $V$ can be supported on an effective finite-dimensional Hilbert space $\mathcal{H}^{\mathrm{eff}}_V$ of dimension at most $\exp(A(\partial V)/4l_P^2)$.
\end{theorem}
\begin{proof}
Let $\{|\psi_i\rangle\}_{i=1}^N$ be orthonormal. Consider the maximally mixed state on their span,
\begin{equation}
    \rho \equiv \frac{1}{N}\sum_{i=1}^{N}|\psi_i\rangle\langle\psi_i|.
\end{equation}
Its von Neumann entropy is $S(\rho)=\ln N$. By definition of $S_{\max}(V)$, one has $\ln N\le S_{\max}(V)$, and the stated bound follows.
\end{proof}

\paragraph{Remark (split property and the continuum limit).}
In theories satisfying the split property, for nested regions $V\subset V_{\epsilon}$ there exists an intermediate type~I factor $\mathfrak{N}$ with $\mathfrak{A}(V)\subset \mathfrak{N}\subset \mathfrak{A}(V_{\epsilon})$ \cite{Buchholz1987}. This provides a controlled notion of localization at finite resolution and an approximate tensor-product structure on the associated state spaces, while the sharp-boundary limit recovers type~III behavior.
In Omega Theory, statements such as $\mathcal{H}\cong\bigotimes_x\mathcal{H}_x$ are to be read in this regulated/code-subspace sense (cf.\ Part~II), not as a claim about exact factorization in continuum QFT.\par

\paragraph{Corollary (area-limited independent degrees of freedom).}
If a microscopic model admits a local tensor-factor description on a graph, but the physically realized states in a region $V$ are restricted to an operational/code subspace obeying the bound above, then the number of independent degrees of freedom in $V$ is limited by area rather than volume. This is the sense in which the holographic constraint enforces redundancy of bulk degrees of freedom in a regulated discrete description.

\subsection{Modular flow as a canonical intrinsic clock}
\label{app:modular_time}
Part~II introduces an intrinsic parameter $\tau$ by writing a one-parameter family of phase-twisted representatives of the Omega ray. In an operator-algebraic setting, a canonical one-parameter flow is provided by modular theory. Since the regulated/QCA description used throughout this manuscript is effectively finite-dimensional on relevant code subspaces, we record the finite-dimensional modular statement explicitly.

\begin{theorem}[Modular automorphism group (finite-dimensional case)]
\label{thm:modular_finite}
Let $\mathcal{H}$ be finite-dimensional and let $\rho$ be a full-rank density matrix on $\mathcal{H}$. Define the faithful state $\omega(A)\equiv \mathrm{Tr}(\rho\,A)$ on $\mathcal{B}(\mathcal{H})$. Then the maps
\begin{equation}
    \sigma_t^\omega(A)\ \equiv\ \rho^{it}\,A\,\rho^{-it}\qquad (t\in\mathbb{R})
\end{equation}
form a one-parameter group of $*$-automorphisms of $\mathcal{B}(\mathcal{H})$ that leaves $\omega$ invariant: $\omega(\sigma_t^\omega(A))=\omega(A)$ for all $t$.
Equivalently, writing the modular Hamiltonian as $K\equiv -\ln\rho$, one has $\sigma_t^\omega(A)=e^{-itK}A\,e^{itK}$.
\end{theorem}
\begin{proof}
Since $\rho$ is positive definite, $\rho^{it}$ is unitary for all $t$, hence $\sigma_t^\omega$ is an inner $*$-automorphism:
\[
    \sigma_t^\omega(AB)=\rho^{it}AB\rho^{-it}=(\rho^{it}A\rho^{-it})(\rho^{it}B\rho^{-it})=\sigma_t^\omega(A)\sigma_t^\omega(B),
    \qquad
    \sigma_t^\omega(A^\dagger)=\sigma_t^\omega(A)^\dagger.
\]
The group property $\sigma_{t+s}^\omega=\sigma_t^\omega\circ\sigma_s^\omega$ follows from $\rho^{i(t+s)}=\rho^{it}\rho^{is}$. Finally, invariance holds because $\rho^{it}$ commutes with $\rho$ and the trace is cyclic:
\[
    \omega(\sigma_t^\omega(A))=\mathrm{Tr}(\rho\,\rho^{it}A\rho^{-it})
    =\mathrm{Tr}(\rho^{it}\rho\rho^{-it}A)=\mathrm{Tr}(\rho A)=\omega(A).
\]
Writing $\rho^{it}=e^{it\ln\rho}=e^{-itK}$ gives the modular-Hamiltonian form.
\end{proof}

\paragraph{Remark (general von Neumann algebras).}
For a general von Neumann algebra $\mathfrak{M}$ and a faithful normal state $\omega$, the Tomita--Takesaki theorem provides a canonical modular automorphism group $\{\sigma_t^\omega\}_{t\in\mathbb{R}}$ characterized (in the GNS representation) by conjugation with the modular operator. This is the natural operator-algebraic sense in which a state determines an intrinsic flow; see \cite{Takesaki2002,ConnesRovelli1994}.

\subsection{Existence of the Golden MERA Isometry}
We record a purely kinematic existence statement for the finite-depth Golden MERA map used throughout this manuscript: under the local tensor assumptions of Assumption~\ref{assump:M3}, the truncated encoding map $\Phi_L$ is an isometry. This matches the isometry component of Axiom~O4 at finite depth; scale-invariant limits and error-correction properties require additional assumptions (Appendix~\ref{app:isometry}).

\begin{theorem}[Golden MERA encoding isometry (finite depth)]
\label{thm:golden_isometry}
Fix a finite-depth truncation $L$ of the Golden MERA network and assume that each disentangler tensor $D^{(s)}$ is unitary and each coarse-graining tensor $W^{(s)}$ is an isometry (so $W^{(s)\dagger}W^{(s)}=I$ on its input legs). Let $\Phi_L:\mathcal{H}_{bulk,L}\to\mathcal{H}_{boundary}$ denote the resulting layer composition. Then $\Phi_L$ is an isometry:
\begin{equation}
    \Phi_L^\dagger \Phi_L = I_{\mathcal{H}_{bulk,L}}.
\end{equation}
\end{theorem}
\begin{proof}
Each layer of the MERA consists of a tensor product of local unitaries (disentanglers) and local isometries (coarse-grainers). A tensor product of unitaries is unitary, and a tensor product of isometries is an isometry. Therefore each layer map is an isometry, and the composition $\Phi_L$ of finitely many layers is again an isometry, giving $\Phi_L^\dagger \Phi_L=I$.
\end{proof}
\paragraph{Remark (error correction vs isometry).}
The isometry statement above is purely kinematic. Approximate OAQEC/entanglement-wedge reconstruction requires additional structure (e.g.\ perfect or $\epsilon$-perfect tensors and a suitable code subspace); see Appendix~\ref{app:isometry} and \cite{Pastawski2015,Harlow2017}.

\subsection{Operator-algebraic superselection viewpoint on baby universes}
\label{app:baby_universes}
The factorization $\mathcal{H} = \otimes \mathcal{H}_x$ appears to contradict the existence of ``baby-universe''/$\alpha$-parameter sectors which are often modeled as implying a direct-sum structure $\mathcal{H} = \oplus_\alpha \mathcal{H}_\alpha$ \cite{MarolfMaxfield2020}. In this manuscript we interpret this tension via the \textbf{Omega Superselection Principle}.

\begin{theorem}[Superselection of the $\alpha$-parameters (operator-algebraic form)]
\label{thm:alpha_superselection}
Let $(\pi,\mathcal{H},|\Omega\rangle)$ be a GNS representation of the Omega state $\omega_\Omega$ and let $\mathfrak{A}_{loc}\subset\mathcal{B}(\mathcal{H})$ denote the von Neumann algebra generated by all quasi-local observables. If a bounded operator $\hat{B}\in\mathcal{B}(\mathcal{H})$ commutes with $\mathfrak{A}_{loc}$,
\begin{equation}
    [\hat{B}, \hat{A}] = 0\qquad \forall\,\hat{A}\in\mathfrak{A}_{loc},
\end{equation}
\noindent
then $\hat{B}$ lies in the commutant $\mathfrak{A}_{loc}'$. In particular, if the representation is irreducible (equivalently $\mathfrak{A}_{loc}'=\mathbb{C}\,I$) or more generally factorial, then $\hat{B}$ is a multiple of the identity and its eigenvalue is a superselection parameter fixed by the choice of $\omega_\Omega$.
\end{theorem}
\begin{proof}
The commutation condition places $\hat{B}$ in the commutant $\mathfrak{A}_{loc}'$. If the representation is irreducible, Schur's lemma implies $\mathfrak{A}_{loc}'=\mathbb{C}\,I$, hence $\hat{B}=\lambda I$ for some $\lambda\in\mathbb{C}$. The same conclusion holds for a factorial representation when $\hat{B}$ is central. Therefore any such $\hat{B}$ acts as a fixed superselection parameter on the physical Hilbert space of the GNS sector determined by $\omega_\Omega$.
\end{proof}
\paragraph{Remark (baby universes as central data).}
In gravitational path-integral discussions, ``baby-universe'' operators are often modeled as global/topological degrees of freedom that commute with the quasi-local algebra. Theorem~\ref{thm:alpha_superselection} states the corresponding operator-algebraic fact: if such operators commute with all quasi-local observables in the physical representation, then they reduce to fixed central data (a choice of sector) rather than dynamical variables within that sector.

\subsection{The Observer Operator}
We distinguish a \emph{conditional} selection/update map (nonlinear because of normalization) from its \emph{unconditioned} statistical average (which is linear and CPTP). This matches the non-linear projector form used in Part~VII.\par
Let $\mathcal{H}_{univ} = \mathcal{H}_{sys} \otimes \mathcal{H}_{obs}$ and let $\{\hat{\Pi}_k\}$ be a projective family of pointer/consistency sectors with $\hat{\Pi}_k\hat{\Pi}_\ell=\delta_{k\ell}\hat{\Pi}_k$ and $\sum_k \hat{\Pi}_k = I$.

\begin{definition}[Conditional selection map (nonlinear)]
For a state $\rho$ with $p_k\equiv \mathrm{Tr}(\hat{\Pi}_k\rho)>0$, define the conditional update associated with outcome/sector $k$ by
\begin{equation}
    \hat{\mathcal{O}}_k[\rho] \equiv \frac{\hat{\Pi}_k \rho \hat{\Pi}_k}{\mathrm{Tr}(\hat{\Pi}_k\rho)}.
\end{equation}
Equivalently, on pure states $|\Psi\rangle$ one has $|\Psi\rangle \mapsto \hat{\Pi}_k|\Psi\rangle/\|\hat{\Pi}_k|\Psi\rangle\|$. This is the basic non-linear ``selection'' map.
\end{definition}

\begin{definition}[Unconditioned channel (CPTP)]
The corresponding unconditioned map (outcome ignored/averaged) is the linear CPTP channel
\begin{equation}
    \mathcal{E}_{\mathcal{O}}[\rho] \equiv \sum_k \hat{\Pi}_k \rho \hat{\Pi}_k
    \;=\;\sum_k p_k\,\hat{\mathcal{O}}_k[\rho].
\end{equation}
\end{definition}

\begin{proposition}[Projective measurement as an instrument (L{\"u}ders/Kraus)]
\label{prop:luders_kraus_instrument}
Let $\{\hat{\Pi}_k\}$ be a projective family with $\hat{\Pi}_k\hat{\Pi}_\ell=\delta_{k\ell}\hat{\Pi}_k$ and $\sum_k \hat{\Pi}_k=I$. Then the unnormalized maps
\begin{equation}
    \rho\ \mapsto\ \hat{\Pi}_k\rho\hat{\Pi}_k
\end{equation}
are completely positive and trace-nonincreasing, and their sum $\rho\mapsto \sum_k \hat{\Pi}_k\rho\hat{\Pi}_k$ is completely positive and trace preserving. Conditioning on outcome $k$ yields the normalized update
\begin{equation}
    \rho\ \mapsto\ \frac{\hat{\Pi}_k\rho\hat{\Pi}_k}{\mathrm{Tr}(\hat{\Pi}_k\rho)},
\end{equation}
which is the standard L{\"u}ders state-update rule for projective measurements \cite{Luders1951,Kraus1983,NielsenChuang2000}.
\end{proposition}
\begin{proof}
Each map $\rho\mapsto \hat{\Pi}_k\rho\hat{\Pi}_k$ has the operator-sum (Kraus) form $K_k\rho K_k^\dagger$ with $K_k=\hat{\Pi}_k$, hence is completely positive. Since $0\le \hat{\Pi}_k\le I$, one has $\mathrm{Tr}(\hat{\Pi}_k\rho\hat{\Pi}_k)=\mathrm{Tr}(\hat{\Pi}_k\rho)\le \mathrm{Tr}(\rho)$, so it is trace-nonincreasing. The sum $\sum_k \hat{\Pi}_k\rho\hat{\Pi}_k$ is completely positive as a sum of CP maps and is trace preserving because
\[
    \sum_k K_k^\dagger K_k=\sum_k \hat{\Pi}_k = I.
\]
Finally, the normalized conditioned state is obtained by dividing the unnormalized post-measurement state by its trace, giving the stated L{\"u}ders update.
\end{proof}

\begin{corollary}[No-signalling of local projective instruments (unconditioned)]
\label{cor:nosignalling_luders}
Let $\rho_{AB}$ be a bipartite state and let $\{\Pi_k\}$ be a projective family acting on $A$. Consider the unconditioned local measurement channel on $A$,
\begin{equation}
    \rho_{AB}\ \mapsto\ \rho'_{AB}\equiv \sum_k (\Pi_k\otimes I_B)\,\rho_{AB}\,(\Pi_k\otimes I_B).
\end{equation}
Then the reduced state on $B$ is unchanged:
\begin{equation}
    \rho'_B \equiv \mathrm{Tr}_A(\rho'_{AB})=\mathrm{Tr}_A(\rho_{AB})\equiv \rho_B.
\end{equation}
In particular, treating $\hat{\mathcal{O}}$ as a conditioning map on outcomes does not enable superluminal signalling: signalling would require control of the outcome-conditioned branch, whereas the locally accessible state with outcomes ignored is invariant on $B$.
\end{corollary}
\begin{proof}
Using linearity of the partial trace and $\sum_k \Pi_k = I_A$,
\begin{align}
    \rho'_B
    &=\mathrm{Tr}_A\!\left(\sum_k (\Pi_k\otimes I_B)\,\rho_{AB}\,(\Pi_k\otimes I_B)\right)
      =\sum_k \mathrm{Tr}_A\!\left((\Pi_k\otimes I_B)\,\rho_{AB}\,(\Pi_k\otimes I_B)\right)\\
    &=\sum_k \mathrm{Tr}_A\!\left((\Pi_k\otimes I_B)\,\rho_{AB}\right)
      =\mathrm{Tr}_A\!\left(\left(\sum_k \Pi_k\otimes I_B\right)\rho_{AB}\right)
      =\mathrm{Tr}_A(\rho_{AB}),
\end{align}
where in the third step we used $\Pi_k^2=\Pi_k$ and cyclicity under the trace on subsystem $A$.
\end{proof}

\subsection{Undecidability in universal QCA (and an oracle interpretation)}
The connection between computation and physics suggests that sufficiently expressive local dynamics can encode undecidable questions. We record a minimal reachability statement and then explain how the selection/oracle language used in Part~VII can be read as an interpretation layer rather than as a derived theorem.

\begin{theorem}[Undecidability of local reachability in a universal QCA]
\label{thm:godel_measurement}
Let $\mathcal{U}$ be a universal quantum cellular automaton whose dynamics can simulate a universal \emph{reversible} Turing machine on an appropriate computational subspace. Then there exist a finite region $K$, a local projector $\Pi_K$ supported on $K$, and a family of (computational-basis) initial states $\{|\psi_{M,x}\rangle\}$ encoding a machine $M$ and input $x$ such that the reachability problem
\begin{equation}
    \text{``Does there exist }t\ge 0\text{ such that }\Pi_K\text{ clicks at time }t\text{?''}
\end{equation}
is algorithmically undecidable as a function of $(M,x)$.
\end{theorem}
\begin{proof}
By universality, for each pair $(M,x)$ there exists an encoding $|\psi_{M,x}\rangle$ whose time evolution under $\mathcal{U}$ simulates the reversible computation of $M$ on input $x$. Choose the local projector $\Pi_K$ to test a designated halting-flag register within the simulation region $K$. Then $\Pi_K$ clicks at some time $t$ if and only if $M$ halts on $x$. If the reachability question were decidable for all $(M,x)$, the Halting Problem would be decidable, a contradiction.
\end{proof}

From the computational viewpoint, such a map can be read as an oracle input that resolves undecidable reachability questions by selecting a consistent continuation. This is an interpretation of how selection could function as an external resource; it is not implied by the undecidability theorem itself.

\subsection{Born Rule from projection-induced readout (counting template)}
We record a symmetry-based counting template in which probabilities are induced by finite-resolution readout (Axiom~O7) together with a sharp-readout symmetry hypothesis on fine-grained micro-realizations (Assumption~\ref{assump:S1}). Related symmetry-based approaches include envariance \cite{Zurek2005}.

\begin{theorem}[Born Rule from projection-induced readout (fine-grained template)]
\label{thm:born_rule}
Let $|\psi\rangle = \sum_k c_k |k\rangle$ be a state in the Omega Network. The probability of observing outcome $k$ is $P_k = |c_k|^2$.
\end{theorem}
\begin{proof}
1. \textbf{Holographic Encoding}: Every state $|k\rangle$ in the bulk corresponds to a boundary code subspace $\mathcal{C}_k$ of dimension $d_k = e^{S(k)}$.
2. \textbf{Fine-Graining}: Approximate the weights by rationals $|c_k|^2 \approx n_k/N_{\mathrm{micro}}$ and represent $|\psi\rangle$ as an equal-weight superposition of $N_{\mathrm{micro}}$ microstates grouped into $n_k$ copies of branch $k$:
\[
|\psi\rangle \propto \sum_{k} \sum_{i=1}^{n_k} |k, i\rangle_{micro}.
\]
3. \textbf{Frame Stability}: The physical "laws" (the Hamiltonian $\hat{H}$) are invariant under unitary swaps of the microstates $|k, i\rangle \leftrightarrow |k, j\rangle$ (Action of the Symmetric Group $S_{N_{\mathrm{micro}}}$).
4. \textbf{Projection-induced micro-measure}: By Axiom~O7, outcome weights are induced by a readout kernel/instrument. Under the sharp-readout symmetry hypothesis (Assumption~\ref{assump:S1}), this kernel is uniform on the admissible micro-realizations within the relevant degeneracy class, so each microstate has probability $p_{micro} = 1/N_{\mathrm{micro}}$.
5. \textbf{Aggregation}: The probability of the macrostate $k$ is the sum over its micro-realizations:
\begin{equation}
    P_k = \sum_{i=1}^{n_k} p_{micro} = \frac{n_k}{N_{\mathrm{micro}}} = |c_k|^2.
\end{equation}
Thus, under fine-graining, Axiom~O7, and Assumption~\ref{assump:S1}, the Born rule follows as a counting identity for the underlying holographic micro-ensemble.
\end{proof}

\subsection{Born rule as the unique noncontextual probability rule (mature theorem)}
The counting template above isolates the structural input needed for a microstate-counting reading: a symmetric sharp-readout limit of the induced readout kernel on fine-grained micro-realizations (Assumption~\ref{assump:S1}). Independently, there is a mathematically sharp characterization of quantum probability rules that can be cited directly.

\begin{theorem}[Gleason--Busch characterization of quantum probabilities]
\label{thm:gleason_busch}
Let $\mathcal{H}$ be a complex Hilbert space. Suppose a probability assignment $\mu$ maps each orthogonal projector $P$ on $\mathcal{H}$ to $\mu(P)\in[0,1]$ and is finitely additive on orthogonal families: for any pairwise orthogonal projectors $\{P_i\}$ with $\sum_i P_i = I$, one has $\sum_i \mu(P_i)=1$.
If $\dim\mathcal{H}\ge 3$, then there exists a unique density matrix $\rho$ such that
\begin{equation}
    \mu(P)=\mathrm{Tr}(\rho\,P)\qquad\text{for all projectors }P.
\end{equation}
In particular, for a pure state $\rho=|\psi\rangle\langle\psi|$ one obtains $\mu(P)=\langle\psi|P|\psi\rangle$ (Born rule for projective measurements).
For $\dim\mathcal{H}=2$, an analogous conclusion holds if one assumes the same additivity on the full set of POVM effects (Busch's extension).
\end{theorem}
\begin{proof}[Reference]
See Gleason's theorem for $\dim\mathcal{H}\ge 3$ \cite{Gleason1957}. For the POVM/effect-algebra extension that covers $\dim\mathcal{H}=2$, see \cite{Busch2003}.
\end{proof}
