
\section{加权有限 \texorpdfstring{$\zeta$}{zeta} 与 Abel 迹接口（有限层）}

定义黄金均值移位的 $m$-块图 $\Gamma_m$（状态为 $X_m$，边为移位追加），其邻接矩阵为 $A_m$。以纤维大小作为列权，定义
$$
M_m=A_m\,\diag(s_m).
$$
定义有限 $\zeta$
$$
\zeta_m(z)=\frac{1}{\det(I-zM_m)}=\det(I-zM_m)^{-1}.
$$
进一步可做归一化 $L_m=M_m/\rho(M_m)$，并研究 Abel-resolvent 迹
$$
T_m(r)=\Tr\big((I-rL_m)^{-1}\big)
$$
在 $\abs{r}<1$ 的全纯性与 $r\to 1^{-}$ 的正则化边界值。该框架与有限型移位/转移算子的经典 $\zeta$ 理论相衔接（参见 \cite{lind1995symbolic,bowenlanford1970zeta}），并为将折叠核嵌入谱—解析域控制提供了一个完全有限、可复现的实验平台。

\subsection{可检验的有限层解析性质}

由于 $M_m$ 为有限维非负矩阵，其谱半径 $\rho(M_m)$ 可由 Perron--Frobenius 理论刻画，并为下述解析展开提供自然收敛域。

\begin{proposition}[有限 $\zeta_m$ 的解析域与迹展开]\label{prop:zeta-analytic}
令 $M$ 为任意复方阵，设其谱半径为 $\rho(M)$。则在 $\abs{z}<1/\rho(M)$ 上，
$$
\det(I-zM)\ne 0,
$$
从而 $\zeta(z):=\det(I-zM)^{-1}$ 在该圆盘内全纯，并满足迹展开
$$
\log \zeta(z)=\sum_{n\ge 1}\frac{z^n}{n}\Tr(M^n),\qquad
\zeta(z)=\exp\!\left(\sum_{n\ge 1}\frac{z^n}{n}\Tr(M^n)\right).
$$
\end{proposition}

\begin{proof}
当 $\abs{z}<1/\rho(M)$ 时 $\rho(zM)<1$，因此 $I-zM$ 可逆且 Neumann 级数收敛：
$$
(I-zM)^{-1}=\sum_{n\ge 0} z^n M^n.
$$
在同一收敛域内矩阵对数可由幂级数定义：
$$
\log(I-zM)=-\sum_{n\ge 1}\frac{z^n}{n}M^n.
$$
两边取迹并使用有限维恒等式 $\log\det(I-zM)=\Tr\log(I-zM)$，得到
$$
\log\det(I-zM)=-\sum_{n\ge 1}\frac{z^n}{n}\Tr(M^n),
$$
从而结论成立。
\end{proof}

\begin{proposition}[Abel-resolvent 迹的幂级数展开]\label{prop:abel-trace}
令 $L$ 为任意方阵。则在 $\abs{r}<1/\rho(L)$ 上，
$$
T(r):=\Tr((I-rL)^{-1})
$$
全纯且满足
$$
T(r)=\sum_{n\ge 0} r^n\,\Tr(L^n).
$$
特别地，对归一化 $L_m=M_m/\rho(M_m)$，在 $\abs{r}<1$ 上该展开成立。
\end{proposition}

\begin{proof}
由 Neumann 级数 $(I-rL)^{-1}=\sum_{n\ge 0} r^nL^n$（当 $\rho(rL)<1$）并逐项取迹即可。
\end{proof}
