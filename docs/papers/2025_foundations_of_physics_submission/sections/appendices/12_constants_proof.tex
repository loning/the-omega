\section{Geometric ans\"atze for dimensionless constants}
\label{app:constants_proof}

This appendix records the \emph{closed-form phase-volume model} used in Part~IV for two benchmark dimensionless constants. The guiding principle is that, in a scan--projection readout (Axioms~O5--O6), operational couplings are encoded by \emph{projective} (ray-space) geometry and therefore naturally involve compact phase spaces and $\mathbb{Z}_2$ quotients. The corresponding invariants are taken to be canonical volumes (or serial sums of such volumes) in the minimal model.

\subsection{Regularization and Scale Definition}
In concrete microscopic realizations, such invariants can be computed as orbit traces and, when necessary, as finite parts of regulated orbit sums tied to the scan (Convention~R1). Whenever a constant term depends on regularization choice, we adopt the orbit finite-part convention of Convention~R1 (Section~\ref{conv:orbit_regularization} and Appendix~\ref{app:orbit_calculus}).
\par
\noindent
For comparison with experiment, we take the CODATA recommended low-energy values for $\alpha$ and $\mu=m_p/m_e$ as reference points \cite{CODATA2022RMP}.

\subsection{Canonical volumes of projective phase spaces}
We record the elementary volume identities used in the benchmark computations below.

\begin{lemma}[Volumes of $U(1)$, $SU(2)$, $SO(3)$, and projective quotients]
\label{lem:group_volumes}
With the standard unit-radius normalizations,
\begin{equation}
    \mathrm{Vol}(U(1))=2\pi,\qquad
    \mathrm{Vol}(SU(2))=\mathrm{Vol}(S^3)=2\pi^2.
\end{equation}
Moreover, the $\mathbb{Z}_2$ quotients satisfy
\begin{equation}
    \mathrm{Vol}(SO(3))=\mathrm{Vol}(SU(2)/\{\pm 1\})=\pi^2,\qquad
    \mathrm{Vol}(\mathbb{R}P^1)=\mathrm{Vol}(U(1)/\{\pm 1\})=\pi.
\end{equation}
Consequently,
\begin{equation}
    \mathrm{Vol}(U(1)\times SU(2))=4\pi^3.
\end{equation}
\end{lemma}
\begin{proof}
The first identity is the circumference of the unit circle. For $SU(2)\cong S^3$, use hyperspherical coordinates on $S^3$ and integrate the standard volume form, yielding $\mathrm{Vol}(S^3)=2\pi^2$ (see, e.g., \cite{Nakahara2003}).
\par
\noindent
The quotient maps $SU(2)\to SO(3)=SU(2)/\{\pm 1\}$ and $U(1)\to \mathbb{R}P^1=U(1)/\{\pm 1\}$ are two-sheeted Riemannian coverings under the induced quotient metrics, so the quotient volumes are halved. The product identity follows by multiplicativity of product volumes.
\end{proof}

\begin{proposition}[Primitive factorization rigidity for the electromagnetic strata]
\label{prop:alpha_phase_factorization_rigidity}
Fix the primitive set
\begin{equation}
    \mathfrak{P}\equiv \{U(1),SU(2),SO(3),\mathbb{R}P^1\},
\end{equation}
with canonical volumes as in Lemma~\ref{lem:group_volumes}. Suppose the bulk stratum is realized as a product of two primitives, while the boundary and line strata are each realized by a single primitive, and impose the target scalings
\begin{equation}
    \mathrm{Vol}(\mathcal{M}_{\mathrm{bulk}})=4\pi^3,\qquad
    \mathrm{Vol}(\mathcal{M}_{\mathrm{boundary}})=\pi^2,\qquad
    \mathrm{Vol}(\mathcal{M}_{\mathrm{line}})=\pi.
\end{equation}
Then, up to ordering, the strata are forced as
\begin{equation}
    \mathcal{M}_{\mathrm{bulk}}\cong U(1)\times SU(2),\qquad
    \mathcal{M}_{\mathrm{boundary}}\cong SO(3),\qquad
    \mathcal{M}_{\mathrm{line}}\cong \mathbb{R}P^1.
\end{equation}
\end{proposition}
\begin{proof}
By Lemma~\ref{lem:group_volumes}, the one-factor values in $\mathfrak{P}$ are $2\pi$, $2\pi^2$, $\pi^2$, and $\pi$. The only way to obtain a two-factor product with $\pi$-power $3$ is the exponent pattern $(1,2)$, and the coefficient $4$ forces both factors to carry a factor of $2$, hence the bulk factors must be $U(1)$ and $SU(2)$. For one-factor realizations, $\pi^2$ is realized uniquely by $SO(3)$ and $\pi$ uniquely by $\mathbb{R}P^1$ in $\mathfrak{P}$.
\end{proof}

\subsection{Fine Structure Constant \texorpdfstring{$\alpha$}{alpha}}
\paragraph{Serial composition (process additivity).}
If a readout protocol must satisfy channel constraints in a fixed hierarchy (no parallel bypass), the natural aggregation rule is serial: costs add. We record this as a minimal lemma.

\begin{lemma}[Sequential constraints imply additive effective cost]
\label{lemma:series_impedance_additivity}
Let $C_1,\dots,C_m$ be constraints that must be satisfied in sequence by a readout/transport process, and let $V_i\ge 0$ be the minimal incremental cost required to satisfy $C_i$ after $C_1,\dots,C_{i-1}$. Then the total process cost is additive:
\begin{equation}
    V_{\mathrm{tot}}=\sum_{i=1}^{m} V_i.
\end{equation}
\end{lemma}
\begin{proof}
By definition of a sequential process, each stage must be traversed, so the total cost is the sum of incremental stage costs.
\end{proof}

\paragraph{Three-stratum phase spaces.}
In the minimal electromagnetic readout model, we take three projective strata (bulk/boundary/line) represented by compact phase spaces
\begin{equation}
    \mathcal{M}_{\mathrm{bulk}} \cong U(1)\times SU(2),\qquad
    \mathcal{M}_{\mathrm{boundary}} \cong SO(3),\qquad
    \mathcal{M}_{\mathrm{line}} \cong \mathbb{R}P^1.
\end{equation}
The $\mathbb{Z}_2$ quotients encode the projective identification intrinsic to ray-space readout (Part~II).

\begin{theorem}[Three-channel geometric impedance for $\alpha$]
\label{thm:alpha_three_channel}
Define the inverse coupling as the serial (additive) impedance across the three strata,
\begin{equation}
    \alpha^{-1}_{\mathrm{geo}}\ \equiv\ \mathrm{Vol}(\mathcal{M}_{\mathrm{bulk}})+\mathrm{Vol}(\mathcal{M}_{\mathrm{boundary}})+\mathrm{Vol}(\mathcal{M}_{\mathrm{line}}).
\end{equation}
Then
\begin{equation}
    \alpha^{-1}_{\mathrm{geo}} = 4\pi^3+\pi^2+\pi \approx 137.0363038.
\end{equation}
\end{theorem}
\begin{proof}
By Lemma~\ref{lem:group_volumes},
$\mathrm{Vol}(\mathcal{M}_{\mathrm{bulk}})=\mathrm{Vol}(U(1)\times SU(2))=4\pi^3$,
$\mathrm{Vol}(\mathcal{M}_{\mathrm{boundary}})=\mathrm{Vol}(SO(3))=\pi^2$,
and $\mathrm{Vol}(\mathcal{M}_{\mathrm{line}})=\mathrm{Vol}(\mathbb{R}P^1)=\pi$.
Summing yields the stated expression.
\end{proof}

\subsubsection{Matching and error budget}
Let $\alpha_{\mathrm{exp}}$ denote the CODATA low-energy value \cite{CODATA2022RMP}. We define the mismatch
\begin{equation}
    \Delta\alpha^{-1}\ \equiv\ \alpha^{-1}_{\mathrm{geo}}-\alpha^{-1}_{\mathrm{exp}}.
    \label{eq:alpha_mismatch_def}
\end{equation}
Using the CODATA 2022 central value $\alpha_{\mathrm{exp}}^{-1}\approx 137.035999177$ \cite{CODATA2022RMP}, one finds
\begin{equation}
    \alpha_{\mathrm{geo}}^{-1}\approx 137.0363037759,\qquad
    \Delta\alpha^{-1}\approx 3.046\times 10^{-4},\qquad
    \frac{\Delta\alpha^{-1}}{\alpha_{\mathrm{exp}}^{-1}}\approx 2.223\times 10^{-6}.
\end{equation}
In the multiplicative readout variable $w=\exp(-\alpha^{-1})$, the corresponding matching factor is
\begin{equation}
    s_\alpha\ \equiv\ \frac{w_{\mathrm{exp}}}{w_{\mathrm{geo}}}
    \;=\;\exp\!\bigl(-(\alpha_{\mathrm{exp}}^{-1}-\alpha_{\mathrm{geo}}^{-1})\bigr)
    \;=\;\exp(\Delta\alpha^{-1}).
\end{equation}
Numerically, $s_\alpha\approx 1.000304645$. In the present manuscript $s_\alpha$ is treated as an interface correction between the canonical geometric impedance and the operationally inferred low-energy coupling.

\subsection{Proton-Electron Mass Ratio \texorpdfstring{$\mu$}{mu}}
We treat $\mu=m_p/m_e$ as a dimensionless mass-ratio invariant and model it by a canonical internal phase-volume (cost) assignment.

\begin{axiom}[Mass invariants as phase-volume costs]
\label{ax:mass_volume}
Assign to each particle class $i$ a compact internal phase space $\mathcal{M}_i$ (or a finite disjoint union of such spaces) and define its invariant cost by
\begin{equation}
    I_i \equiv \mathrm{Vol}(\mathcal{M}_i).
\end{equation}
Normalize the electron reference by $I_e=1$. For a baryon in three-color QCD, the proton invariant is the sum over three identical color sectors \cite{PDGRPP2024,PeskinSchroeder1995}.
\end{axiom}

\begin{axiom}[Proton phase space per color sector]
\label{ax:proton_phase_space}
Model one color sector by
\begin{equation}
    \mathcal{M}_q \cong SO(3)\times SO(3)\times U(1),
\end{equation}
and take $\mathcal{M}_p$ to be the disjoint union of three copies of $\mathcal{M}_q$.
\end{axiom}

\begin{theorem}[Proton--electron mass ratio]
\label{thm:mu_geo}
Under Axioms~\ref{ax:mass_volume}--\ref{ax:proton_phase_space}, the geometric value is
\begin{equation}
    \mu_{\mathrm{geo}} \equiv \frac{m_p}{m_e} = 6\pi^5 \approx 1836.1181087.
\end{equation}
\end{theorem}
\begin{proof}
By Axiom~\ref{ax:mass_volume}, $\mu=I_p/I_e=I_p$ since $I_e=1$. By Axiom~\ref{ax:proton_phase_space},
$I_p=3\,\mathrm{Vol}(SO(3)\times SO(3)\times U(1))$.
Using Lemma~\ref{lem:group_volumes}, $\mathrm{Vol}(SO(3))=\pi^2$ and $\mathrm{Vol}(U(1))=2\pi$, hence
$I_p=3(\pi^2)(\pi^2)(2\pi)=6\pi^5$.
\end{proof}

\begin{proposition}[Minimal factor count for the \texorpdfstring{$\pi^5$}{pi^5} scaling]
\label{prop:mu_min_factor_count}
Fix the primitive set
\begin{equation}
    \mathfrak{P}\equiv \{U(1),SU(2),SO(3),\mathbb{R}P^1\},
\end{equation}
with canonical volumes as in Lemma~\ref{lem:group_volumes}. Then any product of two primitives has $\pi$-power at most $4$. Consequently, achieving the $\pi^5$ scaling in $\mu_{\mathrm{geo}}$ per color sector forces at least three primitive factors.
\end{proposition}
\begin{proof}
In $\mathfrak{P}$, each primitive contributes at most a $\pi^2$ factor to its volume, so a product of two primitives contributes at most $\pi^4$.
\end{proof}

\begin{proposition}[Finite primitive factorization rigidity for \texorpdfstring{$\mu$}{mu}]
\label{prop:mu_phase_factorization_rigidity}
Fix the primitive set $\mathfrak{P}$ and restrict a per-color sector $\mathcal{M}_q$ to be a product of three primitives (order irrelevant). Then the condition
\begin{equation}
    3\,\mathrm{Vol}(\mathcal{M}_q)=6\pi^5
\end{equation}
holds if and only if
\begin{equation}
    \mathcal{M}_q\cong SO(3)\times SO(3)\times U(1)
    \qquad\text{or}\qquad
    \mathcal{M}_q\cong SO(3)\times SU(2)\times \mathbb{R}P^1.
\end{equation}
\end{proposition}
\begin{proof}
Any three-factor product from $\mathfrak{P}$ has the form $\mathrm{Vol}(\mathcal{M}_q)=2^{m}\pi^{k}$, where $m$ counts how many factors are drawn from $\{U(1),SU(2)\}$ and $k$ is the sum of exponents $1$ (for $U(1)$ or $\mathbb{R}P^1$) and $2$ (for $SU(2)$ or $SO(3)$). The constraint $\mathrm{Vol}(\mathcal{M}_q)=2\pi^5$ forces $m=1$ and $k=5$, hence exactly two factors must contribute exponent $2$ and one factor exponent $1$.
\par
\noindent
If the exponent-$1$ factor is $U(1)$, keeping $m=1$ forces both exponent-$2$ factors to be $SO(3)$, giving $SO(3)\times SO(3)\times U(1)$. If the exponent-$1$ factor is $\mathbb{R}P^1$, then exactly one exponent-$2$ factor must be $SU(2)$ and the other $SO(3)$ to achieve $m=1$, giving $SO(3)\times SU(2)\times \mathbb{R}P^1$.
\end{proof}

\subsubsection{Matching and error budget}
Let $\mu_{\mathrm{exp}}$ denote the CODATA value \cite{CODATA2022RMP}. Define the mismatch
\begin{equation}
    \Delta\mu\ \equiv\ \mu_{\mathrm{geo}}-\mu_{\mathrm{exp}}.
    \label{eq:mu_mismatch_def}
\end{equation}
Equivalently, define the multiplicative matching factor
\begin{equation}
    s_\mu \equiv \frac{\mu_{\mathrm{exp}}}{\mu_{\mathrm{geo}}}.
\end{equation}
Using the CODATA 2022 central value $\mu_{\mathrm{exp}}\approx 1836.15267343$ \cite{CODATA2022RMP}, one finds
\begin{equation}
    \mu_{\mathrm{geo}}\approx 1836.1181087117,\qquad
    \Delta\mu\approx -3.456\times 10^{-2},\qquad
    \frac{\Delta\mu}{\mu_{\mathrm{exp}}}\approx -1.882\times 10^{-5},
\end{equation}
so that $s_\mu\approx 1.000018825$. In the closed-theory reading, $s_\mu$ encodes the nonperturbative hadronic and radiative structure that relates the canonical internal phase-volume normalization to the operationally inferred mass ratio.

\subsection{Rigidity bound for local temporal drift of coarse-grained couplings}
\label{app:rigidity_drift}
The drift estimates in Part~VI use only a minimal ``averaging'' mechanism: if an effective coupling is a coarse-grained average over many microscopic contributions, then any bounded-rate microscopic update produces a parametrically suppressed macroscopic drift.

\begin{proposition}[Averaging bound]
\label{prop:averaging_bound}
Let $g(t)$ be an effective (dimensionless) coupling defined as an average over $N_{\mathrm{dof}}$ bounded microscopic contributions (not to be confused with the lapse $N(t)$ used elsewhere in this manuscript),
\begin{equation}
    g(t)=\frac{1}{N_{\mathrm{dof}}}\sum_{i=1}^{N_{\mathrm{dof}}} g_i(t),
\end{equation}
and assume that during a time interval of duration $\Delta t$ at most $M$ of the $g_i$ change, each by at most $|\Delta g_i|\le \delta$. Then
\begin{equation}
    |\Delta g|\ \le\ \frac{M}{N_{\mathrm{dof}}}\,\delta.
\end{equation}
If $g(t)$ is strictly positive and bounded below by $g_{\min}>0$, then the fractional drift obeys
\begin{equation}
    \left|\frac{\Delta g}{g}\right|\ \le\ \frac{M}{N_{\mathrm{dof}}}\,\frac{\delta}{g_{\min}}.
\end{equation}
\end{proposition}
\begin{proof}[Reference]
This is immediate from the triangle inequality and the assumed bounds on the number and magnitude of microscopic changes.
\end{proof}

\paragraph{Holographic scaling of $N_{\mathrm{dof}}$.}
In holographic near-saturation regimes, one may identify $N_{\mathrm{dof}}$ with the number of independent boundary degrees of freedom in a causal patch, $N_{\mathrm{dof}}\sim A/(4l_P^2\ln 2)$, so that any bounded-rate microscopic update yields a very small macroscopic drift. The estimate is purely kinematic; the remaining dynamical question is to determine the effective update rate $M$ and the microscopic step size $\delta$ for a concrete Omega network.
