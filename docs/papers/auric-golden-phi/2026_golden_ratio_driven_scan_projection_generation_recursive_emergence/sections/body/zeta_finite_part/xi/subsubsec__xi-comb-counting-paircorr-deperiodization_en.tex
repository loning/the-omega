\paragraph{Counting constant, correlation degeneracy, and deperiodization at incommensurate scales for single-scale completed combs}
According to the "phase-coordinate gate—finite-window audit—orbit–Levi rigidity" juxtaposition caliber fixed by internal manuscript~\cite{Internal2025ResolutionFoldingPhiPi} (2025), without introducing additional axioms, single-scale completion compresses zero/pole sets into finitely many equally-spaced vertical line clusters; this compression simultaneously provides a class of canonical normalization mechanisms intrinsically determined by rigidity, manifesting analytically as counting constants and correlation degeneracy of comb structures.\footnote{Archive timestamp for this section's entries: 2026-02-14 10:22:24 (Asia/Singapore).}
On the other hand, the golden-base Mellin fingerprint (Theorem~\ref{thm:gm-mellin-meromorphic-fingerprint} and subsequent corollaries) induces another set of base combs with fundamental imaginary period \(2\pi i/\log\varphi\); unless the two scales are commensurate, the two comb classes share no nonzero purely imaginary period. More generally, merely superposing two incommensurate bases suffices to destroy strict imaginary periodicity, thus explaining at the structural level the minimal discrete deperiodization mechanism required for transitioning "from finite lattice to non-lattice."

\begin{proposition}[Leading-constant of single-scale completed zero count: rigidly determined by \texorpdfstring{$\deg P_L$}{deg PL}]\label{prop:xi-single-scale-completed-zero-count-leading-constant}
Fix \(L>1\) and follow the completion notation of \S\ref{subsec:xi-from-self-dual}:
let
\(
P_L(S):=\widehat\Delta(L^{-1/2},S)\in\RR[S]
\)
with \(d:=\deg P_L\), and let
\[
\Phi_L(y):=y^{d}P_L(y+y^{-1})\in\RR[y]
\]
as in Corollary~\ref{cor:xi-rh-unit-circle-reciprocal}.
Denote \(N_{\Xi}(T;\sigma_1,\sigma_2)\) as the number of zeros (counted with multiplicity) of \(\Xi_{\Delta,L}\) in the strip \(\sigma_1\le \Re(s)\le \sigma_2\) with \(|\Im(s)|\le T\).
Then there exists a constant \(C(\sigma_1,\sigma_2)\) such that
\[
N_{\Xi}(T;\sigma_1,\sigma_2)
=\frac{\log L}{\pi}\,T\cdot
\#\!\left\{
y\in\CC:\ \Phi_L(y)=0,\
\sigma_1\le \tfrac12+\frac{\log|y|}{\log L}\le \sigma_2
\right\}
+O(1)\qquad(T\to\infty).
\]
In particular, over the full strip \((\sigma_1,\sigma_2)=(-\infty,\infty)\) we have
\[
N_{\Xi}(T)
=\frac{2d\,\log L}{\pi}\,T+O(1).
\]
\end{proposition}

\begin{proof}
By \(\Phi_L(y)=0\) if and only if \(P_L(y+y^{-1})=0\) (and \(\Phi_L(0)=\mathrm{lc}(P_L)\neq 0\)), and letting \(y:=L^{s-1/2}\) (proof of Proposition~\ref{prop:xi-rh-interval-criterion}), we obtain
\[
\Xi_{\Delta,L}(s)=0\quad\Longleftrightarrow\quad \Phi_L\!\bigl(L^{s-1/2}\bigr)=0.
\]
Thus each root \(y\in\CC^\times\) of \(\Phi_L\) (counted with multiplicity) induces a zero tower
\[
s=\frac12+\frac{\log y+2\pi i k}{\log L},\qquad k\in\ZZ,
\]
where \(\log\) takes any branch and \(2\pi i k\) represents the multi-valuedness of the complex logarithm.
For fixed \(y\), the number of points in this arithmetic progression within the height window \(|\Im(s)|\le T\) is \(\frac{\log L}{\pi}T+O(1)\);
moreover, the strip constraint is equivalent to
\(
\sigma_1\le \frac12+\frac{\log|y|}{\log L}\le \sigma_2
\).
Summing over all roots yields the first formula.
Over the full strip, the number of roots is \(\deg\Phi_L=2d\), yielding
\(
N_{\Xi}(T)=\frac{2d\,\log L}{\pi}T+O(1)
\).
\end{proof}

\begin{proposition}[Two-point correlation degeneracy to Dirac comb for single-scale completion: pure point spectrum under normalized spacing]\label{prop:xi-single-scale-two-point-correlation-comb}
Maintain the setup of Proposition~\ref{prop:xi-single-scale-completed-zero-count-leading-constant}, and let \(\mathcal Z\) be the zero multiset of \(\Xi_{\Delta,L}\).
Denote the imaginary step
\(
\Delta_t(L):=2\pi/\log L
\).
For any compactly-supported continuous function \(\psi\in C_c(\RR)\), define the normalized two-point statistic
\[
\mathcal R_T(\psi)
:=\frac{1}{N_{\Xi}(T)}
\sum_{\substack{s,s'\in\mathcal Z\\ |\Im(s)|,|\Im(s')|\le T}}
\psi\!\left(\frac{\Im(s)-\Im(s')}{\Delta_t(L)}\right).
\]
Then the limit \(\lim_{T\to\infty}\mathcal R_T(\psi)\) exists and is a linear combination of finitely many translated Dirac combs;
more specifically, there exists a finite set \(A\subset[0,1)\) and non-negative weights \((w_a)_{a\in A}\) (determined by phase offsets of \(\Phi_L\) root towers) such that
\[
\lim_{T\to\infty}\mathcal R_T(\psi)
=\sum_{a\in A} w_a\sum_{n\in\ZZ}\psi(n+a).
\]
\end{proposition}

\begin{proof}
By Proposition~\ref{prop:xi-single-scale-completed-zero-count-leading-constant}, \(\mathcal Z\) is the union of finitely many arithmetic vertical line towers (counted with multiplicity).
For any root \(y\), write \(y=|y|e^{i\vartheta}\) and let \(\alpha:=\vartheta/(2\pi)\in[0,1)\); then the imaginary-part set of the corresponding zero tower is
\[
\Im(s)=\Delta_t(L)\,(\ZZ+\alpha).
\]
Thus for any two towers, their imaginary-part difference set lies in
\(
\Delta_t(L)\,(\ZZ+(\alpha-\beta))
\);
after normalization, this becomes \(\ZZ+(\alpha-\beta)\).
Summing over finitely many towers yields a finite set \(A\) (formed by all phase-offset difference congruence classes).
Since each tower has uniform linear count within the height window, the compactly-supported test function \(\psi\) samples only finitely many integer translates, so the limiting weight contribution from each pair of towers exists; summing over all tower pairs yields the displayed form.
\end{proof}

\begin{proposition}[Incommensurability of two comb scales: no common imaginary period except \texorpdfstring{$\log L/\log\varphi\in\QQ$}{log L/log phi in Q}]\label{prop:xi-comb-incommensurability-logL-logphi}
Let \(L>1\), and let the fundamental imaginary period of single-scale Dirichletization be
\(
T_L:=2\pi i/\log L
\)
(Proposition~\ref{prop:single-scale-dirichlet-imag-period}).
The fundamental imaginary period for golden-base Mellin variable \(z=\varphi^{-s}\) is
\(
T_\varphi:=2\pi i/\log\varphi
\).
Then there exists nonzero \(T\in i\RR\) belonging simultaneously to \(T_L\ZZ\) and \(T_\varphi\ZZ\) if and only if \(\log L/\log\varphi\in\QQ\), equivalently \(L=\varphi^{p/q}\) (for some coprime integers \(p,q\)).
In particular, for any integer \(L\ge 2\), we have \(\log L/\log\varphi\notin\QQ\), so the two comb classes cannot share any nonzero common period.
\end{proposition}

\begin{proof}
If \(T\in T_L\ZZ\cap T_\varphi\ZZ\), then there exist nonzero integers \(m,n\) such that
\(
T=2\pi i m/\log L=2\pi i n/\log\varphi
\), so \(\log L/\log\varphi=m/n\in\QQ\).
The converse directly constructs a common period from the same equation.
If \(L\in\ZZ_{\ge2}\) and \(\log L/\log\varphi=p/q\), then \(L^q=\varphi^p\).
The left side is rational while the right side is a quadratic irrational (for \(p\neq 0\), \(\varphi^p\notin\QQ\)), a contradiction.
\end{proof}

\begin{theorem}[Deperiodization by double-base superposition: two incommensurate scales suffice to destroy strict imaginary periodicity]\label{thm:xi-two-base-deperiodization-no-imag-period}
Take \(L_1,L_2>1\) with \(\log L_1/\log L_2\notin\QQ\).
Let \(g,h\) be meromorphic functions of one variable.
Further assume neither has any nontrivial unit-root multiplicative invariance: there do not exist \(\omega\neq 1\) and integer \(n\ge 2\) such that \(\omega^n=1\) and \(g(\omega z)\equiv g(z)\) or \(h(\omega z)\equiv h(z)\).
Define
\[
f(s):=g(L_1^{-s})+h(L_2^{-s}).
\]
Then \(f\) has no nonzero purely imaginary period \(T\in i\RR\setminus\{0\}\).
\end{theorem}

\begin{proof}
Suppose to the contrary there exists \(T=i\tau\neq 0\) such that \(f(s+T)\equiv f(s)\).
Let \(\omega_1:=e^{-T\log L_1}\), \(\omega_2:=e^{-T\log L_2}\); then \(|\omega_1|=|\omega_2|=1\) and
\[
g(\omega_1 L_1^{-s})-g(L_1^{-s})=h(L_2^{-s})-h(\omega_2 L_2^{-s})
\]
holds as a meromorphic identity in \(s\).
\par
Take \(r>0\) such that \(g,h\) have Laurent expansions in the punctured disk \(0<|z|<r\)
\[
g(z)=\sum_{m\ge -M_g} a_m z^m,\qquad
h(z)=\sum_{n\ge -M_h} b_n z^n
\qquad(0<|z|<r),
\]
where \(M_g,M_h\ge 0\).
Take \(\sigma_0\) sufficiently large so that when \(\Re(s)\ge \sigma_0\), both \(0<|L_1^{-s}|<r\) and \(0<|L_2^{-s}|<r\).
Substituting \(z=L_i^{-s}\) gives
\[
g(L_1^{-s})=\sum_{m\ge -M_g} a_m L_1^{-ms},\qquad
h(L_2^{-s})=\sum_{n\ge -M_h} b_n L_2^{-ns}
\qquad(\Re(s)\ge \sigma_0).
\]
By periodicity \(f(s+T)\equiv f(s)\) we have
\[
0=f(s+T)-f(s)
=\sum_{m\ge -M_g} a_m(\omega_1^m-1)L_1^{-ms}
\;+\sum_{n\ge -M_h} b_n(\omega_2^n-1)L_2^{-ns}.
\]
Since \(\log L_1/\log L_2\notin\QQ\), the set \(\{m\log L_1:\ m\in\ZZ\}\cup\{n\log L_2:\ n\in\ZZ\}\) has no overlaps except at \(0\);
by uniqueness of Dirichlet-type exponential series (or exponential polynomials with absolutely-convergent tails), we must have for all \(m,n\)
\(
a_m(\omega_1^m-1)=0
\) and
\(
b_n(\omega_2^n-1)=0
\).
Thus for all \(m\) with \(a_m\neq 0\) we have \(\omega_1^m=1\), and for all \(n\) with \(b_n\neq 0\) we have \(\omega_2^n=1\).
If \(a_m=0\) for all \(m\neq 0\), then \(g\) is constant, so for any unit root \(\omega\neq 1\) we have \(g(\omega z)\equiv g(z)\), contradicting the hypothesis;
hence there exists \(m\neq 0\) with \(a_m\neq 0\), so \(\omega_1\) is a unit root; similarly \(\omega_2\) is also a unit root.
Moreover, on \(0<|z|<r\) we have
\[
g(\omega_1 z)
=\sum_{m\ge -M_g} a_m(\omega_1 z)^m
=\sum_{m\ge -M_g} a_m\omega_1^m z^m
=\sum_{m\ge -M_g} a_m z^m
=g(z),
\]
so by meromorphic continuation \(g(\omega_1 z)\equiv g(z)\);
similarly \(h(\omega_2 z)\equiv h(z)\).
By the unit-root invariance hypothesis, we can only have \(\omega_1=\omega_2=1\).
\par
Thus \(T\in (2\pi i/\log L_1)\ZZ\) and \(T\in (2\pi i/\log L_2)\ZZ\).
By \(\log L_1/\log L_2\notin\QQ\), the intersection of the two lattices contains only \(0\), contradicting \(T\neq 0\).
\end{proof}
