\section{Binary Readout: Mechanical Words and Sturmian Complexity Invariants}

\subsection{Binary Partition Readout and Symbolization}

\begin{definition}[Binary partition readout]\label{def:binary-partition-readout}
Take a measurable partition of $\TT^1$
\[
\mathcal{P}=\{I_0,I_1\},\qquad I_0\cap I_1=\varnothing,\ I_0\cup I_1=\TT^1,
\]
and define the binary readout sequence
\[
w_t:=\ind_{I_1}(\theta_t)\in\{0,1\}.
\]
Call $w=(w_t)_{t\ge 0}$ the mechanical word induced by $(\alpha,\mathcal{P},\theta_0)$.
\end{definition}

\subsection{Sturmian Properties and Minimal Complexity}

\begin{theorem}[Sturmian minimal complexity]\label{thm:sturmian-min-complexity}
If $\alpha$ is irrational and $\mathcal{P}$ is a two-interval partition (excluding degenerate cases where endpoints fall in orbits), then the mechanical word $w$ is a Sturmian sequence; its factor complexity satisfies
\[
p(n)=n+1,\qquad \forall n\ge 1,
\]
where $p(n)$ is the number of distinct subwords of length $n$.
\end{theorem}

\begin{remark}
Proof outline is given in Appendix \ref{app:sturmian-outline}. The core is: symbolization of irrational rotation yields balanced sequences; two-letter non-periodic with minimal complexity is equivalent to Sturmian.
\end{remark}

\subsection{Golden Slope Realization (Optional Model)}

\begin{assumption}[Golden slope choice]\label{assump:golden-slope}
Take the continued fraction partial quotients of $\alpha$ to all be $1$ (equivalent to $\alpha=\varph^{-1}$ or its linear fractional transform image), then the induced Sturmian sequence is a Fibonacci-type special case; this choice has extremal anti-resonance property in the sense of minimal approximation coefficients, and provides the simplest binary canonical form structure in subsequent Zeckendorf stabilization.
\end{assumption}
