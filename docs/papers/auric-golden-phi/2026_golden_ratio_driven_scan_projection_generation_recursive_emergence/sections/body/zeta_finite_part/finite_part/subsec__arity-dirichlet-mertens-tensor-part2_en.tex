\begin{lemma}[Marginal summation equals character projection]\label{lem:arity-character-projection}
Let \(m_1,m_2,m_3\ge 2\), write
\(
G:=(\ZZ/m_1)\times(\ZZ/m_2)\times(\ZZ/m_3)
\)
with \(|G|=m_1m_2m_3\), and let \(\omega_{m_i}:=e^{2\pi i/m_i}\).
Given a constant \(\mathsf{M}\in\RR\) and a family of complex numbers \(C(j_1,j_2,j_3)\) (\(j_i\in\ZZ/m_i\)), define
\[
C^{(m_1,m_2,m_3)}_{a,b,c}
:=
\frac{\mathsf{M}}{|G|}
\frac{1}{|G|}
\!\!\!\sum_{\substack{0\le j_1<m_1\\0\le j_2<m_2\\0\le j_3<m_3\\(j_1,j_2,j_3)\ne(0,0,0)}}
\omega_{m_1}^{-j_1a}\,\omega_{m_2}^{-j_2b}\,\omega_{m_3}^{-j_3c}\,C(j_1,j_2,j_3),
\qquad (a,b,c)\in G.
\]
Then for any \(a\in\ZZ/m_1\) and \(c\in\ZZ/m_3\), the following strict identities hold:
\[
\sum_{b\in\ZZ/m_2}\sum_{c\in\ZZ/m_3} C^{(m_1,m_2,m_3)}_{a,b,c}
=
\frac{\mathsf{M}}{m_1}
\frac{1}{m_1}\sum_{j_1=1}^{m_1-1}\omega_{m_1}^{-j_1a}\,C(j_1,0,0),
\]
\[
\sum_{a\in\ZZ/m_1}\sum_{b\in\ZZ/m_2} C^{(m_1,m_2,m_3)}_{a,b,c}
=
\frac{\mathsf{M}}{m_3}
\frac{1}{m_3}\sum_{j_3=1}^{m_3-1}\omega_{m_3}^{-j_3c}\,C(0,0,j_3).
\]
More generally, marginal summation over any subset of coordinate indices is equivalent to projecting in character space onto the subfamily of characters with zero in the corresponding coordinates; the summed-out coordinates are completely eliminated by character orthogonality, removing all nontrivial Fourier modes in those directions.
\end{lemma}

\begin{proof}
Summing the definition term by term and using character orthogonality on finite cyclic groups:
\[
\sum_{x=0}^{m-1}\omega_m^{-jx}
=
\begin{cases}
m,& j\equiv 0\ (\mathrm{mod}\ m),\\
0,& \text{otherwise}.
\end{cases}
\]
For example, when summing over \(b\) and \(c\), only terms satisfying \(j_2\equiv 0\) and \(j_3\equiv 0\) survive, yielding the first identity; the second identity follows analogously. The general case is entirely analogous. \qed
\end{proof}

\left|\mathcal{C}(j)-\mathcal{C}_K(j)\right|
\;\le\;
\sum_{k>K}\frac{1}{k}\cdot
\frac{d\,(\rho_j/\lambda)^k}{1-(\rho_j/\lambda)^k},
\]
where \(d\) is the dimension of the twisted matrix (in this section, \(d=40\)), so the right-hand side is \(O((\rho_j/\lambda)^{K+1})\).
\end{remark}

\begin{remark}[Projection functoriality of the Dirichlet--Mertens constant tensor under label refinement]\label{thm:dirichlet-mertens-functoriality}
Let \(G\) be a finite abelian label group, \(\pi:G\to H\) a surjection (e.g., coordinate projection onto some factor group/quotient group). Let
\(\{C^G_g\}_{g\in G}\) be the \(G\)-stratified Dirichlet--Mertens constant tensor (Abel/Hadamard finite-part calibration, satisfying \(\sum_{g\in G}C^G_g=\mathsf{M}\)), and similarly define the \(H\)-stratified constants \(\{C^H_h\}_{h\in H}\).
Then the constant tensor satisfies the fiber summation law under projection:
\[
C^H_h=\sum_{g:\ \pi(g)=h}C^G_g.
\]
In particular, performing any marginal summation on the three-dimensional constant tensor \(C_{a,b,c}\) (e.g., \(\sum_c C_{a,b,c}\) or \(\sum_{b,c}C_{a,b,c}\)) yields marginal constants equal to the two-dimensional/one-dimensional stratified constants ``retaining only the corresponding coordinate labels''; hence label refinement does not alter the marginal constants of any existing coordinate, but only introduces ``pure interaction residuals'' at higher dimensions.
\end{remark}

\begin{remark}[Constant-level bias along collision congruence classes (\texorpdfstring{$S_c$}{Sc} decomposition)]\label{cor:arity-335-collision-residue-drift}
Let
\[
S_c:=\sum_{a=0}^{2}\sum_{b=0}^{2}C^{(3,3,5)}_{a,b,c},
\qquad c\in\ZZ/5.
\]
Then under the \(((3,3,5))\) stratification (third axis $c=N_2(\gamma)\bmod 5$) of the real-input 40-state kernel, the reproducible output shows: $S_0$ is significantly positive, while $S_2,S_3,S_4$ are negative, and $\sum_{c=0}^4 S_c=\mathsf{M}$. Therefore at the Abel/Hadamard finite-part (constant-level drift) calibration, the bias of primitive orbits toward collision congruence class $c$ is mainly manifested as ``strong positive drift at $c=0$ + systematic negative drift at remaining congruence classes''.
More strongly, by Corollary \ref{cor:arity-335-marginal-invert-collision}, this \(c\)-axis marginal is entirely determined by the four pure collision twisted constants \(C(0,0,1..4)\), and can be recovered from \(S_c\) by discrete Fourier inversion.
\end{remark}

\begin{remark}[Sign phase-transition threshold and positive-drift sparsity]\label{cor:arity-335-sign-sparsity}
Entry-by-entry inspection of the reproducible output of the five slices of \(((3,3,5))\) shows:
\[
C^{(3,3,5)}_{a,b,c}>0\ \Longrightarrow\ c\in\{0,1\},
\]
and when $c=2,3,4$, all $9$ grid points are strictly negative. Hence positive drift occurs only in the $c=0,1$ slices. Further counting gives
\[
\#\{C>0\}=9,\qquad \#\{C<0\}=36,
\]
and
\[
\sum_{C>0}C\approx 2.303140,\qquad
\sum_{C<0}C\approx -1.980237,\qquad
\sum_{a,b,c}C^{(3,3,5)}_{a,b,c}=\mathsf{M}.
\]
Therefore the total constant $\mathsf{M}$ is a strong cancellation residual between ``large positive drift'' and ``large negative drift'', rather than a one-sided accumulation.
\end{remark}

\begin{remark}[Cancellation factor and resonant cell]\label{rem:arity-335-cancel-resonant}
Define the cancellation factor
\[
\kappa_{\mathrm{cancel}}:=\frac{\sum_{C>0}C}{\sum_{a,b,c}C^{(3,3,5)}_{a,b,c}}
=\frac{\sum_{C>0}C}{\mathsf{M}},
\]
then for the \(((3,3,5))\) output,
\(
\kappa_{\mathrm{cancel}}\approx 7.13
\), showing that $\mathsf{M}$ is the residual after strong positive-negative drift cancellation.
Moreover, from the reproducible table output, the largest single cell occurs at
\(
(a,b,c)=(1,0,0)
\), with value
\(
C^{(3,3,5)}_{1,0,0}\approx 0.95882121
\),
which is already approximately $2.97$ times the total constant $\mathsf{M}$, reflecting the dominant contribution of the ``resonant cell'' to the constant term.
\end{remark}
