\section{Jordan/triality candidate: why ``three'' as an audit-facing internal-sector template (optional)}
\label{app:jordan_triality_candidate}

\paragraph{Scope and status.}
\AuditTag This appendix records a compact candidate template (not used in any closure): if one admits an internal observable algebra described by a formally real Jordan algebra and allows octonionic coordinates, then an exceptional $3\times 3$ structure appears canonically.
The purpose is organizational: it supplies an audit-facing vocabulary for ``three internal sectors'' and their symmetry constraints.
No statement here is used as a premise in theorem-level folding proofs or in SM labeling closure.

\subsection{Formally real Jordan algebras and the exceptional $3\times 3$ case}
\label{subsec:jordan_formally_real_candidate}

\begin{definition}[Jordan algebra and formal reality]\label{def:jordan_formally_real_candidate}
\MathTag A (real) Jordan algebra is a real vector space $J$ with a bilinear product $A\circ B$ that is commutative and satisfies the Jordan identity
\[
A^2\circ(A\circ B)=A\circ(A^2\circ B),\qquad A^2:=A\circ A.
\]
It is \emph{formally real} if $\sum_{i=1}^k A_i^2=0$ implies $A_i=0$ for all $i$.
\end{definition}

\begin{theorem}[Jordan--von Neumann--Wigner classification (standard)]\label{thm:jnww_candidate}
\MathTag Every finite-dimensional formally real Jordan algebra is a direct sum of simple ideals, each isomorphic to one of the following \cite{JordanVonNeumannWigner1934}:
\begin{itemize}
\item Hermitian matrices $H_n(\RR)$, $H_n(\CC)$, or $H_n(\mathbb{H})$ with $A\circ B=\tfrac12(AB+BA)$;
\item a spin factor;
\item the exceptional Albert algebra $J_3(\OO)=H_3(\OO)$ \cite{Albert1934}.
\end{itemize}
Moreover, the octonionic Hermitian construction yields a Jordan algebra only in the $3\times 3$ case.
\end{theorem}

\begin{definition}[Albert algebra]\label{def:albert_candidate}
\MathTag Let $J_3(\OO)$ denote the space of $3\times 3$ Hermitian matrices over $\OO$, equipped with the Jordan product $A\circ B=\tfrac12(AB+BA)$.
\end{definition}

\subsection{Triality as a three-sector symmetry template (audit-facing)}
\label{subsec:triality_candidate}

\AuditTag The exceptional $3\times 3$ form carries three distinguished off-diagonal octonionic slots.
In a representation-theoretic language, the $\mathrm{Spin}(8)$ triality symmetry permutes three $8$-dimensional irreducible representations, and an $S_3$ outer symmetry permutes the corresponding slots; see, e.g., \cite{Baez2002,GunaydinGursey1973}.
In this manuscript, this is recorded only as an optional symmetry template for a ``three internal sectors'' narrative.
