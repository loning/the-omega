
\section{Arithmetic appendix: holographic trace formulas and a conditional Riemann critical-line rigidity theorem}
\label{app:trace_rigidity}

\subsection{Geometric-side holomorphy on the unit disk}
\label{sec:geom_holomorphy}

In the closed layer, Abel-regularized traces are primary objects.
\begin{lemma}[Bounded scan readout implies unit-disk holomorphy]
\label{lem:bounded_trace_holomorphy}
Let $U$ be a unitary on $\mathcal{H}_{\mathrm{eff}}$, let $\omega_{\mathrm{eff}}$ be an effective state, and let $A$ be a bounded operator.
Define the scan-readout sequence
\[
a_t:=\omega_{\mathrm{eff}}\!\left(U^t A U^{-t}\right),\qquad t\in\ZZ_{\ge 0}.
\]
Then $(a_t)$ is bounded, and the Abel generating function
\[
G(r):=\sum_{t\ge 0} a_t r^t
\]
converges absolutely for $|r|<1$ and is holomorphic on the unit disk.
\end{lemma}

\begin{proof}
Since $U$ is unitary, $\|U^t A U^{-t}\|=\|A\|$.
For any state $\omega_{\mathrm{eff}}$ on a $C^\ast$-algebra one has $\abs{\omega_{\mathrm{eff}}(B)}\le \|B\|$, hence $\abs{a_t}\le \|A\|$ for all $t$.
Absolute convergence for $|r|<1$ and holomorphy of the resulting power series are immediate.
\end{proof}

Lemma~\ref{lem:bounded_trace_holomorphy} provides the geometric-side analyticity constraint mandated by bounded scan readout (Axioms~\ref{ax:O5}--\ref{ax:O6}).
This unit-disk holomorphy will be paired with the spectral-side mode structure under HTF.

\begin{remark}[Concrete scan models realizing the holomorphy condition]
\label{rem:concrete_holomorphy_models}
Lemma~\ref{lem:bounded_trace_holomorphy} is intentionally minimal: it isolates the unit-disk analyticity that follows from bounded readout, independent of any arithmetic content.
A concrete realization consistent with the Weyl-pair setting is obtained by taking $\mathcal{H}_{\mathrm{eff}}=L^2(\TT)$, letting $U$ be the irrational rotation shift $(U\psi)(x)=\psi(x+\alpha)$, choosing $A$ as a bounded multiplication operator $(A\psi)(x)=f(x)\psi(x)$ with $f\in L^\infty(\TT)$, and letting $\omega_{\mathrm{eff}}$ be any normal state (e.g.\ a vector state).
Then $a_t=\omega_{\mathrm{eff}}(U^t A U^{-t})$ is a bounded scan-readout sequence, and the Abel generating function $G(r)$ is holomorphic on $|r|<1$.
In any HTF candidate, this holomorphy verification is therefore straightforward once $(U,\omega_{\mathrm{eff}},A)$ are fixed; the nontrivial content is the existence of a trace identity whose spectral mode decomposition contains the zeta data.
\end{remark}

\begin{definition}[Abel-regularized scan trace]
\label{def:abel_scan_trace}
In the setting of Lemma~\ref{lem:bounded_trace_holomorphy}, we call
\[
G(r)=\sum_{t\ge 0}\omega_{\mathrm{eff}}\!\left(U^t A U^{-t}\right) r^t
\]
the \emph{Abel-regularized scan trace} associated to $(U,\omega_{\mathrm{eff}},A)$.
It is holomorphic on $|r|<1$.
\end{definition}

\subsection{HTF as a bridge package (structured assumptions)}
\label{sec:htf_assumption}

The Riemann Hypothesis in this framework is treated as a \emph{protocol consequence} of a trace identity that embeds arithmetic spectral data into scan traces.
We isolate the needed input as a single assumption.

\begin{assumption}[HTF: holographic trace formula bridge (package form)]
\label{ass:HTF}
There exists a choice of $(U,\omega_{\mathrm{eff}},A)$ and an Abel-regularized scan trace $G$ in the sense of Definition~\ref{def:abel_scan_trace}, together with a spectral term $\mathcal{S}$ and a regular term $\mathcal{A}_\infty$ on the unit disk, such that:
\begin{enumerate}
\item \textbf{Meromorphic identity.}
The following identity holds on $|r|<1$ as an equality of meromorphic functions:
\begin{equation}
\label{eq:htf_identity}
G(r)=\mathcal{S}(r)+\mathcal{A}_{\infty}(r),
\qquad |r|<1,
\end{equation}
\item \textbf{Analytic types.}
$G(r)$ is holomorphic on $|r|<1$ by Lemma~\ref{lem:bounded_trace_holomorphy}.
The regular term $\mathcal{A}_{\infty}(r)$ is holomorphic on $|r|<1$ and collects archimedean/renormalization contributions compatible with the Abel-first/finite-part convention.
\item \textbf{Spectral mode structure.}
The spectral term $\mathcal{S}(r)$ is meromorphic on $|r|<1$ and admits a decomposition
\begin{equation}
\label{eq:zero_mode_factor}
\mathcal{S}(r)=\sum_{\rho} c_\rho\,\frac{1}{1-r\,\e^{(\rho-\frac12)}}\;+\;\mathcal{H}(r),
\end{equation}
where the sum ranges over nontrivial zeros $\rho$ of $\zeta$, the coefficients $c_\rho$ are complex numbers, and $\mathcal{H}(r)$ is holomorphic on $|r|<1$.
The sum is understood with a fixed summability/renormalization prescription (for example, symmetric pairing of zeros and an Abel-first limiting procedure) chosen so that the right-hand side defines a meromorphic function on $|r|<1$.
\item \textbf{Non-cancellation of interior poles.}
Whenever $\RePart(\rho)>\tfrac12$ (equivalently $|r_\rho|<1$), the point
\[
r_\rho=\e^{-(\rho-\frac12)}
\]
is an actual pole of $\mathcal{S}(r)$, i.e.\ the principal part of $\mathcal{S}$ at $r=r_\rho$ is nonzero.
\end{enumerate}
\end{assumption}

Assumption~\ref{ass:HTF} is a protocol-level abstraction of the classical ``explicit formula as trace'' paradigm \cite{Connes1999,IwaniecKowalski2004,Titchmarsh1986}, rewritten in Abel $r$-coordinates with the $\Omega$ canonical path; see also the companion protocol trace manuscript \cite{Ma2025RGS}.
The present paper does not attempt to construct such an identity; rather, it isolates the analytic hypotheses and proves the rigidity consequence \emph{if} HTF holds.

\begin{remark}[HTF as a concrete verification checklist]
\label{rem:htf_checklist}
Assumption~\ref{ass:HTF} is formulated so that its components can be checked modularly in any proposed operator/dynamical model:
(i) the geometric-side boundedness and unit-disk holomorphy of $G$ are ensured by the scan-readout interface (Lemma~\ref{lem:bounded_trace_holomorphy});
(ii) the regular term $\mathcal{A}_\infty$ must be holomorphic and compatible with the Abel-first/finite-part discipline (Convention~\ref{ax:R1} and Appendix~\ref{app:abel_finite_part});
(iii) the spectral term must admit a meromorphic mode decomposition with resolvent factors that produce interior poles for off-critical growth, and those poles must not be canceled.
Appendix~\ref{app:htf_function_field} provides a fully explicit model where this checklist is satisfied unconditionally (with the Frobenius spectrum in place of zeta zeros).
\end{remark}

\begin{remark}[A fully explicit HTF instance in the function-field setting]
\label{rem:htf_function_field}
In the setting of zeta functions of curves over finite fields, trace identities of the form \eqref{eq:htf_identity} are explicit and unconditional, with spectral modes given by Frobenius eigenvalues.
Appendix~\ref{app:htf_function_field} records a concrete model showing how the pole-barrier mechanism becomes a transparent spectral-radius statement.
\end{remark}

\subsection{Relation to explicit formulas and operator-trace frameworks}
\label{sec:htf_relation}

\paragraph{Abel coordinate \texorpdfstring{$r$}{r} and Laplace coordinate \texorpdfstring{$s$}{s}.}
Writing $r=\e^{-s}$ with $\RePart(s)>0$ converts Abel weights into a Laplace-like coordinate.
The zero-mode factor in \eqref{eq:zero_mode_factor} becomes
\[
\frac{1}{1-r\,\e^{(\rho-\frac12)}}
\;=\;
\frac{1}{1-\e^{-(s-(\rho-\frac12))}}.
\]
Thus, an off-critical zero with $\RePart(\rho)>\tfrac12$ produces a pole at $r_\rho=\e^{-(\rho-\frac12)}$ strictly inside the unit disk (Lemma~\ref{lem:abel_pole_barrier}).
This is the analytic content of the ``pole barrier'' mechanism: in Abel/Laplace coordinates, exponential-growth modes correspond to interior resolvent singularities.

\paragraph{Test-function viewpoint (explicit-formula dictionary).}
The Abel trace $G(r)=\sum_{t\ge 0} a_t r^t$ is the Laplace transform of the discrete-time sequence $a_t$ under $r=\e^{-s}$:
\[
G(\e^{-s})=\sum_{t\ge 0} a_t\,\e^{-st}.
\]
In this language, the factor $\bigl(1-\e^{-(s-(\rho-\frac12))}\bigr)^{-1}$ is exactly the Laplace image of the geometric mode $t\mapsto \e^{(\rho-\frac12)t}$.
Classical explicit formulas can be viewed as trace identities evaluated against a family of test functions; HTF specializes to a concrete ``geometric-series'' test family adapted to the Abel-first discipline.

\paragraph{Connection to classical explicit formulas and Connes' trace framework.}
In analytic number theory, the Weil explicit formula relates test-function transforms of primes to test-function transforms of the nontrivial zeros of $\zeta$; see, e.g., \cite{Titchmarsh1986,IwaniecKowalski2004}.
Connes' semilocal trace formula provides an operator-theoretic realization of this paradigm in noncommutative geometry, in which the zero spectrum appears through trace identities \cite{Connes1999}.
HTF should be understood as the assertion that a specific Abel-regularized scan trace $G(r)$ admits such a trace realization in a form compatible with the $\Omega$ Abel-first discipline.

\paragraph{Relation to prolate-wave operator constructions.}
Recent work constructs explicit operators whose spectral data encode zeta zeros in novel ways, including prolate wave operators \cite{ConnesConsaniMoscovici2024}.
These constructions exemplify the operator-theoretic Hilbert--P\'olya route that HTF presupposes: one seeks a concrete spectral object and a trace identity whose mode decomposition contains the zeros \cite{BerryKeating1999}.
From the present closed-layer perspective, HTF packages the existence of such a trace bridge into a single auditable assumption, after which the rigidity consequence follows from unit-disk holomorphy.

\subsection{Interior poles from off-critical zeros}
\label{sec:bad_zero_pole}

\begin{theorem}[Riemann critical-line rigidity (conditional)]
\label{thm:rh_conditional}
Assume Axioms~\ref{ax:O5}--\ref{ax:O6}, Convention~\ref{ax:R1}, and the HTF bridge Assumption~\ref{ass:HTF}.
If the geometric-side trace $G(r)$ is holomorphic on the unit disk $|r|<1$, then every nontrivial zero $\rho$ of $\zeta$ satisfies
\[
\RePart(\rho)=\frac12.
\]
\end{theorem}

\begin{proof}
Suppose, for contradiction, that there exists a nontrivial zero $\rho$ with $\RePart(\rho)>\tfrac12$.
By Lemma~\ref{lem:abel_pole_barrier}, the corresponding mode factor
$\bigl(1-r\,\e^{(\rho-\frac12)}\bigr)^{-1}$ has a pole at $r_\rho=\e^{-(\rho-\frac12)}$ with $|r_\rho|<1$.
Under Assumption~\ref{ass:HTF}, the point $r=r_\rho$ is an actual pole of $\mathcal{S}(r)$.
Since $\mathcal{A}_{\infty}(r)$ is holomorphic on $|r|<1$, the right-hand side of \eqref{eq:htf_identity} has an interior pole at $r=r_\rho$.

On the other hand, by the boundedness/holomorphy of Abel traces on $|r|<1$, the left-hand side $G(r)$ is holomorphic on the entire unit disk and has no interior poles.
This contradicts \eqref{eq:htf_identity} as an equality of meromorphic functions on $|r|<1$.
Hence no zero can satisfy $\RePart(\rho)>\tfrac12$.

Finally, the functional equation symmetry $\rho\mapsto 1-\rho$ implies that if a zero with $\RePart(\rho)<\tfrac12$ existed, then $1-\rho$ would be a zero with real part $>\tfrac12$, which has been excluded.
Therefore all nontrivial zeros lie on the critical line.
\end{proof}

\begin{remark}[Status of the result]
Theorem~\ref{thm:rh_conditional} is a closed-layer \emph{protocol consequence} of HTF and unit-disk holomorphy.
The mathematical difficulty is isolated into a single task: construct an HTF bridge compatible with the Abel-first/finite-part convention and with bounded scan readout.
\end{remark}
