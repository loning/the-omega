\paragraph{Cross-interface terminal conclusions: Diagonal zero-temperature selection, \texorpdfstring{$(m,q)$}{(m,q)} dual-scale leading-term separation, and the ``typical--extremal'' freeze discrimination chain}
This paragraph, within the framework of the escort family \(w_{m,q}(x)\propto d_m(x)^q\), records a set dedicated to ``diagonal scaling \(q=\tau m\)'' extremal selection laws and verifiable scalars.
In the general case, Corollary \ref{cor:pom-escort-zero-temperature-superexp-concentration} has already given
\(e^{-\Theta(m^2)}\) super-exponential contraction of the exponential gap set \(A_m(\varepsilon)\) for fixed \(\varepsilon>0\);
here we further promote this contraction, under the assumption of \emph{exponentially separated top spectrum}, to concentration on the \emph{maximal fiber set},
and derive the \(m^2+m\) two-tier leading-term separation of \(\log S_q(m)\), the phase diagram of oracle success rate, as well as the four-quantity chain rigidity \((\alpha_\ast,g_\ast,h_1)\).

\begin{conclusion}[Diagonal zero-temperature selection principle: escorts with \texorpdfstring{$q=\tau m$}{q=tau m} concentrate on the maximal fiber set with \(e^{-\Theta(m^2)}\)]\label{con:xi-terminal-diagonal-zero-temp-maxset-concentration}
Let \(d_m:X_m\to\NN\) be the fiber multiplicity, and denote
\[
M_m:=\max_{x\in X_m}d_m(x),\qquad
A_m^{\max}:=\{x\in X_m:\ d_m(x)=M_m\},\qquad
K_m:=|A_m^{\max}|.
\]
Let
\[
M_m^{(2)}:=\max\{d_m(x):x\in X_m,\ d_m(x)<M_m\}
\]
(if none exists, set \(M_m^{(2)}:=1\)), and assume there exists a strict exponential gap
\[
\delta_\ast:=\liminf_{m\to\infty}\frac{1}{m}\log\frac{M_m}{M_m^{(2)}}\ >\ 0.
\]
For \(q\ge 0\) define the escort
\[
w_{m,q}(x):=\frac{d_m(x)^q}{S_q(m)},\qquad S_q(m):=\sum_{x\in X_m}d_m(x)^q.
\]
Then for any fixed \(\tau>0\), taking integer \(q_m:=\lfloor \tau m\rfloor\), we have
\[
w_{m,q_m}(A_m^{\max})\ \ge\ 1-\exp\!\bigl(-\tau(\delta_\ast-o(1))\,m^2\bigr),
\]
and the total variation distance to the uniform distribution \(U(A_m^{\max})\) on \(A_m^{\max}\) satisfies
\[
D_{\mathrm{TV}}\!\bigl(w_{m,q_m},U(A_m^{\max})\bigr)
\le \exp\!\bigl(-\tau(\delta_\ast-o(1))\,m^2\bigr).
\]
\end{conclusion}
\begin{proof}
For \(x\notin A_m^{\max}\), we have \(d_m(x)\le M_m^{(2)}\), so
\[
\sum_{x\notin A_m^{\max}}d_m(x)^{q_m}\le |X_m|\,(M_m^{(2)})^{q_m}.
\]
Also \(|X_m|=F_{m+2}=\exp(O(m))\).
By the gap assumption we can take \(\varepsilon_m=o(1)\) such that \(\log(M_m/M_m^{(2)})\ge (\delta_\ast-\varepsilon_m)m\)
(whence \(M_m^{(2)}\le M_m\exp\bigl(-(\delta_\ast-\varepsilon_m)m\bigr)\)) and this holds for large \(m\), so
\[
|X_m|\,(M_m^{(2)})^{q_m}
\le
\exp(O(m))\cdot M_m^{q_m}\exp\!\bigl(-(\delta_\ast-\varepsilon_m)q_m m\bigr)
=M_m^{q_m}\exp\!\bigl(-\tau(\delta_\ast-o(1))m^2\bigr).
\]
On the other hand,
\(
\sum_{x\in A_m^{\max}}d_m(x)^{q_m}=K_m M_m^{q_m}\ge M_m^{q_m}
\), so
\[
1-w_{m,q_m}(A_m^{\max})
=\frac{\sum_{x\notin A_m^{\max}}d_m(x)^{q_m}}{S_{q_m}(m)}
\le
\frac{|X_m|\,(M_m^{(2)})^{q_m}}{M_m^{q_m}}
\le \exp\!\bigl(-\tau(\delta_\ast-o(1))m^2\bigr).
\]
On \(A_m^{\max}\) we have \(d_m\equiv M_m\), so the conditional distribution of \(w_{m,q_m}\) on that set is strictly uniform.
Hence
\[
D_{\mathrm{TV}}\!\bigl(w_{m,q_m},U(A_m^{\max})\bigr)
=1-w_{m,q_m}(A_m^{\max}),
\]
and the second formula follows.
\end{proof}

\begin{conclusion}[Dual-scale diagonal moment expansion: \texorpdfstring{$\log S_q(m)$}{log S_q(m)} admits a completely separable \(m^2+m\) two-tier leading term when \texorpdfstring{$q=\tau m+s$}{q=tau m+s}]\label{con:xi-terminal-diagonal-logSq-m2-m-separation}
Under the exponential gap assumption of Conclusion \ref{con:xi-terminal-diagonal-zero-temp-maxset-concentration}, further assume the limits
\[
\alpha_\ast:=\lim_{m\to\infty}\frac{1}{m}\log M_m,\qquad
g_\ast:=\lim_{m\to\infty}\frac{1}{m}\log K_m
\]
exist (can also be interpreted using \(\limsup\)).
Then for any fixed \(\tau>0\) and any fixed \(s\in\RR\), taking the real sequence \(q_m:=\tau m+s\), we have
\[
\log S_{q_m}(m)=\tau\alpha_\ast\,m^2+\bigl(s\alpha_\ast+g_\ast\bigr)m+o(m).
\]
\end{conclusion}
\begin{proof}
By the estimate of Conclusion \ref{con:xi-terminal-diagonal-zero-temp-maxset-concentration} (which also holds for real \(q_m\)) we obtain
\[
S_{q_m}(m)=K_m M_m^{q_m}\Bigl(1+O\bigl(\exp\!\bigl(-\tau(\delta_\ast-o(1))m^2\bigr)\bigr)\Bigr).
\]
Taking logarithm and using \(\log M_m=\alpha_\ast m+o(m)\), \(\log K_m=g_\ast m+o(m)\) yields
\[
\log S_{q_m}(m)=\log K_m+q_m\log M_m+o(m)
=\tau\alpha_\ast m^2+(s\alpha_\ast+g_\ast)m+o(m).
\]
\end{proof}

\begin{remark}[On the sub-leading term for integer rounding \texorpdfstring{$q_m=\lfloor\tau m+s\rfloor$}{q=floor(tau m+s)}]
If we insist on taking integer \(q_m=\lfloor\tau m+s\rfloor\), then \(q_m=\tau m+s_m\) with \(s_m\in[s-1,s]\).
Therefore the separation above still preserves the \(m^2\) leading term unchanged, while the \(m\)-level sub-leading term \(s\alpha_\ast m\) needs to be replaced by \(s_m\alpha_\ast m\);
this replacement corresponds to a bounded oscillation induced by rounding and does not affect the conclusion chain regarding the \(m^2\)-level selection law in this paragraph.
\end{remark}

\begin{conclusion}[Oracle phase diagram of diagonal escort: success rate for budget \texorpdfstring{$B=\beta m$}{B=beta m} is determined solely by \texorpdfstring{$\alpha_\ast$}{alpha*} at the \(m\) scale]\label{con:xi-terminal-diagonal-escort-oracle-phase}
Under the exponential gap assumption of Conclusion \ref{con:xi-terminal-diagonal-zero-temp-maxset-concentration}, define the inversion success rate for budget \(B\) under escort \(w_{m,q}\):
\[
\mathrm{Succ}_{m,q}(B)
:=\sum_{x\in X_m}w_{m,q}(x)\,\min\!\Bigl(1,\frac{2^{B}}{d_m(x)}\Bigr).
\]
Take \(q_m:=\lfloor\tau m\rfloor\) (with \(\tau>0\) fixed) and \(B_m:=\lfloor\beta m\rfloor\).
If \(\alpha_\ast=\lim_{m\to\infty}\frac{1}{m}\log M_m\) exists, then the limit
\[
\lim_{m\to\infty}\frac{1}{m}\log \mathrm{Succ}_{m,q_m}(B_m)
=\min\bigl(0,\ \beta\log 2-\alpha_\ast\bigr)
\]
holds, and when \(\beta\log 2>\alpha_\ast\),
\[
1-\mathrm{Succ}_{m,q_m}(B_m)\ \le\ \exp\!\bigl(-\tau(\delta_\ast-o(1))\,m^2\bigr).
\]
\end{conclusion}
\begin{proof}
By Conclusion \ref{con:xi-terminal-diagonal-zero-temp-maxset-concentration}, \(w_{m,q_m}\) concentrates on \(A_m^{\max}\) with \(e^{-\Theta(m^2)}\).
On \(A_m^{\max}\), \(d_m(x)=M_m\), so
\[
\mathrm{Succ}_{m,q_m}(B_m)
=\min\!\Bigl(1,\frac{2^{B_m}}{M_m}\Bigr)+O\bigl(\exp\!\bigl(-\tau(\delta_\ast-o(1))m^2\bigr)\bigr).
\]
Taking \(\frac1m\log\) and using \(\log M_m=\alpha_\ast m+o(m)\) yields the exponential limit
\(
\min(0,\beta\log 2-\alpha_\ast)
\).
When \(\beta\log2>\alpha_\ast\), \(\frac{2^{B_m}}{M_m}\to\infty\), failure only comes from the probability of leaving \(A_m^{\max}\),
so the second formula follows from the mass bound of Conclusion \ref{con:xi-terminal-diagonal-zero-temp-maxset-concentration}.
\end{proof}

\begin{conclusion}[Diagonal entropy plateau: when \texorpdfstring{$q=\tau m+O(1)$}{q=tau m+O(1)}, both Shannon and Rényi entropy rates of the escort collapse to \texorpdfstring{$g_\ast$}{g*}]\label{con:xi-terminal-diagonal-escort-entropy-plateau}
Under the exponential gap assumption of Conclusion \ref{con:xi-terminal-diagonal-zero-temp-maxset-concentration}, take \(q_m=\tau m+s\) (with \(\tau>0\) fixed, \(s\in\RR\) fixed).
If \(g_\ast=\lim_{m\to\infty}\frac{1}{m}\log K_m\) exists, then for any Rényi order \(r\in(0,\infty]\), the escort entropy satisfies
\[
\frac{1}{m}H_r\!\bigl(w_{m,q_m}\bigr)\longrightarrow g_\ast,
\]
in particular
\(
\frac{1}{m}H\!\bigl(w_{m,q_m}\bigr)\to g_\ast
\)
(where \(H=H_1\) is Shannon entropy, with natural logarithm).
\end{conclusion}
\begin{proof}
Denote \(p_m:=w_{m,q_m}(A_m^{\max})\).
By Conclusion \ref{con:xi-terminal-diagonal-zero-temp-maxset-concentration}, we have \(1-p_m=\exp\!\bigl(-\tau(\delta_\ast-o(1))m^2\bigr)\).
Moreover, \(w_{m,q_m}\) is strictly uniform on \(A_m^{\max}\): for \(x\in A_m^{\max}\), \(w_{m,q_m}(x)=p_m/K_m\).
\par
For any \(r\in(0,\infty)\), we have
\[
\sum_{x\in X_m}w_{m,q_m}(x)^r
=K_m\Bigl(\frac{p_m}{K_m}\Bigr)^r+\sum_{x\notin A_m^{\max}}w_{m,q_m}(x)^r
=p_m^rK_m^{1-r}+O\bigl((1-p_m)^r\bigr),
\]
whence
\(
H_r(w_{m,q_m})=\log K_m+o(m)
\).
When \(r=\infty\),
\(
H_\infty(w_{m,q_m})=-\log\max_x w_{m,q_m}(x)=\log K_m-\log p_m=\log K_m+o(m)
\);
when \(r=1\), by
\(
H(w_{m,q_m})=\log K_m+O\bigl((1-p_m)\log K_m\bigr)+o(m)
\)
we also obtain the same leading term.
Finally using \(\log K_m=g_\ast m+o(m)\) yields the conclusion.
\end{proof}

\begin{conclusion}[\texorpdfstring{$\beta_{\mathrm{crit}}=\alpha_\ast/\log 2$}{beta_crit=alpha*/log2} discriminant: if and only if the pushforward \texorpdfstring{$w_m$}{w_m} freezes to the maximal fiber set at exponential scale]\label{con:xi-terminal-typical-extreme-coaxial-freeze-criterion}
Let the pushforward distribution be \(w_m(x)=d_m(x)/2^m\), and define
\[
a_1:=\lim_{m\to\infty}\frac{1}{m}\mathbb E_{w_m}[\log d_m(X)]\quad(\text{if limit exists}),
\qquad
\beta_{\mathrm{crit}}:=\frac{a_1}{\log2}.
\]
Then we always have \(\beta_{\mathrm{crit}}\le \alpha_\ast/\log2\) (where \(\alpha_\ast=\lim\frac{1}{m}\log M_m\)).
Moreover, under the exponential gap assumption of Conclusion \ref{con:xi-terminal-diagonal-zero-temp-maxset-concentration},
the following two propositions are equivalent:
\begin{enumerate}
  \item \(a_1=\alpha_\ast\) (i.e., \(\beta_{\mathrm{crit}}=\alpha_\ast/\log2\));
  \item \(\lim_{m\to\infty} w_m(A_m^{\max})=1\).
\end{enumerate}
\end{conclusion}
\begin{proof}
From \(\log d_m\le \log M_m\) we get \(a_1\le \alpha_\ast\), so \(\beta_{\mathrm{crit}}\le \alpha_\ast/\log2\).
\par
Assume the exponential gap holds.
If \(a_1=\alpha_\ast\), then
\[
0\le \frac{1}{m}\mathbb E_{w_m}\!\Bigl[\log\frac{M_m}{d_m(X)}\Bigr]
=\frac{1}{m}\log M_m-\frac{1}{m}\mathbb E_{w_m}[\log d_m(X)]\ \to\ 0.
\]
While on the complement \(X_m\setminus A_m^{\max}\) we have \(\log(M_m/d_m)\ge \log(M_m/M_m^{(2)})\ge (\delta_\ast-o(1))m\),
so
\[
\frac{1}{m}\mathbb E_{w_m}\!\Bigl[\log\frac{M_m}{d_m(X)}\Bigr]
\ge (\delta_\ast-o(1))\cdot\bigl(1-w_m(A_m^{\max})\bigr),
\]
whence \(w_m(A_m^{\max})\to 1\).
Conversely: if \(w_m(A_m^{\max})\to 1\), then \(\log d_m(X)=\log M_m\) with \(w_m\)-probability tending to \(1\),
so \(\frac1m\mathbb E_{w_m}[\log d_m]\to \frac1m\log M_m\to\alpha_\ast\), i.e., \(a_1=\alpha_\ast\).
\end{proof}

\begin{conclusion}[Extremal--degeneracy inequality: rigidity of \texorpdfstring{$\alpha_\ast+g_\ast\le \log2$}{alpha*+g*<=log2} and exponential load rate of the maximal set]\label{con:xi-terminal-extreme-degeneracy-inequality}
Denote \(K_m:=|A_m^{\max}|\) and \(M_m:=\max_x d_m(x)\).
Since \(\sum_{x\in X_m}d_m(x)=2^m\), we always have
\[
K_m M_m\ \le\ 2^m.
\]
If \(\alpha_\ast=\lim_{m\to\infty}\frac{1}{m}\log M_m\) and \(g_\ast=\lim_{m\to\infty}\frac{1}{m}\log K_m\) exist, then they must satisfy
\[
\alpha_\ast+g_\ast\le\log2.
\]
Moreover, the pushforward mass satisfies
\[
w_m(A_m^{\max})=\frac{K_mM_m}{2^m},
\qquad
\frac{1}{m}\log w_m(A_m^{\max})\longrightarrow \alpha_\ast+g_\ast-\log2.
\]
Therefore \(\alpha_\ast+g_\ast=\log2\) if and only if \(w_m(A_m^{\max})\) does not decay at a positive exponential rate (i.e., \(w_m(A_m^{\max})=\exp(o(m))\)).
\end{conclusion}
\begin{proof}
The inequality \(K_mM_m\le\sum_x d_m(x)=2^m\) directly yields \(\alpha_\ast+g_\ast\le\log2\).
The identity for \(w_m(A_m^{\max})\) follows from \(w_m(x)=d_m(x)/2^m\) and \(d_m\equiv M_m\) on \(A_m^{\max}\).
Taking \(\frac1m\log\) on both sides and taking limits yields the exponential load rate formula and equality criterion.
\end{proof}

\begin{conclusion}[Four-quantity chain rigidity: \texorpdfstring{$g_\ast\le \log2-\alpha_\ast\le h_1\le \log\varphi$}{g*<=log2-alpha*<=h1<=logphi} and its freeze phase meaning]\label{con:xi-terminal-four-quantity-chain-rigidity}
Let
\[
h_1:=\lim_{m\to\infty}\frac{1}{m}H(w_m)
\quad(\text{if limit exists}),\qquad
\lim_{m\to\infty}\frac{1}{m}\log|X_m|=\log\varphi.
\]
Then when the above limits exist,
\[
g_\ast\le \log2-\alpha_\ast\le h_1\le \log\varphi.
\]
Where:
\begin{itemize}
  \item \(g_\ast\le \log2-\alpha_\ast\) is equivalent to \(\alpha_\ast+g_\ast\le\log2\) in Conclusion \ref{con:xi-terminal-extreme-degeneracy-inequality};
  \item \(\log2-\alpha_\ast\le h_1\) is equivalent to \(a_1\le \alpha_\ast\), since
  \(
  H(w_m)=m\log2-\mathbb E_{w_m}[\log d_m(X)]
  \);
  \item \(h_1\le \log\varphi\) is the exponentialized form of the entropy upper bound \(H(w_m)\le \log|X_m|\).
\end{itemize}
Moreover, equalities at the two ends respectively characterize two types of extremal freeze calibers:
\begin{itemize}
  \item \(g_\ast=\log2-\alpha_\ast\) if and only if \(w_m(A_m^{\max})\) does not decay at a positive exponential rate (Conclusion \ref{con:xi-terminal-extreme-degeneracy-inequality});
  \item Under the exponential gap assumption of Conclusion \ref{con:xi-terminal-diagonal-zero-temp-maxset-concentration},
  \(\log2-\alpha_\ast=h_1\) is equivalent to \(a_1=\alpha_\ast\), and further equivalent to \(w_m(A_m^{\max})\to 1\) (Conclusion \ref{con:xi-terminal-typical-extreme-coaxial-freeze-criterion}).
\end{itemize}
\end{conclusion}
\begin{proof}
The first inequality follows from \(K_mM_m\le 2^m\) by taking logarithm, dividing by \(m\), and taking limits.
The second follows from the identity
\(
H(w_m)=-\sum_x w_m(x)\log w_m(x)
=m\log2-\mathbb E_{w_m}[\log d_m(X)]
\)
which directly yields: \(\log2-\alpha_\ast\le h_1\iff a_1\le\alpha_\ast\).
The third is the Shannon entropy upper bound \(H(w_m)\le \log|X_m|\).
The equality characterizations are given by the equality conditions of the corresponding inequalities and the preceding conclusion chains.
\end{proof}
