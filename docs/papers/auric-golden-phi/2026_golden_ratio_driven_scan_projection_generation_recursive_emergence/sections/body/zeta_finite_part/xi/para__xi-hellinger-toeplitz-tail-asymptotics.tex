\begin{proposition}[Large separation cluster expansion:\texorpdfstring{$S_{1/2}$}{S_{1/2}}'s main term is the two-body exponential energy]\label{prop:xi-hellinger-cluster-expansion-large-separation}
For the general depth spectrum \(\boldsymbol{\sigma}=(\sigma_1,\dots,\sigma_\kappa)\), record the minimum separation
\(
\Delta_{\min}:=\min_{j\neq k}|\sigma_j-\sigma_k|
\)。
There is a consistent expansion when \(\Delta_{\min}\to\infty\)
\[
\boxed{\;
S_{1/2}(\boldsymbol{\sigma})
=\frac12\sum_{j\neq k}A(\sigma_j-\sigma_k)^{2}
+O\!\Bigl(\kappa^{3}(1+\Delta_{\min})^{3}e^{-3\Delta_{\min}/2}\Bigr).\;}
\]
and
\(
A(\Delta)\sim |\Delta|e^{-|\Delta|/2}
\), so that the subject satisfies
\[
\frac12\sum_{j\neq k}A(\sigma_j-\sigma_k)^2
\sim \frac12\sum_{j\neq k}|\sigma_j-\sigma_k|^2e^{-|\sigma_j-\sigma_k|}
\qquad(\Delta_{\min}\to\infty).
\]
\end{proposition}
\begin{proof}
Write \(G(\boldsymbol{\sigma})=I+E\), where \(E_{jj}=0\), \(E_{jk}=A(\sigma_j-\sigma_k)\) (\(j\neq k\)).
The upper bound is obtained by \(A(\Delta)=|\Delta|/(2\sinh(|\Delta|/2))\)
\(
0<A(\Delta)\le C(1+|\Delta|)e^{-|\Delta|/2}
\)
(The hyperbolic function asymptotically gives \(|\Delta|\to\infty\)).
Therefore \(\|E\|_{\max}\le C(1+\Delta_{\min})e^{-\Delta_{\min}/2}\), and when \(\Delta_{\min}\) is large enough \(\|E\|\) can be small.
Expanded by logarithmic determinant
\(
-\log\det(I+E)=\sum_{m\ge 1}\frac{(-1)^m}{m}\tr(E^m)
\)
With \(\tr(E)=0\), the leading term is \(\tr(E^2)/2=\frac12\sum_{j\neq k}E_{jk}^2\).
The remainder is estimated by \(|\tr(E^m)|\le \kappa^m\|E\|_{\max}^m\), resulting in the error bound shown. Certification completed.
\end{proof}

\begin{corollary}[Orthogonalization threshold and volume saturation phase: \texorpdfstring{$\Delta_{\min}\gtrsim 2\log\kappa$}{Delta_min ~ 2 log kappa} forcing \texorpdfstring{$\det G\to 1$}{det G -> 1}]\label{cor:xi-hellinger-quasi-orthogonal-phase}
Follow the notation of the proposition \ref{prop:xi-hellinger-cluster-expansion-large-separation}.
Then there exists a constant \(C>0\) such that
\[
\max_{1\le j\le \kappa}\sum_{k\neq j}\bigl|G(\boldsymbol{\sigma})_{jk}\bigr|
\le C\,\kappa(1+\Delta_{\min})e^{-\Delta_{\min}/2}.
\]
Therefore, if the configuration changes with \(\kappa\) and satisfies \(\kappa(1+\Delta_{\min})e^{-\Delta_{\min}/2}\to 0\), then
\[
\lambda_{\min}(G(\boldsymbol{\sigma}))\to 1,\qquad
\lambda_{\max}(G(\boldsymbol{\sigma}))\to 1,\qquad
\det G(\boldsymbol{\sigma})\to 1,\qquad
S_{1/2}(\boldsymbol{\sigma})\to 0.
\]
In particular, sufficient conditions
\[
\Delta_{\min}\ge 2\log\kappa+\omega(1)
\]
Forced into the "quasi-orthogonal phase" (volume saturated phase): Once the growth in the number of atoms is suppressed by the exponential tail \(e^{-\Delta/2}\), the volume entropy potential collapses to the zero entropy limit.
\end{corollary}
\begin{proof}
By \(G=I+E\) and \(E_{jk}=A(\sigma_j-\sigma_k)\) (\(j\neq k\)), and the upper bound in the proof of the proposition \ref{prop:xi-hellinger-cluster-expansion-large-separation}
\(
A(\Delta)\le C(1+|\Delta|)e^{-|\Delta|/2}
\)，
For each \(j\) there is
\[
\sum_{k\neq j}|G_{jk}|=\sum_{k\neq j}A(\sigma_j-\sigma_k)\le (\kappa-1)\,C(1+\Delta_{\min})e^{-\Delta_{\min}/2},
\]
Get the first formula.
When the right end tends to \(0\), the matrix norm satisfies \(\|E\|_2\le \|E\|_\infty\to 0\), so the spectrum \(\mathrm{Spec}(G)\subset [1-\|E\|_2,1+\|E\|_2]\).
From this \(\lambda_{\min},\lambda_{\max}\to 1\), and from \(\det G=\prod_r\lambda_r\) follows \(\det G\to 1\), thus \(S_{1/2}=-\log\det G\to 0\).
The sufficient condition is established by \(e^{-\Delta_{\min}/2}\le \kappa^{-1}e^{-\omega(1)/2}\). Certification completed.
\end{proof}

\begin{remark}[Cross-domain resolution constant for equidistant symbols: the periodization tail term is controlled by \texorpdfstring{$\exp(-2\pi^2/\Delta)$}{exp(-2pi^2/Delta)}]\label{rem:xi-hellinger-toeplitz-periodization-constant-2pi2}
Let \(\sigma_\Delta^{(0)}(\theta):=\frac{\pi^2}{\Delta}\cosh^{-2}(\pi\theta/\Delta)\) be the \(k=0\) subject of the sum of Poisson periods in the proposition \ref{prop:xi-hellinger-toeplitz-symbol-poisson}.
Then for any \(\theta\in[-\pi,\pi]\) we have
\[
0\le \sigma_\Delta(\theta)-\sigma_\Delta^{(0)}(\theta)
:=\frac{1}{\Delta}\sum_{k\neq 0}\frac{\pi^{2}}{\cosh^{2}\!\Bigl(\pi\frac{\theta+2\pi k}{\Delta}\Bigr)}
\le C\,\frac{\pi^{2}}{\Delta}\,e^{-2\pi^{2}/\Delta},
\]
where \(C>0\) has nothing to do with \(\Delta\).
Therefore, the minimum cross-domain constant of the periodization effect is \(2\pi^2\): it not only controls the dominance of the unit cell at small \(\Delta\), but also gives a unified exponential threshold for the entry of the tail periodic term.
\par
In particular, at \(\theta=0\), there is an exact identity
\[
\sigma_\Delta(0)=\sum_{n\in\ZZ}A(n\Delta)
=\frac{\pi^{2}}{\Delta}\sum_{k\in\ZZ}\mathrm{sech}^{2}\!\Bigl(\frac{2\pi^{2}k}{\Delta}\Bigr),
\]
Thus when \(\Delta\downarrow 0\)
\[
\sigma_\Delta(0)=\frac{\pi^{2}}{\Delta}\Bigl(1+O(e^{-4\pi^{2}/\Delta})\Bigr),
\qquad
\log\sigma_\Delta(0)=\log\frac{\pi^{2}}{\Delta}+O(e^{-4\pi^{2}/\Delta}),
\]
The transcellular correction is sealed by the dual nome \(q_\star=e^{-2\pi^{2}/\Delta}\) with \(O(q_\star^{2})\).
\end{remark}

\begin{remark}[\texorpdfstring{$\zeta(4)$}{zeta(4)}'s first invasion law: the fourth-order coefficient of the frequency domain external potential is forced to appear]\label{rem:xi-hellinger-logw-zeta4-entry}
Let \(w(\xi)=\pi^2/\cosh^2(\pi\xi)\), then there is Taylor expansion at \(\xi=0\)
\[
\log w(\xi)=\log\pi^2-\pi^2\xi^2+\frac{\pi^4}{6}\xi^4+O(\xi^6)
=\log\pi^2-\pi^2\xi^2+15\,\zeta(4)\,\xi^4+O(\xi^6).
\]
Therefore, for any system based on the quadratic external potential (Gaussian approximation), its first indelible systematic correction must appear at the fourth order and be accurately sealed by \(\zeta(4)\);
The fourth-order first intrusion on this spectral side is compatible with the fourth-order closing threshold noted in \ref{rem:xi-scale-q4-threshold-principle}.
\end{remark}

\begin{proposition}[Extremal and modular exponents of equally spaced depth Toeplitz symbols: \texorpdfstring{$\inf\sigma_\Delta$}{inf sigma} of \texorpdfstring{$e^{-2\pi^2/\Delta}$}{exp(-2pi^2/Delta)} scale]\label{prop:xi-hellinger-toeplitz-symbol-extrema-qstar}
Follow the notation of the proposition \ref{prop:xi-hellinger-toeplitz-symbol-poisson}.
Then \(\sigma_\Delta\) takes the global maximum at \(\theta=0\) and takes the global minimum at \(\theta=\pi\):
\[
\sigma_\Delta(0)=\sup_{\theta\in[-\pi,\pi]}\sigma_\Delta(\theta),
\qquad
\sigma_\Delta(\pi)=\inf_{\theta\in[-\pi,\pi]}\sigma_\Delta(\theta).
\]
And when \(\Delta\downarrow 0\) (let the dual nome \(q_\star:=e^{-2\pi^2/\Delta}\)), there is a consistent asymptotic
\[
\sigma_\Delta(0)=\frac{\pi^2}{\Delta}\Bigl(1+O(q_\star^{2})\Bigr),
\qquad
\sigma_\Delta(\pi)=\frac{8\pi^2}{\Delta}\,q_\star\Bigl(1+O(q_\star^{2})\Bigr).
\]
The "minimum spectral height" is therefore exactly capped by the modular dual scale \(2\pi^2/\Delta\) with the institutional exponent of \(q_\star\).
\end{proposition}
\begin{proof}[Proof sketch]
The maximum of \(\theta=0\) can be expressed by Fourier
\(
\sigma_\Delta(\theta)=1+2\sum_{n\ge 1}A(n\Delta)\cos(n\theta)
\)
Directly derived from \(A(n\Delta)>0\), \(\cos(n\theta)\le 1\).
The minimality of \(\theta=\pi\) can be expressed by Poisson
\(
\sigma_\Delta(\theta)=\frac{\pi^2}{\Delta}\sum_{k\in\ZZ}\mathrm{sech}^2\!\bigl(\pi(\theta+2\pi k)/\Delta\bigr)
\)
Combining the strict unimodality of \(\mathrm{sech}^2\) with the lattice symmetry gives: \(\theta=\pi\) is the farthest point to \(2\pi\ZZ\), so that the periodized unimodal sum is minimized at this point.
\par
For \(\theta=\pi\), the exact identity is obtained in Poisson form
\[
\sigma_\Delta(\pi)=\frac{\pi^2}{\Delta}\sum_{k\in\ZZ}\mathrm{sech}^2\!\Bigl(\frac{(2k+1)\pi^2}{\Delta}\Bigr).
\]
When \(\Delta\downarrow 0\), the main contribution comes from \(k=0,-1\) two terms; and use
\(
\mathrm{sech}^2(x)=4e^{-2x}(1+O(e^{-2x}))
\)
You can get
\(
\sigma_\Delta(\pi)=\frac{8\pi^2}{\Delta}e^{-2\pi^2/\Delta}(1+O(e^{-4\pi^2/\Delta}))
\)。
The asymptotics of \(\sigma_\Delta(0)\) are given by the note \ref{rem:xi-hellinger-toeplitz-periodization-constant-2pi2}. Certification completed.
\end{proof}

\begin{proposition}[Conditional exponential law for equidistant depths: the reversibility budget is linearly locked by \texorpdfstring{$2\pi^2/\Delta$}{2pi^2/Delta}]\label{prop:xi-hellinger-toeplitz-conditioning-exponential-law}
Let \(T_\Delta\) be the bilateral Toeplitz convolution operator
\[
(T_\Delta x)_j=\sum_{n\in\ZZ}A((j-n)\Delta)\,x_n,
\]
Its symbol is \(\sigma_\Delta\).
Then \(\mathrm{Spec}(T_\Delta)=\sigma_\Delta([-\pi,\pi])\), so the condition number is
\[
\mathrm{cond}(T_\Delta)=\frac{\sup_\theta\sigma_\Delta(\theta)}{\inf_\theta\sigma_\Delta(\theta)}
=\frac{\sigma_\Delta(0)}{\sigma_\Delta(\pi)}
\sim \frac18\,e^{2\pi^2/\Delta}\qquad(\Delta\downarrow 0).
\]
For the finite-dimensional Toeplitz main block \(G_\kappa(\Delta)=(A((j-k)\Delta))_{1\le j,k\le\kappa}\), its extreme eigenvalues converge to \(\sup\sigma_\Delta,\inf\sigma_\Delta\) under \(\kappa\to\infty\), so under large \(\kappa\) The following \(\mathrm{cond}(G_\kappa(\Delta))\) inherits the same exponential scale.
Therefore, any stable inversion requiring relative error \(\le\varepsilon\) must satisfy the lower bound on the number of digits of accuracy
\[
B\ \gtrsim\ \frac{2\pi^2}{\Delta\ln 2}\ +\ O\bigl(1+\log(1/\varepsilon)\bigr),
\]
The slope of the first term \(2\pi^2/\ln2\) is a non-rewriteable constant.
\end{proposition}
\begin{proof}[Proof sketch]
The spectral conclusion is the standard fact of the Toeplitz operator: \(T_\Delta\) is a multiplier \(\sigma_\Delta\) on the Fourier side, so the spectrum has its value range.
The condition number asymptotically is given by the \(\sigma_\Delta(0)\) and \(\sigma_\Delta(\pi)\) asymptotics of the proposition \ref{prop:xi-hellinger-toeplitz-symbol-extrema-qstar}.
The extreme spectral convergence of finite master blocks is a Grenander--Szeg\H{o} type conclusion; the lower bound on the number of digits of accuracy comes from the necessity of \(\log_2\mathrm{cond}\). Certification completed.
\end{proof}

\begin{theorem}[The "area law" free energy of the dense limit:\texorpdfstring{$f(\Delta)$}{The auditable modular exponential term of f(Delta)}]\label{thm:xi-hellinger-toeplitz-free-energy-area-law}
Let \(\Delta>0\) be fixed and define the Toeplitz free energy density
\[
f(\Delta):=\lim_{\kappa\to\infty}\frac{1}{\kappa}\log\det G_\kappa(\Delta)
=\frac{1}{2\pi}\int_{-\pi}^{\pi}\log\sigma_\Delta(\theta)\,\dd\theta
\]
(For the existence of limits, see theorem \ref{thm:xi-hellinger-toeplitz-szego-thermo-limit}).
Then there is an auditable modular exponential expansion when \(\Delta\downarrow0\)
\[
\boxed{\;
f(\Delta)= -\frac{\pi^{2}}{\Delta}\ +\ \log\!\Bigl(\frac{\pi^{2}}{\Delta}\Bigr)\ +\ 2\log 2\ -\ \frac{\Delta}{16}\ +\ O\!\bigl(e^{-2\pi^{2}/\Delta}\bigr).\;}
\]
Therefore, the dominant term of the volume entropy density \(-f(\Delta)\) is \(\pi^2/\Delta\) (area law term), accompanied by the boundary logarithmic correction and strict linear microcorrection \(\Delta/16\) of \(\log(\pi^2/\Delta)\).
\end{theorem}
\begin{proof}[Proof sketch]
From the \(\vartheta_4\) closed form of the proposition \ref{prop:xi-hellinger-toeplitz-symbol-poisson} and the modular dual scale of the note \ref{rem:xi-hellinger-toeplitz-modular-duality-scale},
The mean value of \(\log\sigma_\Delta\) can be transformed into a series of the dual nome \(q_\star=e^{-2\pi^2/\Delta}\) by Jacobi product and \(\tau\mapsto-1/\tau\) transformation.
Explicit expansion of this series in the composite scale of \(q_\star\) and \(\Delta\) yields the principal terms and error seals shown. Certification completed.
\end{proof}

\begin{proposition}[Exponential additivity of cluster decomposition: depth volume entropy is block additive under large gaps]\label{prop:xi-hellinger-gram-cluster-additivity-exp}
Divide the depth point set into two clusters \(S^{(1)}\sqcup S^{(2)}\), denoted as \(|S^{(\ell)}|=\kappa_\ell\), and set the minimum gap between clusters
\[
L:=\min\{|x-y|:\ x\in S^{(1)},\ y\in S^{(2)}\}.
\]
Corresponding to Gram block
\[
G=\begin{pmatrix}G_1 & C\\ C^{\mathsf T}& G_2\end{pmatrix},
\qquad
|C_{ij}|\le A(L)\le L\,e^{-L/2}.
\]
If \(G_1,G_2\succ0\), then there is a constant \(K\) (which only depends on \(\|G_1^{-1}\|_2,\|G_2^{-1}\|_2\) and \(\kappa_1,\kappa_2\)) so that
\[
\bigl|\log\det G-\log\det G_1-\log\det G_2\bigr|
\le K\,\|C\|_{\mathrm F}^{2}
\le K\,\kappa_1\kappa_2\,L^{2}e^{-L}.
\]
The depth volume entropy \(S_{1/2}=-\log\det G\) is therefore strictly exponentially additive over the cluster spacing \(L\): the interaction term has a natural scale of \(e^{-L}\).
\end{proposition}
\begin{proof}[Proof sketch]
By Schur’s complement formula,
\[
\det G=\det G_1\cdot \det(G_2-C^{\mathsf T}G_1^{-1}C)
=\det G_1\det G_2\cdot \det(I-X),
\]
Among them \(X:=G_2^{-1/2}C^{\mathsf T}G_1^{-1}C\,G_2^{-1/2}\succeq 0\).
When \(\|X\|_2<1\),
\(
|\log\det(I-X)|\le \mathrm{tr}(X)/(1-\|X\|_2)
\)。
and
\[
\mathrm{tr}(X)=\|G_1^{-1/2}C\,G_2^{-1/2}\|_{\mathrm F}^2
\le \|G_1^{-1}\|_2\,\|G_2^{-1}\|_2\,\|C\|_{\mathrm F}^2,
\]
At the same time \(\|X\|_2\le \|G_1^{-1}\|_2\|G_2^{-1}\|_2\|C\|_2^2\).
Merging these bounds into the constant \(K\) gives us the first equation.
The last formula is derived from \(\|C\|_{\mathrm F}^2\le \kappa_1\kappa_2\,\max_{i,j}|C_{ij}|^2\) and \(A(L)\le L e^{-L/2}\). Certification completed.
\end{proof}
