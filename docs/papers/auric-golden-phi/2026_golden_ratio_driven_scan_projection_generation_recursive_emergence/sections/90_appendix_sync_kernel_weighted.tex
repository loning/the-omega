\subsubsection{加权核的压力与大偏差（可审计接口）}
令 $\phi$ 为同步图边上的可观测量（例如输出位 $e$ 的指示函数，或"到达含 $\overline{1}$ 状态"的指示函数）。定义加权矩阵
\[
(B_\theta)_{ij}:=\sum_{e:i\to j}\exp\bigl(\theta\,\phi(e)\bigr),
\]
其中求和遍历所有从 $i$ 到 $j$ 的边。记
\[
\lambda(\theta):=\rho(B_\theta),\qquad P(\theta):=\log\lambda(\theta).
\]
则 $P(\theta)$ 为有限型情形的压力函数，且 $P(0)=\log 3$。令
\[
I(\alpha):=\sup_{\theta\in\RR}\bigl(\theta\alpha-(P(\theta)-P(0))\bigr),
\]
则 $I(\alpha)$ 给出 $\phi$ 的长程平均的大偏差率函数（标准热力学形式口径，见 \cite{Ruelle1978Thermodynamic,ParryPollicott1990Zeta}）。

加权 $\zeta$ 仍由
\[
\zeta_{B,\theta}(z)=\frac{1}{\det(I-zB_\theta)}
\]
给出，并可定义加权 primitive 轨道谱
\[
a_n(\theta):=\mathrm{Tr}(B_\theta^n),\qquad
p_n(\theta)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}(d\theta),
\]
以及相应的加权 Mertens 常数族（对 $z\to 1/\lambda(\theta)$ 取 Abel 有限部）。
可审计数值由脚本 \texttt{scripts/exp\_sync\_kernel\_weighted\_pressure.py} 输出。

\subsubsection{加权 primitive--periodic 的 Witt 分解}\label{app:witt-weighted}
以下取输出位势 $\varphi(e)=e$，令
\[
u=e^\theta,\qquad B(u)=B_0+uB_1,\qquad \Delta(z,u):=\det(I-zB(u)).
\]
其中 $B_0$（resp.\ $B_1$）只计数输出位 $e=0$（resp.\ $e=1$）的边。
则
\[
\zeta(z,u)=\exp\!\left(\sum_{n\ge 1}\frac{\mathrm{Tr}(B(u)^n)}{n}z^n\right)=\Delta(z,u)^{-1}.
\]
令
\[
a_n(u):=\mathrm{Tr}(B(u)^n),\qquad
P(z,u):=\sum_{n\ge 1}p_n(u)z^n,
\]
其中
\(
p_n(u)=\sum_{\tau\ \mathrm{primitive},\,|\tau|=n}u^{S_\tau\varphi}
\)
为 primitive 轨道的加权谱多项式。Euler 乘积给出
\[
\zeta(z,u)=\prod_{\tau\ \mathrm{primitive}}\bigl(1-z^{|\tau|}u^{S_\tau\varphi}\bigr)^{-1}.
\]
取对数并展开幂重复得到 Witt 分解
\[
\log\zeta(z,u)=\sum_{m\ge 1}\frac{1}{m}P(z^m,u^m).
\]
由 Möbius 反演得到
\[
P(z,u)=\sum_{m\ge 1}\frac{\mu(m)}{m}\log\zeta(z^m,u^m),
\]
对系数比较即有
\[
p_n(u)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}(u^d)
\quad\Longleftrightarrow\quad
p_n(\theta)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}(d\theta).
\]

\subsubsection{Big-Witt/necklace 桥包与可检验同余}\label{app:witt-bridge-package}
上式不仅是“可计算公式”，而是 Big-Witt（necklace ring）上的标准坐标变换：$a_n(u)$ 为 ghost 分量，$p_n(u)$ 为 Witt 坐标，而 Euler 乘积是 plethystic exponential。等价地，对任意带参矩阵族 $T(u)$，有严格恒等式
\[
\boxed{\ \zeta_T(z,u)=\mathrm{Exp}_W\bigl(\mathrm{Mob}_W(\mathrm{TrSeq}(T))\bigr)\ }
\]
其中 $\mathrm{TrSeq}(T)=(a_n(u))_{n\ge 1}$、$\mathrm{Mob}_W(a)=p$ 给出 $p_n(u)=\frac1n\sum_{d\mid n}\mu(d)\,a_{n/d}(u^d)$，而 $\mathrm{Exp}_W$ 由
\(
\log\zeta=\sum_{m\ge 1}\frac{1}{m}P(z^m,u^m)
\)
定义。该“桥包”直接产出一批可检验算术约束：

\begin{proposition}[交换图的函子化：intertwiner 下的自然性]\label{prop:witt-bridge-functorial}
取系数环 \(R\)（例如 \(R=\ZZ[u]\) 或 \(R=\ZZ[u,u^{-1}]\)），并令 \(\mathbf{End}_R\) 的对象为 \((V,T)\)（有限秩自由 \(R\)-模与其 \(R\)-线性自同态），态射为 intertwiner \(f:(V,T)\to (W,S)\) 满足 \(f\circ T=S\circ f\)。
令 \(\mathbf{Seq}_R\) 为 \(R\)-值序列范畴（按分量逐项映射），并定义
\[
\mathsf{Ghost}:\mathbf{End}_R\to \mathbf{Seq}_R,\qquad
(V,T)\mapsto (a_n)_{n\ge 1},\ a_n:=\Tr(T^n).
\]
则 \(\mathsf{Ghost}\) 为函子，且对任意 intertwiner \(f\) 有 \(a_n(T)=a_n(S)\)（因此 \(\zeta_T=\zeta_S\)）。
再令 \(\mathsf{Mob}\) 为带 Adams 操作 \(\psi^d(u)=u^d\) 的 Witt--Möbius 变换
\[
\mathsf{Mob}(a)=p,\qquad
p_n(u):=\frac1n\sum_{d\mid n}\mu(d)\,a_{n/d}(u^d),
\]
并令 \(\mathsf{Euler}\) 为由 \(p\) 生成 Euler 乘积（等价于 plethystic exponential）的构造。
则对每个 \((V,T)\) 有严格恒等式
\[
\boxed{\;
\mathsf{Zeta}(V,T)
=\mathsf{Euler}\bigl(\mathsf{Mob}(\mathsf{Ghost}(V,T))\bigr)\; },
\qquad
\mathsf{Zeta}(V,T):=\exp\!\left(\sum_{n\ge 1}\frac{\Tr(T^n)}{n}z^n\right)=\det(I-zT)^{-1}.
\]
因此“trace（时间 ghost）\(\leftrightarrow\) primitive（空间原子）”的交换图不仅是类比，而是由 Big-Witt 坐标变换给出的严格自然变换。
\end{proposition}

\begin{proposition}[necklace 整除律]\label{prop:witt-necklace-div}
对任意 $n\ge 1$，有
\[
n\ \Big|\ \sum_{d\mid n}\mu(d)\,a_{n/d}(u^d)\qquad\text{于}\ \mathbb{Z}[u].
\]
\end{proposition}
\begin{proof}
由于 $p_n(u)\in\mathbb{Z}[u]$ 是 primitive 轨道的加权计数，多项式恒在整系数环内；由 $n\,p_n(u)=\sum_{d\mid n}\mu(d)a_{n/d}(u^d)$ 得整除结论。证毕。
\end{proof}

\begin{corollary}[$p$-幂 Dwork 型同余]\label{cor:witt-dwork-congruence}
取 $n=p^k$（素数 $p$），则
\[
a_{p^k}(u)\equiv a_{p^{k-1}}(u^p)\pmod{p^k}\qquad\text{于}\ \mathbb{Z}[u].
\]
\end{corollary}
\begin{proof}
由 $\mu(1)=1,\mu(p)=-1,\mu(p^2)=0$ 得
\(
p^k\,p_{p^k}(u)=a_{p^k}(u)-a_{p^{k-1}}(u^p)
\)。
利用命题 \ref{prop:witt-necklace-div} 即得结论。证毕。
\end{proof}

\begin{corollary}[\texorpdfstring{$p$}{p}-进超同余：\texorpdfstring{$a_{p^k}(-1)$}{a_{p^k}(-1)} 被 \texorpdfstring{$p^k$}{p^k} 整除]\label{cor:witt-dwork-supercongruence-minus1}
取任意奇素数 \(p\)。则对任意 \(k\ge 1\) 有
\[
\boxed{\ a_{p^k}(-1)\equiv 0\pmod{p^k}\ }.
\]
\end{corollary}
\begin{proof}
由推论 \ref{cor:witt-dwork-congruence}，对 \(k\ge 1\) 有
\[
a_{p^k}(u)\equiv a_{p^{k-1}}(u^p)\pmod{p^k}\qquad\text{于}\ \ZZ[u].
\]
在 \(u=-1\) 处特化并用 \(p\) 为奇数（从而 \((-1)^p=-1\)）得
\[
a_{p^k}(-1)\equiv a_{p^{k-1}}(-1)\pmod{p^k}.
\]
而 \(p^{k-1}\) 为奇数，故由命题 \ref{prop:trace-palindrome} 的结论 \(a_n(-1)=0\)（奇数 \(n\)）知 \(a_{p^{k-1}}(-1)=0\)。
于是 \(a_{p^k}(-1)\equiv 0\pmod{p^k}\)。证毕。
\end{proof}

\begin{remark}[\texorpdfstring{$p$}{p}-幂方向的 Witt 解释：Frobenius 兼容的 ghost 分量]\label{rem:witt-ptypical-ghost}
把 \(a_n(u)\) 视作 Big-Witt（necklace）环上的 ghost 坐标，则 \(\{a_{p^k}(u)\}_{k\ge 0}\) 是其在 \(p\)-幂索引上的 ghost 子序列。
推论 \ref{cor:witt-dwork-congruence} 的 Dwork 型同余表明该子序列在 \(u\mapsto u^p\) 的 Adams/Frobenius 操作下满足强的 \(p\)-进兼容性：
\[
a_{p^k}(u)-a_{p^{k-1}}(u^p)\in p^k\,\ZZ[u].
\]
因此同步核的 primitive/trace 桥包不仅在复解析侧可检验（\(\zeta\)、Euler 乘积），在 \(p\)-进整性侧也携带一条自然的“Frobenius-lift”约束。
在后续若考虑 \(p\)-进方向的插值或对比不同核的“算术指纹”，该同余可作为最基本的可审计输入。
\end{remark}

\subsubsection{同步核的自对偶对称}\label{app:kernel-self-dual}
对同步核存在状态对合 $\iota$，其作用为

设 $z_\star(u)$ 为最小正根、$\lambda(u)=1/z_\star(u)$，则

对按输出 1 次数分层的 primitive 轨道数 $p_{n,k}$，有

等价地 $u^{-1}p_n(u)$ 为次数 $n-2$ 的回文多项式。

\begin{theorem}[自对偶强制的 Gallavotti--Cohen 涨落对称（全域形式）]\label{thm:sync-kernel-gallavotti-cohen}
令 \(u=e^{\theta}\) 且 \(P(\theta)=\log\lambda(e^\theta)\)。定义完成化后的中心化累积母函数
\[
\Lambda(\theta):=P(\theta)-\frac{\theta}{2}.
\]
则对所有实 \(\theta\) 有精确恒等式
\[
\boxed{\ \Lambda(\theta)=\Lambda(-\theta)\ }.
\]
等价地，
\[
\boxed{\ P(\theta)=\theta+P(-\theta)\ }.
\]
此外：
\begin{enumerate}
  \item \textbf{（有限长度镜像对称）} 对任意 \(n\ge 2\) 有 \(p_{n,k}=p_{n,n-k}\)；
  \item \textbf{（率函数对称）} 若 \(I(\alpha)\) 为 \(\alpha=k/n\) 的大偏差率函数，则在其定义域上有
  \(\boxed{\,I(\alpha)=I(1-\alpha)\,}\)；
  \item \textbf{（奇阶累积量消失）} \(\Lambda\) 为偶函数蕴含 centered current 的所有奇阶累积量严格为零：
  \(P^{(2j+1)}(0)=0\ (j\ge 1)\)（参见命题 \ref{prop:fluctuation-even}）。
\end{enumerate}
\end{theorem}
\begin{proof}
\(\Lambda(\theta)\) 为偶函数与 \(P(\theta)=\theta+P(-\theta)\) 在代数上等价。
而 \(P(\theta)=\theta+P(-\theta)\) 正是本节自对偶相似恒等式 \(\Pi^{-1}B(u)\Pi=uB(1/u)\) 对 Perron 根取对数的直接推论。
有限长度镜像对称 \(p_{n,k}=p_{n,n-k}\) 已在上文给出；由 Legendre 对偶 \(I(\alpha)=\sup_{\theta}(\alpha\theta-P(\theta))\) 与 \(P(\theta)=\theta+P(-\theta)\) 即得 \(I(\alpha)=I(1-\alpha)\)。
最后奇阶累积量的结论由 \(\Lambda\) 偶性对 \(\theta=0\) 的导数给出，并与命题 \ref{prop:fluctuation-even} 的表述一致。证毕。
\end{proof}

\begin{proposition}[表示级自对偶：一维角色扭曲的函数方程]\label{prop:kernel-self-dual-character}
沿用 \S\ref{app:kernel-self-dual} 的记号。设 \(G\) 为有限阿贝尔群，\(\chi:G\to\CC^\times\) 为一维角色，且假设群标签只依赖输出位 \(e\in\{0,1\}\)：对 \(e=0\) 取标签为单位元、对 \(e=1\) 取某固定元 \(g_1\in G\)。
则相应的角色扭曲矩阵可写成
\[
B_\chi(u)=B_0+u\,\chi(g_1)\,B_1.
\]
在自对偶对称 \(\Pi^{-1}B_0\Pi=B_1,\ \Pi^{-1}B_1\Pi=B_0\) 下，有相似恒等式
\[
\boxed{\;
\Pi^{-1}B_\chi(u)\Pi
=u\,\chi(g_1)\,B_{\chi^{-1}}(1/u)\; }.
\]
因此行列式与 \(\zeta\)-函数满足表示级函数方程
\[
\boxed{\;
\Delta_\chi(z,u):=\det(I-zB_\chi(u))
=\Delta_{\chi^{-1}}\!\bigl(u\,\chi(g_1)\,z,\ 1/u\bigr)\; },
\]
\[
\boxed{\;
\zeta_\chi(z,u):=\Delta_\chi(z,u)^{-1}
=\zeta_{\chi^{-1}}\!\bigl(u\,\chi(g_1)\,z,\ 1/u\bigr)\; }.
\]
\end{proposition}
\begin{proof}
由 \(\Pi^{-1}B_0\Pi=B_1,\ \Pi^{-1}B_1\Pi=B_0\) 得
\[
\Pi^{-1}B_\chi(u)\Pi
=B_1+u\,\chi(g_1)\,B_0
=u\,\chi(g_1)\Bigl(B_0+\frac{1}{u}\chi(g_1)^{-1}B_1\Bigr)
=u\,\chi(g_1)\,B_{\chi^{-1}}(1/u).
\]
对两边取 \(\det(I-z\,\cdot)\) 并用 \(\det(I-z\alpha A)=\det(I-(\alpha z)A)\) 即得所示行列式函数方程；\(\zeta\) 版本由取倒数得到。证毕。
\end{proof}

\begin{proposition}[周期迹多项式的回文对称]\label{prop:trace-palindrome}
记
\[
a_n(u):=\Tr(B(u)^n)=\sum_{k=0}^{n}a_{n,k}u^k.
\]
则对所有 $n\ge 1$ 有
\[
a_n(u)=u^n a_n(1/u)
\quad\Longleftrightarrow\quad
a_{n,k}=a_{n,n-k}.
\]
\end{proposition}

\begin{proof}
由自对偶 $\Pi^{-1}B(u)\Pi=uB(1/u)$ 与迹不变性，
\[
a_n(u)=\Tr(B(u)^n)=\Tr\!\bigl((uB(1/u))^n\bigr)=u^n\Tr(B(1/u)^n)=u^n a_n(1/u),
\]
比较系数即得结论。证毕。
\end{proof}

\begin{corollary}[奇阶消失律]\label{cor:trace-odd-vanish}

a_n(-1)=0,
\qquad
(u+1)\mid a_n(u)\ \ \text{于}\ \mathbb{Z}[u].

\end{corollary}
\begin{proof}

\end{proof}

\subsubsection{自对偶 \texorpdfstring{$\Delta(z,u)=\Delta(uz,1/u)$}{Delta(z,u)=Delta(uz,1/u)} 的 \texorpdfstring{$s\mapsto 1-s$}{s->1-s} 函数方程}\label{app:kernel-functional-equation}
本小节把二变量自对偶对称压缩为 Dirichlet 变量下的一条单变量函数方程，从而与经典 $\zeta(s)$ 叙事中的“完成化/函数方程/临界线”三件套对齐。

\begin{proposition}[完成化行列式的函数方程]\label{prop:kernel-xi-functional-equation}
设同步核满足自对偶恒等式 $\Delta(z,u)=\Delta(uz,1/u)$（见 \S\ref{app:kernel-self-dual}），并取
\[
\lambda:=\rho(B(1))>1.
\]
定义沿曲线
\[
z(s):=\lambda^{-s},\qquad u(s):=\lambda^{2s-1}
\]
的单变量函数
\[
\Xi(s):=\Delta(z(s),u(s))=\Delta(\lambda^{-s},\lambda^{2s-1}).
\]
则有严格函数方程
\[
\boxed{\ \Xi(s)=\Xi(1-s)\ }.
\]
等价地，定义 $\zeta$ 的同一截面
\[
\zeta_{\mathrm{sd}}(s):=\zeta(z(s),u(s))=\Delta(\lambda^{-s},\lambda^{2s-1})^{-1},
\]
则 $\zeta_{\mathrm{sd}}(s)=\zeta_{\mathrm{sd}}(1-s)$。
\end{proposition}
\begin{proof}
由自对偶恒等式，
\[
\Xi(s)=\Delta(\lambda^{-s},\lambda^{2s-1})
=\Delta(\lambda^{2s-1}\lambda^{-s},\lambda^{1-2s})
=\Delta(\lambda^{s-1},\lambda^{1-2s}).
\]
而对 $1-s$ 有
\(
z(1-s)=\lambda^{-(1-s)}=\lambda^{s-1}
\)
且
\(
u(1-s)=\lambda^{2(1-s)-1}=\lambda^{1-2s}
\)，
故 $\Xi(s)=\Delta(z(1-s),u(1-s))=\Xi(1-s)$。证毕。
\end{proof}

\begin{remark}[临界线与对称固定点]\label{rem:kernel-critical-line-u1}
由定义 $u(s)=\lambda^{2s-1}$ 可见：直线 $\Re(s)=\tfrac12$ 满足 $|u(s)|=1$，并且 $s=\tfrac12$ 为对称固定点（对应 $u=1$）。
因此在该完成化截面上，“临界线”自然等价为“自对偶对称面/单位圆权重”。
\end{remark}

\subsubsection{压力函数的解析性与显式导数}\label{app:pressure-analytic}
对 $u=e^\theta>0$，矩阵 $B(u)$ 仍为 primitive，故 Perron 根 $\lambda(u)$ 在 $u>0$ 上为单根并随 $u$ 实解析。记 $\lambda=\lambda(u)$，其满足六次方程

\begin{proposition}[完成化回文降维：\texorpdfstring{$F(\lambda,u)=0$}{F=0} 等价于 \texorpdfstring{$Q(s,y)=0$}{Q=0}]\label{prop:weighted-completion-Q}

\begin{corollary}[奇长度的严格奇偶平衡：\texorpdfstring{$p_n(-1)=0$}{p_n(-1)=0}]\label{cor:primitive-odd-parity-balance}

\boxed{\ p_n(-1)=\sum_{k\ge 0}(-1)^k p_{n,k}=0\ }.
\]

\end{corollary}
\begin{proof}
由系数镜像对称 \(p_{n,k}=p_{n,n-k}\) 可得（见上文）
\(
p_n(u)=u^n p_n(1/u)

\(
p_n(r^2)=r^{2n}p_n(r^{-2})
\)，

\(
\ZZ[r,r^{-1}]^{\,r\mapsto r^{-1}}=\ZZ[r+r^{-1}]=\ZZ[s]
\)
即得 \(\widehat p_n(s)\in\ZZ[s]\)。
\par

\par

\end{proof}
