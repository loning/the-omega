\begin{theorem}[Scale--time bicomplex: unique obstruction to global unified time given by computable spectral sequence boundary]\label{thm:xi-scale-time-bicomplex-obstruction}
Let the resolution tower \((X_m,p_{m+1,m})\) induce groupoids \(\mathcal G_m\), and at each layer give a strictly additive cocycle \(\ell_m:\mathcal G_m\to\mathbb Z\).
Take bicomplex \(C^{p,q}\): horizontal differential is groupoid cohomology boundary \(\delta\), vertical differential is scale difference \(\partial\) (defined by pullback \(p_{m+1,m}^\ast\)).
Denote \(\ell:=(\ell_m)_m\) as the total cochain composed of length cocycles at each layer, and let the total complex cocycle condition be \((\delta+\partial)\ell=0\).
Then there exists a natural spectral sequence
\[
E_r^{p,q}\Rightarrow H^{p+q}(\mathrm{Tot}(C^{\bullet,\bullet})),
\]
such that "there exists global unified time" if and only if the obstruction class \(\mathfrak o\in E_2^{2,0}\) vanishes.
If \(\mathfrak o\neq 0\), the minimal time central extension rank equals the rank of the abelian subgroup generated by \(\mathfrak o\) in \(H^2\).
Therefore, the unified time problem reduces to a second-order boundary vanishing criterion; in the finitely presented case, this criterion can be computed and determined from finite data.
\end{theorem}

\begin{proof}
Unified time corresponds to a total cocycle being correctable by a global \(1\)-cochain to a strictly additive representative.
This solvability is characterized in the bicomplex by second-order boundary vanishing, i.e., \(E_2^{2,0}\) obstruction.
Nonzero obstruction must externalize a central coordinate, with minimal rank given by the number of independent obstruction generators.
\end{proof}

\begin{theorem}[Time--entropy Pareto frontier: fiber-internal optimal disambiguation strategy must be Gibbs form and temperature forced to global constant by composability]\label{thm:xi-time-entropy-pareto-gibbs-global-temperature}
Let \(\mathrm{Fold}:\Omega\to X\), and given fiber-internal disambiguation time \(\tau:\Omega\to\mathbb N\).
For each \(x\in X\), simultaneously minimize \(\mathbb E[\tau\mid x]\) and maximize \(H(\pi(\cdot\mid x))\) over conditional distribution \(\pi(\cdot\mid x)\).
Then all Pareto-optimal points are uniquely parametrized as
\[
\pi_\lambda(\omega\mid x)
=
\frac{e^{-\lambda\tau(\omega)}}{\sum_{\omega'\in \mathrm{Fold}^{-1}(x)}e^{-\lambda\tau(\omega')}}\qquad(\lambda>0).
\]
If we further require the strategy to be closed under morphism composition (composable optimality), then \(\lambda\) cannot depend on \(x\): there exists a global constant \(\lambda_\ast\) such that all fibers simultaneously take \(\pi_{\lambda_\ast}\).
Therefore, time unification at the statistical level is equivalent to global consistency of the temperature parameter.
\end{theorem}

\begin{proof}
For fixed \(x\), dual-objective Lagrangianization gives the unique exponential family solution, i.e., Gibbs form.
Composition closure requires cross-fiber logarithmic potential additive constants to be consistent; if \(\lambda\) varies with \(x\), it breaks composition covariance.
Thus only a single global temperature parameter is allowed.
\end{proof}

\begin{theorem}[Rigidity of golden base logarithm: time scale of Zeckendorf scanning unique under regularity]\label{thm:xi-logphi-time-rigidity}
Let scale function \(S(m)\) (e.g., maximal representable scale at resolution \(m\) under Zeckendorf/Fibonacci canonical syntax) satisfy Fibonacci recursion
\[
S(m+1)=S(m)+S(m-1),\qquad S(m)\sim c\varphi^m.
\]
Let time function \(t:\mathbb N\to\mathbb R_{\ge 0}\) satisfy:
\begin{enumerate}
  \item Approximately additive: \(|t(ab)-t(a)-t(b)|\le C\);
  \item Locally regular: \(|t(n+1)-t(n)|\le L\);
  \item Resolution consistent: \(t(S(m))=m+O(1)\).
\end{enumerate}
Then there exists a constant \(B\) such that
\[
t(n)=\frac{\log n}{\log\varphi}+B+o(1)\qquad(n\to\infty).
\]
\end{theorem}

\begin{proof}
By \(S(m)\sim c\varphi^m\) and condition (3), \(t\) synchronizes with \(\log_\varphi\) on the Fibonacci subsequence.
Conditions (1)(2) stably extrapolate this synchronization from sparse subsequence to all integers, with error retaining only bounded drift and sub-leading terms.
\end{proof}

\begin{theorem}[Galois adjunction of time--resolution: reciprocal closure and stable type fixed points]\label{thm:xi-time-resolution-galois-adjunction}
On ordered set \((\mathcal R,\preceq)\), let \(T:\mathcal R\to\mathbb R_{\ge 0}\) be minimal realization time, \(R:\mathbb R_{\ge 0}\to\mathcal R\) be finest resolution attainable within budget \(t\).
If \(T\) is monotone and subadditive
\[
T(\rho\vee\rho')\le T(\rho)+T(\rho'),
\]
then the adjunction law holds
\[
\rho\preceq R(t)\ \Longleftrightarrow\ T(\rho)\le t.
\]
Therefore, \(R\circ T\) and \(T\circ R\) are respectively closure operators;
\(\mathrm{Fix}(R\circ T)\) forms a complete lattice, with each fixed point corresponding to a stable type under time progression.
\end{theorem}

\begin{proof}
By definition, \(R(t)\) is the supremal object under the constraint \(T(\rho)\le t\), thus yielding the adjunction iff relation.
Adjunction naturally induces bilateral closures and fixed point lattice structure; stable types are resolution classes where closure no longer changes.
\end{proof}

\begin{theorem}[Mixed curvature zero criterion: unification of addition and multiplication on same time axis iff alternating square defect is zero]\label{thm:xi-mixed-curvature-zero-criterion}
Let two classes of composable evolution \(\circ_+\), \(\circ_\times\) respectively carry length cocycles \(\ell_+,\ell_\times\).
For any alternating square
\(
w\circ_+u\simeq u'\circ_\times w'
\)
define mixed curvature
\[
K(w,u):=\ell_+(w)+\ell_\times(u)-\ell_\times(u')-\ell_+(w').
\]
Then:
\begin{enumerate}
  \item \(K\) is additive under square gluing, forming a \(2\)-cocycle;
  \item There exists a single time function \(\widetilde\ell\) strictly additive for both \(\circ_+\), \(\circ_\times\)
  if and only if \(K\equiv 0\).
\end{enumerate}
If \(K\neq 0\), minimal unified correction is to add to central register the abelian group generated by \(\{K(w,u)\}\).
\end{theorem}

\begin{proof}
\(K\) is essentially the length difference of two paths in an alternating square; differences are additive under gluing.
If a common additive \(\widetilde\ell\) exists, any square difference must be zero; conversely, zero difference gives path-independent length, allowing definition of unified \(\widetilde\ell\).
Nonzero difference can only be absorbed by externalizing to central coordinate.
\end{proof}

\begin{theorem}[Pressure characterization of critical slice: time reversal symmetry equivalent to joint vanishing of zero drift and zero Lyapunov]\label{thm:xi-critical-slice-pressure-zero-drift-lyapunov}
Let \(\theta\mapsto\mathcal L_\theta\) be a family of transfer operators, with time pressure
\(
P(\theta):=\log\rho(\mathcal L_\theta)
\).
Denote drift \(v(\theta):=P'(\theta)\) (when it exists) and principal Lyapunov exponent \(\chi(\theta)\).
If the system satisfies strict time reversal conjugation \(\theta\leftrightarrow-\theta\), then \(P(\theta)=P(-\theta)\) and \(v(0)=0\).
Under auditable microreversibility condition,
\[
\theta=0
\Longleftrightarrow
v(\theta)=0\ \text{and}\ \chi(\theta)=0,
\]
and \(\theta=0\) is the unique dual-zero slice.
\end{theorem}

\begin{proof}
Time reversal makes pressure even-symmetric, thus central point drift is zero.
Under microreversibility, nonzero \(\theta\) will introduce net drift or exponential amplification in at least one; dual-zero holds simultaneously only at the conjugation fixed slice.
\end{proof}

\begin{theorem}[Identifiability of time \(\zeta\)-data: weighted determinant family determines Markov isomorphism class and time cocycle class of protocol]\label{thm:xi-time-zeta-identifiability-markov-cohomology}

\end{theorem}

\begin{proof}
Full-parameter determinant family determines weighted closed orbit counting of all orders, thus determining Markov structural isomorphism class.

\end{proof}

\begin{theorem}[Finite-time distinguishability threshold: proportion of indistinguishable microstates decays exponentially in time, with exponent exactly the per-unit-time mutual information rate]\label{thm:xi-finite-time-distinguishability-mutual-information-rate}
Let underlying microstate process \((\omega_t)\) be stationary ergodic, observation be folded output \(x_t=\mathrm{Fold}(\omega_t)\), and let \(\mathcal F_t:=\sigma(x_0,\dots,x_t)\).
Define time-\(t\) indistinguishability relation: for two initial values \(\omega,\omega'\), if their induced \(\mathcal F_t\)-conditional distributions coincide, then write \(\omega\sim_t\omega'\).
Denote the logarithm of average indistinguishability class size at time \(t\) as \(\Delta(t)\).
Under positive entropy and finite memory realizability conditions, the limit
\[
\lim_{t\to\infty}\frac1t\Delta(t)
=
H(\omega_0\mid x_0)-\lim_{t\to\infty}\frac1t I(\omega_0;x_0,\dots,x_t)
\]
exists; moreover, if \(\omega,\omega'\) are independently drawn from stationary distribution, then the indistinguishability proportion satisfies
\[
\mathbb P(\omega\sim_t\omega')\asymp e^{-t\mathcal I},
\]
where \(\mathcal I:=\lim_{t\to\infty}\frac1t I(\omega_0;x_0,\dots,x_t)\) is the per-unit-time mutual information rate.
Thus time progression resolves fiber degrees of freedom at exponential rate \(\mathcal I\).
\end{theorem}

\begin{proof}
By chain rule and stationary ergodicity, information gain rate limit exists and equals per-unit-time mutual information rate.
Indistinguishability class volume contracts at conditional entropy rate, thus probability ratio exhibits exponential law with exponent \(\mathcal I\).
\end{proof}

\begin{theorem}[Dual reconstruction of time boundary: time filtration tree limit space and protocol cohomology class mutually determine in both directions]\label{thm:xi-time-boundary-double-reconstruction}
Along the notation of Theorem \ref{thm:xi-finite-time-distinguishability-mutual-information-rate}, let \(\sim_t\) be the indistinguishability relation under time budget \(t\), and define the stratified quotient tree

\[
\partial_T:=\varprojlim_{t}\mathcal T_t.

For any \(\lambda>0\), for \(\xi=(\xi_t)_t,\eta=(\eta_t)_t\in\partial_T\), define the maximal common prefix depth
\[
\tau(\xi\wedge\eta):=\sup\{\,t:\ \xi_t=\eta_t\,\}\in[0,\infty]\qquad(\exp(-\lambda\cdot\infty):=0),

  \item \textbf{(Reverse reconstruction of cohomology class)} Given \((\partial_T,d_\lambda)\) produced by any realization and its natural cylinder set measure class (induced by ergodic invariant measure on stratified cylinder set mass), one can reconstruct the time cocycle class in the cohomology sense (up to a gauge coboundary; equivalent to closed loop period sum invariance, see Proposition \ref{prop:xi-cycle-sums-determine-time-cohomology}).
\end{enumerate}
Therefore, time both generates boundary geometry and can be reconstructed from boundary geometry; "existence/visibility" in this sense is equivalent to points on \(\partial_T\).
\end{theorem}
\begin{proof}
By definition, \(\partial_T\) is the compatible limit of indistinguishability partitions progressively refined as \(t\uparrow\infty\); the definition of \(\tau(\xi\wedge\eta)\) gives the standard prefix ultrametric of tree boundary, thus \(d_\lambda\) is an ultrametric.

For (1), gauge coboundary perturbation only changes "which objects are viewed as equal cost/budget at each layer" local thresholds, without changing closed loop period sums and their induced cohomology classes; this is equivalent to a rescaling on the tree boundary preserving cylinder set hierarchy, thus inducing bi-Lipschitz equivalent boundary geometry.

For (2), cylinder set measure assigns values to \(\mathcal T_t\) at hierarchy \(t\), providing scale information along boundary rays; its compatibility on closed loops (verifiable by finite cycle basis under ergodic/strong connectivity) forces this scale information to be carried by a unique time cocycle class in the cohomology sense.
Combining this "period sum consistency" with finite cycle basis discrimination (Proposition \ref{prop:xi-cycle-sums-determine-time-cohomology}), we obtain reverse reconstruction of time cohomology class (up to coboundary). Proof complete.
\end{proof}

\begin{theorem}[Cofinal sparsification criterion for Toeplitz--PSD hierarchy]\label{thm:xi-toeplitz-psd-cofinal-sparsification}
Let
\[
C(w)=c_0+2\sum_{n\ge 1}c_n w^n
\]
be a Carathéodory candidate function on \(\mathbb D\), and let
\[
T_N:=(c_{j-k})_{0\le j,k\le N}.
\]
Then the following two are strictly equivalent:
\begin{enumerate}
  \item For all \(N\ge 0\), \(T_N\succeq 0\);
  \item There exists any cofinal strictly increasing subsequence \(N_1<N_2<\cdots\to\infty\) such that for all \(k\), \(T_{N_k}\succeq 0\).
\end{enumerate}
In particular, in the Toeplitz--PSD formulation of RH, RH is strictly equivalent to "verifying PSD only on any cofinal subsequence (e.g., \(N_k=2^k\))."
\end{theorem}

\begin{proof}
\((1)\Rightarrow(2)\) is trivial.
Reverse direction is obtained by principal submatrix closure: any fixed \(N\) can be embedded as upper-left principal submatrix of some \(N_k\ge N\); if \(T_{N_k}\succeq 0\), then its principal submatrix \(T_N\succeq 0\).
Thus full hierarchy PSD is equivalent to any cofinal subsequence PSD.
\end{proof}

\begin{proposition}[Adams pushforward and Toeplitz principal minor monotonicity]\label{prop:xi-adams-pushforward-toeplitz-principal-minor-monotonicity}
If \(C\) is Carathéodory, then there exists a finite positive measure \(\mu\) such that
\[
c_n=\int_{\mathbb T}\xi^n\,d\mu(\xi).
\]
For any \(m\ge 1\), define pushforward \(\mu_m:=(\xi\mapsto \xi^m)_\ast\mu\), and denote
\[
c_n^{(m)}:=\int_{\mathbb T}\xi^n\,d\mu_m(\xi).
\]
Then
\[
c_n^{(m)}=c_{mn}.
\]
Let
\[
T_N^{(m)}:=(c^{(m)}_{j-k})_{0\le j,k\le N},
\]
then \(T_N^{(m)}\) (after index rearrangement) is a principal submatrix of \(T_{mN}\), thus
\[
T_{mN}\succeq 0\ \Longrightarrow\ T_N^{(m)}\succeq 0.
\]
\end{proposition}

\begin{proof}
Matrix identity is given directly by pushforward definition:
\(
\int \xi^n\,d\mu_m=\int (\xi^m)^n\,d\mu=\int \xi^{mn}\,d\mu
\).
Therefore, \(T_N^{(m)}\) is formed by \(m\)-step sampling from the original matrix sequence, embedded in the principal submatrix of \(T_{mN}\) at corresponding index subset; principal submatrices preserve positive semi-definiteness, yielding conclusion.
\end{proof}

\begin{remark}[Adams coarse-graining refutation backpropagation and "time does not increase information" principle]\label{rem:xi-adams-toeplitz-no-info-increase}
The contrapositive of Proposition \ref{prop:xi-adams-pushforward-toeplitz-principal-minor-monotonicity} indicates: if some derived hierarchy \(T_N^{(m)}\) is not positive semi-definite, then the original hierarchy \(T_{mN}\) must not be positive semi-definite.
Therefore, in the Toeplitz--PSD horizon interface, deeper hierarchies (larger \(N\)) automatically dominate all their derived hierarchies after Adams coarse-graining; any derived hierarchy refutation can strictly backpropagate to original hierarchy refutation.
\end{remark}

\begin{theorem}[Time--horizon uncertainty law: folding scale and horizon order product lower bound]\label{thm:xi-time-horizon-product-lower-bound}
Let defect point \(a\in\mathbb D\), depth
\[
h(a):=1-|a|^2.
\]
For \(m\)-folded position \(a^m\), its depth is
\[
h_m(a):=1-|a|^{2m}=1-(1-h(a))^m.
\]
Under Toeplitz--PSD endpoint area-law type refutation mechanism (triggering order dimensional lower bound \(\gtrsim h^{-1}\)), any mixed strategy with folding scale \(m\) and horizon order \(N\) must satisfy
\[
mN\ \gtrsim\ h(a)^{-1}.
\]
In the Cayley image defect of \(\zeta\), if
\(
w_\rho=\mathcal C(\gamma+i\delta)\in\mathbb D
\)
(\(\delta>0\)), then
\[
h(\rho)=1-|w_\rho|^2
=\frac{4\delta}{\gamma^2+(\delta+1)^2},
\qquad
Q_\rho:=h(\rho)^{-1}
=\frac{\gamma^2+(\delta+1)^2}{4\delta},
\]
thus
\[
mN\ \gtrsim\ Q_\rho\asymp \frac{\gamma^2}{\delta}\quad(|\gamma|\to\infty,\ \delta\ \text{fixed}).
\]
\end{theorem}

\begin{proof}
Folding pushes defect radius by power exponent, thus depth transforms by \(h_m\).
Toeplitz hierarchy triggering threshold grows at inverse dimensional scale of current depth;
combining "folding deepening" with "hierarchy detection" yields product budget lower bound.
For Cayley image, substituting explicit depth formula yields \(Q_\rho\) estimate.
\end{proof}

\begin{proposition}[Optimal allocation square-root law under product constraint]\label{prop:xi-optimal-allocation-under-product-bound}
Under constraint
\(
mN\gtrsim Q
\),
any additive cost (e.g., \(m+N\)) or balanced cost (e.g., \(\max\{m,N\}\)) has order-optimal solution satisfying
\[
m\asymp N\asymp \sqrt Q.
\]
For \(Q=Q_\rho\), we get
\[
m\asymp N\asymp \sqrt{Q_\rho}\asymp \frac{|\gamma|}{\sqrt\delta}.
\]
\end{proposition}

\begin{proof}
By AM--GM inequality,
\(
m+N\ge 2\sqrt{mN}\gtrsim 2\sqrt Q
\),
with order optimality achieved when \(m,N\) are of same order.
\(\max\{m,N\}\) similarly.
\end{proof}

\begin{proposition}[Hyperbolic additivity of double-log time]\label{prop:xi-double-log-time-hyperbolic-additivity}
If defect hyperbolic depth satisfies
\(
h(a)^{-1}\asymp e^{d(0,a)}
\),
then from product lower bound \(mN\gtrsim h(a)^{-1}\) we get
\[
\log(mN)\ \gtrsim\ d(0,a)
\quad\Longleftrightarrow\quad
\log m+\log N\ \gtrsim\ d(0,a).
\]
Thus on logarithmic scale, total time budget is isomorphic of same order with hyperbolic depth.
\end{proposition}

\begin{proof}
Taking logarithm of \(mN\gtrsim h^{-1}\) and substituting \(h^{-1}\asymp e^{d}\) yields the result.
\end{proof}

\begin{theorem}[Height window unified hard lower bound and irreducible quadratic scaling of mixed strategy]\label{thm:xi-height-window-unified-quadratic-hardness}
Under height window \(|\gamma|\le T\) and offset lower bound \(\delta\ge\delta_0>0\),
\[
Q_{\max}(T,\delta_0)
:=\max_{|\gamma|\le T,\ \delta\ge\delta_0}h(\rho)^{-1}
=\frac{T^2+(1+\delta_0)^2}{4\delta_0}.
\]
Therefore, any mixed strategy (any combination of folding scale \(m\) and Toeplitz order \(N\)) guaranteeing refutability within the window satisfies worst-case lower bound
\[
\sup_{|\gamma|\le T,\ \delta\ge\delta_0}\inf\{mN:\text{can trigger refutation}\}
\ \gtrsim\
Q_{\max}(T,\delta_0)
\asymp
\frac{T^2}{\delta_0}.
\]
This lower bound holds uniformly for all folding--horizon mixed strategies, thus quadratic hardness scaling cannot be eliminated by resource reallocation within this interface.
\end{theorem}

\begin{proof}
By \(Q_\rho\) monotone increasing in \(|\gamma|\), worst at lower bound in \(\delta\), yielding explicit formula for \(Q_{\max}\).
Again using Theorem \ref{thm:xi-time-horizon-product-lower-bound}'s product budget uniformly constraining all mixed strategies, conclusion holds.
\end{proof}

\begin{proposition}[Non-substitutability of folding threshold scale versus Toeplitz order]\label{prop:xi-folding-scale-vs-toeplitz-order-nonsubstitutability}
Let the minimal folding scale to move defect to fixed radius threshold \(r_\star\in(0,1)\) be
\[
m_\star(\rho;r_\star)
=
\left\lceil
\frac{\log r_\star^2}{\log|w_\rho|^2}
\right\rceil
\sim
\frac{\gamma^2+(\delta+1)^2}{4\delta}\log\frac1{r_\star^2}
\asymp
\frac{\gamma^2}{\delta}.
\]
Combining \(mN\gtrsim Q_\rho\) shows: \(m\) and \(N\) can only be conservatively reallocated within unified product budget, cannot mutually substitute and break hardness scaling.
Here \(m_\star(\rho;r_\star)\) gives the limiting cost of "pure folding transport" route, while Toeplitz order \(N\) corresponds to the limiting cost of "pure horizon hierarchy" route; saving in any direction must be accompanied by corresponding amplification in the other direction on the same product budget.
\end{proposition}

\begin{proof}
\(m_\star\) is defined by minimal integer solution to \(|w_\rho|^{2m}\le r_\star^2\), with asymptotic expansion given by
\(
\log|w_\rho|^2\sim -4\delta/(\gamma^2+(\delta+1)^2)
\).
Combining with product lower bound yields non-substitutability.
\end{proof}
