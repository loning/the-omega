\section{Zeckendorf shift: primitive events, trace formulas, and zeta identities}
\label{sec:zeckendorf_shift}

\subsection{Local forbidden laws and a primitive-event trace formula}
\label{subsec:zeckendorf_primitive_trace}
In the golden branch, the canonical coordinate of an observable integer (e.g. the staircase count $S_N$ extracted from a $\Theta$-readout prefix) is a Zeckendorf bitstring $(c_k)$ subject to the local constraint $c_k c_{k+1}=0$. This locality turns the Zeckendorf code space into a finite-state symbolic dynamical system, making periodic and primitive structures computable by transfer-matrix traces.

\begin{definition}[Zeckendorf shift]
Let $\Sigma_Z$ be the set of bi-infinite binary sequences
$$
\Sigma_Z:=\{c\in\{0,1\}^{\ZZ}:\ c_k c_{k+1}=0\ \forall k\in\ZZ\}.
$$
Let $\sigma:\Sigma_Z\to\Sigma_Z$ be the left shift $(\sigma c)_k:=c_{k+1}$.
\end{definition}

Equivalently, $\Sigma_Z$ is the topological Markov shift with adjacency matrix
$$
A:=\begin{pmatrix}1&1\\ 1&0\end{pmatrix},
$$
where the states are $\{0,1\}$ and $A_{ij}=1$ encodes an allowed transition $i\to j$.
See e.g.~\cite{LindMarcus1995} for background on shifts of finite type, Markov shifts, and their zeta functions.

\begin{proposition}[Periodic-point trace formula]
\label{prop:zeckendorf_trace}
For $n\ge 1$ let
$$
P_n:=\#\mathrm{Fix}(\sigma^n)=\#\{c\in\Sigma_Z:\sigma^n c=c\}.
$$
Then
$$
P_n=\operatorname{tr}(A^n).
$$
\end{proposition}

\begin{proof}
A point $c\in\Sigma_Z$ satisfies $\sigma^n c=c$ if and only if it is determined by a cyclic word $w=(w_0,\dots,w_{n-1})\in\{0,1\}^n$ such that $w_i w_{i+1}=0$ for $i=0,\dots,n-2$ and $w_{n-1}w_0=0$. Such cyclic words are in bijection with length-$n$ closed walks on the directed graph with adjacency matrix $A$, and the number of closed walks of length $n$ starting and ending at state $i$ is $(A^n)_{ii}$. Summing over $i\in\{0,1\}$ gives $P_n=\sum_i (A^n)_{ii}=\operatorname{tr}(A^n)$.
\end{proof}

\begin{definition}[Primitive events]
A \emph{primitive event of length $n$} is a $\sigma$-periodic orbit $\mathcal{O}\subset\Sigma_Z$ whose minimal period is $n$. Let $\pi_n$ denote the number of primitive events of length $n$.
\end{definition}

\begin{proposition}[M\"obius inversion for primitive event counts]
\label{prop:zeckendorf_mobius}
For every $n\ge 1$ one has
$$
P_n=\sum_{d\mid n} d\,\pi_d,
$$
and hence
$$
\pi_n=\frac{1}{n}\sum_{d\mid n}\mu_{\mathrm{Mob}}(d)\,P_{n/d}
:=\frac{1}{n}\sum_{d\mid n}\mu_{\mathrm{Mob}}(d)\,\operatorname{tr}(A^{n/d}),
$$
where $\mu_{\mathrm{Mob}}:\NN\to\{-1,0,1\}$ is the M\"obius function.
\end{proposition}

\begin{proof}
Each $\sigma$-orbit $\mathcal{O}$ of minimal period $d$ contributes exactly $d$ points to $\mathrm{Fix}(\sigma^n)$ whenever $d\mid n$, and contributes none otherwise. Summing over primitive orbits yields $P_n=\sum_{d\mid n} d\,\pi_d$. The inversion formula is M\"obius inversion on the divisor lattice.
\end{proof}

\begin{proposition}[Zeta and determinant formulas]
\label{prop:zeckendorf_zeta}
Define the dynamical zeta function
$$
\zeta_Z(z):=\exp\Big(\sum_{n\ge 1}\frac{P_n}{n}z^n\Big).
$$
Then for $|z|$ sufficiently small one has the Euler product
$$
\zeta_Z(z)=\prod_{\mathcal{O}\ \mathrm{primitive}}\frac{1}{1-z^{|\mathcal{O}|}}
:=\prod_{n\ge 1}(1-z^n)^{-\pi_n},
$$
and the determinant identity
$$
\zeta_Z(z)=\frac{1}{\det(I-zA)}=\frac{1}{1-z-z^2}.
$$
\end{proposition}

\begin{proof}
Using Proposition~\ref{prop:zeckendorf_mobius} one has
$$
\sum_{n\ge 1}\frac{P_n}{n}z^n
:=\sum_{n\ge 1}\sum_{d\mid n}\pi_d z^n
:=\sum_{d\ge 1}\pi_d\sum_{m\ge 1}\frac{z^{dm}}{m}
:=-\sum_{d\ge 1}\pi_d \log(1-z^d),
$$
so exponentiating gives the Euler product.
For the determinant identity, let $\lambda_1,\lambda_2$ be the eigenvalues of $A$. Then for $|z|<1/\max_i|\lambda_i|$,
$$
-\log\det(I-zA)=-\sum_{i=1}^2\log(1-z\lambda_i)
:=\sum_{n\ge 1}\frac{z^n}{n}\sum_{i=1}^2\lambda_i^n
:=\sum_{n\ge 1}\frac{z^n}{n}\operatorname{tr}(A^n),
$$
hence $\zeta_Z(z)=1/\det(I-zA)$. Computing the determinant for $A=\bigl(\begin{smallmatrix}1&1\\1&0\end{smallmatrix}\bigr)$ yields $\det(I-zA)=1-z-z^2$.
\end{proof}

\begin{proposition}[A canonical injective coding of integers by primitive cycles]
\label{prop:prime_embedding}
Let $N\in\NN_{>0}$ have Zeckendorf digits $(c_1,\dots,c_m)$ as in Theorem~\ref{thm:zeckendorf}, with $c_m=1$. Define the finite word
$$
w(N):=c_m c_{m-1}\cdots c_1\,0\,1\,0^{m+1}\ \in\ \{0,1\}^{2m+3},
$$
and let $x^{(N)}\in\{0,1\}^{\ZZ}$ be the bi-infinite periodic sequence obtained by repeating $w(N)$.
Then $x^{(N)}\in\Sigma_Z$, its $\sigma$-orbit $\mathcal{O}(N):=\{\sigma^k x^{(N)}:k\in\ZZ\}$ is a primitive event, and the assignment
$$
\iota:\NN_{>0}\to\{\text{primitive events in }\Sigma_Z\},\qquad \iota(N):=\mathcal{O}(N)
$$
is injective.
\end{proposition}

\begin{proof}
The Zeckendorf constraint $c_k c_{k+1}=0$ implies the finite block $c_m\cdots c_1$ contains no adjacent ones. Appending $0\,1\,0^{m+1}$ therefore produces a word $w(N)$ with no adjacent ones, and the wrap-around transition from the terminal $0$ of $0^{m+1}$ to the initial symbol $c_m=1$ is also allowed. Hence the periodic repetition $x^{(N)}$ lies in $\Sigma_Z$.

The word $w(N)$ contains a run of exactly $m+1$ consecutive zeros (the terminal block $0^{m+1}$). In the preceding block $c_m\cdots c_1\,0\,1$ every run of zeros has length at most $m$, since this block has total length $m+2$ and begins with $c_m=1$. Thus $0^{m+1}$ is the unique maximal zero run in $w(N)$.
If $w(N)$ were a nontrivial power $u^r$ with $r\ge 2$, then the unique maximal run $0^{m+1}$ would occur $r$ times in the cyclic word, contradiction. Therefore $w(N)$ is primitive and $\mathcal{O}(N)$ is a primitive event.

To see injectivity, note that from the orbit $\mathcal{O}(N)$ one recovers the cyclic word $w(N)$ up to rotation. The unique maximal run $0^{m+1}$ identifies the cut point canonically: rotate so that $0^{m+1}$ is terminal. Removing this terminal block, then removing the preceding symbol $1$ and the preceding appended $0$, leaves the Zeckendorf digit block $c_m\cdots c_1$, hence determines $(c_1,\dots,c_m)$ uniquely, and therefore $N$ uniquely by Theorem~\ref{thm:zeckendorf}.
\end{proof}

\begin{remark}[Coding versus intrinsic primality]
Proposition~\ref{prop:prime_embedding} provides a reversible \emph{coding} of integers by primitive cycles in the Zeckendorf shift. It does not, by itself, internalize multiplicative primality: whether the decoded value $N$ lies in $\PP$ is an arithmetic predicate not decided by the local forbidden law $c_k c_{k+1}=0$.
\end{remark}

\begin{remark}[Finite-state limitation and infinite-state extensions]
Because $\Sigma_Z$ is a finite-state Markov shift, its zeta function is rational (Proposition~\ref{prop:zeckendorf_zeta}) and therefore cannot exhibit the analytic complexity of objects with infinitely many nontrivial zeros. To obtain a nonrational orbit product one must pass beyond the finite-state class, for instance by introducing weights (a potential) and studying the associated transfer operator on an infinite-dimensional function space, or by moving to a countable-state grammar. In the present framework, such infinite-state refinements arise naturally by increasing readout resolution (refining partitions/windows along the $\Theta$-orbit): periodic trajectories remain cycles, but the collection of admissible cycles grows in a tower rather than collapsing to a single finite graph.
\end{remark}
