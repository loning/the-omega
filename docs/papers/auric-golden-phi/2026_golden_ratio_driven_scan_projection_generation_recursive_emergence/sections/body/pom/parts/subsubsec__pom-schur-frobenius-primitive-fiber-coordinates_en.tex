\paragraph{Frobenius Character and Primitive Fiber Coordinates: Integrality of Schur Tomography, Euler Factorization and Modulo \texorpdfstring{$p$}{p} Auditable Histogram}
This paragraph further compresses the role-weighted explicit formula of Theorem \ref{thm:pom-schur-cycle-index-tomography} into a symmetric function identity:
The Schur channel trace is strictly equivalent to the Schur polynomial evaluation of the "fiber multiplicity diagonal matrix."
This identification provides, on one hand, integrality and auditable divisibility, and on the other hand, allows derivation of Euler factorization and primitive fiber coordinates via a single generating function,
thereby obtaining a strict inversion formula for recovering the \(d_m\bmod p\) histogram from high-order collision moments.

\begin{corollary}[Frobenius--Character Identification of Schur Channel Trace: Equivariant MOM Output is \texorpdfstring{$\mathrm{GL}$}{GL}-Trace]\label{cor:pom-schur-trace-frobenius-character-identification}
At resolution \(m\), with the stable type set \(X_m\) and fiber multiplicity \(d_m:X_m\to\NN\) given, let the collision moments be
\[
S_r(m):=\sum_{x\in X_m} d_m(x)^r\qquad(r\ge 1).
\]
Define the diagonal matrix
\[
D_m:=\mathrm{diag}\bigl(d_m(x)\bigr)_{x\in X_m}.
\]
For fixed integer \(q\ge 2\) and any \(\lambda\vdash q\), let \(T_{q,\lambda}(m)\) be the Schur channel trace from Theorem \ref{thm:pom-schur-cycle-index-tomography}.
Denote by \(s_\lambda\) the Schur symmetric function, and evaluate it with the diagonal entries of \(D_m\) as variables; its power sums satisfy
\[
p_r(D_m):=\Tr(D_m^r)=S_r(m).
\]
Then there holds the identity
\[
\boxed{\;
T_{q,\lambda}(m)=f^\lambda\, s_\lambda(D_m)\; } ,
\qquad f^\lambda:=\dim S^\lambda.
\]
Thus for each \((m,q,\lambda)\) we have integrality and auditable divisibility:
\[
\frac{T_{q,\lambda}(m)}{f^\lambda}=s_\lambda(D_m)\in\ZZ,
\qquad\text{and therefore}\qquad
f^\lambda\mid T_{q,\lambda}(m).
\]
In particular, the trivial channel and sign channel respectively satisfy
\[
T_{q,(q)}(m)=h_q(D_m),\qquad
T_{q,(1^q)}(m)=e_q(D_m),
\]
where \(h_q,e_q\) are the complete and elementary symmetric functions.
\end{corollary}
\begin{proof}
By Theorem \ref{thm:pom-schur-cycle-index-tomography},
\[
\frac{T_{q,\lambda}(m)}{f^\lambda}
=\frac{1}{q!}\sum_{\sigma\in S_q}\chi^\lambda(\sigma)\prod_{r\ge 1}S_r(m)^{c_r(\sigma)}.
\]
The Frobenius characteristic isomorphism gives the power sum expansion of Schur functions
\[
s_\lambda=\frac{1}{q!}\sum_{\sigma\in S_q}\chi^\lambda(\sigma)\prod_{r\ge 1}p_r^{c_r(\sigma)}.
\]
Substituting \(p_r=p_r(D_m)=S_r(m)\) yields \(T_{q,\lambda}(m)=f^\lambda s_\lambda(D_m)\).
Since \(s_\lambda\) is an integer-coefficient symmetric polynomial and the diagonal entries of \(D_m\) are integers, we have \(s_\lambda(D_m)\in\ZZ\), hence \(f^\lambda\mid T_{q,\lambda}(m)\).
The last two formulas are special cases of Schur functions at \(\lambda=(q)\) and \(\lambda=(1^q)\).
\end{proof}

\begin{proposition}[Single Generating Function Controls All Schur Tomography: Jacobi--Trudi and Euler Factorization via Primitive Fiber Coordinates]\label{prop:pom-schur-trace-jacobi-trudi-euler-primitive-fiber}
Let the univariate generating function be
\[
H_m(z):=\sum_{q\ge 0}h_q(D_m)\,z^q
=\prod_{x\in X_m}\frac{1}{1-d_m(x)z}
=\exp\!\left(\sum_{r\ge 1}\frac{S_r(m)}{r}z^r\right).
\]
Then for any \(\lambda\vdash q\) we have the Jacobi--Trudi determinant expression
\[
\boxed{\;
s_\lambda(D_m)=\det\bigl(h_{\lambda_i-i+j}(D_m)\bigr)_{1\le i,j\le \ell(\lambda)}\;}
\]
(with the convention \(h_0=1,h_k=0\ (k<0)\)).
\par
Furthermore, introducing primitive fiber coordinates
\[
b_n(m):=\frac{1}{n}\sum_{d\mid n}\mu(d)\,S_{n/d}(m)\qquad(n\ge 1),
\]
there exists the strict Euler factorization
\[
\boxed{\;
H_m(z)=\prod_{n\ge 1}(1-z^n)^{-b_n(m)}\;}
\]
and \(b_n(m)\in\ZZ_{\ge 0}\).
If we let the directed multigraph \(G_m\) have vertex set \(X_m\), and attach exactly \(d_m(x)\) labeled self-loops at each vertex \(x\),
then \(S_n(m)=\Tr(A(G_m)^n)\) is the count of closed walks of length \(n\), and \(b_n(m)\) is precisely the count of primitive closed walks of length \(n\) (quotiented by cyclic shifts) in this graph.
\end{proposition}
\begin{proof}
The product formula for the generating function follows from the definition of \(h_q\) (fundamental generating function for complete symmetric functions);
taking logarithm and using
\(
\log(1-t)^{-1}=\sum_{r\ge 1}t^r/r
\)
and substituting \(p_r(D_m)=S_r(m)\) yields the exponential formula.
Jacobi--Trudi is the standard determinant expression for Schur functions.
\par
Euler factorization is derived from the exponential formula and power series identity:
\[
\log\prod_{n\ge 1}(1-z^n)^{-b_n}
=\sum_{n\ge 1}b_n\sum_{k\ge 1}\frac{z^{nk}}{k}
=\sum_{r\ge 1}\frac{1}{r}\left(\sum_{n\mid r}n b_n\right)z^r.
\]
Setting \(\sum_{n\mid r}n b_n=S_r(m)\) and applying Möbius inversion yields the formula for \(b_n(m)\).
Integrality follows from inversion coefficients being integers; non-negativity can be explained by the necklace decomposition of closed walks in the graph \(G_m\): closed walks of length \(n\) decompose by minimal period, and the primitive count satisfies \(S_n=\sum_{d\mid n} d\,b_d\).
\end{proof}

\begin{proposition}[\texorpdfstring{$p$}{p}-Typical Frobenius Congruence and Teichm\"uller Limit: Complete Histogram Recovery of \texorpdfstring{$d_m\bmod p$}{d\_m mod p} from High-Order Collision Moments]\label{prop:pom-collision-moment-ptypical-congruence-teichmuller-histogram}
Take prime \(p\) and fix \(m\).
\begin{enumerate}
  \item For any integer \(k\ge 1\) the congruence holds:
  \[
  \boxed{\;
  S_{p^k}(m)\equiv S_{p^{k-1}}(m)\pmod{p^k}\;}
  \]
  Thus the sequence \(\{S_{p^k}(m)\}_{k\ge 0}\) is Cauchy in \(\ZZ_p\), and the limit
  \(
  S_{p^\infty}(m):=\lim_{k\to\infty}S_{p^k}(m)\in\ZZ_p
  \)
  exists.
  \item Denote by \(\omega_p:\FF_p\to\ZZ_p\) the Teichm\"uller lift (with the convention \(\omega_p(0)=0\), and for \(a\in\FF_p^\times\) we have \(\omega_p(a)^{p-1}=1\) and \(\omega_p(a)\equiv a\ (\mathrm{mod}\ p)\)).
  For any \(j\in\{1,\dots,p-2\}\), the limit
  \[
  U_j(m):=\lim_{k\to\infty}S_{j p^k}(m)\in\ZZ_p
  \]
  exists and satisfies the identity
  \[
  \boxed{\;
  U_j(m)=\sum_{x\in X_m}\omega_p\!\bigl(d_m(x)\bigr)^j\;}
  \]
  (the right-hand side is summed in \(\ZZ_p\)).
  Additionally define
  \[
  U_0(m):=\lim_{k\to\infty}S_{(p-1)p^k}(m)=\sum_{x\in X_m}\omega_p\!\bigl(d_m(x)\bigr)^{p-1}\in\ZZ
  \]
  (this value equals \(\card{\{x\in X_m:\ p\nmid d_m(x)\}}\), corresponding to the trivial multiplicative character of \(\FF_p^\times\) extended by \(0\mapsto 0\)).
  Let the modulo \(p\) histogram be
  \[
  N_a(m):=\card{\{x\in X_m:\ d_m(x)\equiv a\ (\mathrm{mod}\ p)\}}\qquad(a\in\FF_p),
  \]
  then for any \(a\in\FF_p^\times\) we have the finite Fourier inversion
  \[
  \boxed{\;
  N_a(m)=\frac{1}{p-1}\left(U_0(m)+\sum_{j=1}^{p-2}a^{-j}\,U_j(m)\right),\qquad
  N_0(m)=|X_m|-\sum_{a\in\FF_p^\times}N_a(m)\;}
  \]
  Thus without touching individual fibers, solely through the values of a family of high-order collision moments along \(n=j p^k\) (\(1\le j\le p-2\)) and \(n=(p-1)p^k\) (\(k\to\infty\)), we can strictly recover the complete count of \(d_m\bmod p\).
\end{enumerate}
\end{proposition}
\begin{proof}
For any integer \(a\) we have the basic congruence \(a^{p^k}\equiv a^{p^{k-1}}\ (\mathrm{mod}\ p^k)\); taking \(a=d_m(x)\) for each \(x\) and summing yields (1).
\par
For (2): For any integer \(t\), \(t^{p^k}\) converges in \(\ZZ_p\) to \(\omega_p(t\bmod p)\).
Therefore for \(1\le j\le p-2\) we have \(t^{j p^k}=(t^j)^{p^k}\to \omega_p(t)^j\), while
\(
t^{(p-1)p^k}=(t^{p-1})^{p^k}\to \omega_p(t)^{p-1}
\)
(when \(t\equiv 0\ (\mathrm{mod}\ p)\) the limit is \(0\), when \(t\not\equiv 0\) the limit is \(1\)).
Summing term-by-term yields the expressions for \(U_j(m)\) (\(1\le j\le p-2\)) and \(U_0(m)\).
\par
Finally, applying discrete Fourier inversion on the multiplicative group \(\FF_p^\times\) to the function \(a\mapsto N_a(m)\): from the above formulas,
for \(1\le j\le p-2\) we have
\(
U_j(m)=\sum_{a\in\FF_p^\times}N_a(m)\omega_p(a)^j
\),
while \(U_0(m)=\sum_{a\in\FF_p^\times}N_a(m)\) corresponds to the trivial character.
For fixed \(a\in\FF_p^\times\), summing
\(
U_0+\sum_{j=1}^{p-2}a^{-j}U_j
=\sum_{b\in\FF_p^\times}N_b\sum_{j=0}^{p-2}(b/a)^j
=(p-1)N_a
\)
yields the inversion formula; \(N_0\) follows from conservation of total count.
\end{proof}

\begin{proposition}[Primitive Fiber Expansion of Trivial Channel: \texorpdfstring{$T_{q,(q)}(m)$}{T\_{q,(q)}(m)} is a Positive Integer-Coefficient Polynomial in \texorpdfstring{$b_n(m)$}{b\_n(m)}]\label{prop:pom-schur-trace-trivial-channel-primitive-fiber-expansion}
Following the Euler factorization of Proposition \ref{prop:pom-schur-trace-jacobi-trudi-euler-primitive-fiber}
\(
H_m(z)=\prod_{n\ge 1}(1-z^n)^{-b_n(m)}=\sum_{q\ge 0}h_q(D_m)z^q
\).
Denote the trivial channel \(T_{q,(q)}(m)=h_q(D_m)\).
Then coefficient extraction gives a completely explicit finite sum:
\[
\boxed{\;
T_{q,(q)}(m)=
\sum_{\substack{(k_n)_{n\ge 1}\subset\NN\\ \sum_{n\ge 1}n k_n=q}}
\ \prod_{n\ge 1}\binom{b_n(m)+k_n-1}{k_n}\;}
\]
Thus for fixed \(m\), \(T_{q,(q)}(m)\) is a positive integer-coefficient polynomial in \(\{b_n(m)\}_{1\le n\le q}\).
\end{proposition}
\begin{proof}
By Euler factorization
\(
(1-z^n)^{-b_n}=\sum_{k\ge 0}\binom{b_n+k-1}{k}z^{nk}
\).
Taking the product over all \(n\ge 1\) and extracting the coefficient of \(z^q\), we obtain the closed-form sum over all \((k_n)\) satisfying \(\sum n k_n=q\), which is the stated expression.
\end{proof}
\begin{corollary}[Primitive Fiber Coordinatization of All Schur Channels: \texorpdfstring{$T_{q,\lambda}(m)/f^\lambda\in\ZZ[b_1(m),\dots,b_q(m)]$}{T/f in Z[b]} and Constructible by Finite Determinant]\label{cor:pom-schur-trace-all-channels-primitive-fiber-coordinate}
Following the notation of Proposition \ref{prop:pom-schur-trace-jacobi-trudi-euler-primitive-fiber},
for any \(\lambda\vdash q\), the Jacobi--Trudi determinant gives
\[
\frac{T_{q,\lambda}(m)}{f^\lambda}
=s_\lambda(D_m)
=\det\bigl(h_{\lambda_i-i+j}(D_m)\bigr)_{1\le i,j\le \ell(\lambda)}.
\]
Each \(h_n(D_m)\) is an integer-coefficient polynomial in \(\{b_r(m)\}_{r\le n}\) by Proposition \ref{prop:pom-schur-trace-trivial-channel-primitive-fiber-expansion},
hence
\[
\boxed{\;
\frac{T_{q,\lambda}(m)}{f^\lambda}\in \ZZ[b_1(m),\dots,b_q(m)]\;}
\]
and all Schur channel traces can be explicitly constructed via a single finite determinant calculation.
\end{corollary}
\begin{proof}
Substitute the \(b\)-coordinate polynomial representation of \(h_{\lambda_i-i+j}(D_m)\) term-by-term into the Jacobi--Trudi determinant.
\end{proof}
