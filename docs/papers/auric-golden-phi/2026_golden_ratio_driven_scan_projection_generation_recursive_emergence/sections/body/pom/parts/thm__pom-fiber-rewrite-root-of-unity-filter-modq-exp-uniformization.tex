\begin{theorem}[任意模数下的根单位滤波与指数级均匀化]\label{thm:pom-fiber-rewrite-root-of-unity-filter-modq-exp-uniformization}
固定整数 \(q\ge 2\)，令 \(\zeta_q:=e^{2\pi i/q}\)。
对任意 \(m\ge 1\) 与 \(x\in X_m\)，在纤维 \(\Fold_m^{-1}(x)\) 上取均匀分布并记
\[
R_x:=r(\omega)\in\ZZ_{\ge 0},
\qquad
R_x^{(q)}:=R_x\bmod q\in \ZZ/q\ZZ.
\]
则对任意 \(a\in\ZZ/q\ZZ\) 有精确公式
\[
\PP(R_x^{(q)}=a\mid x)
=\frac{1}{q\,Z_x(1)}\sum_{k=0}^{q-1}\zeta_q^{-ak}\,Z_x(\zeta_q^{k}).
\]
并且存在常数 \(\theta_q\in(0,1)\)（仅依赖于 \(q\)）使得对所有 \(x\in X_m\) 与所有 \(a\in\ZZ/q\ZZ\) 成立
\[
\left|\PP(R_x^{(q)}=a\mid x)-\frac{1}{q}\right|
\le \frac{q-1}{q}\,\theta_q^{\,|V(\Gamma_x)|},
\qquad
\left\|\mathcal L(R_x^{(q)}\mid x)-\mathrm{Unif}(\ZZ/q\ZZ)\right\|_{\mathrm{TV}}
\le \frac{q-1}{2}\,\theta_q^{\,|V(\Gamma_x)|}.
\]
\end{theorem}

\begin{proof}
\textbf{根单位滤波.}
将 \(Z_x(t)\) 写为
\[
Z_x(t)=\sum_{j\ge 0}N_j(x)\,t^j,
\qquad
N_j(x):=\card{\{\omega\in\Fold_m^{-1}(x):\ r(\omega)=j\}}.
\]
则
\[
\card{\{\omega:\ r(\omega)\equiv a\ (\bmod\ q)\}}
=\sum_{j\equiv a\ (q)}N_j(x)
=\frac{1}{q}\sum_{k=0}^{q-1}\zeta_q^{-ak}\sum_{j\ge 0}N_j(x)\,\zeta_q^{kj}
=\frac{1}{q}\sum_{k=0}^{q-1}\zeta_q^{-ak}\,Z_x(\zeta_q^k),
\]
除以 \(Z_x(1)=\sum_j N_j(x)\) 得到概率公式。
\par
\textbf{指数级均匀化.}
由定理 \ref{thm:pom-fiber-indset-factorization}，若 \(\Gamma_x\simeq \bigsqcup_{i=1}^r P_{\ell_i}\)，则
\[
Z_x(t)=\prod_{i=1}^r I_{\ell_i}(t),
\qquad
Z_x(1)=\prod_{i=1}^r I_{\ell_i}(1)=\prod_{i=1}^r F_{\ell_i+2},
\qquad
|V(\Gamma_x)|=\sum_{i=1}^r \ell_i.
\]
并且每个 \(I_{\ell}(t)\) 具有严格非负系数且 \(I_\ell(0)=1\)、\([t]I_\ell(t)>0\)（见定义 \ref{def:pom-fiber-adm-path} 与命题 \ref{prop:pom-path-indset-thermo-constants} 的递推）。
因此对任意单位圆上 \(u\in\CC\) 且 \(u\neq 1\)，有严格不等式
\[
|I_\ell(u)|<I_\ell(1)\qquad(\ell\ge 1).
\]
对固定 \(q\) 与 \(k\in\{1,\dots,q-1\}\)，令线性递推矩阵
\[
M(t):=
\begin{pmatrix}
1 & t\\
1 & 0
\end{pmatrix},
\qquad
\binom{I_\ell(t)}{I_{\ell-1}(t)}=M(t)^{\ell-1}\binom{1+t}{1}\qquad(\ell\ge 1).
\]
记 \(\lambda_+(t)\) 为 \(M(t)\) 的特征根中模最大的一个（当 \(t=1\) 时 \(\lambda_+(1)=\varphi\)）。
当 \(t=\zeta_q^k\neq 1\) 时，由谱半径连续性与上式的严格三角不等式可得
\[
\rho\bigl(M(\zeta_q^k)\bigr)=|\lambda_+(\zeta_q^k)|<\rho(M(1))=\varphi.
\]
因此存在 \(C_{q,k}>0\) 与 \(\vartheta_{q,k}\in(0,1)\) 使得对所有 \(\ell\ge 1\) 有
\[
\frac{|I_\ell(\zeta_q^k)|}{I_\ell(1)}
\le C_{q,k}\,\vartheta_{q,k}^{\,\ell}.
\]
令 \(\theta_q\in(0,1)\) 取为 \(\max_{1\le k\le q-1}\vartheta_{q,k}\) 的任意严格上界（吸收常数 \(C_{q,k}\) 到 \(\theta_q\) 的有限次幂中即可），则
\[
\frac{|Z_x(\zeta_q^k)|}{Z_x(1)}
=\prod_{i=1}^r \frac{|I_{\ell_i}(\zeta_q^k)|}{I_{\ell_i}(1)}
\le \prod_{i=1}^r \theta_q^{\ell_i}
=\theta_q^{\sum_i\ell_i}
=\theta_q^{|V(\Gamma_x)|}.
\]
代回根单位滤波式并对 \(k=1,\dots,q-1\) 求和得到
\[
\left|\PP(R_x^{(q)}=a\mid x)-\frac{1}{q}\right|
\le \frac{1}{q}\sum_{k=1}^{q-1}\frac{|Z_x(\zeta_q^k)|}{Z_x(1)}
\le \frac{q-1}{q}\,\theta_q^{|V(\Gamma_x)|}.
\]
总变差界由恒等式
\(
\|P-U\|_{\mathrm{TV}}\le \tfrac12\sum_a |P(a)-U(a)|
\)
立得。证毕。
\end{proof}
