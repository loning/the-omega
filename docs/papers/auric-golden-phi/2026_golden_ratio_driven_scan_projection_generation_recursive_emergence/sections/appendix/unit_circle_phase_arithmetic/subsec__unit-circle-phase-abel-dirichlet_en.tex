\subsection{Alignment with the Abel--Dirichlet Caliber: Parameter-Free Angular Variable in Disk Polar Coordinates}\label{subsec:unit-circle-phase-abel-dirichlet}
In Appendix~\ref{app:real-input-40-dirichlet}, the Abel unit disk variable is given by
\(
z=\lambda^{-s},\ r=\lambda^{1-s}\in(0,1)\ (s>1)
\),
where \(r\to 1^{-}\) carries the critical singularity at \(s\to 1^{+}\).
This appendix provides an orthogonal, parameter-free angular variable source: let \(x\in\RR\) and take
\[
e^{\mathrm{i}\vartheta(x)}=\iota(x)=\frac{1+\mathrm{i}x}{1-\mathrm{i}x}\in\TT.
\]
Then points inside the disk can be uniformly written as
\begin{equation}\label{eq:unit-circle-phase-disk-polar}
\boxed{\ w=r\,e^{\mathrm{i}\vartheta(x)}\in\mathbb{D},\qquad r\in(0,1),\ x\in\RR\ }.
\end{equation}
Here the radial parameter \(r\) continues to be controlled by the Abel--Dirichlet interface, while the angular variable is generated parameter-free by \(\vartheta(x)=2\arctan(x)\); when restricted to the unitary slice (\(r=1\)), the above degenerates to the parameter-free parametrization \(w=\iota(x)\) of the unit circle boundary.

\paragraph{Tri-fiber injection coordinate: completing the periodic folding of the Dirichlet caliber with a ``phase axis''}
In the Dirichlet--Witt interface, the power substitution \(z\mapsto z^{m}\) translates equivalently to the dilation \(s\mapsto ms\), and the Witt--dilation
\(
(Wf)(s)=\sum_{m\ge 1}\frac{1}{m}f(ms)
\)
yields the strict identity \(\log\mathcal{Z}(s)=(W\mathcal{P})(s)\) as well as the M\"obius stripping of \(\mathcal{P}\) (see Appendix~\ref{app:real-input-40-dirichlet}).
This paragraph makes the ``imaginary-direction periodic folding'' into a coordinate-theoretic fact: the angular variable of \(z=\lambda^{-s}\) records only the \(2\pi\)-congruence class of \(t\log\lambda\); if information conservation is required, an integer axis must be externalized as a winding certificate.

\medskip
Let \(s=\sigma+\mathrm{i}t\in\CC\) with \(\lambda>1\), and write
\begin{equation}\label{eq:dirichlet-z-polar-decomposition}
\boxed{\;
z=\lambda^{-s}=\lambda^{-\sigma}\,e^{-\mathrm{i}t\log\lambda},
\qquad |z|=\lambda^{-\sigma}\in(0,1)\; }.
\end{equation}
Take the principal argument \(\mathrm{Arg}:\CC^{\times}\to(-\pi,\pi]\).
Then \(\mathrm{Arg}(z)\equiv -t\log\lambda\ (\mathrm{mod}\ 2\pi)\), so the observable \(\mathrm{Arg}(z)\) determines only the class of \(t\) in the quotient group
\(\RR/(2\pi/\log\lambda)\ZZ\).

\medskip
Define the parameter-free phase axis (Cayley compactification; coinciding with \(\iota\) from \eqref{eq:unit-circle-cayley-embed} on the boundary)
\begin{equation}\label{eq:dirichlet-cayley-phase-axis}
\boxed{\;
u_{\mathrm C}(t):=\frac{1+\mathrm{i}t}{1-\mathrm{i}t}\in\TT,
\qquad
t(u_{\mathrm C})=-\mathrm{i}\,\frac{u_{\mathrm C}-1}{u_{\mathrm C}+1}
=\mathrm{i}\,\frac{1-u_{\mathrm C}}{1+u_{\mathrm C}}\; }.
\end{equation}
and define the winding certificate as the unique integer satisfying the principal-value constraint:
\begin{equation}\label{eq:dirichlet-winding-certificate}
\boxed{\;
\kappa_{\lambda}(s)\in\ZZ
\ \text{uniquely determined by}\
\mathrm{Arg}(z)=-t\log\lambda+2\pi\,\kappa_{\lambda}(s);
\ \text{equivalently,}\
\kappa_{\lambda}(s)=\frac{t\log\lambda+\mathrm{Arg}(z)}{2\pi}\; }.
\end{equation}

\begin{definition}[Tri-fiber injection map (Dirichlet--Cayley extension)]\label{def:dirichlet-trifiber-injection}
For \(\sigma>0,t\in\RR\), let \(z=\lambda^{-(\sigma+\mathrm{i}t)}\) and take \(\kappa_{\lambda}\) as in \eqref{eq:dirichlet-winding-certificate}.
Define
\begin{equation}\label{eq:dirichlet-trifiber-injection-map}
\boxed{\;
\mathcal{E}_{\lambda}:\{\sigma>0\}\times\RR\to (0,1)\times(\TT\setminus\{-1\})\times\ZZ,\qquad
(\sigma,t)\longmapsto \bigl(|z|,\ u_{\mathrm C}(t),\ \kappa_{\lambda}(\sigma+\mathrm{i}t)\bigr)\; }.
\end{equation}
\end{definition}

\begin{proposition}[Injectivity and consistency constraint: externalized certificate for periodic folding]\label{prop:dirichlet-trifiber-injection-audit}
The map \(\mathcal{E}_{\lambda}\) is injective.
Moreover, \(\kappa_{\lambda}\) provides an auditable consistency constraint:
if the principal value \(\mathrm{Arg}(z)\) records the Dirichlet angular variable and \(u_{\mathrm C}\) records the non-periodic height, then the two must satisfy
\begin{equation}\label{eq:dirichlet-arg-consistency}
\boxed{\;t\log\lambda+\mathrm{Arg}(z)=2\pi\,\kappa_{\lambda}(s)\in 2\pi\ZZ\; }.
\end{equation}
\end{proposition}
\begin{proof}
From \(|z|=\lambda^{-\sigma}\) one recovers \(\sigma=-\log|z|/\log\lambda\).
From the inverse of \eqref{eq:dirichlet-cayley-phase-axis} one recovers \(t=t(u_{\mathrm C})\) from \(u_{\mathrm C}\).
Hence \((|z|,u_{\mathrm C})\) already determines \((\sigma,t)\), so \(\mathcal{E}_{\lambda}\) is injective.
The consistency constraint \eqref{eq:dirichlet-arg-consistency} is a direct rewriting of \eqref{eq:dirichlet-winding-certificate}. \qed
\end{proof}

\begin{remark}[Significance of the minimal extension: from folded angular variable to invertible coordinates]
From \eqref{eq:dirichlet-z-polar-decomposition} one sees that the map \((\sigma,t)\mapsto z=\lambda^{-s}\) necessarily folds in the imaginary direction due to \(2\pi\)-periodicity.
Proposition~\ref{prop:dirichlet-trifiber-injection-audit} shows that once the integer certificate \(\kappa_{\lambda}\) is externalized in the extended space, this folding can be lifted to an invertible coordinate system;
this is precisely one realization of ``extension injectivization'' (Proposition~\ref{prop:unit-circle-phase-extension-preserving}) in the Dirichlet caliber.
\end{remark}

\paragraph{Double-depth diagonalization: Witt dilations simultaneously reduce to translations at both ``infinite endpoints''}
At the \(\sigma\)-end (radial endpoint) and the \(t\)-end (height endpoint), introduce logarithmic depth coordinates
\begin{equation}\label{eq:dirichlet-double-depth-coordinates}
\boxed{\;
Y_{\sigma}:=\log\!\bigl(-\log|z|\bigr)=\log(\sigma\log\lambda),
\qquad
Y_{t}:=-\log\left|\frac{1+u_{\mathrm C}(t)}{1-u_{\mathrm C}(t)}\right|=\log|t|\; }
\qquad(\sigma>0,\ t\neq 0),
\end{equation}
where \(\frac{1+u_{\mathrm C}}{1-u_{\mathrm C}}=\mathfrak{w}(u_{\mathrm C})=\mathrm{i}/t\) is the linearization variable of the Cayley axis (see \eqref{eq:scale-linearization-halfplane-coordinate}--\eqref{eq:scale-linearization-w-vs-x}).

\begin{theorem}[Double-depth translation law and ray invariant]\label{thm:dirichlet-double-depth-translation}
For any integer \(m\ge 1\), the dilation \(s\mapsto ms\) (i.e., \((\sigma,t)\mapsto(m\sigma,mt)\)) satisfies strict additivity in the coordinates \((Y_{\sigma},Y_{t})\):
\begin{equation}\label{eq:dirichlet-double-depth-translation}
\boxed{\;
Y_{\sigma}(m\sigma)=Y_{\sigma}(\sigma)+\log m,
\qquad
Y_{t}(mt)=Y_{t}(t)+\log m\; }.
\end{equation}
Therefore the difference
\begin{equation}\label{eq:dirichlet-ray-invariant}
\boxed{\;\mathcal{I}:=Y_{\sigma}-Y_{t}=\log\!\Bigl(\frac{\sigma\log\lambda}{|t|}\Bigr)\;}
\end{equation}
is invariant under the dilation semigroup action.
\end{theorem}
\begin{proof}
From \(|z|=\lambda^{-\sigma}\) one obtains \(-\log|z|=\sigma\log\lambda\), hence
\(
Y_{\sigma}(m\sigma)=\log(m\sigma\log\lambda)=Y_{\sigma}(\sigma)+\log m
\).
On the other hand, from \(\frac{1+u_{\mathrm C}(t)}{1-u_{\mathrm C}(t)}=\mathrm{i}/t\) one has \(Y_t(t)=\log|t|\), whence
\(
Y_{t}(mt)=\log|mt|=Y_{t}(t)+\log m
\).
The invariance of the difference follows by subtraction. \qed
\end{proof}

\begin{remark}[Alignment with the Witt--dilation: updating only the ``depth ledger'' without introducing new continuous degrees of freedom]
Theorem~\ref{thm:dirichlet-double-depth-translation} shows that the dilation semigroup acts on \((Y_{\sigma},Y_t)\) as a translation along the diagonal direction and strictly preserves the ray invariant \(\mathcal{I}\).
Therefore, in the Dirichlet--Witt caliber, the operator
\(
(Wf)(s)=\sum_{m\ge 1}\frac{1}{m}f(ms)
\)
can be understood as a weighted superposition over the ``depth translation family'': its structural action is concentrated in the one-dimensional accumulation of \(\log m\), without generating an additional continuous parameter axis.
\end{remark}
