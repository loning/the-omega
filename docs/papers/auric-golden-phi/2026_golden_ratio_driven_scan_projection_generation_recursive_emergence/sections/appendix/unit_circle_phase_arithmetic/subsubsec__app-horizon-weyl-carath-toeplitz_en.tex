\subsubsection{Horizon Weyl--Carath\'eodory Positivity and the Toeplitz Hierarchy: PSD Auditable Criterion for the Logarithmic Derivative}\label{subsubsec:app-horizon-weyl-carath-toeplitz}
Building upon the disk completion interface of Appendix \ref{subsec:unit-disk-rh-equivalence}, this subsection provides a closed criterion equivalent to ``no zeros inside the disk'' but more closely aligned with certificate semantics: the RH is rewritten as Carath\'eodory positivity (positive real part) of the horizon logarithmic derivative, which is further equivalent to a Toeplitz positive semidefinite (PSD) hierarchy.
This hierarchy belongs to the standard form of the truncated moment problem, so each level can be regarded as an auditable semidefinite feasibility constraint.
Furthermore, within the subdisk \(|w|<1/3\), the functional equation allows the horizon function to be rewritten as an absolutely convergent Euler product/Dirichlet series, thereby providing an auditable patch ``away from the endpoint'': the arithmetic terms enter entirely into the domain of uniform absolute convergence, and the difficulty is strictly isolated to the analytic continuation and positivity preservation problem in the endpoint neighborhood.
\par
To avoid confusion with the Cayley transform \(\mathcal{C}\) (equation \eqref{eq:app-cayley-disk}), we use the script letter \(\mathscr{C}\) to denote Carath\'eodory class functions.

\paragraph{Horizon variable and the Weyl (logarithmic derivative) function}
Following the completed entire function \(\Xi_{\zeta}\) from equation \eqref{eq:app-xi-classical}, define
\[
E_{\zeta}(z):=\Xi_{\zeta}\!\left(\tfrac12+\mathrm{i}z\right),\qquad z\in\CC .
\]
Then \(E_{\zeta}\) is a real entire function (\(E_{\zeta}(\overline z)=\overline{E_{\zeta}(z)}\)), and the RH is equivalent to all zeros of \(E_{\zeta}\) lying on the real axis: if \(\rho=\beta+\mathrm{i}\gamma\) is a zero of \(\Xi_{\zeta}\), then setting
\[
z:=\gamma+\mathrm{i}\Bigl(\tfrac12-\beta\Bigr)
\]
yields \(E_{\zeta}(z)=0\), and \(\beta<\tfrac12 \iff \Im z>0\).

Define the horizon Weyl (logarithmic derivative) function
\[
m_{\zeta}(z):=-\frac{E_{\zeta}'(z)}{E_{\zeta}(z)}\qquad(z\in\mathbb{H}).
\]

\begin{theorem}[Herglotz equivalence of the horizon Weyl function]\label{thm:app-horizon-weyl-herglotz}
The RH holds if and only if \(m_{\zeta}\) is analytic on the upper half-plane \(\mathbb{H}\) and satisfies
\[
\Im m_{\zeta}(z)\ge 0,\qquad z\in\mathbb{H},
\]
i.e., \(m_{\zeta}\) is a Herglotz (Nevanlinna) function.
\end{theorem}

\begin{proof}
(\(\Rightarrow\)) If the RH holds, then all zeros of \(E_{\zeta}\) are real (counted with multiplicity as \(\{\gamma_n\}\subset\RR\)).
Therefore \(E_{\zeta}'/E_{\zeta}\) is analytic on \(\mathbb{H}\) and admits the partial fraction expansion
\[
\frac{E_{\zeta}'(z)}{E_{\zeta}(z)}=\sum_n\frac{1}{z-\gamma_n}+c,
\qquad c\in\RR,
\]
where the series is understood in the standard sense of convergence.
For any \(\gamma_n\in\RR\) and \(\Im z>0\), one has \(\Im\frac{1}{z-\gamma_n}<0\), whence
\(\Im m_{\zeta}(z)=-\Im\frac{E_{\zeta}'(z)}{E_{\zeta}(z)}\ge 0\).
\par
(\(\Leftarrow\)) If \(m_{\zeta}\) is Herglotz, then \(m_{\zeta}\) is analytic on \(\mathbb{H}\).
The poles of \(m_{\zeta}\) are precisely the zeros of \(E_{\zeta}\) in \(\mathbb{H}\); hence \(E_{\zeta}\) has no zeros in \(\mathbb{H}\).
By the real entirety of \(E_{\zeta}\), any non-real zero must occur in a conjugate pair, one of which necessarily lies in \(\mathbb{H}\), contradicting ``no zeros in \(\mathbb{H}\)''.
Therefore all zeros of \(E_{\zeta}\) lie on \(\RR\), which is equivalent to the RH. \qed
\end{proof}

\paragraph{Disk horizon endpoint and Carath\'eodory positivity}
Following the Cayley transform \(\mathcal{C}:\mathbb{H}\to\mathbb{D}\) and its inverse \(\mathcal{C}^{-1}\) (equation \eqref{eq:app-cayley-disk}), and the disk section \(s(w)=\tfrac12+\mathrm{i}\,\mathcal{C}^{-1}(w)\) (equation \eqref{eq:app-s-of-w}), define the horizon Carath\'eodory (logarithmic derivative) function
\begin{equation}\label{eq:app-horizon-carath-def}
\boxed{\;
\mathscr{C}_{\zeta}(w):=-\frac{\Xi_{\zeta}'(s(w))}{\Xi_{\zeta}(s(w))}\qquad(w\in\mathbb{D}).\;}
\end{equation}
Noting that \(E_{\zeta}'(z)=\mathrm{i}\,\Xi_{\zeta}'(\tfrac12+\mathrm{i}z)\), for \(w=\mathcal{C}(z)\) (\(z\in\mathbb{H}\)) one obtains the strict identity
\begin{equation}\label{eq:app-horizon-m-c-relation}
\boxed{\;
\mathscr{C}_{\zeta}(w)=-\mathrm{i}\,m_{\zeta}\!\bigl(\mathcal{C}^{-1}(w)\bigr),\qquad
m_{\zeta}(z)=\mathrm{i}\,\mathscr{C}_{\zeta}\!\bigl(\mathcal{C}(z)\bigr).\;}
\end{equation}

\begin{corollary}[Disk positivity form of the RH]\label{cor:app-horizon-carath-rh}
The RH holds if and only if \(\mathscr{C}_{\zeta}\) is analytic on \(\mathbb{D}\) and satisfies
\[
\Re \mathscr{C}_{\zeta}(w)\ge 0,\qquad w\in\mathbb{D},
\]
i.e., \(\mathscr{C}_{\zeta}\) belongs to the Carath\'eodory class (positive real part functions).
\end{corollary}

\begin{proof}
By \eqref{eq:app-horizon-m-c-relation}, for \(z\in\mathbb{H}\) one has
\(\Re \mathscr{C}_{\zeta}(\mathcal{C}(z))=\Im m_{\zeta}(z)\).
Therefore \(\Im m_{\zeta}\ge 0\) is equivalent to \(\Re\mathscr{C}_{\zeta}\ge 0\);
combining with Theorem \ref{thm:app-horizon-weyl-herglotz} yields the conclusion.
The analyticity condition is also consistent: if \(\Xi_{\zeta}\) has a zero in \(\mathbb{H}\), then \(m_{\zeta}\) has a pole at that point; equivalently, \(\mathscr{C}_{\zeta}\) has a pole at its disk image, making it impossible to belong to the Carath\'eodory class. \qed
\end{proof}

\paragraph{Toeplitz PSD hierarchy: Carath\'eodory \texorpdfstring{$\Leftrightarrow$}{<->} PSD moment sequence}
Let \(f\) be an analytic function on \(\mathbb{D}\) satisfying \(\Re f(w)\ge 0\).
The Carath\'eodory/Herglotz representation (see, e.g., \cite{Duren1970TheoryHp,Garnett2007BoundedAnalyticFunctions,Simon2005OPUC1}) yields: there exist a finite positive Borel measure \(\mu\) on the unit circle \(\TT\) (or equivalently on \([0,2\pi)\)) and a real constant \(\beta\) such that
\begin{equation}\label{eq:app-horizon-herglotz-disk}
f(w)=\mathrm{i}\beta+\int_{\TT}\frac{\xi+w}{\xi-w}\,\dd\mu(\xi).
\end{equation}
When \(f(\overline w)=\overline{f(w)}\), the Maclaurin coefficients satisfy conjugate symmetry; and one may adopt the normalized power series form
\begin{equation}\label{eq:app-horizon-carath-series}
f(w)=c_0+2\sum_{n\ge 1}c_n w^n,\qquad c_{-n}:=\overline{c_n},
\end{equation}
where \(c_n\) are the Fourier moments of \(\mu\): \(c_n=\int_{\TT}\xi^{-n}\dd\mu(\xi)\).

\paragraph{Explicit spectral measure under the RH: atomization via the Cayley image of zeros}
Corollary \ref{cor:app-horizon-carath-rh} shows that when the RH holds, \(\mathscr{C}_{\zeta}\) belongs to the Carath\'eodory class, so the spectral measure \(\mu\) (and the moments \(\{c_n\}\)) in the representation \eqref{eq:app-horizon-herglotz-disk} are in principle uniquely determined by \(\mathscr{C}_{\zeta}\).
The following theorem provides a closed-form atomization normal form for this spectral measure under the RH: it directly transcribes the zero set into a purely atomic spectral object on the unit circle and simultaneously gives an absolutely convergent summation representation for the moment sequence.

\begin{theorem}[Atomization formula for the horizon spectral measure under the RH]\label{thm:app-horizon-spectral-measure-atomic}
Assume the RH holds.
Let \(E_{\zeta}(z)=\Xi_{\zeta}(\tfrac12+\mathrm{i}z)\), and let \(\Gamma_+\subset(0,\infty)\) be the positive zero multiset of \(E_{\zeta}\) (counted with multiplicity).
For each \(\gamma\in\Gamma_+\), denote its multiplicity by \(\mathrm{mult}(\gamma)\), and define
\[
\xi_{\gamma}:=\mathcal{C}(\gamma)\in\TT,
\qquad
\xi_{-\gamma}=\mathcal{C}(-\gamma)=\xi_{\gamma}^{-1}.
\]
Then in the Herglotz--Carath\'eodory representation \eqref{eq:app-horizon-herglotz-disk} of \(\mathscr{C}_{\zeta}\), one may take \(\beta=0\) and a purely atomic measure \(\mu_{\zeta}\) such that
\begin{equation}\label{eq:app-horizon-mu-atomic}
\boxed{\;
\mu_{\zeta}
=
\sum_{\gamma\in\Gamma_+}\frac{\mathrm{mult}(\gamma)}{1+\gamma^2}\Bigl(\delta_{\xi_{\gamma}}+\delta_{\xi_{\gamma}^{-1}}\Bigr).
\;}
\end{equation}
Under the normalized power series \eqref{eq:app-horizon-carath-series}, the Fourier moments (horizon multipole moments) satisfy the absolutely convergent summation formula: for any \(n\ge 0\),
\begin{equation}\label{eq:app-horizon-moment-sum}
\boxed{\;
c_n=\int_{\TT}\xi^{-n}\,\dd\mu_{\zeta}(\xi)
=
\sum_{\gamma\in\Gamma_+}\frac{\mathrm{mult}(\gamma)}{1+\gamma^2}\bigl(\xi_{\gamma}^{-n}+\xi_{\gamma}^{n}\bigr)
=
2\sum_{\gamma\in\Gamma_+}\frac{\mathrm{mult}(\gamma)}{1+\gamma^2}\cos\!\bigl(n\theta_{\gamma}\bigr),
\qquad \xi_{\gamma}=e^{\mathrm{i}\theta_{\gamma}}.
\;}
\end{equation}
\end{theorem}

\begin{proof}
From \(\Xi_{\zeta}(s)=\Xi_{\zeta}(1-s)\) one obtains \(E_{\zeta}(z)=E_{\zeta}(-z)\), so \(E_{\zeta}\) is an even function, whence \((E_{\zeta}'/E_{\zeta})(z)\) is an odd function.
When the RH holds, all zeros of \(E_{\zeta}\) are real and occur in pairs symmetric about the origin; therefore (counted with multiplicity) one may take a symmetric summation so that
\begin{equation}\label{eq:app-horizon-Elog-odd-pf}
\frac{E_{\zeta}'(z)}{E_{\zeta}(z)}
=
\sum_{\gamma\in\Gamma_+}\mathrm{mult}(\gamma)\!\left(\frac{1}{z-\gamma}+\frac{1}{z+\gamma}\right),
\qquad z\in\mathbb{H},
\end{equation}
where the series is understood in the standard partial fraction sense (consistent with the expansion in Theorem \ref{thm:app-horizon-weyl-herglotz}, with the constant term fixed to zero by the odd symmetry).
Thus
\[
m_{\zeta}(z):=-\frac{E_{\zeta}'(z)}{E_{\zeta}(z)}
=
\sum_{\gamma\in\Gamma_+}\mathrm{mult}(\gamma)\!\left(\frac{1}{\gamma-z}+\frac{1}{-\gamma-z}\right).
\]

Let \(t\in\RR\), \(z\in\mathbb{H}\), and set \(\xi=\mathcal{C}(t)\), \(w=\mathcal{C}(z)\).
From \eqref{eq:app-cayley-disk}, algebraic simplification yields the kernel identity
\begin{equation}\label{eq:app-horizon-cayley-kernel-identity}
\frac{\xi+w}{\xi-w}
=-\mathrm{i}\,\frac{1+t^{2}}{t-z}+\mathrm{i}t,
\end{equation}
equivalently,
\begin{equation}\label{eq:app-horizon-cayley-kernel-solve}
\frac{1}{t-z}
=\frac{\mathrm{i}}{1+t^{2}}\frac{\xi+w}{\xi-w}+\frac{t}{1+t^{2}}.
\end{equation}
Substituting \eqref{eq:app-horizon-cayley-kernel-solve} with \(t=\gamma\) and \(t=-\gamma\) and adding, the constant terms \(\pm\gamma/(1+\gamma^2)\) cancel, yielding
\[
\frac{1}{\gamma-z}+\frac{1}{-\gamma-z}
=\frac{\mathrm{i}}{1+\gamma^{2}}
\left(
\frac{\xi_{\gamma}+w}{\xi_{\gamma}-w}
+
\frac{\xi_{\gamma}^{-1}+w}{\xi_{\gamma}^{-1}-w}
\right).
\]
Substituting back into the expansion of \(m_{\zeta}\) and using the relation \eqref{eq:app-horizon-m-c-relation} (\(\mathscr{C}_{\zeta}(w)=-\mathrm{i}\,m_{\zeta}(\mathcal{C}^{-1}(w))\)), one obtains
\[
\mathscr{C}_{\zeta}(w)
=
\sum_{\gamma\in\Gamma_+}\frac{\mathrm{mult}(\gamma)}{1+\gamma^2}
\left(
\frac{\xi_{\gamma}+w}{\xi_{\gamma}-w}
+
\frac{\xi_{\gamma}^{-1}+w}{\xi_{\gamma}^{-1}-w}
\right),
\qquad w\in\mathbb{D}.
\]
Comparing with the general representation \eqref{eq:app-horizon-herglotz-disk}, one identifies \(\beta=0\) and the spectral measure \(\mu_{\zeta}\) as \eqref{eq:app-horizon-mu-atomic}.
The moment formula \eqref{eq:app-horizon-moment-sum} follows directly from \(c_n=\int_{\TT}\xi^{-n}\dd\mu_{\zeta}(\xi)\) and \(|\xi_{\gamma}|=1\); absolute convergence follows from \(\sum_{\gamma\in\Gamma_+}\mathrm{mult}(\gamma)/(1+\gamma^2)<\infty\).
\qed
\end{proof}

\begin{corollary}[Measure-level realization of Ramanujan self-dual symmetry]\label{cor:app-horizon-ramanujan-selfdual-measure}
Under the representation of Theorem \ref{thm:app-horizon-spectral-measure-atomic}, \(\mu_{\zeta}\) satisfies the strict self-dual symmetry
\[
\mu_{\zeta}\circ(\xi\mapsto \xi^{-1})^{-1}=\mu_{\zeta},
\]
i.e., spectral points occur in pairs \((\xi,\xi^{-1})\) with equal weights; consequently \(\{c_n\}_{n\ge 0}\) is a real sequence satisfying \(c_{-n}=c_n\).
\end{corollary}

\paragraph{Spectral measure of the comoving translation family (proof of the main text tomography theorem)}\label{par:app-horizon-comoving-spectral-measure}
The main text subsection \ref{subsubsec:xi-horizon-spectral-measure-tomography} introduced the comoving translation family
\(
\mathscr{C}_{\zeta,x_0}(w):=-\mathrm{i}\,m_{\zeta}\!\bigl(x_0+\mathcal{C}^{-1}(w)\bigr)
\)
to realize tomographic scanning of the horizon moment sequence.
Here we provide the technical derivation of its closed-form atomization and the ``weighted explicit formula'' interface.

\begin{proof}[Proof of Theorem \ref{thm:xi-horizon-comoving-spectral-measure}]
Fix \(x_0\in\RR\) and define the translated family in the upper half-plane
\(
m_{\zeta,x_0}(z):=m_{\zeta}(z+x_0)
\).
By Theorem \ref{thm:app-horizon-weyl-herglotz}, under the RH \(m_{\zeta}\) is a Herglotz function (\(\Im m_{\zeta}\ge 0\) on \(\mathbb{H}\)), and the translation \(z\mapsto z+x_0\) is a conformal automorphism of \(\mathbb{H}\), so \(m_{\zeta,x_0}\) remains Herglotz.
Compactifying via the Cayley transform to the disk and multiplying by \(-\mathrm{i}\) yields
\[
\mathscr{C}_{\zeta,x_0}(w):=-\mathrm{i}\,m_{\zeta}\!\bigl(x_0+\mathcal{C}^{-1}(w)\bigr),
\qquad w\in\mathbb{D},
\]
whence \(\Re \mathscr{C}_{\zeta,x_0}(w)\ge 0\) on \(\mathbb{D}\), i.e., \(\mathscr{C}_{\zeta,x_0}\) is a Carath\'eodory function.
\par
When the RH holds, all zeros of \(E_{\zeta}\) are real and symmetric about the origin.
Using \eqref{eq:app-horizon-Elog-odd-pf} (counted with multiplicity) and replacing \(z\) by \(z+x_0\), for \(z\in\mathbb{H}\) one obtains
\[
m_{\zeta,x_0}(z)
=
\sum_{\gamma\in\Gamma_+}\mathrm{mult}(\gamma)\!\left(\frac{1}{\gamma-(z+x_0)}+\frac{1}{-\gamma-(z+x_0)}\right).
\]
Writing both terms in the form of the Cauchy kernel \(\frac{1}{t-z}\): let
\(
t_{\pm}:=\pm\gamma-x_0\in\RR
\), then the above becomes
\[
m_{\zeta,x_0}(z)
=
\sum_{\gamma\in\Gamma_+}\mathrm{mult}(\gamma)\!\left(\frac{1}{t_+-z}+\frac{1}{t_--z}\right).
\]
For each real parameter \(t\in\RR\), set \(\xi:=\mathcal{C}(t)\in\TT\) and \(w:=\mathcal{C}(z)\in\mathbb{D}\).
By the kernel identity \eqref{eq:app-horizon-cayley-kernel-solve},
\[
\frac{1}{t-z}
=\frac{\mathrm{i}}{1+t^{2}}\frac{\xi+w}{\xi-w}+\frac{t}{1+t^{2}}.
\]
Substituting term by term and multiplying by \(-\mathrm{i}\) yields
\[
\mathscr{C}_{\zeta,x_0}(w)
=\mathrm{i}\beta(x_0)
+\sum_{\gamma\in\Gamma_+}\mathrm{mult}(\gamma)
\left(
\frac{1}{1+t_+^2}\frac{\xi_{t_+}+w}{\xi_{t_+}-w}
\;+\;
\frac{1}{1+t_-^2}\frac{\xi_{t_-}+w}{\xi_{t_-}-w}
\right).
\]
Here \(\xi_{t_\pm}:=\mathcal{C}(t_\pm)\in\TT\), and \(\beta(x_0)\in\RR\) is the real constant in the Carath\'eodory/Herglotz representation (absorbing all \(t/(1+t^2)\)-type constant contributions); its specific value does not affect the moments or PSD certificates.
Rewriting the summation index from \(\Gamma_+\) to the full zero multiset \(\Gamma:=\Gamma_+\cup(-\Gamma_+)\subset\RR\) (counted with multiplicity), and denoting
\(
\xi_{\gamma,x_0}:=\mathcal{C}(\gamma-x_0)
\), one identifies the purely atomic measure of the Carath\'eodory representation as
\[
\mu_{\zeta,x_0}
=
\sum_{\gamma\in\Gamma}\frac{\mathrm{mult}(\gamma)}{1+(\gamma-x_0)^2}\,\delta_{\xi_{\gamma,x_0}},
\]
yielding the main text equations \eqref{eq:xi-horizon-mu-comoving} and the moment identity \eqref{eq:xi-horizon-moment-comoving} (following directly from \(c_n(x_0)=\int_{\TT}\xi^{-n}\dd\mu_{\zeta,x_0}(\xi)\)).
\par
Finally, we prove \eqref{eq:xi-horizon-c0-im-m}.
Since \(\mathcal{C}^{-1}(0)=\mathrm{i}\) (equation \eqref{eq:app-cayley-disk}), one has
\(
\mathscr{C}_{\zeta,x_0}(0)=-\mathrm{i}\,m_{\zeta}(x_0+\mathrm{i})
\).
On the other hand, for any Carath\'eodory representation \(f(w)=\mathrm{i}\beta+\int_{\TT}\frac{\xi+w}{\xi-w}\dd\mu(\xi)\), one has
\(
\Re f(0)=\mu(\TT)
\).
Taking \(f=\mathscr{C}_{\zeta,x_0}\) and \(\mu=\mu_{\zeta,x_0}\) yields
\[
c_0(x_0)=\mu_{\zeta,x_0}(\TT)=\Re \mathscr{C}_{\zeta,x_0}(0)=\Im m_{\zeta}(x_0+\mathrm{i}).
\]
\qed
\end{proof}

\begin{remark}[Comoving Euler product patch and phase factor for \texorpdfstring{$|w|<1/3$}{|w|<1/3}]\label{rem:app-horizon-euler-patch-comoving}
Following \eqref{eq:app-horizon-fe-rewrite} and \eqref{eq:app-horizon-sigma-def}, note that
\(
1-(s(w)+\mathrm{i}x_0)=\sigma(w)-\mathrm{i}x_0
\),
whence
\[
\mathscr{C}_{\zeta,x_0}(w)
=-\frac{\Xi_{\zeta}'(s(w)+\mathrm{i}x_0)}{\Xi_{\zeta}(s(w)+\mathrm{i}x_0)}
=\frac{\Xi_{\zeta}'(\sigma(w)-\mathrm{i}x_0)}{\Xi_{\zeta}(\sigma(w)-\mathrm{i}x_0)}.
\]
When \(|w|<1/3\), one still has \(\Re\sigma(w)>1\) (Theorem \ref{thm:app-horizon-euler-patch}), so within this subdisk one obtains a uniformly absolutely convergent expansion of the same form as \eqref{eq:app-horizon-euler-patch}, with \(s\) replaced by \(\sigma(w)-\mathrm{i}x_0\):
\[
\mathscr{C}_{\zeta,x_0}(w)
=\frac{1}{\sigma(w)-\mathrm{i}x_0}+\frac{1}{\sigma(w)-1-\mathrm{i}x_0}
-\frac12\log\pi+\frac12\,\psi\!\left(\frac{\sigma(w)-\mathrm{i}x_0}{2}\right)
-\sum_{n\ge2}\frac{\Lambda(n)}{n^{\sigma(w)-\mathrm{i}x_0}}.
\]
The Dirichlet term equals \(-\sum_{n\ge2}\Lambda(n)\,n^{\mathrm{i}x_0}n^{-\sigma(w)}\), i.e., the phase factor \(n^{\mathrm{i}x_0}\) is structurally introduced, while absolute convergence and the validity of term-by-term differentiation/coefficient extraction are unaffected by \(x_0\) (since \(\Re(\sigma(w)-\mathrm{i}x_0)=\Re\sigma(w)\)).
This provides the auditable arithmetic interface for the ``weighted explicit formula family'' stated in the main text Corollary \ref{cor:xi-horizon-weighted-explicit-formula}.
\end{remark}

\begin{theorem}[First-order asymptotics of the endpoint \texorpdfstring{$\xi=-1$}{xi=-1} angular window mass]\label{thm:app-horizon-endpoint-window-mass}
Assume the RH holds and let \(\mu_{\zeta}\) be the horizon spectral measure given by Theorem \ref{thm:app-horizon-spectral-measure-atomic}.
For \(0<\varepsilon<\pi\), define the endpoint angular window
\[
I_{\varepsilon}:=\Bigl\{\xi=e^{\mathrm{i}\theta}\in\TT:\ |\theta-\pi|<\varepsilon\Bigr\},
\qquad \theta\in[0,2\pi).
\]
Then as \(\varepsilon\downarrow 0\),
\[
\mu_{\zeta}(I_{\varepsilon})
=
\frac{\varepsilon}{2\pi}\left(\log\frac{1}{\pi\varepsilon}+1\right)
+o(\varepsilon).
\]
In particular, \(\mu_{\zeta}(I_{\varepsilon})\sim \frac{\varepsilon}{2\pi}\log\frac{1}{\varepsilon}\).
\end{theorem}

\begin{proof}
By the atomization formula \eqref{eq:app-horizon-mu-atomic} of Theorem \ref{thm:app-horizon-spectral-measure-atomic},
\[
\mu_{\zeta}(I_{\varepsilon})
=
2\sum_{\substack{\gamma\in\Gamma_+\\ \xi_{\gamma}\in I_{\varepsilon}}}
\frac{\mathrm{mult}(\gamma)}{1+\gamma^2}.
\]
For \(\gamma>0\), write \(\xi_{\gamma}=e^{\mathrm{i}\theta_{\gamma}}\) (\(\theta_{\gamma}\in(0,\pi)\)).
Since \(\theta_{\gamma}=2\arctan(\gamma)\) (obtained by simplifying \eqref{eq:app-cayley-disk} with \(w=e^{\mathrm{i}\theta}\)),
\(\xi_{\gamma}\in I_{\varepsilon}\iff \theta_{\gamma}>\pi-\varepsilon\iff \gamma\ge \cot(\varepsilon/2)\).
Set \(T_{\varepsilon}:=\cot(\varepsilon/2)\to\infty\) (\(\varepsilon\downarrow 0\)), and let \(N(T)\) be the positive zero counting function (with multiplicity). Then
\[
\mu_{\zeta}(I_{\varepsilon})
=2\int_{T_{\varepsilon}}^{\infty}\frac{1}{1+t^2}\,\dd N(t).
\]
By the leading term of the Riemann--von Mangoldt formula (Remark \ref{rem:app-rvm-endpoint-constant}),
\(
N(t)=M(t)+O(\log t)
\)
with \(M'(t)=\frac{1}{2\pi}\log(\frac{t}{2\pi})\).
Therefore, applying integration by parts to the error term and using the decay of \((1+t^2)^{-1}\) and its derivative,
\[
\mu_{\zeta}(I_{\varepsilon})
=2\int_{T_{\varepsilon}}^{\infty}\frac{1}{1+t^2}M'(t)\,\dd t
+O\!\left(\frac{\log T_{\varepsilon}}{T_{\varepsilon}^2}\right).
\]
For the leading integral, using \( (1+t^2)^{-1}=t^{-2}+O(t^{-4}) \) as \(t\to\infty\),
\[
\int_{T_{\varepsilon}}^{\infty}\frac{1}{1+t^2}M'(t)\,\dd t
=\frac{1}{2\pi}\int_{T_{\varepsilon}}^{\infty}\frac{\log(\frac{t}{2\pi})}{t^2}\,\dd t
+O\!\left(\frac{\log T_{\varepsilon}}{T_{\varepsilon}^3}\right)
=\frac{1}{2\pi}\frac{\log(\frac{T_{\varepsilon}}{2\pi})+1}{T_{\varepsilon}}
+O\!\left(\frac{\log T_{\varepsilon}}{T_{\varepsilon}^3}\right).
\]
Thus
\[
\mu_{\zeta}(I_{\varepsilon})
=\frac{1}{\pi}\frac{\log(\frac{T_{\varepsilon}}{2\pi})+1}{T_{\varepsilon}}
+O\!\left(\frac{\log T_{\varepsilon}}{T_{\varepsilon}^2}\right).
\]
Finally, using \(T_{\varepsilon}^{-1}=\tan(\varepsilon/2)=\varepsilon/2+O(\varepsilon^{3})\) and
\(\log(\tfrac{T_{\varepsilon}}{2\pi})=\log\!\bigl(\tfrac{1}{2\pi\tan(\varepsilon/2)}\bigr)=\log(\tfrac{1}{\pi\varepsilon})+O(\varepsilon^{2})\), one obtains the stated asymptotic formula; the error term is \(o(\varepsilon)\). \qed
\end{proof}

By Remark \ref{rem:app-offcritical-radius-precision-potential},

\).
Therefore the leading-order scale \(\varepsilon\log(1/\varepsilon)\) of the endpoint angular window mass can be interpreted as: the visible endpoint layer mass resulting from the logarithmically weakly divergent \(T\log T\) density induced under phase compactification, after quadratic redshift suppression by the precision potential weight.
\end{remark}

\mathbf{T}_N:=(c_{j-k})_{0\le j,k\le N}.
\]
Then \(\mathbf{T}_N\succeq 0\), since \(\mathbf{T}_N\) is the Gram matrix of the vectors \(\{1,\xi,\dots,\xi^N\}\) in \(L^2(\mu)\).
Conversely, \(\{\mathbf{T}_N\succeq 0\}_{N\ge 0}\) constitutes precisely the full set of principal minor positivity conditions for the truncated moment problem, guaranteeing the existence of a positive measure \(\mu\) such that \eqref{eq:app-horizon-herglotz-disk} holds, and hence \(f\) is Carath\'eodory.

\begin{theorem}[Equivalence of the Toeplitz--PSD hierarchy and the RH]\label{thm:app-horizon-toeplitz-lmi}
Let \(\mathscr{C}_{\zeta}\) be defined as in \eqref{eq:app-horizon-carath-def}, and write its power series at \(w=0\) as
\[
\mathscr{C}_{\zeta}(w)=c_0+2\sum_{n\ge 1}c_n w^n,\qquad c_{-n}:=\overline{c_n}.
\]
Then the following statements are equivalent:\vspace{-2pt}
\begin{enumerate}
  \item The RH holds;
  \item \(\mathscr{C}_{\zeta}\) belongs to the Carath\'eodory class (Corollary \ref{cor:app-horizon-carath-rh});
  \item For all \(N\ge 0\), the Toeplitz principal submatrix
  \[
  \boxed{\;\mathbf{T}_N=(c_{j-k})_{0\le j,k\le N}\succeq 0\;}
  \]
  holds.
\end{enumerate}
In other words, the RH is equivalent to a family of PSD constraints that can be verified level by level, each being a semidefinite feasibility problem.
\end{theorem}

\begin{proof}
\((1)\Leftrightarrow(2)\) follows from Corollary \ref{cor:app-horizon-carath-rh}.
\((2)\Rightarrow(3)\) follows immediately from the Fourier moment expansion of \eqref{eq:app-horizon-herglotz-disk} and the Gram matrix interpretation.
\((3)\Rightarrow(2)\) belongs to the standard completeness of the Carath\'eodory--Fej\'er/truncated moment problem: if all Toeplitz principal submatrices are PSD, then there exists a positive measure \(\mu\) such that the moment sequence \(\{c_n\}\) is realized as the Fourier moments of \(\mu\), whence \(\Re f\ge 0\) by \eqref{eq:app-horizon-herglotz-disk} (see \cite{Duren1970TheoryHp,Simon2005OPUC1}). \qed
\end{proof}

\begin{proposition}[Toeplitz--PSD \texorpdfstring{$\ZZ/2$}{Z/2} sector decomposition (even/odd blocks under the reversal involution)]\label{prop:app-horizon-toeplitz-z2-splitting}
Under the conjugate symmetry convention of \eqref{eq:app-horizon-carath-series}, further assume the coefficients satisfy the real structure
\[
c_n\in\RR\qquad(n\in\ZZ),
\]
so that \(c_{-n}=c_n\) and each \(\mathbf{T}_N\) is a real symmetric Toeplitz matrix.
Let \(\mathbf{J}_N\in\RR^{(N+1)\times(N+1)}\) be the reversal matrix (\( \mathbf{J}_N e_k=e_{N-k} \)), and let
\[
E_N^{\pm}:=\ker(\mathbf{J}_N\mp I).
\]
Then \( \RR^{N+1}=E_N^{+}\oplus E_N^{-} \) is an orthogonal direct sum, and \(\mathbf{T}_N\) commutes with \(\mathbf{J}_N\), so that \(E_N^{\pm}\) are both invariant subspaces of \(\mathbf{T}_N\).
Choosing an orthogonal basis adapted to the decomposition \(E_N^{+}\oplus E_N^{-}\), there exists an orthogonal matrix \(Q_N\) such that
\[
Q_N^{\mathsf T}\mathbf{T}_N Q_N
=
\begin{pmatrix}
\mathbf{T}_N^{+} & 0\\
0 & \mathbf{T}_N^{-}
\end{pmatrix},
\]
where \(\mathbf{T}_N^{\pm}\) are the restriction matrices of \(\mathbf{T}_N\) to \(E_N^{\pm}\) (mutually orthogonally conjugate under different orthogonal bases).
Therefore
\[
\mathbf{T}_N\succeq 0
\quad\Longleftrightarrow\quad
\mathbf{T}_N^{+}\succeq 0\ \text{and}\ \mathbf{T}_N^{-}\succeq 0.
\]
\end{proposition}

\begin{proof}
For any \(0\le j,k\le N\), from \(\mathbf{T}_N=(c_{j-k})\) and \(\mathbf{J}_N e_k=e_{N-k}\) one obtains
\[
(\mathbf{J}_N\mathbf{T}_N)_{jk}=(\mathbf{T}_N)_{N-j,k}=c_{(N-j)-k},
\qquad
(\mathbf{T}_N\mathbf{J}_N)_{jk}=(\mathbf{T}_N)_{j,N-k}=c_{j-(N-k)}.
\]
Noting that \(j-(N-k)=-( (N-j)-k)\), and using \(c_{-n}=c_n\), one obtains \((\mathbf{J}_N\mathbf{T}_N)_{jk}=(\mathbf{T}_N\mathbf{J}_N)_{jk}\), so \(\mathbf{J}_N\mathbf{T}_N=\mathbf{T}_N\mathbf{J}_N\).
Since \(\mathbf{J}_N\) is a symmetric orthogonal involution (\(\mathbf{J}_N^{\mathsf T}=\mathbf{J}_N\), \(\mathbf{J}_N^2=I\)), its \(\pm 1\) eigenspaces \(E_N^{\pm}\) form an orthogonal direct sum of \(\RR^{N+1}\); by commutativity, \(\mathbf{T}_N\) preserves \(E_N^{\pm}\), so choosing the columns of \(Q_N\) as an adapted orthogonal basis for \(E_N^{+}\) and \(E_N^{-}\) yields the block diagonal form.
\par
Finally, let \(x=x_{+}+x_{-}\) be the corresponding decomposition (\(x_{\pm}\in E_N^{\pm}\)). By the symmetry of \(\mathbf{T}_N\) and commutativity,
\[
x_{+}^{\mathsf T}\mathbf{T}_N x_{-}
=
x_{+}^{\mathsf T}\mathbf{J}_N\mathbf{T}_N\mathbf{J}_N x_{-}
=
x_{+}^{\mathsf T}\mathbf{T}_N(-x_{-})
=
-\,x_{+}^{\mathsf T}\mathbf{T}_N x_{-},
\]
so the cross term vanishes, yielding the quadratic form splitting
\(
x^{\mathsf T}\mathbf{T}_N x=x_{+}^{\mathsf T}\mathbf{T}_N x_{+}+x_{-}^{\mathsf T}\mathbf{T}_N x_{-}
\).
Thus \(\mathbf{T}_N\succeq 0\) if and only if the restricted quadratic forms on \(E_N^{+}\) and \(E_N^{-}\) are each PSD; equivalently, \(\mathbf{T}_N^{\pm}\succeq 0\). \qed
\end{proof}

\begin{corollary}[Even/odd certificate splitting and failure sector localization of the Toeplitz--PSD hierarchy]\label{cor:app-horizon-toeplitz-z2-cert-split}
Under the real structure assumption of Proposition \ref{prop:app-horizon-toeplitz-z2-splitting}, the Toeplitz--PSD hierarchy
\(\{\mathbf{T}_N\succeq 0\}_{N\ge 0}\)
splits into two mutually independent PSD sub-hierarchies:
\[
\forall N\ge 0,\qquad
\mathbf{T}_N\succeq 0
\quad\Longleftrightarrow\quad
\mathbf{T}_N^{+}\succeq 0\ \text{and}\ \mathbf{T}_N^{-}\succeq 0.
\]
Therefore, if PSD failure occurs at some level \(N\), there exist a minimal \(N_\star\) and a sector sign \(\epsilon\in\{+,-\}\) such that \(\mathbf{T}_{N_\star}^{\epsilon}\) has a negative eigenvalue; the pair \((N_\star,\epsilon)\) provides an auditable failure localization of the Toeplitz certificate at a finite level.
\end{corollary}

\paragraph{Local encapsulation of coefficients: using only derivatives of \texorpdfstring{$\Xi_{\zeta}$}{Xi_zeta} at \texorpdfstring{$s=-\tfrac12$}{s=-1/2}}
Since \(s(0)=-\tfrac12\) and \(\Xi_{\zeta}(-\tfrac12)\neq 0\), one may define the Taylor coefficients of the logarithmic derivative at this point.
For notational brevity, let
\[
\ell_k:=\left.\frac{\dd^k}{\dd s^k}\log\Xi_{\zeta}(s)\right|_{s=-1/2}\in\RR
\qquad(k\ge 1).
\]
From \(\mathscr{C}_{\zeta}(w)=-\bigl(\log\Xi_{\zeta}\bigr)'\!\bigl(s(w)\bigr)\) and the identity
\begin{equation}\label{eq:app-horizon-s-of-w-shift}
s(w)=-\tfrac12+\frac{2w}{1+w}
\end{equation}
one can express the coefficients of \(\mathscr{C}_{\zeta}\) entirely and explicitly in terms of finitely many \(\ell_k\).

\begin{proposition}[Explicit expressions for the first few coefficients]\label{prop:app-horizon-coeff-local}
Under the normalization \(\mathscr{C}_{\zeta}(w)=c_0+2\sum_{n\ge 1}c_n w^n\),
\begin{equation}\label{eq:app-horizon-c0c3}
\begin{aligned}
c_0&=-\ell_1,\\
c_1&=-\ell_2,\\
c_2&=\ell_2-\ell_3,\\
c_3&=-\ell_2+2\ell_3-\tfrac23\ell_4.
\end{aligned}
\end{equation}
More generally, \(c_n\) is an explicit rational linear combination of \((\ell_2,\dots,\ell_{n+1})\) (obtained by substituting \(\frac{2w}{1+w}=\sum_{m\ge 1}2(-1)^{m-1}w^m\)).
\end{proposition}

\begin{proof}
From \eqref{eq:app-horizon-s-of-w-shift}, let \(u(w):=s(w)+\tfrac12=\frac{2w}{1+w}=2w-2w^2+2w^3+O(w^4)\).
Taylor expanding \( (\log\Xi_{\zeta})'(s) \) at \(s=-\tfrac12\):
\[
\bigl(\log\Xi_{\zeta}\bigr)'\!\left(-\tfrac12+u\right)
=\ell_1+\ell_2 u+\tfrac12\ell_3 u^2+\tfrac16\ell_4 u^3+O(u^4).
\]
Substituting \(u=u(w)\) and taking the negative sign yields the \(w\)-expansion of \(\mathscr{C}_{\zeta}(w)\);
comparing with \(\mathscr{C}_{\zeta}(w)=c_0+2c_1w+2c_2w^2+2c_3w^3+O(w^4)\) gives \eqref{eq:app-horizon-c0c3}.
The general form follows similarly from the uniqueness of composed power series. \qed
\end{proof}

\begin{remark}[Explicit necessary conditions from low-level PSD constraints]\label{rem:app-horizon-low-level-psd}
By Theorem \ref{thm:app-horizon-toeplitz-lmi}, RH \(\Rightarrow\) \(\mathbf{T}_N\succeq 0\) (for all \(N\)).
The first two levels respectively give
\[
\mathbf{T}_0=(c_0)\succeq 0,
\qquad
\mathbf{T}_1=\begin{pmatrix}c_0&c_1\\c_1&c_0\end{pmatrix}\succeq 0,
\]
i.e., \(c_0\ge 0\) and \(c_0^2-c_1^2\ge 0\).
Substituting \eqref{eq:app-horizon-c0c3} yields explicit inequalities in terms of \(\ell_1,\ell_2\); higher-order constraints can likewise be generated level by level from finitely many \(\ell_k\).
These constraints are not individually sufficient to imply the RH, but they provide a completely internal family of necessary conditions that can be verified level by level.
\end{remark}

\paragraph{\texorpdfstring{$|w|<1/3$}{|w|<1/3} Euler product auditable patch: isolating arithmetic terms to the domain of absolute convergence}
Set
\begin{equation}\label{eq:app-horizon-sigma-def}
\boxed{\;\sigma(w):=1-s(w)=\frac{3-w}{2(1+w)}\; }.
\end{equation}
The functional equation \(\Xi_{\zeta}(s)=\Xi_{\zeta}(1-s)\) yields the logarithmic derivative antisymmetry
\(
\frac{\Xi_{\zeta}'(s)}{\Xi_{\zeta}(s)}=-\frac{\Xi_{\zeta}'(1-s)}{\Xi_{\zeta}(1-s)}
\),
whence
\begin{equation}\label{eq:app-horizon-fe-rewrite}
\boxed{\;
\mathscr{C}_{\zeta}(w)
=-\frac{\Xi_{\zeta}'(s(w))}{\Xi_{\zeta}(s(w))}
=\frac{\Xi_{\zeta}'(\sigma(w))}{\Xi_{\zeta}(\sigma(w))}\; }.
\end{equation}

\begin{theorem}[Euler product patch for \texorpdfstring{$|w|<1/3$}{|w|<1/3} (absolutely convergent prime expansion)]\label{thm:app-horizon-euler-patch}
If \(|w|<\frac13\), then \(\Re\sigma(w)>1\).
Therefore within this subdisk, \(\mathscr{C}_{\zeta}\) admits the uniformly absolutely convergent representation
\begin{equation}\label{eq:app-horizon-euler-patch}
\boxed{\;
\mathscr{C}_{\zeta}(w)=\frac{1}{\sigma(w)}+\frac{1}{\sigma(w)-1}
-\frac12\log\pi+\frac12\,\psi\!\left(\frac{\sigma(w)}{2}\right)
-\sum_{n\ge2}\frac{\Lambda(n)}{n^{\sigma(w)}},\qquad \sigma(w)=1-s(w),\;}
\end{equation}
where \(\psi=\Gamma'/\Gamma\) is the digamma function, \(\Lambda\) is the von Mangoldt function; and the series converges uniformly absolutely on any closed subdisk \(|w|\le r<1/3\).
\end{theorem}

\begin{proof}
By \eqref{eq:app-horizon-sigma-def}, \(\sigma\) is a M\"obius transformation with real coefficients, so it maps the circle \(|w|=r\) to a circle symmetric about the real axis; its \(\Re\)-minimum is attained at the leftmost point of the image circle, which necessarily lies on the real axis.
Thus for fixed \(r\in(0,1)\),
\[
\min_{|w|=r}\Re\sigma(w)=\min\{\sigma(r),\sigma(-r)\}=\sigma(r)=\frac{3-r}{2(1+r)}.
\]
When \(r<1/3\), the right-hand side is strictly greater than \(1\), whence \(|w|<1/3\Rightarrow \Re\sigma(w)>1\).
\par
By \eqref{eq:app-horizon-fe-rewrite}, \(\mathscr{C}_{\zeta}(w)=\frac{\Xi_{\zeta}'(\sigma(w))}{\Xi_{\zeta}(\sigma(w))}\).
Taking the logarithmic derivative of \(\Xi_{\zeta}(s)=\frac12\,s(s-1)\pi^{-s/2}\Gamma(\tfrac{s}{2})\zeta(s)\) yields
\[
\frac{\Xi_{\zeta}'(s)}{\Xi_{\zeta}(s)}
=\frac{1}{s}+\frac{1}{s-1}-\frac12\log\pi+\frac12\,\psi\!\left(\frac{s}{2}\right)+\frac{\zeta'}{\zeta}(s).
\]
When \(\Re s>1\), one has the standard identity
\(
\frac{\zeta'}{\zeta}(s)=-\sum_{n\ge 2}\frac{\Lambda(n)}{n^s}
\),
with the right-hand side absolutely convergent.
Substituting \(s=\sigma(w)\) and combining gives \eqref{eq:app-horizon-euler-patch}.
Uniform absolute convergence follows from \(|w|\le r<1/3\Rightarrow \inf\Re\sigma(w)>1\) and the uniform absolute convergence of the Dirichlet series in the half-plane \(\Re s\ge 1+\varepsilon\). \qed
\end{proof}

\begin{corollary}[Prime moment representation of coefficients (auditable interface)]\label{cor:app-horizon-prime-moment-interface}
Within \(|w|<1/3\), the Taylor coefficients \(c_n\) of \(\mathscr{C}_{\zeta}\) can be obtained by differentiating \eqref{eq:app-horizon-euler-patch} term by term at \(w=0\).
The arithmetic contribution is entirely composed of convergent sums
\[
\sum_{n\ge2}\frac{\Lambda(n)(\log n)^k}{n^{3/2}}
\qquad(k\ge 0)
\]
and finitely many rational combinations (arising from \(\sigma(0)=3/2\) and the chain expansion of \(\sigma(w)=\frac{3-w}{2(1+w)}\)).
Therefore, the construction of each finite-level Toeplitz principal submatrix \(\mathbf{T}_N\) can be audited term by term within the domain of absolute convergence; the difficulty is structurally isolated to the endpoint continuation problem of ``extending the local patch positivity across the full disk while maintaining \(\Re\mathscr{C}_{\zeta}\ge 0\)''.
\end{corollary}

\begin{remark}[Horizon multipole moments and reflection positivity (alignment with ``hair'' semantics)]\label{rem:app-horizon-multipole-positivity-hair}
The measure \(\mu\) in the Herglotz representation \eqref{eq:app-horizon-herglotz-disk} can be viewed as a ``horizon spectral measure'':
its Fourier moments \(\{c_n\}\) provide a set of horizon multipole moments (also interpretable as ``visible hair coefficients''), and the Toeplitz PSD condition \(\mathbf{T}_N\succeq 0\) requires that this moment sequence must arise from some positive measure (i.e., from a genuine Hilbert space spectral object).
If there exists a zero inside the disk (equivalent to the failure of the RH), then \(\mathscr{C}_{\zeta}\) has a pole inside the disk and loses Carath\'eodory positivity; at the moment level, this necessarily forces some level \(\mathbf{T}_N\) to have a negative eigenvalue, thereby compressing the ``non-unitarity/reflection positivity breaking'' into a finite-dimensional auditable failure witness.
\end{remark}
\paragraph{Coefficient imprint of interior poles and the detection threshold of the Toeplitz--PSD hierarchy}
This paragraph writes the leading-order contribution of an ``interior pole'' to \(\{c_n\}\) in closed form, and uses this to derive a detection threshold compatible with endpoint compression: the threshold translates ``extreme resonance'' (off-slice information from the endpoint thin layer) into a scaling law for ``how high an order of moments/multipole moments is needed to stably trigger a finite-dimensional falsifier''.

\begin{proposition}[Coefficient imprint of an interior pole (principal part and Taylor leading term)]\label{prop:app-horizon-pole-imprint}
Let \(\rho=\beta+\mathrm{i}\gamma\) be a simple zero of \(\Xi_{\zeta}\) with \(\beta<\tfrac12\).
Set \(\delta:=\tfrac12-\beta>0\), \(z_\rho:=\gamma+\mathrm{i}\delta\in\mathbb{H}\), and let \(w_\rho:=\mathcal{C}(z_\rho)\in\mathbb{D}\) be its disk image (Cayley transform \(\mathcal{C}\) from equation \eqref{eq:app-cayley-disk}).
Then \(\mathscr{C}_{\zeta}\) has a simple pole at \(w=w_\rho\) and can be written as
\begin{equation}\label{eq:app-horizon-pole-principal-part}
\boxed{\;
\mathscr{C}_{\zeta}(w)
=-\frac{(1+w_\rho)^2}{2}\cdot\frac{1}{w-w_\rho}+H_\rho(w),\;}
\end{equation}
where \(H_\rho\) is analytic in \(\mathbb{D}\).
Expanding \(H_\rho\) at \(w=0\) in the same normalized power series
\[
H_\rho(w)=h_0+2\sum_{n\ge 1}h_n w^n,
\]
and using \(\mathscr{C}_{\zeta}(w)=c_0+2\sum_{n\ge 1}c_n w^n\) (equation \eqref{eq:app-horizon-carath-series}), the coefficients satisfy
\[
c_0=\frac{(1+w_\rho)^2}{2w_\rho}+h_0,
\qquad
c_n=\frac{(1+w_\rho)^2}{4w_\rho^{n+1}}+h_n\quad(n\ge 1).
\]
\end{proposition}

\begin{proof}
By definition \eqref{eq:app-horizon-carath-def}, \(\mathscr{C}_{\zeta}(w)=-\bigl(\log\Xi_{\zeta}\bigr)'\!\bigl(s(w)\bigr)\).
If \(\rho\) is a simple zero of \(\Xi_{\zeta}\), then \(\bigl(\log\Xi_{\zeta}\bigr)'(s)\) has a simple pole at \(s=\rho\) with residue \(1\), whence \(\mathscr{C}_{\zeta}\) has a simple pole at \(w=w_\rho\) with residue \(-1\).
Linearizing \(s(w)=s(w_\rho)+s'(w_\rho)(w-w_\rho)+\cdots\) at \(w_\rho\) via the chain rule yields the principal part
\[
\mathscr{C}_{\zeta}(w)
=-\frac{1}{s'(w_\rho)}\cdot\frac{1}{w-w_\rho}+O(1).
\]
From \eqref{eq:app-horizon-s-of-w-shift}, \(s'(w)=\frac{2}{(1+w)^2}\), so \(-1/s'(w_\rho)=-(1+w_\rho)^2/2\), giving \eqref{eq:app-horizon-pole-principal-part}.
\par
Expanding the pole term of \eqref{eq:app-horizon-pole-principal-part} as a geometric series at \(w=0\):
\[
-\frac{(1+w_\rho)^2}{2}\cdot\frac{1}{w-w_\rho}
=\frac{(1+w_\rho)^2}{2}\sum_{n\ge 0}\frac{w^n}{w_\rho^{n+1}},
\]
combining with \(H_\rho(w)=h_0+2\sum_{n\ge 1}h_n w^n\) and comparing coefficients with \(\mathscr{C}_{\zeta}(w)=c_0+2\sum_{n\ge 1}c_n w^n\) yields the stated conclusion. \qed
\end{proof}

\begin{theorem}[Resonance quality factor and the detection threshold of the Toeplitz--PSD hierarchy]\label{thm:app-horizon-toeplitz-detection-threshold}
Under the setting of Proposition \ref{prop:app-horizon-pole-imprint}, define the resonance quality factor
\begin{equation}\label{eq:app-horizon-Qrho-def}
\boxed{\;
Q_\rho:=\frac{1}{1-|w_\rho|^2}
=\frac{\gamma^2+(1+\delta)^2}{4\delta}\;}
\end{equation}
(the latter equality is from Corollary \ref{cor:app-offcritical-horocycle-busemann}).
Assume the remainder coefficients satisfy an auditable upper bound \(|h_n|\le H\) (for all \(n\ge 0\)).
\par
If one wishes to \emph{exclude} this interior pole solely through finite-level Toeplitz--PSD verification (equivalently: to require that \(\mathbf{T}_N\succeq 0\) still holds at some sufficiently large truncation order \(N\), thereby rejecting a ``finite-dimensional failure witness''), then the verification order must satisfy the lower bound
\begin{equation}\label{eq:app-horizon-toeplitz-detect-threshold}
\boxed{\;
N\ \ge\ C^{-1}\,Q_\rho\,\log\!\bigl(1+\delta Q_\rho\bigr)\;},
\end{equation}
where the constant \(C>0\) depends only on \(c_0\) and \(H\) (not on \(\gamma,\delta\)).
In particular, when \(|\gamma|\to\infty\) and \(\delta\) is bounded, \(\delta Q_\rho\asymp \gamma^2\), so the lower bound is equivalent to
\[
N\ \gtrsim\ \frac{\gamma^2}{\delta}\,\log(1+\gamma^2).
\]
\par
Reverse formulation: if for some \(N\) one has verified \(\mathbf{T}_0,\dots,\mathbf{T}_N\succeq 0\), then any potentially existing off-critical-line zero \(\rho=\beta+\mathrm{i}\gamma\) must satisfy
\[
\delta\ \le\ C\,\frac{\gamma^2\,\log(1+\gamma^2)}{N},
\]
so that at large heights \(|\gamma|\gg 1\), a fixed-order Toeplitz--PSD verification can only exclude off-line zeros whose offset does not enter the endpoint thin layer.
\end{theorem}

\begin{proof}
By Theorem \ref{thm:app-horizon-toeplitz-lmi}, \(\mathbf{T}_N\succeq 0\) implies that for any \(1\le n\le N\), the \(\{0,n\}\) principal submatrix
\(
\begin{psmallmatrix}
c_0 & c_n\\
c_{-n} & c_0
\end{psmallmatrix}
\)
is PSD.
Since \(c_{-n}=\overline{c_n}\), this condition is equivalent to
\begin{equation}\label{eq:app-horizon-2x2-necessary}
|c_n|\le c_0,\qquad 1\le n\le N.
\end{equation}
On the other hand, from Proposition \ref{prop:app-horizon-pole-imprint} and \(|h_n|\le H\), one obtains the lower bound for any \(n\ge 1\):
\begin{equation}\label{eq:app-horizon-cn-lowerbound}
|c_n|
\ge
\frac{|1+w_\rho|^2}{4\,|w_\rho|^{n+1}}-H.
\end{equation}
Therefore, if \(\mathbf{T}_N\succeq 0\) holds, then combining \eqref{eq:app-horizon-2x2-necessary}--\eqref{eq:app-horizon-cn-lowerbound} for \(n=N\) gives
\[
\frac{|1+w_\rho|^2}{4\,|w_\rho|^{N+1}}\le c_0+H.
\]
By the identity from Corollary \ref{cor:app-offcritical-horocycle-busemann},
\(
|1+w_\rho|^2=(1-|w_\rho|^2)/\delta
\)
and the definition \(Q_\rho=(1-|w_\rho|^2)^{-1}\), one gets \(|1+w_\rho|^2=(\delta Q_\rho)^{-1}\), whence
\begin{equation}\label{eq:app-horizon-wrho-growth-bound}
|w_\rho|^{-(N+1)}\le 4(c_0+H)\,\delta Q_\rho.
\end{equation}
Let \(r:=|w_\rho|^2=1-1/Q_\rho\in(0,1)\).
For \(r\in(0,1)\), the standard inequality
\(
1-r\le -\log r \le \frac{1-r}{r}
\)
holds, so when \(Q_\rho>1\),
\begin{equation}\label{eq:app-horizon-logw-sandwich}
\frac{1}{2Q_\rho}\ \le\ -\log|w_\rho|\ \le\ \frac{1}{2(Q_\rho-1)}.
\end{equation}
Taking the logarithm of \eqref{eq:app-horizon-wrho-growth-bound} and using the right-hand estimate of \eqref{eq:app-horizon-logw-sandwich} (which makes the left side grow most slowly, thereby yielding the necessary ``detection threshold'' lower bound), one obtains a constant \(C>0\) depending only on \(c_0,H\) such that
\[
N+1
\ge
C^{-1}\,Q_\rho\,\log\!\bigl(1+\delta Q_\rho\bigr),
\]
which gives \eqref{eq:app-horizon-toeplitz-detect-threshold} (absorbing \(\log(4(c_0+H)\delta Q_\rho)\) into \(\log(1+\delta Q_\rho)\) and the constant).
\par
Finally, the closed form \eqref{eq:app-horizon-Qrho-def} and \(\delta Q_\rho\asymp \gamma^2\) (\(|\gamma|\to\infty\), \(\delta\) bounded) yield the stated asymptotic form; the reverse formulation follows by substituting \(Q_\rho\asymp \gamma^2/\delta\) back into \eqref{eq:app-horizon-toeplitz-detect-threshold} and rearranging. \qed
\end{proof}

\begin{remark}[Endpoint focusing of the failure witness (angular resolution aperture)]\label{rem:app-horizon-toeplitz-witness-endpoint-focus}
The finite-level falsifier of the Toeplitz--PSD hierarchy is equivalent to finding a direction in the truncated trigonometric polynomial/OPUC space of degree \(\le N\) that makes the quadratic form negative (Corollary \ref{cor:app-horizon-toeplitz-z2-cert-split} gives the even/odd sector localization).
In the geometric semantics of Proposition \ref{prop:app-horizon-pole-imprint}, an interior pole corresponds to thin-layer compression near the endpoint \(w=-1\): when \(|\gamma|\) is large, \(\arg w_\rho\approx\pi\) and \(\delta Q_\rho=|1+w_\rho|^{-2}\asymp \gamma^2\) (Corollary \ref{cor:app-offcritical-horocycle-busemann}).
Therefore, if the falsifier first appears at order \(N\), its ``angular resolution'' is naturally constrained to order \(N^{-1}\) and tends to focus within the endpoint angular window near \(\theta\approx\pi\); this is consistent with the endpoint window mass asymptotics of Theorem \ref{thm:app-horizon-endpoint-window-mass} and the visibility mechanism of endpoint compactification.
\end{remark}
