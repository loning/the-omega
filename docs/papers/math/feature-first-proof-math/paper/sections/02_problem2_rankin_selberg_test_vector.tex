\section{Problem 2: A Nonvanishing Test Vector for the Local Rankin--Selberg Integral}

\begin{proposition}
Let $F$ be a non-archimedean local field, $\psi$ a nontrivial additive character of conductor $\mathfrak o$, and let
\[
\Pi \text{ (resp. }\pi\text{)}
\]
be a generic irreducible admissible representation of $\mathrm{GL}_{n+1}(F)$ (resp. $\mathrm{GL}_n(F)$), realized in $\mathcal W(\Pi,\psi^{-1})$ (resp. $\mathcal W(\pi,\psi)$). Let $\mathfrak q$ be the conductor ideal of $\pi$, choose $Q\in F^\times$ generating $\mathfrak q^{-1}$, and set
\[
u_Q=I_{n+1}+Q E_{n,n+1}.
\]
Then there exist $W\in\mathcal W(\Pi,\psi^{-1})$ and $V\in\mathcal W(\pi,\psi)$ such that
\[
I(s;W,V):=\int_{N_n\backslash \mathrm{GL}_n(F)}
W(\operatorname{diag}(g,1)u_Q)\,V(g)\,|\det g|^{s-\frac12}\,dg
\]
is finite and nonzero for every $s\in\mathbb C$.
\end{proposition}

\begin{proof}
Write $G_r=\mathrm{GL}_r(F)$ and let
\[
P_{n+1}=\left\{\begin{pmatrix} g & x\\ 0 & 1\end{pmatrix}:g\in G_n,\ x\in F^n\right\}
\]
be the mirabolic subgroup. We use the standard Bernstein--Zelevinsky/Kirillov fact: the restriction of $\mathcal W(\Pi,\psi^{-1})$ to $P_{n+1}$ contains a nonzero $P_{n+1}$-subrepresentation identified with
\[
C_c^\infty(N_{n+1}\backslash P_{n+1},\psi^{-1}),
\]
i.e. $\psi^{-1}$-equivariant functions compactly supported modulo $N_{n+1}$.

Choose $V\in\mathcal W(\pi,\psi)$ with $V(I_n)\neq 0$: pick $v\in\pi$ with nonzero Whittaker functional value, and define $V(g)=\lambda(\pi(g)v)$. By smoothness, $v$ is fixed by some compact open $K_0\subset G_n$; set $K=K_0\cap G_n(\mathfrak o)$ (still compact open). Then
\[
V(k)=V(I_n)\neq 0\qquad (k\in K).
\]

Set
\[
H:=\{\operatorname{diag}(k,1):k\in K\}\subset P_{n+1}.
\]
Since $K\subset G_n(\mathfrak o)$ and $\psi$ has conductor $\mathfrak o$, the generic character is trivial on $N_n\cap K$, hence on $N_{n+1}\cap H$. Define $f\in C_c^\infty(N_{n+1}\backslash P_{n+1},\psi^{-1})$ by
\[
f(p)=\begin{cases}
\psi^{-1}(n),& p\in nHu_Q\text{ for some }n\in N_{n+1},\\
0,&\text{otherwise.}
\end{cases}
\]
This is well-defined (triviality on $N_{n+1}\cap H$), $\psi^{-1}$-equivariant, and compactly supported modulo $N_{n+1}$. Choose $W\in\mathcal W(\Pi,\psi^{-1})$ whose restriction to $P_{n+1}$ equals $f$.

Then:
\[
W(\operatorname{diag}(k,1)u_Q)=1\quad (k\in K),
\]
and if $W(\operatorname{diag}(g,1)u_Q)\neq 0$, then
\[
\operatorname{diag}(g,1)u_Q\in N_{n+1}Hu_Q.
\]
So for some $n\in N_{n+1},k\in K$,
\[
\operatorname{diag}(g,1)u_Q=n\operatorname{diag}(k,1)u_Q,
\]
hence (right-multiplying by $u_Q^{-1}$)
\[
\operatorname{diag}(g,1)=n\operatorname{diag}(k,1).
\]
Writing $n=\begin{psmallmatrix} n_0 & v\\ 0 & 1\end{psmallmatrix}$ with $n_0\in N_n$, comparison of the top-right block gives $v=0$, so $g=n_0k\in N_nK$. Therefore the integrand is supported in the compact subset
\[
(N_n\cap K)\backslash K\subset N_n\backslash G_n,
\]
which implies absolute convergence for all $s\in\mathbb C$.

For $g=n_0k$ with $n_0\in N_n,k\in K$, we get
\[
W(\operatorname{diag}(g,1)u_Q)=\psi^{-1}(n_0),
\qquad
V(g)=\psi(n_0)V(I_n),
\]
so
\[
W(\operatorname{diag}(g,1)u_Q)V(g)=V(I_n).
\]
Also $|\det g|=1$ on $N_nK$ (since $k\in G_n(\mathfrak o)$ and $\det n_0=1$), thus
\[
I(s;W,V)=V(I_n)\cdot \operatorname{vol}\big((N_n\cap K)\backslash K\big),
\]
a nonzero constant independent of $s$. In particular, the integral is finite and nonzero for every $s\in\mathbb C$.
\end{proof}
