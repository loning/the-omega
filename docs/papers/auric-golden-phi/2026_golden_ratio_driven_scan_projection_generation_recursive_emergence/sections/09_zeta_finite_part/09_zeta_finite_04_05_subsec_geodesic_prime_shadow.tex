\subsubsection{Bartholdi 回头 fugacity 变形与真实输入核的无回头（测地）素数影子}\label{subsec:real-input-40-geodesic-prime-shadow}
本小节把注 \ref{rem:rotation-polytope-bartholdi} 中“排除回溯的 Ihara/Bartholdi prime shadow”在真实输入 \(40\) 状态核上具体化：我们在 essential \(20\) 核心的边空间上构造无回头（non-backtracking）边移位，并得到一个出人意料的低阶行列式闭式与新的 Abel/Mertens 有限部常数。
更进一步，引入回头 fugacity 参数 \(t\)，得到一个二变量行列式闭式，把标准 prime shadow（\(t=1\)）与测地 prime shadow（\(t=0\)）置于同一条解析曲线上。

\paragraph{边空间与回头算子}
令 \(\Gamma_{\mathrm{ess}}\) 为命题 \ref{prop:real-input-40-essential-20} 的 essential 有向图（顶点数 \(20\)），并记其有向边集为 \(E\)。
对本核有 \(|E|=48\)（按输入对 \((x,y)\) 细分后的确定性分支边；可由脚本 \texttt{scripts/exp\_real\_input\_40\_rotation\_polytope\_exact.py} 的构造链直接核验）。
把每条边 \(e=(u\to v)\in E\) 视为字母，得到边移位（edge shift）\(X_E\)。

若 \(\bar e:=(v\to u)\in E\) 亦存在，则称 \(\bar e\) 为 \(e\) 的\emph{回头反边}。
定义作用在边空间 \(\RR^{E}\) 上的 \(48\times 48\) 矩阵族 \(H(t)\)（Bartholdi 回头 fugacity 变形）：
\[
H(t)_{e,e'} :=
\begin{cases}
1,& \text{若 } e=(u\to v),\,e'=(v\to w)\ \text{且 }w\neq u,\\
t,& \text{若 } e'=\bar e,\\
0,& \text{否则},
\end{cases}
\qquad (t\in\CC).
\]
于是 \(H(0)\) 禁止立即撤销（无回头/测地），而 \(H(1)\) 恰为完整的边--边转移（允许撤销）。
定义二变量 \(\zeta\) 与行列式分母
\[
\zeta_{\mathrm{bt}}(z,t):=\frac{1}{\Delta_{\mathrm{bt}}(z,t)},\qquad
\Delta_{\mathrm{bt}}(z,t):=\det(I-zH(t)).
\]
并把 \(t=0\) 的无回头（测地）\(\zeta\) 记为 \(\zeta_{\mathrm{nb}}(z):=\zeta_{\mathrm{bt}}(z,0)\)。

\begin{theorem}[无回头（测地）素数影子的 15 次行列式闭式]\label{thm:real-input-40-geodesic-det}
令 \(H_{\mathrm{nb}}:=H(0)\)。则
\[
\zeta_{\mathrm{nb}}(z)=\frac{1}{\det(I-zH_{\mathrm{nb}})},
\]
并且
\[
\boxed{
\det(I-zH_{\mathrm{nb}})
=
1-4z^3-9z^4-12z^5-16z^6-11z^7-11z^8-4z^9-z^{10}+8z^{11}+12z^{12}+10z^{13}+8z^{14}+2z^{15}.
}
\]
因此 \(H_{\mathrm{nb}}\) 有 \(33\) 个零特征值，非零谱由次数 \(15\) 的整系数多项式决定（有效动力学维数从 \(48\) 压缩到 \(15\)）。
\end{theorem}

\begin{corollary}[测地增长率与测地熵]\label{cor:real-input-40-geodesic-entropy}
设 \(\rho_{\mathrm{nb}}:=\rho(H_{\mathrm{nb}})\) 为 Perron 根，则 \(\rho_{\mathrm{nb}}\) 为方程
\[
\rho^{15}-4\rho^{12}-9\rho^{11}-12\rho^{10}-16\rho^{9}-11\rho^{8}-11\rho^{7}-4\rho^{6}-\rho^{5}+8\rho^{4}+12\rho^{3}+10\rho^{2}+8\rho+2=0
\]
的最大实根，数值为
\[
\rho_{\mathrm{nb}}\approx 2.295661870047795\ldots,\qquad
h_{\mathrm{nb}}:=\log\rho_{\mathrm{nb}}\approx 0.831021198469209\ldots.
\]
作为对照，真实输入核在周期点口径下的主增长率为 \(\lambda=\varphi^2\)（\ref{subsec:arity-dirichlet-mertens-tensor}）。
\end{corollary}

\paragraph{测地 primitive 谱与显式起始项}
记测地周期点数 \(a_n^{\mathrm{nb}}:=\Tr(H_{\mathrm{nb}}^n)\)，以及测地 primitive 轨道数 \(p_n^{\mathrm{nb}}\)（按轨道计）。
则仍有 necklace 关系与 Möbius 反演
\[
a_n^{\mathrm{nb}}=\sum_{d\mid n} d\,p_d^{\mathrm{nb}},
\qquad
p_n^{\mathrm{nb}}=\frac1n\sum_{d\mid n}\mu(d)\,a_{n/d}^{\mathrm{nb}}.
\]
本核上前 \(15\) 项完全显式：
\[
(p_n^{\mathrm{nb}})_{n=1}^{15}=(0,0,4,9,12,22,47,95,196,399,843,1775,3790,8038,17258).
\]
并且由 \(a_n^{\mathrm{nb}}\ll \rho_{\mathrm{nb}}^n\) 得到一般估计
\[
p_n^{\mathrm{nb}}=\frac{\rho_{\mathrm{nb}}^n}{n}+O\!\left(\frac{\rho_{\mathrm{nb}}^{n/2}}{n}\right).
\]

\begin{theorem}[测地 Mertens 常数（Abel 有限部）]\label{thm:real-input-40-geodesic-mertens}
定义测地 primitive 生成函数
\[
P_{\mathrm{nb}}(z):=\sum_{n\ge 1}p_n^{\mathrm{nb}}z^n.
\]
在主极点 \(z_\star=\rho_{\mathrm{nb}}^{-1}\) 处定义 Abel 有限部常数 \(\mathfrak{M}_{\mathrm{nb}}\) 使得
\[
P_{\mathrm{nb}}(z)=-\log(1-\rho_{\mathrm{nb}}z)+\log\mathfrak{M}_{\mathrm{nb}}+o(1),
\qquad z\uparrow z_\star.
\]
则数值上
\[
\boxed{
\log\mathfrak{M}_{\mathrm{nb}}\approx -1.465260542946848,\qquad
\mathfrak{M}_{\mathrm{nb}}\approx 0.231017793578888,
}
\]
并因此
\[
\boxed{
\mathsf{M}_{\mathrm{nb}}:=\gamma+\log\mathfrak{M}_{\mathrm{nb}}\approx -0.888044878045315,
}
\]
与真实输入核在周期点口径下的 \(\mathsf{M}\approx 0.3229030\) 形成鲜明对照（\ref{subsec:arity-dirichlet-mertens-tensor}）。
\end{theorem}

\paragraph{Bartholdi 型回头 fugacity 变形：二变量行列式闭式}
令 \(\Delta_{\mathrm{bt}}(z,t)=\det(I-zH(t))\)。
则存在 \(F(z,t)\in\ZZ[z,t]\)（对 \(z\) 次数 \(20\)，对 \(t\) 次数至多 \(9\)）使得：
\begin{theorem}[回头 fugacity 变形的二变量行列式闭式]\label{thm:real-input-40-bartholdi-det}
有严格因式分解
\[
\boxed{
\Delta_{\mathrm{bt}}(z,t)=(1-t^2 z^2)\bigl(1+t(1-t)z^2\bigr)^2\,F(z,t).
}
\]
其中
\[
F(z,t)=\sum_{k=0}^{20}f_k(t)\,z^k,\qquad f_k(t)\in\ZZ[t],\quad \deg f_k\le 9,
\]
且系数可显式写为
\[
\begin{aligned}
f_0(t)&=1,\\
f_1(t)&=-t,\\
f_2(t)&=-2t(2t+1),\\
f_3(t)&=2(2t^3+t^2-2),\\
f_4(t)&=6t^4+6t^3+t^2+4t-9,\\
f_5(t)&=-6t^5-6t^4-t^3+14t^2+9t-12,\\
f_6(t)&=-4t^6-6t^5-2t^4-14t^3+24t^2+15t-16,\\
f_7(t)&=4t^7+6t^6+2t^5-18t^4-22t^3+41t^2-t-11,\\
f_8(t)&=(t-1)(t^7+3t^6+4t^5+22t^4-4t^3-41t^2+21t+11),\\
f_9(t)&=-(t-1)^2(t^7+4t^6+8t^5+2t^4-24t^3+11t^2+50t+4),\\
f_{10}(t)&=-(t-1)(10t^6-6t^5-45t^4+46t^3+54t^2-56t-1),\\
f_{11}(t)&=-(t-1)^2(2t^6+14t^5-15t^4-61t^3+41t^2+36t-8),\\
f_{12}(t)&=(t-1)^2(2t^7-t^6-25t^5+7t^4+85t^3-36t^2-36t+12),\\

f_{14}(t)&=(t-1)^3(4t^6+t^5-46t^4+6t^3+57t^2-28t-8),\\
f_{15}(t)&=2(t-1)^4(2t^5-2t^4-6t^3+7t^2+7t+1),\\
f_{16}(t)&=t(t-1)^4(9t^4+t^3-24t^2+18t+8),\\

\end{aligned}
\]
\end{theorem}

\begin{remark}[两个关键特化：把“标准 prime shadow”与“测地 prime shadow”放到同一条曲线上]\label{rem:real-input-40-bartholdi-specializations}
\begin{itemize}
  \item \textbf{\(t=0\)}：\(\Delta_{\mathrm{bt}}(z,0)=F(z,0)=\det(I-zH_{\mathrm{nb}})\)，回到定理 \ref{thm:real-input-40-geodesic-det} 的测地（无回头）\(\zeta\)。
  \item \textbf{\(t=1\)}：\(\Delta_{\mathrm{bt}}(z,1)=(1-z^2)F(z,1)\) 精确化为完整边--边转移的 \(\zeta\) 分母：
  \[
  \det(I-zH(1))
  =-(z-1)^2(z+1)^3(z^2-3z+1)(z^2-z-1).
  \]
  其中 \(F(z,1)=(z-1)(z+1)^2(z^2-3z+1)(z^2-z-1)\)（此时 \(f_k(1)=0\) 对所有 \(k\ge 8\)）。
\end{itemize}
\end{remark}

\begin{theorem}[回头密度与回头方差的闭式]\label{thm:real-input-40-backtracking-cumulants}
令 \(t=e^{\theta}\)，并记
\[
\lambda(\theta):=\rho\!\bigl(H(e^{\theta})\bigr),\qquad
P(\theta):=\log\lambda(\theta).
\]
则在 \(\theta=0\)（即 \(t=1\)）处有闭式
\[
\boxed{\;
P'(0)=\frac{211-93\sqrt5}{20}\approx 0.1522839046\;},
\]
\[
\boxed{\;
P''(0)=\frac{193219}{100}-\frac{216003}{250}\sqrt5\approx 0.2004346245\; }.
\]
其中 \(P'(0)\) 可解释为“最大熵平衡态下，每步出现立即撤销回头片段的频率”，而 \(P''(0)\) 为回头计数的渐近方差密度（Green--Kubo）。
\end{theorem}
