\section{Appendix A: Notation and Fixed Objects}

\begin{enumerate}
  \item $\varph=\frac{1+\sqrt5}{2}$: the golden ratio; $\alpha=\varph^{-1}$ is the golden branch slope.
  \item $F_1=F_2=1$, $F_{n+2}=F_{n+1}+F_n$: Fibonacci numbers.
  \item $\Omega_m=\{0,1\}^m$, $\mathcal H_m=\ell^2(\Omega_m)$.
  \item $X_m=\{w\in\{0,1\}^m:\ w\ \text{contains no adjacent}\ 11\}$.
  \item $X_m^{\mathrm{cyc}}=\{w\in X_m:\ w_mw_1\neq 11\}$, $X_m^{\mathrm{bdry}}=X_m\setminus X_m^{\mathrm{cyc}}$.
  \item $P_m=\sum_{w\in X_m}\lvert w\rangle\langle w\rvert$: golden existence projection.
  \item $\lvert\Phi_m\rangle=\card{X_m}^{-1/2}\sum_{w\in X_m}\lvert w\rangle$: golden existence state.
\end{enumerate}
