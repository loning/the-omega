\subsection{折叠指标与投影熵率}\label{subsec:pom-projection-entropy}
本节以论文中的“全移位窗口 $\Omega_m$ 到黄金均值 SFT 语言 $X_m$ 的规范化截面”为基底：$X_m$ 是禁词 $(11)$ 的长度 $m$ 语言，$\Fold_m:\Omega_m\to X_m$ 把任意窗口微态投影到该语法块上，$\Phi_m$ 把窗口输出提升为序列级 sofic 因子化；相应的标记图 $G_m$ 由长度 $(m-1)$ 后缀状态与追加一比特构造（定义 \ref{def:G_m}）。在此框架下，纤维多重度 $d_m(x)$ 与条件期望 $E_m$ 是规范投影 $P_Z$ 的两类接口，以下结论给出其尺度结构。
\begin{definition}[折叠指标与投影熵率]\label{def:pom-fold-index}
令
\[
I_m:=\frac{|\Omega_m|}{|X_m|}=\frac{2^m}{F_{m+2}},
\qquad
S_Z(m):=\log I_m,
\qquad
h_Z:=\lim_{m\to\infty}\frac{S_Z(m)}{m}.
\]
$I_m$ 是折叠诱导的平均多重度指标，$S_Z(m)$ 与 $h_Z$ 分别刻画分辨率 $m$ 与单位分辨率的投影熵率。
\end{definition}

\begin{proposition}[折叠指标的渐近]\label{prop:pom-projection-entropy}
有
\[
S_Z(m)=m\log\!\left(\frac{2}{\varphi}\right)+O(1),
\qquad
h_Z=\log\!\left(\frac{2}{\varphi}\right).
\]
\end{proposition}
\begin{proof}
由附录 \ref{app:fold-multiplicity} 的渐近式
\(
I_m\sim C(2/\varphi)^m
\)
直接得到。证毕。
\end{proof}

\begin{corollary}[分辨率不足的空间门槛]\label{cor:pom-resolution-threshold}
若希望“每个稳定类型平均只对应不超过 $R$ 个微态”，则必须满足
\[
m\gtrsim \frac{\log R}{\log(2/\varphi)}.
\]
因此提高可区分性必然以分辨率/空间增长为代价。
\end{corollary}

\begin{remark}[折叠熵的二进制斜率]\label{rem:pom-fold-bits}
定义二进制单位的折叠熵
\[
S_Z^{(2)}(m):=\log_2\frac{2^m}{F_{m+2}}.
\]
则
\[
S_Z^{(2)}(m)=m\log_2\!\left(\frac{2}{\varphi}\right)+O(1)
\approx 0.305758\,m+O(1),
\]
给出每增加一位分辨率所“压扁”的平均信息量常数。
\end{remark}

\subsubsection{窗口折叠的最大纤维多重度}\label{subsec:pom-max-fiber}
记
\[
d_m(x):=\left|\Fold_m^{-1}(x)\right|,
\qquad
D_m:=\max_{x\in X_m} d_m(x).
\]
\subsubsection{信息账本：可见熵 + 纤维体积 $-$ 残差信息}\label{subsec:pom-information-ledger}
固定分辨率 $m$，令 $\Omega_m=\{0,1\}^m$，$X_m$ 为禁词 $(11)$ 的稳定语言块，并设 $\Fold_m:\Omega_m\to X_m$ 为折叠投影。
记概率单纯形为 $\Delta(\Omega_m),\Delta(X_m)$。
定义推前（读出）算子
\[
P_m:=(\Fold_m)_*:\Delta(\Omega_m)\to\Delta(X_m),
\qquad
(P_m\mu)(x)=\sum_{\Fold_m(a)=x}\mu(a),
\]
以及纤维均匀提升（最大对称提升）核
\[
Q_m:\Delta(X_m)\to\Delta(\Omega_m),
\qquad
(Q_m\pi)(a)=\frac{\pi(\Fold_m(a))}{d_m(\Fold_m(a))}.
\]
则 $P_m\circ Q_m=\mathrm{id}_{\Delta(X_m)}$（见附录 \ref{app:op-algebra} 的命题 \ref{prop:fold-pushforward-lift}）。

\begin{theorem}[相对熵缺陷恒等式：信息账本闭式]\label{thm:pom-kl-ledger}
取任意 $\mu\in\Delta(\Omega_m)$，记其稳定推前为 $\pi:=P_m\mu\in\Delta(X_m)$，并令
\[
\mu^{e}:=Q_m\pi\in\Delta(\Omega_m)
\]
为 $\pi$ 的纤维均匀提升（即在每条纤维上条件均匀）。
定义 Shannon 熵与 KL 散度（取自然对数）
\[
H(\nu):=-\sum_{s}\nu(s)\log\nu(s),
\qquad
D_{\mathrm{KL}}(\mu\|\nu):=\sum_{s}\mu(s)\log\frac{\mu(s)}{\nu(s)}.
\]
则有严格恒等式
\[
\boxed{\;
H(\mu)=H(\pi)+\mathbb{E}_{\pi}\!\bigl[\log d_m(X)\bigr]\;-\;D_{\mathrm{KL}}(\mu\|\mu^{e}).
\;}
\]
\end{theorem}
\begin{proof}
由 KL 定义展开得
\(
D_{\mathrm{KL}}(\mu\|\mu^{e})=-H(\mu)-\sum_{a\in\Omega_m}\mu(a)\log\mu^{e}(a)
\)。
而
\(
\log\mu^{e}(a)=\log\pi(x)-\log d_m(x)
\)
其中 $x=\Fold_m(a)$。
因此
\[
\sum_{a}\mu(a)\log\pi(\Fold_m(a))
=\sum_{x\in X_m}\left(\sum_{\Fold_m(a)=x}\mu(a)\right)\log\pi(x)
=\sum_{x}\pi(x)\log\pi(x)
=-H(\pi),
\]
并且
\[
\sum_{a}\mu(a)\log d_m(\Fold_m(a))
=\sum_{x\in X_m}\left(\sum_{\Fold_m(a)=x}\mu(a)\right)\log d_m(x)
=\sum_{x}\pi(x)\log d_m(x)
=\mathbb{E}_{\pi}[\log d_m(X)].
\]
代回整理即得结论。证毕。
\end{proof}

\begin{corollary}[“不丢信息”的严格含义：最大熵提升与残差的唯一承载]\label{cor:pom-kl-ledger-bound}
对任意 $\mu\in\Delta(\Omega_m)$ 与 $\pi=P_m\mu$，有上界
\[
H(\mu)\le H(\pi)+\mathbb{E}_{\pi}[\log d_m(X)],
\]
且等号当且仅当 $\mu=\mu^{e}$（即 $\mu$ 在每条折叠纤维上条件均匀）。
\end{corollary}
\begin{proof}
由定理 \ref{thm:pom-kl-ledger} 与 $D_{\mathrm{KL}}\ge 0$ 得上界；且 $D_{\mathrm{KL}}(\mu\|\mu^e)=0$ 当且仅当 $\mu=\mu^e$。证毕。
\end{proof}

\begin{corollary}[最坏熵缺口上界]\label{cor:pom-entropy-gap-bound}
对任意 $\mu\in\Delta(\Omega_m)$，令 $\pi=P_m\mu$。则
\[
H(\mu)-H(\pi)\le \log D_m.
\]
因此最大可见熵缺口满足
\[
\sup_{\mu\in\Delta(\Omega_m)}\bigl(H(\mu)-H(\pi)\bigr)\le \log D_m.
\]
并由定理 \ref{thm:pom-max-fiber} 得
\[
\limsup_{m\to\infty}\frac{1}{m}\sup_{\mu}\bigl(H(\mu)-H(\pi)\bigr)\le \frac12\log\varphi.
\]
\end{corollary}
\begin{proof}
由定理 \ref{thm:pom-kl-ledger}，
\(
H(\mu)-H(\pi)=\mathbb{E}_{\pi}[\log d_m(X)]-D_{\mathrm{KL}}(\mu\|\mu^e)
\)。
注意该差异可为负，但其上界由 $\mathbb{E}_{\pi}[\log d_m(X)]\le \log D_m$ 给出。取上确界即得前两式。再由定理 \ref{thm:pom-max-fiber} 的渐近得到单位分辨率上界。证毕。
\end{proof}

\begin{lemma}[纤维 KL 分解：残差为纤维内非均匀的平均]\label{lem:pom-fiberwise-kl}
取任意 $\mu\in\Delta(\Omega_m)$，令 $\pi=P_m\mu$。对每个 $x\in X_m$ 定义纤维条件分布
\[
\mu_x(a):=
\begin{cases}
\mu(a)/\pi(x),& a\in \Fold_m^{-1}(x),\ \pi(x)>0,\\
0,& a\in \Fold_m^{-1}(x),\ \pi(x)=0,
\end{cases}
\]
并令 $u_x$ 为该纤维上的均匀分布 $u_x(a)=1/d_m(x)$（$a\in\Fold_m^{-1}(x)$）。
则对纤维均匀提升 $\mu^e=Q_m\pi$，有链式恒等式
\[
\boxed{\;
D_{\mathrm{KL}}(\mu\|\mu^e)=\sum_{x\in X_m}\pi(x)\,D_{\mathrm{KL}}(\mu_x\|u_x).
\;}
\]
\end{lemma}
\begin{proof}
当 $\pi(x)=0$ 时，$\mu$ 在纤维 $\Fold_m^{-1}(x)$ 上亦为零测，因此该纤维对两端均无贡献。
对任意满足 $\pi(x)>0$ 的纤维，若 $a\in\Fold_m^{-1}(x)$，则
\(
\mu(a)=\pi(x)\mu_x(a)
\)
且
\(
\mu^e(a)=\pi(x)u_x(a)
\)。
代入 KL 定义并按纤维分组得
\[
D_{\mathrm{KL}}(\mu\|\mu^e)
=\sum_{x}\sum_{a\in\Fold_m^{-1}(x)}\pi(x)\mu_x(a)\log\frac{\mu_x(a)}{u_x(a)}
=\sum_x \pi(x)\,D_{\mathrm{KL}}(\mu_x\|u_x).
\]
证毕。
\end{proof}

\begin{corollary}[熵增恒等式：最大熵提升的增量等于残差 KL]\label{cor:pom-ent-increase-equals-kl}
沿用上述记号，则有严格恒等式
\[
\boxed{\;
H(\mu^e)-H(\mu)=D_{\mathrm{KL}}(\mu\|\mu^e).
\;}
\]
\end{corollary}
\begin{proof}
由定理 \ref{thm:pom-kl-ledger} 得
\(
H(\mu)=H(\pi)+\mathbb{E}_{\pi}[\log d_m(X)]-D_{\mathrm{KL}}(\mu\|\mu^e)
\)。
另一方面，直接对 $\mu^e$ 代入（或等价地用“纤维条件均匀”）可得
\(
H(\mu^e)=H(\pi)+\mathbb{E}_{\pi}[\log d_m(X)]
\)。
两式相减即得结论。证毕。
\end{proof}

\begin{theorem}[纤维反射子：幂等 Markov 投影与固定点刻画]\label{thm:pom-fiber-reflector}
定义 Markov 算子
\[
K_m:=Q_m\circ P_m:\Delta(\Omega_m)\to\Delta(\Omega_m).
\]
则：
\begin{enumerate}
  \item $K_m$ 幂等：$K_m^2=K_m$；
  \item 对任意 $\mu$，若 $\pi=P_m\mu$，则 $K_m\mu=\mu^e$ 为 $\pi$ 的纤维均匀提升；
  \item 固定点满足：$K_m\mu=\mu$ 当且仅当 $\mu$ 在每条纤维上条件均匀（等价地 $\mu=\mu^e$）。
\end{enumerate}
\end{theorem}
\begin{proof}
由 $P_m\circ Q_m=\mathrm{id}$ 得
\(
K_m^2=Q_mP_mQ_mP_m=Q_m(P_mQ_m)P_m=Q_mP_m=K_m
\)。
对任意 $\mu$，令 $\pi=P_m\mu$，则 $K_m\mu=Q_m\pi=\mu^e$ 是定义。
最后，$K_m$ 的像空间恰为 $\{Q_m\pi:\pi\in\Delta(X_m)\}$；因此 $K_m\mu=\mu$ 当且仅当 $\mu$ 属于该像空间，即在每条纤维上条件均匀。证毕。
\end{proof}

\begin{proposition}[\texorpdfstring{$K_m$}{Km} 的核矩阵：对称、双随机与纤维平均]\label{prop:pom-Km-bistochastic}
把 $K_m$ 视为 $\Omega_m$ 上的 Markov 核，其转移核为
\[
K_m(a,b):=
\begin{cases}
\dfrac{1}{d_m(\Fold_m(a))},& \Fold_m(b)=\Fold_m(a),\\[6pt]
0,& \text{否则}.
\end{cases}
\]
则：
\begin{enumerate}
  \item \textbf{（幂等）} $K_m^2=K_m$；
  \item \textbf{（对称）} $K_m(a,b)=K_m(b,a)$；
  \item \textbf{（双随机）} 对任意 $a$ 有 $\sum_b K_m(a,b)=1$，且对任意 $b$ 有 $\sum_a K_m(a,b)=1$；
  \item \textbf{（函数版 = 纤维平均）} 对任意函数 $f:\Omega_m\to\RR$，
  \[
  (K_m f)(a):=\sum_{b\in\Omega_m}K_m(a,b)f(b)
  =\frac{1}{d_m(\Fold_m(a))}\sum_{\Fold_m(b)=\Fold_m(a)} f(b),
  \]
  即 $K_m$ 在观测量层面就是“按折叠纤维做算术平均”的条件期望接口。
\end{enumerate}
特别地，$\Omega_m$ 上的均匀分布 $u_{\Omega_m}(a)\equiv 2^{-m}$ 为不变分布（由双随机性立得）。
\end{proposition}
\begin{proof}
对固定 $x\in X_m$，将 $\Omega_m$ 分解为纤维并注意：限制在纤维 $\Fold_m^{-1}(x)$ 上，$K_m$ 的矩阵块是常数矩阵 $J/d_m(x)$（其中 $J$ 为全 $1$ 矩阵），其余块为 $0$。
因此 $(J/d)^2=J/d$ 给出幂等性；常数矩阵显然对称；行和为 $1$，而列和也为 $1$（每一列在该纤维内有 $d$ 个相同项 $1/d$）。
函数版公式就是矩阵乘法定义的展开。证毕。
\end{proof}

\begin{corollary}[最大熵提升的 I-投影表述]\label{cor:pom-Iproj-maxent}
记 $u_{\Omega_m}$ 为 $\Omega_m$ 上的均匀分布。对任意固定 $\pi\in\Delta(X_m)$，约束集
\[
\mathcal{C}_\pi:=\{\nu\in\Delta(\Omega_m):P_m\nu=\pi\}
\]
为凸集，且其唯一最大熵元 $\mu^e=Q_m\pi$ 同时满足
\[
\boxed{\;
\mu^e=\arg\min_{\nu\in\mathcal{C}_\pi}D_{\mathrm{KL}}(\nu\|u_{\Omega_m})
\;=\;
\arg\max_{\nu\in\mathcal{C}_\pi}H(\nu).
\;}
\]
\end{corollary}
\begin{proof}
因为 $u_{\Omega_m}(a)\equiv 2^{-m}$，对任意 $\nu$ 有
\(
D_{\mathrm{KL}}(\nu\|u_{\Omega_m})=\sum_a \nu(a)\log\nu(a)+m\log 2=\log(2^m)-H(\nu)
\)，
故在同一约束集 $\mathcal{C}_\pi$ 上，“最小 KL 到均匀”与“最大熵”等价。
而最大熵解的唯一性由定理 \ref{thm:pom-maxent-lift}（或按纤维逐块严格凹性）给出。证毕。
\end{proof}

\begin{theorem}[规范的粗粒化动力学：先提升—再演化—再读出]\label{thm:pom-coarse-grained-kernel}
设 $K:\Delta(\Omega_m)\to\Delta(\Omega_m)$ 为任意 Markov 核（窗口层动力学）。
定义稳定层的粗粒化核
\[
\overline K:=P_m\circ K\circ Q_m:\Delta(X_m)\to\Delta(X_m).
\]
则对任意初始稳定分布 $\pi_0\in\Delta(X_m)$，令 $\mu_0:=Q_m\pi_0$，有
\[
\pi_1=\overline K\,\pi_0=P_m(K\mu_0).
\]
并且 $\overline K$ 是满足该“先提升—再演化—再读出”语义的唯一稳定层 Markov 核。
\end{theorem}
\begin{proof}
第一式为定义展开。若 $\widetilde K$ 亦满足对所有 $\pi_0$ 有
\(
\widetilde K\pi_0=P_m(K(Q_m\pi_0))
\)，
则逐点相等推出 $\widetilde K=\overline K$。证毕。
\end{proof}
\begin{theorem}[最大纤维闭式]\label{thm:pom-max-fiber}
取 Fibonacci 延拓到 $F_0=0,F_1=1$。对 $m\ge 2$，有精确闭式
\[
D_{2k}=F_{k+2},
\qquad
D_{2k+1}=F_{k+2}+F_{k-1}\ (=2F_{k+1}).
\]
因此
\[
D_m\sim C'\,\varphi^{m/2},
\]
即最大纤维增长率为 $\sqrt{\varphi}$。特别地，存在常数 $C_1,C_2>0$ 使
\[
C_1\,\varphi^{m/2}\ \le\ D_m\ \le\ C_2\,\varphi^{m/2}.
\]
\end{theorem}
\begin{proof}
对 $\{0,1\}^m$ 的折叠规范化，唯一会被触发的局部重写是
\(
011\mapsto 100
\)
（Fibonacci 进位）。系统合流意味着每个 $\omega$ 的规范形唯一，因此固定 $x\in X_m$ 时，$\omega\in\Fold_m^{-1}(x)$ 当且仅当 $x$ 能通过若干次逆代换 $100\mapsto 011$ 生成 $\omega$。该逆代换占用长度 $3$ 的局部区间；若两次逆代换的起点相距 $1$ 或 $2$，区间重叠会在某一位产生 $2$，因此可选位点必须满足“不可相邻选择”的约束。令 $k=\lfloor m/2\rfloor$，把可选位点看作一条长度 $k$ 的路径图 $P_k$，记其独立集计数为 $I_k$。则
\[
I_0=1,\qquad I_1=2,\qquad I_k=I_{k-1}+I_{k-2}\ (k\ge 2),
\qquad I_k=F_{k+2}.
\]
于是对任意 $x\in X_m$ 有 $d_m(x)\le I_k$。当 $m=2k+1$ 为奇数时，由引理 \ref{lem:pom-one-bit} 的溢出比特 $b\in\{0,1\}$ 分裂为两支；固定 $b$ 会冻结最顶端一处候选位点，使可选链长度降为 $k-1$，从而
\[
d_m(x)\le 2I_{k-1}=2F_{k+1}.
\]

下面构造达到上界的族。对 $m=2,3$ 取
\(
x^{(2)}=00,\ x^{(3)}=000
\)；其余情形令 $m=4q+r$（$r\in\{0,1,2,3\}$）并定义
\[
x^{(m)}=
\begin{cases}
(0001)^{\,q-1}0000,& r=0,\ q\ge 1,\\[2pt]
(0001)^{\,q-1}00100,& r=1,\ q\ge 1,\\[2pt]
(0001)^{\,q}00,& r=2,\ q\ge 0,\\[2pt]
(0001)^{\,q-1}0000100,& r=3,\ q\ge 1.
\end{cases}
\]
该族满足 $x^{(m)}\in X_m$，且其局部零块足以保证：任取不相邻位点集合 $J\subseteq\{1,\dots,k\}$（奇数时为 $\{1,\dots,k-1\}$），按从高位到低位依次执行逆代换，所需的 $100$ 模式在每一步均可实现，从而得到唯一 $\omega\in\Fold_m^{-1}(x^{(m)})$。因此
\[
d_{2k}(x^{(2k)})=I_k=F_{k+2},
\qquad
d_{2k+1}(x^{(2k+1)})=2I_{k-1}=2F_{k+1}.
\]
上界可达即得 $D_m$ 的闭式；渐近式与常数界由 Fibonacci 渐近 $F_n\sim \varphi^n/\sqrt5$ 直接推出。证毕。
\end{proof}

\begin{corollary}[二步递推与基值]\label{cor:pom-D-rec}
$D_m$ 满足
\[
D_m=D_{m-2}+D_{m-4}\qquad(m\ge 6),
\]
并具有基值
\[
D_2=2,\quad D_3=2,\quad D_4=3,\quad D_5=4.
\]
\end{corollary}
\begin{proof}
由定理 \ref{thm:pom-max-fiber} 的闭式，$D_{2k}=F_{k+2}$ 与 $D_{2k+1}=2F_{k+1}$，而 Fibonacci 数满足 $F_{k+2}=F_{k+1}+F_k$。分别代入 $m=2k$ 与 $m=2k+1$ 即得递推；基值由直接枚举或闭式代入给出。证毕。
\end{proof}

\begin{remark}[最坏投影缺口的比特量化]\label{rem:pom-D-delay}
把“区分纤维内微态”视为必须携带的额外信息量，则最坏缺口可用
\[
\Delta_m^{\max}:=\left\lceil \log_2 D_m\right\rceil
\]
衡量。由定理 \ref{thm:pom-max-fiber} 得
\(
\Delta_m^{\max}\sim \frac{m}{2}\log_2\varphi
\)，它给出一种与分辨率 $m$ 线性增长的“不可并行信息缺口”标尺。
\end{remark}

\begin{remark}[一比特溢出与二步递推的结构来源]\label{rem:pom-overflow-rec}
由引理 \ref{lem:pom-one-bit}，每条纤维只分裂为 $b=0,1$ 两支；且 $b=1$ 必要求末位满足零相邻约束。结合局部逆代换的“不可相邻选择链”，最大纤维的计数等价于在步长为 $2$ 的链上作独立集计数，从而结构上解释了二步递推 $D_m=D_{m-2}+D_{m-4}$。
\end{remark}

\begin{corollary}[空间缺陷的夹逼区间]\label{cor:pom-defect-band}
令平均纤维多重度 $\overline d_m=2^m/F_{m+2}$。则
\[
\log \overline d_m\sim m\log\!\left(\frac{2}{\varphi}\right),
\qquad
\log D_m\sim \frac{m}{2}\log\varphi.
\]
因此空间缺陷在“平均尺度”与“最坏尺度”之间形成硬夹逼区间。
\end{corollary}

\begin{remark}[最坏尺度与平均尺度的偏离]\label{rem:pom-defect-gap}
由渐近式可得
\[
\frac{D_m}{\overline d_m}
=
\Theta\!\left(\left(\frac{\varphi^{3/2}}{2}\right)^m\right),
\qquad

\subsubsection{二阶碰撞矩与 \texorpdfstring{$\mathrm{R\acute{e}nyi}$}{Renyi}--2 投影熵率}\label{subsec:pom-s2}
最大纤维多重度刻画“最坏空间缺陷”，但在统计意义上更稳健的指标来自碰撞矩（\(\ell^2\) 质量）。它给出一个可与在线延迟 \(\delta_{\min}=3\) 并列的、纯由折叠门 \(P_Z\) 决定的指数尺度常数。

\begin{lemma}[碰撞等价于模 \texorpdfstring{$F_{m+2}$}{F\_{m+2}} 同余]\label{lem:pom-fold-congruence}
设 \(\omega,\omega'\in\Omega_m=\{0,1\}^m\)，并定义窗口值
$$
N(\omega):=\sum_{k=1}^m \omega_k F_{k+1}.
$$
则有等价关系
$$
\Fold_m(\omega)=\Fold_m(\omega')
\quad\Longleftrightarrow\quad
N(\omega)\equiv N(\omega')\pmod{F_{m+2}}.
$$
\end{lemma}
\begin{proof}
记 \(x=\Fold_m(\omega)\in X_m\)、\(x'=\Fold_m(\omega')\in X_m\)。由引理 \ref{lem:pom-one-bit}（单隐藏位分解）存在唯一 \(b(\omega),b(\omega')\in\{0,1\}\) 使
$$
N(\omega)=V_m(x)+b(\omega)F_{m+2},\qquad
N(\omega')=V_m(x')+b(\omega')F_{m+2}.
$$
若 \(x=x'\)，则两式相减得 \(N(\omega)-N(\omega')=(b(\omega)-b(\omega'))F_{m+2}\)，从而同余成立。
反之，若 \(N(\omega)\equiv N(\omega')\ (\mathrm{mod}\ F_{m+2})\)，则两者在模 \(F_{m+2}\) 下的余数相同。又由于 \(V_m(\Fold_m(\cdot))\) 正是 Zeckendorf 规范形在前 \(m\) 位对应的值（定义 \ref{def:pom-window-value}），该余数唯一确定折叠后的稳定类型 \(x\in X_m\)，因此 \(x=x'\)，即 \(\Fold_m(\omega)=\Fold_m(\omega')\)。证毕。
\end{proof}

\begin{proposition}[\texorpdfstring{$q$}{q}-重碰撞的同余矩表达]\label{prop:pom-moment-congruence-q}
对任意整数 \(q\ge 2\)，定义
$$
S_q(m):=\sum_{x\in X_m} d_m(x)^q.
$$
并令
$$
c_m(r):=\#\{\omega\in\Omega_m:\ N(\omega)\equiv r\ (\mathrm{mod}\ F_{m+2})\},\qquad r\in\ZZ/F_{m+2}\ZZ.
$$
则有恒等式
$$
\boxed{\ 
S_q(m)=\sum_{r\in\ZZ/F_{m+2}\ZZ} c_m(r)^q.
\ }
$$
\end{proposition}
\begin{proof}
记
$$
R_m^{(q)}:=\{(\omega^{(1)},\dots,\omega^{(q)})\in\Omega_m^q:\ \Fold_m(\omega^{(1)})=\cdots=\Fold_m(\omega^{(q)})\}.
$$
由定义立得 \(|R_m^{(q)}|=\sum_{x\in X_m}d_m(x)^q=S_q(m)\)。
另一方面，由引理 \ref{lem:pom-fold-congruence}，对任意两条窗口 \(\omega,\omega'\) 有
\(
\Fold_m(\omega)=\Fold_m(\omega')\iff N(\omega)\equiv N(\omega')\ (\mathrm{mod}\ F_{m+2})
\)。
因此 \((\omega^{(1)},\dots,\omega^{(q)})\in R_m^{(q)}\) 当且仅当
$$
N(\omega^{(1)})\equiv \cdots \equiv N(\omega^{(q)})\pmod{F_{m+2}}.
$$
按共同余数类 \(r\in\ZZ/F_{m+2}\ZZ\) 分组计数，得到
$$
|R_m^{(q)}|
=\sum_{r\in\ZZ/F_{m+2}\ZZ}\#\{\omega^{(1)}:\ N(\omega^{(1)})\equiv r\}\cdots \#\{\omega^{(q)}:\ N(\omega^{(q)})\equiv r\}
=\sum_{r}c_m(r)^q.
$$
与 \(|R_m^{(q)}|=S_q(m)\) 合并即得结论。证毕。
\end{proof}

\begin{definition}[折叠碰撞矩]\label{def:pom-s2}

2&1&1

2\,\mathbf{1}^{\mathsf T}A_2^{\,1}\mathbf{1}=14,\qquad
2\,\mathbf{1}^{\mathsf T}A_2^{\,2}\mathbf{1}=36,

\end{proof}

\begin{proposition}[\texorpdfstring{$S_2(m)$}{S2(m)} 的 Plancherel 表达与角色因子化]\label{prop:pom-s2-plancherel}
令 \(G_m:=\ZZ/F_{m+2}\ZZ\)，并取其标准角色
\(
\chi_j(r):=\exp(2\pi i jr/F_{m+2})
\)
（\(j\in G_m\)）。对同余计数 \(c_m(r)=\#\{\omega\in\Omega_m:\ N(\omega)\equiv r\}\) 定义离散傅里叶变换
\[
\widehat c_m(j):=\sum_{r\in G_m} c_m(r)\,\chi_j(r).
\]
则二阶碰撞矩满足严格恒等式
\[
\boxed{\;
S_2(m)=\sum_{r\in G_m}c_m(r)^2=\frac{1}{|G_m|}\sum_{j\in G_m}\bigl|\widehat c_m(j)\bigr|^2\;}
\]
并且 \(\widehat c_m(j)\) 具有完全因子化形式
\[
\boxed{\;
\widehat c_m(j)=\sum_{\omega\in\Omega_m}\chi_j\!\bigl(N(\omega)\bigr)
=\prod_{k=1}^m\Bigl(1+\chi_j(F_{k+1})\Bigr)
=\prod_{k=1}^m\left(1+\exp\!\frac{2\pi i j F_{k+1}}{F_{m+2}}\right).\;}
\]
从而
\[
S_2(m)
=\frac{1}{F_{m+2}}\sum_{j=0}^{F_{m+2}-1}\ \prod_{k=1}^m\left|1+\exp\!\frac{2\pi i j F_{k+1}}{F_{m+2}}\right|^2.
\]
\end{proposition}
\begin{proof}
第一式是有限阿贝尔群上的 Plancherel 恒等式。这里给出与本节记号一致的直接推导：由反演公式
\(
c_m(r)=\frac{1}{|G_m|}\sum_{j}\widehat c_m(j)\,\overline{\chi_j(r)}
\)
与角色正交性
\(
\sum_{r}\chi_j(r)\overline{\chi_{j'}(r)}=|G_m|\,\mathbf{1}_{j=j'}
\)
立得
\[
\sum_{r}c_m(r)^2
=\frac{1}{|G_m|}\sum_{j}|\widehat c_m(j)|^2.
\]

对因子化：由定义
\[
\widehat c_m(j)
=\sum_{r}\sum_{\substack{\omega\in\Omega_m\\N(\omega)\equiv r}}\chi_j(r)
=\sum_{\omega\in\Omega_m}\chi_j\!\bigl(N(\omega)\bigr).
\]
再用窗口值的线性分解 \(N(\omega)=\sum_{k=1}^m \omega_k F_{k+1}\)（定义 \ref{def:pom-window-value}）与角色的乘法性，
\[
\chi_j\!\bigl(N(\omega)\bigr)
=\prod_{k=1}^m \chi_j(F_{k+1})^{\omega_k},
\]
对每一位 \(\omega_k\in\{0,1\}\) 求和即可得到
\(
\widehat c_m(j)=\prod_{k=1}^m(1+\chi_j(F_{k+1}))
\)。
最后一式由 \(|G_m|=F_{m+2}\) 与 \(\chi_j(F_{k+1})=\exp(2\pi i jF_{k+1}/F_{m+2})\) 代入得到。证毕。
\end{proof}

\begin{lemma}[三阶递推的最小性]\label{lem:pom-s2-minimal-order}

\begin{remark}[Hankel 最小实现与 \texorpdfstring{$A_2$}{A2} 的非负相似标准形]\label{rem:pom-s2-hankel}

\begin{theorem}[Hankel 秩定理（加权自动机的线性 Myhill--Nerode 口径）]\label{thm:hankel-rank-minimal}
设 \(k\) 为域（例如 \(\mathbb{Q}\) 或 \(\mathbb{R}\)）。给定一个序列 \(a=(a_n)_{n\ge 0}\subset k\)，其 Hankel 矩阵定义为
$$
H(a):=(a_{i+j})_{i,j\ge 0}.
$$
称 \((u,A,v)\) 为 \(a\) 的一个线性表示（linear representation），若 \(u,v\in k^d\)、\(A\in k^{d\times d}\)，且对一切 \(n\ge 0\) 有
$$
a_n=u^{\mathsf T}A^n v.
$$
则有
$$
\mathrm{rank}\bigl(H(a)\bigr)=\min\{d:\exists\ (u,A,v)\text{ 给出维数为 }d\text{ 的线性表示}\}.
$$
等价地，\(\mathrm{rank}(H(a))\) 等于生成该序列的最小加权自动机（有理级数）的状态数（见 \cite{BerstelReutenauer1988RationalSeries}）。
\end{theorem}

因此 \(\zeta_{A_2}(z)\) 的主极点位于 \(z_\star=1/r_2\)，其中 \(r_2\) 为 \(x^3-2x^2-2x+2=0\) 的最大实根（推论 \ref{cor:pom-s2-asymptotic}），并且碰撞核的拓扑熵为
$$
h_{\mathrm{top}}(X_{A_2},\sigma)=\log r_2.
$$
\end{proposition}
\begin{proof}
对 \(A_2\) 直接计算行列式得 \(\det(I-zA_2)=1-2z-2z^2+2z^3\)。主极点对应 \(\det(I-zA_2)\) 的最小模零点，即 \(z_\star=1/\rho(A_2)=1/r_2\)。而 SFT 的拓扑熵等于 \(\log\rho(A_2)\)，故得 \(\log r_2\)。证毕。
\end{proof}

\subsubsection{三阶碰撞矩与 \texorpdfstring{$\mathrm{R\acute{e}nyi}$}{Renyi}--3 投影熵率}\label{subsec:pom-s3}
二阶碰撞矩 \(S_2(m)\) 描述“成对折叠碰撞”的 \(\ell^2\) 质量；更高阶的碰撞矩把纤维分布的尖锐程度分解为一族 \(\mathrm{R\acute{e}nyi}\) 指纹。下面给出三阶（\(q=3\)）的最小三态实现，其结构与命题 \ref{prop:pom-s2-recurrence} 完全同型。

\begin{definition}[三阶折叠碰撞矩]\label{def:pom-s3}
定义三阶碰撞和
$$
S_3(m):=\sum_{x\in X_m} d_m(x)^3.
$$
等价地，从 \(\Omega_m\) 上均匀独立抽取三条窗口 \(\omega_1,\omega_2,\omega_3\)，则
$$
\mathbb{P}\bigl[\Fold_m(\omega_1)=\Fold_m(\omega_2)=\Fold_m(\omega_3)\bigr]=\frac{S_3(m)}{8^m}.
$$
\end{definition}

\begin{proposition}[三阶碰撞核的三态表示与三阶递推]\label{prop:pom-s3-recurrence}
存在一个三态的非负整数矩阵
$$
A_3:=
\begin{pmatrix}
0&0&1\\
0&1&1\\
2&3&1
\end{pmatrix},
$$
以及读出向量
$$
u:=(1,3,1)^{\mathsf T},\qquad \mathbf{1}:=(1,1,1)^{\mathsf T},
$$
使得对一切 \(m\ge 2\),
$$
S_3(m)=2\,u^{\mathsf T}A_3^{\,m-2}\mathbf{1}.
$$
从而 \(S_3(m)\) 满足精确递推（\(m\ge 5\)）
$$
S_3(m)=2S_3(m-1)+4S_3(m-2)-2S_3(m-3),
$$
并具有初值 \(S_3(2)=10,S_3(3)=26,S_3(4)=88\)。
\end{proposition}
\begin{proof}
与命题 \ref{prop:pom-s2-recurrence} 相同，本命题以\textbf{可审计构造}给出：三态非负 realization 可由小 \(m\) 的枚举数据恢复，并由脚本 \texttt{scripts/exp\_fold\_collision\_s3.py} 逐项核验（同时核验递推与读出公式）。给定 \(A_3\) 与读出向量，序列
\(

\)
自动满足由 Cayley--Hamilton 定理诱导的三阶线性递推，其系数由 \(A_3\) 的特征多项式决定。直接计算得
$$
\det(\lambda I-A_3)=\lambda^3-2\lambda^2-4\lambda+2,
$$
再乘以常数 \(2\) 并作指标平移即得所述递推式。证毕。
\end{proof}

\begin{corollary}[三阶碰撞指数与 \texorpdfstring{$h_3$}{h3}]\label{cor:pom-s3-asymptotic}
令 \(r_3\) 为多项式
$$
x^3-2x^2-4x+2=0
$$
的最大实根（\(r_3\approx 3.086130197651\)）。则
$$
S_3(m)=\Theta(r_3^m),
\qquad
\mathbb{P}\bigl[\Fold_m(\omega_1)=\Fold_m(\omega_2)=\Fold_m(\omega_3)\bigr]
=\Theta\!\left(\left(\frac{r_3}{8}\right)^m\right).

$$

=\log\!\left(\frac{8}{r_3}\right).
$$
\end{corollary}
\begin{proof}
由命题 \ref{prop:pom-s3-recurrence} 得 \(S_3(m)\) 的生成函数分母为 \(1-2z-4z^2+2z^3=\det(I-zA_3)\)，其收敛半径由该分母的最小模零点决定，等价于特征多项式的最大实根 \(r_3\)，故 \(S_3(m)=\Theta(r_3^m)\)。又
$$
\sum_{x\in X_m}\pi_m(x)^3=\sum_x \frac{d_m(x)^3}{8^m}=\frac{S_3(m)}{8^m},
$$
代入渐近即得 \(h_3=\log(8/r_3)\)。证毕。
\end{proof}

\begin{proposition}[\texorpdfstring{$h_q$}{hq} 的几何化：投影输出测度的 Rényi 维数谱]\label{prop:pom-renyi-dimension-spectrum}
沿用上文 \(\pi_m(x)=d_m(x)/2^m\) 的输出分布记号。把分辨率 \(m\) 与几何尺度对应为
\[
\varepsilon_m:=\varphi^{-m}.
\]
定义输出测度在尺度 \(\varepsilon_m\) 下的 \(q\)-阶 Rényi 维数近似为
\[
D_q(m):=\frac{1}{q-1}\cdot \frac{\log\sum_{x\in X_m}\pi_m(x)^q}{\log \varepsilon_m}
\qquad(q\ge 2).
\]
若 \(h_q=\lim_{m\to\infty}\frac{1}{m}H_q(m)\) 存在（例如当 \(S_q(m)=\Theta(r_q^m)\) 时成立），则极限
\[
D_q:=\lim_{m\to\infty}D_q(m)
\]
存在且满足闭式
\[
\boxed{\;
D_q=\frac{h_q}{(q-1)\log\varphi}\; }.
\]
\end{proposition}
\begin{proof}
由定义 \(H_q(m)=-\log\sum_x\pi_m(x)^q\) 与 \(\log\varepsilon_m=-m\log\varphi\) 得
\[
D_q(m)=\frac{1}{q-1}\cdot \frac{-H_q(m)}{-m\log\varphi}
=\frac{1}{(q-1)\log\varphi}\cdot \frac{1}{m}H_q(m).
\]
令 \(m\to\infty\) 即得结论。证毕。
\end{proof}

\begin{proposition}[普适夹逼：高阶碰撞主指数与投影熵率仅由 \texorpdfstring{$|X_m|$}{|Xm|} 与最大纤维锁定]\label{prop:pom-rq-universal-bounds}
对整数 \(q\ge 2\)，令
$$
S_q(m):=\sum_{x\in X_m} d_m(x)^q,\qquad d_m(x):=\bigl|\Fold_m^{-1}(x)\bigr|.
$$
并令 \(r_q\) 表示 \(S_q(m)\) 的指数增长率（例如定理 \ref{thm:pom-collision-kernel-ak} 给出的 Perron 根；等价地 \(r_q=\lim_{m\to\infty}S_q(m)^{1/m}\)），定义未归一化 \(q\)-阶投影熵率
$$
h_q:=\log\!\left(\frac{2^q}{r_q}\right)=q\log 2-\log r_q.
$$
则有两侧界
$$
\frac{2^q}{\varphi^{q-1}}\ \le\ r_q\ \le\ 2\,\varphi^{(q-1)/2},
$$
等价地
$$
(q-1)\log\!\left(\frac{2}{\sqrt{\varphi}}\right)\ \le\ h_q\ \le\ (q-1)\log\varphi.
$$
\end{proposition}
\begin{proof}
对任意固定 \(m\) 与 \(q\ge 2\)，由幂均值/Holder 不等式，
$$
S_q(m)=\sum_{x\in X_m}d_m(x)^q\ \ge\ \frac{\bigl(\sum_x d_m(x)\bigr)^q}{|X_m|^{\,q-1}}
=\frac{2^{qm}}{F_{m+2}^{\,q-1}},
$$
其中 \(\sum_x d_m(x)=|\Omega_m|=2^m\) 且 \(|X_m|=F_{m+2}\)。
另一方面令 \(D_m:=\max_{x\in X_m} d_m(x)\)，则
$$
S_q(m)=\sum_x d_m(x)^q\ \le\ D_m^{\,q-1}\sum_x d_m(x)=2^m D_m^{\,q-1}.
$$
两式取对数并除以 \(m\)，再令 \(m\to\infty\)，结合 Fibonacci 渐近 \(F_{m+2}\sim \varphi^{m}/\sqrt5\) 与最大纤维渐近 \(D_m=\Theta(\varphi^{m/2})\)（定理 \ref{thm:pom-max-fiber}），得到
$$
\log r_q=\lim_{m\to\infty}\frac{1}{m}\log S_q(m)\in\Bigl[q\log 2-(q-1)\log\varphi,\ \log 2+\frac{q-1}{2}\log\varphi\Bigr],
$$
从而 \(r_q\) 的界成立。代入 \(h_q=q\log 2-\log r_q\) 即得 \(h_q\) 的等价界。证毕。
\end{proof}

\begin{proposition}[极端阶端点：\texorpdfstring{$r_q^{1/q}\to\sqrt{\varphi}$}{rq^(1/q)->sqrt(phi)} 与归一化熵端点常数]\label{prop:pom-rq-qinfty-endpoint}
令 \(r_q\) 表示 \(S_q(m)\) 的指数增长率（例如由 moment-kernel 的 Perron 根给出），并记 \(\varphi\) 为黄金比例。
则当 \(q\to\infty\) 时有极限
\[
\boxed{\;
\lim_{q\to\infty} r_q^{1/q}=\sqrt{\varphi}\; }.
\]
等价地，
\[
\lim_{q\to\infty}\frac{1}{q}\,h_q
=\log\!\left(\frac{2}{\sqrt{\varphi}}\right),
\qquad
\lim_{q\to\infty}\frac{1}{q-1}\,h_q
=\log\!\left(\frac{2}{\sqrt{\varphi}}\right),
\]
从而 Rényi 维数谱端点满足
\[
\lim_{q\to\infty}D_q
=\frac{\log\!\left(2/\sqrt{\varphi}\right)}{\log\varphi}
\]
（其中 \(D_q\) 见命题 \ref{prop:pom-renyi-dimension-spectrum}）。
\par
并且对任意整数 \(q\ge 2\) 有显式误差界
\[
\boxed{\;
0\ \le\ \log\!\left(\frac{2}{\sqrt{\varphi}}\right)-\frac{h_q}{q}\ \le\ \frac{\log\varphi}{q}\; }.
\]
\end{proposition}
\begin{proof}
对任意 \(m,q\ge 2\)，由
\(
D_m:=\max_x d_m(x)
\)
得两侧夹逼
\[
D_m^q\ \le\ S_q(m)\ \le\ |X_m|\,D_m^q.
\]
两端取 \(1/(mq)\) 次方，得到
\[
D_m^{1/m}\ \le\ S_q(m)^{1/(mq)}\ \le\ D_m^{1/m}\,|X_m|^{1/(mq)}.
\]
令 \(m\to\infty\)，左端由定理 \ref{thm:pom-max-fiber} 给出
\(
\lim_{m\to\infty}D_m^{1/m}=\varphi^{1/2}
\)；
右端的因子满足
\(
\lim_{m\to\infty}|X_m|^{1/(mq)}=\varphi^{1/q}
\)
（因为 \(|X_m|=F_{m+2}\sim \varphi^m/\sqrt5\)）。
因此对每个固定 \(q\ge 2\) 有
\[
\sqrt{\varphi}\ \le\ r_q^{1/q}\ \le\ \sqrt{\varphi}\,\varphi^{1/q}.
\]
令 \(q\to\infty\) 即得 \(\lim_{q\to\infty}r_q^{1/q}=\sqrt{\varphi}\)。
最后由 \(h_q=q\log 2-\log r_q\) 与命题 \ref{prop:pom-renyi-dimension-spectrum} 的闭式，直接得到各端点极限。
\par
由 \(\sqrt{\varphi}\le r_q^{1/q}\le \sqrt{\varphi}\,\varphi^{1/q}\) 得
\[
\frac{1}{2}\log\varphi\ \le\ \frac{1}{q}\log r_q\ \le\ \frac{1}{2}\log\varphi+\frac{1}{q}\log\varphi.
\]
代入 \(\frac{h_q}{q}=\log 2-\frac{1}{q}\log r_q\)，即得
\[
0\ \le\ \log\!\left(\frac{2}{\sqrt{\varphi}}\right)-\frac{h_q}{q}\ \le\ \frac{\log\varphi}{q}.
\]
证毕。
\end{proof}

\begin{proposition}[压力的凸性与指纹的凹性]\label{prop:pom-pressure-convexity}
对任意 \(m\ge 2\)，函数 \(q\mapsto \log S_q(m)\) 在 \(q\ge 0\) 上为凸函数。
因此在 \(S_q(m)=\Theta(r_q^m)\)（例如由命题 \ref{prop:pom-moment-kernel-exists} 的 Perron 主尺度保证）时，压力
\[
P(q):=\log r_q
\]
在整数点上满足离散凸性：对任意整数 \(q_0<q_1<q_2\) 有
\[
\boxed{\;
P(q_1)\ \le\ \frac{q_2-q_1}{q_2-q_0}\,P(q_0)\ +\ \frac{q_1-q_0}{q_2-q_0}\,P(q_2)\; }.
\]
等价地，未归一化 Rényi 指纹 \(h_q=q\log 2-P(q)\) 在整数点上为凹的：
\[
\boxed{\;
h_{q_1}\ \ge\ \frac{q_2-q_1}{q_2-q_0}\,h_{q_0}\ +\ \frac{q_1-q_0}{q_2-q_0}\,h_{q_2}\; }.
\]
\end{proposition}
\begin{proof}
固定 \(m\)。取任意 \(q_0,q_2\ge 0\) 与 \(\theta\in(0,1)\)，令 \(q_1=\theta q_0+(1-\theta)q_2\)。
由 Hölder 不等式（或 log-sum-exp 的凸性）得
\[
S_{q_1}(m)=\sum_{x\in X_m}d_m(x)^{q_1}
\le
\left(\sum_{x\in X_m}d_m(x)^{q_0}\right)^{\theta}
\left(\sum_{x\in X_m}d_m(x)^{q_2}\right)^{1-\theta}
=S_{q_0}(m)^{\theta}S_{q_2}(m)^{1-\theta}.
\]
取对数即得 \(q\mapsto \log S_q(m)\) 的凸性。
\par
令 \(q_0<q_1<q_2\) 为整数并取
\(
\theta=\frac{q_2-q_1}{q_2-q_0}
\)
使得 \(q_1=\theta q_0+(1-\theta)q_2\)。
对上式取 \(\frac{1}{m}\log\) 并令 \(m\to\infty\)，得到
\[
\log r_{q_1}\le \theta\log r_{q_0}+(1-\theta)\log r_{q_2},
\]
即 \(P\) 的离散凸性。最后 \(h_q=q\log 2-P(q)\) 为线性函数减去凸函数，从而在整数点上为凹。证毕。
\end{proof}

\]

\end{remark}

\begin{theorem}[坍缩/共振判别：模态约束导致 Hankel 秩下降]\label{thm:pom-moment-resonance}

$$
S_q(m)=\sum_{\ell=-h}^{h} M_{q,\ell}(m).
$$
并假设这些模态贡献可组织为一个以 \(\{M_{q,\ell}\}\) 为坐标的 \(2h+1\) 维线性 realization（非共振情形下正是“模态闭合推进”的标准实现）。

$$

$$
其特征多项式
$$

\begin{enumerate}

\subsubsection{任意阶碰撞矩的 moment-kernel 构造与多重分形谱}\label{subsec:pom-moment-general}

\paragraph{从 \texorpdfstring{$q$}{q}-重碰撞到有限态 moment-kernel}
二阶与三阶的核心新点是：碰撞矩并非仅是一个“可枚举序列”，而是由一个\textbf{有限状态纤维积核}生成，从而完全落入 Perron--Frobenius/\(\zeta\) 的统一接口。
这一事实对任意 \(q\ge 2\) 都成立：固定 \(q\) 后，“\(q\) 条窗口折叠到同一稳定类型”的约束是一个有限记忆条件，因此可被编译为一个有限图（SFT/sofic）并由整数邻接矩阵计数。

\begin{definition}[\texorpdfstring{$q$}{q}-重碰撞关系与 moment 序列]\label{def:pom-moment-q}
对整数 \(q\ge 2\)，定义 \(q\)-重碰撞关系
$$
R_m^{(q)}:=
\{(\omega^{(1)},\dots,\omega^{(q)})\in\Omega_m^q:\ \Fold_m(\omega^{(1)})=\cdots=\Fold_m(\omega^{(q)})\},
$$
并定义 \(q\)-阶碰撞矩（moment）
$$
S_q(m):=|R_m^{(q)}|=\sum_{x\in X_m} d_m(x)^q.
$$
\end{definition}

\begin{proposition}[任意阶 moment-kernel 的存在与可执行构造]\label{prop:pom-moment-kernel-exists}
对每个固定的 \(q\ge 2\)，存在一个有限状态非负整数核（邻接矩阵）\(A_q\) 与整数读出向量 \(u_q,v_q\) 使得
$$
S_q(m)=u_q^{\mathsf T}A_q^{\,m-m_0}v_q\qquad(m\ge m_0),
$$
从而 \(\{S_q(m)\}_{m\ge m_0}\) 满足某个\textbf{有限阶整系数线性递推}。

并且 \(A_q\) 可按下述“编译路线”显式构造（因此可推演、可重写、可计算）：
\begin{enumerate}
  \item 选取实现 \(\Fold_\infty\) 的确定性在线核（延迟 \(\delta_{\min}=3\) 的同步核，附录给出其显式有限状态表示）；
  \item 取其 \(q\) 份拷贝并做同步直积，把状态扩展为 \(q\)-元组；
  \item 在每一步只保留那些使 \(q\) 份拷贝产生\textbf{相同输出位}的转移，并投影到可达的本质强连通分量；
  \item 得到的有限图即为一个 \(q\)-重碰撞的 fiber-product 核，其邻接矩阵就是所求 \(A_q\)，其路径计数即给出 \(S_q(m)\)。
\end{enumerate}
\end{proposition}

\begin{proof}
把 \(\Fold_\infty\) 视为一个确定性有限转导器：输入为窗口字母（或其由两条合法输入轨道诱导的输入字母），输出为稳定地址比特；其内部状态空间有限且具有常数延迟（命题 \ref{prop:pom-delay-min} 的在线核即是一个具体实现）。
固定 \(q\) 后，约束“\(q\) 份输出逐位相同”是一个局部同步约束：它只需在每一步比较当前输出符号是否一致，因而仍是有限状态可识别的。
因此可对该转导器做同步直积并施加“输出一致”的筛选，得到一个有限有向图 \(G^{(q)}\)。
长度 \(m\) 的合法路径条数等于 \(R_m^{(q)}\) 的计数（即 \(S_q(m)\)），从而可写成某个非负整数邻接矩阵的矩阵幂读出形式 \(u_q^{\mathsf T}A_q^{m-m_0}v_q\)。
线性递推由 Cayley--Hamilton 定理立得，且递推系数为整数（因为 \(A_q\) 为整数矩阵）。证毕。
\end{proof}

\paragraph{从 \texorpdfstring{$r_q$}{r\_q} 到压力与“纤维多重度谱”：凸共轭上界（严格）}
已核验的 \(q\)-阶碰撞核满足 \(S_q(m)=\Theta(r_q^m)\)，其中 \(r_q=\rho(A_q)\) 为 Perron 根（推论 \ref{cor:pom-s2-asymptotic}、\ref{cor:pom-s3-asymptotic}、\ref{cor:pom-s4-asymptotic} 及表 \ref{tab:fold_collision_moment_recursions}）。
因此 moment 指纹不仅给出单点常数 \(h_q=\log(2^q/r_q)\)，还诱导出一条可做 Legendre 变换的“压力曲线”，从而把“空间损失”提升为真正的谱结构。

\begin{definition}[moment 压力与归一化对数矩母函数]\label{def:pom-moment-pressure}
对 \(q\ge 2\) 定义
$$
\tau(q):=\limsup_{m\to\infty}\frac{1}{m}\log S_q(m),
$$
并定义归一化压力（即输出分布 \(\pi_m(x)=d_m(x)/2^m\) 的 \(q\)-阶对数矩母函数）
$$
P(q):=\limsup_{m\to\infty}\frac{1}{m}\log\sum_{x\in X_m}\pi_m(x)^q
=\tau(q)-q\log 2.
$$
当存在 Perron 根 \(r_q\) 使 \(S_q(m)=\Theta(r_q^m)\) 时，有 \(\tau(q)=\log r_q\)，并且 \(P(q)=\log(r_q/2^q)=-h_q\)。
\end{definition}

\begin{definition}[纤维多重度的对数谱函数]\label{def:pom-fiber-spectrum}
令
$$
\gamma_m(x):=\frac{1}{m}\log d_m(x)\qquad(x\in X_m).
$$
对 \(\gamma\ge 0\) 与 \(\varepsilon>0\)，定义“近似层”计数
$$
N_m(\gamma,\varepsilon):=\Bigl|\bigl\{x\in X_m:\ |\gamma_m(x)-\gamma|\le \varepsilon\bigr\}\Bigr|.
$$
定义纤维多重度的（上）谱函数
$$
f(\gamma):=\lim_{\varepsilon\downarrow 0}\ \limsup_{m\to\infty}\ \frac{1}{m}\log N_m(\gamma,\varepsilon).
$$
\end{definition}

\begin{proposition}[碰撞谱的几何含义：纤维指数的指数矩母函数]\label{prop:pom-fiber-index-cgf}
沿用 \(\pi_m(x)=d_m(x)/2^m\) 的输出分布记号，并把 \(\gamma_m(X)\) 视为在 \(X_m\) 上以 \(\pi_m\) 抽样的随机变量。
则对任意整数 \(q\ge 2\) 有严格恒等式
\[
\boxed{\;
\mathbb{E}_{\pi_m}\!\left[\exp\!\bigl((q-1)m\,\gamma_m(X)\bigr)\right]
=\frac{S_q(m)}{2^m}\; }.
\]
从而
\[
\frac{1}{m}\log \mathbb{E}_{\pi_m}\!\left[\exp\!\bigl((q-1)m\,\gamma_m(X)\bigr)\right]
=\frac{1}{m}\log S_q(m)-\log 2.
\]
特别地，若存在 Perron 主尺度 \(S_q(m)=\Theta(r_q^m)\)（例如由 moment-kernel 实现给出），则极限累积母函数在整点 \(t=q-1\) 上满足
\[
\lim_{m\to\infty}\frac{1}{m}\log \mathbb{E}_{\pi_m}\!\left[e^{t m\,\gamma_m(X)}\right]
=\log r_{t+1}-\log 2,
\qquad
h_{t+1}=(t+1)\log 2-\log r_{t+1}.
\]
\end{proposition}
\begin{proof}
按定义展开并用 \(\pi_m(x)=d_m(x)/2^m\)：
\[
\mathbb{E}_{\pi_m}\!\left[\exp\!\bigl((q-1)m\,\gamma_m(X)\bigr)\right]
=\sum_{x\in X_m}\pi_m(x)\,e^{(q-1)\log d_m(x)}
=\sum_{x\in X_m}\frac{d_m(x)}{2^m}\,d_m(x)^{q-1}
=\frac{1}{2^m}\sum_{x\in X_m}d_m(x)^q
=\frac{S_q(m)}{2^m}.
\]
取对数并除以 \(m\) 得第二式。最后一段由 \(S_q(m)=\Theta(r_q^m)\) 的指数尺度与 \(h_q=\log(2^q/r_q)\) 的定义直接推出。证毕。
\end{proof}

\begin{remark}[大偏差/多分形反演（作为可检验目标）]\label{rem:pom-fiber-index-ldp}
命题 \ref{prop:pom-fiber-index-cgf} 表明：碰撞谱 \(q\mapsto h_q\) 等价于纤维指数 \(\gamma_m\) 的指数矩母函数在整点的增长率。
若进一步能把 \(q\)-重 moment-kernel 的“整数参数”推广为实参数 \(t\ge 0\) 的加权核，从而得到一个凸函数
\(
\Lambda(t):=\lim_{m\to\infty}\frac{1}{m}\log \mathbb{E}_{\pi_m}[e^{t m\,\gamma_m(X)}]
\)
并满足标准的可微/本质严格凸条件，则 Gartner--Ellis 定理给出 \(\gamma_m\) 的大偏差率函数
\[
I(\gamma)=\sup_{t\ge 0}\bigl(t\gamma-\Lambda(t)\bigr).
\]
此时 \(I\)（从而纤维厚度谱）可由碰撞谱“反演”得到，构成“从 \(h_q\) 指纹回推纤维几何”的可证伪计算目标。
\end{remark}

\begin{proposition}[Legendre 上界：谱由 moment 压力控制]\label{prop:pom-spectrum-legendre-upper}
对任意 \(q\ge 2\) 与任意 \(\gamma\ge 0\)，有上界
$$
f(\gamma)\ \le\ \inf_{q\ge 2}\bigl(\tau(q)-q\gamma\bigr).
$$
特别地，在 \(\tau(q)=\log r_q\) 的情形下，
$$
f(\gamma)\ \le\ \inf_{q\ge 2}\bigl(\log r_q-q\gamma\bigr).
$$
\end{proposition}

\begin{proof}
固定 \(q\ge 2\) 与 \(\gamma,\varepsilon\)。对任意满足 \(|\gamma_m(x)-\gamma|\le\varepsilon\) 的 \(x\)，有
$$
d_m(x)^q=\exp(qm\gamma_m(x))\ \ge\ \exp\bigl(qm(\gamma-\varepsilon)\bigr).
$$
因此
$$
S_q(m)=\sum_{x\in X_m} d_m(x)^q
\ \ge\ N_m(\gamma,\varepsilon)\,\exp\bigl(qm(\gamma-\varepsilon)\bigr),
$$
即
$$
\frac{1}{m}\log N_m(\gamma,\varepsilon)\ \le\ \frac{1}{m}\log S_q(m)\ -\ q(\gamma-\varepsilon).
$$
取 \(m\to\infty\) 的 \(\limsup\) 并再令 \(\varepsilon\downarrow 0\)，得
$$
f(\gamma)\le \tau(q)-q\gamma.
$$
对所有 \(q\ge 2\) 取下确界即得结论。证毕。
\end{proof}

\begin{remark}[多重分形形式（作为可检验目标）]\label{rem:pom-spectrum-formalism}
命题 \ref{prop:pom-spectrum-legendre-upper} 给出严格的上界：moment 压力曲线 \(\tau(q)\) 的凸共轭控制了纤维多重度的（上）谱。
在标准热力学形式条件下（例如存在与 \(\tau\) 对偶的 Gibbs 结构并具有足够的准乘性），往往可得到相反方向的不等式从而实现等号，即“谱 \(f\) 与压力 \(\tau\) 互为 Legendre 变换”。
在本文框架中，这一等号可被视为一条可证伪的结构目标：它把“空间损失是一条谱”转化为“\(\{r_q\}\) 族决定谱形”的可计算命题。
\end{remark}

\subsubsection{在线延迟的不可约下界}\label{subsec:pom-delay-lb}
\begin{proposition}[单趟在线延迟 $3$ 的最小性]\label{prop:pom-delay-min}
任何实现 $\Fold_\infty:\{0,1,2\}^{\ZZ}\to X_\infty$ 的单趟在线转导器，其最小延迟满足
\[
\delta_{\min}(\Fold_\infty)\ge 3.
\]
结合定理 \ref{thm:online-delay-fold}，可得 $\delta_{\min}(\Fold_\infty)=3$。
\end{proposition}
\begin{proof}
在小节 \ref{subsec:online-delay} 的在线核中，输出满足
\(
e=\lfloor \varphi R(s)+d\varphi^{-3}\rfloor
\)，其中 $s$ 为长度 $3$ 的同步状态、$d\in\{0,1,2\}$ 为当前输入。取状态 $s=002$（其属于状态集 $Q$），则
\[
R(s)=2\varphi^{-3},\qquad
e=\left\lfloor 2\varphi^{-2}+d\varphi^{-3}\right\rfloor.
\]
当 $d=0$ 时 $e=0$，当 $d=1$ 时 $e=1$，故该位输出确实依赖于“向前第 $3$ 位”输入。任何延迟 $2$ 的单趟在线机在输出此位时尚未读到该输入，必产生冲突，从而不可能实现 $\Fold_\infty$。证毕。
\end{proof}

\begin{remark}[空间缺陷与时间缺陷的正交性]\label{rem:pom-space-time-orth}
最大纤维多重度给出“值投影”的最坏空间缺陷（定理 \ref{thm:pom-max-fiber}），而单趟在线延迟给出“步骤投影”的不可约时间缺陷（命题 \ref{prop:pom-delay-min}）。两者来源于不同的投影门，因而互不替代：增大空间并不能把 $\delta_{\min}$ 压到 $2$，而压低延迟也无法消除 $d_m$ 的纤维多重度。
\end{remark}

\begin{definition}[空间—时间—谱三指纹]\label{def:pom-fingerprint}
对实现规范投影 $P_Z$ 的归一化核 $K$，记其分辨率截断为 $K_m:\Omega_m\to X_m$，并令
\(
d^{(K)}_m(x):=|K_m^{-1}(x)|
\)。
定义：
\[
S_K(m):=\max_{x}\log d^{(K)}_m(x)\quad(\text{空间指纹}),
\qquad
T_K:=\delta_{\min}(K)\quad(\text{时间指纹}),
\]
并以
\[
\Pi_K:=\{\zeta_K,\ \log\mathfrak{M}_K,\ \rho/\lambda,\ \text{Dirichlet--Mertens 常数张量等}\}
\]
作为谱指纹族（见第 \ref{sec:zeta-finite-part} 节）。三者分别量化“值投影损失”“在线不可并行延迟”与“机制谱细节”，在同一投影框架下互相正交。
\end{definition}

\begin{remark}[Fold 的三指纹]\label{rem:pom-fold-fingerprint}
对 Zeckendorf 归一化核 $K=\Fold_\infty$，由定理 \ref{thm:pom-max-fiber} 得 $S_K(m)=\log D_m$，由命题 \ref{prop:pom-delay-min} 得 $T_K=3$，而 $\Pi_K$ 可由 10/40 状态同步核的 $\zeta$、$\log\mathfrak{M}$ 与 Dirichlet--Mertens 张量显式抽取（见第 \ref{sec:zeta-finite-part} 节与附录 \ref{app:sync-kernel-basic}、\ref{app:real-input-40-abel}）。因此“空间(值)$\times$时间(步骤)$\times$谱(机制)”形成可比较的统一量化框架。
\end{remark}

\begin{theorem}[统一闭环：在线核—碰撞矩核—\texorpdfstring{$\zeta$/primitive}{zeta/primitive}—几何维数谱]\label{thm:pom-unified-closure}
以 $K=\Fold_\infty$ 为规范投影核。则下述链路在同一套“有限核 $\Rightarrow$ 谱 $\Rightarrow$ 轨道/有限部 $\Rightarrow$ 几何谱”的接口下闭合：
\begin{enumerate}
  \item \textbf{（在线核与时间缺陷）} $\Fold_\infty$ 存在单趟在线同步实现（附录 \ref{app:sync-kernel-basic} 给出显式 10 状态核），并且最小延迟刚性为 $\delta_{\min}=3$（命题 \ref{prop:pom-delay-min}）。
  \item \textbf{（任意阶 moment-kernel 与熵率存在）} 对任意整数 $q\ge 2$，$q$-重碰撞矩 $S_q(m)$ 可由有限状态非负整数核 $A_q$ 计数（命题 \ref{prop:pom-moment-kernel-exists}），从而 $S_q$ 为 \(\mathbb N\)-有理级数并满足有限阶整系数递推；其 Perron 根 $r_q=\rho(A_q)$ 给出指数尺度 $S_q(m)=\Theta(r_q^m)$，并诱导投影熵率
  \[
  h_q=\log\!\left(\frac{2^q}{r_q}\right)
  \]
  的存在性与闭式接口（定义 \ref{def:pom-moment-pressure}）。
  \item \textbf{（闭合维数的内禀不变量）} 在已核验范围 $q\le 8$，序列 $S_q$ 的最小线性实现维数由 Hankel 秩锁定：\(\mathrm{rank}(H)\ge 2\lfloor q/2\rfloor+1\)（定理 \ref{thm:pom-moment-hankel}），并与递推阶数一致从而在该范围内取等号（推论 \ref{cor:pom-moment-minreal}）。
  \item \textbf{（共振/坍缩的机制判别）} 当递推阶数低于 \(2\lfloor q/2\rfloor+1\) 时，差分模态满足线性约束并导致 Hankel 秩下降（定理 \ref{thm:pom-moment-resonance}）；在 \(q=9,\dots,16\) 的已核验递推中该共振被触发。
  \item \textbf{（几何化：Rényi 维数谱）} 当 $h_q$ 存在时，输出测度的 Rényi 维数谱由 $h_q$ 闭式决定：
  \[
  D_q=\frac{h_q}{(q-1)\log\varphi}
  \]
  （命题 \ref{prop:pom-renyi-dimension-spectrum}）。
  \item \textbf{（机制核的周期点闭式与 primitive 抽取）} 在真实算术输入约束下，同步核与输入记忆合成 40 状态机制核 $M$（附录 \ref{app:real-input-40-abel}；构造见命题 \ref{prop:real-input-40-essential-20} 起），其 $\zeta$-函数为显式有理式并具有谱因子分解（命题 \ref{prop:real-input-40-fib-tensor} 与推论 \ref{cor:real-input-40-spectrum-decomp}）。因此周期点计数 $a_n=\Tr(M^n)$ 具有闭式表达（同上证明段落），primitive 轨道数则由 Möbius 反演
  \[
  p_n=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}
  \]
  直接抽取，从而把“prime-orbit/有限部常数/Dirichlet--Mertens 张量”等谱对象纳入同一可计算接口（第 \ref{sec:zeta-finite-part} 节与附录群）。
\end{enumerate}
\end{theorem}

\subsection{新结论簇 I：熵饱和与最大熵提升}\label{subsec:pom-entropy-maxent}
\begin{definition}[窗口值与外表值]\label{def:pom-window-value}
对 $\omega=\omega_1\cdots\omega_m\in\Omega_m$，定义窗口值
\[
N(\omega):=\sum_{k=1}^m \omega_k F_{k+1}.
\]
对 $x=x_1\cdots x_m\in X_m$，定义外表值
\[
V_m(x):=\sum_{k=1}^m x_k F_{k+1}.
\]
\end{definition}

\begin{lemma}[单隐藏位分解]\label{lem:pom-one-bit}
对任意 $\omega\in\Omega_m$，存在唯一 $b(\omega)\in\{0,1\}$ 使
\[
N(\omega)=V_m(\Fold_m(\omega)) + b(\omega)\,F_{m+2}.
\]
\end{lemma}
\begin{proof}
由 $\sum_{k=1}^m F_{k+1}=F_{m+3}-2$ 得 $0\le N(\omega)<F_{m+3}$。因此 $N(\omega)$ 的 Zeckendorf 表示中不可能出现权重 $F_{m+3}$，其非零位仅可能到 $F_{m+2}$。由 $\Fold_m$ 的定义，$\Fold_m(\omega)$ 给出前 $m$ 位，余下唯一可能的高位即 $F_{m+2}$，于是存在唯一 $b(\omega)\in\{0,1\}$ 使等式成立。证毕。
\end{proof}

\begin{proposition}[纤维自由度的二层分解]\label{prop:pom-fiber-decompose}
对任意 $x\in X_m$，记 $I(x)$ 为 $x$ 中可执行逆代换 $100\mapsto 011$ 的局部位点集合（以长度 $3$ 窗口计），并要求所选位点两两不相邻以避免重叠。则任意 $\omega\in\Fold_m^{-1}(x)$ 可视为“一个余量位 $b\in\{0,1\}$”与“一组不相邻逆代换位点”的组合；因此
\[
d_m(x)\le 2\cdot \#\{\text{路径图 }P_{|I(x)|}\text{ 的独立集}\}
\le 2F_{\lfloor m/2\rfloor+2}.
\]
\end{proposition}
\begin{proof}
由引理 \ref{lem:pom-one-bit}，给定 $\omega$ 的超出前 $m$ 位的信息至多是一位 $b$。其余自由度来自逆代换 $100\mapsto 011$ 的选择：当两次逆代换发生重叠时会产生非二元符号，因此合法选择只能来自一条线性链上的不相邻位点集合。路径图独立集计数即为 Fibonacci 数，从而得到上述上界。证毕。
\end{proof}

\begin{theorem}[一步可判别与右解析性]\label{thm:pom-right-resolving}
对任意 $m\ge 2$ 与任意 $v\in V_m=\{0,1\}^{m-1}$，有
\[
\Fold_m(v0)\neq \Fold_m(v1).
\]
因此标记图 $G_m$（定义 \ref{def:G_m}）是右解析（right-resolving）的：同一顶点出发的两条出边具有不同标记。
\end{theorem}
\begin{proof}
记 $\omega_0=v0$、$\omega_1=v1$。则 $N(\omega_1)-N(\omega_0)=F_{m+1}$. 若 $\Fold_m(\omega_0)=\Fold_m(\omega_1)$，由引理 \ref{lem:pom-one-bit} 得
\[
N(\omega_1)-N(\omega_0)\in\{0,F_{m+2}\},
\]
与 $F_{m+1}\neq F_{m+2}$ 矛盾。故两者标记不同，右解析性随之成立。证毕。
\end{proof}

\begin{theorem}[一步可判别与左解析性]\label{thm:pom-left-resolving}
对任意 $m\ge 2$ 与任意 $v\in V_m=\{0,1\}^{m-1}$，有
\[
\Fold_m(0v)\neq \Fold_m(1v).
\]
因此对任意固定的长度 $m-1$ \emph{后缀}，在其左侧补 $0/1$ 的读出结果可一步判别；换言之，$\Fold_m$ 对“首位未知”的二选一也是严格可判别的。
\end{theorem}
\begin{proof}
记 $\omega_0=0v$、$\omega_1=1v$。则 $N(\omega_1)-N(\omega_0)=F_2=1$。
若 $\Fold_m(\omega_0)=\Fold_m(\omega_1)$，由引理 \ref{lem:pom-one-bit} 得
\[
N(\omega_1)-N(\omega_0)\in\{0,F_{m+2}\},
\]
而当 $m\ge 2$ 时 $F_{m+2}\ge 3$，与 $N(\omega_1)-N(\omega_0)=1$ 矛盾。故两者读出不同。证毕。
\end{proof}

\begin{theorem}[熵饱和：折叠不改变信息率]\label{thm:pom-entropy-saturation}
对任意 $m\ge 2$，像空间 $Y_m=\Phi_m(\{0,1\}^{\ZZ})$ 满足
\[
h_{\mathrm{top}}(Y_m)=\log 2.
\]
\end{theorem}
\begin{proof}
由于 $\Phi_m$ 是滑动块因子，必有 $h_{\mathrm{top}}(Y_m)\le \log 2$。另一方面，$G_m$ 每个顶点有两条出边，总长度为 $n$ 的路径数为 $2^{m-1}\cdot 2^{n}$。由定理 \ref{thm:pom-right-resolving}，在固定起点下标记序列决定唯一路径，因此长度 $n$ 的标记词数至少为 $2^{n}$。于是
\[
\log 2\le h_{\mathrm{top}}(Y_m)\le \log 2,
\]
结论成立。证毕。
\end{proof}

\begin{remark}[有限缺陷：从纤维多重度到有限部常数]\label{rem:pom-finite-defect}
熵饱和表明折叠在信息率意义下不压缩；其“丢失”体现为\textbf{有限缺陷}而非指数尺度。具体而言，给定标记序列 $y\in Y_m$，其前像路径数由起始顶点选择给出，故
\[
|\Phi_m^{-1}(y)|\le |V_m|=2^{m-1}.
\]
更强地，该“有限缺陷”可被提升为一个显式的缺陷寄存器：令
\[
D_m:=V_m=\{0,1\}^{m-1}.
\]
定义映射
\[
\Psi_m:\{0,1\}^{\ZZ}\to Y_m\times D_m,\qquad
\Psi_m(s):=\bigl(\Phi_m(s),\,s_0s_1\cdots s_{m-2}\bigr).
\]
则 $\Psi_m$ 为单射：给定 $y=\Phi_m(s)$ 与缺陷寄存器 $\delta:=s_0\cdots s_{m-2}$，可用定理 \ref{thm:pom-right-resolving} 逐步恢复所有 $t\ge 0$ 的后继比特（由窗口末位的二选一可判别），并用定理 \ref{thm:pom-left-resolving} 逐步恢复所有 $t<0$ 的前驱比特（由窗口首位的二选一可判别），从而恢复整条输入序列 $s$。
因此每条 $y\in Y_m$ 的前像数至多为 $|D_m|=2^{m-1}$，并且“折叠丢失”可被理解为一个有限的初态/横截面寄存器，而非熵率损失。

这种有限缺陷可由纤维多重度 $d_m(x)$ 与条件期望 $E_m$ 的块平均结构刻画（命题 \ref{prop:fold-conditional-expectation}），并与 Abel/Hadamard 有限部常数形成互补的“常数级指纹”（定义 \ref{def:abel-finite-part} 及相关附录接口）。因此折叠的“丢失”属于有限分辨率残差，而非熵率损失。
\end{remark}

\begin{remark}[纤维平均与有限部常数的同一接口]\label{rem:pom-em-logm}
窗口尺度上，$E_m$ 通过 $d_m$ 对纤维做平均，压缩的是一次投影的不可观测自由度；轨道尺度上，Abel/Hadamard 有限部常数
\[
\log\mathfrak{M}(\theta)
=
\log C(\theta)
\;+\;
\sum_{k\ge 2}\frac{\mu(k)}{k}\log\zeta_\theta\!\bigl(z_\star(\theta)^k\bigr)
\]
（定义 \ref{def:abel-finite-part} 与命题 \ref{prop:abel-finite-part-mobius-zeta}）压缩的是 primitive 轨道乘积在无限尺度下的有限部漂移。因而 $E_m$ 与 $\log\mathfrak{M}$ 是同一“信息丢失接口”的微观/宏观两种尺度：前者是窗口纤维平均，后者是沿循环重复观测后的重整化常数；其导出的 $\mathsf{M}(\theta)=\gamma+\log\mathfrak{M}(\theta)$ 与 $P(\theta)$ 共同形成主尺度与有限部的双指纹（见 \ref{subsec:chi-ldp}）。
\end{remark}

\begin{definition}[推前与纤维均匀提升]\label{def:pom-maxent-lift}
把 $\mathcal{A}_m\simeq\mathbb{C}^{\Omega_m}$、$\mathcal{B}_m\simeq\mathbb{C}^{X_m}$ 识别为有限集上的函数代数（见命题 \ref{prop:fold-pushforward-lift}）。对任意分布 $\pi\in\Delta(X_m)$，定义纤维均匀提升
\[
\widetilde\pi(a):=\frac{\pi(\Fold_m(a))}{d_m(\Fold_m(a))},\qquad a\in\Omega_m.
\]
则 $(\Fold_m)_*\widetilde\pi=\pi$。
\end{definition}

\begin{theorem}[最大熵提升的唯一性]\label{thm:pom-maxent-lift}
固定任意 $\pi\in\Delta(X_m)$，在所有满足 $(\Fold_m)_*\mu=\pi$ 的分布 $\mu\in\Delta(\Omega_m)$ 中，Shannon 熵
\[
H(\mu):=-\sum_{a\in\Omega_m}\mu(a)\log\mu(a)
\]
的最大值唯一且由 $\widetilde\pi$ 取得。并且
\[
H(\widetilde\pi)=H(\pi)+\mathbb{E}_{\pi}[\log d_m(X)].
\]
\end{theorem}
\begin{proof}
固定 $x\in X_m$ 时，纤维 $\Fold_m^{-1}(x)$ 上的总质量为 $\pi(x)$。在该纤维内，熵在“均匀分配”处唯一取最大值（严格凹性）。对各纤维求和得到全局最大值与唯一性。直接代入 $\widetilde\pi$ 得
\[
H(\widetilde\pi)
=-\sum_{x\in X_m}d_m(x)\frac{\pi(x)}{d_m(x)}\log\!\left(\frac{\pi(x)}{d_m(x)}\right)
=H(\pi)+\sum_{x\in X_m}\pi(x)\log d_m(x).
\]
证毕。
\end{proof}

\begin{definition}[热力学缺陷与空间投影标量]\label{def:pom-kappa}
对任意分布 $\pi\in\Delta(X_m)$，定义
\[
\kappa_m(\pi):=\mathbb{E}_{\pi}[\log d_m(X)].
\]
由定理 \ref{thm:pom-maxent-lift}，$\kappa_m(\pi)=H(\widetilde\pi)-H(\pi)$，因此 $\kappa_m$ 给出“折叠纤维被压掉的自由度”的可观测标量刻画。
\end{definition}

\begin{remark}[窗口缺陷与有限部常数的同型]\label{rem:pom-kappa-logm}
窗口尺度上，$\kappa_m(\pi)$ 是一次折叠投影的对数纤维体积缺陷；轨道尺度上，$\log\mathfrak{M}(\theta)$ 是 primitive 轨道乘积在无限尺度下的 Möbius--Abel 正则化有限部（定义 \ref{def:abel-finite-part}）。二者分别给出微观/宏观的“纤维缺陷”接口，为跨尺度缺陷泛函的对齐提供了统一口径。
\end{remark}

\begin{remark}[与条件期望的对偶性]
纤维均匀提升 $\widetilde\pi$ 与条件期望 $E_m$ 构成对偶接口：$E_m$ 在观测量层面是“纤维平均”，而 $\widetilde\pi$ 在概率层面是“纤维均匀化”。二者分别给出折叠的算子化与信息论刻画。
\end{remark}

\begin{remark}[延迟归一化推迟律]\label{rem:pom-lazy-fold}
若中间演化 $U_t$ 在观测层不区分折叠纤维，即对任意 $g\in\mathcal{A}_m$ 有 $g\circ U_t\in\mathcal{B}_m$ 等价于 $g=\bar g\circ\Fold_m$（命题 \ref{prop:pom-fiber-invariance}），则

\]
因此在此类“纤维不区分”的演化段内，多次 $P_Z$ 可合并为末端一次，实现规范投影的推迟执行。
\end{remark}

\subsection{新结论簇 II：商余作为平移群胚横截面}\label{subsec:pom-division-groupoid}
\begin{definition}[平移群胚与基本域]\label{def:pom-translation-groupoid}
固定整数 $b>0$，令 $\mathbb{Z}$ 作用于 $\mathbb{Z}$：
\[
k\cdot a:=a-kb.
\]
其轨道即同余类 $a\bmod b$。取基本域
\[
\mathcal F_b:=\{0,1,\dots,b-1\},
\]
并定义投影
\[
\Pi_b(a):=\text{唯一满足 }a\equiv r\pmod b\text{ 且 }r\in\mathcal F_b\text{ 的 }r。
\]
\end{definition}

\begin{theorem}[商余作为横截面投影]\label{thm:pom-division-section}
对任意 $a\in\mathbb{Z}$，存在唯一 $(q,r)\in\mathbb{Z}\times\mathcal F_b$ 使
\[
a=qb+r,\qquad r=\Pi_b(a).
\]
其中 $q$ 是轨道坐标，$r$ 是沿平移轨道投影到横截面的像。
\end{theorem}
\begin{proof}
由整数除法算法与同余类代表的唯一性得到。平移群作用的轨道等于同余类，$\mathcal F_b$ 为其横截面，故投影 $\Pi_b$ 给出唯一 $r$，并据此确定 $q$。证毕。
\end{proof}

\begin{corollary}[中心余数横截面]\label{cor:pom-symmetric-remainder}
若改用对称基本域
\[
\mathcal F_b^{\mathrm{sym}}:=\left\{-\left\lfloor\frac{b}{2}\right\rfloor,\dots,\left\lceil\frac{b}{2}\right\rceil-1\right\},
\]
则仍有唯一分解 $a=qb+r$ 且 $r\in\mathcal F_b^{\mathrm{sym}}$。该选择等价于改变横截面而非改变本体演化，可将部分“区间校正”合并到单次序投影中。
\end{corollary}

\begin{remark}[与 Fold 的群胚类比]
定理 \ref{thm:pom-division-section} 与定理 \ref{thm:zeckendorf-transversal} 在结构上同型：二者均为“轨道 = 值纤维/同余类，横截面 = 规范代表”的投影机制；商 $q$ 与折叠残差的群胚坐标具有同一语义地位。
\end{remark}

\subsection{新结论簇 III：包含商余的扩展最小原语集}\label{subsec:pom-primitive-extended}
\begin{theorem}[扩展最小原语集]\label{thm:pom-extended-primitives}
在配置层，$\oplus,\otimes,\mathrm{div},\mathrm{quo},\mathrm{rem}$ 可由以下原语生成：
\begin{enumerate}
  \item 逐点叠加（自由交换幺半群结构）；
  \item Fibonacci 群胚生成元（值保持局部可逆重写）；
  \item 值纤维横截面投影 $\Fold_\infty$（规范投影）；
  \item 平移群胚横截面投影 $\Pi_b$（序/区间投影）。
\end{enumerate}
在过程层，等价表述为：仅以复合 $\circ$ 作为底层原语，再配合值投影 $\Fold_\infty$ 与序投影 $\Pi_b$ 把结果落回外表；原子投影 $P_{\mathrm{prim}}$ 作为谱读出接口独立施加。
\end{theorem}
\begin{proof}
加法与乘法由定理 \ref{thm:composition-two-layer} 与 \ref{thm:arith-composition} 的过程层读出实现，并由 $\Fold_\infty$ 落回稳定地址。商余由定理 \ref{thm:pom-division-section} 作为平移群胚横截面投影给出。其余运算由上述原语的组合闭包得到。证毕。
\end{proof}

\subsection{逆极限自然性与纤维不变量判别}\label{subsec:pom-new-results}
\subsubsection{跨分辨率一致性的 Gauge 原理：fold-aware restriction}\label{subsubsec:pom-gauge}
有限窗口微态空间 $\Omega_m=\{0,1\}^m$ 与稳定类型空间 $X_m$ 的角色并不对称：$X_m$ 在截断映射 $\pi_{m+1\to m}$ 下形成逆系统，而 $\Omega_m$ 在朴素截断下并不携带与折叠兼容的逆系统结构。该差异正是“跨分辨率一致性必须先规范化”的结构来源。

\begin{definition}[朴素截断与折叠感知限制]\label{def:pom-fold-aware-restriction}
设 $\tau_{m+1\to m}:\Omega_{m+1}\to\Omega_m$ 为朴素截断（丢弃最末一位；等价地取长度 $m$ 前缀）。定义折叠感知限制（fold-aware restriction）
$$
r^{\mathrm{e}}_{m+1\to m}:\Omega_{m+1}\to X_m,
\qquad
r^{\mathrm{e}}_{m+1\to m}:=\pi_{m+1\to m}\circ\Fold_{m+1}.
$$
\end{definition}

\begin{proposition}[朴素截断与折叠一般不交换]\label{prop:pom-truncation-not-commute}
一般地，图表
$$
\Omega_{m+1}\xrightarrow{\ \tau_{m+1\to m}\ }\Omega_m\xrightarrow{\ \Fold_m\ }X_m
$$
与
$$
\Omega_{m+1}\xrightarrow{\ \Fold_{m+1}\ }X_{m+1}\xrightarrow{\ \pi_{m+1\to m}\ }X_m
$$
并不一致，即通常存在 $\omega\in\Omega_{m+1}$ 使得
$$
\Fold_m(\tau_{m+1\to m}(\omega))\neq \pi_{m+1\to m}(\Fold_{m+1}(\omega)).
$$
\end{proposition}
\begin{proof}
折叠的局部重写（例如 $011\mapsto 100$）会产生跨位传播：被丢弃的末位可能决定某次局部重写是否触发，从而改变长度 $m$ 前缀在规范化后的形态。因此“先截断再折叠”与“先折叠再截断”在一般情形下不一致。证毕。
\end{proof}

\begin{proposition}[稳定逆系统的自然性：先折叠再限制]\label{prop:pom-gauge-natural}
对任意 $m\ge 1$，折叠感知限制满足
$$
r^{\mathrm{e}}_{m+1\to m}
=
\pi_{m+1\to m}\circ \Fold_{m+1},
$$
从而在稳定层形成严格可交换的跨分辨率图表
$$
\Omega_{m+1}\xrightarrow{\ \Fold_{m+1}\ }X_{m+1}\xrightarrow{\ \pi_{m+1\to m}\ }X_m
\qquad\text{与}\qquad
\Omega_{m+1}\xrightarrow{\ r^{\mathrm{e}}_{m+1\to m}\ }X_m
$$
一致；并且对任意 $m_2\ge m_1$ 有兼容性
$$
\text{令 }r^{\mathrm{e}}_{m_2\to m_1}:=\pi_{m_2\to m_1}\circ\Fold_{m_2},
\ \text{则对任意 }m_3\ge m_2\ge m_1\text{ 有 }r^{\mathrm{e}}_{m_3\to m_1}=\pi_{m_2\to m_1}\circ r^{\mathrm{e}}_{m_3\to m_2},
$$
因而“同一对象的跨分辨率坐标”只能在 $X_\infty\simeq\varprojlim X_m$ 的逆系统口径下被谈论。
\end{proposition}
\begin{proof}
第一式为定义 \ref{def:pom-fold-aware-restriction} 的直接展开。兼容性由稳定层的逆系统映射满足的泛性质给出（定理 \ref{thm:pom-inverse-limit}），而折叠后的坐标才是逆系统对象。证毕。
\end{proof}

\begin{definition}[分辨率反射子（跨分辨率纤维均匀化）]\label{def:pom-resolution-reflector}
固定顶层分辨率 $M$，并把微态空间取为 $\Omega_M$。
对任意 $1\le m\le M$，以折叠感知限制
\(
r^{\mathrm{e}}_{M\to m}=\pi_{M\to m}\circ \Fold_M:\Omega_M\to X_m
\)
为 coarse-graining 映射，定义推前算子
\[
P_{M\to m}:=(r^{\mathrm{e}}_{M\to m})_*:\Delta(\Omega_M)\to\Delta(X_m),
\]
并记其纤维基数为
\[
d_{M\to m}(x):=\bigl|(r^{\mathrm{e}}_{M\to m})^{-1}(x)\bigr|\qquad(x\in X_m).
\]
定义纤维均匀提升算子
\[
Q_{M\to m}:\Delta(X_m)\to\Delta(\Omega_M),
\qquad
(Q_{M\to m}\pi)(\omega):=\frac{\pi(r^{\mathrm{e}}_{M\to m}(\omega))}{d_{M\to m}(r^{\mathrm{e}}_{M\to m}(\omega))}.
\]
则 $P_{M\to m}\circ Q_{M\to m}=\mathrm{id}_{\Delta(X_m)}$。
对任意 $\mu\in\Delta(\Omega_M)$，定义其分辨率 $m$ 的规范化反射（纤维均匀化）为
\[
\mu^{(m)}:=Q_{M\to m}P_{M\to m}\mu\in\Delta(\Omega_M).
\]
当 $m=M$ 时，上述构造退化为小节 \ref{subsec:pom-information-ledger} 的 $P_M,Q_M$ 与 $\mu^{e}$（仅记号不同）。
\end{definition}

\begin{theorem}[KL-毕达哥拉斯：分辨率塔上的残差可加]\label{thm:pom-kl-pythagoras-tower}
沿用定义 \ref{def:pom-resolution-reflector}，并以定理 \ref{thm:pom-kl-ledger} 的口径记 KL 散度。
则对任意 $M\ge m_2\ge m_1\ge 1$ 与任意 $\mu\in\Delta(\Omega_M)$，成立严格恒等式
\[
\boxed{\;
D_{\mathrm{KL}}(\mu\|\mu^{(m_1)})
=D_{\mathrm{KL}}(\mu\|\mu^{(m_2)})
+D_{\mathrm{KL}}(\mu^{(m_2)}\|\mu^{(m_1)})\; }.
\]
\end{theorem}
\begin{proof}
由 KL 定义，
\[
D_{\mathrm{KL}}(\mu\|\mu^{(m_1)})-D_{\mathrm{KL}}(\mu\|\mu^{(m_2)})
=\sum_{\omega\in\Omega_M}\mu(\omega)\log\frac{\mu^{(m_2)}(\omega)}{\mu^{(m_1)}(\omega)}.
\]
对固定的 $y\in X_{m_2}$，在纤维 $(r^{\mathrm{e}}_{M\to m_2})^{-1}(y)$ 上，上式中的比值
\(
\mu^{(m_2)}(\omega)/\mu^{(m_1)}(\omega)
\)
为常数（因为 $\mu^{(m)}$ 在每条 $r^{\mathrm{e}}_{M\to m}$-纤维上均匀）。
因此按 $y$ 分组可得
\[
\sum_{\omega\in\Omega_M}\mu(\omega)\log\frac{\mu^{(m_2)}(\omega)}{\mu^{(m_1)}(\omega)}
=\sum_{y\in X_{m_2}}(P_{M\to m_2}\mu)(y)\,
\log\frac{\mu^{(m_2)}(\omega_y)}{\mu^{(m_1)}(\omega_y)},
\]
其中 $\omega_y$ 为该纤维内任意代表点。
另一方面，由定义 \(\mu^{(m_2)}=Q_{M\to m_2}P_{M\to m_2}\mu\) 得
\(
P_{M\to m_2}\mu^{(m_2)}=P_{M\to m_2}\mu
\)，
从而上式右端可将系数 $(P_{M\to m_2}\mu)(y)$ 替换为 $(P_{M\to m_2}\mu^{(m_2)})(y)$。
再将分组表达“反展开”回 $\Omega_M$，即得
\[
\sum_{\omega\in\Omega_M}\mu(\omega)\log\frac{\mu^{(m_2)}(\omega)}{\mu^{(m_1)}(\omega)}
=\sum_{\omega\in\Omega_M}\mu^{(m_2)}(\omega)\log\frac{\mu^{(m_2)}(\omega)}{\mu^{(m_1)}(\omega)}
=D_{\mathrm{KL}}(\mu^{(m_2)}\|\mu^{(m_1)}).
\]
移项即得结论。证毕。
\end{proof}

\begin{corollary}[跨分辨率的信息守恒账本：望远镜分解]\label{cor:pom-kl-ledger-telescope}
在定理 \ref{thm:pom-kl-pythagoras-tower} 的设定下，定义单步外置量
\[
\Delta_m(\mu):=D_{\mathrm{KL}}(\mu^{(m)}\|\mu^{(m-1)})\qquad(2\le m\le M).
\]
则对任意 $M\ge 1$ 有严格望远镜分解
\[
\boxed{\;
D_{\mathrm{KL}}(\mu\|\mu^{(1)})
=\sum_{m=2}^{M}\Delta_m(\mu)\;+\;D_{\mathrm{KL}}(\mu\|\mu^{(M)})\; }.
\]
\end{corollary}
\begin{proof}
对 $m=2,3,\dots,M$ 依次将定理 \ref{thm:pom-kl-pythagoras-tower} 应用于三元组 $(m_1,m_2)=(m-1,m)$ 并求和，所有中间项望远镜抵消，得到所示恒等式。证毕。
\end{proof}

\begin{remark}[结构判据：伪不变量来自坐标不一致]\label{rem:pom-gauge-criterion}
若某一 ISA/微码/核在跨分辨率上使用朴素截断（$\tau$）来比较对象，而非经 $P_Z$（或等价地经 $r^{\mathrm{e}}$）落到稳定逆系统，则其“规律/不变量”可能只是坐标不一致产生的幻影：同一高分辨率微态在低分辨率下被错误归类，从而制造伪素数影子与伪收敛现象。因而跨分辨率可审计性必须以 fold-aware restriction 作为 Gauge 条件。
\end{remark}

\begin{theorem}[逆极限自然性]\label{thm:pom-inverse-limit}
设对每个分辨率 $m$ 定义稳定层运算 $f_m:X_m\to X_m$，并满足对一切 $m_2\ge m_1$ 的截断投影
\[
\pi_{m_2\to m_1}\circ f_{m_2}=f_{m_1}\circ \pi_{m_2\to m_1}.
\]
则存在唯一的极限运算 $f_\infty:X_\infty\to X_\infty$ 使得对所有 $m$ 有 $\pi_m\circ f_\infty=f_m\circ \pi_m$。
\end{theorem}
\begin{proof}
由定理 \ref{thm:inverse-limit-golden}，$X_\infty\cong\varprojlim X_m$。相容性等式正是逆系统上的自然变换条件，因此由逆极限的泛性质得到唯一的 $f_\infty$。证毕。
\end{proof}

\begin{proposition}[纤维不变量判别]\label{prop:pom-fiber-invariance}
给定 $g:\Omega_m\to\mathbb{R}$，以下条件等价：
\begin{enumerate}
  \item 若 $\Fold_m(a)=\Fold_m(b)$ 则 $g(a)=g(b)$；
  \item 存在 $\bar g:X_m\to\mathbb{R}$ 使得 $g=\bar g\circ\Fold_m$；
  \item $g$（作为可观测量）属于折叠不变量子代数 $\mathcal{B}_m$（命题 \ref{prop:fold-invariant-subalgebra}）。
\end{enumerate}
\end{proposition}
\begin{proof}
$(1)\Rightarrow(2)$：定义 $\bar g(x)=g(a)$ 对任意 $a\in\Fold_m^{-1}(x)$，由 (1) 保证良定义。$(2)\Rightarrow(3)$：$g$ 在每条纤维上常值，故对应于 $\mathcal{B}_m$ 的坐标表示。$(3)\Rightarrow(1)$：$\mathcal{B}_m$ 的元素对纤维常值（命题 \ref{prop:fold-invariant-subalgebra}）。证毕。
\end{proof}

\begin{remark}[时间—空间对偶接口与素数影子]\label{rem:pom-time-space}
命题 \ref{prop:time-space-commuting-diagram} 给出的严格交换图把时间侧 $\Tr(T^n)$ 与空间侧 primitive 轨道谱对齐，从而为“时间—空间正交对偶”提供可审计接口。另一方面，有限核的 primitive 轨道谱与整数素数并不等同（见注 \ref{rem:prime-shadow-vs-primes}）：前者是核动力学的内禀素性，后者需要额外的素数索引算子族。素数寄存器纤维 $\mathcal{P}$ 正是对这一额外索引结构的本体化补充。
\end{remark}

\begin{theorem}[profinite 轴上的 Chebotarev 提升]\label{thm:pom-profinite-axis}
设 $\{G_m,\pi_{m_2\to m_1}\}$ 为有限群的投影系统，$G_\infty:=\varprojlim_m G_m$。对每个 $m$，考虑对应的 $G_m$-扩张核，并记
$$
\lambda_m:=\rho(B_{\mathbf 1,m}),\qquad
\Lambda_m:=\max_{\rho\neq\mathbf 1}\rho(B_{\rho,m}).
$$
令 \(U\subset G_\infty\) 为 cylinder 集合，即存在某个 $m$ 与共轭不变子集 $A\subseteq G_m$ 使
\(
U=\pi_m^{-1}(A)
\)。
则对应的 primitive 轨道计数满足
$$
B_{n,U}
=\mu_{\mathrm{Haar}}(U)\,\frac{\lambda_m^n}{n}
 +O\!\left(\frac{\Lambda_m^n}{n}\right).
$$
其中 \(\mu_{\mathrm{Haar}}(U)=|A|/|G_m|\)。
若进一步满足核版 GRH 条件 $\Lambda_m\le \lambda_m^{1/2}$（对该层 $m$），则误差降为平方根级
$$
B_{n,U}
=\mu_{\mathrm{Haar}}(U)\,\frac{\lambda_m^n}{n}
 +O\!\left(\frac{\lambda_m^{n/2}}{n}\right).
$$
\end{theorem}
\begin{proof}
由于 $U$ 为 cylinder，存在 $m$ 与共轭不变子集 $A\subseteq G_m$ 使 $U=\pi_m^{-1}(A)$。令 $B_{n,c}$ 表示落在共轭类 $c\subseteq G_m$ 的 primitive 轨道计数，则
$$
B_{n,U}=\sum_{c\subseteq A} B_{n,c}.
$$
对每个 $c$，由有限层的 Chebotarev 型估计（定理 \ref{thm:kernel-chebotarev-exp}）有
$
B_{n,c}=\frac{|c|}{|G_m|}\frac{\lambda_m^n}{n}+O(\Lambda_m^n/n)
$。
将各类相加得到主项系数
$
\sum_{c\subseteq A}|c|/|G_m|=|A|/|G_m|=\mu_{\mathrm{Haar}}(U)
$，并因 $A$ 的共轭类数目有限，误差仍为 $O(\Lambda_m^n/n)$。若 $\Lambda_m\le \lambda_m^{1/2}$，则立得平方根级误差。证毕。
\end{proof}
