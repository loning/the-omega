
\section{HPA--Omega axioms: finite information, unitary scanning, and readout rigidity}
\label{sec:omega_axioms}

This section isolates the fragments of the Omega axiom system used by the closed-layer argument.
The role of these axioms is to formalize a simple but strong constraint: \emph{finite-resolution readout of a unitary scan on a compact state space cannot support exponential growth along the canonical Abel path}.

\subsection{Finite information and compact readout}
\label{sec:axiom_O2}

\begin{axiom}[O2 (Finite information / compact readout)]
\label{ax:O2}
For any fixed finite resolution and finite observation horizon, effective observables are represented in a controlled (effectively finite) operator algebra on a compact state space.
In particular, expectations and traces of bounded observables obey uniform boundedness constraints under admissible readout channels.
\end{axiom}

In practice, O2 means that the geometric side of any protocol-defined trace built from bounded effect operators remains bounded whenever its defining series converges.

\subsection{Scan--projection readout}
\label{sec:axiom_O5}

\begin{axiom}[O5 (Scan--projection readout)]
\label{ax:O5}
There exists a pointer observable $V$ with spectral measure $\Pi_V$ and a family of finite-resolution windows $\{w_k^{(\varepsilon)}\}_k$ such that the readout effects are
\[
E_k^{(\varepsilon)}=\int w_k^{(\varepsilon)}(x)\,\dd\Pi_V(x).
\]
Given an effective state $\omega_{\mathrm{eff}}$, the readout channel probabilities are
\[
P_k^{(\varepsilon)}=\omega_{\mathrm{eff}}\!\bigl(E_k^{(\varepsilon)}\bigr).
\]
\end{axiom}

This is a protocol-level encoding of the idea that continuous phases are only accessed through finite-resolution projections, producing finite symbol streams (e.g.\ Sturmian/Fibonacci words) rather than raw continuum data.

\subsection{Unitary scan algebra and Weyl pairs}
\label{sec:axiom_O6}

\begin{axiom}[O6 (Unitary scan algebra / Weyl pair)]
\label{ax:O6}
The scan dynamics is generated by a unitary operator $U_{\mathrm{scan}}$.
Together with the pointer operator $V$, it forms a Weyl pair
\[
U_{\mathrm{scan}}\,V=\e^{2\pi \iu \alpha}\,V\,U_{\mathrm{scan}},
\]
where $\alpha$ is irrational (the golden branch $\alpha=\varphi^{-1}$ is the minimal-complexity model case).
\end{axiom}

Unitarity implies that conjugation preserves operator norms:
\[
\supnorm{U_{\mathrm{scan}}^t A U_{\mathrm{scan}}^{-t}}=\supnorm{A}\qquad (t\in\ZZ).
\]
This yields a basic closed-layer fact: Abel-weighted protocol traces built from bounded single-step contributions are automatically well-defined for every $0<r<1$.

\subsection{Abel-weighted traces are well-defined under boundedness}
\label{sec:abel_trace_defined}

\begin{proposition}[Abel definability for bounded scan contributions]
\label{prop:abel_definability}
Let $A$ be a bounded observable and let $\omega_{\mathrm{eff}}$ be a bounded state functional.
Define the Abel-weighted trace-like quantity
\[
T_A(r):=\sum_{t=0}^{\infty} r^t\,\omega_{\mathrm{eff}}\!\bigl(U_{\mathrm{scan}}^t A U_{\mathrm{scan}}^{-t}\bigr),
\qquad |r|<1.
\]
Then $T_A(r)$ is absolutely convergent for every $|r|<1$, defines a holomorphic function on the open unit disk, and satisfies the bound
\[
|T_A(r)|\le \frac{\supnorm{A}\,\norm{\omega_{\mathrm{eff}}}}{1-|r|}.
\]
\end{proposition}

\begin{proof}
By boundedness of $\omega_{\mathrm{eff}}$ and unitarity,
\[
\left|\omega_{\mathrm{eff}}\!\bigl(U_{\mathrm{scan}}^t A U_{\mathrm{scan}}^{-t}\bigr)\right|
\le
\norm{\omega_{\mathrm{eff}}}\,\supnorm{U_{\mathrm{scan}}^t A U_{\mathrm{scan}}^{-t}}
=\norm{\omega_{\mathrm{eff}}}\,\supnorm{A}.
\]
Hence for $|r|<1$ the series is dominated by $\sum_{t\ge 0} |r|^t$ and converges absolutely.
As a power series in $r$ with radius of convergence at least $1$, it defines a holomorphic function on the open unit disk.
\end{proof}

Proposition~\ref{prop:abel_definability} is purely elementary, but it captures the essential closed-layer rigidity: within Omega, admissible geometric-side traces are defined on the entire interval $(0,1)$.
The additional Omega content is that a \emph{canonical finite-part extraction along $r\uparrow 1$} is treated as part of the protocol specification: only those traces for which the finite part exists on this path are admissible closed-layer objects.
