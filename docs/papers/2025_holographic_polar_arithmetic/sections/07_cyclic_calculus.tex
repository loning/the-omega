\section{Cyclic calculus: difference, Newton expansion, trace, and finite parts along \texorpdfstring{$\Theta$}{Theta}-orbits}
\label{sec:cyclic_calculus}

This section provides the second structural pillar: statements about regularization and finite residuals are formulated via orbit functionals rather than by informal limiting intuition.

\subsection{\texorpdfstring{$\Theta$}{Theta}-difference as a derivative surrogate}
For sufficiently regular $f:X\to\CC$, define the orbit difference operator
$$
(\mathcal{D}_\Theta f)(x):=\frac{f(x+\alpha)-f(x)}{\alpha}.
$$
When $\alpha$ is small this approximates a derivative, but here $\alpha$ is fixed and irrational, so $\mathcal{D}_\Theta$ is an intrinsic ``discrete derivative'' along the scan step.

\subsection{Orbit Newton series: discrete Taylor reconstruction}
\label{subsec:orbit_newton}
Define the forward difference operator along the scan by
$$
(\Delta_\Theta f)(x):=(\Theta-I)f(x)=f(x+\alpha)-f(x)=\alpha\,(\mathcal{D}_\Theta f)(x),
$$
and its higher iterates $\Delta_\Theta^k:=(\Theta-I)^k$ for $k\ge 0$.
The relevant Taylor surrogate in the scan direction is then a Newton-type expansion in the operator $(\Theta-I)$ rather than in a primitive additive increment.

\begin{proposition}[Binomial orbit expansion (discrete Taylor formula)]
\label{prop:orbit_binomial_taylor}
For every integer $n\ge 0$,
$$
\Theta^n=(I+\Delta_\Theta)^n=\sum_{k=0}^{n}\binom{n}{k}\Delta_\Theta^k.
$$
Equivalently, for any $f$ and $x\in X$,
$$
f(x+n\alpha)=(\Theta^n f)(x)=\sum_{k=0}^{n}\binom{n}{k}(\Delta_\Theta^k f)(x)
=\sum_{k=0}^{n}\binom{n}{k}\alpha^k\,(\mathcal{D}_\Theta^k f)(x).
$$
\end{proposition}

\begin{proof}
Since $\Theta=I+\Delta_\Theta$, the operator identity is the binomial theorem.
Applying both sides to $f$ and evaluating at $x$ gives the pointwise formulas.
\end{proof}

\begin{remark}[Explicit $k$th difference]
\label{rem:orbit_kth_difference}
For $k\ge 0$,
$$
\Delta_\Theta^k=\sum_{j=0}^{k}(-1)^{k-j}\binom{k}{j}\Theta^{j},
\qquad
(\Delta_\Theta^k f)(x)=\sum_{j=0}^{k}(-1)^{k-j}\binom{k}{j}f(x+j\alpha).
$$
\end{remark}

\begin{definition}[Orbit Newton polynomial and remainder]
\label{def:orbit_newton_remainder}
For $m\ge 0$ define the order-$m$ orbit Newton polynomial
$$
T_m(n;f)(x):=\sum_{k=0}^{\min\{m,n\}}\binom{n}{k}(\Delta_\Theta^k f)(x),
$$
and the associated remainder
$$
R_m(n;f)(x):=f(x+n\alpha)-T_m(n;f)(x)=\sum_{k=m+1}^{n}\binom{n}{k}(\Delta_\Theta^k f)(x),
$$
with the convention that the sum is empty (hence $0$) when $m\ge n$.
\end{definition}

\subsection{Orbit trace: ergodic averages as an integral surrogate}
Define the orbit average
$$
A_N(f;x_0):=\frac{1}{N}\sum_{k=0}^{N-1} f(x_0+k\alpha).
$$

\begin{theorem}[Unique ergodicity of irrational rotations]
\label{thm:unique_ergodic}
If $f$ is continuous, the limit
$$
\mathrm{Tr}_\Theta(f):=\lim_{N\to\infty} A_N(f;x_0)
$$
exists, is independent of $x_0$, and equals the space average:
$$
\mathrm{Tr}_\Theta(f)=\int_X f(x)\,d\mu(x).
$$
\end{theorem}

Theorem~\ref{thm:unique_ergodic} is a standard consequence of unique ergodicity/equidistribution of irrational circle rotations and provides a canonical replacement for ``integration'' and ``summation'' along the scan orbit; see e.g.~\cite{Weyl1916,KuipersNiederreiter1974}.

\subsection{An Abel-type finite part for divergent orbit sums}
\label{subsec:abel_fp}
Let $u_k:=f(x_0+k\alpha)$ be a (possibly non-absolutely summable) sequence. Define its Abel generating function
$$
S_f(r):=\sum_{k=0}^{\infty} u_k r^k,\qquad 0<r<1.
$$
If $f$ is bounded (e.g. continuous on $X$), then $|u_k|\le \|f\|_\infty$ and $S_f(r)$ converges absolutely for every $0<r<1$. The use of Abel limits and pole subtraction as a regularization principle is classical; see e.g.~\cite{Hardy1949}.
If as $r\uparrow 1$ the function admits an asymptotic expansion of the form
$$
S_f(r)=\sum_{j=1}^{J} \frac{c_j}{(1-r)^j} + C + o(1),
$$
then we define its orbit finite part by
$$
\FP_\Theta(f):=C.
$$
The constant term $C$ is unique whenever such an expansion exists, so $\FP_\Theta$ is well-defined (see Lemma~\ref{lem:fp_unique}).

\begin{lemma}[Uniqueness of the finite part]
\label{lem:fp_unique}
Suppose $S_f(r)$ admits two expansions as $r\uparrow 1$ of the form
$$
S_f(r)=\sum_{j=1}^{J} \frac{c_j}{(1-r)^j} + C + o(1)
$$
and
$$
S_f(r)=\sum_{j=1}^{J'} \frac{c'_j}{(1-r)^j} + C' + o(1).
$$
Then $C=C'$.
\end{lemma}

\begin{proof}
Subtract the two expansions to obtain
$$
0=\sum_{j=1}^{J''}\frac{d_j}{(1-r)^j}+(C-C')+o(1)
$$
as $r\uparrow 1$, where $J'':=\max\{J,J'\}$ and $d_j:=c_j-c'_j$ (with missing coefficients treated as $0$).
Multiplying by $(1-r)^{J''}$ and taking $r\uparrow 1$ gives $d_{J''}=0$. Repeating inductively yields $d_j=0$ for all $j\ge 1$, so $C-C'+o(1)=0$ and therefore $C=C'$.
\end{proof}

This template shifts ``divergence'' from being a defect of the object to being a dependence on a consistent limiting and pole-subtraction scheme.

\subsection{Fourier resolution and a cohomological interpretation of Abel finite parts}
\label{subsec:abel_fourier}
For irrational rotations, Abel orbit sums admit an explicit Fourier--resolvent description.
Write the Fourier coefficients of $f\in L^2(X)$ as
$$
\hat f(m):=\int_X f(x)\,\e^{-2\pi\iu m x}\,d\mu(x),\qquad m\in\ZZ,
$$
so that formally $f(x)\sim\sum_{m\in\ZZ}\hat f(m)\e^{2\pi\iu m x}$.

\begin{proposition}[Fourier symbols and a band-limited remainder bound]
\label{prop:fourier_symbol_remainder}
Let $e_m(x):=\e^{2\pi\iu m x}$. Then $e_m$ is an eigenfunction of $\Theta$:
$$
\Theta e_m=\e^{2\pi\iu m\alpha}\,e_m,
$$
hence $\Delta_\Theta e_m=(\e^{2\pi\iu m\alpha}-1)e_m$ and $\mathcal{D}_\Theta e_m=\frac{\e^{2\pi\iu m\alpha}-1}{\alpha}e_m$.
If $f$ has finite Fourier support $\hat f(m)=0$ for $|m|>M$, set
$$
\eta_M:=\max_{|m|\le M}\bigl|\e^{2\pi\iu m\alpha}-1\bigr|.
$$
Then for all $k\ge 0$,
$$
\|\Delta_\Theta^k f\|_{L^2}\le \eta_M^k\,\|f\|_{L^2},
$$
and for the orbit-Newton remainder from Definition~\ref{def:orbit_newton_remainder},
$$
\|R_m(n;f)\|_{L^2}\le \|f\|_{L^2}\sum_{k=m+1}^{n}\binom{n}{k}\eta_M^k.
$$
\end{proposition}

\begin{proof}
The eigenvalue identity follows from $\Theta f(x)=f(x+\alpha)$.
If $f(x)=\sum_{|m|\le M}\hat f(m)e_m(x)$, then $\Delta_\Theta^k f=\sum_{|m|\le M}\hat f(m)(\e^{2\pi\iu m\alpha}-1)^k e_m$, so by Parseval
$$
\|\Delta_\Theta^k f\|_{L^2}^2=\sum_{|m|\le M}|\hat f(m)|^2\,|\e^{2\pi\iu m\alpha}-1|^{2k}\le \eta_M^{2k}\sum_{|m|\le M}|\hat f(m)|^2=\eta_M^{2k}\|f\|_{L^2}^2.
$$
The remainder bound follows from the triangle inequality and Definition~\ref{def:orbit_newton_remainder}.
\end{proof}

\begin{proposition}[Fourier evaluation of the Abel orbit sum]
\label{prop:abel_fourier_formula}
Let $\alpha\in(0,1)\setminus\QQ$ and $x_0\in X$.
Assume $f$ has absolutely summable Fourier coefficients:
$$
\sum_{m\in\ZZ}|\hat f(m)|<\infty.
$$
Then for every $0<r<1$ one has
$$
S_f(r)=\sum_{m\in\ZZ}\frac{\hat f(m)\,\e^{2\pi\iu m x_0}}{1-r\,\e^{2\pi\iu m\alpha}}.
$$
\end{proposition}

\begin{proof}
For $k\ge 0$,
$$
u_k=f(x_0+k\alpha)=\sum_{m\in\ZZ}\hat f(m)\,\e^{2\pi\iu m (x_0+k\alpha)}
$$
with absolute convergence uniformly in $x$ by the hypothesis $\sum_m|\hat f(m)|<\infty$.
Therefore, for $0<r<1$ we may exchange the summations:
$$
S_f(r)=\sum_{k\ge 0}r^k u_k
=\sum_{m\in\ZZ}\hat f(m)\,\e^{2\pi\iu m x_0}\sum_{k\ge 0}\bigl(r\,\e^{2\pi\iu m\alpha}\bigr)^k,
$$
and the inner geometric series sums to $(1-r\,\e^{2\pi\iu m\alpha})^{-1}$.
\end{proof}

\begin{corollary}[Simple pole and explicit finite part]
\label{cor:abel_fp_fourier}
Under the hypotheses of Proposition~\ref{prop:abel_fourier_formula}, as $r\uparrow 1$ one has
$$
S_f(r)=\frac{\hat f(0)}{1-r}
\;+\;
\sum_{m\in\ZZ\setminus\{0\}}\frac{\hat f(m)\,\e^{2\pi\iu m x_0}}{1-\e^{2\pi\iu m\alpha}}
\;+\;o(1).
$$
In particular, $\FP_\Theta(f)$ exists and equals the constant term
$$
\FP_\Theta(f)=\sum_{m\in\ZZ\setminus\{0\}}\frac{\hat f(m)\,\e^{2\pi\iu m x_0}}{1-\e^{2\pi\iu m\alpha}}.
$$
\end{corollary}

\begin{proof}
In Proposition~\ref{prop:abel_fourier_formula}, the $m=0$ term is $\hat f(0)/(1-r)$.
For $m\ne 0$, irrationality of $\alpha$ implies $\e^{2\pi\iu m\alpha}\ne 1$, so the denominator stays bounded away from $0$ as $r\uparrow 1$.
Dominated convergence (using $\sum_m|\hat f(m)|<\infty$) yields the limit of the $m\ne 0$ part, proving the expansion.
\end{proof}

\begin{proposition}[A quantitative remainder bound for the Abel finite part]
\label{prop:abel_fp_error}
Assume the hypotheses of Proposition~\ref{prop:abel_fourier_formula} and assume additionally that $\alpha$ is of bounded type, i.e. $\sup_i a_i<\infty$ in the continued fraction expansion $\alpha=[0;a_1,a_2,\dots]$.
Assume further that
$$
\sum_{m\in\ZZ}(1+|m|^2)\,|\hat f(m)|<\infty.
$$
Then there exists a constant $C=C(f,\alpha)$ such that for all $r\in(0,1)$,
$$
\left|S_f(r)-\frac{\hat f(0)}{1-r}-\FP_\Theta(f)\right|\le C\,(1-r).
$$
\end{proposition}

\begin{proof}
Write $\theta_m:=2\pi m\alpha$.
By Proposition~\ref{prop:abel_fourier_formula} and Corollary~\ref{cor:abel_fp_fourier},
$$
S_f(r)-\frac{\hat f(0)}{1-r}-\FP_\Theta(f)
=\sum_{m\ne 0}\hat f(m)\e^{2\pi\iu m x_0}
\left(\frac{1}{1-r\e^{\iu\theta_m}}-\frac{1}{1-\e^{\iu\theta_m}}\right).
$$
For $m\ne 0$,
$$
\left|\frac{1}{1-r\e^{\iu\theta_m}}-\frac{1}{1-\e^{\iu\theta_m}}\right|
=\frac{1-r}{|1-r\e^{\iu\theta_m}|\;|1-\e^{\iu\theta_m}|}.
$$
Split the sum into the regimes $|1-\e^{\iu\theta_m}|\ge 2(1-r)$ and $|1-\e^{\iu\theta_m}|<2(1-r)$.

If $|1-\e^{\iu\theta_m}|\ge 2(1-r)$, then
$$
|1-r\e^{\iu\theta_m}|\ge |1-\e^{\iu\theta_m}|-(1-r)\ge \tfrac12|1-\e^{\iu\theta_m}|,
$$
hence
$$
\left|\frac{1}{1-r\e^{\iu\theta_m}}-\frac{1}{1-\e^{\iu\theta_m}}\right|
\le \frac{2(1-r)}{|1-\e^{\iu\theta_m}|^2}.
$$
If $|1-\e^{\iu\theta_m}|<2(1-r)$, then $|1-r\e^{\iu\theta_m}|\ge 1-r$, so
$$
\left|\frac{1}{1-r\e^{\iu\theta_m}}-\frac{1}{1-\e^{\iu\theta_m}}\right|
\le \frac{1}{|1-\e^{\iu\theta_m}|}.
$$

Since $\alpha$ is of bounded type, there exists $c_\alpha>0$ such that $\|m\alpha\|\ge c_\alpha/|m|$ for all $m\ne 0$, where $\|t\|:=\inf_{n\in\ZZ}|t-n|$ denotes distance to the nearest integer.
Using $|1-\e^{2\pi\iu t}|=2|\sin(\pi t)|\ge 4\|t\|$, we obtain
$$
|1-\e^{\iu\theta_m}|=|1-\e^{2\pi\iu m\alpha}|\ge \frac{4c_\alpha}{|m|}.
$$
Therefore $|1-\e^{\iu\theta_m}|^{-2}\le C'_\alpha\,|m|^2$ and $|1-\e^{\iu\theta_m}|^{-1}\le C''_\alpha\,|m|$.
Combining the two regimes yields
$$
\left|\frac{1}{1-r\e^{\iu\theta_m}}-\frac{1}{1-\e^{\iu\theta_m}}\right|
\le 2(1-r)\,C'_\alpha\,|m|^2 + C''_\alpha\,|m|\ind_{\{|1-\e^{\iu\theta_m}|<2(1-r)\}}.
$$
The first term is summable by the hypothesis $\sum_m|m|^2|\hat f(m)|<\infty$.
For the second term, note that $|1-\e^{\iu\theta_m}|<2(1-r)$ implies $|m|>2c_\alpha/(1-r)$, hence
$$
\sum_{m\ne 0}|m|\,|\hat f(m)|\,\ind_{\{|1-\e^{\iu\theta_m}|<2(1-r)\}}
\le \sum_{|m|>2c_\alpha/(1-r)} |m|\,|\hat f(m)|
=o(1)
$$
as $r\uparrow 1$.
In particular, this term is $O(1-r)$ for $r$ sufficiently close to $1$ (after increasing the constant if needed).
Putting the estimates together proves the claim.
\end{proof}

\begin{proposition}[Cohomological equation and the finite part]
\label{prop:cohomological_fp}
Assume the hypotheses of Proposition~\ref{prop:abel_fourier_formula} and set $f_0:=f-\hat f(0)$.
Define
$$
g(x):=\sum_{m\in\ZZ\setminus\{0\}}\frac{\hat f(m)}{1-\e^{2\pi\iu m\alpha}}\,\e^{2\pi\iu m x}.
$$
Then the series converges uniformly, $g$ is continuous, and it solves the cohomological equation
$$
(I-\Theta)g=f_0.
$$
Moreover, the Abel limit of orbit sums is the cohomological potential:
$$
\lim_{r\uparrow 1}\sum_{k\ge 0} r^k f_0(x_0+k\alpha)=g(x_0)=\FP_\Theta(f).
$$
\end{proposition}

\begin{proof}
Uniform convergence follows as in Proposition~\ref{prop:abel_fourier_formula}.
Compute Fourier coefficients. Since $\Theta$ acts by translation, it sends the Fourier mode $\e^{2\pi\iu m x}$ to $\e^{2\pi\iu m\alpha}\e^{2\pi\iu m x}$, hence
$$
\widehat{(I-\Theta)g}(m)=(1-\e^{2\pi\iu m\alpha})\hat g(m)=\hat f(m)\qquad (m\ne 0),
$$
and $\widehat{(I-\Theta)g}(0)=0=\widehat{f_0}(0)$, so $(I-\Theta)g=f_0$.
Finally, apply Corollary~\ref{cor:abel_fp_fourier} to $f_0$ (for which $\hat f_0(0)=0$) to obtain the Abel limit and identify it with $g(x_0)$.
\end{proof}

\subsection{Mellin comparison: Abel finite parts and zeta regularization}
\label{subsec:mellin_zeta_fp}
The orbit prescription above becomes classical zeta regularization when the ``orbit'' is replaced by the standard counting orbit $n\mapsto n$.
We record a representative statement that makes the relationship to Mellin transform analytic continuation explicit; see e.g.~\cite{FlajoletGourdonDumas1995,Hardy1949}.

\begin{theorem}[Abel finite part of polynomial sums equals zeta regularization]
\label{thm:abel_fp_zeta}
Let $k\in\NN_0$ and define, for $\varepsilon>0$,
$$
S_k(\varepsilon):=\sum_{n=1}^{\infty} n^k\,\e^{-n\varepsilon}.
$$
Then as $\varepsilon\downarrow 0$ the function $S_k(\varepsilon)$ admits an asymptotic expansion of the form
$$
S_k(\varepsilon)=\sum_{j=1}^{k+1}\frac{c_j}{\varepsilon^j}+\zeta(-k)+o(1),
$$
and therefore
$$
\FP_{\varepsilon\downarrow 0}\,S_k(\varepsilon)=\zeta(-k).
$$
In particular, the Abel finite part agrees with zeta regularization of the divergent sum $\sum_{n\ge 1}n^k$.
\end{theorem}

\begin{proof}
For $\Re(s)>0$ one has the Mellin identity
$$
\e^{-n\varepsilon}=\frac{1}{2\pi\iu}\int_{\sigma-\iu\infty}^{\sigma+\iu\infty}\Gamma(s)\,(n\varepsilon)^{-s}\,ds
$$
for any $\sigma>0$.
Multiplying by $n^k$ and summing over $n\ge 1$ gives, for $\sigma>k+1$ (justifying absolute convergence),
$$
S_k(\varepsilon)=\frac{1}{2\pi\iu}\int_{\sigma-\iu\infty}^{\sigma+\iu\infty}\Gamma(s)\,\zeta(s-k)\,\varepsilon^{-s}\,ds.
$$
The integrand extends meromorphically to $s\in\CC$, with a simple pole at $s=k+1$ coming from $\zeta(s-k)$ and simple poles at $s=0,-1,-2,\dots$ coming from $\Gamma(s)$.
Shifting the contour to the left and summing residues yields an asymptotic expansion in powers of $\varepsilon$ as $\varepsilon\downarrow 0$.
The constant term comes from the residue at $s=0$:
since $\Gamma(s)=\frac{1}{s}+O(1)$ and $\zeta(s-k)=\zeta(-k)+O(s)$ near $s=0$, the residue equals $\zeta(-k)$.
This proves that the finite-part constant term is $\zeta(-k)$.
\end{proof}

\begin{remark}[Hadamard finite parts and ``constant-term extraction'']
The operation $\FP_{\varepsilon\downarrow 0}$ in Theorem~\ref{thm:abel_fp_zeta} is a Hadamard-type finite part: it extracts the constant term in a full asymptotic expansion after subtracting divergent negative powers of $\varepsilon$.
This is the same conceptual mechanism as in Subsection~\ref{subsec:abel_fp}, with the regulator $(1-r)$ replaced by $\varepsilon$ via $r=\e^{-\varepsilon}$; see e.g.~\cite{EstradaKanwal2002}.
\end{remark}

\subsection{Finite-part traces and regularized determinants}
\label{subsec:fp_determinants}
To interface orbit finite parts with spectral/trace formulas one needs a determinant language compatible with regularization. A minimal template is as follows.

\begin{definition}[Regularized trace and finite-part trace]
Let $\mathcal{A}$ be a (not necessarily commutative) algebra of operators on a Hilbert space. A \emph{regularized trace scheme} is a family of linear functionals $\{\mathrm{Tr}_\varepsilon\}_{\varepsilon>0}$ on a subalgebra $\mathcal{A}_0\subset\mathcal{A}$ such that for each $T\in\mathcal{A}_0$ the scalar $\mathrm{Tr}_\varepsilon(T)$ is well-defined for $\varepsilon>0$ and admits, as $\varepsilon\downarrow 0$, an asymptotic expansion
$$
\mathrm{Tr}_\varepsilon(T)=\sum_{j=1}^{J}\frac{c_j(T)}{\varepsilon^{j}}+\mathrm{Tr}_{\FP}(T)+o(1).
$$
The constant term $\mathrm{Tr}_{\FP}(T)$ is called the \emph{finite-part trace} of $T$ (relative to the chosen scheme).
\end{definition}

\begin{remark}
The Abel orbit finite part of Subsection~\ref{subsec:abel_fp} is a special case of this definition with the regulator $\varepsilon\sim 1-r$ and $T$ replaced by an orbit-sampled observable. In analytic number theory one often uses heat-kernel or zeta regularizations; see e.g.~\cite{Connes1994,Hardy1949}.
\end{remark}

\begin{assumption}[Cyclicity on the admissible class]
\label{ass:fp_cyclicity}
On $\mathcal{A}_0$ the finite-part trace is cyclic:
$$
\mathrm{Tr}_{\FP}(AB)=\mathrm{Tr}_{\FP}(BA)\qquad (A,B\in\mathcal{A}_0),
$$
equivalently $\mathrm{Tr}_{\FP}([A,B])=0$.
\end{assumption}

\begin{definition}[Finite-part determinant]
\label{def:fp_det}
Let $T\in\mathcal{A}_0$ be such that $\mathrm{Tr}_{\FP}(T^n)$ is defined for all $n\ge 1$ and obeys an exponential growth bound $\big|\mathrm{Tr}_{\FP}(T^n)\big|\le C R^n$ for some $C,R>0$. For $|u|<1/R$ define the finite-part determinant by
$$
\log\det_{\FP}(I-uT):=-\sum_{n=1}^{\infty}\frac{u^n}{n}\,\mathrm{Tr}_{\FP}(T^n),
\qquad
\det_{\FP}(I-uT):=\exp\!\big(\log\det_{\FP}(I-uT)\big).
$$
\end{definition}

\begin{proposition}[Analyticity near the origin]
\label{prop:fp_det_analytic}
Under the growth hypothesis in Definition~\ref{def:fp_det}, the series defining $\log\det_{\FP}(I-uT)$ converges absolutely for $|u|<1/R$ and defines a holomorphic function there.
\end{proposition}

\begin{proof}
For $|u|<1/R$ one has
$$
\sum_{n\ge 1}\left|\frac{u^n}{n}\mathrm{Tr}_{\FP}(T^n)\right|
\le \sum_{n\ge 1}\frac{C(|u|R)^n}{n}
<\infty,
$$
so the series converges uniformly on compact subsets and defines a holomorphic function.
\end{proof}

\begin{proposition}[Invariance of the zero set under finite-part scheme changes]
\label{prop:fp_det_invariance}
Let $\mathrm{Tr}_{\FP}$ and $\mathrm{Tr}'_{\FP}$ arise from two regularized trace schemes on $\mathcal{A}_0$, and assume that for a given $T\in\mathcal{A}_0$ the difference of the corresponding logarithmic determinants,
$$
\Delta(u):=\log\det'_{\FP}(I-uT)-\log\det_{\FP}(I-uT),
$$
extends holomorphically to a neighborhood of $u=0$. Then in that neighborhood
$$
\det'_{\FP}(I-uT)=\e^{\Delta(u)}\,\det_{\FP}(I-uT),
$$
so the two determinants have the same zero set (with multiplicities).
\end{proposition}

\begin{proof}
By definition $\det'_{\FP}(I-uT)/\det_{\FP}(I-uT)=\exp(\Delta(u))$. If $\Delta$ is holomorphic then $\exp(\Delta)$ is holomorphic and nowhere vanishing, hence it cannot create or remove zeros.
\end{proof}

\begin{remark}[Why cyclicity matters]
Assumption~\ref{ass:fp_cyclicity} is the minimal condition ensuring that $\mathrm{Tr}_{\FP}$ behaves as a trace and that determinant identities are stable under cyclic rearrangements (the mechanism behind many trace formulas). Verifying cyclicity and identifying a canonical admissible class $\mathcal{A}_0$ is model-dependent; in noncommutative settings, canonical traces often arise from unique ergodicity or $C^*$-algebraic trace states, while in analytic settings they arise from heat-kernel/zeta regularization.
\end{remark}
