\begin{theorem}[Intrinsic Self-Calibration of the Golden Ratio: Recovery of \texorpdfstring{$\varphi$}{phi} via Terminal-Bit Conditional Mean Ratio]\label{thm:xi-foldbin-intrinsic-self-calibration-phi-lastbit-means}

let \(d_m^{\mathrm{bin}}(x):=\bigl|(\mathrm{Fold}^{\mathrm{bin}}_m)^{-1}(x)\bigr|\) for \(x\in X_m\), and partition by terminal bit:
\[
X_m^{(0)}:=\{x\in X_m:\ x_m=0\},
\qquad
X_m^{(1)}:=\{x\in X_m:\ x_m=1\}.
\]
Define the observable statistic
\[
\widehat\varphi_m
\ :=\
\frac{\frac{1}{|X_m^{(0)}|}\sum_{x\in X_m^{(0)}} d_m^{\mathrm{bin}}(x)}{\frac{1}{|X_m^{(1)}|}\sum_{x\in X_m^{(1)}} d_m^{\mathrm{bin}}(x)}.
\]
Then there exists a constant \(C>0\) independent of \(m\) such that
\[
\bigl|\widehat\varphi_m-\varphi\bigr|\ \le\ C\left(\frac{\varphi}{2}\right)^m,
\]
so that \(\widehat\varphi_m\) converges to \(\varphi\) with exponential precision.
This result provides a self-calibration mechanism requiring no external calibration: the growth constant is recovered solely from the fold-fiber multiplicity and terminal visible bit.
\end{theorem}

\begin{proof}
By Theorem \ref{thm:fold-bin-two-state-asymptotic}, there exists a constant \(C_0>0\) such that for sufficiently large \(m\) and any \(x\in X_m\), we have a uniform estimate
\[
\Bigl|d_m^{\mathrm{bin}}(x)-\varphi^{-x_m}\Bigl(\frac{2}{\varphi}\Bigr)^m\Bigr|\le C_0.
\]
Taking the mean over \(X_m^{(0)}\) and \(X_m^{(1)}\) respectively, we obtain
\[
\frac{1}{|X_m^{(0)}|}\sum_{x\in X_m^{(0)}} d_m^{\mathrm{bin}}(x)
=\Bigl(\frac{2}{\varphi}\Bigr)^m+O(1),
\qquad
\frac{1}{|X_m^{(1)}|}\sum_{x\in X_m^{(1)}} d_m^{\mathrm{bin}}(x)
=\varphi^{-1}\Bigl(\frac{2}{\varphi}\Bigr)^m+O(1),
\]
where the \(O(1)\) constant is independent of \(m\).
Dividing the two expressions and using \((2/\varphi)^{-m}=(\varphi/2)^m\), we get
\[
\widehat\varphi_m=\varphi\cdot\frac{1+O((\varphi/2)^m)}{1+O((\varphi/2)^m)}
=\varphi+O\!\left(\left(\frac{\varphi}{2}\right)^m\right),
\]
from which there exists a constant \(C\) satisfying the stated error bound. This completes the proof.
\end{proof}
