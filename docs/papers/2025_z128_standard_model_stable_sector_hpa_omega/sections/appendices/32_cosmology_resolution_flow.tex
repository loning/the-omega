\subsection[Cosmology as resolution flow]{Cosmology as resolution flow (self-contained interface extension)}
\label{app:cosmology_resolution_flow}

This appendix records a minimal cosmology interface in the same protocol language used throughout this paper.
The central idea is that changing resolution parameters $(m,n)$ is a physical operation (a protocol flow step), and the growth laws of the stable sector $|X_m|=F_{m+2}$ and of the mean degeneracy $2^m/|X_m|$ provide a quantitative capacity-growth backbone.
\AuditTag In the joint protocol-state convention, cosmological interface readouts are conditional on $(m,n,K)$ (Appendix~\ref{subsec:tick_cap_joint_key_contract}): $m,n$ control capacity and uplift staging, while $K$ specifies how stable labels are weighted when forming scalar summaries (Appendix~\ref{app:kernel_family_cap_closure}).
When this appendix uses the pushforward distribution $\mu_m(w)=|P(w)|/2^m$, it is adopting the microstate-pushforward kernel as a concrete audit-facing baseline.

\subsubsection{Initialization: the big bang as resolution bootstrapping}
\label{subsec:cosmo_init}

\InterfaceTag In a readout cosmology, the ``big bang'' is interpreted as resolution initialization of the readout hardware:
the system starts at small window length $m$ and small addressing order $n$, and rapidly enters a regime where nontrivial stable types exist and can be maintained under projection.
The balanced interface $m=2n$ provides a minimal coupling between a spatial grid and a local readout alphabet, and the anchor $(m,n)=(6,3)$ supplies the smallest fully explicit computable layer with nontrivial folding $64\to 21$.

\subsubsection{Inflation as exponential growth of stable capacity}
\label{subsec:cosmo_inflation_capacity}

As window length increases by $\Delta m$, the stable type count grows as
\[
|X_m|=F_{m+2}\sim \frac{\varphi^{m+2}}{\sqrt{5}},
\]
so stable capacity grows exponentially in $m$.
If an early epoch has approximately linear resolution growth $m(t)\approx m_0+\alpha t$, then
\[
|X_{m(t)}|\asymp \varphi^{m(t)}\asymp \e^{(\alpha\log\varphi)\,t},
\]
which provides a purely combinatorial capacity-growth analogue of an inflationary exponential stage (no additional inflaton field is introduced at the protocol level).

\subsubsection{Hidden-sector dominance at high resolution}
\label{subsec:cosmo_hidden_fraction}

The folding map compresses $2^m$ microstates into $F_{m+2}$ stable types.
Define the stable fraction and hidden fraction at window length $m$ by
\[
f_{\mathrm{stab}}(m):=\frac{|X_m|}{|\Omega_m|}=\frac{F_{m+2}}{2^m},
\qquad
f_{\mathrm{hid}}(m):=1-f_{\mathrm{stab}}(m),
\]
and the mean degeneracy
\[
d_m:=\frac{|\Omega_m|}{|X_m|}=\frac{2^m}{F_{m+2}}.
\]
Since $F_{m+2}\asymp \varphi^m$ and $\varphi<2$, one has $f_{\mathrm{stab}}(m)\to 0$ and $f_{\mathrm{hid}}(m)\to 1$ as $m\to\infty$; equivalently $d_m\asymp (2/\varphi)^m$ grows exponentially.

\begin{lemma}[Monotonicity of the stable fraction]\label{lem:f_stab_monotone}
\MathTag For all $m\ge 1$, $f_{\mathrm{stab}}(m)$ is strictly decreasing in $m$, and $d_m=1/f_{\mathrm{stab}}(m)$ is strictly increasing in $m$.
\end{lemma}
\begin{proof}
Recall $f_{\mathrm{stab}}(m)=F_{m+2}/2^m$ with Fibonacci numbers satisfying $F_{k+1}=F_k+F_{k-1}$ and $F_k>0$.
For $m\ge 1$ one has $F_{m+1}<F_{m+2}$ since $F_{m+2}=F_{m+1}+F_m$ with $F_m>0$.
Hence
\[
F_{m+3}=F_{m+2}+F_{m+1}<2F_{m+2},
\]
and therefore
\[
f_{\mathrm{stab}}(m+1)=\frac{F_{m+3}}{2^{m+1}}<\frac{2F_{m+2}}{2^{m+1}}=\frac{F_{m+2}}{2^m}=f_{\mathrm{stab}}(m).
\]
Thus $f_{\mathrm{stab}}(m)$ is strictly decreasing, and $d_m=1/f_{\mathrm{stab}}(m)$ is strictly increasing.
\end{proof}

\paragraph{Degeneracy as fiber size and pushforward occupancy.}
\MathTag At fixed $m$, the equivalence semantics of Section~\ref{app:equivalence_semantics} identifies stable labels with projection fibers.
Let $P(w):=\mathrm{Fold}_m^{-1}(w)$ be the microstate fiber of a stable type $w\in X_m$.
Then the mean degeneracy $d_m$ is exactly the mean fiber size, and the mean fiber-count entropy is controlled by $d_m$ (Lemma~\ref{lem:mean_degeneracy_fiber_entropy_bounds}).
Under the maximal-ignorance counting measure on $\Omega_m$, the induced (pushforward) distribution on stable labels is therefore
\[
\mu_m(w):=\frac{|P(w)|}{|\Omega_m|}=\frac{|P(w)|}{2^m},
\]
so stable labels are weighted in proportion to degeneracy.

\begin{lemma}[Entropy gap and hidden exponent]\label{lem:entropy_gap_hidden_exponent_cosmo}
\MathTag Let $\varphi=(1+\sqrt{5})/2$ and define $f_{\mathrm{stab}}(m)=F_{m+2}/2^m$ and $d_m=2^m/F_{m+2}$ as above.
Then
\[
\lim_{m\to\infty}\frac{1}{m}\log f_{\mathrm{stab}}(m)=\log\varphi-\log 2,
\qquad
\lim_{m\to\infty}\frac{1}{m}\log d_m=\log(2/\varphi).
\]
Moreover, writing $\psi:=(1-\sqrt{5})/2=-\varphi^{-1}$, one has the exact Binet identity
\[
F_{m+2}=\frac{\varphi^{m+2}-\psi^{m+2}}{\sqrt{5}}
=\frac{\varphi^{m+2}}{\sqrt{5}}\Bigl(1-(-\varphi^{-2})^{m+2}\Bigr),
\]
and therefore
\[
\log f_{\mathrm{stab}}(m)
=m\log(\varphi/2)+\log(\varphi^2/\sqrt{5})+\eta_m,
\qquad
|\eta_m|\le 2\,\varphi^{-2m-4},
\]
so the convergence to $\log(\varphi/2)$ is exponentially fast in $m$.
\end{lemma}
\begin{proof}
By Binet's formula (see, e.g., \cite{HardyWright2008}),
$
F_{m+2}=(\varphi^{m+2}-\psi^{m+2})/\sqrt{5}
$
with $\psi=-\varphi^{-1}$, giving the stated exact identity.
Dividing by $2^m$ yields $f_{\mathrm{stab}}(m)$ and taking reciprocals yields $d_m$, hence
\[
\frac{1}{m}\log f_{\mathrm{stab}}(m)=\log(\varphi/2)+\frac{1}{m}\log(\varphi^2/\sqrt{5})+\frac{1}{m}\log\!\left(1-(-\varphi^{-2})^{m+2}\right),
\]
so the limit is $\log(\varphi/2)=\log\varphi-\log 2$.
The statement for $d_m$ follows because $\log d_m=-\log f_{\mathrm{stab}}(m)$.
For the error bound, set $x_m:=-(-\varphi^{-2})^{m+2}$.
Then $|x_m|=\varphi^{-2m-4}<1/2$ for all $m\ge 1$, and the elementary bound $|\log(1+x)|\le 2|x|$ for $|x|\le 1/2$ gives
$
|\eta_m|=|\log(1+x_m)|\le 2|x_m|=2\varphi^{-2m-4}.
$
\end{proof}

\begin{proposition}[Closed-form matching estimate for $m_\ast$]\label{prop:cosmo_mstar_closed_form}
\MathTag Fix a target $\Omega_{\mathrm{vis},0}\in(0,1)$ and define the continuous approximation
\[
\widehat m(\Omega_{\mathrm{vis},0})
:=\frac{\log(\Omega_{\mathrm{vis},0})-\log(\varphi^2/\sqrt{5})}{\log(\varphi/2)}
=\frac{\log(\Omega_{\mathrm{vis},0}\sqrt{5}/\varphi^2)}{\log(\varphi/2)}.
\]
If an integer $m$ satisfies the relative-fit condition
\[
\left|\log\!\left(\frac{f_{\mathrm{stab}}(m)}{\Omega_{\mathrm{vis},0}}\right)\right|\le \varepsilon
\]
for some $\varepsilon\ge 0$ (e.g.\ a declared log-uncertainty budget), then
\[
\left|m-\widehat m(\Omega_{\mathrm{vis},0})\right|
\le \frac{\varepsilon+2\,\varphi^{-2m-4}}{|\log(\varphi/2)|}.
\]
\end{proposition}
\begin{proof}
From Lemma~\ref{lem:entropy_gap_hidden_exponent_cosmo},
\[
\log f_{\mathrm{stab}}(m)=m\log(\varphi/2)+\log(\varphi^2/\sqrt{5})+\eta_m,\qquad |\eta_m|\le 2\varphi^{-2m-4}.
\]
Subtract $\log\Omega_{\mathrm{vis},0}$ from both sides and divide by $\log(\varphi/2)\neq 0$, to obtain
\[
m-\widehat m(\Omega_{\mathrm{vis},0})
\;=\;\frac{\log(f_{\mathrm{stab}}(m)/\Omega_{\mathrm{vis},0})-\eta_m}{\log(\varphi/2)}.
\]
Taking absolute values and using the assumed relative-fit bound and $|\eta_m|\le 2\varphi^{-2m-4}$ gives the claim.
\end{proof}

\subsubsection{A discrete energy-budget fit (interface hypothesis)}
\label{subsec:cosmo_energy_budget_fit}

\begin{assumption}[Readout-occupancy energy accounting (interface)]
\label{ass:occupancy_energy_z128}
Fix an effective window length $m$.
We assume a \emph{holographic equipartition} principle: the energy cost to maintain a readout state is uniform per microstate (1 bit of capacity $\leftrightarrow$ characteristic energy unit).
Consequently, the fractional contribution of a sector to the cosmic energy budget is proportional to the count of accessible microstates in that sector.
Under the maximum-entropy (maximal-ignorance) approximation on $\Omega_m$ (Appendix~\ref{subsec:entropy_as_counting}), this yields
\[
\Omega_{\mathrm{vis},0} = \frac{|X_m|}{|\Omega_m|} = f_{\mathrm{stab}}(m),
\qquad
\Omega_{\mathrm{dark},0} = \frac{|\Omega_m|-|X_m|}{|\Omega_m|} = f_{\mathrm{hid}}(m).
\]
\end{assumption}

\begin{remark}[Derivation from Landauer's principle]
This assumption is formally equivalent to applying Landauer's principle \cite{Landauer1961} globally: if erasing/resetting each microstate bit costs $k_B T \ln 2$, and the readout hardware cycles through all $|\Omega_m|$ states ergodically, then the time-averaged energy budget allocated to the stable sector is strictly the fraction of states it occupies.
This removes the need for ad-hoc ``dark fluids''; the ``dark sector'' is simply the energetic cost of the vast hidden state space required to support the sparse stable computational surface.
\end{remark}

\InterfaceTag Given a target present-day visible fraction $\Omega_{\mathrm{vis},0}$ (e.g.\ baryon fraction as a minimal proxy), Assumption~\ref{ass:occupancy_energy_z128} selects an effective integer window length $m_\ast$ by discrete matching.
\MatchTag Concretely, we select $m_\ast$ as the unique minimizer (with a deterministic tie-break) of the absolute log-mismatch $|\log(f_{\mathrm{stab}}(m)/\Omega_{\mathrm{vis},0})|$ over a declared finite scan window; the generator \texttt{scripts/exp\_cosmology\_energy\_budget\_fit.py} implements this rule.
Lemma~\ref{lem:f_stab_monotone} implies $f_{\mathrm{stab}}(m)$ is strictly decreasing, so the induced Voronoi cells in this one-dimensional matching problem are contiguous and the stability diagnostic is well defined.
\begin{lemma}[Voronoi boundaries for absolute log-mismatch]\label{lem:log_voronoi_geometric_mean}
\MathTag Fix two positive values $a>b>0$ and consider the distance on $(0,\infty)$ given by $d(x,y)=|\log(x/y)|$.
Then the set of targets $x>0$ that are at least as close to $a$ as to $b$ is the interval
\[
\{x>0:\ d(x,a)\le d(x,b)\}=[\sqrt{ab},\infty).
\]
Equivalently, the bisector between $a$ and $b$ under $d$ is the geometric mean $\sqrt{ab}$.
\end{lemma}
\begin{proof}
Since $a>b>0$, for $x>0$ one has
\[
|\log(x/a)|\le |\log(x/b)|
\quad\Longleftrightarrow\quad
|\log x-\log a|\le |\log x-\log b|.
\]
On the real line, the set of points closer to $\log a$ than to $\log b$ is the half-line $\log x\ge (\log a+\log b)/2$.
Exponentiating yields $x\ge \exp\bigl((\log a+\log b)/2\bigr)=\sqrt{ab}$.
\end{proof}

\paragraph{Status and falsifiability.}
\AuditTag The assumption is explicitly flagged as an interface hypothesis.
Its falsifiability route is to compare the implied hidden/stable ratio $f_{\mathrm{hid}}(m_\ast)/f_{\mathrm{stab}}(m_\ast)=d_{m_\ast}-1$ with an observationally declared reference ratio.
\MatchTag Concretely,
\[
\frac{f_{\mathrm{hid}}(m_\ast)}{f_{\mathrm{stab}}(m_\ast)}
=\frac{1-f_{\mathrm{stab}}(m_\ast)}{f_{\mathrm{stab}}(m_\ast)}
  =\frac{1}{f_{\mathrm{stab}}(m_\ast)}-1
  =d_{m_\ast}-1,
\]
so a log-mismatch against a chosen reference ratio $R_{\mathrm{ref}}$ is
$\log\!\bigl((d_{m_\ast}-1)/R_{\mathrm{ref}}\bigr)$; this is reported in the generated summary fragment for transparency.
\MatchTag Under the literal reading of Assumption~\ref{ass:occupancy_energy_z128} with $\Omega_{\mathrm{dark},0}=1-\Omega_{\mathrm{vis},0}$, the reference ratio is $(1-\Omega_{\mathrm{vis},0})/\Omega_{\mathrm{vis},0}$ (fixed once $\Omega_{\mathrm{vis},0}$ is chosen).
\MatchTag If instead one interprets ``dark'' as ``dark matter only'' while treating dark energy separately, one may compare to $\Omega_{\mathrm{DM}}/\Omega_{\mathrm{b}}$ (Planck-2018 baseline; \cite{Planck2018Parameters}), as reported in the generated summary for transparency.
