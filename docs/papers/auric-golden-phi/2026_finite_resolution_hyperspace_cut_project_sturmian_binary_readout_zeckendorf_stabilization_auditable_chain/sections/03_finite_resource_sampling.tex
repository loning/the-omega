\section{有限资源采样：内部相位的环面平移与周期近似}

\subsection{内部相位动力学与一维因子}

\begin{definition}[内部相位平移]\label{def:internal-phase-translation}
令内部空间的商空间
\[
\TT_{\perp}:=E_{\perp}/\pi_{\perp}(L)
\]
为紧致阿贝尔群（典型为三维环面）。取平移向量 $\beta\in \TT_{\perp}$。定义离散时间动力学
\[
u_{t+1}=u_t+\beta,\qquad t\in\ZZ_{\ge 0}.
\]
\end{definition}

\begin{definition}[一维观测因子]\label{def:one-dimensional-factor}
取连续群同态（或可测因子映射）
\[
\phi:\TT_{\perp}\to \TT^1=\RR/\ZZ,
\]
令
\[
\theta_t:=\phi(u_t)\in\TT^1.
\]
则 $\theta_t$ 满足一维旋转
\[
\theta_{t+1}=\theta_t+\alpha\quad (\mathrm{mod}\ 1),
\]
其中 $\alpha:=\phi(\beta)\in \TT^1$。
\end{definition}

\subsection{周期近似作为有限资源可实现机制（表述替代）}

\begin{assumption}[有限资源近似原则]\label{assump:finite-resource-cf}
观测协议在有限计算/有限记忆条件下不能无限精确地实现无理旋转参数 $\alpha$，其可实现参数集合由连分数截断产生的收敛子序列 $\alpha_k=p_k/q_k$ 给出，其中
\[
\alpha_k\to \alpha,\qquad \gcd(p_k,q_k)=1.
\]
因此，在有限时间尺度内观测轨道可由周期 $q_k$ 的闭合轨道近似；随资源增长（$k$ 增大）近似精度提升，周期近似体形成层级序列。
\end{assumption}

\begin{remark}
上述表述将“可达采样”从直觉性语言替换为“资源约束下的参数逼近域”；其数学内容为连分数理论与环面平移的周期近似。
\end{remark}
