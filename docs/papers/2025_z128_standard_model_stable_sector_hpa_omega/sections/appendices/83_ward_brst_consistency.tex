\section{Ward/BRST consistency for the finite-order EFT representative}
\label{app:ward_brst_consistency}

This appendix records the minimal theorem-level (algebraic) consistency statements associated with gauge symmetry in the EFT representative of Appendix~\ref{app:eft_representative_construction}.
The role of these statements is to separate:
(i) purely algebraic identities that follow from gauge invariance (classical BRST closure),
from (ii) quantization/renormalization questions (regularization, anomalies, counterterms) that are explicitly scoped in Appendix~\ref{app:renormalization_dictionary_and_boundaries}.

\subsection{BRST differential as a nilpotent symmetry operator (algebraic core)}

\paragraph{Setup (schematic).}
Fix the gauge group $G=U(1)\times SU(2)\times SU(3)$ and the field content used by the EFT representative (Appendix~\ref{app:eft_representative_construction}):
gauge fields $A_\mu$, chiral fermions $\psi$, and a Higgs doublet $H$.
Introduce the standard ghost fields $c$ valued in the Lie algebra $\mathfrak{g}$ and, when gauge fixing is discussed, the auxiliary (Nakanishi--Lautrup) fields $b$.
We treat all fields as formal variables in a graded algebra of local functionals.

\begin{definition}[BRST differential (schematic)]\label{def:z128_brst_differential}
\MathTag The BRST differential $s$ is the graded derivation on the algebra of local functionals generated by the rules
\[
sA_\mu = D_\mu c,\qquad sc = -\tfrac12[c,c],
\]
with $s$ acting on matter fields as an infinitesimal gauge transformation with parameter $c$ (representation-dependent),
and extended to products by the graded Leibniz rule.
For gauge fixing in a standard BRST-exact form one also introduces
\[
s\bar c = b,\qquad sb = 0,
\]
where $\bar c$ is the antighost.
\end{definition}

\begin{lemma}[Nilpotency]\label{lem:brst_nilpotent}
\MathTag The BRST differential satisfies $s^2=0$ on the generators and hence on all local functionals.
\end{lemma}
\begin{proof}
This is standard: $s^2$ vanishes on $A_\mu$ by the Jacobi identity and the definition of $sc$; it vanishes on $c$ because $[c,[c,c]]=0$ in a Lie algebra; and it vanishes on $(\bar c,b)$ by definition.
The graded Leibniz rule extends the result to all polynomials/functionals.
\end{proof}

\subsection{Gauge invariance implies BRST closure (finite-order EFT)}

The finite-order EFT candidate family $\Theta_{\mathrm{EFT}}(B,D_{\max})$ was defined using \emph{gauge-invariant operator types} (Definition~\ref{def:eft_operator_basis}).
This has an immediate algebraic consequence.

\begin{proposition}[BRST closure of gauge-invariant operator types]\label{prop:brst_closure_operator_types}
\MathTag Let $\mathcal{O}$ be a gauge-invariant local operator (under $G$) built from $(A_\mu,\psi,H)$ and covariant derivatives.
Then $s\mathcal{O}=0$.
\end{proposition}
\begin{proof}
By definition, $s$ acts as an infinitesimal gauge transformation on fields, with parameter $c$.
Gauge invariance of $\mathcal{O}$ means its infinitesimal gauge variation vanishes; hence its BRST variation vanishes.
\end{proof}

\begin{corollary}[BRST closure of the finite-order EFT representative]\label{cor:brst_closure_eft_representative}
\MathTag Let $\mathcal{L}_{\mathrm{EFT}}$ be any Lagrangian density in $\Theta_{\mathrm{EFT}}(B,D_{\max})$ built from gauge-invariant operator types.
Then $s\mathcal{L}_{\mathrm{EFT}}=0$ and therefore $sS_{\mathrm{EFT}}=0$ for the corresponding action functional $S_{\mathrm{EFT}}=\int\dd^4x\,\mathcal{L}_{\mathrm{EFT}}$.
\end{corollary}
\begin{proof}
Linearity of $s$ and Proposition~\ref{prop:brst_closure_operator_types}.
\end{proof}

\subsection{Ward/Slavnov--Taylor identities: theorem statement and scoped prerequisites}

To speak about Ward identities one must specify how observables are evaluated in a state.
Appendix~\ref{app:observable_algebra_state_base} records the unified observable-algebra/state base package used in this manuscript.

\begin{definition}[BRST-exact observables]\label{def:brst_exact_observable}
\MathTag An observable (local functional) $X$ is \emph{BRST-exact} if $X=sY$ for some $Y$.
\end{definition}

\begin{assumption}[BRST-invariant quantization map]\label{ass:brst_invariant_quantization}
\AuditTag Fix a quantization/expectation functional $\langle\cdot\rangle$ on a class of observables for the EFT representative (for example, a perturbative construction at fixed scheme and scale).
Assume it is compatible with BRST symmetry in the sense that
\[
\langle sX\rangle = 0
\]
for all admissible observables $X$ (or, equivalently, that the corresponding generating functional satisfies the Slavnov--Taylor identity).
\end{assumption}

\begin{theorem}[Ward identity / BRST decoupling under BRST-invariant quantization]\label{thm:ward_brst_decoupling}
\MathTag Under Assumption~\ref{ass:brst_invariant_quantization}, all BRST-exact observables have vanishing expectation:
if $X=sY$ then $\langle X\rangle=0$.
\end{theorem}
\begin{proof}
Immediate from Assumption~\ref{ass:brst_invariant_quantization}.
\end{proof}

\paragraph{Interpretation and scope.}
\AuditTag Theorem~\ref{thm:ward_brst_decoupling} is recorded as the theorem-level \emph{logical form} of the Ward/Slavnov--Taylor constraint.
The substantive work (and the potential failure modes) lies in establishing Assumption~\ref{ass:brst_invariant_quantization} in a concrete renormalization scheme at finite order and under truncation; this is why renormalization is explicitly scoped in Appendix~\ref{app:renormalization_dictionary_and_boundaries}.

\subsection{Anomaly obstruction as a cohomological failure mode (minimal)}

\begin{definition}[BRST anomaly functional (minimal form)]\label{def:brst_anomaly_functional}
\MathTag A BRST anomaly is a functional $\mathcal{A}$ (of ghost number $1$) such that the BRST identity fails by
\[
\langle sX\rangle = \langle \mathcal{A}\cdot X\rangle
\]
for a suitable class of observables $X$.
\end{definition}

\begin{lemma}[Wess--Zumino consistency (schematic)]\label{lem:wess_zumino_consistency}
\MathTag If an anomaly functional $\mathcal{A}$ arises from a deformation of a nilpotent BRST symmetry, then it satisfies the consistency condition
\[
s\mathcal{A}=0.
\]
\end{lemma}
\begin{proof}
Apply $s$ to the anomalous Ward identity and use $s^2=0$ (Lemma~\ref{lem:brst_nilpotent}).
\end{proof}

\begin{remark}[Trivial vs.\ nontrivial anomalies]\label{rem:trivial_nontrivial_anomalies}
\AuditTag If $\mathcal{A}=s\mathcal{B}$ for some $\mathcal{B}$, then the anomaly is cohomologically trivial and can be removed by a local counterterm/renormalization redefinition at the level of the effective action.
Nontrivial anomalies correspond to nonzero cohomology classes and are genuine obstructions to Assumption~\ref{ass:brst_invariant_quantization}.
In the present manuscript, anomaly cancellation constraints for the closed SM labeling content are treated as non-negotiable interface consistency requirements (Section~\ref{sec:sm_labeling_closure}), precisely to avoid such obstructions at the level of gauge symmetry.
\end{remark}

\subsection{Minimal failure points (audit-facing)}

\AuditTag For later theoremization steps (T1-3/T1-4), the minimal failure points for Ward/BRST consistency are:
\begin{itemize}
\item \textbf{(WBR1) Regularization/scheme breaking.} A chosen regularization or renormalization prescription may violate BRST invariance, requiring restoration by counterterms; this is scheme-scoped (Appendix~\ref{app:renormalization_dictionary_and_boundaries}).
\item \textbf{(WBR2) Gauge anomaly.} A nontrivial BRST anomaly class obstructs the Ward identity (Definition~\ref{def:brst_anomaly_functional} and Remark~\ref{rem:trivial_nontrivial_anomalies}).
\item \textbf{(WBR3) Truncation mismatch.} If one enforces Ward/BRST identities only up to dimension $D_{\max}$, the remainder must be recorded explicitly as an $O(\Lambda^{-(D_{\max}-4)})$ effect in the EFT power counting; this belongs to the quantitative error-budget layer (T1-4).
\end{itemize}
