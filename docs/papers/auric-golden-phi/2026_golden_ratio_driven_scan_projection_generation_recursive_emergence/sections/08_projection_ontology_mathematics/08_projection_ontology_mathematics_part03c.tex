\begin{proposition}[\texorpdfstring{$h_q$}{hq} 的几何化：投影输出测度的 Rényi 维数谱]\label{prop:pom-renyi-dimension-spectrum}

\[

令 \(m\to\infty\) 即得结论。证毕。

\begin{proposition}[普适夹逼：高阶碰撞主指数与投影熵率仅由 \texorpdfstring{$|X_m|$}{|Xm|} 与最大纤维锁定]\label{prop:pom-rq-universal-bounds}
对整数 \(q\ge 2\)，令
$$
S_q(m):=\sum_{x\in X_m} d_m(x)^q,\qquad d_m(x):=\bigl|\Fold_m^{-1}(x)\bigr|.
$$
并令 \(r_q\) 表示 \(S_q(m)\) 的指数增长率（例如定理 \ref{thm:pom-collision-kernel-ak} 给出的 Perron 根；等价地 \(r_q=\lim_{m\to\infty}S_q(m)^{1/m}\)），定义未归一化 \(q\)-阶投影熵率
$$
h_q:=\log\!\left(\frac{2^q}{r_q}\right)=q\log 2-\log r_q.
$$
则有两侧界
$$
\frac{2^q}{\varphi^{q-1}}\ \le\ r_q\ \le\ 2\,\varphi^{(q-1)/2},
$$
等价地
$$
(q-1)\log\!\left(\frac{2}{\sqrt{\varphi}}\right)\ \le\ h_q\ \le\ (q-1)\log\varphi.
$$
\end{proposition}
\begin{proof}
对任意固定 \(m\) 与 \(q\ge 2\)，由幂均值/Holder 不等式，
$$
S_q(m)=\sum_{x\in X_m}d_m(x)^q\ \ge\ \frac{\bigl(\sum_x d_m(x)\bigr)^q}{|X_m|^{\,q-1}}
=\frac{2^{qm}}{F_{m+2}^{\,q-1}},
$$
其中 \(\sum_x d_m(x)=|\Omega_m|=2^m\) 且 \(|X_m|=F_{m+2}\)。
另一方面令 \(D_m:=\max_{x\in X_m} d_m(x)\)，则
$$
S_q(m)=\sum_x d_m(x)^q\ \le\ D_m^{\,q-1}\sum_x d_m(x)=2^m D_m^{\,q-1}.
$$
两式取对数并除以 \(m\)，再令 \(m\to\infty\)，结合 Fibonacci 渐近 \(F_{m+2}\sim \varphi^{m}/\sqrt5\) 与最大纤维渐近 \(D_m=\Theta(\varphi^{m/2})\)（定理 \ref{thm:pom-max-fiber}），得到
$$
\log r_q=\lim_{m\to\infty}\frac{1}{m}\log S_q(m)\in\Bigl[q\log 2-(q-1)\log\varphi,\ \log 2+\frac{q-1}{2}\log\varphi\Bigr],
$$
从而 \(r_q\) 的界成立。代入 \(h_q=q\log 2-\log r_q\) 即得 \(h_q\) 的等价界。证毕。
\end{proof}

\begin{proposition}[极端阶端点：\texorpdfstring{$r_q^{1/q}\to\sqrt{\varphi}$}{rq^(1/q)->sqrt(phi)} 与归一化熵端点常数]\label{prop:pom-rq-qinfty-endpoint}
令 \(r_q\) 表示 \(S_q(m)\) 的指数增长率（例如由 moment-kernel 的 Perron 根给出），并记 \(\varphi\) 为黄金比例。
则当 \(q\to\infty\) 时有极限
\[
\boxed{\;
\lim_{q\to\infty} r_q^{1/q}=\sqrt{\varphi}\; }.
\]
等价地，
\[
\lim_{q\to\infty}\frac{1}{q}\,h_q
=\log\!\left(\frac{2}{\sqrt{\varphi}}\right),
\qquad
\lim_{q\to\infty}\frac{1}{q-1}\,h_q
=\log\!\left(\frac{2}{\sqrt{\varphi}}\right),
\]
从而 Rényi 维数谱端点满足
\[
\lim_{q\to\infty}D_q
=\frac{\log\!\left(2/\sqrt{\varphi}\right)}{\log\varphi}
\]
（其中 \(D_q\) 见命题 \ref{prop:pom-renyi-dimension-spectrum}）。
\par
并且对任意整数 \(q\ge 2\) 有显式误差界
\[
\boxed{\;
0\ \le\ \log\!\left(\frac{2}{\sqrt{\varphi}}\right)-\frac{h_q}{q}\ \le\ \frac{\log\varphi}{q}\; }.
\]
\end{proposition}
\begin{proof}
对任意 \(m,q\ge 2\)，由
\(
D_m:=\max_x d_m(x)
\)
得两侧夹逼
\[
D_m^q\ \le\ S_q(m)\ \le\ |X_m|\,D_m^q.
\]
两端取 \(1/(mq)\) 次方，得到
\[
D_m^{1/m}\ \le\ S_q(m)^{1/(mq)}\ \le\ D_m^{1/m}\,|X_m|^{1/(mq)}.
\]
令 \(m\to\infty\)，左端由定理 \ref{thm:pom-max-fiber} 给出
\(
\lim_{m\to\infty}D_m^{1/m}=\varphi^{1/2}
\)；
右端的因子满足
\(
\lim_{m\to\infty}|X_m|^{1/(mq)}=\varphi^{1/q}
\)
（因为 \(|X_m|=F_{m+2}\sim \varphi^m/\sqrt5\)）。
因此对每个固定 \(q\ge 2\) 有
\[
\sqrt{\varphi}\ \le\ r_q^{1/q}\ \le\ \sqrt{\varphi}\,\varphi^{1/q}.
\]
令 \(q\to\infty\) 即得 \(\lim_{q\to\infty}r_q^{1/q}=\sqrt{\varphi}\)。
最后由 \(h_q=q\log 2-\log r_q\) 与命题 \ref{prop:pom-renyi-dimension-spectrum} 的闭式，直接得到各端点极限。
\par
由 \(\sqrt{\varphi}\le r_q^{1/q}\le \sqrt{\varphi}\,\varphi^{1/q}\) 得
\[
\frac{1}{2}\log\varphi\ \le\ \frac{1}{q}\log r_q\ \le\ \frac{1}{2}\log\varphi+\frac{1}{q}\log\varphi.
\]
代入 \(\frac{h_q}{q}=\log 2-\frac{1}{q}\log r_q\)，即得
\[
0\ \le\ \log\!\left(\frac{2}{\sqrt{\varphi}}\right)-\frac{h_q}{q}\ \le\ \frac{\log\varphi}{q}.
\]
证毕。
\end{proof}

\begin{proposition}[压力的凸性与指纹的凹性]\label{prop:pom-pressure-convexity}
对任意 \(m\ge 2\)，函数 \(q\mapsto \log S_q(m)\) 在 \(q\ge 0\) 上为凸函数。
因此在 \(S_q(m)=\Theta(r_q^m)\)（例如由命题 \ref{prop:pom-moment-kernel-exists} 的 Perron 主尺度保证）时，压力
\[
P(q):=\log r_q
\]
在整数点上满足离散凸性：对任意整数 \(q_0<q_1<q_2\) 有
\[
\boxed{\;
P(q_1)\ \le\ \frac{q_2-q_1}{q_2-q_0}\,P(q_0)\ +\ \frac{q_1-q_0}{q_2-q_0}\,P(q_2)\; }.
\]
等价地，未归一化 Rényi 指纹 \(h_q=q\log 2-P(q)\) 在整数点上为凹的：
\[
\boxed{\;
h_{q_1}\ \ge\ \frac{q_2-q_1}{q_2-q_0}\,h_{q_0}\ +\ \frac{q_1-q_0}{q_2-q_0}\,h_{q_2}\; }.
\]
\end{proposition}
\begin{proof}
固定 \(m\)。取任意 \(q_0,q_2\ge 0\) 与 \(\theta\in(0,1)\)，令 \(q_1=\theta q_0+(1-\theta)q_2\)。
由 Hölder 不等式（或 log-sum-exp 的凸性）得
\[
S_{q_1}(m)=\sum_{x\in X_m}d_m(x)^{q_1}
\le
\left(\sum_{x\in X_m}d_m(x)^{q_0}\right)^{\theta}
\left(\sum_{x\in X_m}d_m(x)^{q_2}\right)^{1-\theta}
=S_{q_0}(m)^{\theta}S_{q_2}(m)^{1-\theta}.
\]
取对数即得 \(q\mapsto \log S_q(m)\) 的凸性。
\par
令 \(q_0<q_1<q_2\) 为整数并取
\(
\theta=\frac{q_2-q_1}{q_2-q_0}
\)
使得 \(q_1=\theta q_0+(1-\theta)q_2\)。
对上式取 \(\frac{1}{m}\log\) 并令 \(m\to\infty\)，得到
\[
\log r_{q_1}\le \theta\log r_{q_0}+(1-\theta)\log r_{q_2},
\]
即 \(P\) 的离散凸性。最后 \(h_q=q\log 2-P(q)\) 为线性函数减去凸函数，从而在整数点上为凹。证毕。
\end{proof}

\end{remark}

\begin{theorem}[坍缩/共振判别：模态约束导致 Hankel 秩下降]\label{thm:pom-moment-resonance}
设 \(q\ge 2\)，记 \(h=\lfloor q/2\rfloor\)。在差分模态截断下（注 \ref{rem:pom-closure-dim-mode-trunc}），令 \(M_{q,\ell}(m)\) 表示净偏置 \(\ell=a-b\) 的模态贡献，则
$$
S_q(m)=\sum_{\ell=-h}^{h} M_{q,\ell}(m).
$$
并假设这些模态贡献可组织为一个以 \(\{M_{q,\ell}\}\) 为坐标的 \(2h+1\) 维线性 realization（非共振情形下正是“模态闭合推进”的标准实现）。
设存在从某个 \(m_0\) 起成立的整系数线性递推
$$
S_q(m)=\sum_{i=1}^{d} c_i S_q(m-i)\qquad(m\ge m_0),
$$
其特征多项式
$$
P_q(\lambda)=\lambda^d-\sum_{i=1}^d c_i\lambda^{d-i}.
$$
若 \(d<2h+1\)，则：
\begin{enumerate}
  \item \(S_q\) 的 Hankel 秩满足 \(\mathrm{rank}(H)\le d<2h+1\)；
  \item 该 \(2h+1\) 维线性 realization 必非最小，从而存在非零系数 \(\alpha_{-h},\dots,\alpha_h\) 使
  $$
  \sum_{\ell=-h}^{h}\alpha_\ell M_{q,\ell}(m)=0\qquad(m\ge m_0),

\end{proof}

\end{corollary}

\subsubsection{任意阶碰撞矩的 moment-kernel 构造与多重分形谱}\label{subsec:pom-moment-general}

\paragraph{从 \texorpdfstring{$q$}{q}-重碰撞到有限态 moment-kernel}
二阶与三阶的核心新点是：碰撞矩并非仅是一个“可枚举序列”，而是由一个\textbf{有限状态纤维积核}生成，从而完全落入 Perron--Frobenius/\(\zeta\) 的统一接口。
这一事实对任意 \(q\ge 2\) 都成立：固定 \(q\) 后，“\(q\) 条窗口折叠到同一稳定类型”的约束是一个有限记忆条件，因此可被编译为一个有限图（SFT/sofic）并由整数邻接矩阵计数。

\begin{definition}[\texorpdfstring{$q$}{q}-重碰撞关系与 moment 序列]\label{def:pom-moment-q}
对整数 \(q\ge 2\)，定义 \(q\)-重碰撞关系
$$
R_m^{(q)}:=
\{(\omega^{(1)},\dots,\omega^{(q)})\in\Omega_m^q:\ \Fold_m(\omega^{(1)})=\cdots=\Fold_m(\omega^{(q)})\},
$$
并定义 \(q\)-阶碰撞矩（moment）
$$
S_q(m):=|R_m^{(q)}|=\sum_{x\in X_m} d_m(x)^q.
$$
\end{definition}

\begin{proposition}[任意阶 moment-kernel 的存在与可执行构造]\label{prop:pom-moment-kernel-exists}
对每个固定的 \(q\ge 2\)，存在一个有限状态非负整数核（邻接矩阵）\(A_q\) 与整数读出向量 \(u_q,v_q\) 使得
$$
S_q(m)=u_q^{\mathsf T}A_q^{\,m-m_0}v_q\qquad(m\ge m_0),
$$
从而 \(\{S_q(m)\}_{m\ge m_0}\) 满足某个\textbf{有限阶整系数线性递推}。

并且 \(A_q\) 可按下述“编译路线”显式构造（因此可推演、可重写、可计算）：
\begin{enumerate}
  \item 选取实现 \(\Fold_\infty\) 的确定性在线核（延迟 \(\delta_{\min}=3\) 的同步核，附录给出其显式有限状态表示）；
  \item 取其 \(q\) 份拷贝并做同步直积，把状态扩展为 \(q\)-元组；
  \item 在每一步只保留那些使 \(q\) 份拷贝产生\textbf{相同输出位}的转移，并投影到可达的本质强连通分量；
  \item 得到的有限图即为一个 \(q\)-重碰撞的 fiber-product 核，其邻接矩阵就是所求 \(A_q\)，其路径计数即给出 \(S_q(m)\)。
\end{enumerate}
\end{proposition}

\begin{proof}
把 \(\Fold_\infty\) 视为一个确定性有限转导器：输入为窗口字母（或其由两条合法输入轨道诱导的输入字母），输出为稳定地址比特；其内部状态空间有限且具有常数延迟（命题 \ref{prop:pom-delay-min} 的在线核即是一个具体实现）。
固定 \(q\) 后，约束“\(q\) 份输出逐位相同”是一个局部同步约束：它只需在每一步比较当前输出符号是否一致，因而仍是有限状态可识别的。
因此可对该转导器做同步直积并施加“输出一致”的筛选，得到一个有限有向图 \(G^{(q)}\)。
长度 \(m\) 的合法路径条数等于 \(R_m^{(q)}\) 的计数（即 \(S_q(m)\)），从而可写成某个非负整数邻接矩阵的矩阵幂读出形式 \(u_q^{\mathsf T}A_q^{m-m_0}v_q\)。
线性递推由 Cayley--Hamilton 定理立得，且递推系数为整数（因为 \(A_q\) 为整数矩阵）。证毕。
\end{proof}

\paragraph{从 \texorpdfstring{$r_q$}{r\_q} 到压力与“纤维多重度谱”：凸共轭上界（严格）}
已核验的 \(q\)-阶碰撞核满足 \(S_q(m)=\Theta(r_q^m)\)，其中 \(r_q=\rho(A_q)\) 为 Perron 根（推论 \ref{cor:pom-s2-asymptotic}、\ref{cor:pom-s3-asymptotic}、\ref{cor:pom-s4-asymptotic} 及表 \ref{tab:fold_collision_moment_recursions}）。
因此 moment 指纹不仅给出单点常数 \(h_q=\log(2^q/r_q)\)，还诱导出一条可做 Legendre 变换的“压力曲线”，从而把“空间损失”提升为真正的谱结构。

\begin{definition}[moment 压力与归一化对数矩母函数]\label{def:pom-moment-pressure}
对 \(q\ge 2\) 定义
$$
\tau(q):=\limsup_{m\to\infty}\frac{1}{m}\log S_q(m),
$$
并定义归一化压力（即输出分布 \(\pi_m(x)=d_m(x)/2^m\) 的 \(q\)-阶对数矩母函数）
$$
P(q):=\limsup_{m\to\infty}\frac{1}{m}\log\sum_{x\in X_m}\pi_m(x)^q
=\tau(q)-q\log 2.
$$
当存在 Perron 根 \(r_q\) 使 \(S_q(m)=\Theta(r_q^m)\) 时，有 \(\tau(q)=\log r_q\)，并且 \(P(q)=\log(r_q/2^q)=-h_q\)。
\end{definition}

\begin{definition}[纤维多重度的对数谱函数]\label{def:pom-fiber-spectrum}
令
$$
\gamma_m(x):=\frac{1}{m}\log d_m(x)\qquad(x\in X_m).
$$
