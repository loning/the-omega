\section{Ostrowski numeration and Zeckendorf coordinates: canonical discrete arithmetic}
\label{sec:zeckendorf}

\subsection{Ostrowski representation for a general irrational slope}
Let $\alpha\in(0,1)\setminus\QQ$ have continued fraction expansion
$$
\alpha=[0;a_1,a_2,a_3,\dots],\qquad a_i\in\NN.
$$
Let $q_k$ be the denominators of convergents, defined by
$$
q_{-1}=0,\quad q_0=1,\quad q_{k+1}=a_{k+1}q_k+q_{k-1}.
$$
Ostrowski numeration asserts that every $N\in\NN_{>0}$ admits a unique expansion (see e.g.~\cite{AlloucheShallit2003,Lothaire2002})
$$
N=\sum_{k=0}^{m} b_{k+1} q_k,
$$
where the digits satisfy
\begin{equation}
\label{eq:ostrowski_digits}
0\le b_1<a_1,\qquad
0\le b_{k+1}\le a_{k+1}\ (k\ge 1),\qquad
b_{k+1}=a_{k+1}\ \Rightarrow\ b_k=0\ (k\ge 1).
\end{equation}
This numeration shares a common source with Sturmian mechanical words: both are controlled by the continued fraction data $(a_i)$; see e.g.~\cite{AlloucheShallit2003,Lothaire2002}.

\subsection{Golden degeneration: Zeckendorf as the binary canonical form}
For $\alpha=\varphi^{-1}=[0;1,1,1,\dots]$, we have $a_i=1$ for all $i$, hence
$$
q_{k+1}=q_k+q_{k-1},
$$
so $(q_k)$ is Fibonacci-type. The digit constraints degenerate to
$$
b_1=0,\qquad b_{k+1}\in\{0,1\}\ (k\ge 1),\qquad b_{k+1}=1\ \Rightarrow\ b_k=0\ (k\ge 1),
$$
i.e. no adjacent ones.

\begin{lemma}[Convergent denominators and Fibonacci bases]
\label{lem:qk_fib}
For $\alpha=\varphi^{-1}$ the convergent denominators satisfy $q_0=1$, and for every $k\ge 1$ one has
$$
q_k=F_k,
$$
where $(F_k)$ is the Zeckendorf Fibonacci base defined below by $F_1=1$, $F_2=2$, $F_{n+2}=F_{n+1}+F_n$.
\end{lemma}

\begin{proof}
From the recurrence $q_{k+1}=q_k+q_{k-1}$ with $q_0=1$ and $q_1=1$, we obtain
$$
(q_k)_{k\ge 0}=(1,1,2,3,5,\dots).
$$
The Zeckendorf base satisfies the same recurrence with initial data $F_1=1$ and $F_2=2$, hence
$$
(F_k)_{k\ge 1}=(1,2,3,5,\dots).
$$
Thus $q_1=1=F_1$ and by induction $q_k=F_k$ for all $k\ge 1$.
\end{proof}

To align with the classical Zeckendorf convention, define the Fibonacci base
$$
F_1=1,\qquad F_2=2,\qquad F_{n+2}=F_{n+1}+F_n.
$$

\begin{theorem}[Zeckendorf uniqueness]
\label{thm:zeckendorf}
Every $N\in\NN_{>0}$ admits a unique representation
$$
N=\sum_{k=1}^{m} c_k F_k,
\qquad c_k\in\{0,1\},\qquad c_k c_{k+1}=0.
$$
\end{theorem}

\begin{proof}
See~\cite{Zeckendorf1972}.
\end{proof}

\subsection{Algorithms and interface to readout}
\begin{itemize}
    \item \textbf{Canonical encoding.} Given $N$ and the convergent denominators $(q_k)$ of $\alpha$, one computes the Ostrowski digits $(b_{k+1})$ by a greedy algorithm subject to the constraints in \eqref{eq:ostrowski_digits}; see e.g.~\cite{AlloucheShallit2003,Lothaire2002}. In the golden branch this degenerates to the Zeckendorf greedy algorithm, yielding the canonical bitstring $(c_k)$.
    \item \textbf{Readout interface.} Given a finite readout prefix $(s_0,\dots,s_{N-1})$, form the staircase coordinate $S_N=\sum_{k=0}^{N-1}s_k$. In the golden branch, canonicalization is achieved by Zeckendorf normalization (Theorem~\ref{thm:zeckendorf}); for a general irrational slope, the analogous canonical coding is given by Ostrowski digits $(b_{k+1})$.
\end{itemize}
