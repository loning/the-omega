
\section{\texorpdfstring{$S$}{S}-inversion as scale exchange and algebraic equivalence}
\label{sec:S_duality_morita}

\subsection{Modular inversion and cusp endpoint identification}
\label{sec:modular_S_inversion}

The modular group is generated by
\begin{equation}
T:\tau\mapsto \tau+1,\qquad S:\tau\mapsto -\frac{1}{\tau}.
\end{equation}
On the modular curve $X(1)$, the cusp orbit is unique, so $0\sim \infty$ in the quotient. The inversion $S$ exchanges these endpoints and therefore provides a strict mathematical template for ``deep/shallow'' identification.

Explicitly, for $\tau=x+\iu y$,
\begin{equation}
S(\tau)=-\frac{1}{\tau}=-\frac{x-\iu y}{x^2+y^2},
\qquad
\Im S(\tau)=\frac{y}{x^2+y^2}.
\end{equation}
Thus, when $x$ is not large, $y$ is exchanged with $1/y$ at the level of scale magnitude: large height and small height are directly related by $S$.

\subsection{Morita equivalence of noncommutative tori and the \texorpdfstring{$SL_2(\ZZ)$}{SL2(Z)} action}
\label{sec:morita_equivalence_nct}

Noncommutative tori admit a canonical $SL_2(\ZZ)$ action on the slope parameter:
\begin{equation}
\alpha'=\frac{a\alpha+b}{c\alpha+d},
\qquad
\gamma=\begin{pmatrix}a&b\\c&d\end{pmatrix}\in SL_2(\ZZ).
\end{equation}
A foundational result is that $A_\alpha$ and $A_{\alpha'}$ are Morita equivalent (and therefore represent the same ``noncommutative geometry'' up to equivalence of module categories) \cite{Rieffel1981,Rieffel1988,Connes1994}.

\begin{theorem}[Morita equivalence classification (standard)]
\label{thm:morita_classification_nct}
For irrational parameters $\alpha,\beta\in\RR\setminus\QQ$, the noncommutative tori $A_\alpha$ and $A_\beta$ are (strongly) Morita equivalent if and only if there exists $\gamma=\begin{pmatrix}a&b\\c&d\end{pmatrix}\in SL_2(\ZZ)$ such that
\begin{equation}
\beta=\frac{a\alpha+b}{c\alpha+d}.
\end{equation}
\end{theorem}
\begin{remark}
See \cite{Rieffel1988,Connes1994} for proofs and further structure (projective modules and $K$-theory invariants).
\end{remark}

In particular, the modular inversion
\begin{equation}
S=\begin{pmatrix}0&-1\\1&0\end{pmatrix}
\end{equation}
acts by $\alpha\mapsto -1/\alpha$, which is structurally aligned with the Gauss-map inversion step underlying continued fractions.

\subsection{Fourier exchange: swapping scan and readout roles}
\label{sec:fourier_exchange_weyl}

In the covariant representation~\eqref{eq:covariant_model}, the Fourier transform exchanges translation and multiplication. Abstractly, this realizes a ``scan--readout swap'': in one representation, $U$ acts as a tick shift and $V$ acts as a phase; in the Fourier-dual representation, these roles are interchanged.

\begin{proposition}[Fourier exchange for the covariant Weyl pair]
\label{prop:fourier_exchange_weyl}
Let $U,V$ act on $\hilbert=L^2(\RR/\ZZ)$ by $(U\psi)(x)=\psi(x+\alpha)$ and $(V\psi)(x)=\e^{2\pi\iu x}\psi(x)$ as in~\eqref{eq:covariant_model}. Let $\mathcal{F}:\hilbert\to \ell^2(\ZZ)$ be the Fourier transform
\begin{equation}
(\mathcal{F}\psi)(k)=\int_0^1 \psi(x)\,\e^{-2\pi\iu k x}\,\dd x.
\end{equation}
Then on $\ell^2(\ZZ)$ one has
\begin{equation}
\mathcal{F}U\mathcal{F}^{-1}:\widehat\psi(k)\mapsto \e^{2\pi\iu k\alpha}\widehat\psi(k),
\qquad
\mathcal{F}V\mathcal{F}^{-1}:\widehat\psi(k)\mapsto \widehat\psi(k-1).
\end{equation}
In particular, translation (scan shift) becomes phase multiplication in Fourier space, while phase multiplication becomes an index shift, realizing a concrete scan--readout exchange.
\end{proposition}
\begin{proof}
For the first identity, compute
\begin{equation}
(\mathcal{F}U\psi)(k)=\int_0^1 \psi(x+\alpha)\e^{-2\pi\iu k x}\,\dd x
=\e^{2\pi\iu k\alpha}\int_0^1 \psi(u)\e^{-2\pi\iu k u}\,\dd u
=\e^{2\pi\iu k\alpha}(\mathcal{F}\psi)(k),
\end{equation}
using the substitution $u=x+\alpha$ and periodicity on $\RR/\ZZ$. The second identity follows similarly:
\begin{equation}
(\mathcal{F}V\psi)(k)=\int_0^1 \e^{2\pi\iu x}\psi(x)\e^{-2\pi\iu k x}\,\dd x
=\int_0^1 \psi(x)\e^{-2\pi\iu (k-1)x}\,\dd x
=(\mathcal{F}\psi)(k-1).
\end{equation}
\end{proof}

\begin{remark}[interpretation-layer language]
It is tempting to interpret $S$ and Morita/Fourier equivalence as a ``scale-exchange wormhole'' that short-circuits intermediate scales. Such language is Layer~2 semantics. The Layer~0/1 content used here is strictly the symmetry/equivalence structure: cusp endpoint identification on $X(1)$ and Morita/Fourier exchange on $A_\alpha$.
\end{remark}
