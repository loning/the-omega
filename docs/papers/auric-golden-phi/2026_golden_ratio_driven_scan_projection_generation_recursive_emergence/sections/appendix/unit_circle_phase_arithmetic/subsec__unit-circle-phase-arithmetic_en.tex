\subsection{Arithmetic on the Phase Box: Structure Transport (Lift--Operate--Read Out)}\label{subsec:unit-circle-phase-arithmetic}
Write the phase box as
\[
\mathbb{U}:=\Bigl(-\tfrac12,\tfrac12\Bigr).
\]
Viewing \(\upsilon:\RR\to\mathbb{U}\) as the lift gate and \(\upsilon^{-1}(u)=\tan(\pi u)\) as the read-out gate, one can define, via structure transport on \(\mathbb{U}\), operations strictly isomorphic to real-number arithmetic.

\begin{definition}[Phase box arithmetic (structure transport)]\label{def:unit-circle-phase-transport-arith}
For any \(u,v\in\mathbb{U}\), define
\[
u\boxplus v
:=\upsilon\!\bigl(\upsilon^{-1}(u)+\upsilon^{-1}(v)\bigr),
\qquad
u\boxtimes v
:=\upsilon\!\bigl(\upsilon^{-1}(u)\cdot\upsilon^{-1}(v)\bigr).
\]
Equivalently,
\[
u\boxplus v=\frac{1}{\pi}\arctan\!\Bigl(\tan(\pi u)+\tan(\pi v)\Bigr),
\qquad
u\boxtimes v=\frac{1}{\pi}\arctan\!\Bigl(\tan(\pi u)\tan(\pi v)\Bigr).
\]
The zero element and identity element are
\[
0_{\mathbb{U}}:=\upsilon(0)=0,
\qquad
1_{\mathbb{U}}:=\upsilon(1)=\frac{1}{\pi}\arctan(1)=\frac14.
\]
\end{definition}

\begin{theorem}[Strict isomorphism with the real number field]\label{thm:unit-circle-phase-field-iso}
The map
\(
\upsilon:(\RR,+,\cdot)\to(\mathbb{U},\boxplus,\boxtimes)
\)
is a field isomorphism; that is, for any \(x,y\in\RR\),
\[
\upsilon(x+y)=\upsilon(x)\boxplus\upsilon(y),
\qquad
\upsilon(xy)=\upsilon(x)\boxtimes\upsilon(y).
\]
In particular, \((\mathbb{U},\boxplus,\boxtimes)\) satisfies commutativity, associativity, distributivity, and invertibility (except at \(0_{\mathbb{U}}\)).
\end{theorem}
\begin{proof}
By Definition~\ref{def:unit-circle-phase-transport-arith}, \(\boxplus,\boxtimes\) are the structure transport along the bijection \(\upsilon\); the isomorphism identities follow immediately from \(\upsilon^{-1}\circ \upsilon=\mathrm{id}_\RR\). The remaining field-structure properties are inherited from \((\RR,+,\cdot)\) via the isomorphism. \qed
\end{proof}

\begin{remark}[Bounded representation and boundary-carried divergence]\label{rem:unit-circle-phase-bounded}
The phase representation always lies within the bounded box \(\mathbb{U}\); ``numerical divergence'' is equivalently transported to the phase approaching the boundary:
\[
x\to\pm\infty\qquad\Longleftrightarrow\qquad \upsilon(x)\to\pm\tfrac12
\qquad\Longleftrightarrow\qquad \vartheta(x)\to\pm\pi.
\]
Hence this coordinate gate can serve as a parameter-free representation layer for ``overflow avoidance'': the principal scale no longer manifests as numerical magnitude but rather as ``boundary margin.''
\end{remark}

\begin{remark}[Definitional isomorphism with the arithmetic of Section~\ref{sec:emergent-arithmetic}]\label{rem:unit-circle-phase-transport-analogy}
Section~\ref{sec:emergent-arithmetic} defines arithmetic at a fixed stable address via the pullback \( \mathrm{Val}^{-1}(\mathrm{Val}(c)\pm\mathrm{Val}(d)) \) (see Definitions~\ref{def:stable-add} and \ref{def:stable-mul}).
Here one merely replaces the ``value map'' by \(\upsilon^{-1}=\tan(\pi\cdot)\) and the codomain by \(\RR\), thereby obtaining a continuous version of the same structure transport.
\end{remark}
