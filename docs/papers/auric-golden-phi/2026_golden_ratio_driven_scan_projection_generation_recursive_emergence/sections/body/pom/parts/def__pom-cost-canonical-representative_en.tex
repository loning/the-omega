\begin{definition}[Canonical representative selection on bounded slices (Pareto front section)]\label{def:pom-cost-canonical-representative}
In the setup of Proposition \ref{prop:pom-cost-optimization-decidable}, fix a total order \(\triangleleft\) for ranking the interface normal forms (or their deterministic encodings) in the bounded slice \(\mathbf{PW}^{\ast}_{\le}\).
For any semantic equivalence class \([w]\subseteq \mathbf{PW}^{\ast}_{\le}\), denote the realization set corresponding to its \(\mathcal{C}\)-Pareto front as
\[
\mathrm{Front}_{\mathcal C}([w])
:=
\left\{\,w'\in [w]\ :\ \mathcal{C}(w')\ \text{is Pareto minimal in}\ \mathcal{C}([w])\,\right\}.
\]
Define the \textbf{canonical representative} selection as
\[
\boxed{\;
\mathrm{Can}_{\mathcal C}([w])
:=
\min_{\triangleleft}\,\mathrm{Front}_{\mathcal C}([w])\; }.
\]
\end{definition}

\begin{corollary}[Canonical representative functor: computable section with implementation-independent optimality]\label{cor:pom-cost-canonical-representative-functor}
In the bounded slice calibration of Proposition \ref{prop:pom-cost-optimization-decidable}, \(\mathrm{Can}_{\mathcal C}\) is well-defined and computable, and satisfies:
\begin{enumerate}
  \item \textbf{(Implementation independence)} If \(w\simeq w'\), then \(\mathrm{Can}_{\mathcal C}([w])=\mathrm{Can}_{\mathcal C}([w'])\);
  \item \textbf{(Pareto optimality)} \(\mathcal{C}\bigl(\mathrm{Can}_{\mathcal C}([w])\bigr)\) is a Pareto minimal element of \(\mathcal{C}([w])\);
  \item \textbf{(Section)} \(\mathrm{Can}_{\mathcal C}\) provides a reproducible section of the natural projection \(\mathbf{PW}^{\ast}_{\le}\twoheadrightarrow \mathbf{PW}^{\ast}_{\le}/\!\!\simeq\); when the quotient set is viewed as a discrete groupoid, it can equivalently be expressed as a section functor
  \[
  \mathrm{Can}_{\mathcal C}:\ \mathbf{PW}^{\ast}_{\le}/\!\!\simeq\ \longrightarrow\ \mathbf{PW}^{\ast}_{\le}.
  \]
\end{enumerate}
\end{corollary}
\begin{proof}
By Proposition \ref{prop:pom-cost-optimization-decidable}, \(\mathrm{Front}_{\mathcal C}([w])\) is a finite set and can be computed by finite enumeration; therefore its minimum under the fixed total order \(\triangleleft\) exists and is computable, thus \(\mathrm{Can}_{\mathcal C}\) is well-defined.
Item (1) follows directly from the definition that \([w]\) depends only on the \(2\)-isomorphism class; item (2) is obtained from \(\mathrm{Can}_{\mathcal C}([w])\in \mathrm{Front}_{\mathcal C}([w])\); item (3) is the definitional expression of \(\mathrm{Can}_{\mathcal C}\).
\end{proof}
