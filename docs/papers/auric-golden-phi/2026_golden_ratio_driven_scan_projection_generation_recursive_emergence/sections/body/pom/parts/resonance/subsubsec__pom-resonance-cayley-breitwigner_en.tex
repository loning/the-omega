\begin{corollary}[Cayley Tomography: Directly Reading Out the Center--Width Parameters of Breit--Wigner from the Shadow Spectral Packet, with Closed-Form Constant \texorpdfstring{$-\sqrt{15}/5$}{-sqrt15/5}]\label{cor:pom-resonance-cayley-breitwigner-centerwidth}
Under the setting of Remark \ref{rem:pom-resonance-poisson-lorentz-prelimit}, for any point \(a=\eta e^{\mathrm{i}\theta}\in\mathbb{D}\) in the disk, define the Cayley half-plane image
\[
w:=\frac{1+a}{1-a}=\delta+\mathrm{i}\gamma\qquad(\delta>0).
\]
Then we have the explicit formulas
\[
\boxed{\;
\delta=\frac{1-\eta^{2}}{1-2\eta\cos\theta+\eta^{2}},
\qquad
\gamma=\frac{2\eta\sin\theta}{1-2\eta\cos\theta+\eta^{2}}\; }.
\]
In the Poisson (Breit--Wigner) superposition law for the phase derivative of the inner function, \(w\) corresponds to the "peak position \(\gamma\) and half-width \(\delta\)" parameters of a Lorentz kernel
\(
\delta\big/((t-\gamma)^2+\delta^2)
\).
\par
Applying this to the resonance-window shadow spectral packet, we take
\(
a_q:=\eta_q e^{\mathrm{i}\theta_q}=\mu_q/r_{q-2}
\)
and denote \(w_q=\delta_q+\mathrm{i}\gamma_q\).
If furthermore \(\eta_q\to 1\) and \(\theta_q\to \theta^\ast:=\arg(z_\ast)\) (where \(z_\ast=(-1-\mathrm{i}\sqrt{15})/4\), so \(\cos\theta^\ast=-\tfrac14\)), then \(w_q\to \mathrm{i}\gamma^\ast\) with
\[
\boxed{\;
\gamma^\ast=\cot\!\Bigl(\frac{\theta^\ast}{2}\Bigr)=-\frac{\sqrt{15}}{5},
\qquad
\delta_q\sim \frac{4}{5}\,(1-\eta_q)\; }.
\]
\end{corollary}
\begin{proof}
From
\(
\frac{1+a}{1-a}=\frac{(1+a)(1-\overline a)}{|1-a|^2}
\)
we obtain
\(
\delta=\Re(w)=\frac{1-|a|^2}{|1-a|^2}
\)
and
\(
\gamma=\Im(w)=\frac{2\Im(a)}{|1-a|^2}
\);
substituting \(a=\eta e^{\mathrm{i}\theta}\) and writing \(|1-a|^2=1-2\eta\cos\theta+\eta^2\) gives the two explicit formulas.
\par
If \(\eta_q\to 1\) and \(\theta_q\to\theta^\ast\), then \(a_q\to e^{\mathrm{i}\theta^\ast}\), so
\(
w_q\to\frac{1+e^{\mathrm{i}\theta^\ast}}{1-e^{\mathrm{i}\theta^\ast}}
=\mathrm{i}\cot(\theta^\ast/2)
\), i.e., \(w_q\to \mathrm{i}\gamma^\ast\).
From \(\cos\theta^\ast=-\tfrac14\) and \(\sin\theta^\ast=-\tfrac{\sqrt{15}}{4}\) we get
\[
\gamma^\ast=\cot\!\Bigl(\frac{\theta^\ast}{2}\Bigr)=\frac{1+\cos\theta^\ast}{\sin\theta^\ast}
=\frac{3/4}{-\sqrt{15}/4}=-\frac{\sqrt{15}}{5}.
\]
Finally, \(\delta=\frac{1-\eta^2}{1-2\eta\cos\theta+\eta^2}\) as \((\eta,\theta)\to(1,\theta^\ast)\) satisfies
\[
\delta\sim \frac{(1-\eta)(1+\eta)}{2-2\cos\theta^\ast}
\sim \frac{2(1-\eta)}{2(1-\cos\theta^\ast)}
=\frac{1-\eta}{1-\cos\theta^\ast}
=\frac{4}{5}(1-\eta),
\]
where we used \(1-\cos\theta^\ast=1+\tfrac14=\tfrac54\). This completes the proof.
\end{proof}

\begin{corollary}[Geometric Essence of the Resonance Window: Narrowing Orbit at Fixed Peak Position—"Energy Nearly Stationary, Width Rapidly Collapsing" (Reproducible)]\label{cor:pom-resonance-cayley-narrowing-orbit}
For the resonance window \(q=9,\dots,17\), take the two-step modulus and phase from Table \ref{tab:fold_collision_shadow_spectral_packet_q9_17}
\(
\eta_q=|\mu_q|/r_{q-2}
\)、\(\theta_q=\arg(\mu_q)
\), and define as in Corollary \ref{cor:pom-resonance-cayley-breitwigner-centerwidth}
\[
w_q=\frac{1+a_q}{1-a_q}=\delta_q+\mathrm{i}\gamma_q,
\qquad a_q:=\eta_q e^{\mathrm{i}\theta_q}.
\]
Then by direct substitution of the data from the table we obtain:
\begin{enumerate}
  \item \textbf{Peak position nearly fixed:} \(\gamma_q\) is compressed within an interval of length
  \[
  \max_{9\le q\le 17}\gamma_q-\min_{9\le q\le 17}\gamma_q\ <\ 1.1\times 10^{-2}
  \]
  (numerically \(\gamma_q\in[-0.7711,-0.7608]\)).
  \item \textbf{Half-width monotonically narrowing:} \(\delta_q\) decreases monotonically with \(q\), reaching at the window's end
  \[
  \delta_{17}\approx 3.16\times 10^{-3}=O(10^{-3}).
  \]
\end{enumerate}
Consequently, the shadow spectral packet forms a narrowing orbit approaching the boundary in the Cayley half-plane: its peak position (center) remains nearly stationary, while the half-width rapidly closes according to \(1-\eta_q\).
\end{corollary}
\begin{proof}
By the explicit formulas of Corollary \ref{cor:pom-resonance-cayley-breitwigner-centerwidth}
\[
\delta_q=\frac{1-\eta_q^{2}}{1-2\eta_q\cos\theta_q+\eta_q^{2}},
\qquad
\gamma_q=\frac{2\eta_q\sin\theta_q}{1-2\eta_q\cos\theta_q+\eta_q^{2}},
\]
substituting the values of \((\eta_q,\theta_q)\) (rounded readings) from Table \ref{tab:fold_collision_shadow_spectral_packet_q9_17} term by term yields the stated interval and monotonicity conclusions. This completes the proof.
\end{proof}
