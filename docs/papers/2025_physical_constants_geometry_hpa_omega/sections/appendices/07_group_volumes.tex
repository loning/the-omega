\section{Canonical volumes of the phase spaces}
\label{app:group_volumes}

We record the elementary volume computations used in Theorem~\ref{thm:alpha_three_channel}.

\begin{lemma}[Volumes of $U(1)$, $SU(2)$, $SO(3)$, and projective quotients]
\label{lem:group_volumes}
With the standard (unit-radius) normalizations,
\begin{equation}
\mathrm{Vol}(U(1))=2\pi,\qquad \mathrm{Vol}(SU(2))=\mathrm{Vol}(S^3)=2\pi^2.
\end{equation}
Moreover, the $\mathbb{Z}_2$ quotients satisfy
\begin{equation}
\mathrm{Vol}(SO(3))=\mathrm{Vol}(SU(2)/\{\pm 1\})=\pi^2,\qquad
\mathrm{Vol}(\mathbb{R}P^1)=\mathrm{Vol}(U(1)/\{\pm 1\})=\pi.
\end{equation}
Consequently,
\begin{equation}
\mathrm{Vol}(U(1)\times SU(2))=4\pi^3.
\end{equation}
\end{lemma}

\begin{proof}
The first identity is the circumference of the unit circle.

For $SU(2)\cong S^3$, use hyperspherical coordinates on $S^3$ with volume element
\begin{equation}
\diff V=\sin^2\chi\,\sin\theta\,\diff\chi\,\diff\theta\,\diff\phi,
\qquad
\chi\in[0,\pi],\ \theta\in[0,\pi],\ \phi\in[0,2\pi].
\end{equation}
Then
\begin{equation}
\mathrm{Vol}(S^3)=\int_0^{\pi}\!\sin^2\chi\,\diff\chi\int_0^{\pi}\!\sin\theta\,\diff\theta\int_0^{2\pi}\!\diff\phi
=\left(\frac{\pi}{2}\right)\cdot 2\cdot 2\pi = 2\pi^2.
\end{equation}

The quotient maps $SU(2)\to SU(2)/\{\pm 1\}\cong SO(3)$ and $U(1)\to U(1)/\{\pm 1\}\cong \mathbb{R}P^1$ are two-sheeted Riemannian coverings under the induced quotient metrics. Let $\pi:M\to M/\{\pm 1\}$ denote either map and $\diff V,\diff \bar V$ the corresponding volume forms. Since $\pi$ is a local isometry one has $\pi^\ast(\diff \bar V)=\diff V$, and since the covering has degree $2$,
\[
\mathrm{Vol}(M)=\int_M \diff V=\int_M \pi^\ast(\diff \bar V)=2\int_{M/\{\pm 1\}}\diff \bar V=2\,\mathrm{Vol}(M/\{\pm 1\}),
\]
so the quotient volumes are halved.

Finally, $\mathrm{Vol}(U(1)\times SU(2))=\mathrm{Vol}(U(1))\,\mathrm{Vol}(SU(2))=(2\pi)(2\pi^2)=4\pi^3$.
\end{proof}
