\section{Functorial refinement under window uplift}
\label{app:functorial_refinement}

This appendix records a minimal, auditable notion of functorial refinement of the field-level labeling map under window uplift $m\mapsto m'$.
\paragraph{Audit note (tables as deterministic checks).}
All tabulated refinement multiplicities and refinement-index catalogs below are theorem-level consequences of the prefix-projection definition together with Zeckendorf admissibility.
Scripts referenced in captions reproduce these deterministic counts and write \LaTeX{} fragments; they do not introduce additional free choices beyond the stated definitions.

\subsection{Prefix projection and a functorial lift}

For $m\ge 6$, define the truncation (prefix) projection
\[
\pi_{m\to 6}:X_m\to X_6,\qquad
\pi_{m\to 6}(w_1\cdots w_m):=w_1\cdots w_6.
\]

\begin{lemma}[Well-definedness and surjectivity of $\pi_{m\to 6}$]\label{lem:pi_m6_surj}
For every $m\ge 6$, the map $\pi_{m\to 6}$ is well-defined and surjective.
\end{lemma}
\begin{proof}
If $w\in X_m$ has no adjacent ones, then any prefix has no adjacent ones, hence $\pi_{m\to 6}(w)\in X_6$ and the map is well-defined.
For surjectivity, given any $u\in X_6$, the word $u0\cdots 0\in X_m$ (padding with $m-6$ zeros) satisfies $\pi_{m\to 6}(u0\cdots 0)=u$.
\end{proof}

Given the closed labeling map $\mathcal{L}_{\mathrm{SM}}:X_6\to \mathcal{F}_{\mathrm{SM}}\sqcup\mathcal{G}_{\mathrm{SM}}$ (Section~\ref{sec:sm_labeling_closure}),
we define its functorial lift to window length $m$ by composition:
\[
\mathcal{L}_m:=\mathcal{L}_{\mathrm{SM}}\circ \pi_{m\to 6}:X_m\to \mathcal{F}_{\mathrm{SM}}\sqcup\mathcal{G}_{\mathrm{SM}}.
\]
This definition makes the refinement under successive truncations explicit:
for $m\ge k\ge 6$, one has $\pi_{m\to 6}=\pi_{k\to 6}\circ \pi_{m\to k}$, hence $\mathcal{L}_m=\mathcal{L}_k\circ \pi_{m\to k}$.

\subsection{Deterministic refinement multiplicities}

Although $\mathcal{L}_m$ uses only the first six digits to assign an SM label, each base type in $X_6$ admits multiple higher-window extensions in $X_m$.
This yields a computable refinement multiplicity per base label.

\begin{proposition}[Extension counts depend only on the last bit]\label{prop:prefix_extension_count}
Fix $m\ge 6$ and $u=u_1\cdots u_6\in X_6$.
Let
\[
\mathrm{Ext}_m(u):=\{w\in X_m:\ \pi_{m\to 6}(w)=u\}.
\]
Then $|\mathrm{Ext}_m(u)|$ depends only on $u_6$:
\[
|\mathrm{Ext}_m(u)|=
\begin{cases}
F_{m-4}, & u_6=0,\\
F_{m-5}, & u_6=1,
\end{cases}
\]
where $(F_k)$ are Fibonacci numbers with $F_1=F_2=1$.
\end{proposition}
\begin{proof}
Let $L:=m-6$ be the extension length.
If $u_6=1$, the next bit must be $0$, and the remaining $L-1$ bits form an admissible word in $X_{L-1}$, so $|\mathrm{Ext}_m(u)|=|X_{L-1}|=F_{(L-1)+2}=F_{m-5}$ by Lemma~\ref{lem:xm_fib}.
If $u_6=0$, there is no forced initial $0$ and the $L$ extension bits form an admissible word in $X_L$, so $|\mathrm{Ext}_m(u)|=|X_L|=F_{L+2}=F_{m-4}$.
\end{proof}

\begin{remark}[Matrix viewpoint via the golden-mean transition graph]\label{rem:extension_counts_matrix_view}
\MathTag Let
\[
A=\begin{pmatrix}1&1\\ 1&0\end{pmatrix}
\]
be the golden-mean transition matrix (Lemma~\ref{lem:golden_zeta_matrix}).
Let $\mathbf{1}=(1,1)^{\mathsf{T}}$ and let $e_0=(1,0)^{\mathsf{T}}$, $e_1=(0,1)^{\mathsf{T}}$.
Writing $L=m-6$ for the extension length, one has
\[
|\mathrm{Ext}_m(u)|=e_{u_6}^{\mathsf{T}}\,A^{L}\,\mathbf{1}.
\]
Equivalently, lift multiplicities are row-sums of powers of the same $2\times2$ matrix that controls the $\pi$/$\e$-channel periodic-point and zeta data (Section~\ref{subsec:e_channel}).
\end{remark}

\subsection{A trace recurrence (Lucas-type) and a Hecke-like comparison}
\label{subsec:trace_recurrence_hecke_comparison}

\begin{lemma}[Trace recurrence for $2\times 2$ matrices]\label{lem:trace_recurrence_2x2}
\MathTag Let $M\in M_2(\CC)$ and set $t_n:=\mathrm{tr}(M^n)$ for $n\ge 0$ (with $M^0=I$).
Then $t_0=2$, $t_1=\mathrm{tr}(M)$, and for all $n\ge 1$ one has the second-order recurrence
\[
t_{n+1}=\mathrm{tr}(M)\,t_n-\det(M)\,t_{n-1}.
\]
\end{lemma}
\begin{proof}
The Cayley--Hamilton identity for $2\times 2$ matrices is
\[
M^2-(\mathrm{tr}M)\,M+(\det M)\,I=0.
\]
Multiplying by $M^{n-1}$ gives $M^{n+1}=(\mathrm{tr}M)M^n-(\det M)M^{n-1}$ for $n\ge 1$.
Taking traces yields the stated recurrence.
\end{proof}

\begin{corollary}[Lucas recurrence for the golden-mean matrix]\label{cor:lucas_trace_golden_matrix}
\MathTag For $A=\bigl(\begin{smallmatrix}1&1\\1&0\end{smallmatrix}\bigr)$ one has $\mathrm{tr}(A)=1$ and $\det(A)=-1$, hence
\[
\mathrm{tr}(A^{n+1})=\mathrm{tr}(A^n)+\mathrm{tr}(A^{n-1})\qquad(n\ge 1),
\]
with $\mathrm{tr}(A^0)=2$ and $\mathrm{tr}(A^1)=1$.
Therefore $\mathrm{tr}(A^n)=L_n$ (Lucas numbers), consistent with the $\pi$-channel periodic-point counts (Lemma~\ref{lem:cyc_periodic_points}).
\end{corollary}

\begin{remark}[Hecke prime-power recurrence as a structural analogue]\label{rem:hecke_prime_power_structural_analogue}
\AuditTag The Hecke prime-power relation
\[
T_{p^{r+1}} = T_p T_{p^r} - p^{k-1} T_{p^{r-1}}
\]
(Equation~\eqref{eq:hecke_prime_power_z128}) has the same second-order skeleton as Lemma~\ref{lem:trace_recurrence_2x2}.
This paper uses the resemblance only as a \emph{structural template}: a small set of generators together with a rigid recurrence propagates constraints across scales.
No identification between Hecke operators and the protocol lift matrices is asserted.
\end{remark}

\subsection{A canonical suffix index for lift refinement}
\label{subsec:lift_suffix_index}

The lift multiplicities in Table~\ref{tab:label_lift_counts} depend only on the adjacency constraint between the prefix last bit $u_6$ and the first extension bit.
To obtain a concrete refinement of the coarse lift $\mathcal{L}_m$ (beyond counting multiplicities), it is useful to attach a deterministic index to each lift in $\mathrm{Ext}_m(u)$.

\begin{definition}[Free suffix index]\label{def:free_suffix_index}
Fix $m\ge 6$ and a base type $u\in X_6$.
Write any lift $w\in \mathrm{Ext}_m(u)$ as $w=u\,s$, where $s$ is the length-$(m-6)$ suffix.
If $u_6=0$, define the free suffix to be $t:=s$.
If $u_6=1$, admissibility forces $s$ to begin with $0$, and we define the free suffix to be $t:=s_2\cdots s_{m-6}$ (drop the forced leading $0$).
Define the free suffix index
\[
\rho(w):=V_Z(t),
\]
where $V_Z$ is the Zeckendorf/Fibonacci value computed from the binary word $t$ using the standard Fibonacci weights (starting at $F_2=1$).
\end{definition}

At fixed $(m,u_6)$, the possible free suffixes form an admissible set and the index $\rho$ enumerates them deterministically.
