\section{$(m=6)$ 的 $(64\to 21)$ 纤维化：空间态与时间纤维}

\subsection{微态空间与数字轨道}

考虑区间 $\{0,\dots,63\}$。
由于 $F_{11}=89>63$，任意 $n\le 63$ 的 Zeckendorf 正规形仅需使用 $F_2,\dots,F_{10}$。
为便于局域重写实现，我们使用长度 $12$ 的 digit 带
$$
d_{12}d_{11}\cdots d_2d_1,\qquad \val(d)=\sum_{k=1}^{12} d_k F_k,
$$
并约定正规形满足 $d_1=d_{11}=d_{12}=0$、$d_k\in\{0,1\}$、且 $d_k d_{k+1}=0$。

\begin{definition}[微态闭合集]
定义
$$
\M_6:=\{d:\ \val(d)\in\{0,\dots,63\}\ \text{且 } d \text{为 Zeckendorf 正规形}\}.
$$
\end{definition}

显然 $\abs{\M_6}=64$，并可将 $\M_6$ 与集合 $\{0,\dots,63\}$ 以及循环群 $\ZZ_{64}$ 逐点对应。

\subsection{空间态 $\X_6$（低位投影）}

取低 $6$ 位（对应 $F_7,F_6,F_5,F_4,F_3,F_2$）定义
$$
w=(d_7,d_6,d_5,d_4,d_3,d_2)\in\{0,1\}^6,\qquad d_i d_{i+1}=0,
$$
并令
$$
\X_6:=\{w\in\{0,1\}^6:\ w_i w_{i+1}=0\}.
$$
经典计数给出 $\abs{\X_6}=F_8=21$（长度为 $m$ 的无相邻 $1$ 二进制串数为 $F_{m+2}$）\cite{WikipediaZeckendorf}.

\begin{definition}[空间值]
定义
$$
V(w):=\sum_{k=2}^{7} d_k F_k=13d_7+8d_6+5d_5+3d_4+2d_3+d_2.
$$
\end{definition}
由 $w\in\X_6$ 可知 $V(w)\in\{0,\dots,20\}$。

\subsection{时间纤维 $u$（高位残余）}

取高位残余
$$
u:=d_8F_8+d_9F_9+d_{10}F_{10}\in\{0,21,34,55\}.
$$
于是每个微态 $n\in\M_6$ 都可写为
$$
n=V(w)+u,
$$
并定义投影
$$
\pi:\M_6\to \X_6,\qquad \pi(n)=w.
$$

\subsection{纤维大小分类定理}

\begin{theorem}[纤维分类]
对任意 $w\in\X_6$，定义纤维集合
$$
U(w):=\{u\in\{0,21,34,55\}:\ V(w)+u\le 63\ \text{且与 Zeckendorf 相邻约束相容}\}.
$$
则 $U(w)$ 只可能是下列三类之一：
$$
U(w)=
\begin{cases}
\{0,21,34,55\}, & V(w)\le 8,\\
\{0,21,34\}, & 9\le V(w)\le 12,\\
\{0,34\}, & 13\le V(w)\le 20.
\end{cases}
$$
因此纤维大小仅为 $4,3,2$ 三类。
\end{theorem}

\begin{proof}
由相邻约束 $d_7d_8=0$，若 $d_7=1$ 则必有 $d_8=0$，因此 $u$ 不可能取 $21$。
又因为 $d_9d_{10}=0$，在 $d_8=0$ 的前提下高位仅可能为
$$
u\in\{0,34,55\}.
$$
当 $d_7=1$ 时 $V(w)\ge F_7=13$，并且为了满足 $V(w)+55\le 63$ 必须有 $V(w)\le 8$，这与 $V(w)\ge 13$ 矛盾，
故此时 $u$ 只能取 $\{0,34\}$，得到第三类区间 $13\le V(w)\le 20$。

当 $d_7=0$ 时，相邻约束不排斥 $d_8=1$，因此 $u$ 至少可取 $\{0,21,34\}$。
进一步，$u=55$ 要求 $d_{10}=1$ 且 $d_9=0$，并且由于 $F_{10}=55$，要满足 $V(w)+55\le 63$ 等价于 $V(w)\le 8$。
因此当且仅当 $V(w)\le 8$ 时 $u$ 取满 $\{0,21,34,55\}$；当 $9\le V(w)\le 12$ 时只能取 $\{0,21,34\}$。
综上得到分类。
\end{proof}
