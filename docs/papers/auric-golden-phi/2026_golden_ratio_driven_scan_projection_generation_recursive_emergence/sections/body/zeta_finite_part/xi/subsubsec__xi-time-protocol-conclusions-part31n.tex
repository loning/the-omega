\paragraph{二字母自由半群字迹重写与隧穿主导律}
本段将结论 \ref{con:xi-terminal-replica-softcore-trace-moment-cycle-subgraph-independent-sets}、
\ref{con:xi-terminal-replica-softcore-exceptional-power-sum-cycle-sector-replacement} 与
\ref{con:xi-terminal-replica-softcore-large-q-two-layer-asymptotics}
改写为二字母自由半群字迹口径，以便把 \(q\)-方向指数底数与 Fibonacci 乘积结构直接对应。

\begin{conclusion}[时间纤维二字母提升：\texorpdfstring{$\Tr((T^{(q)})^m)$}{Tr((T^(q))^m)} 的自由半群字迹分解]\label{con:xi-terminal-replica-softcore-trace-free-semigroup-lift}
固定 \(q\ge 2\)、\(m\ge 1\)。令
\[
K:=\begin{pmatrix}1&1\\[2pt]1&0\end{pmatrix},
\qquad
P:=\begin{pmatrix}1&1\\[2pt]1&1\end{pmatrix},
\qquad
D_q:=K^{\otimes q},
\qquad
J_q:=P^{\otimes q}.
\]
则
\[
T^{(q)}=\frac12\bigl(J_q+D_q\bigr),
\]
并且对任意 \(m\ge 1\) 有严格恒等式
\[
\boxed{\;
\Tr\!\bigl((T^{(q)})^m\bigr)
=\frac1{2^m}\sum_{w\in\{P,K\}^m}\Bigl(\Tr(M(w))\Bigr)^q\; },
\]
其中 \(w=(w_1,\dots,w_m)\)，\(M(w):=w_1\cdots w_m\in\ZZ^{2\times2}\)。
特别地，
\[
\Tr\!\bigl((T^{(q)})^m\bigr)\in 2^{-m}\ZZ,
\]
且其 \(q\)-方向仅由有限集合 \(\{\Tr(M(w)):w\in\{P,K\}^m\}\subset\ZZ\) 的 \(q\) 次幂线性组合决定。
\end{conclusion}
\begin{proof}
由 \(P=\mathbf 1\mathbf 1^{\mathsf T}\) 得 \(P^{\otimes q}=J_q\)，并由定义 \(D_q=K^{\otimes q}\)。
故 \(T^{(q)}=\frac12(P^{\otimes q}+K^{\otimes q})\)。
展开 \((P^{\otimes q}+K^{\otimes q})^m\) 得
\[
\sum_{w\in\{P,K\}^m}(w_1^{\otimes q})\cdots(w_m^{\otimes q})
=\sum_{w\in\{P,K\}^m}(M(w))^{\otimes q},
\]
其中用了 \((A^{\otimes q})(B^{\otimes q})=(AB)^{\otimes q}\)。
取迹并用 \(\Tr(X^{\otimes q})=(\Tr X)^q\) 即得结论。
\end{proof}

\begin{conclusion}[时间纤维分裂的 Fibonacci 因子化]\label{con:xi-terminal-replica-softcore-word-trace-fibonacci-factorization}
沿结论 \ref{con:xi-terminal-replica-softcore-trace-free-semigroup-lift} 的记号，设 \(w\in\{P,K\}^m\) 且含至少一个 \(P\)。
按循环同构可写为
\[
w\sim P\,K^{\ell_1}\,P\,K^{\ell_2}\cdots P\,K^{\ell_t},
\qquad
t\ge 1,\ \ell_i\in\ZZ_{\ge 0},\ \sum_{i=1}^t\ell_i=m-t.
\]
则
\[
\boxed{\;
\Tr(M(w))=\prod_{i=1}^{t}F_{\ell_i+3}\in\ZZ_{>0}\; }.
\]
此外唯一不含 \(P\) 的字词 \(K^m\) 满足 \(\Tr(K^m)=L_m\)。
故
\[
\{\Tr(M(w)):w\in\{P,K\}^m\}
=\{L_m\}\cup
\left\{\prod_{i=1}^{t}F_{\ell_i+3}:\ t\ge1,\ \ell_i\ge0,\ \sum_i\ell_i=m-t\right\}.
\]
\end{conclusion}
\begin{proof}
令 \(e:=(1,1)^{\mathsf T}\)，则 \(P=ee^{\mathsf T}\) 为秩一矩阵。
对 \(\ell\ge0\)，有 \(e^{\mathsf T}K^\ell e=F_{\ell+3}\)。
将 \(w\) 写成所示循环代表后，
\(
PK^{\ell_i}=e(e^{\mathsf T}K^{\ell_i})
\)
仍为秩一。
用恒等式
\[
\Tr(u_1v_1^{\mathsf T}\cdots u_tv_t^{\mathsf T})
=\prod_{i=1}^{t}(v_i^{\mathsf T}u_{i+1}),
\]
（指标按 \(t\) 循环）即得
\[
\Tr(M(w))
=\prod_{i=1}^{t}(e^{\mathsf T}K^{\ell_i}e)
=\prod_{i=1}^{t}F_{\ell_i+3}.
\]
而 \(w=K^m\) 时 \(\Tr(K^m)=L_m\) 由 Fibonacci--Lucas 矩阵表示给出。
\end{proof}

\begin{conclusion}[例外幂和的全阶字词闭式与递推闭包]\label{con:xi-terminal-replica-softcore-exceptional-power-sum-word-closed-form}
令
\[
S_m(q):=\sum_{i=0}^{q}(\lambda_i^{(q)})^m,
\]
并沿用 \(\Omega_m(q)\)（结论 \ref{con:xi-terminal-replica-softcore-lucas-pell-omega-unified}）。
则对任意 \(q\ge2,\ m\ge1\) 有
\[
\boxed{\;
S_m(q)=\frac1{2^m}\left(
\Omega_m(q)+
\sum_{\substack{w\in\{P,K\}^m\\ w\neq K^m}}
\bigl(\Tr(M(w))\bigr)^q
\right)\; }.
\]
并且：
\begin{enumerate}
  \item \(2^mS_m(q)\in\ZZ\)；
  \item 记
  \[
  \mathcal T_m:=\{\Tr(M(w)):w\in\{P,K\}^m,\ w\neq K^m\}\subset\ZZ_{>0},
  \]
  则 \(q\mapsto S_m(q)\) 属于
  \[
  \mathrm{span}_{\QQ}\{\alpha^q:\alpha\in\mathcal T_m\}
  \ \oplus\
  \mathrm{span}_{\QQ}\{\Omega_m(q)\},
  \]
  从而满足首一整数系数线性递推；其湮灭多项式可取
  \[
  \boxed{\;
  \Bigl(x^2-L_mx+(-1)^m\Bigr)\prod_{\alpha\in\mathcal T_m}(x-\alpha)\; }.
  \]
\end{enumerate}
\end{conclusion}
\begin{proof}
由结论 \ref{con:xi-terminal-replica-softcore-trace-free-semigroup-lift}，
\[
\Tr\!\bigl((T^{(q)})^m\bigr)
=2^{-m}\sum_{w\in\{P,K\}^m}\bigl(\Tr(M(w))\bigr)^q.
\]
而 \(w=K^m\) 时 \(\Tr(M(w))=L_m\)。
再由结论 \ref{con:xi-terminal-replica-softcore-exceptional-power-sum-cycle-sector-replacement}，
\[
\Tr\!\bigl((T^{(q)})^m\bigr)-S_m(q)=2^{-m}\bigl(L_m^q-\Omega_m(q)\bigr),
\]
两式相减即得闭式。
整性由右端整数项直接推出。
\(\Omega_m(q)\) 满足二阶递推
\(
\Omega_m(q)=L_m\Omega_m(q-1)-(-1)^m\Omega_m(q-2)
\)，
每个 \(\alpha^q\) 满足一阶递推 \(x-\alpha\)，故乘积多项式湮灭和序列。
\end{proof}

\begin{corollary}[二次幂和的完全闭式]\label{cor:xi-terminal-replica-softcore-exceptional-power-sum-m2-closed-form}
对任意 \(q\ge2\)，
\[
\boxed{\;
S_2(q)=\frac14\Bigl(4^q+2\cdot3^q+F_{2q+2}\Bigr)\; }.
\]
\end{corollary}
\begin{proof}
由结论 \ref{con:xi-terminal-replica-softcore-exceptional-power-sum-word-closed-form} 取 \(m=2\)：
\[
S_2(q)=\frac14\Bigl(\Omega_2(q)+
\sum_{\substack{w\in\{P,K\}^2\\ w\neq K^2}}\bigl(\Tr(M(w))\bigr)^q\Bigr).
\]
长度 \(2\) 的非 \(K^2\) 字词为 \(PP,PK,KP\)，且
\[
\Tr(PP)=4,\qquad \Tr(PK)=\Tr(KP)=3.
\]
又 \(\Omega_2(q)=F_{2q+2}\)，代入即得。
\end{proof}

\begin{conclusion}[六次幂和的全指数展开]\label{con:xi-terminal-replica-softcore-exceptional-power-sum-m6-word-expanded}
对任意 \(q\ge2\)，
\[
\boxed{\;
\begin{aligned}
S_6(q)=\frac1{64}\Big(
&64^{q}
+6\cdot 48^{q}
+6\cdot 40^{q}
+9\cdot 36^{q}
+6\cdot 32^{q}
+12\cdot 30^{q}\\
&+2\cdot 27^{q}
+6\cdot 26^{q}
+3\cdot 25^{q}
+6\cdot 24^{q}
+6\cdot 21^{q}
+\Omega_6(q)
\Big)\; .
\end{aligned}
}
\]
等价地，
\[
\Tr\!\bigl((T^{(q)})^6\bigr)
=S_6(q)+\frac1{64}\bigl(L_6^q-\Omega_6(q)\bigr)
=\frac1{64}\bigl(\cdots+18^q\bigr),
\]
其中 \(L_6=18\)。
\end{conclusion}
\begin{proof}
该式与结论 \ref{con:xi-terminal-replica-softcore-exceptional-power-sum-m6} 一致。
亦可由结论 \ref{con:xi-terminal-replica-softcore-word-trace-fibonacci-factorization}
对长度 \(6\) 字词按 \(K\)-段型分组计数后，代入
结论 \ref{con:xi-terminal-replica-softcore-exceptional-power-sum-word-closed-form} 得到。
\end{proof}

\begin{conclusion}[隧穿通道主导律：唯一极大指数与次极大指数]\label{con:xi-terminal-replica-softcore-tunneling-channel-dominance}
固定 \(m\ge3\)，定义
\[
\Theta_m:=\max\{\Tr(M(w)):w\in\{P,K\}^m,\ w\neq K^m\}.
\]
则 \(\Theta_m=2^m\)，且仅由 \(w=P^m\) 取得。
并且对所有 \(w\neq P^m\) 有
\[
\Tr(M(w))\le 3\cdot2^{m-2},
\]
等号当且仅当 \(w\) 含恰一枚 \(K\)（重数为 \(m\)）。
因此当 \(q\to\infty\) 时，
\[
\boxed{\;
S_m(q)
=\frac1{2^m}\Big((2^m)^q+m(3\cdot2^{m-2})^q\Big)
+O_m\!\big((5\cdot2^{m-3})^q\big)\; }.
\]
\end{conclusion}
\begin{proof}
由结论 \ref{con:xi-terminal-replica-softcore-word-trace-fibonacci-factorization}，
任一含 \(P\) 字词可写为
\(
\prod_i F_{\ell_i+3}
\)，且 \(\sum_i(\ell_i+1)=m\)。
由 \(F_{\ell+3}\le 2^{\ell+1}\)（\(\ell\ge0\)，且仅 \(\ell=0\) 取等）得
\[
\Tr(M(w))\le 2^m,
\]
且等号当且仅当全部 \(\ell_i=0\)，即 \(w=P^m\)。
若 \(w\neq P^m\)，至少有一段 \(\ell_j\ge1\)。
在固定 \(\sum_i\ell_i\) 下，由
\[
F_{a+3}F_{b+3}=2F_{a+b+3}-F_aF_b<2F_{a+b+3}\qquad(a,b\ge1)
\]
可知把正段合并会严格增大乘积，故最优形态为唯一正段。
若正段长度为 \(S\ge1\)，对应值为
\[
F_{S+3}\,2^{m-S-1}.
\]
且 \(F_{S+3}/2^{S-1}\) 随 \(S\ge1\) 严格递减，故次极大在 \(S=1\) 取得：
\(
3\cdot2^{m-2}
\)，重数即单个 \(K\) 的位置数 \(m\)。
下一层由 \(S=2\) 给出底数 \(5\cdot2^{m-3}\)。
此外 \(\Omega_m(q)\) 满足二阶 Lucas--Pell 递推，故 \(\Omega_m(q)=O(\varphi^{mq})\)；当 \(m\ge3\) 时
\[
\varphi^{m}\le 5\cdot2^{m-3},
\]
因此 \(\Omega_m\)-通道被同一误差底数吸收。
把该分层代入结论 \ref{con:xi-terminal-replica-softcore-exceptional-power-sum-word-closed-form} 即得渐近式。
\end{proof}
