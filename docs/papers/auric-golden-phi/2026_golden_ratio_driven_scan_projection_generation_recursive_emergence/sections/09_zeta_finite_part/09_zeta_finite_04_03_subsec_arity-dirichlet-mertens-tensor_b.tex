\begin{remark}[\texorpdfstring{$\ZZ/5$}{Z/5} 轴的傅里叶压缩特征量（精确恒等分解）]\label{prop:arity-335-fourier-fingerprint}

\widehat f_{a,b}(j):=\frac15\sum_{c=0}^{4} f_{a,b}(c)\,\omega_5^{-jc},
\qquad j\in\{0,1,2,3,4\}.
\]
则对所有 $c$ 有恒等分解
\[
f_{a,b}(c)
=\widehat f_{a,b}(0)
+2\Re\bigl(\widehat f_{a,b}(1)\,\omega_5^{c}\bigr)
+2\Re\bigl(\widehat f_{a,b}(2)\,\omega_5^{2c}\bigr),
\]

\end{remark}

\begin{remark}[\texorpdfstring{$c$}{c} 轴强度的精确标尺：全局 \texorpdfstring{$c$}{c}-方差的 Parseval 恒等式]\label{thm:arity-335-c-variance-identity}

V_c

\ }.

\begin{remark}[单位圆扭曲的偶性与无三次项]\label{lem:unit-circle-evenness}
设 $M_\theta$ 为一个实矩阵族，且其条目关于 $e^{\mathrm{i}\theta}$ 实解析，并满足
\[
M_{-\theta}=\overline{M_\theta}\qquad(\theta\in\RR).
\]
记 $\rho(\theta):=\rho(M_\theta)$ 为谱半径，并设在 $\theta=0$ 附近存在简单的主特征值分支 $\Lambda(\theta)$（$\Lambda(0)>0$）。
则 \(\rho(\theta)=\rho(-\theta)\) 且 \(\rho\) 在 \(0\) 的邻域为实解析偶函数；因此 \(\log(\rho(\theta)/\rho(0))\) 的 Taylor 展开只含偶次项，特别地
\[
\log\frac{\rho(\theta)}{\rho(0)}=-c_2\theta^2+c_4\theta^4+O(\theta^6),
\qquad \theta\to 0.
\]
\end{remark}

\begin{remark}[小角扭曲的二次律：$\kappa$ 的方差密度刻画]\label{prop:arity-335-kappa-variance}
取碰撞 cocycle \(\xi=\mathbf{1}_{\{d=2\}}\)（沿边取值 \(0/1\)），并令三轴扭曲中第三轴取 \(N_2(\gamma)\bmod p\)。
对 \(\theta\in\RR\) 定义纯碰撞扭曲矩阵
$$
M_\theta:=M(1,1,e^{\mathrm{i}\theta})\qquad(\text{仅对 }\xi\text{ 施加相位}).
$$
记 \(\Lambda(\theta)\) 为 \(M_\theta\) 的主特征值分支（\(\Lambda(0)=\lambda\)）。
由于本系统为有限维原始非负矩阵的解析扰动，\(\Lambda(\theta)\) 在 \(\theta=0\) 的邻域解析（参见一般的有限维谱扰动理论）。
则存在常数 \(\mu_\xi\in\RR\) 与 \(\sigma_\xi^2\ge 0\) 使得
$$
\log\frac{\Lambda(\theta)}{\lambda}
=
\mathrm{i}\mu_\xi\,\theta-\frac{\sigma_\xi^2}{2}\theta^2+O(\theta^3),
\qquad \theta\to 0,
$$
从而
$$
\log\frac{|\Lambda(\theta)|}{\lambda}
=
-\frac{\sigma_\xi^2}{2}\theta^2+O(\theta^4).
$$
特别地，当 \(\theta=2\pi/p\) 时（\(p\) 为奇素数）有
$$
1-\frac{\rho_{3,3,p}}{\lambda}
=
\frac{\sigma_\xi^2}{2}\left(\frac{2\pi}{p}\right)^2
+
O\!\left(p^{-4}\right),
$$
其中 \(\rho_{3,3,p}\) 为 \( ((3,3,p))\) 的最坏扭曲谱半径（且在可复现输出中由纯碰撞扭曲达到）。
因此二次律常数的本体为
$$
\kappa_\infty:=\lim_{p\to\infty}\frac{1-\rho_{3,3,p}/\lambda}{(2\pi/p)^2}
=\frac{\sigma_\xi^2}{2}.
$$
并且 \(\sigma_\xi^2\) 可由实参倾斜压力
\(
P(t):=\log\rho(M(1,1,e^t))
\)
给出：\(\sigma_\xi^2=P''(0)\)（与命题 \ref{prop:kernel-clt-ldp} 的有限状态 Markov 压力导数口径一致）。
\end{remark}

\begin{remark}[\texorpdfstring{$\kappa_\infty$}{kappa} 的变分/Poisson 方程表达（可计算）]\label{rem:arity-335-kappa-poisson}

同一常数亦可被刻画为一个严格凸的最小能量问题，并因此可由一把线性代数直接求出。

\begin{remark}[near-coboundary \texorpdfstring{$\Rightarrow$}{->} 小偏差：带 \texorpdfstring{$4\sin^2(\pi/p)$}{4 sin^2} 的缩放上界]\label{cor:arity-335-eps-trig-bound}

$$
1-\frac{\rho_{3,3,p}}{\lambda}
=
\frac{\sigma_\xi^2}{2}\left(\frac{2\pi}{p}\right)^2+O(p^{-4})
\ \le\
\frac{\varepsilon_\pi^{(2)}}{2}\left(\frac{2\pi}{p}\right)^2+O(p^{-4}).
$$

\le
\frac{\varepsilon_\pi^{(2)}}{2}\,|1-\omega_p|^2\ +\ O(p^{-4}),
$$
从而在 \(p\to\infty\) 的缩放意义下得到 \(\kappa_\infty\le \varepsilon_\pi^{(2)}/2\)。
\end{remark}

\begin{remark}[小角主曲线与普适塌缩：二次主项 \texorpdfstring{$+$}{+} 四次修正]\label{prop:arity-335-master-curve}

=

\kappa_\infty=\sigma_\xi^2/2.
$$
\end{remark}

\begin{remark}[谐波能量特征量 \texorpdfstring{$\mathcal{E}_1$}{E1}：把“低秩可压缩”表述为可解释结构量]\label{rem:arity-335-harmonic-energy}

\begin{remark}[碰撞 cocycle 的共边界缺陷证书：best \texorpdfstring{$\ZZ/5$}{Z/5} 着色]\label{rem:arity-335-best-5-coloring}
把 40 状态图的每条边赋 cocycle $\xi=\mathbf{1}_{\{d=2\}}\in\ZZ/5$。若存在 $h:V\to\ZZ/5$ 使得 $h(v)-h(u)\equiv \xi(u\to v)\pmod 5$ 在全部边上成立，则第三轴扭曲可由共轭消去，因而不会降低谱半径（$\rho/\lambda=1$）。二级摘要输出给出一个可复现的“best found” 着色缺陷 $\varepsilon$（不一致边比例），作为“near-coboundary” 的组合学证书；其值与谱缺口 $1-\rho_{3,3,5}/\lambda$ 应在机制上相关。
\end{remark}

\begin{remark}[近共边界缺陷的组合测度与谱测度同阶]\label{rem:arity-335-eps-vs-gap}

$$
\delta_{\mathrm{spec}}<\varepsilon.
$$

\begin{conjecture}[共边界缺陷 $\Rightarrow$ 谱隙（定量 GRH 证书）]\label{conj:coboundary-defect-gap}
设 $M$ 为原始非负核，$\chi$ 为单位根角色扭曲，记
$$
\lambda:=\rho(M),\qquad \rho_\chi:=\rho(M_\chi).
$$
令 \(\varepsilon\) 为最小共边界缺陷：在所有赋值 \(f:V\to U(1)\) 中，使
\(
\chi(e)=f(t(e))\overline{f(s(e))}
\)
成立的边比例最大化，其余比例即为 \(\varepsilon\)；并令 \(\varepsilon_\pi\) 为以无扭曲 PF 平衡边测度加权的对应缺陷。则存在只依赖于图的结构常数 \(c>0\) 使
$$
1-\frac{\rho_\chi}{\lambda}\ \ge\ c\,\varepsilon^2
\qquad(\text{或 }c\,\varepsilon_\pi^2),
$$
并且反向成立：若 \(1-\rho_\chi/\lambda\le \delta\)，则存在某个 \(f\) 使共边界缺陷满足 \(\varepsilon\lesssim \sqrt{\delta}\)（或 \(\varepsilon_\pi\lesssim \sqrt{\delta}\)）。因此“近共边界”与“小谱隙”在定量上等价，成为核版 GRH 的可审计证书接口。
\end{conjecture}

\begin{remark}[证明路线图：从能量泛函到离散 Poincar\'e 不等式]\label{rem:coboundary-defect-proof}
把 $M$ 归一化到 PF 平衡测度下的 Markov/transfer 算子，并把扭曲看作边上的相位扰动。由谱半径的变分表述，可将 $1-\rho_\chi/\lambda$ 写为某个“能量下降”泛函的极小值；而共边界缺陷提供一个离散 Poincar\'e/cheeger 型下界，迫使能量至少下降 \(\gtrsim \varepsilon^2\)。反向方向来自“近最大向量”的相位一致性：若谱隙极小，则存在 $g$ 使得相位在大多数边上可由梯度吸收，从而得到 \(\varepsilon\lesssim \sqrt{\delta}\)。
\end{remark}

\begin{remark}[从“最佳着色”得到的可证明上界：\texorpdfstring{$\kappa_\infty\le \varepsilon_\pi^{(2)}/2$}{kappa <= eps/2}]\label{rem:arity-335-eps-bound}
