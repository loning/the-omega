\paragraph{Threshold Intensity and Endpoint Spectral Dimension: Zeroth-Order Atom \texorpdfstring{$\mathfrak m_{-1}$}{m(-1)} and Power-Law Decay}
Decompose the representing measure at the endpoint into atomic and non-atomic parts:
\[
\mu=\mathfrak m_{-1}\delta_{-1}+\mu_{\neq -1},
\qquad
\mathfrak m_{-1}:=\mu(\{-1\})\ge 0,
\qquad
\mu_{\neq -1}:=\mu\!\big|_{\TT\setminus\{-1\}}.
\]

\begin{corollary}[Strict Separation of the Atomic Part]\label{cor:xi-endpoint-atom-separation}
Under the hypotheses of Theorem \ref{thm:xi-endpoint-heat-kernel-probe}, for every \(N\ge 0\) one has
\[
\boxed{\;
a_N
=\mathfrak m_{-1}
\;+\;
\int_{\TT\setminus\{-1\}}\left|\frac{1-\xi}{2}\right|^{2N}\,d\mu_{\neq -1}(\xi)\;}
\]
and \(a_N-\mathfrak m_{-1}\downarrow 0\).
Consequently, \(\mathfrak m_{-1}\) is the unique ``zeroth-order intensity parameter'' carried by the endpoint threshold, and the remaining visible structure at the endpoint is entirely captured by the decay law of the remainder \(a_N-\mathfrak m_{-1}\).
\end{corollary}

\begin{proof}
The first identity follows immediately from the measure decomposition and \eqref{eq:xi-endpoint-heat-kernel-identity}; then \eqref{eq:xi-endpoint-heat-kernel-limit} yields \(a_N\downarrow \mathfrak m_{-1}\), so that the remainder decreases monotonically to \(0\). \qed
\end{proof}

\begin{theorem}[Laplace-Type Classification of the Endpoint Spectral Dimension]\label{thm:xi-endpoint-spectral-dimension-laplace}
Suppose \(\mu_{\neq -1}\) satisfies a local power law in a neighborhood of \(\xi=-1\): there exist \(\alpha>-1\) and constants \(K_\pm\ge 0\) such that as \(\varepsilon\downarrow 0\),
\[
\mu_{\neq -1}\bigl(\{\pi<\arg\xi<\pi+\varepsilon\}\bigr)
=K_+\,\varepsilon^{\alpha+1}+o(\varepsilon^{\alpha+1}),
\qquad
\mu_{\neq -1}\bigl(\{\pi-\varepsilon<\arg\xi<\pi\}\bigr)
=K_-\,\varepsilon^{\alpha+1}+o(\varepsilon^{\alpha+1}).
\]
Then the endpoint probe remainder satisfies the power-law asymptotic
\begin{equation}\label{eq:xi-endpoint-laplace-asymptotic}
\boxed{\;
a_N-\mathfrak m_{-1}
\sim
(\alpha+1)\,2^{\alpha}\,
\Gamma\!\left(\frac{\alpha+1}{2}\right)\,
(K_++K_-)\,N^{-(\alpha+1)/2}\qquad(N\to\infty).}
\end{equation}
In particular, if \(\mu\) is absolutely continuous in a neighborhood of \(\xi=-1\) with angular density \(f(\theta)\) continuous at \(\theta=\pi\) and \(f(\pi)>0\), then \(\alpha=0\), \(K_+=K_-=f(\pi)\), and hence
\[
\boxed{\;
a_N-\mathfrak m_{-1}\sim 2f(\pi)\sqrt{\frac{\pi}{N}}\qquad(N\to\infty).}
\]
\end{theorem}

\begin{proof}
Write \(\xi=e^{\mathrm{i}\theta}=-e^{\mathrm{i}\varphi}\) (\(\theta=\pi+\varphi\), \(|\varphi|\ll 1\)), so that
\[
\left|\frac{1-\xi}{2}\right|
=\left|\sin\frac{\theta}{2}\right|
=\left|\cos\frac{\varphi}{2}\right|
=1-\frac{\varphi^2}{8}+O(\varphi^4),
\]
whence \(\left|\frac{1-\xi}{2}\right|^{2N}=\exp\!\bigl(-\frac{N\varphi^2}{4}+O(N\varphi^4)\bigr)\), with the dominant contribution coming from \(|\varphi|\lesssim N^{-1/2}\).
Decompose \(a_N-\mathfrak m_{-1}=\int_{\TT\setminus\{-1\}}\left|\frac{1-\xi}{2}\right|^{2N}\,d\mu_{\neq-1}(\xi)\) into the \(\varphi>0\) and \(\varphi<0\) sides, and approximate the local increments of \(\mu_{\neq-1}\) by \((\alpha+1)K_\pm|\varphi|^{\alpha}\,d\varphi\) using the hypothesis.
For each side, the substitution \(t=\frac{N}{4}\varphi^2\) gives
\[
\int_{0}^{\infty}e^{-N\varphi^2/4}\,(\alpha+1)K_+\,\varphi^{\alpha}\,d\varphi
=(\alpha+1)2^{\alpha}K_+\,N^{-(\alpha+1)/2}\,
\Gamma\!\left(\frac{\alpha+1}{2}\right),
\]
and the negative side follows analogously with \(K_+\) replaced by \(K_-\). Combining yields \eqref{eq:xi-endpoint-laplace-asymptotic}; the absolutely continuous case is obtained by setting \(\alpha=0\). \qed
\end{proof}

\begin{definition}[Endpoint Spectral Dimension]\label{def:xi-endpoint-spectral-dimension}
Under the power-law regime of Theorem \ref{thm:xi-endpoint-spectral-dimension-laplace}, define the endpoint ``spectral dimension''
\[
\boxed{\;
d_{-1}:=\alpha+1,\qquad
a_N-\mathfrak m_{-1}\asymp N^{-d_{-1}/2}.}
\]
This dimension is determined entirely by the local structure of \(\mu\) at the endpoint \(-1\), and can be uniformly estimated from the power-law decay rate of the probe sequence \(\{a_N\}\) generated by finite Toeplitz data.
\end{definition}

\paragraph{Intrinsic Exponent of Horizon Renormalization: Complementary Contraction Rates of Endpoint Atomic and Continuous Modes}
The M\"obius semigroup \(\{\psi_m\}_{m>0}\subset\mathrm{Aut}(\mathbb{D})\) of Appendix \eqref{eq:scale-psi-m} linearizes strictly in endpoint coordinates and satisfies
\(\mathcal{B}_{-1}(\psi_m(w))=m\,\mathcal{B}_{-1}(w)\) (Theorem \ref{thm:xi-endpoint-poisson-weight-scaling}).
On the other hand, the Carath\'eodory representation gives \(\Re \mathscr{C}_{\zeta}(w)=\int_{\TT}P_w(\xi)\,d\mu(\xi)\) (Poisson integral).
Combining these two facts promotes the endpoint spectral dimension \(d_{-1}\) to an intrinsic contraction exponent along the renormalization orbit.

\begin{theorem}[Renormalization Index of the Endpoint Continuous Spectrum]\label{thm:xi-endpoint-renormalization-index}
Assume RH holds, so that \(\mathscr{C}_{\zeta}\) is Carath\'eodory and represented by the measure \(\mu\).
Let \(m>1\) and write \(\psi_{m^n}=\psi_m^{\circ n}\).
Under the endpoint spectral dimension regime (Definition \ref{def:xi-endpoint-spectral-dimension}), for any compact subset \(K\Subset\mathbb{D}\) there exists a constant \(C_K>0\) such that for all \(n\ge 0\),
\[
\sup_{w\in K}\left|
m^{-n}\Re \mathscr{C}_{\zeta}\!\bigl(\psi_{m^n}(w)\bigr)
-\mathfrak m_{-1}\,\mathcal{B}_{-1}(w)
\right|
\le C_K\,m^{-n d_{-1}/2}.
\]
In particular: if \(\mathfrak m_{-1}>0\), then the endpoint atom produces a non-decaying threshold mode
\(
m^{-n}\Re \mathscr{C}_{\zeta}(\psi_{m^n}(w))\to \mathfrak m_{-1}\mathcal{B}_{-1}(w)
\);
if \(\mathfrak m_{-1}=0\), then \(m^{-n}\Re \mathscr{C}_{\zeta}(\psi_{m^n}(w))=O(m^{-n d_{-1}/2})\).
\end{theorem}

\begin{proof}
By the Poisson representation,
\(
\Re \mathscr{C}_{\zeta}(\psi_{m^n}(w))
=\int_{\TT}P_{\psi_{m^n}(w)}(\xi)\,d\mu(\xi)
\).
For the endpoint atomic term \(\xi=-1\), Theorem \ref{thm:xi-endpoint-poisson-weight-scaling} yields
\(
P_{\psi_{m^n}(w)}(-1)=m^n P_w(-1)=m^n\mathcal{B}_{-1}(w)
\), so that after \(m^{-n}\) normalization this is exactly \(\mathfrak m_{-1}\mathcal{B}_{-1}(w)\).
For the remainder, when \(w\in K\), \(\mathcal{B}_{-1}(w)\) is bounded, so the endpoint layer depth of \(\psi_{m^n}(w)\) is of order \(m^n\);
the effective angular width of the Poisson kernel in the endpoint neighborhood is \(m^{-n/2}\), and its action on \(\mu_{\neq-1}\) is equivalent to a class of endpoint heat-kernel averages.
The uniform decay bound \(m^{-n d_{-1}/2}\) then follows from the Laplace power-law classification of Theorem \ref{thm:xi-endpoint-spectral-dimension-laplace}. \qed
\end{proof}

\paragraph{Multichannel Strictification: Operator-Valued Endpoint Atoms and Completely Positive Spectral Dimension}
On the strictification visible quotient of \(\mathcal{A}=\CC[G^{\mathrm{BC}}_m]\), the completely positive Carath\'eodory family is characterized by the operator-valued Herglotz representation and block Toeplitz--PSD completeness (Theorem \ref{thm:xi-operator-herglotz-toeplitz-completeness} and Theorem \ref{thm:xi-gbc-maximal-cp-visible-realization}).
The endpoint probe can be defined via the same binomial template at the block Toeplitz level, thereby promoting ``endpoint threshold loading'' to a completely positive, state-wise auditable family of invariants.

\begin{theorem}[Operator-Valued Endpoint Atoms and Spectral Dimension]\label{thm:xi-operator-endpoint-heat-kernel-dimension}
Let \(\mathcal{A}\) be a finite-dimensional \(C^\ast\)-algebra and let \(F:\mathbb{D}\to\mathcal{A}\) be an operator-valued Carath\'eodory function.
Write \(F(w)=c_0+2\sum_{n\ge 1}c_n w^n\) and let the block Toeplitz truncation be
\(
\mathbb{T}_N(F):=(c_{j-k})_{0\le j,k\le N}\in M_{N+1}(\mathcal{A})
\).
For the binomial vector \(b^{(N)}\), define the operator-valued endpoint probe
\[
\boxed{\;
\mathbf{a}_N
:=\frac{(b^{(N)}\otimes I)^\ast\,\mathbb{T}_N(F)\,(b^{(N)}\otimes I)}{4^N}\ \in\ \mathcal{A}_+.\;}
\]
Then there exists an \(\mathcal{A}\)-valued positive measure \(\boldsymbol\mu\) such that \(F\) has the Herglotz representation \eqref{eq:xi-operator-herglotz}, and there exists an endpoint atom
\(\mathbf{m}_{-1}:=\boldsymbol\mu(\{-1\})\in\mathcal{A}_+\) satisfying
\[
\boxed{\;
\mathbf{a}_N\downarrow \mathbf{m}_{-1}\qquad(N\to\infty)\;}
\]
(in the strong operator topology; equivalently, for any state \(\omega\), \(\omega(\mathbf{a}_N)\downarrow \omega(\mathbf{m}_{-1})\)).
If furthermore, for each state \(\omega\), the induced scalar measure \(\mu_{\omega}:=\omega\circ\boldsymbol\mu\) has spectral dimension \(d_{-1}(\omega)\) at the endpoint, then
\[
\omega(\mathbf{a}_N-\mathbf{m}_{-1})\asymp N^{-d_{-1}(\omega)/2},
\]
so that the endpoint spectral dimension after strictification becomes a completely positive, state-wise verifiable family of invariants.
\end{theorem}

\begin{proof}
By Theorem \ref{thm:xi-operator-herglotz-toeplitz-completeness}, \(F\) admits an \(\mathcal{A}\)-valued positive measure representation.
For any state \(\omega\), composition yields a scalar Carath\'eodory function \(\omega\circ F\) and its positive measure \(\mu_{\omega}\);
applying Theorem \ref{thm:xi-endpoint-heat-kernel-probe} and Theorem \ref{thm:xi-endpoint-spectral-dimension-laplace} to \(\mu_{\omega}\) gives \(\omega(\mathbf{a}_N)\downarrow \omega(\mathbf{m}_{-1})\) and the power-law estimate.
Since states separate positive elements in a finite-dimensional \(C^\ast\)-algebra, we obtain strong operator convergence \(\mathbf{a}_N\downarrow \mathbf{m}_{-1}\) and the state-wise spectral dimension conclusion. \qed
\end{proof}

\paragraph{Endpoint Focusing Falsification in the Non-RH Phase: Detection Threshold of the Binomial Template and True Resonance Poles}
When RH does not hold, \(\mathscr{C}_{\zeta}\) has poles inside \(\mathbb{D}\), forcing negative inertia in some Toeplitz truncation (Theorem \ref{thm:xi-finite-window-true-resonance-index}).
The following result shows that, under a ``nearest pole dominant'' finite-type regime, the same endpoint binomial template inevitably produces negative values, thereby yielding a reproducible endpoint-focused falsifier; its detection threshold is controlled by the nearest pole \(r_{\min}\) (equivalently, by the non-temperedness exponent of the multipole moments \(\Lambda_{\mathrm{mom}}\)).

\begin{theorem}[Endpoint Binomial Falsifier (Nearest Pole Dominant Regime)]\label{thm:xi-endpoint-binomial-falsifier}
Assume RH does not hold, and suppose \(\mathscr{C}_{\zeta}\) has a pole \(w_\ast\) inside \(\mathbb{D}\) realizing
\(
r_{\min}:=\min\{|w|:\ \mathscr{C}_{\zeta}\ \text{has a pole at}\ w\}\in(0,1)
\)
(see \eqref{eq:xi-horizon-rmin-pole}).
If this deepest pole is isolated and dominant in modulus (e.g., the modulus minimizer is unique), then there exists an infinite sequence \(N_j\to\infty\) such that
\[
\boxed{\;
(b^{(N_j)})^{\mathsf T}\mathbf{T}_{N_j}\,b^{(N_j)}<0\;}
\]
so that the endpoint binomial template yields a strictly negative finite-dimensional falsifier at certain depth levels.
Moreover, the detection threshold satisfies
\[
N_j\ \gtrsim\ \Lambda_{\mathrm{mom}}^{-1},
\qquad
\Lambda_{\mathrm{mom}}:=-\log r_{\min}>0,
\]
where \(\Lambda_{\mathrm{mom}}\) can be estimated in closed form from the quadratic compression at high altitudes in the endpoint horocycle depth (Theorem \ref{thm:xi-horizon-multipole-temperedness-dichotomy}, equations \eqref{eq:xi-horizon-Lambda-closed}--\eqref{eq:xi-horizon-Lambda-asymptotic}).
\end{theorem}

\begin{proof}
By Theorem \ref{thm:xi-horizon-forced-geometric-tail}, the nearest pole \(w_\ast\) forces a geometric tail in the coefficients:
\(
c_n=-\frac{R_\ast}{2w_\ast^{n+1}}+h_n
\)
(\(n\ge 1\), \(R_\ast\neq 0\)), where \(\{h_n\}\) comes from the remainder analytic in a larger disk.
On the other hand, the binomial vector corresponds to the polynomial \(p_N(\xi)=(1-\xi)^N\), whose autocorrelation Laurent polynomial satisfies
\(
|1-\xi|^{2N}=\sum_{n=-N}^{N}(-1)^n\binom{2N}{N+n}\xi^n
\);
therefore, for any Toeplitz data, one has the combinatorial identity
\[
(b^{(N)})^{\mathsf T}\mathbf{T}_N b^{(N)}
=\sum_{n=-N}^{N}(-1)^n\binom{2N}{N+n}\,c_n.
\]
Decomposing \(c_n\) into the pole-dominant term and the remainder, the dominant term has amplitude \(\asymp 4^N r_{\min}^{-(N+1)}\) while the remainder is at most \(\asymp 4^N\);
thus, when \(N\) is large enough for \(r_{\min}^{-N}\) to dominate, the sign is determined by the phase oscillation of the dominant term, which must take negative values for infinitely many \(N\).
Finally, the threshold follows from the condition that \(r_{\min}^{-N}\) overwhelms the \(O(1)\) remainder, i.e., \(N\gtrsim (-\log r_{\min})^{-1}=\Lambda_{\mathrm{mom}}^{-1}\). \qed
\end{proof}
