\section{Problem 5: $\mathcal O$-Adapted Slice Filtration and Geometric Fixed Points}

Let $G$ be a finite group and let $\mathcal O$ be an incomplete transfer system (equivalently, an $N_\infty$ indexing system). Write $\mathrm{Sp}^{\mathcal O}_G$ for the corresponding incomplete equivariant stable category.

\subsection*{External inputs used}

We use the following standard structural results.

\begin{enumerate}
\item \textbf{Restriction/induction and Mackey formula.}
For each $H\le G$, there are adjoint functors
\[
\operatorname{Ind}^G_H:\mathrm{Sp}^{\mathcal O|_H}_H \rightleftarrows \mathrm{Sp}^{\mathcal O}_G:\operatorname{Res}^G_H,
\]
and restriction of induced objects satisfies the usual double-coset (Mackey) decomposition.
\item \textbf{Geometric fixed points via isotropy separation.}
If $\mathcal P$ is the family of proper subgroups of $G$, then for all $E\in\mathrm{Sp}^{\mathcal O}_G$ there is a cofiber sequence
\[
E\mathcal P_+\wedge E\longrightarrow E\longrightarrow \widetilde E\mathcal P\wedge E.
\]
The geometric fixed point functor is
\[
\Phi^G(E)\simeq \bigl((\widetilde E\mathcal P\wedge E)^{\mathrm{fib}}\bigr)^G.
\]
\item \textbf{Local geometric equivalence.}
The localization by $\widetilde E\mathcal P\wedge(-)$ kills the localizing subcategory generated by proper orbits, and categorical fixed points identify the localized category with $\mathrm{Sp}$.
Equivalently: on geometric (proper-isotropy-free) objects, $\Phi^G$ is an exact equivalence to ordinary spectra.
\item \textbf{Representation-sphere behavior.}
\[
\Phi^G(S^{V})\simeq S^{V^G}
\]
for genuine representation spheres; in particular $\Phi^G(S^{k\rho_G})\simeq S^k$.
\end{enumerate}

Items (1)--(3) are exactly the structural framework developed by Blumberg--Hill for incomplete transfer systems; item (4) is the standard geometric fixed-point formula on representation spheres.

\subsection*{Definition of the $\mathcal O$-adapted regular slice filtration}

For each subgroup $H\le G$, let $\rho_H$ be the real regular representation of $H$.
For $n\in\mathbb Z$, define
\[
\tau^{\mathcal O,G}_{\ge n}
:=
\operatorname{Loc}
\left\{
G_+\wedge_H S^{k\rho_H}
\;:\;
H\le G,\ k\ge 0,\ k|H|\ge n
\right\}
\subset \mathrm{Sp}^{\mathcal O}_G.
\]
Here $\operatorname{Loc}$ means closure under equivalences, cofibers, suspensions/desuspensions, retracts, and arbitrary coproducts (hence homotopy colimits).

An object $E\in\mathrm{Sp}^{\mathcal O}_G$ is \emph{$\mathcal O$-slice $n$-connective} if $E\in\tau^{\mathcal O,G}_{\ge n}$.

\subsection*{Main theorem}

\begin{theorem}[Geometric fixed-point criterion]\label{thm:q5-main}
Let $E\in\mathrm{Sp}^{\mathcal O}_G$ be connective and let $n\ge 0$. Then
\[
E\in\tau^{\mathcal O,G}_{\ge n}
\quad\Longleftrightarrow\quad
\forall H\le G,\ \Phi^H(E)\in\mathrm{Sp}_{\ge \lceil n/|H|\rceil}.
\]
\end{theorem}

\subsection*{Preparatory lemmas}

\begin{lemma}[Restriction of regular representations]\label{lem:q5-rho-restrict}
For $L\le K\le G$,
\[
\operatorname{Res}^K_L(\rho_K)\cong \rho_L^{\oplus [K:L]}.
\]
Hence $k\rho_K$ restricts to $k[K:L]\rho_L$, and
\[
k[K:L]|L|=k|K|.
\]
\end{lemma}

\begin{proof}
As an $L$-set, $K$ is a disjoint union of $[K:L]$ left cosets, each isomorphic to $L$. Passing to permutation representations gives the formula.
\end{proof}

\begin{lemma}[Restriction preserves slice-connectivity]\label{lem:q5-res-preserve}
For every $H\le G$,
\[
\operatorname{Res}^G_H\bigl(\tau^{\mathcal O,G}_{\ge n}\bigr)\subseteq \tau^{\mathcal O|_H,H}_{\ge n}.
\]
\end{lemma}

\begin{proof}
It suffices to check generators.
Take $G_+\wedge_K S^{k\rho_K}$ with $k|K|\ge n$.
By Mackey decomposition,
\[
\operatorname{Res}^G_H(G_+\wedge_K S^{k\rho_K})
\simeq
\bigvee_{HgK\in H\backslash G/K}
H_+\wedge_{J_g}
\operatorname{Res}^{gKg^{-1}}_{J_g}(S^{k\rho_{gKg^{-1}}}),
\]
where $J_g:=H\cap gKg^{-1}$.
By Lemma~\ref{lem:q5-rho-restrict},
\[
\operatorname{Res}^{gKg^{-1}}_{J_g}(S^{k\rho_{gKg^{-1}}})
\simeq
S^{k[gKg^{-1}:J_g]\rho_{J_g}}.
\]
Its slice degree is
\[
k[gKg^{-1}:J_g]|J_g|
=k|gKg^{-1}|
=k|K|
\ge n.
\]
So every summand is a generator of $\tau^{\mathcal O|_H,H}_{\ge n}$, hence so is the wedge.
\end{proof}

\begin{lemma}[Induction preserves slice-connectivity]\label{lem:q5-ind-preserve}
For $H\le G$,
\[
\operatorname{Ind}^G_H\bigl(\tau^{\mathcal O|_H,H}_{\ge n}\bigr)\subseteq \tau^{\mathcal O,G}_{\ge n}.
\]
\end{lemma}

\begin{proof}
Again on generators:
\[
\operatorname{Ind}^G_H\left(H_+\wedge_K S^{k\rho_K}\right)
\simeq
G_+\wedge_K S^{k\rho_K},
\]
which is a generator of $\tau^{\mathcal O,G}_{\ge n}$ when $k|K|\ge n$.
Extend by locality.
\end{proof}

\begin{lemma}[Isotropy separation]\label{lem:q5-isotropy}
Let $\mathcal P$ be the family of proper subgroups of $G$. For all $E\in\mathrm{Sp}^{\mathcal O}_G$:
\[
E\mathcal P_+\wedge E\longrightarrow E\longrightarrow \widetilde E\mathcal P\wedge E
\]
is a cofiber sequence.
\end{lemma}

\begin{lemma}[Proper-isotropy criterion]\label{lem:q5-proper}
For $n\in\mathbb Z$, the following are equivalent:
\begin{enumerate}
\item $E\mathcal P_+\wedge E\in \tau^{\mathcal O,G}_{\ge n}$.
\item $\operatorname{Res}^G_H E\in\tau^{\mathcal O|_H,H}_{\ge n}$ for every proper $H<G$.
\end{enumerate}
\end{lemma}

\begin{proof}
\textbf{(1)$\Rightarrow$(2).}
Restrict and apply Lemma~\ref{lem:q5-res-preserve}:
\[
\operatorname{Res}^G_H(E\mathcal P_+\wedge E)\in\tau^{\mathcal O|_H,H}_{\ge n}.
\]
Since $H\in\mathcal P$, $(E\mathcal P)^H$ is contractible; hence
\[
\operatorname{Res}^G_H(E\mathcal P_+\wedge E)\simeq \operatorname{Res}^G_H E.
\]
So $\operatorname{Res}^G_H E\in\tau^{\mathcal O|_H,H}_{\ge n}$.

\textbf{(2)$\Rightarrow$(1).}
$E\mathcal P$ is a homotopy colimit of $G/H$ with $H<G$, therefore
\[
E\mathcal P_+\wedge E
\simeq
\operatorname*{hocolim}_{H<G}(G/H_+\wedge E)
\simeq
\operatorname*{hocolim}_{H<G}\operatorname{Ind}^G_H\operatorname{Res}^G_H E.
\]
By hypothesis and Lemma~\ref{lem:q5-ind-preserve}, each term is in $\tau^{\mathcal O,G}_{\ge n}$.
Locality gives the homotopy colimit in $\tau^{\mathcal O,G}_{\ge n}$.
\end{proof}

\begin{lemma}[Geometric-piece criterion]\label{lem:q5-geometric-piece}
For $n\in\mathbb Z$:
\[
\widetilde E\mathcal P\wedge E\in\tau^{\mathcal O,G}_{\ge n}
\Longleftrightarrow
\Phi^G(E)\in\mathrm{Sp}_{\ge \lceil n/|G|\rceil}.
\]
\end{lemma}

\begin{proof}
Set $L:=\widetilde E\mathcal P\wedge(-)$ and $X:=L(E)$.
Then $X$ is geometric.

Apply $L$ to the definition of $\tau^{\mathcal O,G}_{\ge n}$:
\[
L\bigl(\tau^{\mathcal O,G}_{\ge n}\bigr)
=
\operatorname{Loc}
\left\{
L\!\left(G_+\wedge_H S^{k\rho_H}\right): k|H|\ge n
\right\}.
\]
For $H<G$, $L(G_+\wedge_H Y)=0$ (proper isotropy is killed by localization).
For $H=G$, $L(S^{k\rho_G})=S^{k\rho_G}$.
Hence
\[
L\bigl(\tau^{\mathcal O,G}_{\ge n}\bigr)
=
\operatorname{Loc}\{S^{k\rho_G}:k|G|\ge n\}
\subset \mathrm{Geom}(\mathrm{Sp}^{\mathcal O}_G).
\]

Because $X$ is already geometric, we have
\[
X\in\tau^{\mathcal O,G}_{\ge n}
\iff
X\in L\bigl(\tau^{\mathcal O,G}_{\ge n}\bigr)
\iff
X\in \operatorname{Loc}\{S^{k\rho_G}:k|G|\ge n\}.
\]

Now use External Inputs (3)--(4): on geometric objects, $\Phi^G$ is an exact equivalence to $\mathrm{Sp}$ and
\[
\Phi^G(S^{k\rho_G})\simeq S^k.
\]
Therefore
\[
X\in \operatorname{Loc}\{S^{k\rho_G}:k|G|\ge n\}
\iff
\Phi^G(X)\in \operatorname{Loc}\{S^k:k\ge\lceil n/|G|\rceil\}.
\]
The right-hand localizing subcategory is exactly $\mathrm{Sp}_{\ge\lceil n/|G|\rceil}$.
Finally $\Phi^G(X)=\Phi^G(E)$ by definition of geometric fixed points.
\end{proof}

\subsection*{Proof of Theorem~\ref{thm:q5-main}}

\begin{proof}
We prove by induction on $|G|$.

\paragraph{Base case $|G|=1$.}
Then $\mathrm{Sp}^{\mathcal O}_G=\mathrm{Sp}$, $\rho_G$ is $1$-dimensional, and $\tau^{\mathcal O,G}_{\ge n}=\mathrm{Sp}_{\ge n}$, so the statement is tautological.

\paragraph{Forward implication.}
Assume $E\in\tau^{\mathcal O,G}_{\ge n}$.
Fix $H\le G$. By Lemma~\ref{lem:q5-res-preserve},
\[
\operatorname{Res}^G_H E\in \tau^{\mathcal O|_H,H}_{\ge n}.
\]
Apply Lemma~\ref{lem:q5-geometric-piece} inside group $H$:
\[
\Phi^H(E)=\Phi^H(\operatorname{Res}^G_H E)\in \mathrm{Sp}_{\ge \lceil n/|H|\rceil}.
\]

\paragraph{Reverse implication.}
Assume
\[
\forall H\le G,\ \Phi^H(E)\in\mathrm{Sp}_{\ge \lceil n/|H|\rceil}.
\]
We must show $E\in\tau^{\mathcal O,G}_{\ge n}$.

By Lemma~\ref{lem:q5-isotropy}, it suffices to show both end terms are in $\tau^{\mathcal O,G}_{\ge n}$.

\emph{Geometric term.}
From the case $H=G$ and Lemma~\ref{lem:q5-geometric-piece},
\[
\widetilde E\mathcal P\wedge E\in\tau^{\mathcal O,G}_{\ge n}.
\]

\emph{Proper-isotropy term.}
Take any proper subgroup $H<G$.
For every $K\le H$, we have
\[
\Phi^K(\operatorname{Res}^G_H E)=\Phi^K(E)\in\mathrm{Sp}_{\ge \lceil n/|K|\rceil}
\]
by hypothesis.
Since $|H|<|G|$, the induction hypothesis applied to $H$ gives
\[
\operatorname{Res}^G_H E\in\tau^{\mathcal O|_H,H}_{\ge n}.
\]
Thus condition (2) of Lemma~\ref{lem:q5-proper} holds, so
\[
E\mathcal P_+\wedge E\in\tau^{\mathcal O,G}_{\ge n}.
\]

Now both ends of the isotropy-separation cofiber sequence lie in $\tau^{\mathcal O,G}_{\ge n}$; closure under extensions gives
\[
E\in\tau^{\mathcal O,G}_{\ge n}.
\]
This proves the theorem.
\end{proof}

\begin{remark}
The right-hand condition depends only on subgroup-indexed geometric fixed points. The proof shows this is sufficient because the proper-isotropy and geometric pieces are controlled separately by Lemmas~\ref{lem:q5-proper} and~\ref{lem:q5-geometric-piece}.
\end{remark}
