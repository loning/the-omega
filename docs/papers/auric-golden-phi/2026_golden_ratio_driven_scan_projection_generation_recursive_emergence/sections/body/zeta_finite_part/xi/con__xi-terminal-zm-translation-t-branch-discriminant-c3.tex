\begin{theorem}[双参数四次模型的分歧判别式闭式与 \texorpdfstring{$C_{3}$}{C3} 协变律]\label{thm:xi-terminal-zm-translation-t-branch-discriminant-c3-closed}
定义
\[
F_t(\mu,u):=\mu^4-\mu^3-(tu+1)\mu^2+\mu(tu-u^3)+tu.
\]
令
\[
\operatorname{Disc}_{\mu}\!\bigl(F_t(\mu,u)\bigr)=u\,D_t(u),
\qquad D_t(u)\in\ZZ[t,u].
\]
则 \(\deg_u D_t=11\)，并且
\[
\boxed{
\begin{aligned}
D_t(u)=\;&-27u^{11}+90t\,u^9+(4t^3-22)u^8-239t^2u^7-(8t^4+70t)u^6\\
&+(236t^3+5)u^5+(20t^5-164t^2)u^4-(120t^4+108t)u^3\\
&+220t^3u^2-120t^2u+20t.
\end{aligned}}
\]
此外，对任意满足 \(\zeta^3=1\) 的三次单位根 \(\zeta\)，有严格协变关系
\[
\boxed{\,D_{\zeta t}(u)=\zeta\,D_t(\zeta u)\, }.
\]
\end{theorem}
\begin{proof}
由判别式定义
\[
\operatorname{Disc}_{\mu}(F)=(-1)^{n(n-1)/2}\operatorname{Res}_{\mu}(F,\partial_{\mu}F),\qquad n=4,
\]
对 \(F_t\) 直接消元可得
\[
\operatorname{Disc}_{\mu}(F_t)=u\,D_t(u),
\]
并得到上述闭式系数。
\par
对协变性，先验证恒等式
\[
F_{\zeta t}(\mu,\zeta^{-1}u)=F_t(\mu,u),
\]
因为 \(u^3\) 在 \(u\mapsto \zeta^{-1}u\) 下不变，且 \(tu\) 在 \((t,u)\mapsto(\zeta t,\zeta^{-1}u)\) 下不变。
对两边取 \(\mu\)-判别式：
\[
u\,D_t(u)=\zeta^{-1}u\,D_{\zeta t}(\zeta^{-1}u).
\]
将 \(u\) 替换为 \(\zeta u\) 即得
\[
D_{\zeta t}(u)=\zeta D_t(\zeta u).
\]
可复现核验脚本：\texttt{scripts/exp\_fold\_zm\_elliptic\_translation\_t\_discriminant\_c3\_audit.py}。证毕。
\end{proof}

\begin{theorem}[二次判别式 \texorpdfstring{$\Delta(t)=\operatorname{Disc}_u(D_t)$}{Delta(t)=Disc_u(Dt)} 的完全分解]\label{thm:xi-terminal-zm-translation-t-branch-discriminant-c3-delta-factor}
令
\[
\Delta(t):=\operatorname{Disc}_u\!\bigl(D_t(u)\bigr)\in\ZZ[t].
\]
则有完全分解
\[
\boxed{
\begin{aligned}
\Delta(t)=\;&515978035200\;t^4\,(80t^6+16424t^3-84375)\,(256t^6-1312t^3+3125)^2\\
&\times(8000t^{12}+429800t^9+96378087t^6-2034256t^3-64)^3.
\end{aligned}}
\]
等价地，记 \(s=t^3\)，则
\[
\Delta(t)=515978035200\;t^4\,\Phi_1(s)\,\Phi_2(s)^2\,\Phi_3(s)^3,
\]
其中
\[
\Phi_1(s)=80s^2+16424s-84375,\qquad
\Phi_2(s)=256s^2-1312s+3125,
\]
\[
\Phi_3(s)=8000s^4+429800s^3+96378087s^2-2034256s-64.
\]
\end{theorem}
\begin{proof}
由 \(\deg_u D_t=11\) 与判别式定义
\[
\operatorname{Disc}_u(D_t)=(-1)^{m(m-1)/2}\operatorname{Res}_u(D_t,\partial_u D_t),\qquad m=11,
\]
直接消元并因式分解得所示公式。
\par
由定理 \ref{thm:xi-terminal-zm-translation-t-branch-discriminant-c3-closed} 的协变律
\[
D_{\zeta t}(u)=\zeta D_t(\zeta u),
\]
并使用判别式缩放公式
\[
\operatorname{Disc}_u\!\bigl(c\,P(au)\bigr)=c^{2m-2}a^{m(m-1)}\operatorname{Disc}_u(P),\qquad m=11,
\]
得
\[
\Delta(\zeta t)=\zeta^{20}\zeta^{110}\Delta(t)=\zeta\,\Delta(t).
\]
因此 \(\Delta\) 的幂次仅落在 \(1\pmod 3\) 的同余类上，等价写为 \(t^4\) 乘 \(t^3\)-多项式；上式给出了该结构的精确因子。证毕。
\end{proof}

\begin{corollary}[\texorpdfstring{$\RR$}{R}-参数下分歧碰撞点的精确刻画]\label{cor:xi-terminal-zm-translation-t-branch-discriminant-c3-real-collision}
对 \(t\in\RR\)，多项式 \(D_t(u)\) 在 \(\CC\) 上出现重根当且仅当
\[
t=0,\quad\text{或}\quad 80t^6+16424t^3-84375=0,\quad\text{或}\quad
8000t^{12}+429800t^9+96378087t^6-2034256t^3-64=0.
\]
其中：
\begin{enumerate}
  \item \(80t^6+16424t^3-84375=0\) 的两实解为
  \[
  t^3=-\frac{2053}{20}\pm\frac{247}{10}\sqrt{19},
  \qquad
  t=\sqrt[3]{-\frac{2053}{20}\pm\frac{247}{10}\sqrt{19}},
  \]
  数值为
  \[
  t\approx 1.7116619053346387,\qquad t\approx -5.946890580731152;
  \]
  \item \(8000t^{12}+429800t^9+96378087t^6-2034256t^3-64=0\) 恰有两实根，数值为
  \[
  t\approx 0.2764887319883657,\qquad t\approx -0.0315531574228415;
  \]
  \item \(256t^6-1312t^3+3125=0\) 在 \(\RR\) 上无解。
\end{enumerate}
\end{corollary}
\begin{proof}
由定理 \ref{thm:xi-terminal-zm-translation-t-branch-discriminant-c3-delta-factor}，
\(\Delta(t)=0\) 当且仅当其因子之一为零。
对 \(\Phi_2(s)=256s^2-1312s+3125\)，判别式为
\[
(-1312)^2-4\cdot 256\cdot 3125=-1478656<0,
\]
故无实根。
\par
\(\Phi_1\) 为二次式，直接求根得到闭式表达。
\(\Phi_3\) 为四次式，且其导数三次式判别式为
\[
\operatorname{Disc}\!\bigl(\Phi_3'(s)\bigr)=-855217962210024019631145312000<0,
\]
故 \(\Phi_3'\) 仅有一个实根，从而 \(\Phi_3\) 仅有一个实极值点，实根数至多为 \(2\)。
又 \(\Phi_3(0)=-64<0\)，且 \(\Phi_3(s)\to +\infty\)（\(s\to\pm\infty\)），故恰有一正一负两个实根。
映射 \(t\mapsto t^3\) 在 \(\RR\) 上双射，因此对应地恰有两实 \(t\)-根。证毕。
\end{proof}

\begin{theorem}[四次因子的严格正性与唯一实 \texorpdfstring{$u=1$}{u=1} 共振]\label{thm:xi-terminal-zm-translation-t-branch-discriminant-c3-u1-positivity}
