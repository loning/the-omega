\subsection{AQFT states, representations, and the GNS net}
\label{app:aqft_states_representations_gns_nets}

This appendix upgrades the state-functional/GNS background (Appendix~\ref{app:state_gns_background}) to the setting of local nets.
It is a structural bridge: it makes explicit how a state on the quasi-local algebra induces a concrete Hilbert-space representation in which the net is realized by operator algebras.

\subsubsection{States on the quasi-local algebra}
\paragraph{State.}
A state on the quasi-local algebra $\mathcal{A}$ (Appendix~\ref{app:aqft_axioms_local_nets}) is a positive normalized linear functional $\omega:\mathcal{A}\to\CC$.
It is the abstract version of $\omega(A)=\Tr(\rho A)$ in concrete Hilbert models (Appendix~\ref{app:state_gns_background}).

\subsubsection{GNS representation and local von Neumann algebras}
\begin{theorem}[GNS representation (structural form)]
\label{thm:z128_gns_for_nets}
Given a state $\omega$ on $\mathcal{A}$, there exists a triple $(\pi_\omega,\mathcal{H}_\omega,\Omega_\omega)$ such that:
\begin{itemize}
\item $\pi_\omega:\mathcal{A}\to\mathcal{B}(\mathcal{H}_\omega)$ is a $^\ast$-representation,
\item $\Omega_\omega\in\mathcal{H}_\omega$ is cyclic for $\pi_\omega(\mathcal{A})$,
\item $\omega(A)=\langle \Omega_\omega|\pi_\omega(A)\Omega_\omega\rangle$ for all $A\in\mathcal{A}$.
\end{itemize}
\end{theorem}
\begin{proof}[Reference]
See Appendix~\ref{app:state_gns_background}.
\end{proof}

\paragraph{Local algebras in the representation.}
Given a region $\mathcal{O}$, define the represented local algebra
\[
\mathcal{A}_\omega(\mathcal{O}) := \pi_\omega(\mathcal{A}(\mathcal{O})) \subseteq \mathcal{B}(\mathcal{H}_\omega),
\]
and the associated local von Neumann algebra as the double commutant
\[
\mathcal{M}_\omega(\mathcal{O}) := \mathcal{A}_\omega(\mathcal{O})''.
\]
The map $\mathcal{O}\mapsto \mathcal{M}_\omega(\mathcal{O})$ is the \emph{GNS net} induced by $\omega$.

\subsubsection{Covariance implementation in the GNS picture}
\paragraph{Structural statement.}
If $\omega$ is invariant under a covariance action $\alpha_g$ on $\mathcal{A}$ (Definition~\ref{def:z128_poincare_covariance_net}), then covariance is implemented in the GNS representation by unitaries $U_\omega(g)$ satisfying
\[
U_\omega(g)\,\pi_\omega(A)\,U_\omega(g)^\dagger=\pi_\omega(\alpha_g(A)),
\qquad
U_\omega(g)\Omega_\omega=\Omega_\omega,
\]
for all $A\in\mathcal{A}$.

\paragraph{Interface reading.}
\InterfaceTag This is the operator-algebraic form of ``a state defines a concrete observer Hilbert space'':
the choice of state fixes which representation is physically realized, while locality and covariance constrain how observables are organized by regions and symmetries.

\subsubsection{Scope note}
\AuditTag In full AQFT one often imposes additional conditions (e.g.\ cyclicity/separating properties such as Reeh--Schlieder for the vacuum, or split/nuclearity conditions).
We treat such properties as optional extensions that must be explicitly declared if used.
