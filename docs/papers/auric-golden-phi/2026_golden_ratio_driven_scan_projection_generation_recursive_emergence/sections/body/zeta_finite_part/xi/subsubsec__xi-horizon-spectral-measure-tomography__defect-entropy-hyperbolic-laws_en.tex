\paragraph{Hyperbolic geometry laws of defect entropy: area, horocycle length, and scale fingerprints}
This paragraph provides several scalar consequences that depend only on \(\{(\gamma_j,\delta_j,m_j)\}\): they strictly identify
\(\displaystyle S_{\mathrm{def}}=\sum_j m_j\frac{\delta_j}{1+\delta_j}\)
as hyperbolic area and horocycle length deficit, and provide optimal feasible intervals enforced by linear flux and integer indices.
Furthermore, through scale coarse-graining, a completely monotone Stieltjes trajectory is obtained, thereby further compressing the depth spectrum into an auditable single-variable fingerprint.

\begin{proposition}[Hyperbolic area law of defect entropy and \texorpdfstring{$4\pi$}{4pi} normalization]\label{prop:xi-defect-entropy-hyperbolic-area-law-4pi}
In the upper half-plane model \(\mathbb{H}=\{x+\mathrm{i}y:\ y>0\}\) take the hyperbolic metric
\(
ds^2=(dx^2+dy^2)/y^2
\), with area element
\(
\dd A_{\mathbb{H}}=\frac{\dd x\,\dd y}{y^2}
\).
For each point defect parameter \((\gamma_j,\delta_j,m_j)\) (\(\delta_j>0\)) define the unit-width vertical strip domain
\[
R_j:=
\bigl[\gamma_j-\tfrac12,\gamma_j+\tfrac12\bigr]\times\bigl[1,1+\delta_j\bigr]\subset\mathbb{H}.
\]
Then its hyperbolic area satisfies the strict identity
\begin{equation}\label{eq:xi-hyperbolic-area-vertical-strip}
\boxed{\;
A_{\mathbb{H}}(R_j)
:=\int_{R_j}\dd A_{\mathbb{H}}
=\frac{\delta_j}{1+\delta_j}\; }.
\end{equation}
Thus the defect entropy possesses a hyperbolic area law
\begin{equation}\label{eq:xi-defect-entropy-hyperbolic-area-law}
\boxed{\;
S_{\mathrm{def}}
=\sum_{j=1}^{\kappa}m_j\,A_{\mathbb{H}}(R_j)\; }.
\end{equation}
On the other hand, the comoving scan profile \(H\) (\eqref{eq:xi-comoving-scan-profile-H}) satisfies the atom-wise identity
\begin{equation}\label{eq:xi-atom-mass-4pi-area-alignment}
\boxed{\;
\int_{\RR} m_j\,\frac{4\delta_j}{(x-\gamma_j)^2+(1+\delta_j)^2}\,\dd x
=4\pi\,m_j\,A_{\mathbb{H}}(R_j)\; }.
\end{equation}
Therefore
\(
\frac{1}{4\pi}\int_{\RR}H(x)\dd x
=S_{\mathrm{def}}
\)
agrees with Definition \ref{def:xi-defect-entropy-functional};
the role of "\(4\pi\)" is precisely to align the Euclidean integral of the boundary Poisson trace with hyperbolic area under the same normalization.
\end{proposition}

\begin{proof}
By \(\dd A_{\mathbb{H}}=\frac{\dd x\,\dd y}{y^2}\) compute directly
\[
A_{\mathbb{H}}(R_j)
=\int_{\gamma_j-1/2}^{\gamma_j+1/2}\int_{1}^{1+\delta_j}y^{-2}\,\dd y\,\dd x
=\int_{1}^{1+\delta_j}y^{-2}\,\dd y
=1-\frac{1}{1+\delta_j}
=\frac{\delta_j}{1+\delta_j},
\]
obtaining \eqref{eq:xi-hyperbolic-area-vertical-strip}; substituting into \eqref{eq:xi-defect-entropy-closed-form} yields \eqref{eq:xi-defect-entropy-hyperbolic-area-law}.
\par
For \eqref{eq:xi-atom-mass-4pi-area-alignment}, use the standard integral
\(
\int_{\RR}\frac{\dd x}{(x-\gamma)^2+a^2}=\frac{\pi}{a}
\) (\(a>0\)) with \(a=1+\delta_j\) to obtain
\[
\int_{\RR} m_j\,\frac{4\delta_j}{(x-\gamma_j)^2+(1+\delta_j)^2}\,\dd x
=4\pi\,m_j\,\frac{\delta_j}{1+\delta_j}
=4\pi\,m_j\,A_{\mathbb{H}}(R_j).
\]
\end{proof}

\begin{corollary}[Boundary length deficit formula: entropy as net dissipation of horocycle length]\label{cor:xi-defect-entropy-horocycle-length-deficit}
Under the same hyperbolic metric, let the hyperbolic length of a horizontal line segment of Euclidean length \(1\) at height \(y\) be
\[
L(y):=\int_{-1/2}^{1/2}\frac{\dd x}{y}=\frac{1}{y}.
\]
Then for each strip domain \(R_j\) (Proposition \ref{prop:xi-defect-entropy-hyperbolic-area-law-4pi}) we have
\[
A_{\mathbb{H}}(R_j)=L(1)-L(1+\delta_j)=1-\frac{1}{1+\delta_j}.
\]
Hence the defect entropy satisfies the strict identity
\begin{equation}\label{eq:xi-defect-entropy-horocycle-length-deficit}
\boxed{\;
S_{\mathrm{def}}
=\sum_{j=1}^{\kappa}m_j\bigl(L(1)-L(1+\delta_j)\bigr)
=\kappa_{\mathrm{ind}}-\sum_{j=1}^{\kappa}\frac{m_j}{1+\delta_j}\; }.
\end{equation}
This identity interprets \(S_{\mathrm{def}}\) as: on the bottom horocycle (\(y=1\)), the hyperbolic length deficit produced by each defect lifting a same-width strip to height \(1+\delta_j\); entropy is the summable total of the deficits.
\end{corollary}

\begin{proof}
From \(A_{\mathbb{H}}(R_j)=\int_{1}^{1+\delta_j}y^{-2}\dd y\) and the definition of \(L(y)=1/y\), obtain
\(
A_{\mathbb{H}}(R_j)=L(1)-L(1+\delta_j)
\).
Substituting this identity into
\(
S_{\mathrm{def}}=\sum_j m_jA_{\mathbb{H}}(R_j)
\)
and using \(L(1)=1\) yields \eqref{eq:xi-defect-entropy-horocycle-length-deficit};
this agrees with \eqref{eq:xi-defect-entropy-index-decomposition}.
\end{proof}

\begin{proposition}[Optimal feasible interval of entropy--flux--index and harmonic mean lower bound]\label{prop:xi-defect-entropy-feasible-interval-flux-index}
Let the linear flux (total depth)
\[
\Phi:=\sum_{j=1}^{\kappa}m_j\,\delta_j
\qquad\bigl(=\ M_0^{\mathrm{def}}\ \text{in the notation of Proposition \ref{prop:xi-poisson-coarsegraining-nohair}}\bigr),
\]
and denote the integer defect index \(\kappa_{\mathrm{ind}}:=\sum_{j=1}^{\kappa}m_j\).
Then the defect entropy satisfies sharp bounds
\begin{equation}\label{eq:xi-defect-entropy-feasible-interval}
\boxed{\;
\frac{\Phi}{1+\Phi}
\ \le\
S_{\mathrm{def}}
\ \le\
\frac{\kappa_{\mathrm{ind}}\Phi}{\kappa_{\mathrm{ind}}+\Phi}\; }.
\end{equation}
The upper bound is attained in the "uniform depth" case (constant \(\delta\) when expanded by multiplicity); the lower bound approaches its infimum in the "extreme concentration" case (concentrating all depth on a single unit-multiplicity atom, with remaining depths tending to \(0\)).
In particular, when \(\Phi>0\), the upper bound equivalently rewrites as a harmonic mean type inequality
\begin{equation}\label{eq:xi-defect-entropy-harmonic-mean-bound}
\boxed{\;
\frac{1}{S_{\mathrm{def}}}\ \ge\ \frac{1}{\Phi}+\frac{1}{\kappa_{\mathrm{ind}}}\; }.
\end{equation}
\end{proposition}

\begin{proof}
Let \(f(\delta):=\frac{\delta}{1+\delta}\), then \(f\) is increasing and strictly concave on \((0,\infty)\):
\(
f''(\delta)=-2(1+\delta)^{-3}<0
\).
\par
\textbf{Upper bound:}
Apply Jensen's inequality with weights \(w_j:=m_j/\kappa_{\mathrm{ind}}\) (\(\sum_j w_j=1\)):
\[
\frac{S_{\mathrm{def}}}{\kappa_{\mathrm{ind}}}
=\sum_{j}w_j f(\delta_j)
\le f\!\Bigl(\sum_j w_j\delta_j\Bigr)
=f\!\Bigl(\frac{\Phi}{\kappa_{\mathrm{ind}}}\Bigr)
=\frac{\Phi}{\kappa_{\mathrm{ind}}+\Phi},
\]
thus \(S_{\mathrm{def}}\le \frac{\kappa_{\mathrm{ind}}\Phi}{\kappa_{\mathrm{ind}}+\Phi}\).
\par
\textbf{Lower bound:}
Since \(m_j\in\NN\) (setting of this section), we have \(\delta_j\le m_j\delta_j\le \Phi\), hence
\(
\frac{\delta_j}{1+\delta_j}\ge \frac{\delta_j}{1+\Phi}
\).
Multiplying term-wise by \(m_j\) and summing:
\[
S_{\mathrm{def}}
=\sum_j m_j\frac{\delta_j}{1+\delta_j}
\ge \sum_j m_j\frac{\delta_j}{1+\Phi}
=\frac{\Phi}{1+\Phi}.
\]
Finally, \eqref{eq:xi-defect-entropy-harmonic-mean-bound} is obtained by taking the reciprocal of the upper bound and rearranging.
\end{proof}

\begin{proposition}[Schur-concavity: majorization optimization principle of defect entropy with respect to depth vector]\label{prop:xi-defect-entropy-schur-concavity}
Fix \(\kappa_{\mathrm{ind}}\ge 2\) and expand depths by multiplicity into vector
\(\boldsymbol{\delta}=(\delta_1,\dots,\delta_{\kappa_{\mathrm{ind}}})\in(0,\infty)^{\kappa_{\mathrm{ind}}}\).
Define
\[
\mathcal{S}(\boldsymbol{\delta})
:=\sum_{i=1}^{\kappa_{\mathrm{ind}}}\frac{\delta_i}{1+\delta_i}.
\]
Then \(\mathcal{S}\) is a strictly Schur-concave function: if \(\boldsymbol{\delta}\) is \textbf{more concentrated} in majorization order (i.e., \(\boldsymbol{\delta}\succ \boldsymbol{\delta}'\) and they differ beyond permutation), then
\[
\mathcal{S}(\boldsymbol{\delta})\ <\ \mathcal{S}(\boldsymbol{\delta}').
\]
Therefore under any linear constraint \(\sum_{i=1}^{\kappa_{\mathrm{ind}}}\delta_i=\Phi\),
the maximum of \(\mathcal{S}\) is attained by the equal-depth configuration \(\delta_i\equiv \Phi/\kappa_{\mathrm{ind}}\), while its infimum is approached by extreme concentration configurations (one component tending to \(\Phi\), remaining components tending to \(0\)).
This majorization principle fully agrees with Proposition \ref{prop:xi-defect-entropy-feasible-interval-flux-index}'s "upper bound attained at uniform depth, lower bound approached at extreme concentration", and elevates it to a structural optimization law.
\end{proposition}
\begin{proof}
The function \(f(\delta)=\delta/(1+\delta)\) is strictly concave on \((0,\infty)\) (see the computation in Proposition \ref{prop:xi-defect-entropy-feasible-interval-flux-index}),
hence by Karamata's inequality (strict case) we know \(\sum_i f(\delta_i)\) is strictly Schur-concave.
Under a fixed sum constraint, equal-depth configuration is the minimal element in majorization order, thus gives the maximum; extreme concentration is the maximal endpoint in majorization order, thus gives the infimum.
\end{proof}

\begin{proposition}[Entropy renormalization inequality under uniform scaling: Jensen envelope of \texorpdfstring{$\delta\mapsto m\delta$}{delta -> m delta}]\label{prop:xi-defect-entropy-uniform-scaling-jensen-envelope}
Let
\(
p_j:=\delta_j/(1+\delta_j)\in(0,1)
\), and denote
\(
S:=S_{\mathrm{def}}=\sum_j m_j p_j
\),
\(
\kappa:=\kappa_{\mathrm{ind}}=\sum_j m_j
\).
For any \(m>0\) apply uniform scaling \(\delta_j\mapsto m\delta_j\), and define new entropy
\[
S_m:=\sum_j m_j\,\frac{m\delta_j}{1+m\delta_j}.
\]
Then in \(p\)-coordinates this transformation is given by the Möbius transformation
\begin{equation}\label{eq:xi-defect-entropy-scaling-mobius}
\boxed{\;
\Psi_m(p):=\frac{mp}{1+(m-1)p}\qquad(p\in[0,1])\;}
\end{equation}
with \(S_m=\sum_j m_j\,\Psi_m(p_j)\).
Moreover:
\begin{enumerate}
  \item If \(m>1\), then \(\Psi_m\) is strictly concave on \([0,1]\), so by Jensen's inequality
  \[
  \boxed{\;
  S_m\ \le\ \kappa\,\Psi_m\!\left(\frac{S}{\kappa}\right),\;}
  \]
  with equality if and only if all \(p_j\) (expanded by multiplicity) are equal;
  \item If \(0<m<1\), then \(\Psi_m\) is strictly convex on \([0,1]\), so similarly
  \[
  \boxed{\;
  S_m\ \ge\ \kappa\,\Psi_m\!\left(\frac{S}{\kappa}\right),\;}
  \]
  with equality if and only if all \(p_j\) (expanded by multiplicity) are equal.
\end{enumerate}
Therefore, "the reachable domain of entropy under uniform scale change" is compressed to a single-variable Möbius envelope, and "uniform depth" is the unique saturating configuration.
\end{proposition}

\begin{proof}
From \(p=\delta/(1+\delta)\) solve \(\delta=p/(1-p)\), substitute into \(m\delta\) and simplify to get
\(\frac{m\delta}{1+m\delta}=\Psi_m(p)\), hence \(S_m=\sum_j m_j\Psi_m(p_j)\).
Compute
\[
\Psi_m'(p)=\frac{m}{(1+(m-1)p)^2},
\qquad
\Psi_m''(p)=-\frac{2m(m-1)}{(1+(m-1)p)^3},
\]
so when \(m>1\) we have \(\Psi_m''<0\) (strictly concave), when \(0<m<1\) we have \(\Psi_m''>0\) (strictly convex).
Applying Jensen with weights \(m_j/\kappa\) yields the two inequalities; strict equality conditions are given by the strict convexity/concavity equality condition.
\end{proof}

\begin{proposition}[Variance gap identity of continuous renormalization flow: deviation from uniformity equals recycling intensity]\label{prop:xi-defect-entropy-scaling-variance-gap-identity}
Parameterize uniform scaling \(\delta\mapsto e^{s}\delta\) by logarithmic scale \(s=\log m\), and let
\(
p_j(s):=\Psi_{e^{s}}(p_j(0))
\),
\(
S(s):=\sum_j m_j p_j(s)
\),
\(
\bar p(s):=S(s)/\kappa
\).
Then for all \(s\in\RR\) we have the derivative closed form
\begin{equation}\label{eq:xi-defect-entropy-logistic-variance-gap}
\boxed{\;
\frac{\dd}{\dd s}S(s)
=\sum_j m_j\,p_j(s)\bigl(1-p_j(s)\bigr)
=S(s)\Bigl(1-\frac{S(s)}{\kappa}\Bigr)-\kappa\,\mathrm{Var}_m\!\bigl(p(s)\bigr),\;}
\end{equation}
where the weighted variance is defined as
\[
\mathrm{Var}_m(p):=\frac{1}{\kappa}\sum_j m_j\bigl(p_j-\bar p\bigr)^2\ \ge\ 0.
\]
Therefore
\[
\boxed{\;
\frac{\dd}{\dd s}S(s)\ \le\ S(s)\Bigl(1-\frac{S(s)}{\kappa}\Bigr),\;}
\]
with strict inequality if and only if the depth is non-uniform (\(\mathrm{Var}_m(p(s))>0\)).
This identity quantifies the "recycler intensity" as a computable gap term: uniformization is the unique case achieving the logistic rate upper bound, while any non-uniformity is further erased at the regime level by variance.
\end{proposition}

\begin{proof}
From the closed form \eqref{eq:xi-defect-entropy-scaling-mobius} we can directly verify
\[
\frac{\partial}{\partial s}\Psi_{e^s}(p)
=\frac{e^sp}{(1+(e^s-1)p)^2}
=\Psi_{e^s}(p)\bigl(1-\Psi_{e^s}(p)\bigr),
\]
so \(p_j'(s)=p_j(s)(1-p_j(s))\), thus \(\frac{\dd}{\dd s}S(s)=\sum_j m_j p_j(s)(1-p_j(s))\).
Then use the identity
\[
\sum_j m_j p_j(1-p_j)
\;=\;
\kappa\,\bar p(1-\bar p)\;-\;\kappa\,\mathrm{Var}_m(p),
\]
and substitute \(\bar p=S/\kappa\) to obtain \eqref{eq:xi-defect-entropy-logistic-variance-gap}.
\end{proof}

\begin{corollary}[Certificate complexity hyperbolic exponential law: finite-order moment certificates cover only logarithmic radius]\label{cor:xi-certificate-complexity-hyperbolic-exponential-law}
Let \(a\in\mathbb{D}\) be an interior disk point defect (equivalent to the horizon logarithmic derivative having a pole at \(a\); Theorem \ref{thm:xi-disk-zero-pole-index-reduction}), and denote its Euclidean boundary depth
\(
h:=1-|a|\in(0,1)
\)
as well as the hyperbolic distance from disk center to \(a\)
\[
d:=\log\frac{1+|a|}{1-|a|}.
\]
If some finite-order Toeplitz-PSD certificate (or equivalent finite CMV/Schur certificate) can witness this defect in a stable negative direction at regime level, then its order \(N\) must satisfy
\[
\boxed{\;
N\ \gtrsim\ h^{-1}\;}
\]
(see Theorem \ref{thm:xi-toeplitz-length-vs-boundary-depth} for the coarse-scale necessary lower bound).
From the identity \(h=\frac{2}{e^{d}+1}\) we obtain the equivalent hyperbolic exponential lower bound
\[
\boxed{\;
N\ \gtrsim\ e^{d}\;}
\]
thus given certificate order \(N\), the hyperbolic distance it can stably cover is at most of order \(d\lesssim \log N\).
Therefore, the coverage radius of finite-order moment certificates grows only as \(\log N\) in hyperbolic geometry, while certificate resources inflate exponentially with hyperbolic depth.
\end{corollary}

\begin{proof}
From \(d=\log\frac{1+|a|}{1-|a|}\) solve
\(
|a|=\frac{e^{d}-1}{e^{d}+1}
\), hence
\(
h=1-|a|=\frac{2}{e^{d}+1}
\).
Substituting Theorem \ref{thm:xi-toeplitz-length-vs-boundary-depth}'s coarse-scale necessary lower bound \(N\gtrsim h^{-1}\) yields \(N\gtrsim e^{d}\);
the reverse conclusion follows directly from \(d=\log\frac{2-h}{h}\le \log\frac{2}{h}\) and \(h\gtrsim N^{-1}\).
\end{proof}

\begin{proposition}[Hyperbolic height linearization: exponential complementarity and semigroup law of entropy and top edge length]\label{prop:xi-defect-entropy-hyperbolic-height-semigroup}
Introduce hyperbolic height variable
\[
s_j:=\log(1+\delta_j)\qquad(1+\delta_j=e^{s_j}).
\]
Then the entropy contribution of each atom satisfies
\[
\frac{\delta_j}{1+\delta_j}=1-e^{-s_j},
\]
thus defect entropy has exponential linearization expression
\begin{equation}\label{eq:xi-defect-entropy-hyperbolic-height-linearization}
\boxed{\;
S_{\mathrm{def}}=\sum_{j=1}^{\kappa}m_j\bigl(1-e^{-s_j}\bigr)\;}
\end{equation}
and the "top edge total length" (i.e., weighted total length of unit-width horocycle segments at height \(y=1+\delta_j\))
\[
L_{\mathrm{top}}
:=\sum_{j=1}^{\kappa}\frac{m_j}{1+\delta_j}
=\sum_{j=1}^{\kappa}m_j e^{-s_j}
=\kappa_{\mathrm{ind}}-S_{\mathrm{def}}.
\]
In particular, under uniform vertical translation \(s_j\mapsto s_j+\tau\) (\(\tau\ge 0\)) for all atoms,
\[
L_{\mathrm{top}}\mapsto e^{-\tau}L_{\mathrm{top}},
\qquad
S_{\mathrm{def}}\mapsto \kappa_{\mathrm{ind}}-e^{-\tau}\bigl(\kappa_{\mathrm{ind}}-S_{\mathrm{def}}\bigr),
\]
thus \(\tau\) generates a one-parameter semigroup, and gives the strict differential
\[
\dd S_{\mathrm{def}}=\sum_{j=1}^{\kappa} m_j e^{-s_j}\,\dd s_j.
\]
\end{proposition}

\begin{proof}
From \(1+\delta_j=e^{s_j}\) obtain
\(
\frac{\delta_j}{1+\delta_j}=1-\frac{1}{1+\delta_j}=1-e^{-s_j}
\), substituting into \eqref{eq:xi-defect-entropy-closed-form} yields \eqref{eq:xi-defect-entropy-hyperbolic-height-linearization}.
The remaining formulas follow term-wise from algebraic identities
\(
L_{\mathrm{top}}=\sum_j m_j e^{-s_j}
\)
and
\(
S_{\mathrm{def}}=\kappa_{\mathrm{ind}}-L_{\mathrm{top}}
\);
the differential follows from term-wise differentiation with respect to \(s_j\).
\end{proof}

\begin{corollary}[Exponential law of hyperbolic height budget: optimal upper bound of defect entropy when fixing \texorpdfstring{$\sum m_j\log(1+\delta_j)$}{sum m log(1+delta)}]\label{cor:xi-defect-entropy-height-budget-exponential-law}
Under the setting of Proposition \ref{prop:xi-defect-entropy-hyperbolic-height-semigroup}, let
\[
S_{\mathrm{ht}}:=\sum_{j=1}^{\kappa}m_j\log(1+\delta_j)=\sum_{j=1}^{\kappa}m_j s_j,
\qquad
\kappa_{\mathrm{ind}}:=\sum_{j=1}^{\kappa}m_j.
\]
Then defect entropy under fixed height budget \(S_{\mathrm{ht}}\) satisfies sharp upper bound
\begin{equation}\label{eq:xi-defect-entropy-height-budget-exp-law}
\boxed{\;
S_{\mathrm{def}}
\ \le\
\kappa_{\mathrm{ind}}\Bigl(1-e^{-S_{\mathrm{ht}}/\kappa_{\mathrm{ind}}}\Bigr).\;}
\end{equation}
Equality holds if and only if all \(s_j\) (counting by multiplicity) are equal (equivalent to all \(\delta_j\) equal).
This inequality provides the exponential saturation law of defect entropy under "area/height budget" constraint, and uniquely identifies the optimal state as uniform depth configuration.
\end{corollary}
\begin{proof}
From \eqref{eq:xi-defect-entropy-hyperbolic-height-linearization},
\(
S_{\mathrm{def}}=\sum_j m_j g(s_j)
\)
where \(g(s)=1-e^{-s}\) is strictly concave on \([0,\infty)\).
Apply Jensen's inequality with weights \(w_j:=m_j/\kappa_{\mathrm{ind}}\) (\(\sum_j w_j=1\)):
\[
\frac{S_{\mathrm{def}}}{\kappa_{\mathrm{ind}}}
=\sum_j w_j g(s_j)
\le g\!\Bigl(\sum_j w_j s_j\Bigr)
=1-\exp\!\Bigl(-\frac{S_{\mathrm{ht}}}{\kappa_{\mathrm{ind}}}\Bigr),
\]
yielding \eqref{eq:xi-defect-entropy-height-budget-exp-law}.
Strict concavity provides the equality condition: if and only if \(s_j\) (counting by multiplicity) are all equal.
\end{proof}

\begin{corollary}[Detectability function: normalized peak amplitude is strict quadratic image of entropy contribution]\label{cor:xi-detectability-amplitude-is-quadratic-image}
For single-atom profile kernel
\(
H_{(\gamma,\delta,m)}(x)=m\,\frac{4\delta}{(x-\gamma)^2+(1+\delta)^2}
\)
define normalized peak amplitude
\[
a:=\frac{H_{(\gamma,\delta,m)}(\gamma)}{m}=\frac{4\delta}{(1+\delta)^2}\in(0,1].
\]
Let entropy contribution
\(
p:=\frac{\delta}{1+\delta}\in(0,1)
\), then there holds strict algebraic identity
\begin{equation}\label{eq:xi-amplitude-as-quadratic-image-of-entropy}
\boxed{\;
a=4p(1-p)\; }.
\end{equation}
Therefore peak amplitude is completely determined by entropy contribution, and attains maximum \(a_{\max}=1\) at \(p=\tfrac12\) (equivalent to \(\delta=1\)).
Conversely, given \(a\in(0,1]\), entropy contribution has only two possible branches
\begin{equation}\label{eq:xi-entropy-from-amplitude-two-branches}
\boxed{\;
p=\frac12\bigl(1\pm\sqrt{1-a}\bigr)\; }.
\end{equation}
This two-branch bifurcation corresponds to the intrinsic binary indeterminacy of "amplitude alone cannot determine whether depth lies at \(\delta<1\) or \(\delta>1\)"; disambiguation requires analytic information such as phase/pole direction (see the branch discussion in Proposition \ref{prop:xi-lorentzian-rigidity-amplitude-halfwidth}).
\end{corollary}

\begin{proof}
From \(p=\delta/(1+\delta)\) obtain \(1-p=1/(1+\delta)\), so
\(
4p(1-p)=4\delta/(1+\delta)^2=a
\), yielding \eqref{eq:xi-amplitude-as-quadratic-image-of-entropy}.
For \eqref{eq:xi-entropy-from-amplitude-two-branches}, view \(a=4p(1-p)\) as a quadratic equation in \(p\) and solve; maximum and branches follow from downward-opening parabola and symmetry.
\end{proof}

\begin{proposition}[Complete monotonicity of coarse-grained entropy curve: scale trajectory as Stieltjes fingerprint]\label{prop:xi-defect-entropy-scale-fingerprint-stieltjes}
Introduce scale parameter \(t\ge 0\) Poisson coarse-graining, rewriting the half-width of each atom kernel from \(1+\delta_j\) to \(t+1+\delta_j\), and define scale entropy
\begin{equation}\label{eq:xi-defect-entropy-scale-curve}
\boxed{\;
S_{\mathrm{def}}(t):=\sum_{j=1}^{\kappa}m_j\,\frac{\delta_j}{t+1+\delta_j}\;}
\end{equation}
(thus \(S_{\mathrm{def}}(0)=S_{\mathrm{def}}\) and \(S_{\mathrm{def}}(t)\downarrow 0\) as \(t\to\infty\)).
Then \(t\mapsto S_{\mathrm{def}}(t)\) is a completely monotone function: for all integers \(n\ge 0\)
\begin{equation}\label{eq:xi-defect-entropy-scale-complete-monotone}
\boxed{\;
(-1)^n\frac{d^n}{dt^n}S_{\mathrm{def}}(t)
=n!\sum_{j=1}^{\kappa}m_j\,\frac{\delta_j}{(t+1+\delta_j)^{n+1}}
\ge 0\; }.
\end{equation}
Therefore \(S_{\mathrm{def}}(t)\) is an auditable Stieltjes-type fingerprint curve: under finite defect caliber,
its pole set and residues in the rational extension to the complex plane completely determine \(\{(\delta_j,m_j)\}\) (up to permutation and merging same-depth atoms),
while the height parameters \(\{\gamma_j\}\) do not appear in this scalar scale trajectory.
\end{proposition}

\begin{proof}
Direct differentiation of each term \(\frac{\delta_j}{t+1+\delta_j}\) gives
\(
\frac{d^n}{dt^n}\frac{\delta_j}{t+1+\delta_j}
=(-1)^n n!\,\frac{\delta_j}{(t+1+\delta_j)^{n+1}}
\), weighted summation yields \eqref{eq:xi-defect-entropy-scale-complete-monotone}, thus complete monotonicity holds.
\par
Under further assumption of finite defects, \(S_{\mathrm{def}}(t)\) is a rational function
with poles at \(t=-1-\delta_j\) and residues \(m_j\delta_j\).
Therefore pole locations recover \(\{\delta_j\}\), and residues divided by \(\delta_j\) recover \(\{m_j\}\) (obtaining total multiplicity when merging same-\(\delta\) cases).
Height parameters \(\gamma_j\) do not appear in \eqref{eq:xi-defect-entropy-scale-curve}, hence cannot be identified by this scalar trajectory.
\end{proof}
\begin{proposition}[Riccati pincer of scale entropy flow: inevitable recycling rate from coarse-graining sealed by \texorpdfstring{$(\kappa_{\mathrm{ind}},\delta_{\min},\delta_{\max})$}{(kappa,delta_min,delta_max)}]\label{prop:xi-defect-entropy-riccati-sandwich}
Under the setting of Proposition \ref{prop:xi-defect-entropy-scale-fingerprint-stieltjes}, denote
\[
\kappa_{\mathrm{ind}}:=\sum_{j=1}^{\kappa}m_j,
\qquad
\delta_{\min}:=\min_{1\le j\le\kappa}\delta_j,
\qquad
\delta_{\max}:=\max_{1\le j\le\kappa}\delta_j.
\]
Then \(S_{\mathrm{def}}(t)\) is differentiable and satisfies exact derivative formula
\[
S_{\mathrm{def}}'(t)=-\sum_{j=1}^{\kappa}m_j\,\frac{\delta_j}{(t+1+\delta_j)^2}.
\]
Moreover, for all \(t\ge 0\) there holds two-sided Riccati-type pincer
\begin{equation}\label{eq:xi-defect-entropy-riccati-sandwich}
\boxed{\;
-\frac{1}{\kappa_{\mathrm{ind}}\delta_{\min}}\,S_{\mathrm{def}}(t)^2
\ \le\
S_{\mathrm{def}}'(t)
\ \le\
-\frac{1}{\kappa_{\mathrm{ind}}\delta_{\max}}\,S_{\mathrm{def}}(t)^2.\;}
\end{equation}
Equivalently,
\begin{equation}\label{eq:xi-defect-entropy-inv-linear-bracket}
\boxed{\;
\frac{1}{S_{\mathrm{def}}(t)}-\frac{1}{S_{\mathrm{def}}(0)}
\ \in\
\Bigl[\frac{t}{\kappa_{\mathrm{ind}}\delta_{\max}},\ \frac{t}{\kappa_{\mathrm{ind}}\delta_{\min}}\Bigr].\;}
\end{equation}
Therefore growth of scale \(t\) forces \(S_{\mathrm{def}}(t)\) to decay into recycling state with certifiable \(t^{-1}\) order; this decay rate at regime level depends only on defect index and depth interval, decoupling from height parameters \(\{\gamma_j\}\).
\end{proposition}
\begin{proof}
Let
\(
B_j(t):=m_j\frac{\delta_j}{(t+1+\delta_j)^2}>0
\).
Then
\[
-S_{\mathrm{def}}'(t)=\sum_{j=1}^{\kappa}B_j(t),
\qquad
S_{\mathrm{def}}(t)=\sum_{j=1}^{\kappa}(t+1+\delta_j)\,B_j(t),
\]
thus the ratio
\[
\frac{S_{\mathrm{def}}(t)}{-S_{\mathrm{def}}'(t)}
=\frac{\sum_j (t+1+\delta_j)B_j(t)}{\sum_j B_j(t)}
\]
is a weighted average of \(\{t+1+\delta_j\}\), hence
\begin{equation}\label{eq:xi-defect-entropy-weighted-average-bracket}
t+1+\delta_{\min}
\ \le\
\frac{S_{\mathrm{def}}(t)}{-S_{\mathrm{def}}'(t)}
\ \le\
t+1+\delta_{\max}.
\end{equation}
On the other hand, from monotonicity \(\delta_{\min}\le \delta_j\le \delta_{\max}\) and increasing function \(\delta\mapsto \delta/(t+1+\delta)\), obtain
\[
\kappa_{\mathrm{ind}}\frac{\delta_{\min}}{t+1+\delta_{\min}}
\ \le\
S_{\mathrm{def}}(t)
\ \le\
\kappa_{\mathrm{ind}}\frac{\delta_{\max}}{t+1+\delta_{\max}}.
\]
Multiplying these with left and right sides of \eqref{eq:xi-defect-entropy-weighted-average-bracket} respectively gives
\[
\kappa_{\mathrm{ind}}\delta_{\min}
\ \le\
\frac{S_{\mathrm{def}}(t)^2}{-S_{\mathrm{def}}'(t)}
\ \le\
\kappa_{\mathrm{ind}}\delta_{\max},
\]
i.e.,
\(
\frac{1}{\kappa_{\mathrm{ind}}\delta_{\max}}
\le
(-S_{\mathrm{def}}'(t))/S_{\mathrm{def}}(t)^2
\le
\frac{1}{\kappa_{\mathrm{ind}}\delta_{\min}}
\), equivalent to \eqref{eq:xi-defect-entropy-riccati-sandwich}.
For the identity
\(
\frac{d}{dt}\frac{1}{S_{\mathrm{def}}(t)}=-\frac{S_{\mathrm{def}}'(t)}{S_{\mathrm{def}}(t)^2}
\),
integration yields \eqref{eq:xi-defect-entropy-inv-linear-bracket}.
\end{proof}
