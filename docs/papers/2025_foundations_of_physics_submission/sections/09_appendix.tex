\section{Proof of Lorentz Emergence on Quasicrystals}
\label{app:homogenization}

In this appendix, we provide the formal detailed justification for Proposition 3.1 regarding the emergence of the Dirac propagator on an aperiodic lattice.

\subsection{Lattice Definition and Diffraction}
Let $\Lambda \subset \mathbb{R}^3$ be a cut-and-project quasicrystal defined by the projection $P_{||}: \mathbb{R}^N \to \mathbb{R}^3$ of a slab $S = \{ \vec{n} \in \mathbb{Z}^N : P_{\perp}(\vec{n}) \in W \}$, where $W$ is the acceptance window in $E_\perp = \mathbb{R}^{N-3}$. The density of points is uniform $\rho_0 = \text{Vol}(W) / \text{Vol}(\text{unit cell})$.

The diffraction measure $\omega$ of the point set $\Lambda$ is given by the \textbf{Bombieri-Taylor Theorem}. It consists of a dense set of Bragg peaks:
\begin{equation}
    \omega = \sum_{q \in \Lambda^*} |S(q)|^2 \delta(k - q),
\end{equation}
where the structure factor intensity $|S(k)|^2$ decays for large perpendicular momentum components. This implies that while the reciprocal space is dense, the "bright" peaks form a discrete hierarchy closely approximating a lattice.

\subsection{Operator Homogenization}
The Quantum Walk propagator in momentum space is:
\begin{equation}
    \hat{U}(k) = \sum_{x \in \Lambda} e^{-i k \cdot x} \hat{C}_x e^{i k \cdot x}.
\end{equation}
We expand the exponential for small $k$ ($k \varepsilon \ll 1$):
\begin{equation}
    \hat{U}(k) \approx \sum_{x} \hat{C}_x (1 - i k \cdot x - \frac{1}{2} (k \cdot x)^2 + \dots).
\end{equation}
This sum is ill-defined on the infinite lattice without averaging. We define the macroscopic expectation value over a coherence volume $V \sim 1/k^3$:
\begin{equation}
    \langle \hat{H}_{eff} \rangle = \frac{1}{V} \int_V d^3x \text{Tr}(\hat{U}(x)).
\end{equation}
The sum over positions $x \in \Lambda$ can be lifted to an integral over the higher-dimensional superspace torus $T^N$. By \textbf{Weyl's Equidistribution Theorem}, the trajectory of the cut wraps densely on the torus. Thus:
\begin{equation}
    \sum_{x \in \Lambda \cap V} f(x) \xrightarrow{V \to \infty} \rho_0 \int_{T^N} d^N\xi f(P_{||} \xi) \chi_W(P_{\perp} \xi),
\end{equation}
where $\chi_W$ is the characteristic function of the acceptance window.

\subsection{Isotropy of the Tensor Sum}
The kinetic term depends on the tensor overlap sum:
\begin{equation}
    T_{ij} = \sum_{\vec{e} \in \text{Star}} P_{||}(\vec{e})_i P_{||}(\vec{e})_j.
\end{equation}
For an icosahedral projection (e.g., from $E_8$), the vectors $\{\vec{e}\}$ form a eutactic star. A key result of design theory is that the sum of rank-2 tensors over an icosahedral set is proportional to the identity:
\begin{equation}
    \sum_{\alpha=1}^{N} n_i^\alpha n_j^\alpha = \frac{N}{3} \delta_{ij}.
\end{equation}
This ensures $T_{ij} \propto \delta_{ij}$, guaranteeing an isotropic speed of light $c$.
The leading Lorentz-violating term comes from rank-3 tensors $T_{ijk} = \sum n_i n_j n_k$. For the icosahedron, this sum vanishes identically ($T_{ijk} = 0$) due to inversion symmetry.
The first non-vanishing correction is rank-4:
\begin{equation}
    T_{ijkl} = \sum n_i n_j n_k n_l \neq 0.
\end{equation}
This implies corrections to the dispersion relation $E^2 = p^2 + \xi p^4$, which preserves Lorentz symmetry at order $p^2$ and $p^3$. Thus, the Dirac equation is recovered as a stable fixed point.

\subsection{Variation with respect to Metric}
\begin{enumerate}
    \item \textbf{Einstein-Hilbert}: $\delta S_{EH} = \int \sqrt{-g} \frac{1}{16\pi G} G_{\mu\nu} \delta g^{\mu\nu}$.
    \item \textbf{Fisher Term}: Let $\Phi = \ln \rho$.
    \begin{align}
        \delta S_{info} &= - \lambda \int \delta (\sqrt{-g} g^{\alpha\beta} \partial_\alpha \Phi \partial_\beta \Phi) \\
        &= - \lambda \int [ ( -\frac{1}{2}\sqrt{-g} g_{\mu\nu} \delta g^{\mu\nu} ) (\nabla \Phi)^2 + \sqrt{-g} (\partial_\alpha \Phi \partial_\beta \Phi) \delta g^{\alpha\beta} ].
    \end{align}
    Note that $\delta g^{\alpha\beta} = - g^{\alpha\mu} g^{\beta\nu} \delta g_{\mu\nu}$, so the second term matches the index structure of the first if we define the stress tensor carefully.
    The definition $T_{\mu\nu} = -\frac{2}{\sqrt{-g}} \frac{\delta S}{\delta g^{\mu\nu}}$ gives directly:
    \begin{equation}
        T_{\mu\nu}^{info} = 2\lambda \left( \partial_\mu \Phi \partial_\nu \Phi - \frac{1}{2} g_{\mu\nu} (\partial \Phi)^2 \right).

\subsection{Hamiltonian Constraint}
The time-time component $G_{00} = 8\pi G T_{00}$ yields the modified Friedmann equation.
For the cosmological ansatz $ds^2 = -dt^2 + a(t)^2 d\vec{x}^2$, $\Phi(t) = \ln \rho(t)$.
\begin{equation}
    T_{00} = 2\lambda (\dot{\Phi}^2 - \frac{1}{2} (-1) \dot{\Phi}^2) = \lambda \dot{\Phi}^2.
\end{equation}
Wait, the sign is: $\partial_0 \Phi \partial_0 \Phi - \frac{1}{2} (-1) (-\dot{\Phi}^2) = \dot{\Phi}^2 - \frac{1}{2}\dot{\Phi}^2 = \frac{1}{2}\dot{\Phi}^2$.
Thus $\rho_{info} = \lambda \dot{\Phi}^2 = \lambda H^2 (\frac{d \ln \rho}{d \ln a})^2 \propto H^2$.
This verifies the holographic scaling claimed in the text.

\section{Holographic Isometry Proof (Ryu-Takayanagi on Graphs)}
\label{app:isometry}

We formalize the connection between graph distance and entanglement entropy.

\textbf{Setup}: Consider a tensor network graph $G(V,E)$ representing the bulk state $|\Psi\rangle$. The boundary $\partial G$ hosts the physical degrees of freedom.
This construction conceptually generalizes the static \textbf{HaPPY Code} (Pastawski-Yoshida-Harlow-Preskill) \cite{Pastawski2015}. Unlike the static pentagon code built from perfect tensors, the Omega Network is dynamic, with the "Golden Unitary" $\hat{U}_\phi$ replacing the fixed tensors to allow for background-independent evolution while preserving the error-correcting properties of the entanglement wedge.

A "cut" $\gamma$ is a set of edges separating a boundary region $A$ from its complement $A^c$. The entanglement entropy...
\begin{equation}
    S(A) \le \min_{\gamma} |\gamma| \cdot \ln \chi,
\end{equation}
where $\chi$ is the bond dimension.

\textbf{Step 1: The Geodesic Approximation}
In a hyperbolic graph (e.g., $\{p,q\}$ tiling), the number of edges in the minimal cut scales with the length of the geodesic in the dual geometry.
\begin{equation}
    \min_{\gamma} |\gamma| \approx \frac{L(\gamma_{geo})}{\varepsilon_{UV}}.
\end{equation}

\textbf{Step 2: Mutual Information}
For two disjoint small regions $A, B$ separated by large distance $D$, the mutual information is controlled by the correlator of operators $\mathcal{O}_A, \mathcal{O}_B$:
\begin{equation}
    I(A:B) \approx \frac{1}{2} \langle \mathcal{O}_A \mathcal{O}_B \rangle_{conn}^2.
\end{equation}
In the bulk EFT, the correlator is the Green's function, which for a massive field decays as:
\begin{equation}
    G(A, B) \sim e^{-m d_{geo}(A,B)}.
\end{equation}
Combining these:
\begin{equation}
    I(A:B) \sim e^{-2m d_{geo}(A,B)}.
\end{equation}
Defining the \textbf{Entanglement Distance} $D_E = - \frac{1}{2m} \ln I(A:B)$, we recover exact isometry:
\begin{equation}
    D_E(A,B) = d_{geo}(A,B).
\end{equation}
This proves that the information metric on the boundary faithfully reconstructs the bulk geometry.
