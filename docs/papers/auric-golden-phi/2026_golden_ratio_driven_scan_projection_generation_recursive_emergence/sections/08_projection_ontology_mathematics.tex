\subsection{定位与对照：在既有骨架上提升到统一语法}\label{subsec:pom-position}
本节把“投影—本体—稳定读出”的核心器件在同一套形式系统中闭合：以现有折叠 $\Fold_m/\Fold_\infty$ 与序列级因子 $\Phi_m$（第 \ref{sec:folding} 节）、算子代数上的条件期望 $E_m$（附录 \ref{app:op-algebra}）、在线归一化延迟（定理 \ref{thm:online-delay-fold}）以及时间—空间交换图 $Z=E\circ M\circ T$（命题 \ref{prop:time-space-commuting-diagram}）为底座，引入\textbf{三条先验纤维}并把\textbf{加减乘除商余}统一为“同一本体演化 $U$ + 两类投影 $(P_Z,P_{\le})$ + 线性化深度 $D_U$”的投影词语法。

\Fold_\infty(a)=Z(\mathrm{Val}(a))

\subsection{三条先验纤维}\label{subsec:pom-fibers}
\begin{definition}[尺度纤维 $\mathcal{S}_\varphi$]\label{def:pom-scale-fiber}
取无理数 $\alpha\in(0,1)$ 的连分数 $\alpha=[0;a_1,a_2,\dots]$，Ostrowski 表示给出
\[
N=\sum_{n\ge 1} d_n q_{n-1},
\]
其中 $q_n$ 为收敛分母，数字 $(d_n)$ 满足合法约束。取 $\alpha=\varphi^{-1}$ 时，退化为 Fibonacci 权重与 Zeckendorf 合法语法（禁止相邻 $11$）。定义尺度纤维为该语法下的稳定地址空间（有限分辨率为 $X_m$，极限为 $X_\infty$），其折叠归一化由 $\Fold_m/\Fold_\infty$ 给出（第 \ref{sec:folding} 节）。
\end{definition}

\begin{definition}[相位纤维 $\mathcal{T}$]\label{def:pom-phase-fiber}
设相空间为紧阿贝尔群（例如 $\TT^n$），其角色基为
\[
\chi_k(x)=e^{2\pi i\langle k,x\rangle},\quad k\in\ZZ^n.
\]
相位纤维 $\mathcal{T}$ 指由该角色结构诱导的谱坐标层，用于把可加观测量提升为可乘权重与卷积对角化（见第 \ref{sec:zeta-finite-part} 节与第 \ref{sec:high-dimensional-cut-project} 节）。
\end{definition}

\begin{definition}[素数寄存器纤维 $\mathcal{P}$]\label{def:pom-prime-fiber}
令 $\mathbb{P}$ 为素数集合，定义有限支撑指数向量空间
\[
\mathcal{P}:=\mathbb{N}^{(\mathbb{P})}=\{\mathbf{r}:\mathbb{P}\to\mathbb{N}\mid \mathbf{r}(p)=0\ \text{除有限多个 }p\}.
\]
定义值映射
\[
N(\mathbf{r})=\prod_{p\in\mathbb{P}} p^{\mathbf{r}(p)}.
\]
由算术基本定理（参见 \cite{Apostol1976}），$N(\cdot)$ 给出 $\mathcal{P}$ 与 $\mathbb{N}_{\ge 1}$ 的同构，从而把乘法线性化为坐标加法。由于每个状态仅激活有限多素数轴，该纤维满足“无限坐标系 + 局部触发”的可实现口径。
\end{definition}

\begin{remark}[两套自由交换幺半群坐标的并存]\label{rem:pom-two-monoids}
Zeckendorf 坐标与素数指数坐标分别线性化加法与乘法；同一整数对象拥有两套自由交换幺半群坐标（见注 \ref{rem:two-coordinate-monoids}）。本节所加的素数纤维即是该坐标系统在本体层的显式化。
\end{remark}

\subsection{本体态空间与局部演化}\label{subsec:pom-state-evolution}
\begin{definition}[分辨率态空间]\label{def:pom-state-space}
对分辨率 $n$，定义本体态空间
\[
\mathcal{H}_n:=\mathcal{X}_n\times \mathcal{M}_n,
\]
其中 $\mathcal{X}_n$ 为外表坐标（如 $X_n$ 或 $X_\infty$ 的截断），$\mathcal{M}_n$ 为隐纤维（可包含 $\mathcal{S}_\varphi,\mathcal{T},\mathcal{P}$ 的局部激活块、同步核状态与余量等）。
\end{definition}

\begin{axiom}[局部性与有限传播]\label{ax:pom-locality}
存在本体一步演化算子
\[
U:\mathcal{H}_n\to\mathcal{H}_n
\]
使得每一步仅依赖有限窗口并只更新有限半径的局部块。该公理与第 \ref{sec:A-kernel-compare-main} 节的局部核/在线归一化口径一致。
\end{axiom}

\subsection{两类投影：规范投影与序投影}\label{subsec:pom-projections}
\begin{definition}[规范投影 $P_Z$]\label{def:pom-proj-norm}
在窗口层，$P_Z$ 定义为 $\Fold_m:\Omega_m\to X_m$ 的规范化投影（定义 \ref{def:fold-word}），其实现为值保持的局部重写系统并满足终止与合流（定理 \ref{thm:fold-suite}）。在算子代数层，$P_Z$ 由折叠诱导的条件期望
\[
E_m:\mathcal{A}_m\to\mathcal{B}_m
\]
实现（命题 \ref{prop:fold-conditional-expectation}），因此 $P_Z$ 为幂等投影并与归约顺序无关。
\end{definition}

\begin{definition}[序投影 $P_{\le}$]\label{def:pom-proj-order}
序投影 $P_{\le}$ 指把本体态映射到有序约束的读出证书（如符号、比较结果或区间约束），其目标是把“$0\le r<b$”一类的全局序关系编码为可观测输出。与 $P_Z$ 不同，$P_{\le}$ 一般不是环同态，也不保证局部可并行化；其读出通常需要跨窗口聚合，因此是不可并行增长的主要来源。
\end{definition}

\subsection{投影词语法与成本模型}\label{subsec:pom-words}
\begin{definition}[投影词与半群]\label{def:pom-projection-word}
令投影字母表 $\{Z,\le\}$ 分别代表一次 $P_Z$ 与一次 $P_{\le}$ 读出。对任意指令/微码 $I$，定义投影词
\[
W(I)=(k_Z(I),k_{\le}(I)),
\]
其中 $k_Z,k_{\le}$ 为两类投影的调用次数。令
\[
\mathcal{W}=\langle Z,\le\rangle^*/\sim,\qquad Z^2\sim Z
\]
为投影词半群（幂等律来自 $P_Z\circ P_Z=P_Z$）；其他等价关系可按具体实现的因果锥/局部可交换性附加。
\end{definition}

\begin{definition}[成本模型]\label{def:pom-cost}
令 $\delta_Z$ 为一次规范投影的基准读出深度（例如在线归一化的固定延迟，见定理 \ref{thm:online-delay-fold}），令 $T_{\le}(n)$ 为一次序投影的聚合深度，令 $D_U(n)$ 为在本体演化中完成线性化所需的深度。则定义
\[
T_I(n)=k_Z(I)\,\delta_Z+k_{\le}(I)\,T_{\le}(n)+D_U(n).
\]
该模型把投影读出与演化深度分离，用于比较不同实现把“混合”置于时间还是空间的策略差异。
\end{definition}

\subsection{统一算术：同一演化，不同读出}\label{subsec:pom-unified-arith}
\paragraph{加法}由“逐位叠加 $\Rightarrow$ 规范投影”给出：
\[
a\oplus b=\Fold_\infty(a+b),
\]
其投影词为 $W=(1,0)$（推论 \ref{cor:add-as-fold}）。

\paragraph{乘法}同样可视为“线性化 $\Rightarrow$ 规范投影”，其投影词仍为 $W=(1,0)$；差异体现在 $D_U(n)$ 与所选纤维：相位纤维路线把卷积对角化为点wise 乘，素数纤维路线把乘法线性化为指数坐标的加法（定义 \ref{def:pom-prime-fiber}）。该分解不改变本体原语，仅改变线性化深度与空间开销的分配。

\paragraph{减法}有两条读出路径：
（i）先以 $P_{\le}$ 判定符号/大小，再执行归一化；其投影词 $W=(1,1)$。
（ii）在冗余字母表中把符号内化到 $P_Z$ 的更强归一化中，避免显式 $P_{\le}$，从而 $W=(1,0)$ 但 $\delta_Z$ 增大。这一分解把“序信息”转移到规范投影的本体核中。

\paragraph{除法/商余}满足
\[
a=bq+r,\quad 0\le r<b,
\]
因此 $P_{\le}$ 是不可约读出步骤。由注 \ref{rem:division-as-selection} 的“解方程 + 规范形选择”口径可得：长除式对应 $k_{\le}$ 的线性增长；倒数/迭代式把多次乘法吸收到 $D_U(n)$ 中并以常数次 $P_{\le}$ 校正。两者的差异是投影词的不同实现，而非本体原语的差异。

\subsection{折叠指标与投影熵率}\label{subsec:pom-projection-entropy}
\begin{definition}[折叠指标与投影熵率]\label{def:pom-fold-index}
令
\[
I_m:=\frac{|\Omega_m|}{|X_m|}=\frac{2^m}{F_{m+2}},
\qquad
S_Z(m):=\log I_m,
\qquad
h_Z:=\lim_{m\to\infty}\frac{S_Z(m)}{m}.
\]
$I_m$ 是折叠诱导的平均多重度指标，$S_Z(m)$ 与 $h_Z$ 分别刻画分辨率 $m$ 与单位分辨率的投影熵率。
\end{definition}

\begin{proposition}[折叠指标的渐近]\label{prop:pom-projection-entropy}
有
\[
S_Z(m)=m\log\!\left(\frac{2}{\varphi}\right)+O(1),
\qquad
h_Z=\log\!\left(\frac{2}{\varphi}\right).
\]
\end{proposition}
\begin{proof}
由附录 \ref{app:fold-multiplicity} 的渐近式
\(
I_m\sim C(2/\varphi)^m
\)
直接得到。证毕。
\end{proof}

\begin{corollary}[分辨率不足的空间门槛]\label{cor:pom-resolution-threshold}
若希望“每个稳定类型平均只对应不超过 $R$ 个微态”，则必须满足
\[
m\gtrsim \frac{\log R}{\log(2/\varphi)}.
\]
因此提高可区分性必然以分辨率/空间增长为代价。
\end{corollary}

\subsection{逆极限自然性与纤维不变量判别}\label{subsec:pom-new-results}
\begin{theorem}[逆极限自然性]\label{thm:pom-inverse-limit}
设对每个分辨率 $m$ 定义稳定层运算 $f_m:X_m\to X_m$，并满足对一切 $m_2\ge m_1$ 的截断投影
\[
\pi_{m_2\to m_1}\circ f_{m_2}=f_{m_1}\circ \pi_{m_2\to m_1}.
\]
则存在唯一的极限运算 $f_\infty:X_\infty\to X_\infty$ 使得对所有 $m$ 有 $\pi_m\circ f_\infty=f_m\circ \pi_m$。
\end{theorem}
\begin{proof}
由定理 \ref{thm:inverse-limit-golden}，$X_\infty\cong\varprojlim X_m$。相容性等式正是逆系统上的自然变换条件，因此由逆极限的泛性质得到唯一的 $f_\infty$。证毕。
\end{proof}

\begin{proposition}[纤维不变量判别]\label{prop:pom-fiber-invariance}
给定 $g:\Omega_m\to\mathbb{R}$，以下条件等价：
\begin{enumerate}
  \item 若 $\Fold_m(a)=\Fold_m(b)$ 则 $g(a)=g(b)$；
  \item 存在 $\bar g:X_m\to\mathbb{R}$ 使得 $g=\bar g\circ\Fold_m$；
  \item $g$（作为可观测量）属于折叠不变量子代数 $\mathcal{B}_m$（命题 \ref{prop:fold-invariant-subalgebra}）。
\end{enumerate}
\end{proposition}
\begin{proof}
$(1)\Rightarrow(2)$：定义 $\bar g(x)=g(a)$ 对任意 $a\in\Fold_m^{-1}(x)$，由 (1) 保证良定义。$(2)\Rightarrow(3)$：$g$ 在每条纤维上常值，故对应于 $\mathcal{B}_m$ 的坐标表示。$(3)\Rightarrow(1)$：$\mathcal{B}_m$ 的元素对纤维常值（命题 \ref{prop:fold-invariant-subalgebra}）。证毕。
\end{proof}

\begin{remark}[时间—空间对偶接口与素数影子]\label{rem:pom-time-space}
命题 \ref{prop:time-space-commuting-diagram} 给出的严格交换图把时间侧 $\Tr(T^n)$ 与空间侧 primitive 轨道谱对齐，从而为“时间—空间正交对偶”提供可审计接口。另一方面，有限核的 primitive 轨道谱与整数素数并不等同（见注 \ref{rem:prime-shadow-vs-primes}）：前者是核动力学的内禀素性，后者需要额外的素数索引算子族。素数寄存器纤维 $\mathcal{P}$ 正是对这一额外索引结构的本体化补充。
\end{remark}

\subsection{谱隙驱动的时间尺度与读出稳定}\label{subsec:pom-spectral-gap}
\begin{corollary}[谱隙控制的读出时间尺度]\label{cor:pom-spectral-gap-time}
设 $M_K(u)$ 原始，主特征值为 $\lambda_1(u)$，其余特征值最大模为 $\lambda_2(u)$。由定理 \ref{thm:kernel-weighted-prime-orbit}，
\[
B_n(u)=\frac{\lambda_1(u)^n}{n}+O\!\left(\frac{|\lambda_2(u)|^n}{n}\right),
\]
从而相对误差满足
\[
\left|\frac{nB_n(u)}{\lambda_1(u)^n}-1\right|
\le C\left(\frac{|\lambda_2(u)|}{\lambda_1(u)}\right)^n.
\]
给定目标精度 $\varepsilon\in(0,1)$，满足
\[
n\ \ge\ \frac{\log(C/\varepsilon)}{\log\!\bigl(\lambda_1(u)/|\lambda_2(u)|\bigr)}
\]
即可保证相对误差不超过 $\varepsilon$。若进一步满足核版 RH/GRH 条件（定义 \ref{def:kernel-rh}），则可用 $\lambda_1(u)^{1/2}$ 替代 $|\lambda_2(u)|$，从而得到平方根级衰减尺度（推论 \ref{cor:kernel-rh-sqrt-error}）。
\end{corollary}

\subsection{素数影子到素数寄存器的桥接}\label{subsec:pom-prime-bridge}
\paragraph{有限轴束的扭曲与分解}
对有限群标签 $G$ 的扭曲矩阵族 $M_{K,\rho}(u)$ 与扭曲 $\zeta/L$-函数，命题 \ref{prop:kernel-artin-factorization} 给出 Artin 型因子化。该分解表明：有限轴束本质上是以 $\widehat G$ 为谱坐标系的“可审计纤维化”。

\paragraph{新增模轴的不可约性}
命题 \ref{prop:arity-335-n2-nondegenerate} 证明了第三轴 $N_2(\gamma)\bmod 5$ 无法由前两轴消去，因此新增模轴引入不可约的新自由度。这一事实把“增加轴束不是重标记”在核层面固定为可审计结论。

\begin{definition}[素数寄存器纤维的极限口径]\label{def:pom-profinite}
取模系统 $\{\ZZ/m\ZZ\}_{m\ge 1}$ 的射影极限
\[
\widehat{\ZZ}:=\varprojlim_m \ZZ/m\ZZ\ \cong\ \prod_{p\in\mathbb{P}}\ZZ_p,
\]
并将其作为“无限素数轴束”的极限对象。该极限把所有有限模轴束统一为同一纤维坐标体系，从而为“素数寄存器”提供结构性定义（参见 \cite{Apostol1976} 中关于 CRT 与 p-进分解的标准事实）。
\end{definition}

\begin{remark}[从有限扭曲到无限寄存器]
有限群扭曲提供“有限模轴束”的可审计谱分解；而新增轴的不可约性意味着该分解在模数增大时严格增加自由度。因而 $\widehat{\ZZ}$ 的极限口径与素数寄存器纤维 $\mathcal{P}$ 在结构上是兼容的：前者是“模轴束的完备化”，后者是“乘法坐标的线性化轴束”。
\end{remark}

\subsection{投影词正规形与读出层级塔}\label{subsec:pom-normal-form}
\begin{theorem}[半环闭包的一次投影正规形]\label{thm:pom-one-fold-normal-form}
令 $t$ 为由 $\oplus,\otimes,0_X,1_X$ 生成的任意项，变量取值于 $X_\infty$。则存在多项式
\(
P\in\NN[x_1,\dots,x_k]
\)
使得对任意 $c_1,\dots,c_k\in X_\infty$，
\[
t(c_1,\dots,c_k)=Z\!\left(P\bigl(\mathrm{Val}(c_1),\dots,\mathrm{Val}(c_k)\bigr)\right).
\]
因此在半环闭包内，规范投影 $P_Z$ 可在末端统一调用一次，而中间步骤可完全留在过程层。
\end{theorem}
\begin{proof}
由定理 \ref{thm:add-definitional} 与 \ref{thm:mul-definitional}，$\mathrm{Val}$ 给出 $(X_\infty,\oplus,\otimes)$ 与 $(\NN,+,\times)$ 的半环同构。任意项 $t$ 在 $\NN$ 上对应一个多项式 $P$；于是
\(
\mathrm{Val}(t(c_1,\dots,c_k))=P(\mathrm{Val}(c_1),\dots,\mathrm{Val}(c_k))
\)。
再由 $Z=\mathrm{Val}^{-1}$ 得结论。证毕。
\end{proof}

\begin{remark}[纲领：读出层级塔]\label{rem:pom-hyper-iteration}
定理 \ref{thm:pom-one-fold-normal-form} 表明，在半环闭包内“规范投影只需末端一次”。进一步可沿定理 \ref{thm:composition-two-layer} 的思路构造读出层级塔：第 $0$ 层以合成 $\circ$ 的一次迭代读出对应加法，第 $1$ 层以“加法迭代”对应乘法；按同样方式递归定义第 $k$ 层的“运算”为第 $k{-}1$ 层的迭代读出。该层级塔在过程层不引入新原语，而将复杂度转移为读出层级的提升。
\end{remark}

\subsection{序投影的空间化}\label{subsec:pom-order-spatialization}
\begin{theorem}[序投影的空间化定律]\label{thm:pom-order-spatialization}
设 $B\ge 1$，取两两互素的模 $p_1,\dots,p_m$，令 $P=\prod_{i=1}^m p_i$。若 $P>2B$，则任意 $x\in[-B,B]$ 由其在模 $p_i$ 下的剩余类唯一确定。因而比较、符号与区间约束等序投影可以先在各模轴上点wise 计算，再通过 CRT 重建完成（参见 \cite{Apostol1976}）。
\end{theorem}
\begin{proof}
由中国剩余定理，模 $P$ 的剩余类与模 $\{p_i\}$ 的剩余类一一对应。若 $P>2B$，则区间 $[-B,B]$ 中的整数在模 $P$ 下唯一代表，故由模轴信息即可恢复 $x$ 并完成序判定。证毕。
\end{proof}

\begin{remark}[时间—空间转移的结构后果]
定理 \ref{thm:pom-order-spatialization} 表明：序投影的时间瓶颈可通过增加模轴束转移为可并行的空间代价。在并行 CRT 口径下，序判定的深度由重建电路决定，而非随位长线性增长；因此 $P_{\le}$ 的“不可并行增长”可被视为\emph{未激活足够模轴时}的结构性症状。
\end{remark}

\subsection{熵—延迟互补的结构猜想}\label{subsec:pom-entropy-delay}
\begin{conjecture}[延迟与 carry-free 复杂度的互补]\label{conj:pom-entropy-delay}
对固定基数与同一值不变量模板（零式 $+$ $q$-场），若在线归一化核的延迟 $\delta_Z$ 被压低，则其 carry-free 骨架的分支复杂度（例如 $\rho(A_0)$ 或对应 $\zeta$ 的谱指纹）必须上升；反之，若要求 carry-free 骨架更紧（低熵），则必须付出更长的延迟。该趋势与 $9/13/21$-local 三核的 carry-free $\zeta$ 三分分类一致（定理 \ref{thm:carry-free-zeta-trichotomy}）。
\end{conjecture}

\subsection{ISA 与微码：投影词的抽象层实现}\label{subsec:pom-isa}
\paragraph{可见寄存器（外表）}
设通用寄存器 $Z_0,\dots,Z_{31}$ 的内容始终保持在 $X_\infty$ 的规范地址上；写回时默认触发 $P_Z$。

\paragraph{隐纤维（程序不可见）}
隐纤维包含尺度纤维 $\mathcal{S}_\varphi$、相位纤维 $\mathcal{T}$、素数纤维 $\mathcal{P}$，以及同步核状态与余量（例如 $|\Sigma|=40$ 的真实输入核，见附录关于真实输入核的构造）。

\paragraph{指令层（投影词标注）}
\begin{itemize}
  \item \texttt{ZADD rd,ra,rb}：$W=(1,0)$。
  \item \texttt{ZMUL rd,ra,rb}：$W=(1,0)$（$D_U$ 由线性化纤维选择决定）。
  \item \texttt{ZSUB rd,ra,rb}（比较路线）：$W=(1,1)$。
  \item \texttt{ZSUBR rd,ra,rb}（符号内化路线）：$W=(1,0)$ 但 $\delta_Z$ 增大。
  \item \texttt{ZCMP rt,ra,rb}：$W=(0,1)$。
  \item \texttt{ZDIVMOD rq,rr,ra,rb}：依实现可取 $W\approx(2,\Theta(n))$ 或 $W\approx(2,2)$。
  \item \texttt{ZNORM rd,rs}：显式 $P_Z$，$W=(1,0)$。
\end{itemize}

\paragraph{微码原语（英文标签，用于可审计性）}
\begin{itemize}
  \item \texttt{LIFT\_SCALE}：进入 $\mathcal{S}_\varphi$；
  \item \texttt{LIFT\_PHASE}：进入 $\mathcal{T}$；
  \item \texttt{LIFT\_PRIME}：进入 $\mathcal{P}$；
  \item \texttt{EVOLVE(op)}：执行一步本体演化 $U$；
  \item \texttt{PROJ\_NORM}：执行 $P_Z$；
  \item \texttt{PROJ\_ORDER}：执行 $P_{\le}$；
  \item \texttt{CLEAN}：排空有限余量并回写稳定地址。
\end{itemize}

\paragraph{统一微码模板}
\begin{itemize}
  \item \texttt{ZADD}：\texttt{LIFT\_SCALE; EVOLVE(add); PROJ\_NORM; CLEAN}.
  \item \texttt{ZMUL}（phase-route）：\texttt{LIFT\_PHASE; EVOLVE(phase-linearize); EVOLVE(pointwise-mul); PROJ\_NORM; CLEAN}.
  \item \texttt{ZMUL}（prime-route）：\texttt{LIFT\_PRIME; EVOLVE(pointwise-mul); PROJ\_NORM; CLEAN}.
  \item \texttt{ZSUB}：\texttt{PROJ\_ORDER(cmp); EVOLVE(sub); PROJ\_NORM; CLEAN}.
  \item \texttt{ZSUBR}：\texttt{LIFT\_SCALE; EVOLVE(signed-digit-sub); PROJ\_NORM(strong); CLEAN}.
  \item \texttt{ZDIVMOD}（reciprocal-route）：\texttt{EVOLVE(recip-iterate); PROJ\_ORDER(q); EVOLVE(r=a-bq); PROJ\_ORDER(fix); PROJ\_NORM(q); PROJ\_NORM(r); CLEAN}.
\end{itemize}

\begin{remark}[\texorpdfstring{$p$}{p}-local 与在线延迟的对应]\label{rem:pom-plocal-delay}
第 \ref{sec:A-kernel-compare-main} 节把并行归一化核的滑动窗口范围（$p$-local）与 MSDF 在线延迟建立对应，并给出单流在线机的最小状态规格（命题 \ref{prop:min-state-spec-flow}）。因此 $\delta_Z$ 并非外加常数，而是由所选归一化核的局部性与在线编译结构决定。
\end{remark}

以上 ISA/微码不引入新的算术原语，而是将“同一演化 + 两类投影 + 线性化深度”的差异组织为可审计的投影词。该抽象层与第 \ref{sec:folding} 节的折叠接口与第 \ref{sec:A-kernel-compare-main} 节的核指纹体系兼容。
