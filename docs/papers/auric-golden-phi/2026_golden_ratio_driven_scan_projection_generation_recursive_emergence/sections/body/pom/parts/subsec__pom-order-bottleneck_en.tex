\subsection{Corollary Cluster V: Order Projection as Structural Boundary}\label{subsec:pom-order-bottleneck}
\begin{corollary}[Irreducibility of Order Projection Readout]\label{cor:pom-order-bottleneck}
Under fixed resolution $m$ without introducing additional fiber coordinates, if a readout quantity $g:\Omega_m\to\mathbb{R}$ does not belong to the fold-invariant subalgebra $\mathcal{B}_m$, then gauge projection $P_Z$ alone cannot determine $g$; to obtain this readout, one must either increase resolution or introduce order projection $P_{\le}$ (or an equivalent global aggregation mechanism). This conclusion is a direct structural corollary of Proposition \ref{prop:pom-fiber-invariance}.
\end{corollary}

\begin{corollary}[Mixed Time Scales of Order Projection]\label{cor:pom-order-mixing}
Let the classification readout be given by finite group mark extension, and let $\lambda_1(u)$ denote the principal spectral radius and $\Lambda(u)$ the worst-case non-trivial twisted spectral radius (Theorem \ref{thm:kernel-chebotarev-exp}). Then the error in classification counting decays at exponential rate $(\Lambda(u)/\lambda_1(u))^n$, so that iteration depth to reach precision $\varepsilon$ satisfies
\[
n\ \gtrsim\ \frac{\log(1/\varepsilon)}{\log\!\bigl(\lambda_1(u)/\Lambda(u)\bigr)}.
\]
Therefore the time cost of order projection can be explicitly quantified by the twisted spectral gap.
\end{corollary}

\begin{remark}[Spectral Gauge Calibration of Order Projection (Spectral Gap = Ontological Time Scale)]\label{rem:pom-order-mixing-scale}
This conclusion indicates that the time cost of $P_{\le}$ should not be characterized solely by input size, but rather by the "worst-case twist ratio"
\[
\eta_\infty(u):=\frac{\Lambda(u)}{\lambda_1(u)}
=\sup_{\chi\neq 1}\frac{\rho(M_\chi(u))}{\rho(M_1(u))}
\in(0,1)
\]
as the fundamental scale (twist block caliber in Theorem \ref{thm:kernel-chebotarev-exp}; or the profinite character diagonalization form in Proposition \ref{prop:pom-profinite-artin-hilbert}).
Equivalently, one can take
\[
\tau_{\mathrm{mix}}(u):=\frac{1}{-\log \eta_\infty(u)}=\frac{1}{\log(\lambda_1(u)/\Lambda(u))}
\]
as the intrinsic time unit of the order projection gate, and logarithmically scale with error threshold $\varepsilon$:
\(
n\gtrsim \tau_{\mathrm{mix}}(u)\log(1/\varepsilon)
\)
(by Corollary \ref{cor:pom-order-mixing}).
\end{remark}

\begin{remark}[Mixing Scale and Numerical Anchor of Character Projection]\label{rem:pom-chi-mixing}
For character projection $P_\chi$, one can define mixing scale
\[
\tau_{\mathrm{mix}}:=\frac{1}{-\log(\rho/\lambda)},
\]
where $\rho$ is the spectral radius of the worst non-trivial character and $\lambda$ is the Perron root. For the $(2,2,2)$ stratification of the 40-state real input kernel, the generating table gives
\(
\rho_{2,2,2}/\lambda\approx 0.927372804879
\)
(see \ref{subsec:arity-dirichlet-mertens-tensor}), yielding $\tau_{\mathrm{mix}}\approx 13.2$. This scale provides the intrinsic readout time scale for $P_\chi$, forming a time projection quantity orthogonal to the constant delay of $P_Z$.
For more complete prime-modulo scan and running diagnostics, see Table \ref{tab:real-input-40-rho-ppp-prime-scan}.
\end{remark}

\begin{remark}[Sum Conservation of Stratification Constants]\label{rem:pom-chi-sum}
For the Dirichlet--Mertens constant tensor $C_{r_1r_2r_3}$ under residue stratification, one has
\[
\sum_{r_1,r_2,r_3} C_{r_1r_2r_3}=\mathsf{M}.
\]
This is not an accidental numerical phenomenon, but a consistency constraint of character projection $P_\chi$: stratification slice merging must recover the global finite-part constant. The sum error in the generating table has reached $10^{-16}$ order (see \ref{subsec:arity-dirichlet-mertens-tensor}).
\end{remark}

\begin{remark}[Prime Shadows of Finite Kernels and the Boundary with Arithmetic Primes]
In the finite kernel framework, the prime-like objects yielded by Euler product and Möbius inversion are primitive orbit spectra (Proposition \ref{prop:time-space-commuting-diagram}). This primality differs from integer primes (see Remark \ref{rem:prime-shadow-vs-primes}), and their difference suggests the necessity of "order projection/extra fiber" as a structural boundary: to express \emph{arithmetic primes} at the ontological layer requires introducing additional fiber bundles or new projection mechanisms. An extreme but clean example is Conway's FRACTRAN: it literally makes "primes=registers" and implements deterministic evolution via first-fit order gate control; see Subsection \ref{subsec:pom-fractran}.
\end{remark}

\subsection{Spatialization of Order Projection}\label{subsec:pom-order-spatialization}
\begin{theorem}[Spatialization Law of Order Projection]\label{thm:pom-order-spatialization}
Let $B\ge 1$, take pairwise coprime moduli $p_1,\dots,p_m$, and let $P=\prod_{i=1}^m p_i$. If $P>2B$, then any $x\in[-B,B]$ is uniquely determined by its residue classes modulo $p_i$. Therefore order projections such as comparison, sign, and interval constraints can be computed pointwise on each modulo axis, then completed via CRT reconstruction (cf. \cite{Apostol1976}).
\end{theorem}
\begin{proof}
By the Chinese Remainder Theorem, residue classes modulo $P$ and residue classes modulo $\{p_i\}$ are in one-to-one correspondence. If $P>2B$, then integers in $[-B,B]$ have unique representatives modulo $P$, so $x$ can be recovered from modulo axis information and order determination can be completed. This completes the proof.
\end{proof}

\begin{remark}[Structural Consequences of Time–Space Transfer]
Theorem \ref{thm:pom-order-spatialization} shows that the time bottleneck of order projection can be transferred to parallelizable spatial cost by adding modulo axis bundles. Under parallel CRT caliber, the depth of order determination is determined by reconstruction circuits, rather than linear growth with bit length; therefore the "non-parallelizable growth" of $P_{\le}$ can be viewed as a structural symptom \emph{when insufficient modulo axes are activated}.
\end{remark}

\begin{remark}[Prime Support Count: Natural Factorization of Modulo Register Occupation]\label{rem:pom-mod-support-count}
Let \(m=\prod_{p\mid m}p^{k_p}\).
Truncating the congruence axis \(\widehat{\ZZ}\cong\prod_p\ZZ_p\) (Definition \ref{def:pom-profinite}) to modulo \(m\) is equivalent to: touching only finitely many \(p\)-axes, and on each axis requiring only \(k_p\) bits of \(p\)-adic depth.
Therefore one can decompose "modulo register occupation" into two auditable counts:
\[
S_{\mathrm{mod}}(m):=\#\{p:\ p\mid m\},\qquad
R_{\mathrm{mod}}(m):=\sum_{p\mid m}k_p.
\]
The former characterizes how many prime axes need to be activated in parallel, while the latter characterizes total depth on these axes (isomorphic to finite support caliber of prime-fiber; Definition \ref{def:pom-prime-fiber}).
Under ISA resource caliber, the spatial overhead of \(P_\chi(\bmod m)\) is structurally controlled by \((S_{\mathrm{mod}}(m),R_{\mathrm{mod}}(m))\);
once one needs to select the \emph{canonical representative} from congruence classes (such as minimal non-negative residue), the problem returns to irreducible readout of order projection \(P_{\le}\) (Corollary \ref{cor:pom-order-bottleneck}).
\end{remark}

\subsection{Structural Conjecture of Entropy–Delay Complementarity}\label{subsec:pom-entropy-delay}
\begin{conjecture}[Complementarity of Delay and Carry-Free Complexity]\label{conj:pom-entropy-delay}
For fixed radix and the same value-invariant template (zero-pattern $+$ $q$-field), if the delay $\delta_Z$ of the online normalization kernel is suppressed, then the branching complexity of its carry-free skeleton (e.g., $\rho(A_0)$ or corresponding $\zeta$ spectral fingerprint) must increase; conversely, if one requires a tighter carry-free skeleton (low entropy), one must pay a longer delay. This trend is consistent with the carry-free $\zeta$ trichotomy classification of the $9/13/21$-local three kernels (Theorem \ref{thm:carry-free-zeta-trichotomy}).
\end{conjecture}

\subsection{Closed-Loop Summary: Three Core Conclusions}\label{subsec:pom-summary}
The following three conclusions constitute the core convergence of this paper in the POM context:
\begin{enumerate}
  \item \textbf{Projection entropy rate locked by $\varphi$:} The spatial cost of gauge projection is controlled by fold index $I_m$, and the entropy rate per unit resolution is
  \[
  h_Z=\log\!\left(\frac{2}{\varphi}\right)
  \]
  (Proposition \ref{prop:pom-projection-entropy}).
  \item \textbf{One-fold normal form for semiring closure:} Any term generated by $\oplus,\otimes$ can be written as a value-layer polynomial followed by a single $P_Z$ back to stable address (Theorem \ref{thm:pom-one-fold-normal-form}).
  \item \textbf{Growth boundary at order projection:} Once the readout quantity is no longer a fold-fiber invariant, one must introduce $P_{\le}$ or increase fiber/resolution (Corollary \ref{cor:pom-order-bottleneck}); its time bottleneck can be transferred via modulo axis bundle spatialization (Theorem \ref{thm:pom-order-spatialization}).
\end{enumerate}
\paragraph{Structural Reduction (Summary)}
The ontological structure of mathematics is "local evolution $U$ + three projection gates": $P_Z$ projects value-fiber groupoid orbits to Zeckendorf transversal, whose spatial defect is described by $d_m$ and $E_m$ and renormalized at primitive cycle scale to $\log\mathfrak{M}$ and Dirichlet--Mertens constant tensor; $P_{\mathrm{prim}}$ projects time traces to primitive atomic layer and strictly aligns with the commutative diagram $Z=E\circ M\circ T$; $P_{\le,b}$ projects translation fiber orbits to fundamental domain transversal and yields quotient-remainder, while character/congruence readout is uniformly encoded by $P_\chi$ (twist matrices and constant tensor implementation), whose time scale is given by $\rho/\lambda$. Differences among addition, subtraction, multiplication, division, quotient, and remainder are only manifested in projection word combination mode and linearization depth (prime-fiber or phase-fiber route) choices.

\paragraph{Non-Parallel Overhead of Addition, Subtraction, Multiplication, Division, Quotient, and Remainder Is the Irreducible Cost of the Same Class of Projection Gates}
In the cost model
\(
T_I(n;m)=k_Z\delta_Z+k_{\le}T_{\le}(n)+k_{\downarrow}T_{\downarrow}(m)+D_U(n)
\)
(Definition \ref{def:pom-cost}), the difference among "six arithmetic operations" does not come from new ontological primitives, but from two points:
\begin{enumerate}
  \item \textbf{Which prior fiber linearization is placed on:} The same semantics can choose to complete linearization on scale-fiber/phase-fiber/prime-register/congruence-axis (\(\mathcal{S}_\varphi,\mathcal{T},\mathcal{P},\widehat{\ZZ}\); Definition \ref{def:pom-fiber-kernel-choice}), thereby leaving growth terms in evolution depth \(D_U\), or externalizing as spatial occupation of axis bundles (e.g., \((S_{\mathrm{mod}},R_{\mathrm{mod}})\), Remark \ref{rem:pom-mod-support-count}).
  \item \textbf{Whether order gate appears unavoidably in projection word:} Once semantics includes minimality/uniqueness/inequality and other global constraints, order projection \(P_{\le}\) (or its softening \(\mathrm{PROJ}_u\)) appears irreducibly (Corollary \ref{cor:pom-order-bottleneck}), and introduces ontological time scale \(\tau_{\mathrm{mix}}(u)\) controlled by worst-case twisted spectral gap (Corollary \ref{cor:pom-order-mixing} and Remark \ref{rem:pom-order-mixing-scale}).
\end{enumerate}
In particular: if \(k_{\le}=0\), then non-parallel delay contains only constant-level \(\delta_Z\) (gauge projection gate), independent of input size; if \(k_{\le}>0\), one must pay time cost calibrated by \(\tau_{\mathrm{mix}}(u)\), and can spatialize this time bottleneck by transfer via modulo axis bundle according to Theorem \ref{thm:pom-order-spatialization}.

\begin{remark}[Order Gate's \texorpdfstring{$(m,n,N)$}{(m,n,N)} Three-Budget Certificate (Time Tail Term \texorpdfstring{$+$}{+} Sampling Projection Term)]\label{rem:pom-order-three-budget}
When implementing order projection \(P_{\le}\) (or its character/congruence readout \(P_\chi\)) as "uniform distribution determination on layer \(m\) congruence cylinder", its finite precision certificate can be decomposed into two orthogonal budget axes: dynamical tail term of primitive depth \(n\), and projection/sampling term of sample size \(N\).
Specifically, for any cylinder \(C\subseteq G_m\), the finite error satisfies (Theorem \ref{thm:auditable-profinite-chebotarev-finite}; for deterministic version under rotation scan see Corollary \ref{cor:auditable-profinite-chebotarev-three-budget-deterministic})
\[
\left|\Pi_{n,m}(C)-\frac{|C|}{|G_m|}\right|
\ \le\
K_1\,\frac{|G_m|}{\lambda^{(1/2+\varepsilon)n}}
\ +\
K_2\cdot D_{\mathrm{TV}}(\hat{\pi}_{m,N},\pi_m^{\mathrm{fold}})
\ \le\
K_1\,\frac{|G_m|}{\lambda^{(1/2+\varepsilon)n}}
\ +\
K_2(m+1)D_N^*(\alpha;x_0).
\]
Therefore when semantics must include order gate (such as division/quotient-remainder's \(0\le r<b\) constraint), non-parallel overhead is irreducibly constrained by both \(n\) (time tail term) and \(N\) (sampling projection term), which are structurally non-interchangeable.
\end{remark}
Prime axes play a common index role here: they are both the coordinate axis for multiplication linearization (prime-register fiber \(\mathcal{P}\), Definition \ref{def:pom-prime-fiber}), and the completion axis for congruence/character readout (\(\widehat{\ZZ}\cong\prod_p\ZZ_p\), Definition \ref{def:pom-profinite}); phase/character coordinatization necessarily introduces exponential weight \(u=e^\theta\) and normalization constant \(e^{2\pi i\langle k,x\rangle}\) (Remark \ref{rem:pom-congruence-invlimit}).
Table \ref{tab:pom-isa-minsig} implements this principle as an ISA-level minimal signature table: as long as \(P_{\le}\) appears, mixing time calibrated by spectral gap necessarily appears; as long as \(\mathrm{RESCALE}\downarrow\) is explicitly called, support writeback budget calibrated by gauge difference \(G_m\) necessarily appears; all other non-parallel terms are constant online delay of gauge projection gate.

\begin{remark}[Time Defect Dichotomy: Sync Delay and Spatializable Order Depth]\label{rem:pom-time-defect-dichotomy}
In this paper's framework, "time complexity" splits into at least two non-equivalent sources:
\begin{enumerate}
  \item \textbf{Sync delay class (non-spatializable time):} Represented by \(\delta_{\min}(\Fold_\infty)=3\), it is forced by local dependencies (margin conservation/sync state), and is orthogonal to expansion of prime-register / modulo axis bundle; therefore simply adding modulo axes cannot eliminate this class of delay, but can only change the \emph{spatial} allocation of parallel deployment.
  \item \textbf{Order projection class (spatializable time):} The serial bottleneck brought by comparison, sign, and interval constraints and other total-order aggregation can be compiled into spatialized residue parallel readout and CRT reconstruction according to Theorem \ref{thm:pom-order-spatialization} and rewrite pattern (RCRT) (Definition \ref{def:pom-order-elimination-rewrite}; Proposition \ref{prop:pom-rcrt-epsilon}); its cost is transferred to prime-register capacity budget \(\log P_m\), and calibrated by quantile quota law (Corollary \ref{cor:pom-prime-register-quantile-budget}).
\end{enumerate}
Therefore, "space/time duality" does not hold globally: what holds is only spatialization transfer of order projection class time; sync delay constitutes an independent and non-spatializable time invariant.
At abstract ISA layer, one can also make the "coordinate ledger" of prior fibers explicit as two optional register file groups: prime exponent registers \texttt{P[i]} (\(\mathcal{P}\) finitely supported exponent vector, for linearizing multiplication/division to coordinate-wise addition/subtraction), and phase/character registers \texttt{T[i]} (\(\mathcal{T}\) phase/character coordinates, for diagonalizing linear parts and phase/character readout). Under this paper's caliber, they only change the allocation of spatial budget \(S(n)\) and linearization depth \(D_U(n)\), without introducing new non-parallel readout gates: non-parallel overhead is still completely locked by the three fingerprints of \(P_Z,P_{\le}\) and \(\mathrm{RESCALE}\downarrow\).
\end{remark}

\begin{remark}[Typical External Face Rewrite Constant of Cross-Resolution \texttt{RESCALE}$\downarrow$]\label{rem:pom-rescale-space-constant}

T_{\downarrow}(m)\ \asymp\ \frac{4}{9}m,

which characterizes \emph{how many register bits to write back}, and is counted in the unified cost model as independent spatial term $k_{\downarrow}(I)\,T_{\downarrow}(m)$ (Definition \ref{def:pom-cost}); this term is orthogonal to online delay constants (e.g., \(\delta_{\min}(\Fold_\infty)=3\), Proposition \ref{prop:pom-delay-min}).

\subsubsection{FRACTRAN Compatibility Layer: First-Fit Zero-Temperature Order Gate for Prime Registers}\label{subsec:pom-fractran}
Conway's FRACTRAN language makes "primes=registers" literally equivalent: machine configuration is an integer \(N\), whose prime factorization exponent vector \((v_p(N))_{p\in\mathbb{P}}\) is precisely the coordinate of prime-register fiber \(\mathcal{P}\) (Definition \ref{def:pom-prime-fiber}; Remark \ref{rem:two-coordinate-monoids}). Its execution rule is: given an ordered string of fractions, always take the \emph{first} fraction that can make \(N\cdot f\) remain an integer to multiply \cite{Conway1987FRACTRAN}.

\begin{definition}[FRACTRAN Program and One-Step Semantics]\label{def:pom-fractran}
A FRACTRAN program is a sequence of reduced positive fractions
\(
f_1,\dots,f_k
 \)
(\(f_i=a_i/b_i\), \(\gcd(a_i,b_i)=1\)).
Given state integer \(N\in\NN_{\ge 1}\), if there exists minimal \(i\) such that \(Nf_i\in\NN_{\ge 1}\), then update \(N\leftarrow Nf_i\); if none exists then halt.
\end{definition}

\begin{proposition}[FRACTRAN = Order-Gated Translation on \texorpdfstring{$\mathcal{P}$}{P}]\label{prop:pom-fractran-prime-translation}
Let \(r=v(N)\in\mathcal{P}\) denote the prime exponent vector of \(N\) (i.e., \(r(p)=v_p(N)\)).
For each instruction \(f_i=a_i/b_i\) define
\[
\beta_i:=v(b_i)\in\mathcal{P},\qquad
\Delta_i:=v(a_i)-v(b_i)\in \ZZ^{(\mathbb{P})}.
\]
Then \(Nf_i\in\NN_{\ge 1}\) if and only if \(r\ge \beta_i\) (coordinate-wise partial order over primes), and this step update is equivalent to
\[
r\ \longleftarrow\ r+\Delta_i.
\]
Let
\[
i^\ast(r):=\min\{\,i:\ r\ge \beta_i\,\},
\]
if set is empty then halt; otherwise FRACTRAN one step is equivalent to deterministic transition
\[
T(r)=r+\Delta_{i^\ast(r)}.
\]
\end{proposition}
\begin{proof}
By \(\gcd(a_i,b_i)=1\), have \(Nf_i\in\NN\iff b_i\mid N\). In exponent vector caliber, this is equivalent to \(v_p(N)\ge v_p(b_i)\) for all \(p\), i.e., \(r\ge \beta_i\). When executable,
\(
v_p(Nf_i)=v_p(N)+v_p(a_i)-v_p(b_i)
\)
holds coordinate-wise, thus \(r\leftarrow r+\Delta_i\). First-fit's "take minimal \(i\)" corresponds to definition of \(i^\ast(r)\). This completes the proof.
\end{proof}

\begin{remark}[First-Fit Is \texorpdfstring{$u\to 0$}{u->0} Zero-Temperature Hardening Limit]\label{rem:pom-fractran-zero-temp}
For fixed \(r\), let feasible set \(I(r):=\{i:\ r\ge \beta_i\}\). Define a "soft first-fit" distribution
\[
\Pr_u(i\mid r)\ \propto\ u^{i}\,\ind\!\left[i\in I(r)\right],\qquad u\in(0,1].
\]
Then as \(u\to 0\), mass concentrates to \(\min I(r)\), and recovers the hard first-fit rule of Definition \ref{def:pom-fractran} (deterministic selection \(i^\ast(r)\)).
This provides a minimal example: deterministic priority selection can be viewed as zero-temperature hardening of temperatured order projection \(\mathrm{PROJ}_u\) (Definition \ref{def:pom-proj-temp}, Remark \ref{rem:pom-order-as-zero-temp}).
\end{remark}

\begin{proposition}[Thermal Deformation of Order Projection: Analytic Semantics Family and Stochastic Dominance of Soft First-Fit]\label{prop:pom-fractran-soft-firstfit-analytic-dominance}
For fixed \(r\) with \(I(r)\neq\varnothing\), let
\[
Z_r(u):=\sum_{j\in I(r)}u^{j}\qquad(u\in(0,1])
\]
and normalize the distribution of Remark \ref{rem:pom-fractran-zero-temp} as
\[
\Pr_u(i\mid r):=\frac{u^{i}}{Z_r(u)}\,\ind[i\in I(r)].
\]
Then:
\begin{enumerate}
  \item \textbf{(Analyticity)} The map \(u\mapsto (\Pr_u(i\mid r))_{i=1}^{k}\) is a real-analytic probability simplex curve on \(u\in(0,1]\);
  \item \textbf{(Zero-temperature limit)} \(\Pr_u(\cdot\mid r)\) converges to \(\delta_{i^\ast(r)}\) as \(u\downarrow 0\) (recovering hard first-fit);
  \item \textbf{(Stochastic dominance monotonically strengthens as \(u\downarrow 0\))} If \(0<u_1<u_2\le 1\), then for any threshold \(t\in\{1,\dots,k\}\) we have
  \[
  \sum_{i\le t}\Pr_{u_1}(i\mid r)\ \ge\ \sum_{i\le t}\Pr_{u_2}(i\mid r),
  \]
  i.e., mass monotonically concentrates toward smaller indices as \(u\downarrow 0\) (first-order stochastic dominance in index order).
\end{enumerate}
\end{proposition}
\begin{proof}
Since \(I(r)\subseteq\{1,\dots,k\}\) is a finite set, \(Z_r(u)\) is a nonzero polynomial strictly positive on \(u\in(0,1]\), so each \(\Pr_u(i\mid r)\) is a ratio of polynomials, hence real-analytic on \(u\in(0,1]\), yielding (1).
Let \(i_\ast:=\min I(r)=i^\ast(r)\). Writing \(Z_r(u)\) as \(u^{i_\ast}\bigl(1+\sum_{j\in I(r)\setminus\{i_\ast\}}u^{j-i_\ast}\bigr)\) immediately gives \(\Pr_u(i_\ast\mid r)\to 1\) and other coordinates tend to \(0\) as \(u\downarrow 0\), yielding (2).
\par
For (3), note that for any \(u\) and \(i\in I(r)\) we have \(\Pr_u(i\mid r)\propto u^{i}\), so when \(0<u_1<u_2\),
\[
\frac{\Pr_{u_1}(i\mid r)}{\Pr_{u_2}(i\mid r)}
=
\frac{Z_r(u_2)}{Z_r(u_1)}\left(\frac{u_1}{u_2}\right)^{i}
\]
is monotonically decreasing in \(i\); therefore this family satisfies monotone likelihood ratio (MLR) order, and by standard corollary yields first-order stochastic dominance in index order:
\(
\Pr_{u_1}(i\ge t\mid r)\le \Pr_{u_2}(i\ge t\mid r)
\),
equivalent to the shown cumulative distribution inequality. This completes the proof.
\end{proof}

\begin{remark}[Control Flow Externalization: State Machine as Order Projection Selection]\label{rem:pom-fractran-controlflow}
In FRACTRAN, "state machine" need not be additionally understood as graph-theoretic object: feasibility predicate \(r\ge \beta_i\) yields a family of piecewise linear reachability regions, and first-fit rule is precisely the \emph{selection projection} on this decomposition (mapping \(r\) to index \(i^\ast(r)\) and selecting update accordingly).
FRACTRAN can also externalize program counter and finite control state to \(\mathcal{P}\)'s coordinate ledger by introducing dedicated prime axes (using exponents of these primes to carry current instruction segment/branch state), thereby compiling control flow also as addressable coordinates of prime registers.
In POM language, this is equivalent to moving control structure of projection words to \(\mathcal{P}\) fiber, while non-parallel overhead is still dominated by selection depth of order gate.
\end{remark}

\begin{remark}[Cost Lower Bound: Scan Depth \texorpdfstring{$\times$}{x} Support Scale]\label{rem:pom-fractran-cost}
Within prime-core, \(r\mapsto r+\Delta_i\) is linear update on \(\mathcal{P}\), so evolution depth \(D_U\) can be viewed as constant-level; non-parallel term comes entirely from order gate.
Implementing one \texttt{FSTEP} as one "priority selection" type \texttt{PROJ\_ORDER(first-fit)} call, its projection word can take \(W=(0,1)\), but its one order projection aggregation depth \(T_{\le}\) contains at least priority scan over instruction table (worst-case \(k\)).
Meanwhile, feasibility test touches only \(|\mathrm{supp}(b_i)|\) coordinates of denominator \(b_i\)'s prime support (coordinate-wise comparison), so "how many register axes to read per step" and "how deep to scan" are structurally orthogonal.
If output must be explicitly written back to external face \(X_\infty\), an additional \texttt{PROJ\_NORM} is needed, corresponding to \(k_Z\) increasing \(1\) (Definition \ref{def:pom-cost}).
\end{remark}

\begin{proposition}[Intrinsic Seriality of Turing Completeness: Ineliminable Lower Bound for First-Fit Priority Selection Depth]\label{prop:pom-fractran-firstfit-serial-depth}
Fix a FRACTRAN program \(f_1,\dots,f_k\) of length \(k\) (Definition \ref{def:pom-fractran}).
In a PW/ISA semantic model that only allows implementing priority selection via order gate \(\mathrm{PROJ\_ORDER}(\text{first-fit})\), any one-step implementation consistent with hard first-fit semantics satisfies worst-case order projection aggregation depth
\[
T_{\le}\ =\ \Omega(k).
\]
Moreover, this lower bound does not change when control flow is externalized to \(\mathcal{P}\) fiber or normalization rewriting is imposed: non-parallel overhead remains locked in priority selection depth.
\end{proposition}
\begin{proof}
Take state \(r\) such that feasible set \(I(r)=\{k\}\) (e.g., make \(r\ge \beta_k\) and \(r\not\ge \beta_i\) for all \(i<k\); under finite instruction table this can always be realized via coordinate-wise prime selection).
Then hard first-fit must output \(i^\ast(r)=k\).
In a model that only allows determining \(i^\ast(r)\) through priority selection order gate, to exclude feasibility of all \(i<k\) and confirm \(k\) as the minimal feasible index, worst-case requires priority scan over \(k\) instructions; therefore \(T_{\le}\) is at least linearly controlled by \(k\).
Control flow externalization and normalization rewriting only change implementation's fiber bearing and constant layer organization (Remark \ref{rem:pom-fractran-controlflow}), without changing the selection semantics itself that "output must equal \(\min I(r)\)"; thus this lower bound is ineliminable. This completes the proof.
\end{proof}

\subsubsection{Projection Word Rewriting: Termination and Normal Form}\label{subsec:pom-rewriting-calculus}
This subsection elevates "interface concatenation" to derivable calculus: at word level giving a terminating rewrite system, making "eliminability of online gauge projection" and "spatialization of order projection" land in rule form; their correctness is guaranteed by structural theorems already proved in this paper (especially one-fold normal form and CRT spatialization law).

\begin{definition}[Value-Layer Projection Word Language and Normalization Rewriting]\label{def:pom-rewriting-val}
Consider sublanguage \(\mathbf{W}_{\mathrm{val}}\) of value-layer microcode words, whose generators include ontological evolution step \texttt{EVOLVE(op)} (where \texttt{op} belongs to local combinations expressible by value-layer semiring closure \(\oplus,\otimes\)) and gauge projection \texttt{PROJ\_NORM} (i.e., \(P_Z\)), and allows finite composition with \texttt{CLEAN}.
Define rewrite relation \(\Rightarrow\) as minimal compatible closure (closed for any context) generated by the following two rules:
\[
\text{(RZ)}\quad
\texttt{PROJ\_NORM;\ PROJ\_NORM}\ \Rightarrow\ \texttt{PROJ\_NORM},
\]
\[
\text{(RNF)}\quad
\texttt{PROJ\_NORM;\ w;\ PROJ\_NORM}\ \Rightarrow\ \texttt{w;\ PROJ\_NORM}
\quad (\text{where }w\in\mathbf{W}_{\mathrm{val}}\text{ does not contain }\texttt{PROJ\_NORM}).
\]
Here (RNF) semantically corresponds to the "one-fold normal form" of Theorem \ref{thm:pom-one-fold-normal-form}: intermediate normalization of value-layer semiring closure can be pushed to one completion at the end.
\end{definition}

\begin{proposition}[Termination and Normal Form Existence (Value Layer)]\label{prop:pom-rewriting-terminating}
On \(\mathbf{W}_{\mathrm{val}}\) of Definition \ref{def:pom-rewriting-val}, rewrite system \((\mathbf{W}_{\mathrm{val}},\Rightarrow)\) terminates; and each word can be reduced to a normal form
\[
w\ \Rightarrow^\ast\ \widetilde w\cdot \texttt{PROJ\_NORM},
\]
where \(\widetilde w\) does not contain \texttt{PROJ\_NORM}, and terminal \texttt{PROJ\_NORM} appears at most once.
\end{proposition}
\begin{proof}
Take descending measure as binary order
\(
\big(\#(\texttt{PROJ\_NORM}),\ \text{number of }\texttt{PROJ\_NORM}\text{ occurrences outside terminal}\big)
\)
by lexicographic order comparison.
Rule (RZ) strictly decreases first component; rule (RNF) does not increase first component but strictly decreases second component.
Thus there exists no infinite descending chain, system terminates. After termination, if \texttt{PROJ\_NORM} still exists not at terminal, then (RNF) can be applied again, contradiction; therefore normal form must be of stated form. This completes the proof.
\end{proof}
