\section{Appendix E: Proof that Independent Gated Sparsification Preserves Equidistribution (Theorem \ref{thm:sparse-gating})}
\label{app:sparse-gating}

\subsection{E.1 Counting Decomposition}

Fix an interval $J=[0,u)$. Let
\[
A_N=\sum_{n=0}^{N-1}S_n\,\ind_J(x_n),\qquad M_N=\sum_{n=0}^{N-1}S_n.
\]
The goal is to prove $A_N/M_N\to u$ almost surely. Write
\[
A_N-p\sum_{n=0}^{N-1}\ind_J(x_n)=\sum_{n=0}^{N-1}(S_n-p)\ind_J(x_n)
\]
as a sum of independent centered random variables.

\subsection{E.2 Equidistribution Term of the Original Sequence}

Since $(x_n)$ is equidistributed,
\[
\frac{1}{N}\sum_{n=0}^{N-1}\ind_J(x_n)\to u.
\]
Hence
\[
\frac{p}{N}\sum_{n=0}^{N-1}\ind_J(x_n)\to pu.
\]

\subsection{E.3 Gating-Fluctuation Term}

Because $(S_n)$ is i.i.d. and independent of $(x_n)$, conditioning on $(x_n)$ shows that $(S_n-p)\ind_J(x_n)$ remains independent, mean-zero, and bounded. By Kolmogorov's strong law (or Hoeffding + Borel--Cantelli),
\[
\frac{1}{N}\sum_{n=0}^{N-1}(S_n-p)\ind_J(x_n)\to 0\quad\text{almost surely}.
\]
Therefore,
\[
\frac{A_N}{N}=\frac{p}{N}\sum_{n=0}^{N-1}\ind_J(x_n)+o(1)\to pu.
\]
Similarly, the strong law gives $M_N/N\to p$ almost surely. Consequently,
\[
\frac{A_N}{M_N}=\frac{A_N/N}{M_N/N}\to \frac{pu}{p}=u\quad\text{almost surely}.
\]

\subsection{E.4 From Half-Open Intervals to General Intervals}

Take a countable dense family of rational-endpoint intervals and preserve probability-one events under countable union. Then extend to all intervals and general Riemann-integrable functions by monotone-class approximation and boundary-zero continuity. The proof is complete.

\subsection{E.5 Finite-Sample Error Decomposition (for Testing)}

For fixed $J$, one has
\[
\left|\frac{A_N}{M_N}-u\right|
\le
\underbrace{\left|\frac{A_N}{M_N}-\frac{p}{M_N}\sum_{n=0}^{N-1}\ind_J(x_n)\right|}_{\text{gating fluctuation}}
\;+\;
\underbrace{\left|\frac{p}{M_N}\sum_{n=0}^{N-1}\ind_J(x_n)-u\right|}_{\text{original-sequence deviation + density deviation}}.
\]
The first term admits exponential tails via Hoeffding bounds; the second is controlled by the original discrepancy $D_N^*$ together with concentration bounds on $\abs{M_N-pN}$. This decomposition provides a direct error-budget interface for falsifiable numerical tests when the gating probability is chosen as $\varphi^{-1}$.
