\section{固定带宽递归读出与分割细化}

\subsection{对象层定义：固定带宽递归读出}

本节在主文固定带宽 $m=6$ 的前提下，引入“递归观测（分割细化）”的对象层构造。其核心思想是：不通过增大 $m$，而通过对同一底系统施加逐级条件化读出，形成嵌套纤维链，从而提升有效分辨率。

\begin{definition}[第一级输出与预像纤维]\label{def:recursive-level0}
在给定观察者 $O$ 下，记
$$
x^{(0)}_t=\mathcal{R}_{O,6}(s_t)\in X_6^{\mathrm Z}.
$$
对任意 $x^{(0)}\in X_6^{\mathrm Z}$，其预像纤维为
$$
\mathcal{F}^{(0)}(x^{(0)})=\mathcal{F}_6(x^{(0)})=\mathrm{Fold}_6^{-1}(x^{(0)})\subseteq \Omega_6.
$$
\end{definition}

\begin{definition}[条件读出与第 $k$ 级输出]\label{def:recursive-conditioned-readout}
给定历史输出
$$
x^{(0:k-1)}=(x^{(0)},x^{(1)},\dots,x^{(k-1)})\in (X_6^{\mathrm Z})^{k},
$$
定义第 $k$ 级的条件读出协议为在该历史条件下的可测读出
$$
q_6^{(k)}:\Sigma\to \Omega_6,
$$
并定义第 $k$ 级输出
$$
x^{(k)}=\mathrm{Fold}_6\!\left(q_6^{(k)}(\cdot)\right)\in X_6^{\mathrm Z}.
$$
\end{definition}

\begin{definition}[嵌套纤维链]\label{def:nested-fiber-chain}
对每一级输出 $x^{(k)}$，定义对应纤维 $\mathcal{F}^{(k)}$ 为在前级条件下可见等价类的预像集合。递归观测诱导嵌套链
$$
\mathcal{F}^{(0)}\supseteq \mathcal{F}^{(1)}\supseteq \cdots \supseteq \mathcal{F}^{(K)}.
$$
\end{definition}

\begin{definition}[递归输出序列]\label{def:recursive-output-sequence}
递归深度为 $K$ 的递归输出记为
$$
(x^{(0)},x^{(1)},\dots,x^{(K)})\in (X_6^{\mathrm Z})^{K+1}.
$$
\end{definition}

\subsection{可核查命题}

\begin{proposition}[R1：递归读出等价于分割的逐级细化]\label{prop:R1-refinement}
在固定底系统与固定 $m=6$ 下，递归读出产生的历史条件化等价于对可观测 $\sigma$-代数施加逐级细化：递归深度 $K$ 对应一个嵌套分割族，其柱集族随 $K$ 单调细化。
\end{proposition}

\begin{proposition}[R2：条件不确定性的单调性]\label{prop:R2-monotone-conditional-uncertainty}
令 $G$ 为目标细节量（可测函数或有限值观测量）。则条件不确定性（例如条件熵）
$$
H(G\mid x^{(0:K)})
$$
随递归深度 $K$ 非增，即 $K'\ge K$ 时有 $H(G\mid x^{(0:K')})\le H(G\mid x^{(0:K)})$。
\end{proposition}

\begin{proposition}[R3：与增大 $m$ 的关系（仅信息量同阶）]\label{prop:R3-K-vs-m}
在可比资源口径下，递归深度增加与一次性增大窗口长度 $m$ 均可提升可区分信息量。本文仅将二者作为实证对照：比较给定目标量 $G$ 时 $H(G\mid x^{(0:K)})$ 的下降效率，不在一般性上宣称严格等价。
\end{proposition}

\subsection{固定 $m=6$ 的协议层动机（对象层表述）}
固定 $m=6$ 的选择在本文仅作为协议层约定，其动机为：\textbf{(i)} 与秩 $6$ 的超空间闭合模型类对齐；\textbf{(ii)} 与 $(3+3)$ 分解并行，使每一级读出可解释为对几何与内部摘要的联合编码；\textbf{(iii)} 最小闭合可验证性（例如 $2^6\to F_8$ 的全枚举平台）。
