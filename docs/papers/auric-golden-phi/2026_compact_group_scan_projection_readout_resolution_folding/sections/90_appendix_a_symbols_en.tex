\section{Appendix A: Symbols and Basic Objects}

\begin{enumerate}
  \item $\TT=\RR/\ZZ$, $\TT^d$ is the $d$-dimensional torus.
  \item $\varphi=(1+\sqrt5)/2$ is the golden ratio; commonly used slope $\alpha=\varphi^{-1}$ or equivalent transformations thereof.
  \item Scanning orbit: $x_n=x_0+n\alpha\pmod1$.
  \item Binary readout: $s_n=\ind_W(x_n)$.
  \item $D_N^*$: one-dimensional star discrepancy.
  \item Fibonacci sequence: $F_1=F_2=1$, $F_{n+2}=F_{n+1}+F_n$.
  \item Golden mean language: $X_m=\{w\in\{0,1\}^m:\text{no subword }11\}$, $|X_m|=F_{m+2}$.
  \item Resolution folding: $\Fold_m:\{0,1\}^m\to X_m$ (induced by Zeckendorf normalization).
  \item Cyclic admissibility:
  \[
  X_m^{\mathrm{cyc}}=\{w\in X_m:w_mw_1\neq 11\};
  \qquad
  X_m^{\mathrm{bdry}}=X_m\setminus X_m^{\mathrm{cyc}}.
  \]
  \item Weyl pair: $U_{\mathrm{scan}}V=e^{2\pi i\alpha}VU_{\mathrm{scan}}$.
\end{enumerate}
