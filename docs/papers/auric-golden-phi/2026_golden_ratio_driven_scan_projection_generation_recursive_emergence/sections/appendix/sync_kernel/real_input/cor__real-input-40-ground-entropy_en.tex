\begin{corollary}[Zero-temperature limit yields positive ground entropy: \texorpdfstring{$h_{\mathrm{ground}}=\log c>0$}{h_ground=log c>0}]\label{cor:real-input-40-ground-entropy}
Under the output potential weighting of Proposition~\ref{prop:real-input-40-pressure-freezing}, let
\(
Z_n(u):=\mathbf{1}^{\mathsf T}M(u)^n\mathbf{1}
\)
be the partition function of length-\(n\) paths (weighting each edge by \(u^{e}\) and summing over all start and end states).
For each \(n\), let
\[
A_{n,k}:=\#\{\text{feasible paths of length }n\text{ with total output }1\text{ count equal to }k\},
\qquad 0\le k\le n,
\]
and write the maximum attainable count as
\(
k_{\max}(n):=\max\{k:\ A_{n,k}>0\}.
\)
Then:
\begin{enumerate}
  \item \(k_{\max}(n)=\lfloor n/2\rfloor\) (consistent with \(\lim_{\theta\to+\infty}P'(\theta)=1/2\));
  \item The exponential growth rate of the maximum-density layer (the ``ground set'') satisfies
  \[
  \boxed{\;
  \lim_{n\to\infty}\frac{1}{n}\log A_{n,k_{\max}(n)}=\log c\; }.
  \]
  Hence the ground subsystem possesses positive per-step entropy density \(h_{\mathrm{ground}}=\log c>0\) (numerically \(\log c\approx 0.6407\)).
\end{enumerate}
\end{corollary}
\begin{proof}
By definition, \(Z_n(u)=\sum_{k=0}^{n}A_{n,k}u^{k}\).
Let \(u=e^\theta\) and take \(P(\theta)=\lim_{n\to\infty}\frac{1}{n}\log Z_n(e^\theta)\) (under a finite matrix family this limit equals \(\log\rho(M(e^\theta))\), i.e., the \(P\) from Proposition~\ref{prop:real-input-40-pressure-freezing}).

\textbf{(i) Maximum density is \(1/2\).}
Proposition~\ref{prop:real-input-40-pressure-freezing} gives \(\lim_{\theta\to+\infty}P'(\theta)=1/2\), which is equivalent to saying that the maximum average density of output \(1\) over feasible paths is \(1/2\); therefore \(k_{\max}(n)=\lfloor n/2\rfloor\).

\textbf{(ii) Extracting ground entropy via the endpoint asymptotics.}
For any \(u>1\), one has the sandwich
\[
A_{n,k_{\max}(n)}u^{k_{\max}(n)}
\ \le\
Z_n(u)
\ \le\
(n+1)\,A_{n,k_{\max}(n)}u^{k_{\max}(n)}.
\]
Taking logarithms, dividing by \(n\), and letting \(n\to\infty\) yields
\[
\lim_{n\to\infty}\frac{1}{n}\log Z_n(u)
\ =\
\lim_{n\to\infty}\frac{1}{n}\log A_{n,k_{\max}(n)}
\ +\
\lim_{n\to\infty}\frac{k_{\max}(n)}{n}\log u.
\]
From \(k_{\max}(n)/n\to 1/2\), this gives
\[
P(\theta)=\lim_{n\to\infty}\frac{1}{n}\log A_{n,k_{\max}(n)}+\frac{\theta}{2}.
\]
Finally, letting \(\theta\to+\infty\) and using the endpoint expansion from Proposition~\ref{prop:real-input-40-pressure-freezing},
\(
P(\theta)=\theta/2+\log c+o(1)
\), one deduces
\(
\lim_{n\to\infty}\frac{1}{n}\log A_{n,k_{\max}(n)}=\log c
\).
Since \(c>1\), it follows that \(h_{\mathrm{ground}}=\log c>0\). \qed
\end{proof}
