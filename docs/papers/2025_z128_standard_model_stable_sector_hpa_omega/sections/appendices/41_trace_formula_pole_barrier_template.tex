\section{Holomorphy versus interior poles: a pole-barrier rigidity template (notes)}
\label{app:trace_pole_barrier_template}

\AuditTag This appendix records an abstract rigidity motif used repeatedly in Abel-first analytic stability arguments: if a protocol object is defined as a holomorphic function on the unit disk, then any competing representation that would force an interior pole is incompatible.
The template is included for reuse and is not used as a premise in theorem-level folding proofs.

\subsection{Exponential modes force interior poles}
\label{subsec:interior_pole_from_exponential_mode}

\begin{lemma}[Interior pole from an off-stable exponential mode]
\label{lem:interior_pole_exponential_mode}
\MathTag Let $\lambda\in\CC$ with $\RePart(\lambda)>0$ and define the mode generating function
\[
M_\lambda(r):=\sum_{t=0}^{\infty} r^t\,\e^{\lambda t},
\qquad |r|<\e^{-\RePart(\lambda)}.
\]
Then $M_\lambda$ has the meromorphic closed form
\[
M_\lambda(r)=\frac{1}{1-r\,\e^{\lambda}},
\]
with a pole at $r_\lambda=\e^{-\lambda}$ satisfying $|r_\lambda|=\e^{-\RePart(\lambda)}<1$.
In particular, any expression that contains a nonzero multiple of $M_\lambda$ cannot extend to a holomorphic function on the full unit disk $\{|r|<1\}$.
\end{lemma}
\begin{proof}
The series is geometric with ratio $r\,\e^\lambda$ and therefore sums to $1/(1-r\,\e^\lambda)$ on its domain of absolute convergence.
The pole occurs where the denominator vanishes, at $r=\e^{-\lambda}$.
Its modulus is $\e^{-\RePart(\lambda)}<1$ when $\RePart(\lambda)>0$.
\end{proof}

\subsection{Holomorphic--meromorphic incompatibility at a pole}
\label{subsec:holomorphic_meromorphic_incompatibility}

\begin{lemma}[Holomorphic--meromorphic incompatibility]
\label{lem:holomorphic_meromorphic_incompatibility}
\MathTag Let $U\subset\CC$ be open and let $r_0\in U$.
If $F$ is holomorphic on $U$ and $G$ is meromorphic on $U$ with a pole at $r_0$, then $F\neq G$ on $U\setminus\{r_0\}$.
\end{lemma}
\begin{proof}
Assume for contradiction that $F=G$ on $U\setminus\{r_0\}$.
On a small disk centered at $r_0$, $F$ has a Taylor series while $G$ has a Laurent expansion with a nontrivial principal part.
Equality on the punctured disk forces the principal part to vanish, contradicting that $G$ has a pole.
\end{proof}
