\section{Observer Difference and Statistical Equivalence}

\subsection{Local Differences from Different Phases}

\begin{proposition}[Local realization difference]\label{prop:local-difference}
For a general window $W$ and different phases $u\neq u'$, the finite patch collections of $\Lambda(u)$ and $\Lambda(u')$ may differ; thus ``observer difference'' manifests rigorously as patch realization difference.
\end{proposition}

\begin{remark}
This proposition is a direct consequence of geometric construction: $u$ controls which lattice points fall into the window, hence controls the local arrangement of projected points.
\end{remark}

\subsection{Statistical Equivalence under Ergodic/Uniquely Ergodic Conditions}

\begin{theorem}[Patch frequencies phase-independent: uniquely ergodic case]\label{thm:unique-ergodic-frequencies}
If the cut-and-project hull system defined by $(E_{\parallel},E_{\perp},W)$ is uniquely ergodic, then for any patch type $P$, the patch frequency $\mathrm{freq}(P;u)$ exists and is independent of $u$.
\end{theorem}

\begin{remark}
Unique ergodicity ensures time average equals space average and is independent of initial conditions; patch counts can be expressed as orbit averages of continuous (or integrable) functions on the hull, hence the limit and $u$-independence follow from ergodicity (see the argument framework in Appendix \ref{app:smb-conditions}).
\end{remark}

\begin{corollary}[Statistical holography: single realization recovers statistical law]\label{cor:statistical-holography}
Under the conditions of Theorem \ref{thm:unique-ergodic-frequencies}, the long-orbit statistics of a single typical $u$ suffice to recover global statistical laws consistent with the hull's invariant measure; thus ``local difference'' and ``statistical agreement'' coexist in a rigorous sense.
\end{corollary}
