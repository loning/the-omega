
\section{Mathematical notes and auxiliary lemmas}
\label{app:math_notes}

\subsection{Hecke prime-power recursion and prime generation}
\label{app:hecke_recursion}

Starting from the Hecke multiplication relation~\eqref{eq:hecke_multiplication_law}, take $m=p^r$ and $n=p$ for a prime $p$. The divisors of $(p^r,p)$ are $d\in\{1,p\}$, so
\begin{equation}
T_{p^r}T_p
=\sum_{d\mid(p^r,p)} d^{k-1}\,T_{p^{r+1}/d^2}
=T_{p^{r+1}}+p^{k-1}T_{p^{r-1}}.
\end{equation}
Rearranging gives the prime-power recursion~\eqref{eq:hecke_prime_power}:
\begin{equation}
T_{p^{r+1}}=T_pT_{p^r}-p^{k-1}T_{p^{r-1}}.
\end{equation}
This shows that all $T_{p^r}$ are generated by $T_p$. Together with~\eqref{eq:hecke_multiplication_law} and coprime multiplicativity, it implies that the full Hecke algebra is generated by $\{T_p\}_{p\ \mathrm{prime}}$.

\subsection{Star discrepancy and the Koksma inequality}
\label{app:koksma_inequality}

For a point set $P_N=\{x_0,\dots,x_{N-1}\}\subset[0,1)$, the one-dimensional star discrepancy is
\begin{equation}
D_N^\ast(P_N)=\sup_{0\le u\le 1}\left|\frac{1}{N}\sum_{t=0}^{N-1}\ind_{[0,u)}(x_t)-u\right|.
\end{equation}

\begin{definition}[variation for complex-valued functions]
\label{def:complex_variation}
For a complex-valued function $f:[0,1]\to\CC$, define its (Jordan) variation by
\begin{equation}
\Var(f):=\Var(\Re f)+\Var(\Im f),
\end{equation}
where $\Var(\Re f)$ and $\Var(\Im f)$ are the usual real-valued Jordan variations.
\end{definition}

\begin{theorem}[Koksma inequality, 1D]
\label{thm:koksma_inequality_1d}
If $f:[0,1]\to\CC$ has bounded variation (Jordan variation) $\Var(f)<\infty$, then
\begin{equation}
\left|\frac{1}{N}\sum_{t=0}^{N-1} f(x_t)-\int_0^1 f(x)\,\dd x\right|
\le \Var(f)\,D_N^\ast(P_N).
\end{equation}
\end{theorem}
\begin{remark}
See \cite{Koksma1935,KuipersNiederreiter1974} for classical statements and proofs.
With Definition~\ref{def:complex_variation}, the complex-valued inequality follows immediately by applying the real-valued Koksma inequality to $\Re f$ and $\Im f$ and using the triangle inequality:
\begin{equation}
\left|\Delta(f)\right|
\le \left|\Delta(\Re f)\right|+\left|\Delta(\Im f)\right|
\le \big(\Var(\Re f)+\Var(\Im f)\big)D_N^\ast(P_N)
=\Var(f)\,D_N^\ast(P_N),
\end{equation}
where $\Delta(g)=\frac{1}{N}\sum g(x_t)-\int_0^1 g$.
\end{remark}

\subsection{Denjoy--Koksma at convergent lengths}
\label{app:denjoy_koksma}

Let $\alpha=[0;a_1,a_2,\dots]$ be irrational with convergents $p_n/q_n$. For the irrational rotation $R_\alpha(x)=x+\alpha\pmod 1$, the Denjoy--Koksma inequality controls Birkhoff sums at convergent lengths \cite{Denjoy1932,Koksma1935,KatokHasselblatt1995}.

\begin{theorem}[Denjoy--Koksma inequality (rotation)]
\label{thm:denjoy_koksma}
If $f:[0,1]\to\RR$ has bounded variation and $\int_0^1 f(x)\,\dd x=0$, then for every $n$ and all $x$,
\begin{equation}
\left|\sum_{t=0}^{q_n-1} f(x+t\alpha)\right|\le \Var(f).
\end{equation}
\end{theorem}

\subsection{Ostrowski block decomposition and the digit bound}
\label{app:dk_koksma_ostrowski}

Write $N$ in its Ostrowski expansion $N=\sum_{j=0}^m b_j q_j$ associated with $\alpha$. Then the orbit segment $\{x+t\alpha\}_{t=0}^{N-1}$ can be partitioned into blocks of consecutive lengths $q_j$ repeated $b_j$ times, up to shifts of the starting point. Applying Theorem~\ref{thm:denjoy_koksma} to each block yields a standard digit bound for Birkhoff sums:
\begin{equation}
\left|\sum_{t=0}^{N-1} f(x+t\alpha) - N\int_0^1 f\right|
\le \Var(f)\sum_{j=0}^m b_j.
\end{equation}

Applying this to interval indicator functions and combining with Theorem~\ref{thm:koksma_inequality_1d} yields the discrepancy bound used in Corollary~\ref{cor:ostrowski_digit_discrepancy}. Concretely, for $f_u(x)=\ind_{[0,u)}(x)-u$ we have $\int_0^1 f_u=0$ and $\Var(f_u)\le 2$, hence
\begin{equation}
\left|\frac{1}{N}\sum_{t=0}^{N-1}\ind_{[0,u)}(x+t\alpha)-u\right|
\le \frac{2}{N}\sum_{j=0}^m b_j,
\end{equation}
and taking the supremum over $u\in[0,1]$ gives~\eqref{eq:ostrowski_discrepancy_bound}.

\paragraph{A constant-type corollary (explicit \texorpdfstring{$(\log N)/N$}{(log N)/N} rate).}
If $\alpha=[0;a_1,a_2,\dots]$ has bounded partial quotients $a_n\le A$, then Ostrowski digits satisfy $b_j\le a_{j+1}\le A$, hence $\sum_{j=0}^{m} b_j\le A(m+1)$.
Moreover, the convergent denominators satisfy $q_{n+1}=a_{n+1}q_n+q_{n-1}\ge q_n+q_{n-1}$, so $q_n\ge F_n$ for Fibonacci numbers $F_n$. If $q_m\le N<q_{m+1}$, then $F_m\le N$, hence $m\le 2+\log_\varphi N$.
Combining these estimates with~\eqref{eq:ostrowski_discrepancy_bound} yields
\begin{equation}
D_N^\ast(P_N)\le \frac{2A(2+\log_\varphi N)}{N},
\end{equation}
recovering the standard $O((\log N)/N)$ discrepancy rate for constant-type rotations (cf.\ \cite{KuipersNiederreiter1974}).

\subsection{Exact star discrepancy at convergent lengths (closed form)}
\label{app:convergent_star_discrepancy}

\begin{proposition}[Convergent-length star discrepancy for a Kronecker orbit]
\label{prop:convergent_star_discrepancy}
Let $\alpha\in(0,1)\setminus\QQ$ and let $p/q$ be a reduced rational satisfying
\begin{equation}
0<\left|\alpha-\frac{p}{q}\right|<\frac{1}{q^2}.
\end{equation}
Consider the Kronecker point set
\begin{equation}
P_q(\alpha):=\bigl\{\{t\alpha\}: t=0,1,\ldots,q-1\bigr\}\subset[0,1).
\end{equation}
Then the one-dimensional star discrepancy satisfies
\begin{equation}
D_q^*\!\left(P_q(\alpha)\right)=
\begin{cases}
\dfrac{1}{q}, & \dfrac{p}{q}<\alpha,\\[6pt]
\dfrac{1}{q}+(q-1)\!\left(\dfrac{p}{q}-\alpha\right), & \dfrac{p}{q}>\alpha.
\end{cases}
\end{equation}
\end{proposition}
\begin{proof}
Write $\delta:=\alpha-\frac{p}{q}$ and note that for $0\le t\le q-1$ one has $|t\delta|<(q-1)/q^2<1/q$.
Let $r_t\in\{0,1,\ldots,q-1\}$ be the residue $r_t\equiv tp\ (\mathrm{mod}\ q)$; since $\gcd(p,q)=1$, the map $t\mapsto r_t$ is a permutation.

\paragraph{Case 1: $\delta>0$ (i.e.\ $p/q<\alpha$).}
For each $t$,
\begin{equation}
\{t\alpha\}=\left\{\frac{tp}{q}+t\delta\right\}= \frac{r_t}{q}+t\delta \in \left[\frac{r_t}{q},\frac{r_t+1}{q}\right),
\end{equation}
where the last inclusion uses $0\le t\delta<1/q$ and $r_t=q-1$ is interpreted modulo $1$.
Hence each interval $[j/q,(j+1)/q)$ contains exactly one point of $P_q(\alpha)$, which implies $D_q^*(P_q(\alpha))\le 1/q$.
Since $0\in P_q(\alpha)$, one also has $D_q^*(P_q(\alpha))\ge 1/q$ (take $u\downarrow 0$), so equality holds.

\paragraph{Case 2: $\delta<0$ (i.e.\ $p/q>\alpha$).}
Let $\delta':=-\delta=\frac{p}{q}-\alpha>0$. For $t\ge 1$, $r_t\neq 0$ so $r_t/q\ge 1/q$, and
\begin{equation}
\{t\alpha\}=\left\{\frac{tp}{q}-t\delta'\right\}=\frac{r_t}{q}-t\delta' \in \left(\frac{r_t-1}{q},\frac{r_t}{q}\right),
\end{equation}
since $0<t\delta'<1/q$.
Thus every interval $[j/q,(j+1)/q)$ with $j=0,\ldots,q-2$ contains exactly one point coming from the unique $t\in\{1,\ldots,q-1\}$ with $r_t=j+1$, while the point $0\in P_q(\alpha)$ lies in $[0,1/q)$.
The maximal discrepancy occurs at the right endpoints and equals $\frac{1}{q}+(q-1)\delta'$.
\end{proof}

\subsection{Golden branch: Fibonacci convergents and the \texorpdfstring{$1+1/\sqrt{5}$}{1+1/sqrt(5)} constant}
\label{app:golden_constant}

Let $\varphi=(1+\sqrt{5})/2$ and $\alpha=\varphi^{-1}$. The convergents of $\alpha$ are Fibonacci ratios $\frac{p_n}{q_n}=\frac{F_{n-1}}{F_n}$.
Proposition~\ref{prop:convergent_star_discrepancy} implies that for odd $n$ (when $p_n/q_n<\alpha$) one has $D_{F_n}^*=1/F_n$, while for even $n$ (when $p_n/q_n>\alpha$) one has
\begin{equation}
D_{F_n}^*=\frac{1}{F_n}+(F_n-1)\!\left(\frac{F_{n-1}}{F_n}-\alpha\right).
\end{equation}
Using Binet's formula and the asymptotic $F_n\sim \varphi^n/\sqrt{5}$, one obtains
\begin{equation}
F_n^2\left(\frac{F_{n-1}}{F_n}-\alpha\right)\to \frac{1}{\sqrt{5}}
\qquad (n\to\infty,\ n\ \mathrm{even}),
\end{equation}
and therefore
\begin{equation}
F_n\,D_{F_n}^*\to 1+\frac{1}{\sqrt{5}}
\qquad (n\to\infty,\ n\ \mathrm{even}),
\end{equation}
which is the constant observed in discrepancy scaling on the golden branch.

\subsection{A crude variation bound for the sliced integrand}
\label{app:variation_bound_gny}

In Theorem~\ref{thm:finiteN_coeff_bound}, the bounded-variation assumption is imposed on
\begin{equation}
g_{n,y}(x)=F_y(x)\,\e^{-2\pi\iu n x}=f(x+\iu y)\,\e^{-2\pi\iu n x}.
\end{equation}
In concrete modular-form settings, $F_y$ is smooth in $x$ for every fixed $y>0$ and admits termwise differentiation under absolute convergence of~\eqref{eq:qexp_general}. A crude bound is obtained from total variation via an $L^1$ derivative estimate when $g_{n,y}$ is absolutely continuous:
\begin{equation}
\Var(g_{n,y})\le \int_0^1 |g_{n,y}'(x)|\,\dd x
\le \int_0^1 |F_y'(x)|\,\dd x + 2\pi n \int_0^1 |F_y(x)|\,\dd x.
\end{equation}
For specific choices of $f$ (e.g.\ $E_4,E_6,\Delta,j$), the $L^1$ norms above can be bounded using coefficient growth and geometric tail bounds in $|q|=\e^{-2\pi y}$.

\subsection{Uniform convergence on slices and sufficient conditions for coefficient recovery}
\label{app:qexp_uniform_convergence}

The slice--projection identity of Theorem~\ref{thm:slice_coefficient_reconstruction} is stated under an absolute-uniform convergence assumption at fixed $y>0$. For the modular objects used in this manuscript, that assumption follows from standard coefficient-growth bounds.

\begin{proposition}[polynomial growth implies uniform slice convergence]
\label{prop:poly_growth_uniform_slice}
Let $(a_n)_{n\ge 0}$ satisfy a polynomial growth bound $|a_n|\le C(1+n)^A$ for some constants $C,A>0$, and define
\begin{equation}
f(\tau)=\sum_{n\ge 0} a_n \e^{2\pi \iu n \tau},\qquad \tau=x+\iu y.
\end{equation}
Then for every $y_0>0$ the series for $f$ converges absolutely and uniformly on $\{(x,y): x\in[0,1],\ y\ge y_0\}$. In particular, for each fixed $y>0$, the slice $F_y(x)=f(x+\iu y)$ admits termwise integration on $[0,1]$ and Theorem~\ref{thm:slice_coefficient_reconstruction} applies.
\end{proposition}
\begin{proof}
For $y\ge y_0$ one has $|\e^{2\pi \iu n\tau}|=\e^{-2\pi n y}\le \e^{-2\pi n y_0}$. Thus
\begin{equation}
|a_n \e^{2\pi \iu n\tau}|\le C(1+n)^A \e^{-2\pi n y_0}.
\end{equation}
Since $\sum_{n\ge 0}(1+n)^A \e^{-2\pi n y_0}<\infty$, the Weierstrass $M$-test gives absolute uniform convergence on the stated domain. Termwise integration on $[0,1]$ follows from uniform absolute convergence at fixed $y>0$.
\end{proof}

\begin{remark}[bounded variation via Fourier coefficient sums]
Under the same polynomial bound, the derivative series with respect to $x$ also converges absolutely and uniformly for $y\ge y_0$ because an extra factor $n$ preserves summability against $\e^{-2\pi n y_0}$. Hence $F_y$ is $C^1$ in $x$, and $g_{n,y}(x)=F_y(x)\e^{-2\pi\iu n x}$ is absolutely continuous. Moreover, if $g_{n,y}(x)=\sum_{m\in\ZZ} c_m \e^{2\pi\iu m x}$ is its Fourier series, then
\begin{equation}
\Var(g_{n,y})=\int_0^1 |g_{n,y}'(x)|\,\dd x
\le 2\pi \sum_{m\in\ZZ} |m|\,|c_m|.
\end{equation}
For $q$-expansions with nonnegative indices, one has $c_{k-n}=a_k \e^{-2\pi k y}$, so $\Var(g_{n,y})$ can be bounded explicitly in terms of coefficient-growth majorants, as in Appendix~\ref{app:variation_bound_e4_explicit}.
\end{remark}

\begin{remark}[modular-form coefficient growth (standard input)]
For holomorphic modular forms of level~$1$, Fourier coefficients have polynomial growth. For Eisenstein series, one has $a_n=O(n^{k-1})$ (e.g.\ \cite{Apostol1990,DiamondShurman2005}). For cusp forms, Deligne's theorem (Ramanujan--Petersson) implies $a_n=O_\varepsilon(n^{(k-1)/2+\varepsilon})$ (e.g.\ \cite{Deligne1974,Serre1973}). Thus Proposition~\ref{prop:poly_growth_uniform_slice} applies to the modular-form objects used in the slice--sampling experiments.
\end{remark}

\subsection{Deligne/Hecke bounds for Ramanujan's \texorpdfstring{$\tau(n)$}{tau(n)} (explicit majorants)}
\label{app:tau_bounds}

Let $\Delta(\tau)=\sum_{n\ge 1}\tau(n)q^n$ denote the normalized weight-$12$ Hecke eigen cusp form on $\SL_2(\ZZ)$.

\begin{proposition}[Ramanujan--Petersson/Deligne bound (standard)]
\label{prop:deligne_tau_bound}
For every prime $p$ one has
\begin{equation}
|\tau(p)|\le 2p^{11/2}.
\end{equation}
\end{proposition}
\begin{remark}
This is a special case of Deligne's proof of the Weil conjectures. See \cite{Deligne1974,Deligne1971Bourbaki} for the Ramanujan--Petersson bounds for Hecke eigenvalues of modular forms.
\end{remark}

\begin{proposition}[global coefficient bound via Hecke multiplicativity]
\label{prop:tau_global_bound}
For every $n\ge 1$,
\begin{equation}
|\tau(n)|\le d(n)\,n^{11/2},
\end{equation}
where $d(n)$ is the divisor-counting function. In particular, using the trivial estimate $d(n)\le 2\sqrt{n}$ one obtains the explicit crude majorant
\begin{equation}
|\tau(n)|\le 2n^6,\qquad n\ge 1.
\end{equation}
\end{proposition}
\begin{proof}
For prime powers one has the Hecke recursion
\begin{equation}
\tau(p^{r+1})=\tau(p)\tau(p^r)-p^{11}\tau(p^{r-1}),\qquad r\ge 1,
\end{equation}
so the local Euler factor is $1-\tau(p)p^{-s}+p^{11}p^{-2s}$ and the standard bound $|\tau(p^r)|\le (r+1)p^{11r/2}$ follows from Proposition~\ref{prop:deligne_tau_bound} (see, e.g., \cite{Serre1973,DiamondShurman2005}).
Multiplicativity on coprime integers then yields $|\tau(n)|\le d(n)n^{11/2}$ for general $n$.
Finally, $d(n)\le 2\sqrt{n}$ is immediate from pairing divisors $d\leftrightarrow n/d$.
\end{proof}

\subsection{An explicit \texorpdfstring{$\Var(g_{n,y})$}{Var(g\_n,y)} bound for \texorpdfstring{$E_4$}{E4}}
\label{app:variation_bound_e4_explicit}

For the weight-$4$ Eisenstein series
\begin{equation}
E_4(\tau)=1+240\sum_{k\ge 1}\sigma_3(k)\,q^k,\qquad q=\e^{2\pi\iu\tau},
\end{equation}
define, as in Section~\ref{sec:cusp_qexp_slice_sampling},
\begin{equation}
g_{n,y}(x)=E_4(x+\iu y)\,\e^{-2\pi\iu n x}.
\end{equation}
Fix $y>0$ and write $r=\e^{-2\pi y}=|q|\in(0,1)$. Then $g_{n,y}$ has an absolutely convergent Fourier expansion
\begin{equation}
g_{n,y}(x)=\sum_{m\in\ZZ} c_m\,\e^{2\pi\iu m x},
\qquad
c_{k-n}=
\begin{cases}
r^0, & k=0,\\
240\,\sigma_3(k)\,r^k, & k\ge 1,
\end{cases}
\end{equation}
and is absolutely continuous. Hence $\Var(g_{n,y})=\int_0^1|g_{n,y}'(x)|\,\dd x$.
Moreover,
\begin{equation}
g_{n,y}'(x)=2\pi\iu \sum_{m\in\ZZ} m c_m\,\e^{2\pi\iu m x},
\qquad
|g_{n,y}'(x)|\le 2\pi \sum_{m\in\ZZ} |m|\,|c_m|,
\end{equation}
so
\begin{equation}
\label{eq:var_bound_by_coeff_sum}
\Var(g_{n,y})
\le 2\pi \sum_{m\in\ZZ} |m|\,|c_m|
 = 2\pi\left(n + 240\sum_{k\ge 1}|k-n|\sigma_3(k)\,r^k\right).
\end{equation}
Using the classical bound $\sigma_3(k)\le \zeta(3)\,k^3$ (e.g.\ \cite{Apostol1990}), we obtain the explicit estimate
\begin{equation}
\label{eq:var_e4_explicit_bound}
\Var(g_{n,y})
\le 2\pi\left(
n
 +240\,\zeta(3)\sum_{k\ge 1}(k+n)\,k^3\,r^k
\right)
=2\pi\left(
n
 +240\,\zeta(3)\big(S_4(r)+nS_3(r)\big)
\right),
\end{equation}
where
\begin{equation}
S_3(r)=\sum_{k\ge 1}k^3r^k=\frac{r(1+4r+r^2)}{(1-r)^4},
\qquad
S_4(r)=\sum_{k\ge 1}k^4r^k=\frac{r(1+11r+11r^2+r^3)}{(1-r)^5}.
\end{equation}
Combining~\eqref{eq:var_e4_explicit_bound} with Theorem~\ref{thm:finiteN_coeff_bound} yields a fully explicit coefficient-recovery bound for $E_4$, which is audited numerically in Table~\ref{tab:e4_slice_sampling_bound}.

\subsection{An explicit \texorpdfstring{$\Var(g_{n,y})$}{Var(g\_n,y)} bound for \texorpdfstring{$E_6$}{E6}}
\label{app:variation_bound_e6_explicit}

For the weight-$6$ Eisenstein series
\begin{equation}
E_6(\tau)=1-504\sum_{k\ge 1}\sigma_5(k)\,q^k,\qquad q=\e^{2\pi\iu\tau},
\end{equation}
define
\begin{equation}
g_{n,y}(x)=E_6(x+\iu y)\,\e^{-2\pi\iu n x}.
\end{equation}
Fix $y>0$ and write $r=\e^{-2\pi y}=|q|\in(0,1)$. As in the $E_4$ case, $g_{n,y}$ is absolutely continuous and
\begin{equation}
\Var(g_{n,y})\le 2\pi\left(n+504\sum_{k\ge 1}|k-n|\,\sigma_5(k)\,r^k\right).
\end{equation}
Using $\sigma_5(k)\le \zeta(5)\,k^5$ (e.g.\ \cite{Apostol1990}) and $|k-n|\le k+n$, we obtain
\begin{equation}
\label{eq:var_e6_explicit_bound}
\Var(g_{n,y})
\le 2\pi\left(n+504\,\zeta(5)\sum_{k\ge 1}(k+n)\,k^5\,r^k\right)
=2\pi\left(n+504\,\zeta(5)\big(S_6(r)+nS_5(r)\big)\right),
\end{equation}
where
\begin{align}
S_5(r)&=\sum_{k\ge 1}k^5 r^k=\frac{r(1+26r+66r^2+26r^3+r^4)}{(1-r)^6},\\
S_6(r)&=\sum_{k\ge 1}k^6 r^k=\frac{r(1+57r+302r^2+302r^3+57r^4+r^5)}{(1-r)^7}.
\end{align}
Combining~\eqref{eq:var_e6_explicit_bound} with Theorem~\ref{thm:finiteN_coeff_bound} yields a fully explicit coefficient-recovery bound for $E_6$, which is audited numerically in Table~\ref{tab:e6_slice_sampling_bound}.

\subsection{Tail bounds for Eisenstein \texorpdfstring{$q$}{q}-series (explicit constants)}
\label{app:eisenstein_tail_bounds}

For level~$1$, the Eisenstein series admit the classical $q$-expansions (e.g.\ \cite{Apostol1990,DiamondShurman2005})
\begin{equation}
E_4(\tau)=1+240\sum_{n\ge 1}\sigma_3(n)\,q^n,
\qquad
E_6(\tau)=1-504\sum_{n\ge 1}\sigma_5(n)\,q^n,
\qquad q=\e^{2\pi\iu\tau}.
\end{equation}
Fix $|q|=r\in(0,1)$ and define the truncations $E_{4}^{(N)}=1+240\sum_{n=1}^{N}\sigma_3(n)q^n$ and $E_{6}^{(N)}=1-504\sum_{n=1}^{N}\sigma_5(n)q^n$.
Using the divisor-sum bound $\sigma_k(n)\le \zeta(k)\,n^k$ (e.g.\ \cite{Apostol1990}), we obtain explicit tail bounds.

\begin{proposition}[explicit tails for $E_4$ and $E_6$]
\label{prop:eisenstein_tail_bounds}
Let $r\in(0,1)$ and $N\ge 1$. Then
\begin{align}
\label{eq:e4_tail_bound}
|E_4(\tau)-E_4^{(N)}(\tau)|
&\le 240\,\zeta(3)\,\sum_{n=N+1}^\infty n^3 r^n
\\
&\le 240\,\zeta(3)\,r^{N+1}\Bigg(
\frac{(N+1)^3}{1-r}
 +\frac{3(N+1)^2 r}{(1-r)^2}
\\
&\qquad +\frac{3(N+1) r(1+r)}{(1-r)^3}
 +\frac{r(1+4r+r^2)}{(1-r)^4}
\Bigg), \notag
\end{align}
and
\begin{align}
\label{eq:e6_tail_bound}
|E_6(\tau)-E_6^{(N)}(\tau)|
&\le 504\,\zeta(5)\,\sum_{n=N+1}^\infty n^5 r^n
\;\le\;
504\,\zeta(5)\,r^{N+1}\Bigg(
\frac{(N+1)^5}{1-r}
\notag\\
&\qquad +\frac{5(N+1)^4 r}{(1-r)^2}
\notag\\
&\qquad +\frac{10(N+1)^3 r(1+r)}{(1-r)^3}
\notag\\
&\qquad +\frac{10(N+1)^2 r(1+4r+r^2)}{(1-r)^4}
\notag\\
&\qquad +\frac{5(N+1) r(1+11r+11r^2+r^3)}{(1-r)^5}
\notag\\
&\qquad +\frac{r(1+26r+66r^2+26r^3+r^4)}{(1-r)^6}
\Bigg),
\end{align}
\end{proposition}

\begin{remark}
The bounds~\eqref{eq:e4_tail_bound} and~\eqref{eq:e6_tail_bound} are obtained by expanding $(N+1+m)^p$ and using the standard generating functions for $\sum_{m\ge 0} m^j r^m$. In particular,
\begin{align}
\sum_{m\ge 0} m^4 r^m &= \frac{r(1+11r+11r^2+r^3)}{(1-r)^5},\\
\sum_{m\ge 0} m^5 r^m &= \frac{r(1+26r+66r^2+26r^3+r^4)}{(1-r)^6},
\end{align}
which appear explicitly in the $E_6$ tail bound.
\end{remark}

\subsection{Gauss measure mean of the roof function}
\label{app:gauss_roof_mean}

Let $\mu$ denote the Gauss invariant measure for the Gauss map,
\begin{equation}
\dd\mu(\xi)=\frac{1}{\log 2}\frac{\dd\xi}{1+\xi},\qquad \xi\in(0,1).
\end{equation}
The roof function in Theorem~\ref{thm:series_suspension} is $r(\xi)=-2\log\xi$.

\begin{proposition}[closed-form mean]
\label{prop:gauss_roof_mean}
One has
\begin{equation}
\int_0^1 (-\log \xi)\,\dd\mu(\xi)=\frac{\pi^2}{12\log 2},
\qquad
\int_0^1 r(\xi)\,\dd\mu(\xi)=\frac{\pi^2}{6\log 2}.
\end{equation}
\end{proposition}
\begin{proof}
Expand $(1+\xi)^{-1}=\sum_{n=0}^\infty (-1)^n\xi^n$ for $\xi\in(0,1)$ and integrate termwise:
\begin{align}
\int_0^1 \frac{-\log \xi}{1+\xi}\,\dd\xi
&=\sum_{n=0}^\infty (-1)^n\int_0^1 \xi^n(-\log\xi)\,\dd\xi
=\sum_{n=0}^\infty (-1)^n\frac{1}{(n+1)^2}
\\
&=\sum_{m=1}^\infty \frac{(-1)^{m-1}}{m^2}
=(1-2^{1-2})\zeta(2)
=\frac{1}{2}\cdot\frac{\pi^2}{6}
=\frac{\pi^2}{12}.
\end{align}
Dividing by $\log 2$ gives the first identity, and the second follows by multiplying by $2$.
\end{proof}

\subsection{Derivation of the Gauss digit law}
\label{app:gauss_digit_law_derivation}

Write the continued fraction of $\xi\in(0,1)$ as $\xi=[0;a_1,a_2,\dots]$ so that $a_1=\lfloor 1/\xi\rfloor$.

\begin{proposition}[Gauss digit probabilities]
\label{prop:gauss_digit_law}
For the Gauss invariant measure $\dd\mu(\xi)=\frac{1}{\log 2}\frac{\dd\xi}{1+\xi}$ one has, for every $k\ge 1$,
\begin{equation}
\mu(a_1=k)=\log_2\!\left(1+\frac{1}{k(k+2)}\right).
\end{equation}
\end{proposition}
\begin{proof}
The event $\{a_1=k\}$ is equivalent to $\frac{1}{k+1}<\xi\le \frac{1}{k}$. Therefore
\begin{align}
\mu(a_1=k)
&=\frac{1}{\log 2}\int_{1/(k+1)}^{1/k}\frac{\dd\xi}{1+\xi}
=\frac{1}{\log 2}\left[\log(1+\xi)\right]_{1/(k+1)}^{1/k}
\\
&=\frac{1}{\log 2}\log\!\left(\frac{1+1/k}{1+1/(k+1)}\right)
=\frac{1}{\log 2}\log\!\left(\frac{(k+1)^2}{k(k+2)}\right)
=\log_2\!\left(1+\frac{1}{k(k+2)}\right).
\end{align}
\end{proof}

\subsection{Gauss--Kuzmin convergence (standard exponential relaxation)}
\label{app:gauss_kuzmin}

\begin{theorem}[Gauss--Kuzmin]
\label{thm:gauss_kuzmin}
Let $G:(0,1)\to(0,1)$ be the Gauss map and let $\mu$ be its invariant probability measure. For a broad class of absolutely continuous initial distributions $\nu$ on $(0,1)$ (e.g.\ densities of bounded variation), there exist constants $C>0$ and $0<\lambda<1$ such that for all $n\ge 1$,
\begin{equation}
\sup_{x\in(0,1]}\left|\nu\!\left(G^{-n}\big((0,x]\big)\right)-\mu\big((0,x]\big)\right|\le C\,\lambda^n.
\end{equation}
\end{theorem}
\begin{remark}
This is the classical Gauss--Kuzmin theorem (and its refinements via transfer-operator spectral gaps). We treat it as a standard input from metrical continued-fraction theory; see \cite{IosifescuKraaikamp2002}.
\end{remark}
