\begin{proposition}[Discrete variational threshold for moment order selection: \texorpdfstring{$q^\star(c)$}{q*(c)} interval characterization under exponential risk budget]\label{prop:pom-qstar-discrete-variational-threshold}
Let \(P(q):=\log r_q\) (integer \(q\ge 1\)), and denote the secant slope
\[
\beta_q:=\frac{P(q)-P(1)}{q-1}
=\frac{\log r_q-\log 2}{q-1}
\qquad(q\ge 2),
\]
where \(r_1=2\).
For exponential risk budget \(\varepsilon_m=e^{-cm}\) (\(c>0\)), define the \(q\)-family of certificate exponents
\[
\gamma^{(q)}(c):=\beta_q+\frac{c}{q-1},
\qquad
\gamma^{\mathrm{cert}}(c):=\inf_{q\in\ZZ_{\ge 2}}\gamma^{(q)}(c),
\qquad
q^\star(c)\in\arg\min_{q\in\ZZ_{\ge 2}}\gamma^{(q)}(c),
\]
which are isomorphic to the finite-\(m\) design formula in Remark \ref{rem:pom-quantile-budget-computable} (taking \(c=\lim_{m\to\infty}\frac1m\log(1/\varepsilon_m)\)).
Then for any optimal moment order satisfying \(q^\star(c)=q\ge 2\), the threshold inequality must hold
\[
\boxed{\ ;
(q-1)(q-2)\bigl(\beta_q-\beta_{q-1}\bigr)\le c\le q(q-1)\bigl(\beta_{q+1}-\beta_{q}\bigr)\;}
\]
where the left-hand side is valid only for \(q\ge 3\) (when \(q=2\) this condition degenerates to vacuum).
Therefore \(c\mapsto q^\star(c)\) is a piecewise-constant, globally monotone non-decreasing discrete phase diagram.
\end{proposition}
\begin{proof}
By definition, \(q^\star(c)=q\) is equivalent to simultaneously satisfying
\(
\gamma^{(q)}(c)\le \gamma^{(q-1)}(c)
\)
and
\(
\gamma^{(q)}(c)\le \gamma^{(q+1)}(c)
\)
(the latter is meaningful for \(q\ge 2\); the former for \(q\ge 3\)).

\smallskip\noindent\textbf{(i) Comparison with \(q-1\) (yields lower bound)}
When \(q\ge 3\),
\[
\beta_q+\frac{c}{q-1}\le \beta_{q-1}+\frac{c}{q-2}
\quad\Longleftrightarrow\quad
\beta_q-\beta_{q-1}\le \frac{c}{(q-1)(q-2)},
\]
i.e., \(c\ge (q-1)(q-2)(\beta_q-\beta_{q-1})\).

\smallskip\noindent\textbf{(ii) Comparison with \(q+1\) (yields upper bound)}
For any \(q\ge 2\),
\[
\beta_q+\frac{c}{q-1}\le \beta_{q+1}+\frac{c}{q}
\quad\Longleftrightarrow\quad
\beta_{q+1}-\beta_q\ge \frac{c}{q(q-1)},
\]
i.e., \(c\le q(q-1)(\beta_{q+1}-\beta_q)\).

\smallskip
Finally, by the discrete convexity of Proposition \ref{prop:pom-pressure-convexity}, \(\beta_q\) is monotone non-decreasing in \(q\) (see the derivation in Remark \ref{rem:pom-quantile-budget-computable}),
thus these thresholds are ordered in the \(q\) direction and yield a piecewise-constant selection spectrum; moreover, when \(c\) increases, the term \(c/(q-1)\) tends to favor larger \(q\), so \(q^\star(c)\) is globally monotone non-decreasing. \qed
\end{proof}
