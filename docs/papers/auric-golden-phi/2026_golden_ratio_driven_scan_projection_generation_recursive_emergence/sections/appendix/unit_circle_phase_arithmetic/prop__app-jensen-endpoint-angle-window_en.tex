\begin{proposition}[Endpoint angle window law: window measure for the high imaginary-part threshold]\label{prop:app-jensen-endpoint-angle-window}
Adopting the notation of Proposition~\ref{prop:app-jensen-defect-upperhalf-circle-average}, let
\[
y(\varrho,\theta):=\Im\,t(\varrho,\theta)=\frac{1-\varrho^2}{1+\varrho^2+2\varrho\cos\theta},
\qquad 0<\varrho<1,\ \theta\in[0,2\pi).
\]
For any threshold \(Y>0\) define the angle window
\[
A_Y(\varrho):=\{\theta\in[0,2\pi): y(\varrho,\theta)\ge Y\}.
\]
and write
\(
y_{-}(\varrho)=\frac{1-\varrho}{1+\varrho}
\)
and
\(
y_{+}(\varrho)=\frac{1+\varrho}{1-\varrho}
\)
(Proposition~\ref{prop:app-cayley-preimage-modulus-circle}).
Then the following piecewise measure formula holds:
\begin{enumerate}
  \item If \(0<Y\le y_{-}(\varrho)\), then \(A_Y(\varrho)=[0,2\pi)\), so \(\mathrm{meas}\,A_Y(\varrho)=2\pi\);
  \item If \(y_{-}(\varrho)<Y<y_{+}(\varrho)\), then \(A_Y(\varrho)\) is a symmetric interval centered at \(\pi\), and
  \[
  \boxed{\;
  \mathrm{meas}\,A_Y(\varrho)=2\Bigl(\pi-\arccos\bigl(c_Y(\varrho)\bigr)\Bigr),\qquad
  c_Y(\varrho):=\frac{(1-\varrho^2)-Y(1+\varrho^2)}{2\varrho Y}\; }.
  \]
  \item If \(Y\ge y_{+}(\varrho)\), then \(A_Y(\varrho)=\varnothing\), so \(\mathrm{meas}\,A_Y(\varrho)=0\).
\end{enumerate}
Furthermore, when \(\varrho=1-\eta\) and \(\eta\downarrow 0\) (equivalently \(\varrho\uparrow 1\)) with \(Y>0\) fixed, one has the boundary-layer asymptotics
\[
\boxed{\;
\mathrm{meas}\,A_Y(\varrho)
=2\sqrt{\frac{2\eta}{Y}}\ \bigl(1+o(1)\bigr)\qquad(\eta\downarrow 0)\; }.
\]
\end{proposition}

\begin{proof}
The condition \(y(\varrho,\theta)\ge Y\) is equivalent to
\[
1-\varrho^2\ \ge\ Y(1+\varrho^2+2\varrho\cos\theta)
\quad\Longleftrightarrow\quad
\cos\theta\ \le\ c_Y(\varrho).
\]
From \(y(\varrho,0)=y_{-}(\varrho)\) and \(y(\varrho,\pi)=y_{+}(\varrho)\) (Proposition~\ref{prop:app-cayley-preimage-modulus-circle}) one sees:
when \(Y\le y_{-}(\varrho)\) the inequality holds for all \(\theta\); when \(Y\ge y_{+}(\varrho)\) there is no solution; and in the intermediate case \(y_{-}(\varrho)<Y<y_{+}(\varrho)\) one has \(c_Y(\varrho)\in(-1,1)\), and the solution set is a symmetric interval of \(\theta\) centered at \(\pi\).
Since the set \(\{\theta\in[0,2\pi):\cos\theta\ge c\}\) has measure \(2\arccos(c)\), taking the complement yields
\(\mathrm{meas}\,A_Y(\varrho)=2\pi-2\arccos(c_Y(\varrho))\), which is the stated closed form.
\par
Finally let \(\varrho=1-\eta\) with \(\eta\downarrow 0\). For fixed \(Y>0\) one has
\(
c_Y(\varrho)=-1+\eta/Y+O(\eta^2)
\).
Using the endpoint expansion \(\arccos(-1+\epsilon)=\pi-\sqrt{2\epsilon}+o(\sqrt{\epsilon})\) (\(\epsilon\downarrow 0\)), one obtains
\[
\mathrm{meas}\,A_Y(\varrho)
=2\bigl(\pi-\arccos(c_Y(\varrho))\bigr)
=2\sqrt{\frac{2\eta}{Y}}\,(1+o(1)).
\]
\qed
\end{proof}

\begin{theorem}[Endpoint angle window rarity exponent and closed-form matching with the min-entropy endpoint constant]\label{thm:app-endpoint-window-hinfty-match}
Let \(h_\infty=\log(2/\sqrt{\varphi})\) be the min-entropy rate endpoint constant of the fold output measure (Corollary~\ref{cor:pom-min-entropy-rate}).
Take a constant \(c>0\), and for \(m\to\infty\) define
\[
Y_m:=\Bigl(\frac{4}{\varphi^{2}}\Bigr)^{m},
\qquad
\varrho_m:=1-\eta_m,\qquad \eta_m:=c\,\varphi^{-m}.
\]
Then as \(m\to\infty\) one has the exponential matching
\[
\mathrm{meas}\,A_{Y_m}(\varrho_m)
=2\sqrt{2c}\,e^{-m h_\infty}\,\bigl(1+o(1)\bigr).
\]
Equivalently, if \(\Theta\sim\mathrm{Unif}[0,2\pi)\), then the rare-event probability satisfies
\[
\mathbb{P}\!\left(\Im\,t(\varrho_m,\Theta)\ge Y_m\right)
=\frac{1}{2\pi}\,\mathrm{meas}\,A_{Y_m}(\varrho_m)
=\frac{\sqrt{2c}}{\pi}\,e^{-m h_\infty}\,\bigl(1+o(1)\bigr).
\]
\end{theorem}

\begin{proof}
From the definitions \(y_\pm(\varrho)=\frac{1\pm\varrho}{1\mp\varrho}\) and \(\eta_m=c\varphi^{-m}\) one obtains
\[
y_-(\varrho_m)\asymp \eta_m,\qquad
y_+(\varrho_m)\asymp \eta_m^{-1}\asymp \varphi^{m}.
\]
On the other hand \(Y_m=(4/\varphi^2)^m\), and
\(
Y_m/y_+(\varrho_m)\asymp (4/\varphi^3)^m\to 0
\), so for sufficiently large \(m\) one must have \(y_-(\varrho_m)<Y_m<y_+(\varrho_m)\), and the intermediate-case closed form of Proposition~\ref{prop:app-jensen-endpoint-angle-window} can be used:
\[
\mathrm{meas}\,A_{Y_m}(\varrho_m)
=2\Bigl(\pi-\arccos\bigl(c_{Y_m}(\varrho_m)\bigr)\Bigr),
\qquad
c_Y(\varrho):=\frac{(1-\varrho^2)-Y(1+\varrho^2)}{2\varrho Y}.
\]
Set \(\varrho_m=1-\eta_m\). Direct expansion gives
\[
c_{Y_m}(\varrho_m)
=-1+\frac{\eta_m}{Y_m}+O(\eta_m^2),
\]
and since
\(
\eta_m^2/(\eta_m/Y_m)=\eta_m Y_m=c(4/\varphi^3)^m\to 0
\), one may also write
\(
c_{Y_m}(\varrho_m)=-1+\frac{\eta_m}{Y_m}\,(1+o(1))
\).
Using the endpoint expansion \(\arccos(-1+\epsilon)=\pi-\sqrt{2\epsilon}+o(\sqrt\epsilon)\) (\(\epsilon\downarrow 0\)), one obtains
\[
\mathrm{meas}\,A_{Y_m}(\varrho_m)
=2\sqrt{\frac{2\eta_m}{Y_m}}\,(1+o(1)).
\]
Substituting \(\eta_m=c\varphi^{-m}\) and \(Y_m=(4/\varphi^2)^m\) gives
\[
\mathrm{meas}\,A_{Y_m}(\varrho_m)
=2\sqrt{2c}\Bigl(\frac{\sqrt{\varphi}}{2}\Bigr)^m(1+o(1))
=2\sqrt{2c}\,e^{-m\log(2/\sqrt{\varphi})}(1+o(1)).
\]
The final step uses \(h_\infty=\log(2/\sqrt{\varphi})\) (Corollary~\ref{cor:pom-min-entropy-rate}) to obtain the stated result.
\end{proof}

\begin{remark}[Endpoint angle window readout at the window-\(6\) anchor point]\label{rem:app-endpoint-window-window6}
Taking \(m=6\), the above theorem gives the rare-measure exponential rate of the endpoint angle window at threshold \(Y_6=(4/\varphi^2)^6\) and radius \(\varrho_6=1-c\varphi^{-6}\):
\[
\mathrm{meas}\,A_{Y_6}(\varrho_6)
=2\sqrt{2c}\,e^{-6h_\infty}\,\bigl(1+o(1)\bigr),
\qquad
\mathbb{P}\!\left(\Im\,t(\varrho_6,\Theta)\ge Y_6\right)
=\frac{\sqrt{2c}}{\pi}\,e^{-6h_\infty}\,\bigl(1+o(1)\bigr).
\]
Therefore, at the minimal channel anchor point of window-\(6\), the min-entropy endpoint constant \(h_\infty\) can be directly translated as the exponential decay rate of the endpoint horizon angle window rare event.
\end{remark}

\begin{theorem}[Defect upper bound implies explicit upper bound on off-critical-line shift]\label{thm:app-jensen-defect-controls-offcritical-shift}
Fix \(0<\varrho<1\).
If there exists \(\varepsilon\ge 0\) such that \(D_{\zeta}(\varrho)\le \varepsilon\), let
\[
r_0:=\varrho e^{-\varepsilon}\in(0,1).
\]
Then for any zero \(\rho=\beta+\mathrm{i}\gamma\) satisfying \(\beta<\tfrac12\) (if such exists), its off-critical-line shift \(\delta:=\tfrac12-\beta>0\) and disk radius \(r=|w_{\rho}|\) (Proposition~\ref{prop:app-offcritical-radius-compression}) satisfy
\[
r=|w_{\rho}|\ \ge\ r_0.
\]
Moreover, under the condition
\[
1-\frac{(1-r_0^2)^2}{(1+r_0^2)^2}\,(\gamma^2+1)\ \ge\ 0
\]
(so that the following square root is real), one has the explicit upper bound
\[
\boxed{\;
\delta\ \le\ \Delta(\gamma;r_0),\qquad
\Delta(\gamma;r)
:=\frac{1+r^{2}}{1-r^{2}}
\left(1-\sqrt{1-\frac{(1-r^{2})^{2}}{(1+r^{2})^{2}}(\gamma^{2}+1)}\right)\; }.
\]
\end{theorem}

\begin{proof}
By Corollary~\ref{cor:app-jensen-visible-threshold-offcritical}, when \(|w_{\rho}|<\varrho\) one has
\(
D_{\zeta}(\varrho)\ge \log(\varrho/|w_{\rho}|)
\);
combining with \(D_{\zeta}(\varrho)\le \varepsilon\) yields \(|w_{\rho}|\ge \varrho e^{-\varepsilon}=r_0\).
When \(|w_{\rho}|\ge \varrho\) the above lower bound does not apply, but \(|w_{\rho}|\ge \varrho\ge r_0\) still holds, so in all cases \(r\ge r_0\).
\par
On the other hand, Proposition~\ref{prop:app-offcritical-radius-compression} gives the closed-form relation
\[
r^{2}=\frac{\gamma^{2}+(\delta-1)^{2}}{\gamma^{2}+(\delta+1)^{2}}.
\]
For fixed \(\gamma\), the right-hand side is strictly decreasing in \(\delta\) on \([0,\sqrt{\gamma^{2}+1}]\), so \(r\ge r_0\) implies \(\delta\le \delta_0\), where \(\delta_0\) is the (smaller) real root of the equation at \(r=r_0\).
Solving this equation explicitly for \(\delta\) (taking the smaller root) yields the stated \(\Delta(\gamma;r_0)\). \qed
\end{proof}

\begin{corollary}[Boundary-layer expansion: leading-order upper bound as \texorpdfstring{$r_0\uparrow 1$}{r0->1}]\label{cor:app-jensen-offcritical-shift-boundary}
Under the conditions of Theorem~\ref{thm:app-jensen-defect-controls-offcritical-shift}, if furthermore \(r_0\uparrow 1\) with \(\gamma\) fixed, then one has the asymptotic expansion
\[
\Delta(\gamma;r_0)
=\frac{\gamma^2+1}{2}\cdot\frac{1-r_0^2}{1+r_0^2}
+O\!\left((\gamma^2+1)^2(1-r_0^2)^2\right)
\qquad(r_0\uparrow 1).
\]
\end{corollary}

\begin{proof}
As \(r_0\uparrow 1\),
\(
q:=\frac{(1-r_0^2)^2}{(1+r_0^2)^2}(\gamma^2+1)\to 0
\).
Substituting
\(
\sqrt{1-q}=1-\frac{q}{2}+O(q^2)
\)
into the closed form for \(\Delta(\gamma;r_0)\) and simplifying yields the stated expansion. \qed
\end{proof}
