即完成化后的系数在单位圆上为实数。
\end{corollary}

\begin{corollary}[有限长度的精确镜像对称]\label{cor:primitive-finite-sym}
令 $\pi_{n,k}:=p_{n,k}/\sum_j p_{n,j}$ 为长度 $n$ 的 primitive 轨道在 $k$-分层上的归一化分布，则

\sum_k k\,\pi_{n,k}=\frac{n}{2},\qquad
\sum_k\left(k-\frac{n}{2}\right)^{2m+1}\pi_{n,k}=0\quad(m\ge 0).

\end{corollary}

\begin{corollary}[奇长度 primitive 的整除律]\label{cor:primitive-odd-div}

\end{corollary}

\subsubsection{完成化行列式与严格偶性}\label{app:kernel-completion}
令 $u=e^\theta$，定义完成化行列式
\[
\widetilde{\Delta}(z,\theta):=\Delta(e^{-\theta/2}z,e^\theta),
\qquad
\widetilde{\zeta}(z,\theta):=\widetilde{\Delta}(z,\theta)^{-1}.
\]
由 $\Delta(z,u)=\Delta(uz,1/u)$ 得
\[
\widetilde{\Delta}(z,-\theta)=\widetilde{\Delta}(z,\theta),
\qquad
\widetilde{\zeta}(z,-\theta)=\widetilde{\zeta}(z,\theta),
\]
即完成化后在参数 $\theta$ 上呈严格偶性。

\subsubsection{自对偶 \texorpdfstring{$\Delta(z,u)=\Delta(uz,1/u)$}{Delta(z,u)=Delta(uz,1/u)} 的 s->1-s 函数方程}\label{app:kernel-functional-equation}
本小节把二变量自对偶对称压缩为 Dirichlet 变量下的一条单变量函数方程，从而与经典 $\zeta(s)$ 叙事中的“完成化/函数方程/临界线”三件套对齐。

\begin{proposition}[完成化行列式的函数方程]\label{prop:kernel-xi-functional-equation}
设同步核满足自对偶恒等式 $\Delta(z,u)=\Delta(uz,1/u)$（见 \S\ref{app:kernel-self-dual}），并取
\[
\lambda:=\rho(B(1))>1.
\]
定义沿曲线
\[
z(s):=\lambda^{-s},\qquad u(s):=\lambda^{2s-1}
\]
的单变量函数
\[
\Xi(s):=\Delta(z(s),u(s))=\Delta(\lambda^{-s},\lambda^{2s-1}).
\]
则有严格函数方程
\[
\boxed{\ \Xi(s)=\Xi(1-s)\ }.
\]
等价地，定义 $\zeta$ 的同一截面
\[
\zeta_{\mathrm{sd}}(s):=\zeta(z(s),u(s))=\Delta(\lambda^{-s},\lambda^{2s-1})^{-1},
\]
则 $\zeta_{\mathrm{sd}}(s)=\zeta_{\mathrm{sd}}(1-s)$。
\end{proposition}
\begin{proof}
由自对偶恒等式，
\[
\Xi(s)=\Delta(\lambda^{-s},\lambda^{2s-1})
=\Delta(\lambda^{2s-1}\lambda^{-s},\lambda^{1-2s})
=\Delta(\lambda^{s-1},\lambda^{1-2s}).
\]
而对 $1-s$ 有
\(
z(1-s)=\lambda^{-(1-s)}=\lambda^{s-1}
\)
且
\(
u(1-s)=\lambda^{2(1-s)-1}=\lambda^{1-2s}
\)，
故 $\Xi(s)=\Delta(z(1-s),u(1-s))=\Xi(1-s)$。证毕。
\end{proof}

\begin{remark}[临界线与对称固定点]\label{rem:kernel-critical-line-u1}
由定义 $u(s)=\lambda^{2s-1}$ 可见：直线 $\Re(s)=\tfrac12$ 满足 $|u(s)|=1$，并且 $s=\tfrac12$ 为对称固定点（对应 $u=1$）。
因此在该完成化截面上，“临界线”自然等价为“自对偶对称面/单位圆权重”。
\end{remark}

\begin{definition}[核 Hardy 函数：临界线实值化]\label{def:kernel-hardy-function}
沿用命题 \ref{prop:kernel-xi-functional-equation} 的记号。定义
$$
Z(t):=\widetilde{\Delta}\!\bigl(\lambda^{-1/2},\,2it\log\lambda\bigr),
\qquad t\in\RR,
$$
并称其为同步核完成化行列式的 Hardy 型实值化。
\end{definition}

\begin{proposition}[Hardy 实值性、偶性与临界线截面恒等式]\label{prop:kernel-hardy-real-even}
对一切实数 \(t\in\RR\)，有
$$
\boxed{\ Z(t)\in\RR,\qquad Z(t)=Z(-t)\ }.
$$
并且若令 \(s=\tfrac12+it\)，则临界线上的截面恒等式成立：
$$
\boxed{\ Z(t)=\Xi\!\left(\tfrac12+it\right)\ }.

\end{proposition}

\[
Z(t)=\widetilde{\Delta}(\lambda^{-1/2},\theta)

\end{itemize}
\par
\noindent\textbf{端点簇的真实分岔指数为 \(1/2\)。}
端点“簇的阶数”不仅是 \(r_0\) 里出现 \((4\alpha-1)^4\) 这一代数事实，还决定了 Perron 分支靠近端点的\emph{速度}与相应的求根\emph{条件数}。
由证书中
\(
r_0(\alpha)=\alpha(\alpha-1)(4\alpha-1)^4
\)、
\(
r_1(\alpha)=(4\alpha-1)^2(\cdots)
\)，且括号多项式在 \(\alpha=\tfrac14\) 处非零，可知在 \(u\to 0\) 的 Perron 率曲线分支上，主平衡满足 \((\alpha-\tfrac14)^4\) 与 \(u(\alpha-\tfrac14)^2\) 同阶，从而存在 Puiseux 型端点标度
\[
\boxed{\ \alpha(u)-\tfrac14=\Theta(u^{1/2})\quad(u\to0)\ }.
\]
由对偶自对称 \((u,\alpha)\mapsto(1/u,1-\alpha)\) 立刻得到对称端点
\[
\boxed{\ \tfrac34-\alpha(u)=\Theta(u^{-1/2})\quad(u\to\infty)\ }.
\]
该 \(1/2\) 指数比“端点四重簇”的纯代数口径更直接决定端点附近的误差放大与自适应步长选取。

由 \(R(\tfrac12,u)=-(u-1)^6Q(u)/64\) 且 \(Q(1)\neq0\) 可知 \(u=1\) 在 \(\alpha=\tfrac12\) 截面上为恰 \(6\) 重根，从而

\subsubsection{能量分层 primitive 谱的局部极限定理（prime-local CLT）}\label{app:prime-local-clt}
本节把上一小节的 LDP（率函数 \(I\)）提升为长度 \(n\) 的能量分层 primitive 计数 \(p_{n,k}\) 的\emph{局部极限定理}，并显式给出 \(n^{-3/2}\) 的前因子（其中 \(n^{-1}\) 来自 primitive 的 Möbius/Witt 反演，额外的 \(n^{-1/2}\) 来自高斯局部化）。

\begin{theorem}[prime-local CLT：能量层的高斯局部极限与显式前因子]\label{thm:prime-local-clt}
令 \(p_{n,k}=[u^k]p_n(u)\) 表示长度 \(n\) 的 primitive 轨道中能量为 \(k\) 的计数（本附录记号），并记
\[
\sigma^2:=P''(0)=\frac{11}{102}.
\]
则对任意固定 \(X>0\)，当 \(n\to\infty\) 且
\[
k=\frac{n}{2}+x\sqrt{n},\qquad |x|\le X,
\]
有一致渐近
\[
\boxed{\;
p_{n,k}
=\frac{1}{n}\cdot \frac{3^n}{\sqrt{2\pi\sigma^2\,n}}\,
\exp\!\left(-\frac{x^2}{2\sigma^2}\right)\cdot \bigl(1+o(1)\bigr)\; }.
\]
等价地，
\[
p_{n,k}\sim \frac{3^n}{\sqrt{2\pi\sigma^2}}\;n^{-3/2}\,
\exp\!\left(-\frac{(k-\frac{n}{2})^2}{2\sigma^2 n}\right),
\qquad n\to\infty,
\]
且上述 \(\sim\) 在 \(|x|\le X\) 上一致成立。
\end{theorem}
\begin{remark}[证明路线（与草稿现有接口完全兼容）]\label{rem:prime-local-clt-proof-idea}
要点是把“周期迹层”的谱扰动局部 CLT 与 primitive 的 Möbius 反演解耦：
\begin{itemize}
  \item 对周期迹 \(a_n(u)=\mathrm{Tr}(B(u)^n)\)，由 \(\lambda(e^\theta)\) 的实解析性与单位圆上的谱隙，可对 \(u\) 的复参数做 Fourier 反演/鞍点法，得到 \(a_{n,k}=[u^k]a_n(u)\) 的局部高斯极限及 \(n^{-1/2}\) 前因子；
  \item 对 primitive 多项式 \(p_n(u)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}(u^d)\)，主项来自 \(d=1\)；而 \(d\ge 2\) 的项在局部尺度上指数小（由 \(\rho(u)<\lambda(u)\) 与 \(n/d\) 缩短），故不影响主阶形状，只留下额外的 primitive 因子 \(1/n\)。
\end{itemize}
因此局部 CLT 的“形状常数”由同一压力 Hessian \(\sigma^2=P''(0)\) 决定，前因子则严格分解为 primitive 的 \(1/n\) 与局部高斯的 \(n^{-1/2}\)。
\end{remark}

\end{proposition}

\subsubsection{加权 Mertens 常数与 Abel 有限部}\label{app:mertens-weighted}
令 $\lambda(u)$ 为 Perron 根，$\rho(u)$ 为其余特征值最大模，则
\[
a_n(u)=\mathrm{Tr}(B(u)^n)=\lambda(u)^n+O(\rho(u)^n),
\]
由 Witt 反演得
\[
p_n(u)=\frac{\lambda(u)^n}{n}+O\!\left(\frac{\rho(u)^n}{n}\right).
\]
因此
\[
\Pi_\theta(N):=\sum_{n\le N}p_n(\theta)\sim \frac{\lambda(\theta)}{\lambda(\theta)-1}\cdot\frac{\lambda(\theta)^N}{N}.
\]
记
\[
\mathcal{M}_\theta(N):=\sum_{n\le N}\frac{p_n(u)}{\lambda(u)^n},
\]
则
\[
\mathcal{M}_\theta(N)=\log N+M(\theta)+o(1),
\qquad
M(\theta)=\gamma+\sum_{n\ge 1}\left(\frac{p_n(u)}{\lambda(u)^n}-\frac{1}{n}\right),
\]

\begin{proposition}[Mertens 常数的偶性]\label{prop:mertens-even}
由 $p_n(u)=u^n p_n(1/u)$ 与 $\lambda(u)=u\lambda(1/u)$ 得
\[
\frac{p_n(u)}{\lambda(u)^n}=\frac{p_n(1/u)}{\lambda(1/u)^n},
\]
从而
\[
M(\theta)=M(-\theta),\qquad
\log\mathfrak{M}(\theta)=\log\mathfrak{M}(-\theta).
\]
\end{proposition}
其中 $\gamma$ 为 Euler 常数。设
\[
P_\theta(r):=\sum_{n\ge 1}\frac{p_n(u)}{\lambda(u)^n}r^n,\quad 0<r<1,
\]
则 Abel 有限部给出
\[
\log\mathfrak{M}(\theta):=\mathrm{fp}_{r=1}\bigl(P_\theta(r)+\log(1-r)\bigr)
=\sum_{n\ge 1}\left(\frac{p_n(u)}{\lambda(u)^n}-\frac{1}{n}\right),
\]
从而
\[
M(\theta)=\gamma+\log\mathfrak{M}(\theta).
\]
在 $\theta=0$ 处有
\[
M(0)\approx 0.2738887860704445,\qquad
\log\mathfrak{M}(0)\approx -0.3033268788310883.
\]

\subsubsection{仅用 \texorpdfstring{$\Delta$}{Delta} 的闭式计算}\label{app:delta-only}
令 $z_\star(u)=\lambda(u)^{-1}$ 为 $\Delta(z,u)$ 的最小正根，定义
\[
C(u):=\lim_{z\to z_\star}(1-z/z_\star)\zeta(z,u)=\frac{1}{-z_\star\,\partial_z\Delta(z_\star,u)}.
\]
由 Witt 反演与 $\zeta=1/\Delta$ 可得
\[
\log\mathfrak{M}(\theta)
=\log C(u)+\sum_{m\ge 2}\frac{\mu(m)}{m}\log\zeta(z_\star^m,u^m)
=\log C(u)-\sum_{m\ge 2}\frac{\mu(m)}{m}\log\Delta(z_\star^m,u^m),
\]
其中级数以指数速度收敛。

\begin{corollary}[完成化的纯实计算：单位圆上 \texorpdfstring{$\Delta(z_\star^m,u^m)$}{Delta(z*^m,u^m)} 的实代数化]\label{cor:sync-kernel-weighted-logM-completed-real}
取 \(u=e^{i\theta}\) 并令 \(r=e^{i\theta/2}\)、\(s=r+r^{-1}=2\cos(\theta/2)\in[-2,2]\)。记 \(w_\star(s)\) 为 \(\widehat\Delta(w,s)=0\) 的最小模根，则 \(z_\star(u)=w_\star(s)/r\)。
对任意 \(m\ge 1\)，令
\[
S_m(s):=r^m+r^{-m}=2\cos(m\theta/2)=:C_m(s)\in\ZZ[s]
\]
（\(C_m\) 为 Chebyshev--Adams 运算；见推论 \ref{cor:primitive-completion-hatp}）。
则有恒等式
\[
\boxed{\ \Delta(z_\star(u)^m,u^m)=\widehat\Delta\!\bigl(w_\star(s)^m,\ S_m(s)\bigr)\ }.
\]
因此在单位圆扭曲上，\(\log\mathfrak{M}(\theta)\) 的 \(\Delta\)-only 级数每一项都可在实域上计算：输入仅为实参数 \(s\in[-2,2]\) 与实数 \(w_\star(s)\) 及其幂。
\end{corollary}

\begin{corollary}[primitive 多项式的完成化：\texorpdfstring{$\widehat p_n(s)\in\ZZ[s]$}{hat p in Z[s]} 与 Chebyshev--Adams 运算]\label{cor:primitive-completion-hatp}
取
\[
u=r^2,\qquad s:=r+r^{-1},
\]
并对 \(n\ge 1\) 定义 primitive 多项式的完成化
\[
\widehat p_n(s):=r^{-n}p_n(r^2).
\]
则对所有 \(n\ge 1\) 有 \(\widehat p_n(s)\in\ZZ[s]\)。并且：
\begin{enumerate}
  \item 若 \(n\) 为偶数，则 \(\widehat p_n(s)\) 为 \(s\) 的偶多项式；
  \item 若 \(n\) 为奇数，则 \(\widehat p_n(s)\) 为 \(s\) 的奇多项式。
\end{enumerate}
进一步，对任意整数 \(d\ge 1\) 定义
\[
C_d(s):=r^d+r^{-d}\in\ZZ[s],
\qquad
C_d(s)=2T_d(s/2)
\]
（\(T_d\) 为第一类 Chebyshev 多项式）。则完成化坐标下的幂替换满足严格的“倍角律”：
\[
r^{-dn}\,p_n(r^{2d})=\widehat p_n\!\bigl(C_d(s)\bigr),
\]
即 Witt/Möbius 反演中的 \(u\mapsto u^d\) 在完成化后等价于 \(s\mapsto C_d(s)\) 的 Chebyshev--Adams 运算。
\end{corollary}

\begin{corollary}[奇长度的严格奇偶平衡：\texorpdfstring{$p_n(-1)=0$}{p_n(-1)=0}]\label{cor:primitive-odd-parity-balance}
若 \(n\ge 3\) 为奇数，则由推论 \ref{cor:primitive-odd-div} 的整除律 \(u(u+1)\mid p_n(u)\) 得
\[
\boxed{\ p_n(-1)=\sum_{k\ge 0}(-1)^k p_{n,k}=0\ }.
\]
因此奇长度 primitive 轨道在“能量奇偶”两侧的计数严格相等（以符号加权意义表述）。
\end{corollary}
\begin{proof}
由系数镜像对称 \(p_{n,k}=p_{n,n-k}\) 可得（见上文）
\(
p_n(u)=u^n p_n(1/u)
\)。
令 \(u=r^2\) 得
\(
p_n(r^2)=r^{2n}p_n(r^{-2})
\)，
从而 \(r^{-n}p_n(r^2)\) 在 \(r\mapsto r^{-1}\) 下不变。由不变量子环恒等式
\(
\ZZ[r,r^{-1}]^{\,r\mapsto r^{-1}}=\ZZ[r+r^{-1}]=\ZZ[s]
\)
即得 \(\widehat p_n(s)\in\ZZ[s]\)。
\par
奇偶性来自指数奇偶：\(p_n(r^2)\) 只含 \(r^{2k}\)，乘上 \(r^{-n}\) 后幂指数与 \(n\) 同奇偶；而 \(\ZZ[s]\) 中 \(s\mapsto -s\) 的奇偶分解与 \(r\mapsto -r\) 的幂奇偶一致，故得到 \(\widehat p_n\) 的奇偶性结论。
\par
最后令 \(s=r+r^{-1}\)，则
\(
C_d(s)=r^d+r^{-d}
\)
满足递推 \(C_{d+1}=sC_d-C_{d-1}\)，因而 \(C_d\in\ZZ[s]\)，并与 Chebyshev 多项式关系 \(C_d(s)=2T_d(s/2)\) 等价。代入 \(u=r^2\) 及 \(u^d=r^{2d}\) 得
\[
r^{-dn}p_n(r^{2d})=\widehat p_n(r^d+r^{-d})=\widehat p_n(C_d(s)),
\]
即所述倍角律。证毕。
\end{proof}

\begin{corollary}[完成化的 Chebyshev--Witt 变换：\texorpdfstring{$\widehat a\leftrightarrow \widehat p$}{hat a <-> hat p}]\label{cor:chebyshev-witt-completed}
在同一完成化坐标 \(u=r^2,s=r+r^{-1}\) 下，定义 completed ghost（迹）多项式
\[
\widehat a_n(s):=r^{-n}a_n(r^2)=r^{-n}\Tr(B(r^2)^n)\in\ZZ[s].
\]
则对任意 \(n\ge 1\) 有严格恒等式
\[
\boxed{\ 
\widehat p_n(s)=\frac1n\sum_{d\mid n}\mu(d)\,\widehat a_{n/d}\!\bigl(C_d(s)\bigr),
\qquad
\widehat a_n(s)=\sum_{d\mid n}d\,\widehat p_d\!\bigl(C_{n/d}(s)\bigr)
\ }.
\]
\end{corollary}
\begin{proof}
对 \(u=r^2\) 的 Witt/Möbius 反演
\(
p_n(u)=\frac1n\sum_{d\mid n}\mu(d)a_{n/d}(u^d)
\)
两侧同乘 \(r^{-n}\)，并用完成化倍角律
\(
r^{-dn}f(r^{2d})=\widehat f(C_d(s))
\)
（对 \(f=a_{n/d}\) 与 \(f=p_d\) 分别应用）即可得到第一式；第二式同理由
\(
a_n(u)=\sum_{d\mid n} d\,p_d(u^{n/d})
\)
完成化得到。
\end{proof}

\begin{corollary}[素数长度同余：\texorpdfstring{$\widehat a_p(s)\equiv C_p(s)\ (\mathrm{mod}\ p)$}{hat a_p congruent Cp mod p}]\label{cor:completed-prime-congruence}
沿用推论 \ref{cor:chebyshev-witt-completed} 的完成化记号。若 \(p\) 为素数，则有整系数同余
\[
\boxed{\ \widehat a_p(s)\equiv C_p(s)\pmod p\ }.
\]
\end{corollary}
\begin{proof}
对素数 \(p\)，由 \(\widehat p_p(s)=\frac1p\bigl(\widehat a_p(s)-\widehat a_1(C_p(s))\bigr)\)（推论 \ref{cor:chebyshev-witt-completed} 的第一式，因子仅为 \(d=1,p\)）可知
\[
\widehat a_p(s)-\widehat a_1(C_p(s))\in p\,\ZZ[s].
\]
而在本核上 \(a_1(u)=\Tr(B(u))=p_1(u)=1+u\)，故
\(
\widehat a_1(s)=r^{-1}(1+r^2)=r+r^{-1}=s
\)，从而 \(\widehat a_1(C_p(s))=C_p(s)\)。
代回即得同余。证毕。
\end{proof}
