\subsubsection{Operator-Valued Horizon Positivity and Completely Positive Visible Realization: Multi-Channel Carath\'eodory--Toeplitz Completeness, Fourier--Pontryagin Decomposition, and Universal Strictification}\label{subsubsec:xi-operator-valued-carath-toeplitz-complete-positivity}
This section has previously established the horizon positivity criterion for RH at scalar caliber: the Carath\'eodory positivity of \(\mathscr{C}_{\zeta}\) is strictly equivalent to the Toeplitz--PSD hierarchy (Appendix \ref{subsubsec:app-horizon-weyl-carath-toeplitz}).
Simultaneously, strict commutativity on the calculus side has been compressed to vanishing anomaly signature, and strictified by the universal commutativization group quotient \(G\twoheadrightarrow G^{\mathrm{BC}}_m\) (\S\ref{subsec:pom-three-generator-residual-closure}; Theorem \ref{thm:dephys-offline-hair-curvature-reduction}).
\par
This subsection unifies the above two chains at the \emph{multi-channel/operator-valued} level: it promotes ``per-channel positivity'' to a \textbf{completely positive} horizon object valued in a finite-dimensional \(C^\ast\)-algebra, and uses the Fourier--Pontryagin (Peter--Weyl) decomposition to strictly align block Toeplitz--PSD with per-channel PSD; this interprets strictification as the maximal quotient for ``completely positive visible realization existence,'' and rigorously characterizes offline hair (genuine resonances or non-strictifiable residuals) as finite-dimensional falsifiers of complete positivity.

\paragraph{Finite-dimensional \(C^\ast\)-algebras and operator-valued Carath\'eodory class}
Let \(\mathcal{A}\) be a finite-dimensional \(C^\ast\)-algebra.
In the strictification context of this paper, the typical choice is
\[
\mathcal{A}_m:=\CC\bigl[G^{\mathrm{BC}}_m\bigr],
\]
whose \(C^\ast\)-structure is induced by the right regular representation, equivalent to its Peter--Weyl decomposition into a finite direct sum of matrix algebras:
\begin{equation}\label{eq:xi-A-peter-weyl}
\boxed{\;
\mathcal{A}
\ \cong\
\bigoplus_{\pi\in\widehat{\mathcal{A}}} M_{d_\pi}(\CC)\; },
\end{equation}
where \(\widehat{\mathcal{A}}\) denotes the set of irreducible \(\ast\)-representations (for a group algebra this is \(\widehat{G^{\mathrm{BC}}_m}\)).
Therefore \(a\in\mathcal{A}\) is a positive element if and only if \(a_\pi\succeq 0\) for every irreducible component \(\pi\).
\par
An analytic map \(F:\mathbb{D}\to\mathcal{A}\) is called an \textbf{operator-valued Carath\'eodory function} if for all \(w\in\mathbb{D}\),
\[
\Re F(w):=\frac{F(w)+F(w)^\ast}{2}\succeq 0
\qquad(\text{in the self-adjoint order of }\mathcal{A}).
\]
Write its normalized power series as
\begin{equation}\label{eq:xi-operator-carath-series}
\boxed{\;
F(w)=c_0+2\sum_{n\ge 1}c_n w^n,\qquad c_{-n}:=c_n^\ast\;}
\end{equation}
(where \(c_n\in\mathcal{A}\)).
For each \(N\ge 0\), define the block Toeplitz truncation
\begin{equation}\label{eq:xi-block-toeplitz-def}
\boxed{\;
\mathbf{T}^{\mathcal{A}}_N(F):=(c_{j-k})_{0\le j,k\le N}\ \in\ M_{N+1}(\mathcal{A}).\;}
\end{equation}

\begin{theorem}[Operator-valued Herglotz representation and block Toeplitz completeness]\label{thm:xi-operator-herglotz-toeplitz-completeness}
Let \(F:\mathbb{D}\to\mathcal{A}\) be analytic and written as \eqref{eq:xi-operator-carath-series}.
Then the following statements are equivalent:
\begin{enumerate}
  \item \(\Re F(w)\succeq 0\) for all \(w\in\mathbb{D}\) (operator-valued Carath\'eodory);
  \item There exists an \(\mathcal{A}\)-valued positive measure\footnote{In the finite-dimensional case this can be understood as: for each Borel set \(E\subseteq\TT\), the assignment \(\boldsymbol\mu(E)\in\mathcal{A}_+\) is countably additive for disjoint unions and satisfies \(\boldsymbol\mu(\TT)\) finite.} \(\boldsymbol\mu\) and a self-adjoint constant \(\beta\in\mathcal{A}\) such that
  \begin{equation}\label{eq:xi-operator-herglotz}
  \boxed{\;
  F(w)=\mathrm{i}\beta+\int_{\TT}\frac{\xi+w}{\xi-w}\,d\boldsymbol\mu(\xi)\qquad(w\in\mathbb{D});\;}
  \end{equation}
  \item For all \(N\ge 0\), the block Toeplitz truncation is positive semidefinite:
  \begin{equation}\label{eq:xi-block-toeplitz-psd}
  \boxed{\;\mathbf{T}^{\mathcal{A}}_N(F)\succeq 0\qquad(\forall N\ge 0).\;}
  \end{equation}
\end{enumerate}
In the scalar case \(\mathcal{A}=\CC\), \eqref{eq:xi-operator-herglotz}--\eqref{eq:xi-block-toeplitz-psd} reduces to the Toeplitz--PSD completeness of Appendix \eqref{eq:app-horizon-herglotz-disk} and Theorem \ref{thm:app-horizon-toeplitz-lmi}.
\end{theorem}

\begin{proof}
Since \(\mathcal{A}\) is finite-dimensional, the decomposition \eqref{eq:xi-A-peter-weyl} exists.
For each irreducible component \(\pi\), let \(F_\pi:=\pi(F):\mathbb{D}\to M_{d_\pi}(\CC)\); then
\(\Re F\succeq 0\) if and only if \(\Re F_\pi\succeq 0\) for all \(\pi\).
The matrix-valued Carath\'eodory/Herglotz representation yields: \(\Re F_\pi\succeq 0\) if and only if there exists a positive \(M_{d_\pi}(\CC)\)-valued measure \(\mu_\pi\) and a self-adjoint constant \(\beta_\pi\) such that
\(
F_\pi(w)=\mathrm{i}\beta_\pi+\int_{\TT}\frac{\xi+w}{\xi-w}\,d\mu_\pi(\xi)
\);
moreover the block Toeplitz truncation of its moment sequence is positive semidefinite in \(M_{(N+1)d_\pi}(\CC)\) if and only if the above representation exists.
Assembling \(\{\mu_\pi\}_{\pi}\) and \(\{\beta_\pi\}_{\pi}\) blockwise into \(\boldsymbol\mu\) and \(\beta\) yields \((1)\Leftrightarrow(2)\).
Similarly, \(\mathbf{T}^{\mathcal{A}}_N(F)\succeq 0\) if and only if for all \(\pi\) we have \(\pi(\mathbf{T}^{\mathcal{A}}_N(F))=\mathbf{T}_N(F_\pi)\succeq 0\), thus yielding \((1)\Leftrightarrow(3)\). \qed
\end{proof}

\paragraph{Fourier--Pontryagin decomposition: per-channel PSD and completely positive block PSD}
In the case \(\mathcal{A}=\CC[H]\) (finite group algebra), \eqref{eq:xi-A-peter-weyl} becomes the Peter--Weyl decomposition of the group algebra;
when \(H\) is a finite abelian group, this further reduces to the Fourier--Pontryagin dual decomposition \(\CC[H]\cong \CC[\widehat H]\) (diagonalization).

\begin{theorem}[Block Toeplitz criterion under representation decomposition]\label{thm:xi-fourier-pontryagin-block-toeplitz}
Let \(\mathcal{A}\) be a finite-dimensional \(C^\ast\)-algebra with decomposition \eqref{eq:xi-A-peter-weyl}.
For an analytic \(F:\mathbb{D}\to\mathcal{A}\), the following are equivalent:
\begin{enumerate}
  \item \(\Re F(w)\succeq 0\) (operator-valued Carath\'eodory);
  \item For each \(\pi\in\widehat{\mathcal{A}}\), the component \(F_\pi\) satisfies \(\Re F_\pi(w)\succeq 0\) (per-channel Carath\'eodory);
  \item For each \(N\ge 0\), the block Toeplitz matrix \(\mathbf{T}^{\mathcal{A}}_N(F)\succeq 0\);
  \item For each \(\pi\in\widehat{\mathcal{A}}\) and each \(N\ge 0\), the matrix truncation
  \[
  \pi\!\bigl(\mathbf{T}^{\mathcal{A}}_N(F)\bigr)
  \ =\
  \mathbf{T}_N(F_\pi)\ \in\ M_{(N+1)d_\pi}(\CC)
  \]
  is positive semidefinite.
\end{enumerate}
Therefore ``per-channel PSD'' (Toeplitz--PSD for all \(\pi\)) and ``completely positive block PSD'' (a single block Toeplitz feasibility domain) are strictly equivalent.
\end{theorem}

\begin{proof}
By the direct sum decomposition \eqref{eq:xi-A-peter-weyl} of the finite-dimensional \(C^\ast\)-algebra, the positivity of any self-adjoint element is characterized blockwise by the positivity of each irreducible component;
similarly, the positivity of \(M_{N+1}(\mathcal{A})\) is characterized by the positivity under each amplified representation \(\pi^{(N+1)}:=\mathrm{id}_{N+1}\otimes \pi\).
Applying this observation separately to \(\Re F(w)\) and \(\mathbf{T}^{\mathcal{A}}_N(F)\) and combining with Theorem \ref{thm:xi-operator-herglotz-toeplitz-completeness} yields the result. \qed
\end{proof}

\begin{corollary}[Per-channel diagonalization for finite abelian groups: single block PSD \texorpdfstring{$\Longleftrightarrow$}{<->} all-channel PSD]\label{cor:xi-abelian-block-toeplitz-directsum}
Let \(H\) be a finite abelian group and take \(\mathcal{A}=\CC[H]\).
Under the Fourier--Pontryagin dual decomposition \(\CC[H]\cong \bigoplus_{\chi\in\widehat H}\CC\),
for analytic \(F:\mathbb{D}\to\CC[H]\), denote its character channel components by \(F_\chi\).
Then for each \(N\ge 0\), the block Toeplitz truncation \(\mathbf{T}^{\mathcal{A}}_N(F)\), viewed under the regular representation as a Hermitian block matrix in \(M_{(N+1)|H|}(\CC)\), is unitarily equivalent to the direct sum matrix
\[
\bigoplus_{\chi\in\widehat H}\mathbf{T}_N(F_\chi).
\]
In particular,
\[
\boxed{\;
\mathbf{T}^{\mathcal{A}}_N(F)\succeq 0
\ \Longleftrightarrow\
\forall\chi\in\widehat H,\ \mathbf{T}_N(F_\chi)\succeq 0\;}
\]
so that ``per-channel PSD verification'' under the abelian visible quotient caliber can be unified into a single block Toeplitz--PSD feasibility domain (a single semidefinite certificate).
\end{corollary}

\begin{proof}
This is a direct specialization of Theorem \ref{thm:xi-fourier-pontryagin-block-toeplitz} to the abelian case \(d_\pi=1\): irreducible representations are characters \(\chi\), and the amplified representation diagonalizes the block Toeplitz truncation into the direct sum of per-channel Toeplitz truncations. \qed
\end{proof}

\paragraph{Completely positive interpretation of strictification: \texorpdfstring{$G^{\mathrm{BC}}_m$}{GBCm} as the maximal visible quotient}
In the projection-word closed-loop interface, strict commutativity is equivalent to vanishing anomaly signature (Corollary \ref{cor:pom-zero-anom-pullback-exactness}), and is strictified by the universal quotient \(G\twoheadrightarrow G^{\mathrm{BC}}_m\) (Proposition \ref{prop:pom-bc-quotient-strictify} and Proposition \ref{prop:pom-bc-quotient-universal}).
Combining this universality with Theorem \ref{thm:xi-operator-herglotz-toeplitz-completeness} allows strictification to be interpreted as the maximal quotient for ``multi-channel completely positive visible realization existence.''

\begin{theorem}[Maximal completely positive visible realization: operator-valued characterization of strictification]\label{thm:xi-gbc-maximal-cp-visible-realization}
Fix resolution \(m\) and let \(\mathcal{A}_m:=\CC[G^{\mathrm{BC}}_m]\).
Suppose on the set of visible twist channels \(\widehat{G^{\mathrm{BC}}_m}\), a family of horizon functions (e.g., twisted output horizon log-derivatives/Carath\'eodory family)
\(\{F_{\overline\chi}\}_{\overline\chi\in\widehat{G^{\mathrm{BC}}_m}}\) is given, and assume they can be assembled into an \(\mathcal{A}_m\)-valued analytic function \(F_m:\mathbb{D}\to\mathcal{A}_m\) such that for each \(\overline\chi\), \(\pi_{\overline\chi}(F_m)=F_{\overline\chi}\) (where \(\pi_{\overline\chi}\) is the corresponding channel \(\ast\)-representation).
If the anomaly vanishes strictly on visible channels (Proposition \ref{prop:pom-bc-quotient-strictify}), and \(F_m\) satisfies the block Toeplitz--PSD hierarchy \(\mathbf{T}^{\mathcal{A}_m}_N(F_m)\succeq 0\) (\(\forall N\)), then there exists an \(\mathcal{A}_m\)-valued positive measure \(\boldsymbol\mu_m\) such that \(F_m\) admits the Herglotz representation \eqref{eq:xi-operator-herglotz}.
Consequently, all channels \(F_{\overline\chi}\) are simultaneously given by the Fourier--Pontryagin projection of the same \(\boldsymbol\mu_m\) (completely positive visible realization).
\par
Conversely, if there exists a group quotient \(G\twoheadrightarrow Q\) such that the above completely positive visible realization holds on \(Q\) and the closed-loop interface commutes strictly, then this group quotient must uniquely factorize through \(G^{\mathrm{BC}}_m\) (Proposition \ref{prop:pom-bc-quotient-universal}).
\end{theorem}

\begin{proof}
The first part follows from ``anomaly vanishes on visible channels,'' which guarantees that the visible channel assembly after strictification is compatible with the closed-loop interface; Theorem \ref{thm:xi-operator-herglotz-toeplitz-completeness} \((3)\Rightarrow(2)\) then yields the \(\mathcal{A}_m\)-valued positive measure representation.
The second part is a direct application of the universality of \(G^{\mathrm{BC}}_m\) (Proposition \ref{prop:pom-bc-quotient-universal}). \qed
\end{proof}

\begin{theorem}[Centrality of endpoint atoms (threshold support) and channel strength decomposition]\label{thm:xi-endpoint-atom-central-channel-strength}
Under the completely positive realization of Theorem \ref{thm:xi-gbc-maximal-cp-visible-realization}, the atom at endpoint \(-1\),
\[
\mathbf{M}_{-1}:=\boldsymbol\mu_m(\{-1\})\in(\mathcal{A}_m)_+,
\]
exists and is a positive element.
\begin{enumerate}
  \item If \(G^{\mathrm{BC}}_m\) is abelian, then \(\mathcal{A}_m=\CC[G^{\mathrm{BC}}_m]\) is commutative, and hence \(\mathbf{M}_{-1}\in Z(\mathcal{A}_m)\) (center).
  \item More generally, if \(\boldsymbol\mu_m\) takes values in the center subalgebra (central measure), then \(\mathbf{M}_{-1}\in Z(\mathcal{A}_m)\), and for any irreducible component \(\pi\),
  \begin{equation}\label{eq:xi-endpoint-atom-channel-strength}
  \boxed{\;
  (\mathbf{M}_{-1})_\pi=\lambda_\pi\,I_{d_\pi},\qquad \lambda_\pi\ge 0.\;}
  \end{equation}
\end{enumerate}
The family \(\{\lambda_\pi\}\) gives a finite-dimensional ``channel strength decomposition'' of the endpoint threshold support; in the abelian case, \(d_\pi=1\) and \(\lambda_\pi\) reduces to the endpoint load on each character channel.
\end{theorem}

\begin{proof}
The first assertion follows immediately from the commutativity of \(\mathcal{A}_m\).
For the second, by assumption \(\boldsymbol\mu_m(E)\in Z(\mathcal{A}_m)\) for all Borel sets \(E\), so \(\mathbf{M}_{-1}=\boldsymbol\mu_m(\{-1\})\in Z(\mathcal{A}_m)\).
By Schur's lemma, the image of a central element under each irreducible representation must be a scalar matrix, written as \((\mathbf{M}_{-1})_\pi=\lambda_\pi I_{d_\pi}\);
since \(\mathbf{M}_{-1}\succeq 0\), we have \(\lambda_\pi\ge 0\). \qed
\end{proof}

\paragraph{Finite-dimensional falsifiers: auditable witnesses of complete positivity breakdown}
Since each level \(\mathbf{T}^{\mathcal{A}}_N(F)\succeq 0\) is a finite-dimensional semidefinite constraint, the failure of complete positivity (operator-valued Carath\'eodory) must yield a finite-dimensional falsifier at some level.

\begin{corollary}[Complete positivity breakdown \texorpdfstring{$\Rightarrow$}{=>} finite-order block negative inertia]\label{cor:xi-cp-breakdown-finite-witness}
Suppose \(F:\mathbb{D}\to\mathcal{A}\) is analytic and does not belong to the operator-valued Carath\'eodory class (i.e., there exists \(w\in\mathbb{D}\) such that \(\Re F(w)\nsucceq 0\)).
Then there exist a finite \(N\) and an irreducible component \(\pi\in\widehat{\mathcal{A}}\) such that the matrix truncation
\[
\pi\!\bigl(\mathbf{T}^{\mathcal{A}}_N(F)\bigr)=\mathbf{T}_N(F_\pi)
\]
has a negative eigenvalue.
In particular, any channel exhibiting a pole within the stability domain (genuine resonance) forces a certain level of block Toeplitz--PSD failure, thereby producing a realization-independent, offline-verifiable, finite-dimensional falsifier.
\end{corollary}

\begin{proof}
If truncations at all \(N\) and all \(\pi\) were positive semidefinite, Theorem \ref{thm:xi-fourier-pontryagin-block-toeplitz} would yield \(\Re F\succeq 0\), contradicting the hypothesis; hence there must exist a witness pair \((N,\pi)\). \qed
\end{proof}

\begin{remark}[Anomaly as an obstruction to ``completely positive visible realization'']\label{rem:xi-anom-as-cp-visible-obstruction}
In Theorem \ref{thm:xi-gbc-maximal-cp-visible-realization}, ``anomaly vanishes strictly on visible channels'' is a structural prerequisite for the completely positive visible realization:
when the anomaly is nonzero, the strict commutativity of the closed-loop interface fails at the given resolution (Corollary \ref{cor:pom-zero-anom-pullback-exactness}), making it impossible to assemble multi-channel outputs into a single \(\mathcal{A}_m\)-valued object compatible with the interface on the chosen visible quotient.
This failure itself constitutes a finite-dimensional auditable falsifier (anomaly signature/anomaly cohomology class), and can be amplified in the linear sense by the MOM\((q)\) gate (Proposition \ref{prop:pom-anom-moment-amplification}).
Orthogonally, genuine resonances (poles within the stability domain) manifest through finite-level failure of block Toeplitz--PSD (Corollary \ref{cor:xi-cp-breakdown-finite-witness}).
\end{remark}
