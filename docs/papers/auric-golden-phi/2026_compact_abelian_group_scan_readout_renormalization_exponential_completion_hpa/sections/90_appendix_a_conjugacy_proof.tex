\section{附录 A：尺度共轭命题证明（命题 \ref{prop:conjugacy}）}

对任意 $x\in\RR$，记 $[x]_L$ 为模 $L$ 等价类。定义 $h_L([x]_L)=[x/L]_1$。
\begin{enumerate}
  \item $h_L$ 良定义：若 $x-y\in L\ZZ$，则 $(x/L)-(y/L)\in\ZZ$，故 $[x/L]_1=[y/L]_1$。
  \item $h_L$ 为同胚：其逆映射为 $h_L^{-1}([u]_1)=[Lu]_L$。连续性由商拓扑与线性映射连续性继承。
  \item 共轭：注意
  \[
  h_L(R_\delta^{(L)}([x]_L))=h_L([x+\delta]_L)=[(x+\delta)/L]_1=[x/L+\delta/L]_1,
  \]
  而
  \[
  R_{\delta/L}^{(1)}(h_L([x]_L))=R_{\delta/L}^{(1)}([x/L]_1)=[x/L+\delta/L]_1.
  \]
  故 $h_L\circ R_\delta^{(L)}=R_{\delta/L}^{(1)}\circ h_L$。
\end{enumerate}
