\section{Notation and glossary}\label{sec:appendix:notation}

This appendix collects the core symbols used throughout the review.

\subsection{Geometric objects}
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $E=\RR^d$: physical space.
  \item $H=\RR^n$: internal space.
  \item $G=E\times H$: superspace.
  \item $\Gamma\subset G$: lattice (CPS data).
  \item $\pi_E,\pi_H$: projections $G\to E$ and $G\to H$.
  \item $x^\star$: star map image of $x\in \Gamma_E$ (Definition~\ref{def:cps}).
  \item $W\subset H$: window / acceptance domain.
  \item $\Lambda(W)\subset E$: model set (Definition~\ref{def:model-set}).
\end{itemize}

\subsubsection*{Terminology disambiguation (GRG-facing)}
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item \textbf{Superspace (this paper)}: the cut-and-project embedding space $G=E\times H$ used in model-set/quasicrystal theory; a representation space (Section~\ref{sec:superspace:cps} and Section~\ref{sec:intro:superspace-terminology}).
  \item \textbf{Supersymmetric superspace (not used)}: spacetime extended by Grassmann coordinates in SUSY QFT.
  \item \textbf{Wheeler superspace (not used)}: configuration space of 3-geometries (3-metrics modulo diffeomorphisms) in canonical GR.
\end{itemize}

\subsection{Protocol/readout objects}
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $\Theta$: protocol parameter tuple (orientation, phase, window, resolution, scan rule).
  \item $R_\Theta:H\to\mathcal{A}$: readout map into a finite alphabet $\mathcal{A}$ (Definition~\ref{def:readout}).
  \item $a_t\in\mathcal{A}$: symbolic readout sequence generated by scanning internal states.
  \item $\mathcal{P}$: the partition of $H$ induced by $R_\Theta$.
  \item $C(a_{0:t-1})\subset H$: cylinder set of internal states consistent with a length-$t$ prefix.
\end{itemize}

\subsection{Stabilization/fibers and time}
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $X_m\subset\{0,1\}^m$: golden-mean legal language (no substring $11$), Definition~\ref{def:Xm}.
  \item $\Fold_m:\mathcal{A}^m\twoheadrightarrow X_m$: stabilization (canonicalization) map, Definition~\ref{def:fold}.
  \item $\mathcal{F}_m(x)=\Fold_m^{-1}(x)$: raw-word fiber over a stabilized type $x$.
  \item $\widetilde{\mathcal{F}}_m(x)=\{h\in H:(\Fold_m\circ R_\Theta)(h)=x\}$: stabilized fiber in internal space (Proposition~\ref{prop:fiber-template}).
  \item $\tau(t)=-\log\mu(C(a_{0:t-1}))$: entropy-rate protocol time, Definition~\ref{def:entropy-time}.
\end{itemize}
