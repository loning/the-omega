\paragraph{KMS Locking at Code Length \texorpdfstring{$(1,2)$}{(1,2)} and the Golden Scale: Modular Scale Group, Reconstruction Threshold, and Three-Domain Alignment of the Two-State Fiber Law}
Under the framework of this paper, where time is fixed as the iteration count of a single-generator scan operator and binary window readout with normalization layers serve as the protocol constraints, the prefix branches of code length \((1,2)\) (\(0\) and \(10\)) yield a fully explicit operator-theoretic realization: the KMS inverse temperature is locked at \(\log\varphi\), while the information gap caused by folding is quantified by \(\log(2/\varphi)=\log 2-\log\varphi\); these two are not independent constants, but structural invariants jointly externalized under the same KMS--fiber two-state law.
\par
The following results formalize this alignment as an independently citable chain of theorems.
\par

\begin{theorem}[KMS Factor at Code Length \texorpdfstring{$(1,2)$}{(1,2)} and the Discrete Modular Scale Group: Rigidity of \texorpdfstring{$\varphi^{\mathbb Z}$}{phi^Z}]\label{thm:xi-kms-code12-discrete-modular-scale-group}
Denote the Cuntz algebra \(\mathcal{O}_2=C^\ast(\mathsf{S}_0,\mathsf{S}_1)\), where \(\mathsf{S}_0,\mathsf{S}_1\) are isometries satisfying
\[
\mathsf{S}_i^\ast\mathsf{S}_j=\delta_{ij}1,
\qquad
\mathsf{S}_0\mathsf{S}_0^\ast+\mathsf{S}_1\mathsf{S}_1^\ast=1.
\]
Define the quasi-free dynamics \((\alpha_t)_{t\in\mathbb R}\) on \(\mathcal{O}_2\):
\[
\alpha_t(\mathsf{S}_0)=e^{it}\mathsf{S}_0,
\qquad
\alpha_t(\mathsf{S}_1)=e^{2it}\mathsf{S}_1.
\]
Then there exists a unique KMS state \(\omega\), whose inverse temperature \(\beta\) is uniquely determined by the equation \(e^{-\beta}+e^{-2\beta}=1\), hence
\[
\beta=\log\varphi.
\]
Let \((\pi_\omega,\mathcal H_\omega,\Omega_\omega)\) be the GNS representation and denote \(\mathcal M:=\pi_\omega(\mathcal{O}_2)''\).
Then the positive spectrum of the modular operator \(\Delta_\omega\) of \(\omega\) is discrete and satisfies
\[
\mathrm{spec}(\Delta_\omega)\cap\mathbb R_{>0}=\varphi^{\mathbb Z},
\]
so that the Connes scale invariant satisfies
\[
S_{\mathrm{Connes}}(\mathcal M)=\{0\}\cup\varphi^{\mathbb Z}.
\]
In particular,
\[
\mathbb Z[\varphi]_{>0}^\times=\varphi^{\mathbb Z},
\]
so the modular scale group coincides with the positive unit group of the quadratic integer ring \(\mathbb Z[\varphi]\).
\end{theorem}

\begin{theorem}[Limiting Optimal Success Probability of Invertible Reconstruction: Piecewise Closed Form under Constant Bias]\label{thm:xi-optimal-reconstruction-success-probability-constant-bias}
Let \(U_m\sim\mathrm{Unif}(\Omega_m)\), and denote the macroscopic observation random variable
\(
\mathsf{X}_m:=\mathrm{Fold}^{\mathrm{bin}}_m(U_m)\in X_m
\).
Allow an auxiliary register of size \(M_m\in\mathbb N\), and let the encoder produce an auxiliary tag \(A\in\{1,\dots,M_m\}\) given \(\omega\in\Omega_m\) and \(x=\mathrm{Fold}^{\mathrm{bin}}_m(\omega)\); the decoder outputs an estimate \(\widehat\omega\) based on \((x,A)\).
The average optimal success probability under uniform input is
\[
P_m^\star(M_m):=\sup \mathbb P[\widehat\omega=U_m].
\]
Then \(P_m^\star(M_m)\) has the closed form:
\[
P_m^\star(M_m)=\frac{1}{2^m}\sum_{x\in X_m}\min\!\bigl\{d_m^{\mathrm{bin}}(x),\,M_m\bigr\}.
\]
Further, taking the constant bias scaling
\[
M_m=\exp\!\Bigl(m\log\!\Bigl(\frac{2}{\varphi}\Bigr)+c\Bigr),
\qquad c\in\mathbb R\ \text{fixed}.
\]
Then the limit
\(
P_\star(c):=\lim_{m\to\infty}P_m^\star(M_m)
\)
exists and yields the explicit piecewise formula
\[
P_\star(c)=
\begin{cases}
1, & c\ge 0,\\[6pt]
\displaystyle \frac{\varphi}{\sqrt5}\,e^{c}+\frac{1}{\varphi\sqrt5}, & -\log\varphi\le c<0,\\[10pt]
\displaystyle \frac{\varphi^2}{\sqrt5}\,e^{c}, & c<-\log\varphi.
\end{cases}
\]
\end{theorem}

\begin{proof}
The closed form \(P_m^\star(M_m)=2^{-m}\sum_x\min\{d_m^{\mathrm{bin}}(x),M_m\}\) follows from: for each fiber \(x\), the optimal strategy achieves injective encoding on \(M_m\) microstates, with the remainder necessarily producing collisions, so the intra-fiber optimal success probability is \(\min\{1,M_m/d_m^{\mathrm{bin}}(x)\}\); averaging over uniform input yields the result.
\par
Denote the output distribution \(\mu_m(x):=d_m^{\mathrm{bin}}(x)/2^m\) and let \(W_m\sim\mu_m\), so that
\[
P_m^\star(M_m)=\mathbb E_{\mu_m}\!\left[\min\!\left\{1,\frac{M_m}{d_m^{\mathrm{bin}}(W_m)}\right\}\right].
\]
By Corollary \ref{cor:fold-bin-two-point-limit-law},
\(
(\varphi/2)^m d_m^{\mathrm{bin}}(W_m)\Rightarrow Z\in\{1,\varphi^{-1}\}
\)
with
\(
\mathbb P(Z=1)=\varphi/\sqrt5
\),
\(
\mathbb P(Z=\varphi^{-1})=1/(\varphi\sqrt5)
\).
Moreover, \(M_m/d_m^{\mathrm{bin}}(W_m)=e^{c}/Z+o(1)\) (in distribution). Computing \(\min\{1,e^{c}/Z\}\) pointwise on the two values of \(Z\) and separating by the thresholds in \(c\) yields the stated limiting formula.
\end{proof}

\begin{corollary}[Closed-Form Inversion of the Constant Term \texorpdfstring{$b(\varepsilon)$}{b(eps)}: Exact Quantification of the Zero-Dispersion Second-Order Term]\label{cor:xi-constant-bias-inversion-b-eps}
Let \(L_m(\varepsilon)\) be the minimum additional information (in natural logarithm units) required to achieve average error rate \(\le\varepsilon\), i.e., the minimum \(L\) for which there exists an encoding-decoding scheme with success probability \(\ge 1-\varepsilon\) and \(|\mathrm{aux}|\le e^{L}\).
Then there exists a constant term \(b(\varepsilon)\) such that
\[
L_m(\varepsilon)=m\log\!\Bigl(\frac{2}{\varphi}\Bigr)+b(\varepsilon)+o(1).
\]
Let \(s:=1-\varepsilon\) and denote the threshold \(s_c:=\varphi/\sqrt5\). Then
\[
b(\varepsilon)=
\begin{cases}
0, & s=1,\\[6pt]
\displaystyle \log\!\Bigl(\frac{\sqrt5\,s-\varphi^{-1}}{\varphi}\Bigr), & s_c\le s<1,\\[12pt]
\displaystyle \log\!\Bigl(\frac{\sqrt5\,s}{\varphi^2}\Bigr), & 0<s<s_c.
\end{cases}
\]
\end{corollary}

\begin{proof}
By Theorem \ref{thm:xi-optimal-reconstruction-success-probability-constant-bias}, the limiting success probability depends only on the bias \(c\), and \(P_\star(c)\) is continuous and strictly increasing.
Therefore \(b(\varepsilon)\) is the minimal solution of the equation \(P_\star(c)=1-\varepsilon\); solving piecewise for the two linear segments yields the stated closed form.
\end{proof}

\begin{theorem}[Strong Converse Theorem and Exact Exponent: Exponential Collapse of Success Probability below the Critical Rate]\label{thm:xi-strong-converse-exponent-below-critical-rate}
Denote the auxiliary register rate \(R_m:=\frac{1}{m}\log M_m\).
If there exists \(\delta>0\) such that for sufficiently large \(m\),
\[
R_m\le \log\!\Bigl(\frac{2}{\varphi}\Bigr)-\delta,
\]
then the average success probability of any scheme satisfies the exponential upper bound
\[
P_m^\star(M_m)\ \le\ \frac{\varphi^2}{\sqrt5}\,e^{-\delta m}\,(1+o(1)).
\]
Conversely, if there exists \(\delta>0\) such that \(R_m\ge \log(2/\varphi)+\delta\) and \(M_m\ge d^{\mathrm{bin}}_{\max}(m)\), then for sufficiently large \(m\) there exists a scheme with strictly zero error rate.
Therefore, invertible reconstruction exhibits a hard threshold at the exponential scale, with the phase transition slope being precisely \(\delta\).
\end{theorem}

\begin{proof}
The first part follows from the third segment (\(c\to-\infty\)) of Theorem \ref{thm:xi-optimal-reconstruction-success-probability-constant-bias}, setting \(c=-\delta m+o(m)\).
The second part follows directly from the necessary and sufficient condition that "injective encoding within each fiber is possible if and only if \(M_m\ge d^{\mathrm{bin}}_{\max}(m)\)."
\end{proof}

\begin{theorem}[Rigidity Characterization of the Scale Constant: Three Structural Axioms Uniquely Determine \texorpdfstring{$\varphi$}{phi}]\label{thm:xi-phi-rigidity-from-kms-entropy-density}

Therefore the golden ratio constant can be uniquely characterized by purely operator-dynamical conditions (KMS uniqueness and discrete modular spectrum) together with information density (relative entropy rate), without a priori introduction of Fibonacci combinatorics.
\end{theorem}
