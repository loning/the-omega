\section{Appendix A: Construction Algorithm for \texorpdfstring{$\mathrm{Fold}_m$}{Fold\_m} (Canonical Form)}

\begin{verbatim}
Input: n in Y_m
Output: w = Fold_m(n) in X_m

1. Construct Fibonacci weights F_2, F_3, ..., F_K such that F_K <= n < F_{K+1}.
2. For k = K down to 2:
     if F_k <= remainder:
         set z_k = 1, remainder -= F_k, and skip k-1 (set z_{k-1} = 0)
     else:
         set z_k = 0
3. Output window w = (z_{m+1}, z_m, ..., z_2).
\end{verbatim}
