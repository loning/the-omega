\subsection{Blaschke Data and Orthogonal Decomposition: Boundary Modulus and Disk-Interior Zero Factorization (Conditional Caliber)}\label{subsec:unit-disk-inner-outer}
The Jensen defect of the preceding subsection compresses ``disk-interior zeros'' into a family of finite integrals; within the Hardy/Smirnov class, zero information can further be promoted to a more structured object: the \textbf{inner factor}.

\begin{remark}[Inner--outer factorization and ``hidden phase'']
If \(F_{\zeta}\) (or a normalized version obtained after explicit growth stripping) belongs to some Hardy class \(H^p(\mathbb{D})\) (\(p>0\)) or the Smirnov class \(N^+(\mathbb{D})\), then a standard factorization exists:
\[
F_{\zeta}(w)=B(w)\,S(w)\,O(w),
\]
where \(B\) is a Blaschke product (encoding all disk-interior zeros), \(S\) is a singular inner factor, and \(O\) is an outer factor; moreover, on the boundary one has almost everywhere \(|B(e^{\mathrm{i}\theta})S(e^{\mathrm{i}\theta})|=1\), while \(|O(e^{\mathrm{i}\theta})|\) is determined by \(|F_{\zeta}(e^{\mathrm{i}\theta})|\) (see \cite{Duren1970TheoryHp,Garnett2007BoundedAnalyticFunctions}).
\par
Therefore, under this conditional caliber:
\begin{itemize}
  \item \textbf{Boundary-visible quantity:} \(|F_{\zeta}(e^{\mathrm{i}\theta})|\), equivalent to \(|\Xi_{\zeta}(\tfrac12+\mathrm{i}\tan(\theta/2))|\);
  \item \textbf{Disk-interior hidden quantity:} the inner factor data carried by \(B\) (and possibly \(S\)); the zero set \(\{a_k\}\) of \(B\) constitutes exactly the equivalent encoding of ``deviation from the critical line.''
\end{itemize}
In this sense, RH is equivalent to ``the inner factor carries no zero data inside the disk'' (\(B\equiv 1\), and under the stronger condition of no singular inner factor, \(S\equiv 1\)).
Furthermore, if an off-critical-line zero \(\rho=\beta+\mathrm{i}\gamma\) (\(\beta<\tfrac12\)) exists, then its disk image \(w_{\rho}\) satisfies the Busemann coordinate with basepoint \(-1\):
\(
\delta=\tfrac12-\beta
=\mathcal{B}_{-1}(w_{\rho})=\frac{1-|w_{\rho}|^{2}}{|1+w_{\rho}|^{2}}
\)
(Proposition~\ref{prop:app-busemann-poisson-minus1} and Corollary~\ref{cor:app-offcritical-horocycle-busemann}), which belongs entirely to the Blaschke zero data;
thus the ``shift'' structurally belongs to the hidden component carried by the inner factor, rather than the outer factor component determined by the boundary modulus.
\end{remark}

\paragraph{Inner factor phase current: from ``invisible modulus'' to endpoint observable}
Under the boundary modulus constraint \(|BS|=1\), the inner factor is invisible to boundary \emph{modulus} auditing; however, its disk-interior zeros and singular support leave an uncancellable positive contribution on the boundary \emph{phase}.
This contribution can be completely recorded by a non-negative measure (the phase current measure), and at the endpoint \(-1\) it yields a single-point readable lower-bound certificate.

\medskip
For notational convenience, first introduce the unit circle Poisson kernel: for \(a\in\mathbb{D}\) and \(e^{\mathrm{i}\theta}\in\TT\) define
\begin{equation}\label{eq:app-poisson-kernel}
\boxed{\;
P_a(e^{\mathrm{i}\theta})
:=\frac{1-|a|^2}{|e^{\mathrm{i}\theta}-a|^2}\; }.
\end{equation}
Note that at the endpoint \(-1=e^{\mathrm{i}\pi}\) one has
\begin{equation}\label{eq:app-poisson-kernel-endpoint-minus1}
\boxed{\;
P_a(-1)=\frac{1-|a|^2}{|1+a|^2}=\mathcal{B}_{-1}(a)\;}
\end{equation}
(Proposition~\ref{prop:app-busemann-poisson-minus1}).

\begin{definition}[Inner factor phase current measure]\label{def:app-inner-phase-current}
Under the Hardy/Smirnov conditional caliber of this subsection, write \(F_{\zeta}=BSO\) and let the inner factor be \(I(w):=B(w)S(w)\).
Let \(\{a_k\}\subset\mathbb{D}\) be the disk-interior zeros of \(B\) (counted with multiplicity).
Let \(\nu\) be the (positive) singular Borel measure corresponding to \(S\), so that
\begin{equation}\label{eq:app-singular-inner-factor-measure}
\boxed{\;
S(w)
=\exp\!\left(
-\int_{0}^{2\pi}\frac{e^{\mathrm{i}t}+w}{e^{\mathrm{i}t}-w}\,d\nu(t)
\right)\; }.
\end{equation}
Define the \textbf{phase current measure} \(\mu_{\mathrm{in}}\) as
\begin{equation}\label{eq:app-inner-phase-current-measure}
\boxed{\;
d\mu_{\mathrm{in}}(\theta)
:=
\left(\sum_k P_{a_k}(e^{\mathrm{i}\theta})\right)\frac{d\theta}{2\pi}
\;+\;
d\nu(\theta)\; }.
\end{equation}
and define its absolutely continuous density (allowing the value \(+\infty\)) as
\begin{equation}\label{eq:app-inner-phase-current-density}
\boxed{\;
j_{\mathrm{in}}(\theta):=\sum_k P_{a_k}(e^{\mathrm{i}\theta})\in[0,\infty]\;}
\end{equation}
(in the Lebesgue almost-everywhere sense, the above gives the Radon--Nikodym density of \(\mu_{\mathrm{in}}\)).
\end{definition}

\begin{theorem}[Poisson decomposition of the phase current and endpoint positivity]\label{thm:app-inner-phase-current-poisson}
Under the conditions of Definition~\ref{def:app-inner-phase-current}, \(\mu_{\mathrm{in}}\) is a non-negative Borel measure on \(\TT\), and there exists a real-valued function \(\phi\) (understood modulo \(2\pi\), which may be taken as the ``unwound'' version of the inner factor boundary phase) such that in the Lebesgue almost-everywhere sense
\[
I(e^{\mathrm{i}\theta})=e^{\mathrm{i}\phi(\theta)},
\]
and its distributional derivative satisfies
\begin{equation}\label{eq:app-inner-phase-current-as-arg-derivative}
\boxed{\;
\frac{1}{2\pi}\,d\phi(\theta)=d\mu_{\mathrm{in}}(\theta)\; }.
\end{equation}
Therefore the contribution of disk-interior zeros to the phase current is entirely positive and uncancellable.
In particular, the phase current density at the endpoint \(-1\) satisfies
\begin{equation}\label{eq:app-endpoint-phase-current-lowerbound}
\boxed{\;
j_{\mathrm{in}}(\pi)
=\sum_k P_{a_k}(-1)
=\sum_k \mathcal{B}_{-1}(a_k)\ \in[0,\infty]\; }.
\end{equation}
\end{theorem}

\begin{proof}
For a single Blaschke factor \(b_a(w):=\frac{|a|}{a}\frac{a-w}{1-\overline a\,w}\), its boundary value satisfies \(|b_a(e^{\mathrm{i}\theta})|=1\), and direct computation gives
\(
\frac{d}{d\theta}\arg b_a(e^{\mathrm{i}\theta})=P_a(e^{\mathrm{i}\theta})
\).
For a finite Blaschke product, summing term by term yields the density formula; for a general Blaschke product, the result still holds Lebesgue almost everywhere under the Blaschke condition (see \cite{Duren1970TheoryHp,Garnett2007BoundedAnalyticFunctions} for the inner factor theory and phase variation discussion).
The standard representation \eqref{eq:app-singular-inner-factor-measure} of the singular inner factor \(S\) shows that its phase singular component is carried by \(\nu\), thus yielding \eqref{eq:app-inner-phase-current-measure} and \eqref{eq:app-inner-phase-current-as-arg-derivative}.
The endpoint formula \eqref{eq:app-endpoint-phase-current-lowerbound} follows directly from \eqref{eq:app-poisson-kernel-endpoint-minus1} and Definition~\ref{def:app-inner-phase-current}. \qed
\end{proof}

\begin{corollary}[Endpoint phase lower bound for off-critical-line shifts]\label{cor:app-endpoint-phase-current-offcritical}
Under the conditions of Theorem~\ref{thm:app-inner-phase-current-poisson}, if the disk-interior zero set contains points \(w_{\rho_k}\in\mathbb{D}\) transported from off-critical-line zeros \(\rho_k=\beta_k+\mathrm{i}\gamma_k\) (\(0<\beta_k<\tfrac12\)), and writing \(\delta_k:=\tfrac12-\beta_k>0\), then
\[
\boxed{\;
j_{\mathrm{in}}(\pi)\ \ge\ \sum_k \delta_k\;}
\]
where \(\delta_k=\mathcal{B}_{-1}(w_{\rho_k})\) (Corollary~\ref{cor:app-offcritical-horocycle-busemann}).
\end{corollary}

\begin{corollary}[Endpoint phase certificate]\label{cor:app-endpoint-phase-certificate}
Under the above conditions, further assume \(S\equiv 1\) (no singular inner factor).
If \(j_{\mathrm{in}}(\pi)=0\), then \(B\equiv 1\), so that \(F_{\zeta}\) has no zeros in \(\mathbb{D}\); by Theorem~\ref{thm:app-rh-iff-disk-zero-free}, RH follows.
\par
More generally, if an auditable upper bound \(j_{\mathrm{in}}(\pi)\le M<\infty\) exists, then the total off-critical-line shift satisfies
\[
\boxed{\;\sum_k \delta_k\le M\;}
\]
(where \(\delta_k\) ranges over the shifts of all off-critical-line zeros within the strip).
\end{corollary}

\begin{proof}
Under \(S\equiv 1\), by \eqref{eq:app-endpoint-phase-current-lowerbound} one has \(j_{\mathrm{in}}(\pi)=\sum_k \mathcal{B}_{-1}(a_k)\).
Each term \(\mathcal{B}_{-1}(a_k)=\frac{1-|a_k|^2}{|1+a_k|^2}\) is strictly positive, so \(j_{\mathrm{in}}(\pi)=0\) if and only if the zero set is empty, whence \(B\) is a constant inner factor; under standard normalization one may take \(B\equiv 1\).
Thus \(F_{\zeta}\) has no zeros inside the disk, and by Theorem~\ref{thm:app-rh-iff-disk-zero-free} RH follows.
The upper-bound case follows directly from Corollary~\ref{cor:app-endpoint-phase-current-offcritical}. \qed
\end{proof}

\begin{remark}[Endpoint hair flux and radial absorption coefficient]\label{rem:app-endpoint-hair-flux-absorption}
The endpoint density in \eqref{eq:app-endpoint-phase-current-lowerbound},
\(
j_{\mathrm{in}}(\pi)=\sum_k P_{a_k}(-1)
\),
is not only a single-point certificate for the inner factor phase, but can also be interpreted as the \emph{hair flux} (or absorption coefficient) of the Blaschke zero data at the endpoint \(-1\):
it compresses the disk-interior zero set into a non-negative scalar, readable under radial endpoint approach as a modulus decay rate (Proposition~\ref{prop:app-endpoint-blaschke-radial-absorption}).
\end{remark}

\begin{proposition}[Endpoint radial absorption law (finite Blaschke product)]\label{prop:app-endpoint-blaschke-radial-absorption}
Let \(B\) be a finite Blaschke product with zero set \(\{a_k\}_{k=1}^{N}\subset\mathbb{D}\) (counted with multiplicity), and write
\[
\Phi_{-1}(B):=\sum_{k=1}^{N}P_{a_k}(-1)=\sum_{k=1}^{N}\frac{1-|a_k|^2}{|1+a_k|^2}.
\]
Then as \(r\uparrow 1\), along the endpoint radial direction \(w=-r\) one has the asymptotic law
\begin{equation}\label{eq:app-endpoint-blaschke-radial-absorption}
\boxed{\;
\log|B(-r)|=-(1-r)\,\Phi_{-1}(B)+o(1-r)\; }.
\end{equation}
In particular, \(\Phi_{-1}(B)=0\) if and only if \(B\equiv 1\).
\end{proposition}

\begin{proof}
For a single Blaschke factor \(b_a(w)=\frac{|a|}{a}\frac{a-w}{1-\overline a\,w}\) (ignoring constant phase factors) one has
\[
\log|b_a(-r)|
=\log|a+r|-\log|1+r\overline a|.
\]
Direct differentiation gives
\[
\left.\frac{d}{dr}\log|b_a(-r)|\right|_{r=1^-}
=\Re\!\left(\frac{1}{1+a}-\frac{\overline a}{1+\overline a}\right)
=-\,\frac{1-|a|^2}{|1+a|^2}
=-P_a(-1),
\]
so a Taylor expansion gives
\(
\log|b_a(-r)|=-(1-r)P_a(-1)+o(1-r)
\).
For a finite product \(B=\prod_{k=1}^{N}b_{a_k}\), summing term by term yields \eqref{eq:app-endpoint-blaschke-radial-absorption};
the final statement follows from the term-by-term positivity \(P_{a_k}(-1)>0\). \qed
\end{proof}
