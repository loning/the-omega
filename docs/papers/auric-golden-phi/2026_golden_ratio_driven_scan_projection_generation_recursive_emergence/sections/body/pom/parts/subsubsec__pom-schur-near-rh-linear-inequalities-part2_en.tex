\paragraph{Final Quasi-Periodic Linear Law for Maximal Partition Energy and Large-\texorpdfstring{$q$}{q} Necessary Condition for near-RH}
We continue with the pressure notation \(P(r)\) and per-unit pressure \(a_r:=P(r)/r\) from \S\ref{subsubsec:pom-schur-character-sieve-thermodynamic}.
This section compresses the "maximal partition linear pattern" into a principal linear term plus quasi-periodic correction, and accordingly derives a linear necessary condition for near-RH at large \(q\).

\begin{proposition}[Final Quasi-Periodic Linear Law for Maximal Partition Energy: Bounded Deficit Determined by the Per-Unit Pressure Peak Set]\label{prop:pom-schur-partition-energy-eventual-quasiperiodic-linear-law}
Let \(a_r:=P(r)/r\) be the per-unit pressure, and define the maximal partition energy
\[
L^\star(q):=\max\left\{\sum_{r\ge 1}c_r P(r):\ c_r\in\NN_0,\ \sum_{r\ge 1}r c_r=q\right\}.
\]
Let
\[
a_*:=\max_{r\ge 1}a_r
\]
be attained on the finite nonempty set
\(
R_*:=\{r:\ a_r=a_*\}
\),
and suppose there exists a strict gap
\[
\eta:=a_*-\max_{r\notin R_*}a_r>0,
\qquad
g:=\gcd(R_*).
\]
Then there exist constants \(Q_0,C>0\) and a \(g\)-periodic function \(\beta:\ZZ/g\ZZ\to[-C,0]\) such that for all \(q\ge Q_0\) one has
\[
\boxed{\;
L^\star(q)=q\,a_*+\beta(q\bmod g)\; }.
\]
Moreover, any partition attaining \(L^\star(q)\) uses parts outside \(R_*\) only in a bounded "correction window":
more precisely, there exists a constant \(M\) independent of \(q\) such that any optimal solution satisfies
\[
\sum_{r\notin R_*} r c_r\le M,
\]
and its energy loss is at least \(\eta\) times the left-hand side above.
\end{proposition}
\begin{proof}
For any feasible \((c_r)\), one has
\[
\sum_r c_r P(r)=\sum_r (r c_r)a_r
\le a_*\sum_r r c_r-\eta\sum_{r\notin R_*}r c_r
=q\,a_*-\eta\sum_{r\notin R_*}r c_r.
\]
Therefore, in any optimal solution, if the total length of non-peak parts \(\sum_{r\notin R_*}r c_r\) were to tend to infinity, the objective value would decline relative to \(q a_*\) at a linear rate of at least \(\eta\), and thus could not remain optimal;
hence this total length must be bounded by some constant \(M\) independent of \(q\), simultaneously giving the energy-loss lower bound.
\par
On this basis, write \(q\) as
\(
q=q_0+\sum_{r\in R_*} r c_r
\), where \(q_0:=\sum_{r\notin R_*} r c_r\) lies in the finite set \(\{0,1,\dots,M\}\).
Because the sum semigroup of \(R_*\) is eventually cofinite in each congruence class modulo \(g\) (cofiniteness of numerical semigroups),
for each congruence class \(q\bmod g\) one can enumerate all possible \(q_0\) and corresponding peak-set combinations in a finite window, obtaining a maximum deficit
\(
\beta(q\bmod g)\in[-C,0]
\)
depending only on the congruence class;
once \(q\) is sufficiently large, this enumeration window already covers all optimal correction patterns, so \(L^\star(q)=q a_*+\beta(q\bmod g)\) holds stably.
\end{proof}

\begin{corollary}[Linear Necessary Condition for near-RH at Large \texorpdfstring{$q$}{q}, Quasi-Periodized: Square-Root Threshold Enforces a Linear Lower Bound on \texorpdfstring{$P(q)$}{P(q)}]\label{cor:pom-schur-near-rh-linear-necessary-quasiperiodic}
Under the setting of Proposition \ref{prop:pom-schur-partition-energy-eventual-quasiperiodic-linear-law},
further assume that the Schur stratification is nondegenerate for each \(q\) so that the nontrivial Schur envelope index is given by the partition energy, i.e.,
\[
\max_{\lambda\vdash q,\ \lambda\neq(q)}\log\rho(B_{q,\lambda})=L^\star(q).
\]
Then the level-\(q\) near-RH (square-root threshold) condition
\[
\max_{\lambda\neq(q)}\rho(B_{q,\lambda})\le \rho(B_{q,(q)})^{1/2}
\qquad(\rho(B_{q,(q)})=e^{P(q)})
\]
implies the linear necessary condition for all sufficiently large \(q\)
\[
\boxed{\;
P(q)\ \ge\ 2L^\star(q)\ =\ 2q\,a_*+2\beta(q\bmod g)\; }.
\]
\end{corollary}
\begin{proof}
near-RH is equivalent to
\(
\max_{\lambda\neq(q)}\log\rho(B_{q,\lambda})\le P(q)/2
\).
Substituting the nondegeneracy assumption \(\max_{\lambda\neq(q)}\log\rho(B_{q,\lambda})=L^\star(q)\) gives \(P(q)\ge 2L^\star(q)\).
Then applying the quasi-periodic expansion from Proposition \ref{prop:pom-schur-partition-energy-eventual-quasiperiodic-linear-law} yields the result.
\end{proof}
