\section{黄金分支的稳定折叠：Zeckendorf 语法与分辨率压缩}\label{sec:folding}

本节给出一个完全可计算的“稳定类型折叠”机制，用以将分辨率 $m$ 下的全部微态 $\Omega_m$ 压缩为 $\abs{X_m}=F_{m+2}$ 个稳定类型，并展示其递归结构由 $\varphi$ 控制。更一般的 cut-and-project 场景下，局部几何信息可经滑动窗口编码为二元词，再施加同样的稳定语法约束并折叠为稳定类型直方图（附录 F 给出接口表述）。

\subsection{Fibonacci 权重与稳定语法集合}

令 Fibonacci 数列 $(F_k)_{k\ge1}$ 满足
\[
F_1=F_2=1,\quad F_{k+2}=F_{k+1}+F_k.
\]
定义长度 $m$ 的稳定类型集合
\[
X_m:=\{w\in\{0,1\}^m:\text{不含子词 }11\}.
\]
即 $X_m$ 为禁止相邻 $1$ 的二元语法（亦称“黄金均值约束”）。其基数满足递归
\[
|X_m|=|X_{m-1}|+|X_{m-2}|,\quad |X_1|=2,\ |X_2|=3,
\]
因此
\[
|X_m|=F_{m+2}\sim \frac{\varphi^{m+2}}{\sqrt5}.
\]
作为符号动力系统对象，$\{X_m\}$ 对应一个两状态、禁止 $11$ 的有限型子移位（golden mean shift），其谱半径与最大熵测度均由 Perron--Frobenius 理论控制（见 \cite{LindMarcus1995} 与第 8 节）。

\subsection{Zeckendorf 规范形}

对任意 $N\in\NN$，存在唯一表示
\[
N=\sum_{k\ge1}c_k F_{k+1},\qquad c_k\in\{0,1\},\quad c_kc_{k+1}=0.
\]
将 $(c_k)_{k\ge1}$ 视为 $N$ 的 Zeckendorf 数字序列，记为 $Z(N)=(c_1,c_2,\dots)$。

定义截断映射
\[
\pi_m(c_1,c_2,\dots)=(c_1,\dots,c_m)\in X_m.
\]

\subsection{多尺度一致性：restriction 逆系统与相容性口径}\label{subsec:folding-multiscale}

当比较不同窗口长度 $m$ 的输出时，“一致”必须被精确定义为跨尺度相容性：高分辨率输出经由自然的截断（restriction）投影到低分辨率后，应复现低分辨率输出所包含的内容。下面给出对象层的标准构造（完整证明见附录 G）。

\subsubsection{折叠与因子化的核心定理（汇总陈述）}\label{subsec:fold-theorem-suite}
为回应“折叠是否提供了一个自洽且可审计的新接口”的审稿口径，本小节把由折叠专门引起的关键结构压缩为一个汇总定理：它给出折叠的精确定义、归约顺序无关（终止/合流）、幂等/满射、多尺度 restriction 逆系统相容、序列级 sofic 因子与最小右分辨覆盖，以及差异度证书到折叠直方图 TV 偏差的定量链条。

\begin{theorem}[有限窗口折叠：合流、逆系统相容、sofic 因子与定量证书]\label{thm:fold-suite}
设 $\Omega_m=\{0,1\}^m$，$X_m=\{w\in\{0,1\}^m:\text{不含子词 }11\}$，并令 $\Fold_m:\Omega_m\to X_m$ 如定义 \ref{def:fold-word}。则：
\begin{enumerate}
  \item \textbf{（归约顺序无关的规范化）}折叠可由值保持的局部重写系统实现，且该系统强终止并合流；因此 $\Fold_m$ 的实现与归约顺序无关。
  \item \textbf{（幂等与满射）}\(\Fold_m(\omega)\in X_m\)；并且 \(\Fold_m|_{X_m}=\mathrm{id}\)、\(\Fold_m:\Omega_m\twoheadrightarrow X_m\)。
  \item \textbf{（restriction 逆系统与逆极限）}对任意 $m_2\ge m_1$，截断 $\pi_{m_2\to m_1}:X_{m_2}\to X_{m_1}$ 构成 restriction 逆系统，且存在自然同构 \(X_\infty\cong\varprojlim X_m\)（定理 \ref{thm:inverse-limit-golden}）。
  \item \textbf{（跨分辨率相容的交换图）}朴素微态截断一般不与折叠交换；但定义 \ref{subsec:fold-omega-restriction} 的折叠感知 restriction \(\widetilde r_{m+1\to m}(\omega)=\pi_{m+1\to m}(\Fold_{m+1}(\omega))\) 使得交换图成立（命题 \ref{prop:fold-omega-commute}）。
  \item \textbf{（序列级因子与显式 sofic 图像）}由 \(\Fold_m\) 沿滑动窗口诱导的 \(\Phi_m\) 是滑动块码，记忆为 $0$、预期性为 $m-1$（命题 \ref{prop:Phi_m-radius}）；其像空间 \(Y_m\subset X_m^{\ZZ}\) 是 sofic 子移位，并可由显式标记图 \(G_m\) 给出（定理 \ref{thm:Phi_m-sofic-graph}）。
  \item \textbf{（最小右分辨覆盖）}对 \(Y_m\)，右分辨表示可由子集确定化得到（命题 \ref{prop:Phi_m-right-resolving}）；进一步按 follower 集最小化得到右 Fischer cover，且其右分辨/最小性满足命题 \ref{prop:fischer-minimal} 的结论。
  \item \textbf{（定量证书：差异度 \(\Rightarrow\) 折叠直方图 TV 偏差）}在旋转扫描（Sturmian）模型下，对任意 $N\ge 1$，
  \[
  D_{\mathrm{TV}}(\hat{\pi}_{m,N},\pi_m^{\mathrm{fold}})\ \le\ (m+1)\,D_N^*(\alpha;x_0),
  \]
  且到 Parry 基线的误差分解满足推论 \ref{cor:tv-decomposition-parry}。
\end{enumerate}
\end{theorem}

\begin{proof}
(1) 由命题 \ref{prop:fold-rewrite-newman}（强终止 + 合流）与推论 \ref{cor:foldm-order-indep}。

(2) 由命题 \ref{prop:fold-basic}。

(3) restriction 一致性由引理 \ref{lem:pi-functorial-golden}；逆极限同构由定理 \ref{thm:inverse-limit-golden}（完整证明见附录 G 的定理 \ref{app:multiscale-inverse-limit}）。

(4) 由小节 \ref{subsec:fold-omega-restriction} 的定义与命题 \ref{prop:fold-omega-commute}。

(5) 半径由命题 \ref{prop:Phi_m-radius}；sofic 图像与显式标记图由定理 \ref{thm:Phi_m-sofic-graph}。

(6) 由命题 \ref{prop:Phi_m-right-resolving} 与命题 \ref{prop:fischer-minimal}。

(7) 由第 \ref{sec:experiments} 节的定理 \ref{thm:tv-certificate-hist}（其证明给出显式系数 $m+1$），并由推论 \ref{cor:tv-decomposition-parry} 得到到 Parry 基线的误差分解。
\end{proof}

\begin{definition}[尺度投影（restriction）]\label{def:pi-m2-m1-golden}
对任意 $m_2\ge m_1$，定义截断投影
\[
\pi_{m_2\to m_1}:X_{m_2}\to X_{m_1},\qquad
\pi_{m_2\to m_1}(x_1,\dots,x_{m_2})=(x_1,\dots,x_{m_1}).
\]
\end{definition}

\begin{lemma}[一致性]\label{lem:pi-functorial-golden}
对任意 $m_3\ge m_2\ge m_1$ 有
\[
\pi_{m_3\to m_1}=\pi_{m_2\to m_1}\circ \pi_{m_3\to m_2}.
\]
并且对任意无穷合法序列 $c=(c_1,c_2,\dots)$ 有
\[
\pi_{m_2\to m_1}\bigl(\pi_{m_2}(c)\bigr)=\pi_{m_1}(c).
\]
\end{lemma}

\begin{theorem}[逆极限同构（分辨率无关对象）]\label{thm:inverse-limit-golden}
令
\[
X_\infty:=\{(c_1,c_2,\dots)\in\{0,1\}^{\mathbb{N}}:\ c_kc_{k+1}=0\ \forall k\ge 1\}.
\]
则存在自然同构
\[
X_\infty\ \cong\ \varprojlim X_m,
\]
并且 $X_m$ 可视为 $X_\infty$ 的有限截断因子：不同 $m$ 的差异仅在“可见前缀长度”上，而非在统一对象本身上。
\end{theorem}

\subsubsection{关于微态截断与折叠的相容性：正确的交换图}\label{subsec:fold-omega-restriction}
上述逆系统结构发生在稳定语法集合 $\{X_m\}$ 上，其 restriction 映射是定义 \ref{def:pi-m2-m1-golden} 的截断 $\pi_{m_2\to m_1}:X_{m_2}\to X_{m_1}$。一个自然但需要澄清的问题是：能否在微态空间 $\Omega_m=\{0,1\}^m$ 上直接用“去掉末位”的截断与折叠 $\Fold_m$ 形成交换图？

\paragraph{朴素截断一般不与折叠交换}
令
\[
r_{m+1\to m}:\Omega_{m+1}\to\Omega_m,\qquad
r_{m+1\to m}(\omega_1\cdots\omega_m\omega_{m+1}):=\omega_1\cdots\omega_m
\]
为微态上的普通截断。一般地，
\[
\Fold_m\circ r_{m+1\to m}\ \neq\ \pi_{m+1\to m}\circ \Fold_{m+1}.
\]
例如取 $m=1$、$\omega=11$，则 $r_{2\to 1}(11)=1$，从而 $(\Fold_1\circ r_{2\to 1})(11)=\Fold_1(1)=1$；另一方面 $N(11)=F_2+F_3=3$ 的 Zeckendorf 规范形为 $3=F_4$，故 $\Fold_2(11)=00$，从而 $(\pi_{2\to 1}\circ \Fold_2)(11)=0$。

\paragraph{折叠感知的 restriction（在 $\Omega$ 上实现交换）}
为在微态层面得到严格交换的形式化，我们定义一个与折叠相容的 restriction：
\[
\widetilde r_{m+1\to m}:\Omega_{m+1}\to \Omega_m,\qquad
\widetilde r_{m+1\to m}(\omega):=\pi_{m+1\to m}\bigl(\Fold_{m+1}(\omega)\bigr).
\]
注意 $\widetilde r_{m+1\to m}(\omega)\in X_m\subset\Omega_m$，因此该定义确实给出一条 $\Omega_{m+1}\to\Omega_m$ 的映射（它不是“原始截断”，而是“先折叠后截断”的可审计 coarse-graining）。

\begin{proposition}[跨分辨率相容的交换图]\label{prop:fold-omega-commute}
对任意 $m\ge 1$，下图交换：
\[
\begin{CD}
\Omega_{m+1} @>{\Fold_{m+1}}>> X_{m+1}\\
@V{\widetilde r_{m+1\to m}}VV @VV{\pi_{m+1\to m}}V\\
\Omega_m @>{\Fold_m}>> X_m.
\end{CD}
\]
即对任意 $\omega\in\Omega_{m+1}$ 有
\[
\Fold_m\bigl(\widetilde r_{m+1\to m}(\omega)\bigr)
\ =\
\pi_{m+1\to m}\bigl(\Fold_{m+1}(\omega)\bigr).
\]
\end{proposition}

\begin{proof}
由定义，$\widetilde r_{m+1\to m}(\omega)=\pi_{m+1\to m}(\Fold_{m+1}(\omega))\in X_m$。而命题 \ref{prop:fold-basic} 的幂等性给出：对任意 $x\in X_m$，$\Fold_m(x)=x$。因此
\[
\Fold_m\bigl(\widetilde r_{m+1\to m}(\omega)\bigr)
=\Fold_m\!\left(\pi_{m+1\to m}(\Fold_{m+1}(\omega))\right)
=\pi_{m+1\to m}(\Fold_{m+1}(\omega)).
\]
证毕。
\end{proof}

\subsection{分辨率折叠映射（folding）}

折叠的目标是：把任意窗口微态 $\omega=\omega_1\cdots \omega_m\in\Omega_m$ 规范化为一个合法词 $x\in X_m$。

\begin{lemma}[权重的刚性：二阶递推强制 Fibonacci]\label{lem:fib-weight-rigidity}
设权重序列 $(a_k)_{k\ge 1}$ 满足二阶线性递推 $a_{k+2}=a_{k+1}+a_k$ 且归一化 $a_1=a_2=1$。则对所有 $k\ge 1$ 有 $a_k=F_k$。
\end{lemma}

\begin{proof}
由初值 $a_1=a_2=1$ 与递推 $a_{k+2}=a_{k+1}+a_k$ 归纳得 $a_3=2,a_4=3,\dots$，并与 Fibonacci 数列定义一致，故 $a_k=F_k$。
\end{proof}

为避免任何“人为编号”，本文直接以 Fibonacci 权重给微态赋一个可审计整数值，再取 Zeckendorf 规范形：
\begin{definition}[Fibonacci 加权值与折叠算子]\label{def:fold-word}
对 $\omega=\omega_1\cdots\omega_m\in\Omega_m$ 定义其 Fibonacci 加权值
\[
N(\omega):=\sum_{k=1}^{m}\omega_k F_{k+1}\ \in\ \NN.
\]
令 $Z(N(\omega))=(c_1,c_2,\dots)$ 为 $N(\omega)$ 的 Zeckendorf 数字（\cite{Zeckendorf1972}），定义分辨率 $m$ 的折叠
\[
\Fold_m:\Omega_m\to X_m,\qquad
\Fold_m(\omega):=\pi_m\!\bigl(Z(N(\omega))\bigr).
\]
\end{definition}

\subsubsection{折叠的局部重写实现：终止、合流与顺序无关}\label{subsec:fold-local-rewrite}
定义 \ref{def:fold-word} 以 ``先取值、再取 Zeckendorf 规范形'' 给出 $\Fold_m$。为了将其落实为\textbf{局部、可执行}的归一化过程，并澄清 ``归约顺序不影响结果''，我们把输入窗口嵌入到允许中间态出现冗余数字的配置空间上实施值保持重写。

\paragraph{配置空间与值函数}
令
\[
\mathcal{D}:=\Bigl\{a=(a_1,a_2,\dots): a_k\in\NN,\ \text{仅有限个 }a_k\neq 0\Bigr\}
\]
为有限支撑的非负整数数字配置。定义值同态
\[
\mathrm{Val}(a):=\sum_{k\ge 1} a_k F_{k+1}.
\]
对给定 $\omega\in\Omega_m$，记其嵌入为
\[
\iota_m(\omega):=(\omega_1,\dots,\omega_m,0,0,\dots)\in\mathcal{D},
\]
则 $\mathrm{Val}(\iota_m(\omega))=N(\omega)$。

\paragraph{局部重写规则（值保持）}
对 $a\in\mathcal{D}$，允许在满足条件时对局部模式做如下替换（其中 $e_k$ 为第 $k$ 位单位向量）：
\begin{itemize}
  \item \textbf{(R\_adj) 相邻合并：}若 $a_k\ge 1$ 且 $a_{k+1}\ge 1$，则
  \[
  a\ \to\ a-e_k-e_{k+1}+e_{k+2}.
  \]
  \item \textbf{(R\_1) 底位去重：}若 $a_1\ge 2$，则
  \[
  a\ \to\ a-2e_1+e_2.
  \]
  \item \textbf{(R\_2) 第二位去重：}若 $a_2\ge 2$，则
  \[
  a\ \to\ a-2e_2+e_1+e_3.
  \]
  \item \textbf{(R\_k) 一般位去重（$k\ge 3$）：}若 $a_k\ge 2$，则
  \[
  a\ \to\ a-2e_k+e_{k-2}+e_{k+1}.
  \]
\end{itemize}
每条规则都只改动有限个坐标，因此是严格的\textbf{局部}重写；并且它们分别对应恒等式
\(
F_{k+1}+F_{k+2}=F_{k+3},
2F_2=F_3,
2F_3=F_2+F_4,
2F_{k+1}=F_{k-1}+F_{k+2}
\)，因此每一步都保持 $\mathrm{Val}$ 不变。

\paragraph{不可约项与 Zeckendorf 规范形}
记 $X_\infty=\{(c_1,c_2,\dots)\in\{0,1\}^{\mathbb{N}}:\ c_kc_{k+1}=0\}$。显然 $a\in X_\infty$ 时上述规则均不可用，故 $X_\infty$ 中元素不可约；反之，若 $a$ 不可约，则必有 $a_k\le 1$ 且不存在相邻的正数，故 $a\in X_\infty$。因此不可约项恰为 $X_\infty$。

\begin{proposition}[终止性、局部合流与合流（纽曼引理）]\label{prop:fold-rewrite-newman}
上述重写系统在 $\mathcal{D}$ 上强终止，且局部合流；因此由纽曼引理它是合流的。于是每个 $a\in\mathcal{D}$ 的正规形存在且唯一，记为 $\mathrm{NF}(a)\in X_\infty$，并满足
\[
\mathrm{Val}(\mathrm{NF}(a))=\mathrm{Val}(a).
\]
\end{proposition}

\begin{proof}
\textbf{（终止性）}令
\[
A(a)=\sum_{k\ge 1} a_k,\qquad
B(a)=\sum_{k\ge 1} k\,a_k,\qquad
C(a)=a_2,
\qquad
\mu(a)=(A(a),B(a),C(a))
\]
按字典序比较。则 (R\_adj) 与 (R\_1) 使 $A$ 减少 1；$(R_k,\,k\ge 3)$ 使 $B$ 减少 1；(R\_2) 保持 $A,B$ 不变但使 $C$ 减少 2。故每步重写都严格降低 $\mu$，系统强终止。

\textbf{（局部合流）}任取 $a$ 的两条一步归约 $a\to a_1$ 与 $a\to a_2$。由终止性，$a_1,a_2$ 均可继续归约到不可约正规形 $\mathrm{NF}(a_1),\mathrm{NF}(a_2)\in X_\infty$。又因每一步保持 $\mathrm{Val}$，有
\[
\mathrm{Val}(\mathrm{NF}(a_1))=\mathrm{Val}(a_1)=\mathrm{Val}(a)=\mathrm{Val}(a_2)=\mathrm{Val}(\mathrm{NF}(a_2)).
\]
而在 $X_\infty$ 中，同值元素由 Zeckendorf 唯一性必相同，故 $\mathrm{NF}(a_1)=\mathrm{NF}(a_2)$，从而两条一步归约可汇合，系统局部合流。

\textbf{（合流）}由纽曼引理（强终止 $\Rightarrow$ 局部合流等价于合流）得系统合流。唯一正规形与值保持性质随之成立。
\end{proof}

\begin{corollary}[\texorpdfstring{$\Fold_m$}{Fold\_m} 的归约顺序无关实现]\label{cor:foldm-order-indep}
对任意 $\omega\in\Omega_m$，
\[
\Fold_m(\omega)\ =\ \pi_m\bigl(\mathrm{NF}(\iota_m(\omega))\bigr),
\]
且右端与归约时选取规则/位置的顺序无关。
\end{corollary}

\begin{proposition}[折叠的规范性、幂等性与满射性]\label{prop:fold-basic}
折叠算子 $\Fold_m$ 良定义且满足：
\begin{enumerate}
  \item \textbf{（规范性）} $\Fold_m(\omega)\in X_m$；
  \item \textbf{（幂等性）} 若 $\omega\in X_m$，则 $\Fold_m(\omega)=\omega$；
  \item \textbf{（满射性）} $\Fold_m:\Omega_m\twoheadrightarrow X_m$ 为满射。
\end{enumerate}
\end{proposition}

\begin{proof}
\textbf{（规范性）}由定义 \ref{def:fold-word}，$Z(N(\omega))\in X_\infty$，故 $\pi_m(Z(N(\omega)))\in X_m$。

\textbf{（幂等性）}若 $\omega\in X_m$，令 $N=\sum_{k=1}^m \omega_k F_{k+1}$。由于 $\omega$ 已是合法 Zeckendorf 数字的长度-$m$ 前缀，且 Zeckendorf 表示唯一，故 $Z(N)$ 的前 $m$ 位必等于 $\omega$，从而 $\Fold_m(\omega)=\omega$。

\textbf{（满射性）}对任意 $x\in X_m$，取 $\omega=x\in\Omega_m$，由幂等性得 $\Fold_m(\omega)=x$，故 $\Fold_m$ 满射。
\end{proof}

\subsection{稳定压缩率与分辨率不足的形式化}

折叠压缩率由
\[
\frac{|\Omega_m|}{|X_m|}=\frac{2^m}{F_{m+2}}
\]
刻画。由于 $F_{m+2}\asymp \varphi^m$，平均折叠多重度具有指数增长率 $(2/\varphi)^m$。因此，即便稳定类型数量随 $m$ 指数增长，微态数量增长更快，导致折叠必然多对一：不同微态在有限分辨率下不可区分而被归入同一稳定类型。该结论为“分辨率不足导致规律不可读出”的严格表述：不可读出并非缺乏规律，而是稳定投影将大量微差异压缩到同一类中，从而在微观层面丢失可区分信息。

\subsection{序列层滑动块因子：由 \texorpdfstring{$\Fold_m$}{Fold\_m} 诱导的 \texorpdfstring{$\Phi_m$}{Phi\_m}}

本节回答一个结构性问题：$\Fold_m$ 本身是\emph{窗口层}映射 $\Omega_m\to X_m$，并不直接对应双边子移位之间的因子映射；然而若将其施加于双边序列的滑动窗口，则得到一个与时间平移相容（移位等变）的序列层算子，从而可在符号动力系统框架内讨论其像空间的转移结构。

记 $\sigma$ 为左移位：对 $s\in\{0,1\}^{\ZZ}$，定义 $(\sigma s)_t:=s_{t+1}$。将有限集合 $X_m$ 视为字母表，记 $X_m^{\ZZ}$ 的左移位同样记为 $\sigma$。

\begin{definition}[序列层折叠（滑动块码）]\label{def:Phi_m}
对任意分辨率 $m\ge 1$，定义映射
\[
\Phi_m:\{0,1\}^{\ZZ}\to X_m^{\ZZ},\qquad
(\Phi_m(s))_t:=\Fold_m\bigl(s_t s_{t+1}\cdots s_{t+m-1}\bigr).
\]
\end{definition}

\begin{proposition}[记忆/预期性（窗口半径）]\label{prop:Phi_m-radius}
$\Phi_m$ 是一个滑动块码，其\textbf{记忆}为 $0$、\textbf{预期性}为 $m-1$：即对每个时刻 $t$，输出符号 $(\Phi_m(s))_t$ 仅依赖于输入坐标
\(
s_t,s_{t+1},\dots,s_{t+m-1}
\)。
\end{proposition}

\begin{proof}
由定义 \ref{def:Phi_m} 直接可见；这也是滑动块码“记忆/预期性（左/右半径）”的标准口径（参见 \cite{LindMarcus1995,Kitchens1998SymbolicDynamics}）。
\end{proof}

\begin{proposition}[移位等变性（时间相容）]\label{prop:Phi_m-equivariant}
对任意 $m\ge 1$ 有 $\Phi_m\circ\sigma=\sigma\circ\Phi_m$。
\end{proposition}

\begin{proof}
由定义 \ref{def:Phi_m}，对任意 $t\in\ZZ$，
\[
(\Phi_m(\sigma s))_t
=\Fold_m\bigl((\sigma s)_t\cdots(\sigma s)_{t+m-1}\bigr)
=\Fold_m\bigl(s_{t+1}\cdots s_{t+m}\bigr)
=(\Phi_m(s))_{t+1}
=(\sigma\Phi_m(s))_t.
\]
证毕。
\end{proof}

\begin{definition}[像空间]\label{def:Y_m}
定义 $\Phi_m$ 的像
\[
Y_m:=\Phi_m\bigl(\{0,1\}^{\ZZ}\bigr)\ \subset\ X_m^{\ZZ}.
\]
\end{definition}

\begin{remark}[不要将 $Y_m$ 与“黄金分割移位”混淆]\label{rem:Ym-not-golden-mean}
需要强调：$X_m$ 是“禁止相邻 $11$”语法在长度 $m$ 上的合法词集合（字母表为 $X_m$），而 $Y_m$ 是字母表 $X_m$ 上的一个子移位；因此 $Y_m$ 并不是字母表 $\{0,1\}$ 上的黄金分割 SFT（禁词 $11$ 的 golden mean shift）。

更具体地：
\begin{itemize}
  \item 当 $m=1$ 时，$X_1=\{0,1\}$ 且 $\Fold_1$ 为恒等，因此 $\Phi_1$ 是 $\{0,1\}^{\ZZ}$ 上的恒等映射，从而 $Y_1=\{0,1\}^{\ZZ}$（全移位），这与黄金分割移位不同。
  \item 当 $m\ge 2$ 时，字母表大小为 $|X_m|=F_{m+2}>2$，故 $Y_m$ 的自然字母表不是 $\{0,1\}$；其结构需在 $X_m^{\ZZ}$ 上讨论，并由 sofic 图像理论刻画（见命题 \ref{prop:Y_m-sofic} 与定理 \ref{thm:Phi_m-sofic-graph}）。
\end{itemize}
\end{remark}

\begin{proposition}[对子移位的因子性质（序列级折叠的最小口径）]\label{prop:Phi_m-factor-on-subshift}
设 $S\subset\{0,1\}^{\ZZ}$ 为任意子移位（紧致、移位不变闭集）。则：
\begin{enumerate}
  \item $\Phi_m|_S:S\to X_m^{\ZZ}$ 连续且与移位交换；
  \item 其像 $Y_m(S):=\Phi_m(S)$ 是 $X_m^{\ZZ}$ 的一个子移位；
  \item 映射 $\Phi_m|_S:S\twoheadrightarrow Y_m(S)$ 是一个因子映射。
\end{enumerate}
\end{proposition}

\begin{proof}
由于 $\Phi_m$ 是长度 $m$ 的滑动块码（定义 \ref{def:Phi_m}），它对全移位是连续映射；限制到闭子集 $S$ 仍连续。命题 \ref{prop:Phi_m-equivariant} 已证 $\Phi_m$ 与移位交换，限制后同样成立。

又因为 $S$ 紧致且 $\Phi_m$ 连续，$Y_m(S)=\Phi_m(S)$ 为紧致闭集；并且对任意 $y=\Phi_m(s)\in Y_m(S)$，有
\[
\sigma y=\sigma\Phi_m(s)=\Phi_m(\sigma s)\in \Phi_m(S)=Y_m(S),
\]
故 $Y_m(S)$ 移位不变，从而是子移位。最后，“因子映射”按定义即为连续、满射且移位等变的映射；这些性质已逐条建立。
\end{proof}

\begin{proposition}[像空间为 sofic 子移位]\label{prop:Y_m-sofic}
$Y_m$ 是一个 sofic 子移位。更具体地，它可由一个显式有限标记图给出。
\end{proposition}

\begin{proof}
像空间是经典符号动力系统事实：任意滑动块码作用于有限字母表全移位，其像是 sofic 子移位（参见 \cite{LindMarcus1995} 中关于滑动块码与 sofic 系统的标准章节）。为方便审计，我们在下一个命题中给出一个可计算的有限图构造，从而将该标准事实转为本文所需的“可复现转移结构描述”。证毕。
\end{proof}

\subsection{诱导过程与转移结构：显式有限标记图}\label{subsec:Phi_m-graph}

下面给出 $Y_m$ 的一个完全显式的 sofic 图表示，它也提供了“把 $\Fold_m$ 沿长轨迹连续重叠窗口施加后，输出过程的状态/转移结构是什么”的回答。

\begin{definition}[长度 \texorpdfstring{$m-1$}{m-1} 后缀状态图]\label{def:G_m}
令顶点集为
\[
V_m:=\{0,1\}^{m-1}.
\]
对任意顶点 $v=v_1\cdots v_{m-1}\in V_m$ 与任意 $b\in\{0,1\}$，定义一条有向边
\[
e=(v,b):\ v\to v':=v_2\cdots v_{m-1}b,
\]
并用 $\ell_m(e)\in X_m$ 标记该边：
\[
\ell_m(e):=\Fold_m(v_1\cdots v_{m-1}b)\in X_m.
\]
由此得到一个有限标记图 $G_m=(V_m,E_m,\ell_m)$，其中 $|V_m|=2^{m-1}$，$|E_m|=2^{m}$。
\end{definition}

\begin{theorem}[显式 sofic 表示与可计算诱导结构]\label{thm:Phi_m-sofic-graph}
令 $Z_m\subset E_m^{\ZZ}$ 为图 $G_m$ 的双边边移位（即双边路径空间），并令
\[
(\mathcal{L}_m(e))_t:=\ell_m(e_t)\in X_m
\]
为逐边取标记得到的序列。则
\[
Y_m=\mathcal{L}_m(Z_m),
\]
从而 $Y_m$ 是一个由 $G_m$ 给出的 sofic 子移位；并且 $G_m$ 给出一个可审计的“有限状态诱导结构”：状态为长度 $m-1$ 的输入后缀，转移为追加一比特，输出为对应窗口经 $\Fold_m$ 的稳定类型。
\end{theorem}

\begin{proof}
\textbf{（包含 $\subset$）} 给定任意输入序列 $s\in\{0,1\}^{\ZZ}$，定义
\[
v_t:=s_t s_{t+1}\cdots s_{t+m-2}\in V_m,\qquad
e_t:=(v_t,s_{t+m-1})\in E_m.
\]
则 $e=(e_t)_{t\in\ZZ}$ 是 $G_m$ 上的一条双边路径（因为 $v_{t+1}$ 恰为 $v_t$ 去首加尾的更新）。并且由标记定义，
\[
(\mathcal{L}_m(e))_t=\ell_m(e_t)=\Fold_m(s_t\cdots s_{t+m-1})=(\Phi_m(s))_t.
\]
因此 $\Phi_m(s)\in \mathcal{L}_m(Z_m)$，即 $Y_m\subset \mathcal{L}_m(Z_m)$。

\textbf{（包含 $\supset$）} 反之，取任意双边路径 $e=(e_t)\in Z_m$。每条边 $e_t$ 具有形式 $(v_t,b_t)$，且路径条件保证 $v_{t+1}$ 等于 $v_t$ 去首加尾的更新。由此可一致地构造一个二元序列 $s\in\{0,1\}^{\ZZ}$：令 $s_{t+m-1}:=b_t$，并用 $v_t$ 提供窗口的其余 $m-1$ 位（路径相容性保证该定义不矛盾）。于是对该 $s$ 有
\[
(\Phi_m(s))_t=\Fold_m(s_t\cdots s_{t+m-1})=\Fold_m(v_t b_t)=\ell_m(e_t)=(\mathcal{L}_m(e))_t.
\]
因此 $\mathcal{L}_m(e)\in Y_m$，从而 $\mathcal{L}_m(Z_m)\subset Y_m$。两边相等，证毕。
\end{proof}

\begin{remark}[关于 Markov 性与可计算近似]\label{rem:Phi_m-not-markov}
定理 \ref{thm:Phi_m-sofic-graph} 给出的是\emph{拓扑/组合}层面的有限状态表示：$Y_m$ 的所有允许序列由有限图刻画。但对任意输入过程（例如第 \ref{sec:experiments} 节的旋转扫描产生的 Sturmian 序列），其推前测度 $(\Phi_m)_*\mu$ 在 $Y_m$ 上未必是一阶马尔可夫测度。实践上若需要一个“诱导转移结构”的数值近似，可在图 $G_m$ 上用长样本估计边/状态频率，从而得到一个可审计的有限状态 Markov 近似；该近似与第 \ref{sec:quant-fit} 节的直方图/偏差指标天然对齐。
\end{remark}

\subsection{从 sofic 图像到可计算的 Markov 覆盖：邻接矩阵/熵/最大熵测度与 \texorpdfstring{$\zeta$}{zeta}}\label{subsec:Phi_m-fischer}
上一小节把 $Y_m$ 写成一个显式标记图 $G_m$ 的像，从而给出“$\Phi_m$ 的像空间是 sofic”的可核验表示。为了把该表示提升为 ETDS 语境下常用的可计算不变量（邻接矩阵、熵、最大熵测度、以及 \(\zeta\) 的行列式接口），需要引入 sofic shift 的标准 Markov 覆盖（cover）理论。

\subsubsection{右解析（right-resolving）标记图：从 \texorpdfstring{$G_m$}{Gm} 到 Fischer cover 的可计算构造}
本段把“$Y_m$ 是 sofic”进一步提升为一个\textbf{右解析}（right-resolving）的标记图表示，从而得到一条可直接导出邻接关系、熵与 Parry（最大熵）测度的计算链条。

\begin{definition}[从 \texorpdfstring{$G_m$}{Gm} 的子集构造得到右解析标记图]\label{def:Phi_m-right-resolving}
记 $G_m=(V_m,E_m,\ell_m)$ 为定义 \ref{def:G_m} 的有限标记图（字母表为 $X_m$）。定义一个确定性右解析标记图
\[
H_m^{\mathrm{det}}=(\mathcal{S}_m,\mathcal{E}_m,\lambda_m),
\]
其顶点集为若干非空子集 $S\subset V_m$，边按如下\textbf{转移函数}给出：对任意 $S\subset V_m$ 与任意输出符号 $a\in X_m$，令
\[
\delta_m(S,a):=\Bigl\{v'\in V_m:\ \exists\, e=(v,b)\in E_m\ \text{s.t.}\ v\in S,\ \ell_m(e)=a,\ \text{且 } e:v\to v'\Bigr\}.
\]
若 $\delta_m(S,a)\neq\varnothing$，则在 $H_m^{\mathrm{det}}$ 中加入一条标记为 $a$ 的边
\[
S \xrightarrow{\ a\ } \delta_m(S,a).
\]
并令 $\mathcal{S}_m$ 为从初始集合 $V_m$ 出发，在上述转移下可达的全部状态集合（从而 $\mathcal{S}_m$ 有限且可算法生成）。
\end{definition}

\begin{proposition}[右解析性与 sofic 表示]\label{prop:Phi_m-right-resolving}
由定义 \ref{def:Phi_m-right-resolving} 构造的标记图 $H_m^{\mathrm{det}}$ 满足：
\begin{enumerate}
  \item \textbf{（右解析）}对任意顶点 $S\in\mathcal{S}_m$，从 $S$ 出发的两条不同出边具有不同标记；
  \item \textbf{（同一像空间）}$H_m^{\mathrm{det}}$ 所呈现的标记移位等于 $Y_m$：其双边标记路径的标记序列恰为 $Y_m$。
\end{enumerate}
\end{proposition}

\begin{proof}
\textbf{(1)} 由构造可知：对固定 $S$ 与固定标记 $a$，若 $\delta_m(S,a)\neq\varnothing$ 则对应的出边终点被定义为唯一集合 $\delta_m(S,a)$，故同一 $S$ 不可能有两条出边共享同一标记 $a$。

\textbf{(2)} 从 $G_m$ 到 $H_m^{\mathrm{det}}$ 的子集构造是正则语言/有限自动机理论的标准确定化过程：$H_m^{\mathrm{det}}$ 识别的正是 $G_m$ 的标记语言，因此两者给出相同的标记移位。结合定理 \ref{thm:Phi_m-sofic-graph} 的 $Y_m=\mathcal{L}_m(Z_m)$ 即得结论。关于该构造在符号动力系统中的标准表述，可参见 \cite{LindMarcus1995,Kitchens1998SymbolicDynamics} 中对 sofic shift 的右解析表示与 Fischer cover 的讨论。
\end{proof}

\begin{remark}[右 Fischer cover（最小右解析表示）]\label{rem:fischer-cover}
进一步将 $H_m^{\mathrm{det}}$ 做\textbf{状态最小化}（合并具有相同 follower 集的状态），即可得到 $Y_m$ 的右 Fischer cover：一个右解析且 follower-separated 的标记图表示，其同构类由 $Y_m$ 唯一确定（\cite{LindMarcus1995,Kitchens1998SymbolicDynamics}）。下文关于熵与 Parry 测度的计算在任何这样的右解析表示上均可执行；采用 Fischer cover 的优势在于它给出\textbf{最小}的可计算状态空间。
\end{remark}

\begin{definition}[follower 集与右 Fischer cover 的显式商图]\label{def:fischer-quotient}
在右解析标记图 $H_m^{\mathrm{det}}=(\mathcal{S}_m,\mathcal{E}_m,\lambda_m)$ 上，对任意状态 $S\in\mathcal{S}_m$ 定义其 follower 集为
\[
\mathsf{F}(S):=\Bigl\{a_0a_1a_2\cdots\in X_m^{\mathbb{N}}:\ \exists\ \text{从 }S\text{ 出发的右无限路径 }e_0e_1e_2\cdots\ \text{s.t.}\ \lambda_m(e_k)=a_k\Bigr\}.
\]
定义等价关系 $S\sim S'$ 当且仅当 $\mathsf{F}(S)=\mathsf{F}(S')$。令 $\mathcal{S}_m^{\mathrm{Fis}}:=\mathcal{S}_m/\!\sim$ 为等价类集合。

定义商标记图
\[
H_m^{\mathrm{Fis}}=(\mathcal{S}_m^{\mathrm{Fis}},\mathcal{E}_m^{\mathrm{Fis}},\lambda_m^{\mathrm{Fis}})
\]
如下：若 $H_m^{\mathrm{det}}$ 中有边 $S\xrightarrow{\,a\,}T$，则在 $H_m^{\mathrm{Fis}}$ 中加入边
\[
[S]\xrightarrow{\,a\,}[T],
\]
其中 $[S]$ 表示 $S$ 的等价类。并取 $\lambda_m^{\mathrm{Fis}}$ 为该边的标记 $a$。
\end{definition}

\begin{proposition}[Fischer cover 的右分辨与最小性]\label{prop:fischer-minimal}
商图 $H_m^{\mathrm{Fis}}$ 满足：
\begin{enumerate}
  \item \textbf{（右分辨）}$H_m^{\mathrm{Fis}}$ 是右解析（right-resolving）的；
  \item \textbf{（follower-separated）}不同状态具有不同 follower 集；
  \item \textbf{（同一像空间）}$H_m^{\mathrm{Fis}}$ 呈现的标记移位等于 $Y_m$；
  \item \textbf{（最小性）}在所有呈现 $Y_m$ 的右解析标记图中，$H_m^{\mathrm{Fis}}$ 的状态数最小；更具体地，对任意右解析呈现 $G$（其标记移位为 $Y_m$），存在一个满射的标记图态射 $G\twoheadrightarrow H_m^{\mathrm{Fis}}$。
\end{enumerate}
\end{proposition}

\begin{proof}
\textbf{(1)} 由命题 \ref{prop:Phi_m-right-resolving} 知 $H_m^{\mathrm{det}}$ 右解析：同一状态 $S$ 的不同出边标记不同。商映射只会合并状态而不会改变“从同一状态出发、按标记 $a$ 的后继状态唯一”的确定性，因此 $H_m^{\mathrm{Fis}}$ 仍右解析。

\textbf{(2)} 由构造，$H_m^{\mathrm{Fis}}$ 的状态即为 follower 集等价类，因此天然 follower-separated。

\textbf{(3)} 合并 follower 相同的状态不会改变可实现的右无限标记集合，从而不改变双边标记语言；因此商图与原图呈现同一 sofic 移位。结合命题 \ref{prop:Phi_m-right-resolving} 的“$H_m^{\mathrm{det}}$ 呈现 $Y_m$”即得。

\textbf{(4)} 设 $G$ 为任意右解析呈现 $Y_m$ 的标记图。对每个状态 $u\in G$，定义其 follower 集 $\mathsf{F}_G(u)$（定义同上）。因为 $G$ 与 $H_m^{\mathrm{det}}$ 呈现同一标记移位 $Y_m$，每个 $u$ 的 follower 集必等于某个 $S\in\mathcal{S}_m$ 的 follower 集；并且右解析性保证从 $u$ 出发按标记推进得到的 follower 集在商图上有唯一对应的后继，因此映射
\[
u\ \longmapsto\ [S]\in\mathcal{S}_m^{\mathrm{Fis}}\quad\text{其中 }\mathsf{F}_G(u)=\mathsf{F}(S)
\]
可提升为一个标记图态射 $G\to H_m^{\mathrm{Fis}}$，且对边/标记相容。该态射对可达状态是满射（否则 $H_m^{\mathrm{Fis}}$ 存在一个 follower 类在 $Y_m$ 中出现却不被 $G$ 覆盖，与“呈现同一 $Y_m$”矛盾）。因此 $H_m^{\mathrm{Fis}}$ 在右解析呈现类中最小。更系统的标准证明与等价表述可参见 \cite{LindMarcus1995,Kitchens1998SymbolicDynamics} 关于 Fischer cover 的章节。
\end{proof}

\subsubsection{邻接关系、拓扑熵与 Parry（最大熵）测度}
取一个右解析且 follower-separated 的表示（例如上面的 Fischer cover），记为
\[
H_m=(\mathcal{S}_m,\mathcal{E}_m,\lambda_m),
\]
其中每条边 $e\in\mathcal{E}_m$ 具有起点 $s(e)\in\mathcal{S}_m$、终点 $t(e)\in\mathcal{S}_m$ 与标记 $\lambda_m(e)\in X_m$。

\begin{definition}[覆盖 SFT 的邻接矩阵（边移位）]\label{def:Phi_m-adjacency}
令 $X_{A_m}$ 为图 $H_m$ 的\textbf{边移位}（把边当作字母的 SFT）。其邻接矩阵 $A_m$ 为 $0$--$1$ 矩阵，索引集为 $\mathcal{E}_m$，并由
\[
(A_m)_{e,e'}:=\mathbf{1}\{\,t(e)=s(e')\,\}
\]
给出。这一邻接关系是完全显式的：它只依赖于 $H_m$ 的端点相容性。
\end{definition}

\begin{proposition}[拓扑熵的可计算公式]\label{prop:Phi_m-entropy}
设 $H_m$ 不可约（或取其最大熵不可约分量），则
\[
h_{\mathrm{top}}(Y_m)=h_{\mathrm{top}}(X_{A_m})=\log\rho(A_m).
\]
\end{proposition}

\begin{proof}
$X_{A_m}$ 通过逐边取标记映射到 $Y_m$（一块码），当 $H_m$ 为 Fischer cover 时该标记映射是有限对一的因子码，因此熵保持：$h_{\mathrm{top}}(Y_m)=h_{\mathrm{top}}(X_{A_m})$。而对边移位（SFT）有经典公式 $h_{\mathrm{top}}(X_{A_m})=\log\rho(A_m)$（\cite{LindMarcus1995,Kitchens1998SymbolicDynamics}）。
\end{proof}

\begin{proposition}[Parry 最大熵测度（覆盖 \texorpdfstring{$X_{A_m}$}{XAm} 上）]\label{prop:Phi_m-parry}
令 $\lambda=\rho(A_m)$，取 Perron--Frobenius 左/右特征向量 $l,r>0$ 满足
\(
l^{\mathsf T}A_m=\lambda l^{\mathsf T},\ A_m r=\lambda r
\)
并规范化为 $l^{\mathsf T}r=1$。定义状态权重
\[
\pi(e):=l_e r_e\qquad (e\in\mathcal{E}_m),
\]
以及转移概率
\[
P_{e\to e'}:=\frac{(A_m)_{e,e'}\,r_{e'}}{\lambda\, r_e}.
\]
则由 $(\pi,P)$ 给出的平稳马尔可夫测度是 $X_{A_m}$ 上唯一的最大熵测度（Parry 测度），记为 $\mu_{A_m}^{\mathrm{Parry}}$。
其对边柱集 $[e_0e_1\cdots e_{n-1}]$ 的概率满足
\[
\mu_{A_m}^{\mathrm{Parry}}([e_0\cdots e_{n-1}])=\pi(e_0)\prod_{j=0}^{n-2} P_{e_j\to e_{j+1}}.
\]
\end{proposition}

\begin{proposition}[把 Parry 测度推前到 \texorpdfstring{$Y_m$}{Ym}]\label{prop:Phi_m-parry-pushforward}
令 $\pi_m:X_{A_m}\to Y_m$ 为逐边取标记的因子映射
\[
(\pi_m((e_t)_{t\in\ZZ}))_t:=\lambda_m(e_t)\in X_m.
\]
则 $\nu_m:=({\pi_m})_*\mu_{A_m}^{\mathrm{Parry}}$ 是 $Y_m$ 的最大熵测度（当 $Y_m$ 不可约时唯一）。对任意输出柱集 $[a_0\cdots a_{n-1}]$（$a_j\in X_m$），有完全显式的求和公式
\[
\nu_m([a_0\cdots a_{n-1}])
=
\sum_{\substack{e_0,\dots,e_{n-1}\in\mathcal{E}_m\\ \lambda_m(e_j)=a_j}}
\mu_{A_m}^{\mathrm{Parry}}([e_0\cdots e_{n-1}]),
\]
其中 $\mu_{A_m}^{\mathrm{Parry}}([e_0\cdots e_{n-1}])$ 由命题 \ref{prop:Phi_m-parry} 的 PF 数据给出。
\end{proposition}

\begin{proof}
Fischer cover 的 Parry 测度推前为 sofic shift 的最大熵测度是标准结论；上述柱集公式仅是“推前测度 = 对纤维求和”的展开。参见 \cite{LindMarcus1995,Kitchens1998SymbolicDynamics} 中关于 sofic shift 最大熵测度与 Fischer cover 的章节。
\end{proof}

\subsubsection{滑动块因子 \texorpdfstring{$Y_m$}{Ym} 的谱/\texorpdfstring{$\zeta$}{zeta}：cover 与内禀对象的精确分离}\label{subsec:Ym-zeta-spectrum}
上一段已解释“为何本文主要使用 cover 的 \(\zeta\)”。为回应“折叠诱导滑动块因子的谱/\(\zeta\) 计算应明确到可执行层”的审稿口径，本小节把两类对象的计算链条明确列出，并给出一个对 \(Y_m\) 内禀 \(\zeta_{Y_m}\) 的\textbf{可计算}口径（不强行把它误写成 \(\det(I-zA_m)\)）。

\paragraph{(A) cover-SFT 的谱与 \texorpdfstring{$\zeta$}{zeta}（本文主接口）}
取 \(Y_m\) 的右 Fischer cover（或任一 follower-separated 右解析 cover）\(H_m\)，按定义 \ref{def:Phi_m-adjacency} 得到边移位 \(X_{A_m}\) 的邻接矩阵 \(A_m\)。则：
\begin{itemize}
  \item \textbf{谱/熵：} \(h_{\mathrm{top}}(Y_m)=\log\rho(A_m)\)（命题 \ref{prop:Phi_m-entropy}）；
  \item \textbf{最大熵测度：} Parry 测度由 \(A_m\) 的 Perron--Frobenius 左/右本征向量显式给出（命题 \ref{prop:Phi_m-parry}），并可推前得到 \(Y_m\) 的最大熵测度（命题 \ref{prop:Phi_m-parry-pushforward}）；
  \item \textbf{\(\zeta\)：} 覆盖 SFT 的 Artin--Mazur \(\zeta\) 满足
  \[
  \zeta_{X_{A_m}}(z)=\frac{1}{\det(I-zA_m)},
  \qquad
  \#\mathrm{Fix}(\sigma^n|_{X_{A_m}})=\mathrm{Tr}(A_m^n),
  \]
  并因此
  \(
  \#\mathrm{Fix}(\sigma^n|_{X_{A_m}})=\rho(A_m)^n+O(\rho_2^n)
  \)
  （\(\rho_2<\rho(A_m)\) 为次大谱半径，取不可约非周期分量时成立）。
\end{itemize}
这套链条之所以“可审计”，在于一切输入仅是有限图/矩阵：所有数值不变量（熵、主特征值、留数常数、primitive 轨道谱等）都可由线性代数直接复现。

\paragraph{(B) sofic 因子 \texorpdfstring{$Y_m$}{Ym} 的内禀 \texorpdfstring{$\zeta_{Y_m}$}{zetaYm}（可计算但不等于 cover 行列式）}
内禀 \(\zeta_{Y_m}\) 由周期点计数定义：
\[
\zeta_{Y_m}(z):=\exp\left(\sum_{n\ge 1}\frac{\#\mathrm{Fix}(\sigma^n|_{Y_m})}{n}z^n\right).
\]
由于 \(Y_m\) 是 sofic shift（有限标记图像；见 \cite{Weiss1973SFTandSofic,CovenPaul1975SoficSystems,LindMarcus1995,Kitchens1998SymbolicDynamics}），其周期点集是正则对象，因此 \(\zeta_{Y_m}\) 为有理函数；这类“有理性”结论在动力系统语境中可追溯到 Manning 关于 \(\zeta\) 有理性的经典结果（\cite{Manning1971RationalZeta}，并可在 sofic 系统框架下由上述标准文本吸收）。关键是：\(\#\mathrm{Fix}(\sigma^n|_{Y_m})\) 计数的是\textbf{标记序列}的周期点，而 \(\mathrm{Tr}(A_m^n)\) 计数的是 cover 上\textbf{边序列}的周期点；当标记因子不是一一对应时，这两者不同。

\paragraph{一个可执行的计数口径（基于右解析表示的“转移半群自动机”）}
令 \(H_m^{\mathrm{det}}\) 为定义 \ref{def:Phi_m-right-resolving} 的右解析确定化图（或直接取右 Fischer cover；二者同口径）。对每个输出符号 \(a\in X_m\)，由右解析性可定义一个\emph{部分转移函数}
\[
\tau_a:\mathcal{S}_m\to\mathcal{S}_m,\qquad
\tau_a(S)=T\ \Longleftrightarrow\ \text{在 }H_m^{\mathrm{det}}\text{ 中存在边 }S\xrightarrow{\,a\,}T,
\]
并在无此边时令 \(\tau_a\) 在该点未定义。对长度 \(n\) 的词 \(w=a_0a_1\cdots a_{n-1}\in X_m^n\)，记复合部分映射
\[
\tau_w:=\tau_{a_{n-1}}\circ\cdots\circ\tau_{a_1}\circ\tau_{a_0}.
\]
则 \(w\) 能产生一个 \(n\)-周期点（即其周期延拓 \(w^\infty\in Y_m\)）当且仅当存在某状态 \(S\in\mathcal{S}_m\) 使得 \(\tau_w(S)\) 定义且 \(\tau_w(S)=S\)（这等价于在标记图中存在一条标记为 \(w\) 的闭路径）。

注意到：\(\{\tau_w: w\in X_m^*\}\) 作为\textbf{有限集合上的部分映射半群}是有限的（至多 \( (|\mathcal{S}_m|+1)^{|\mathcal{S}_m|}\) 个），于是可构造一个有限自动机：
\begin{itemize}
  \item \textbf{状态：}所有可达的部分映射 \(\tau_w\)（从恒等映射出发，逐字母右乘）；
  \item \textbf{输入字母：} \(a\in X_m\)；
  \item \textbf{转移：} \(\tau \xrightarrow{a} \tau_a\circ\tau\)；
  \item \textbf{接受条件：} \(\tau\) 具有至少一个不动点（存在 \(S\) 使 \(\tau(S)=S\) 且定义）。
\end{itemize}
令该自动机的状态集合为 \(\mathcal{T}_m\)，其 \(0\)–\(1\) 邻接矩阵为 \(B_m\)。则
\[
\#\mathrm{Fix}(\sigma^n|_{Y_m})=\mathrm{Tr}(B_m^n),
\qquad
\zeta_{Y_m}(z)=\frac{1}{\det(I-zB_m)}.
\]
这一口径把“\(Y_m\) 的内禀 \(\zeta\)”也还原为一个行列式对象，但输入矩阵是\textbf{转移半群自动机}的邻接矩阵 \(B_m\)，而非 cover 的 \(A_m\)。它同时解释了两点：
\begin{itemize}
  \item \textbf{为何 \(\zeta_{Y_m}\) 有理：} \(\mathrm{Tr}(B_m^n)\) 满足线性递推；
  \item \textbf{为何一般不等于 cover 行列式：} 因子映射把多个 cover 周期轨道折叠为同一标记周期点，计数发生不可逆压缩。
\end{itemize}
在本文的数值与证明链中，我们把 \(\det(I-zA_m)\) 用作“语法/cover 层可计算接口”；若需要 \(Y_m\) 内禀 \(\zeta\) 的显式有理函数，可按上述自动机构造逐 \(m\) 计算（参见 \cite{Nasu1995TextileSystems} 对 sofic 系统以自动机/cover 表达周期点与 \(\zeta\) 的标准处理）。

\begin{proposition}[内禀 \texorpdfstring{$\zeta_{Y_m}$}{zetaYm} 的主极点与熵：与 cover 同一指数尺度]\label{prop:Ym-zeta-leading-pole}
设 \(Y_m\) 不可约（或取其最大熵本质分量）。则 \(Y_m\) 的周期点增长率满足
\[
\limsup_{n\to\infty}\frac{1}{n}\log\#\mathrm{Fix}(\sigma^n|_{Y_m})
\ =\ h_{\mathrm{top}}(Y_m),
\]
从而内禀 \(\zeta_{Y_m}(z)\) 的收敛半径为
\[
R_{Y_m}=e^{-h_{\mathrm{top}}(Y_m)}.
\]
另一方面，cover 接口给出 \(h_{\mathrm{top}}(Y_m)=\log\rho(A_m)\)（命题 \ref{prop:Phi_m-entropy}），因此
\[
R_{Y_m}=\rho(A_m)^{-1}.
\]
换言之：尽管 \(\zeta_{Y_m}\) 与 \(\zeta_{X_{A_m}}=1/\det(I-zA_m)\) 一般不相等，但二者在主极点模长（即指数尺度）上由同一熵决定。
\end{proposition}

\begin{proof}
对任意 sofic shift，周期点集是正则对象，且周期点计数的指数增长率与拓扑熵一致；更具体地，可由一个有限标记图/右解析表示把“长度 \(n\) 的可允许周期块”与“长度 \(n\) 的闭路径”联系起来，从而得到
\(
\limsup_{n\to\infty}\frac{1}{n}\log\#\mathrm{Fix}(\sigma^n)=h_{\mathrm{top}}
\)
（参见 \cite{Weiss1973SFTandSofic,CovenPaul1975SoficSystems,LindMarcus1995,Kitchens1998SymbolicDynamics} 中关于 sofic shift 的语言增长/熵与周期点的标准章节）。

在本文的具体构造中，我们已得到 \(\#\mathrm{Fix}(\sigma^n|_{Y_m})=\mathrm{Tr}(B_m^n)\)。对有限矩阵 \(B_m\)，Perron--Frobenius 理论给出
\(
\limsup_{n\to\infty}\frac{1}{n}\log\mathrm{Tr}(B_m^n)=\log\rho(B_m)
\)；
因此 \(\log\rho(B_m)=h_{\mathrm{top}}(Y_m)\)，并由 \(\zeta_{Y_m}=1/\det(I-zB_m)\) 的谱半径控制得到收敛半径 \(R_{Y_m}=\rho(B_m)^{-1}=e^{-h_{\mathrm{top}}(Y_m)}\)。
最后，命题 \ref{prop:Phi_m-entropy} 已给出 \(h_{\mathrm{top}}(Y_m)=\log\rho(A_m)\)，于是 \(R_{Y_m}=\rho(A_m)^{-1}\)。
\end{proof}

\begin{proposition}[周期点计数的显式夹逼：cover \texorpdfstring{$\Rightarrow$}{=>} sofic]\label{prop:periodic-count-sandwich}
取 \(Y_m\) 的一个右解析标记图表示 \(H_m=(\mathcal{S}_m,\mathcal{E}_m,\lambda_m)\)（例如定义 \ref{def:Phi_m-right-resolving} 的 \(H_m^{\mathrm{det}}\)，或进一步最小化得到的 Fischer cover）。令 \(X_{A_m}\) 为其边移位（定义 \ref{def:Phi_m-adjacency}），并令 \(\pi_m:X_{A_m}\to Y_m\) 为逐边取标记的因子映射（定义同命题 \ref{prop:Phi_m-parry-pushforward}）。则对任意 \(n\ge 1\) 有
\[
\#\mathrm{Fix}(\sigma^n|_{Y_m})
\ \le\
\#\mathrm{Fix}(\sigma^n|_{X_{A_m}})
\ =\
\mathrm{Tr}(A_m^n),
\]
并且还存在完全显式的下界
\[
\#\mathrm{Fix}(\sigma^n|_{Y_m})
\ \ge\
\frac{1}{|\mathcal{S}_m|}\,\#\mathrm{Fix}(\sigma^n|_{X_{A_m}})
\ =\
\frac{1}{|\mathcal{S}_m|}\,\mathrm{Tr}(A_m^n).
\]
因此 \(\#\mathrm{Fix}(\sigma^n|_{Y_m})\) 与 \(\mathrm{Tr}(A_m^n)\) 在指数尺度上一致（差异至多为常数因子），并与命题 \ref{prop:Ym-zeta-leading-pole} 的“同一主极点模长”结论一致。
\end{proposition}

\begin{proof}
上界由周期点对因子映射的推前显然成立：若 \(y\in\mathrm{Fix}(\sigma^n|_{Y_m})\)，则任选其一个前像 \(x\in X_{A_m}\) 必满足 \(\sigma^n x=x\)，从而
\(
\#\mathrm{Fix}(Y_m)\le \#\mathrm{Fix}(X_{A_m})
\)。

下界使用右解析性给出对前像多重度的显式控制。对任意 \(y\in Y_m\)，其在 \(H_m\) 上的 lift 等价于选择一个起始状态 \(S\in\mathcal{S}_m\)，然后按标记序列逐步推进得到唯一的边路径（右解析保证“给定状态 + 给定标记”时出边唯一）。因此任意 \(y\) 的前像数满足
\[
|\pi_m^{-1}(y)|\le |\mathcal{S}_m|.
\]
将该不等式限制到 \(n\)-周期点集合并求和：
\[
\#\mathrm{Fix}(\sigma^n|_{X_{A_m}})
=\sum_{y\in \mathrm{Fix}(\sigma^n|_{Y_m})} |\pi_m^{-1}(y)|
\le |\mathcal{S}_m|\,\#\mathrm{Fix}(\sigma^n|_{Y_m}),
\]
整理即得下界。最后等式 \(\#\mathrm{Fix}(\sigma^n|_{X_{A_m}})=\mathrm{Tr}(A_m^n)\) 是边移位的标准计数恒等式（参见 \cite{LindMarcus1995}）；而 sofic 系统可实现为同熵的有限对一 SFT 因子这一结构性事实可参见 \cite{CovenPaul1975SoficSystems}。
\end{proof}


\subsection{推广：一般 Ostrowski 计数下的稳定折叠}\label{subsec:ostrowski-generalized-fold}

本节把“Fibonacci/Zeckendorf 折叠”推广到一般无理数的 Ostrowski 计数系统，以回应“扩展到更广泛计数框架”的审稿要求。黄金分支的特殊性在于：当 $\alpha=\varphi^{-1}=[0;1,1,1,\dots]$ 时，Ostrowski 计数退化为 Fibonacci 权重与 Zeckendorf 语法，因此稳定类型空间成为最简单的二元 SFT（禁止相邻 $11$），并使 PF/$\zeta$ 接口闭式可得。

\subsubsection{一般 Ostrowski 权重与合法数字}

设 $\alpha\in(0,1)\setminus\QQ$ 的连分数展开为
\[
\alpha=[0;a_1,a_2,\dots],\qquad a_k\in\NN.
\]
令其收敛分母 $(q_n)_{n\ge 0}$ 由递推定义
\[
q_{-1}=0,\ q_0=1,\qquad q_{n+1}=a_{n+1}q_n+q_{n-1}\quad(n\ge 0).
\]

\begin{theorem}[Ostrowski 表示（标准）]\label{thm:ostrowski-representation}
对任意 $N\in\NN$，存在唯一数字序列 $(d_n)_{n\ge 1}$ 满足：
\begin{enumerate}
  \item $0\le d_n\le a_n$；
  \item 若 $d_n=a_n$，则 $d_{n-1}=0$（对 $n\ge 2$）；
  \item 仅有限个 $d_n\neq 0$；
  \item 数值分解为
  \[
  N=\sum_{n\ge 1} d_n q_{n-1}.
  \]
\end{enumerate}
称该唯一序列为 $N$ 的 Ostrowski 数字，记为 $Z_\alpha(N)=(d_1,d_2,\dots)$（参见 \cite{Ostrowski1922} 及后续综述中的标准陈述）。
\end{theorem}

\begin{definition}[一般稳定类型空间]\label{def:Xm-alpha}
对任意 $m\ge 1$，定义长度-$m$ 的 Ostrowski 合法集合
\[
X_m^{(\alpha)}:=\Bigl\{(d_1,\dots,d_m):\ 0\le d_n\le a_n,\ \ d_n=a_n\Rightarrow d_{n-1}=0\ (n\ge 2)\Bigr\}.
\]
并定义截断 $\pi_m^{(\alpha)}(d_1,d_2,\dots)=(d_1,\dots,d_m)$。
\end{definition}

\begin{proposition}[计数与递推（分母即计数）]\label{prop:Xm-alpha-cardinality}
对任意 $m\ge 1$，有
\[
\bigl|X_m^{(\alpha)}\bigr|=q_{m+1}.
\]
特别地，当 $\alpha=\varphi^{-1}$（即 $a_n\equiv 1$）时，$q_{m+1}=F_{m+2}$，而 $X_m^{(\alpha)}$ 退化为二元集合 $\{0,1\}^m$ 中禁止相邻 $11$ 的黄金均值语法 $X_m$。
\end{proposition}

\subsubsection{一般折叠算子 \texorpdfstring{$\Fold_m^{(\alpha)}$}{Fold\_m(alpha)}}

为了与本文的“折叠 = 取值 + 取规范形”机制保持一致，我们把一般 Ostrowski 折叠定义为：先将“未规范数字”（允许违反相邻最大值约束）赋一个可审计整数值，再取 Ostrowski 规范形并截断。

\begin{definition}[Ostrowski 加权值与折叠]\label{def:fold-alpha}
令
\[
\Omega_m^{(\alpha)}:=\prod_{n=1}^{m}\{0,1,\dots,a_n\}
\]
为长度-$m$ 的“原始数字空间”。对 $\omega=(\omega_1,\dots,\omega_m)\in\Omega_m^{(\alpha)}$ 定义其 Ostrowski 加权值
\[
N_\alpha(\omega):=\sum_{n=1}^{m}\omega_n q_{n-1}\in\NN.
\]
定义一般折叠
\[
\Fold_m^{(\alpha)}:\Omega_m^{(\alpha)}\to X_m^{(\alpha)},\qquad
\Fold_m^{(\alpha)}(\omega):=\pi_m^{(\alpha)}\!\bigl(Z_\alpha(N_\alpha(\omega))\bigr).
\]
\end{definition}

\begin{proposition}[一般折叠的基本性质]\label{prop:fold-alpha-basic}
$\Fold_m^{(\alpha)}$ 良定义且满足：
\begin{enumerate}
  \item \textbf{（规范性）} $\Fold_m^{(\alpha)}(\omega)\in X_m^{(\alpha)}$；
  \item \textbf{（幂等性）} 若 $\omega\in X_m^{(\alpha)}$，则 $\Fold_m^{(\alpha)}(\omega)=\omega$；
  \item \textbf{（满射性）} $\Fold_m^{(\alpha)}:\Omega_m^{(\alpha)}\twoheadrightarrow X_m^{(\alpha)}$ 为满射；
  \item \textbf{（多尺度相容）} 对 $m_2\ge m_1$，有
  \[
  \pi_{m_2\to m_1}^{(\alpha)}\circ \Fold_{m_2}^{(\alpha)}=\Fold_{m_1}^{(\alpha)}\circ \pi_{m_2\to m_1}^{(\alpha)},
  \]
  其中 $\pi_{m_2\to m_1}^{(\alpha)}$ 为截断到前 $m_1$ 位的投影。
\end{enumerate}
\end{proposition}

\begin{proof}
第 1 条由 Ostrowski 表示的合法性立得。第 2 条：若 $\omega$ 已合法，则 $\omega$ 即为某个整数的规范 Ostrowski 数字的前 $m$ 位，唯一性迫使规范化后截断不变。第 3 条：对任意 $x\in X_m^{(\alpha)}$，取 $\omega=x$ 即有 $\Fold_m^{(\alpha)}(\omega)=x$。第 4 条由截断与“先取值再取规范形”的函子性得到：对任意 $\omega$，两边都等于 $\pi_{m_1}^{(\alpha)}(Z_\alpha(N_\alpha(\omega)))$。
\end{proof}

\begin{remark}[与本文二元读出的关系]
本文的主线实验固定二元读出与黄金语法（便于得到闭式 PF/Parry/$\zeta$ 基线）。上述一般化表明：若允许更一般的计数/地址字母表（或更一般的观测通道），则“折叠 = 取值 + Ostrowski 规范形”的机制可原则化推广；黄金分支只是其中最简、最可闭式分析的特例。
\end{remark}
