Joukowsky 椭圆族在本文中承担“双尺度外置”的几何通道：一方面它由尺度参数 \(r>1\) 决定并以
\(
E_r=\{\,w=r z+r^{-1}z^{-1}:|z|=1\,\}
\)
给出（见定理 \ref{thm:group-jg-elliptic-two-scale-stokes-mean} 的记号）；
另一方面，当 \(r\) 取为素数寄存器范数的平方根时，它把 Gödel 位长与 Lorentz 合成、容量/环模数以及高维体积在同一对数坐标下统一。

\begin{theorem}[Gödel--Lorentz 代数化：Joukowsky 椭圆的 Minkowski 不变量与合成律]\label{thm:group-jg-godel-lorentz-algebraization}
沿用定理 \ref{thm:group-jg-ellipse-prime-homomorphism} 的素数寄存器 \(\mathsf P=\NN^{(\mathbb P)}\) 与范数 \(N:\mathsf P\to\NN_{>0}\)。
对 \(r\in\mathsf P\) 且 \(N(r)>1\)，令
\[
L:=N(r),\qquad r_{\mathrm{geo}}:=\sqrt{L}>1,\qquad
\Delta(r):=\log r_{\mathrm{geo}}=\frac12\log N(r).
\]
记 Joukowsky 椭圆
\[
E_J(r):=\{\,w=r_{\mathrm{geo}} z+r_{\mathrm{geo}}^{-1}z^{-1}:|z|=1\,\}\subset\CC,
\]
并记其长短半轴
\[
a(r):=r_{\mathrm{geo}}+r_{\mathrm{geo}}^{-1},\qquad b(r):=r_{\mathrm{geo}}-r_{\mathrm{geo}}^{-1}.
\]
定义
\[
\Lambda(r):=
\begin{pmatrix}
\frac{a(r)}{2} & \frac{b(r)}{2}\\[2pt]
\frac{b(r)}{2} & \frac{a(r)}{2}
\end{pmatrix}\in\GL_2(\RR).
\]
则：
\begin{enumerate}
  \item \(a(r)^2-b(r)^2=4\)，且 \(\det\Lambda(r)=1\)。因此 \((a(r),b(r))\) 落在双曲线 \(x^2-y^2=4\) 上，等价地，\(\{E_J(r)\}\) 为一族共焦椭圆（焦距恒为 \(2\)）。
  \item 令 \(\Delta=\Delta(r)\)，则
  \[
  a(r)=2\cosh\Delta,\qquad b(r)=2\sinh\Delta,
  \]
  从而 \(\Lambda(r)\in\mathrm{SO}^+(1,1)\subset \mathrm{SL}_2(\RR)\)，并且 \(\Lambda(r)\) 的特征值为 \(e^{\pm\Delta}\)。
  \item 映射 \(\Lambda:\mathsf P\to \mathrm{SO}^+(1,1)\) 为交换幺半群同态：
  \[
  \Lambda(r_1+r_2)=\Lambda(r_1)\Lambda(r_2).
  \]
  等价地，若 \((a_i,b_i)=(a(r_i),b(r_i))\)，则
  \[
  a(r_1+r_2)=\frac{a_1a_2+b_1b_2}{2},\qquad
  b(r_1+r_2)=\frac{a_1b_2+b_1a_2}{2}.
  \]
  \item 谱半径满足
  \[
  \log \rho\bigl(\Lambda(r)\bigr)=\Delta(r)=\frac12\log N(r).
  \]
\end{enumerate}
\end{theorem}

\begin{proof}
由 \(a=r_{\mathrm{geo}}+r_{\mathrm{geo}}^{-1}\)、\(b=r_{\mathrm{geo}}-r_{\mathrm{geo}}^{-1}\) 直接计算得
\(
a^2-b^2=4
\)；
并令 \(\Delta=\log r_{\mathrm{geo}}\) 得 \(r_{\mathrm{geo}}=e^{\Delta}\)，故
\(
a=2\cosh\Delta,\ b=2\sinh\Delta
\)。
于是
\(
\det\Lambda=\cosh^2\Delta-\sinh^2\Delta=1
\)，且 \(\Lambda\) 的特征值为 \(\cosh\Delta\pm\sinh\Delta=e^{\pm\Delta}\)。
\par
由 \(N(r_1+r_2)=N(r_1)N(r_2)\) 得 \(\Delta(r_1+r_2)=\Delta(r_1)+\Delta(r_2)\)，故
\[
\Lambda(r_1+r_2)
=
\begin{pmatrix}\cosh(\Delta_1+\Delta_2)&\sinh(\Delta_1+\Delta_2)\\ \sinh(\Delta_1+\Delta_2)&\cosh(\Delta_1+\Delta_2)\end{pmatrix}
=\Lambda(r_1)\Lambda(r_2),
\]
并由双曲函数加法公式得到半轴的双线性合成式。
最后 \(\rho(\Lambda)=e^{\Delta}\) 给出谱半径恒等式。证毕。
\end{proof}

\begin{corollary}[Gödel 范数与 \texorpdfstring{$2$}{2}-范数条件数恒等式]\label{cor:group-jg-godel-condition-number-identity}
在定理 \ref{thm:group-jg-godel-lorentz-algebraization} 的记号下，矩阵 \(\Lambda(r)\) 的 \(2\)-范数条件数满足
\[
\kappa_2\bigl(\Lambda(r)\bigr)=N(r).
\]
等价地，
\(
\log \kappa_2(\Lambda(r))=\log N(r)=2\Delta(r)
\)。
\end{corollary}
\begin{proof}
由定理 \ref{thm:group-jg-godel-lorentz-algebraization}，\(\Lambda(r)\) 为对称矩阵且特征值为 \(e^{\pm\Delta(r)}\)。
因此其奇异值等于特征值的绝对值，得到
\(
\kappa_2(\Lambda)=\frac{e^{\Delta}}{e^{-\Delta}}=e^{2\Delta}=N(r)
\)。
证毕。
\end{proof}

\begin{theorem}[面积守恒规范化的 Cayley 共轭与 Einstein 加法]\label{thm:group-jg-joukowsky-area-preserving-cayley}
沿用定理 \ref{thm:group-jg-godel-lorentz-algebraization} 的记号，并记
\[
E(a):=\{(x,y)\in\RR^2:\ x^2/a^2+y^2a^2=1\}\qquad(a>0)
\]
为面积守恒椭圆族（定理 \ref{thm:group-jg-ellipse-prime-homomorphism}）。
对 \(r\in\mathsf P\) 定义面积守恒规范化
\[
\widehat E_J(r):=(a(r)b(r))^{-1/2}\cdot E_J(r),
\]
即在平面坐标上施以相似缩放因子 \((a(r)b(r))^{-1/2}\)。
则：
\begin{enumerate}
  \item \(\widehat E_J(r)\) 属于 \(E(a)\) 族，且其参数可写为
  \[
  \alpha(r):=\sqrt{\frac{a(r)}{b(r)}}
  =\sqrt{\frac{L+1}{L-1}}
  =\sqrt{\coth(\Delta(r))}\in(1,\infty).
  \]
  \item 令 \(u(r):=\alpha(r)^2=\coth(\Delta(r))\)，则存在交换结合律
  \[
  u(r_1+r_2)=u(r_1)\odot u(r_2),
  \qquad
  u\odot v:=\frac{uv+1}{u+v}.
  \]
  等价地，令 \(t(r):=u(r)^{-1}=\tanh(\Delta(r))\in(0,1)\)，则满足 Einstein 加法
  \[
  t(r_1+r_2)=\frac{t(r_1)+t(r_2)}{1+t(r_1)t(r_2)}.
  \]
\end{enumerate}
\end{theorem}

\begin{proof}
由定义 \(\widehat E_J\) 的半轴为
\[
\hat a=\frac{a}{\sqrt{ab}}=\sqrt{\frac{a}{b}},\qquad
\hat b=\frac{b}{\sqrt{ab}}=\sqrt{\frac{b}{a}}=\hat a^{-1},
\]
故 \(\widehat E_J(r)\) 具有方程 \(x^2/\hat a^2+y^2\hat a^2=1\)，即属于 \(E(a)\) 族且参数为 \(\alpha=\hat a=\sqrt{a/b}\)。
代入 \(a=r_{\mathrm{geo}}+r_{\mathrm{geo}}^{-1}\)、\(b=r_{\mathrm{geo}}-r_{\mathrm{geo}}^{-1}\) 得
\(
a/b=(L+1)/(L-1)
\)，再由 \(r_{\mathrm{geo}}=e^{\Delta}\) 得 \((L+1)/(L-1)=\coth\Delta\)，从而 \(\alpha^2=\coth\Delta\)。
最后由双曲恒等式
\(
\coth(\Delta_1+\Delta_2)=\frac{\coth\Delta_1\coth\Delta_2+1}{\coth\Delta_1+\coth\Delta_2}
\)
与
\(
\tanh(\Delta_1+\Delta_2)=\frac{\tanh\Delta_1+\tanh\Delta_2}{1+\tanh\Delta_1\tanh\Delta_2}
\)
并结合 \(\Delta(r_1+r_2)=\Delta(r_1)+\Delta(r_2)\) 得结论。证毕。
\end{proof}

\begin{proposition}[边界信息体积的 primorial 下界与椭圆资源上界]\label{prop:group-jg-boundary-primorial-ellipse-bound}
设 \(b\ge 1\)，并考虑群 \((\ZZ/2\ZZ)^b\) 的平方自由 Gödel 外置：取互异素数 \(q_1,\dots,q_b\)，并定义
\[
g(\varepsilon_1,\dots,\varepsilon_b):=\prod_{i=1}^b q_i^{\varepsilon_i}\qquad(\varepsilon_i\in\{0,1\}).
\]
记最坏情形范数 \(L_{\max}:=\max_{\varepsilon} g(\varepsilon)=\prod_{i=1}^b q_i\)，并记 primorial \(p_b\#:=\prod_{i=1}^b p_i\)（\(p_i\) 为第 \(i\) 个素数）。
则：
\begin{enumerate}
  \item \(L_{\max}\ge p_b\#\)，且当且仅当 \(\{q_i\}_{i=1}^b=\{p_1,\dots,p_b\}\)（忽略排列）时取等号。
  \item 对应的 Koenigs 位移 \(\Delta_{\max}:=\frac12\log L_{\max}\) 满足
  \[
  \Delta_{\max}\ge \frac12\log(p_b\#).
  \]
  \item 相应的 Joukowsky 椭圆 \(E_J(L_{\max})\) 的半轴与面积满足
  \[
  a_{\max}\ge \sqrt{p_b\#}+\frac1{\sqrt{p_b\#}},\qquad
  b_{\max}\ge \sqrt{p_b\#}-\frac1{\sqrt{p_b\#}},
  \]
  \[
  \mathrm{Area}\bigl(E_J(L_{\max})\bigr)=\pi\,\bigl(L_{\max}-L_{\max}^{-1}\bigr)\ge \pi\,\bigl(p_b\#-(p_b\#)^{-1}\bigr).
  \]
  并且偏心率 \(e=2/a\) 满足
  \[
  e_{\max}\le \frac{2\sqrt{p_b\#}}{p_b\#+1}.
  \]
  \item 面积守恒规范化参数 \(\alpha_{\max}=\sqrt{(L_{\max}+1)/(L_{\max}-1)}\) 满足
  \[
  \alpha_{\max}\le \sqrt{\frac{p_b\#+1}{p_b\#-1}},
  \]
  因而当 \(b\to\infty\) 时其上界单调趋于 \(1\)。
\end{enumerate}
\end{proposition}

\begin{proof}
将 \(\{q_i\}\) 递增排列为 \(q_{(1)}\le\cdots\le q_{(b)}\)。由素数序性质 \(q_{(i)}\ge p_i\)，故
\(
L_{\max}=\prod_i q_{(i)}\ge \prod_i p_i=p_b\#
\)；
等号当且仅当逐项相等。
其余结论由 \(a(L)=\sqrt L+1/\sqrt L\)、\(b(L)=\sqrt L-1/\sqrt L\)、\(\Delta=\tfrac12\log L\) 与 \(\mathrm{Area}=\pi a b=\pi(L-L^{-1})\) 的单调性直接推出。
偏心率恒为 \(e=2/a\)（焦距为 \(2\)），故 \(L_{\max}\ge p_b\#\Rightarrow e_{\max}\le e(p_b\#)\)。
对 \(\alpha^2=(L+1)/(L-1)\) 的递减性同理。证毕。
\end{proof}

\begin{corollary}[dyadic 边界 Gödel 化诱导的椭圆病态下界]\label{cor:group-jg-dyadic-boundary-godel-ellipse-illconditioning}
固定 \(n\ge 2\) 与分辨率 \(m\ge 1\)，令 \(U\subset I^n\) 为任意 dyadic polycube，并以平方自由边界 Gödel 码 \(H(U)\) 表示其边界面素数乘积（例如定理 \ref{thm:spg-boundary-godelization-holographic-dictionary} 的 \(H_m(U)\)）。
取素数寄存器元 \(r_U\in\mathsf P\) 使 \(N(r_U)=H(U)\)，并令 \(A_U:=\Lambda(r_U)\in \mathrm{SL}_2(\RR)\)。
则
\[
\kappa_2(A_U)=H(U),
\]
并且当 \(N(U)\to\infty\) 时存在仅依赖于 \(n\) 的常数 \(c_n>0\) 使
\[
\log \kappa_2(A_U)\ \ge\ c_n\,N(U)^{\frac{n-1}{n}}\log N(U).
\]
\end{corollary}
\begin{proof}
由推论 \ref{cor:group-jg-godel-condition-number-identity} 有 \(\kappa_2(A_U)=N(r_U)=H(U)\)。
另一方面，定理 \ref{thm:spg-dyadic-polyclube-discrete-isoperimetry} 与定理 \ref{thm:spg-boundary-godel-bitcomplexity-isoperimetric}（或其同型 primorial 位长下界）给出：当 \(N(U)\) 足够大时 \(\log H(U)\ge c_n N(U)^{(n-1)/n}\log N(U)\)。
将 \(\log \kappa_2(A_U)=\log H(U)\) 代入即得。证毕。
\end{proof}

\begin{corollary}[容量--环模数--Koenigs 位移的一致]\label{cor:group-jg-capacity-modulus-koenigs}
沿用定理 \ref{thm:group-jg-godel-lorentz-algebraization} 的记号。则 Joukowsky 椭圆 \(E_J(r)\) 的对数容量满足
\[
\mathrm{cap}\bigl(E_J(r)\bigr)=r_{\mathrm{geo}}=\sqrt{N(r)},
\qquad
\log N(r)=2\log\mathrm{cap}\bigl(E_J(r)\bigr).
\]
并且环域
\(
A_{r_{\mathrm{geo}}}:=\{z\in\CC:\ 1<|z|<r_{\mathrm{geo}}\}
\)
的共形模数为
\[
\mathrm{mod}(A_{r_{\mathrm{geo}}})=\frac{1}{2\pi}\log r_{\mathrm{geo}}=\frac{1}{2\pi}\Delta(r),
\qquad
\log N(r)=4\pi\,\mathrm{mod}(A_{r_{\mathrm{geo}}}).
\]
\end{corollary}

\begin{proof}
环域模数公式 \(\mathrm{mod}(\{r<|z|<R\})=\frac{1}{2\pi}\log(R/r)\) 为经典结论，取 \(r=1,\ R=r_{\mathrm{geo}}\) 即得。
对容量，Joukowsky 外映射 \(w=z+z^{-1}\) 在无穷远处满足 \(w=z+O(z^{-1})\)，故外域的共形半径由外圆半径给出；而 \(w=r_{\mathrm{geo}}z+r_{\mathrm{geo}}^{-1}z^{-1}\) 为尺度变换后的同型外映射，因而 \(\mathrm{cap}(E_J(r))=r_{\mathrm{geo}}\)。证毕。
\end{proof}

\begin{theorem}[高维椭球函子：Gödel 范数的体积对数坐标]\label{thm:group-jg-godel-ellipsoid-volume}
设 \(r\in\mathsf P\) 的支撑为互异素数集合 \(\{p_1,\dots,p_d\}\)，并记 \(r_i:=r(p_i)\)。
定义对角椭球
\[
\mathcal E_d(r):=\left\{x\in\RR^d:\ \sum_{i=1}^d \frac{x_i^2}{p_i^{r_i}}\le 1\right\}.
\]
则体积满足显式等式
\[
\mathrm{Vol}\bigl(\mathcal E_d(r)\bigr)
=\mathrm{Vol}(B_d)\prod_{i=1}^d p_i^{r_i/2}
=\mathrm{Vol}(B_d)\,\sqrt{N(r)}.
\]
从而
\[
\log\mathrm{Vol}\bigl(\mathcal E_d(r)\bigr)-\log\mathrm{Vol}(B_d)=\frac12\log N(r)=\Delta(r).
\]
\end{theorem}

\begin{proof}
\(\mathcal E_d(r)\) 为单位球 \(B_d\) 在对角线性变换 \(x_i\mapsto p_i^{r_i/2}x_i\) 下的像，体积按行列式缩放：
\(
\mathrm{Vol}(T(B_d))=|\det T|\mathrm{Vol}(B_d)
\)。
此处 \(|\det T|=\prod_i p_i^{r_i/2}=\sqrt{N(r)}\)。证毕。
\end{proof}

\begin{corollary}[体积差与椭圆面积的解析同一化]\label{cor:group-jg-ellipse-area-ellipsoid-gap}
沿用定理 \ref{thm:group-jg-godel-lorentz-algebraization} 与定理 \ref{thm:group-jg-godel-ellipsoid-volume} 的记号。
对任意 \(r\in\mathsf P\) 且 \(N(r)>1\)，令 \(\Delta(r)=\frac12\log N(r)\)。
则 Joukowsky 椭圆 \(E_J(r)\) 的欧氏面积满足
\[
\mathrm{Area}\bigl(E_J(r)\bigr)=\pi\bigl(e^{2\Delta(r)}-e^{-2\Delta(r)}\bigr)=\pi\bigl(N(r)-N(r)^{-1}\bigr).
\]
因此
\[
\log \mathrm{Area}\bigl(E_J(r)\bigr)=2\Delta(r)+\log\pi+\log\bigl(1-e^{-4\Delta(r)}\bigr),
\]
并在 \(\Delta(r)\to\infty\) 时有渐近 \(\log\mathrm{Area}(E_J(r))=2\Delta(r)+\log\pi+o(1)\)。
\end{corollary}
\begin{proof}
由定理 \ref{thm:group-jg-godel-lorentz-algebraization}，\(E_J(r)\) 的半轴为
\(
a(r)=r_{\mathrm{geo}}+r_{\mathrm{geo}}^{-1}
\)
与
\(
b(r)=r_{\mathrm{geo}}-r_{\mathrm{geo}}^{-1}
\)，其中 \(r_{\mathrm{geo}}=\sqrt{N(r)}=e^{\Delta(r)}\)。
故
\[
\mathrm{Area}\bigl(E_J(r)\bigr)=\pi a(r)b(r)=\pi(r_{\mathrm{geo}}^2-r_{\mathrm{geo}}^{-2})
=\pi\bigl(N(r)-N(r)^{-1}\bigr)
=\pi\bigl(e^{2\Delta(r)}-e^{-2\Delta(r)}\bigr).
\]
取对数即得第二式；当 \(\Delta\to\infty\) 时 \(\log(1-e^{-4\Delta})=o(1)\)，从而得到所示渐近。证毕。
\end{proof}

\begin{proposition}[有限码本的最小总 Gödel 成本与 primorial 耦合]\label{prop:group-jg-min-total-godel-cost-primorial}
令 \(X\) 为有限集，\(|X|=n\ge 1\)。
考虑所有注入码本
\[
\iota: X\hookrightarrow \NN_{\mathrm{sf},>1},
\]
其中 \(\NN_{\mathrm{sf},>1}\) 为大于 \(1\) 的平方自由正整数集合。
定义总 Gödel 成本
\[
C(\iota):=\sum_{x\in X}\log \iota(x).
\]
则
\[
\min_{\iota} C(\iota)=\log(p_n\#),
\]
且极小码本可取为把 \(X\) 双射到 \(\{p_1,\dots,p_n\}\)（忽略排列）。
\end{proposition}

\begin{proof}
将 \(\{\iota(x):x\in X\}\) 递增排列为 \(q_{(1)}\le\cdots\le q_{(n)}\)。
每个 \(q_{(i)}\) 至少含一个素因子，且平方自由与互异性迫使其最小可能选择为不同素数；
由素数序性质得 \(q_{(i)}\ge p_i\)，故
\[
\prod_{x\in X}\iota(x)=\prod_{i=1}^n q_{(i)}\ge \prod_{i=1}^n p_i=p_n\#.
\]
两边取对数即得下界。
取 \(\iota(X)=\{p_1,\dots,p_n\}\) 达到该下界，故极小值为 \(\log(p_n\#)\)。证毕。
\end{proof}
