\begin{remark}[Strippable individual Euler factor]\label{rem:real-input-40-single-euler}

\begin{proposition}[Primitive orbit surgery: exact constant correction from a strippable Euler factor]\label{prop:primitive-orbit-surgery}
Suppose the \(\zeta\)-function of some weighted SFT/finite matrix family decomposes in the formal power series ring as
\[
\zeta(z,u)=\frac{1}{(1-u^{E}z^{\ell})^{m}}\cdot \zeta_{\mathrm{core}}(z,u),
\qquad m\in\ZZ_{\ge 1},\ \ell\ge 1,\ E\ge 0,
\]
and the factor \(1-u^{E}z^{\ell}\) is analytic and nonzero at the principal pole \(z=z_\star(u)\) (so that the principal growth rate \(\lambda(u)=z_\star(u)^{-1}\) is determined by the pole of \(\zeta_{\mathrm{core}}\)).
Then:
\begin{enumerate}
  \item In the Euler product, exactly \(m\) primitive orbits satisfy \((|\tau|,E(\tau))=(\ell,E)\);
  \item The primitive generating function satisfies the exact ``surgery'' decomposition
  \[
  P(z,u)=P_{\mathrm{core}}(z,u)+m\,u^{E}z^{\ell},
  \]
  where \(P(z,u)=\sum_{n\ge 1}p_n(u)z^n\) and \(P_{\mathrm{core}}\) corresponds to the primitive series of \(\zeta_{\mathrm{core}}\);
  \item If one defines the Abel finite-part constant (for fixed \(u>0\))
  \[
  \log\mathfrak{M}(u):=\sum_{n\ge 1}\left(\frac{p_n(u)}{\lambda(u)^n}-\frac{1}{n}\right),
  \]
  then the exact constant correction holds:
  \[
  \boxed{\ \log\mathfrak{M}(u)=\log\mathfrak{M}_{\mathrm{core}}(u)+m\,u^{E}\,z_\star(u)^{\ell}\ }.
  \]
\end{enumerate}
\end{proposition}
\begin{proof}
Taking logarithms of the decomposition yields
\(
\log\zeta(z,u)=m\sum_{k\ge 1}\frac{u^{kE}}{k}z^{k\ell}+\log\zeta_{\mathrm{core}}(z,u)
\).
Comparing coefficients with the Witt decomposition \(\log\zeta=\sum_{r\ge 1}\frac{1}{r}P(z^{r},u^{r})\), one sees that the primitive series gains only \(m\) atoms at \((\ell,E)\), i.e.,
\(
P(z,u)=P_{\mathrm{core}}(z,u)+m u^{E}z^{\ell}
\), so that the number of primitive orbits with \((\ell,E)\) in the Euler product is \(m\).
Finally, substituting the change in \(P\) into the series definition of \(\log\mathfrak{M}(u)\), the constant term receives an increment only at \(n=\ell\) equal to \(m u^{E}/\lambda(u)^{\ell}=m u^{E}z_\star(u)^{\ell}\), yielding the stated identity.
\end{proof}

\begin{corollary}[Spectral interpretation of the factor \texorpdfstring{$(1-uz^2)$}{(1-uz^2)}: \texorpdfstring{$\pm\sqrt u$}{+-sqrt(u)}]\label{cor:real-input-40-factor-sqrtu-eigs}
If for some weighted matrix family \(M(u)\) one has
\(
\Delta(z,u)=\det(I-zM(u))=(1-uz^2)F(z,u)
\)
and \(u>0\), then the spectrum of \(M(u)\) necessarily contains two explicit eigenvalues
\[
\boxed{\ \lambda_\pm(u)=\pm\sqrt u\ }.
\]
Therefore, for any \(n\ge 1\), there is a completely explicit trace contribution
\[
\mathrm{Tr}(M(u)^n)
=\bigl(\sqrt u\bigr)^n+\bigl(-\sqrt u\bigr)^n+\text{\rm(remaining spectral terms)}
=
\begin{cases}
2u^{n/2}+\cdots,& n\ \text{even},\\
0+\cdots,& n\ \text{odd}.
\end{cases}
\]
\end{corollary}
\begin{proof}
From
\(
\det(I-zM(u))=\prod_i(1-z\nu_i(u))
\)
(\(\nu_i(u)\) being the eigenvalues of \(M(u)\), counted with multiplicity), the factor \(1-uz^2\) must decompose as
\((1-z\sqrt u)(1+z\sqrt u)\), so \(\pm\sqrt u\) must be eigenvalues of \(M(u)\). The trace formula follows from the power sums of eigenvalues. \qed
\end{proof}

\paragraph{Sub-spectral near-degeneracy and the third scale of explicit branches}
The factor \((1-uz^2)\) yields two explicit eigenvalues \(\lambda_\pm(u)=\pm\sqrt u\) (Corollary~\ref{cor:real-input-40-factor-sqrtu-eigs}).
They do not become the Perron principal root (Proposition~\ref{prop:real-input-40-pressure-freezing} gives \(\lambda(u)\sim c\sqrt u\) with \(c>1\)),
and at the saturation endpoint \(u\to+\infty\) of this kernel, the dominant sub-spectrum does \emph{not} come from the \(\sqrt u\) branch but rather from the ``mirror Perron branch'' inside the \(F\)-block:

\begin{proposition}[Mirror Perron branch and sub-spectral near-degeneracy]\label{prop:real-input-40-output-mirror-branch}
Following the notation of Proposition~\ref{prop:real-input-40-pressure-freezing}, let \(G(\lambda,u)=\lambda^8F(\lambda^{-1},u)\).
Then as \(u\to+\infty\), besides the Perron branch, the equation \(G(\lambda,u)=0\) admits a negative real branch \(\lambda_{\mathrm{mir}}(u)<0\), and the two branches form a strict pair of Puiseux jets:
\[
\boxed{\ \lambda_1(u)=c\sqrt u+d+\frac{e}{\sqrt u}+\frac{f}{u}+O\!\left(u^{-3/2}\right)\ },
\qquad
\boxed{\ \lambda_{\mathrm{mir}}(u)=-c\sqrt u+d-\frac{e}{\sqrt u}+\frac{f}{u}+O\!\left(u^{-3/2}\right)\ }.
\]
Therefore the largest non-principal modulus eigenvalue satisfies
\[
\boxed{\ \Lambda_2(u)=|\lambda_{\mathrm{mir}}(u)|\sim c\sqrt u\ },
\qquad\text{so that}\qquad
\boxed{\ \lim_{u\to+\infty}\frac{\Lambda_2(u)}{\lambda_1(u)}=1\ }.
\]
Moreover, the relative spectral gap has the explicit leading order
\[
\boxed{\ 1-\frac{\Lambda_2(u)}{\lambda_1(u)}=\frac{2d}{c}\,u^{-1/2}+O(u^{-1})\ }.
\]
On the other hand, the explicit branch satisfies a strictly sub-leading scale
\[
\boxed{\ \frac{\sqrt u}{\lambda_1(u)}\longrightarrow \frac{1}{c}\in(0,1)\qquad(u\to+\infty)\ }.
\]
\end{proposition}
\begin{proof}
Under the rescaling \(\lambda=\sqrt u\,x\) as \(u\to+\infty\), the argument from Proposition~\ref{prop:real-input-40-pressure-freezing} shows that the limiting principal equation can be written as a quartic in \(y=x^2\):
\(
y^4-6y^3+9y^2-y-1=0
\),
whose largest positive root is \(y_{\max}=c^2\).
Thus at the saturation endpoint \(u\to+\infty\), the equation \(G(\lambda,u)=0\) must produce two branches at the principal scale related by sign reversal:
\(
\lambda\approx \pm\sqrt{y_{\max}}\,\sqrt u=\pm c\sqrt u
\).
The positive branch is the Perron root (by non-negativity and Perron--Frobenius uniqueness), while the negative branch is \(\lambda_{\mathrm{mir}}(u)\), yielding the stated near-degeneracy conclusion.
For the higher-order terms, set \(t:=u^{-1/2}\to 0^+\) and use the Laurent/Puiseux ansatz
\(\lambda_1(t)=c\,t^{-1}+d+e\,t+f\,t^2+\cdots\), \(\lambda_{\mathrm{mir}}(t)=-c\,t^{-1}+d-e\,t+f\,t^2+\cdots\), substitute into \(G(\lambda,t^{-2})=0\), and compare coefficients order by order to solve for \(d,e,f\) (see the script \texttt{scripts/exp\_real\_input\_40\_output\_potential\_puiseux\_asymptotics.py}). \qed
\end{proof}

\begin{remark}[Interpretation: mixing time divergence at the saturation endpoint is driven by the mirror branch]\label{rem:real-input-40-output-mirror-mixing}
Proposition~\ref{prop:real-input-40-output-mirror-branch} shows that as \(u\to+\infty\), the sub-spectral ratio \(\Lambda_2(u)/\lambda_1(u)\to 1\), and
\(
1-\Lambda_2(u)/\lambda_1(u)=\Theta(u^{-1/2})
\).
Therefore any mixing scale controlled by the spectral gap ratio
\(
\tau_{\mathrm{mix}}(u)\asymp 1/\bigl(-\log(\Lambda_2(u)/\lambda_1(u))\bigr)
\)
diverges at rate \(\sqrt u\) with \(u\); the dominant mechanism comes from the mirror branch of the \(F\)-block, not from the explicit \(\sqrt u\) branch.
The latter still provides an independent, traceable third-scale candidate \(\sqrt u/\lambda_1(u)\to 1/c\), which can be used to separate the residual spectral layers beyond the ``near-degenerate slow mode.''
\end{remark}

\begin{proposition}[Dirichlet twist spectrum for output $1$ counting modulo $m$: systematic failure of the square-root error]\label{prop:real-input-40-output-dirichlet-nonrh}

\rho_{m,j}:=\rho\!\bigl(M(\omega_m^j)\bigr).
\]
Equivalently, from \(\Delta(z,u)=\det(I-zM(u))\),
\[
\rho_{m,j}=\frac{1}{\min\{|z|:\Delta(z,\omega_m^j)=0\}}.
\]

\[
\theta_{m,j}:=\frac{\log \rho_{m,j}}{\log \lambda},

\rho_m:=\max_{1\le j\le m-1}\rho_{m,j},\quad
\theta_m:=\max_{1\le j\le m-1}\theta_{m,j}.
\]
Then the nontrivial Fourier components of the output $1$ count modulo $m$ (at both the periodic and primitive levels) decay at the exponential rate \(O(\rho_m^n)\); and on this kernel there exist explicit values of \(m\) such that

Thus within this window, the fourth-order correction further compresses the error relative to the second-order approximation by factors of approximately \(5.3\times10^2\), \(1.3\times10^3\), and \(3.7\times10^3\) (for \(m=5,10,20\), respectively).
\par
\noindent Furthermore, the fact that \(j_{\mathrm{worst}}\) is consistently \(1\) in the table suggests a reusable algorithmic rule: when performing large-modulus sweeps, one can prioritize computing only the \(j=1\) channel (with a small number of spot checks to verify whether it remains the worst-case mode), thereby reducing ``full \(j\) sweep'' to ``prediction + sparse verification''---an auditable workflow.
\end{remark}

\begin{remark}[Zero eigenvalue branch activated by the weight]\label{rem:real-input-40-zero-activation}
In $\Delta(z,u)=(1-uz^2)F(z,u)$, the leading coefficient of $F$ is $(u^3-u^4)z^8=u^3(1-u)z^8$. Therefore when $u=1$ this term vanishes and the degree of $F$ drops from $8$ to $7$; when $u\neq 1$ the degree is restored to $8$. This means that one zero-eigenvalue direction in the unweighted kernel is ``activated'' into a nonzero eigenvalue branch under weight twisting, and this branch degenerates back to $0$ as $u\to 1$.
\end{remark}

\begin{proposition}[Critical exponent of the activated branch: \texorpdfstring{$\lambda_{\mathrm{act}}(u)\sim u-1$}{lambda_act(u) ~ u-1}]\label{prop:real-input-40-activated-branch-linear}
Following the polynomial from Proposition~\ref{prop:real-input-40-pressure-freezing},
\[
G(\lambda,u)=\lambda^8\,F(\lambda^{-1},u).
\]
There exists a unique (simple) small-eigenvalue branch \(\lambda_{\mathrm{act}}(u)\) at \(u=1\) satisfying
\[
G(\lambda_{\mathrm{act}}(u),u)=0,\qquad \lambda_{\mathrm{act}}(1)=0,
\]
and it has linear asymptotics as \(u\to 1\):
\[
\boxed{\ \lambda_{\mathrm{act}}(u)=(u-1)+O\!\bigl((u-1)^2\bigr)\ }.
\]
Therefore the critical exponent of this branch is \(1\): \(\lambda_{\mathrm{act}}(u)=\Theta(u-1)\).
\end{proposition}
\begin{proof}
From the explicit form of \(G(\lambda,u)\) (given in \eqref{prop:real-input-40-pressure-freezing}), its constant term is
\(
G(0,u)=u^3-u^4=u^3(1-u)
\), so \(G(0,1)=0\), i.e., \(\lambda=0\) is a root when \(u=1\).
Meanwhile the coefficient of the first-order term is \((4u^3-3u^2)\lambda\), so
\(
\partial_\lambda G(0,1)=4-3=1\neq 0
\), and thus \(\lambda=0\) is a simple root at \(u=1\).
By the implicit function theorem, there exists an analytic function \(\lambda_{\mathrm{act}}(u)\) satisfying \(G(\lambda_{\mathrm{act}}(u),u)=0\) and \(\lambda_{\mathrm{act}}(1)=0\), with
\[
\lambda_{\mathrm{act}}'(1)=-\frac{\partial_u G(0,1)}{\partial_\lambda G(0,1)}.
\]
Since \(\partial_u G(0,u)=\frac{d}{du}(u^3-u^4)=3u^2-4u^3\), one has \(\partial_u G(0,1)=-1\). Therefore \(\lambda'_{\mathrm{act}}(1)=1\), giving
\(
\lambda_{\mathrm{act}}(u)=(u-1)+O((u-1)^2)
\).
\qed
\end{proof}

\begin{remark}[Algorithmic selection rule: first deflate \texorpdfstring{$\lambda_{\mathrm{act}}$}{lambda_act}, then eliminate/root-find]\label{rem:real-input-40-deflate-activated-branch}
The activated branch \(\lambda_{\mathrm{act}}(u)\) satisfies \(\lambda_{\mathrm{act}}(u)=\Theta(u-1)\) near \(u\approx 1\) (Proposition~\ref{prop:real-input-40-activated-branch-linear}), with a higher-order Taylor expansion (Remark~\ref{rem:real-input-40-activated-branch-series}).
Numerically, this means that when \(u\) is close to \(1\), the root scales of the algebraic curve \(G(\lambda,u)=0\) exhibit strong separation---one root tends to \(0\) while the others remain at \(O(1)\)---thereby introducing significant scale mismatch and coefficient cancellation in root-finding/elimination.
\par
\noindent Therefore, in any elimination or numerical root-finding task involving \(u\approx 1\) (e.g., tracking spectral branches in a complex neighborhood of \(u\), or performing stable root computation for \(\Delta(z,u)\)), one should first perform an \emph{explicit deflation} of the activated branch:
\begin{itemize}
  \item Use the expansion from Remark~\ref{rem:real-input-40-activated-branch-series} to evaluate \(\lambda_{\mathrm{act}}(u)\) at a given \(u\) (the second-order coefficient is \(0\); truncating to sixth order already significantly stabilizes the numerical scale);
  \item View \(G(\lambda,u)\) as a univariate polynomial in \(\lambda\) and perform one factor deflation
  \(
  G(\lambda,u)=(\lambda-\lambda_{\mathrm{act}}(u))\,\widetilde G(\lambda,u)
  \)
  (implementable via synthetic division or companion matrix deflation);
  \item Perform subsequent root-finding, resultant computation, or branch tracking on \(\widetilde G\).
\end{itemize}
After deflation, the pathological scaling near \(u\to 1\) is substantially mitigated, bringing subsequent computations back to the condition number level of standard polynomial problems; this also condenses the hint that ``this expansion can be used for branch deflation'' into a reusable algorithmic selection rule: \textbf{first deflate the activated branch, then perform elimination/root-finding}.
\end{remark}

\begin{proposition}[Finite-part constant limit under extreme biasing]\label{prop:real-input-40-logM-infty}
Let $z_\star(u)=\lambda(u)^{-1}$, and define
\[
b:=\lim_{u\to\infty}u\,z_\star(u)^2=\frac{1}{c^2}.
\]
Then $b$ is the largest positive root of the equation
\[
1-6b+9b^2-b^3-b^4=0
\]
with numerical value $b\approx 0.27807480214113$. Write
\[
C(u):=\lim_{z\to z_\star(u)}(1-\lambda(u)z)\,\zeta(z,u),
\]
then the limit
\[
C_\infty:=\lim_{u\to\infty}C(u)=\frac{1}{2b(1-b)(6-18b+3b^2+4b^3)}\approx 1.89745149363
\]
exists. Moreover, the Abel finite-part constant satisfies the limit
\[
\log\mathfrak{M}_\infty:=\lim_{u\to\infty}\log\mathfrak{M}(u),
\]
with the rapidly convergent M\"obius--Euler expression
\[
\log\mathfrak{M}_\infty
=\log C_\infty
-\sum_{m\ge 2}\frac{\mu(m)}{m}\log\Bigl[(1-b^m)\bigl(1-6b^m+9b^{2m}-b^{3m}-b^{4m}\bigr)\Bigr].

\end{proposition}
