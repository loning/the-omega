\section{Conjugacy Between Modulo \texorpdfstring{$L$}{L} and Modulo \texorpdfstring{$1$}{1}: One-Dimensional Scan and Golden Parameters}

\subsection{Scale Conjugacy}

Let $C_L=\RR/L\ZZ$. Consider translation
\[
R_\delta^{(L)}(x)=x+\delta\pmod{L}.
\]

\begin{proposition}[Scale conjugacy]\label{prop:conjugacy}
The map
\[
h_L:C_L\to C_1,\qquad h_L([x]_L)=[x/L]_1
\]
is a homeomorphism and satisfies
\[
h_L\circ R_\delta^{(L)}=R_{\delta/L}^{(1)}\circ h_L.
\]
Hence translation dynamics modulo $L$ can be uniformly reduced to the step-ratio parameter $\alpha=\delta/L$.
\end{proposition}

\subsection{Density and Unique Ergodicity}

On $C_1$, if $\alpha\notin\QQ$, the orbit is dense and the system is uniquely ergodic. For any Riemann-integrable $f$,
\[
\lim_{N\to\infty}\frac{1}{N}\sum_{n=0}^{N-1}f(x_0+n\alpha)=\int_0^1 f(x)\,\dd x.
\]
Taking $f=\ind_W$ (with zero boundary measure) gives window frequency equal to window measure.

\subsection{Normalized Definition of the Golden Step Length}

By scale conjugacy, the ``golden step modulo $\pi$'' can be equivalently defined as
\[
\alpha=\frac{1}{\varphi},\qquad \delta=\frac{\pi}{\varphi}.
\]
Its special role is not mere density (all irrationals are dense) but an extremal relation between finite-sample discrepancy, suppression of short returns, and Diophantine approximation properties (see Appendix \ref{app:golden-approx}).

\subsection{Near Returns and Finite-Resolution Periodization}

\begin{definition}[$\varepsilon$-return]
Given a metric $d$, if there exists $N>0$ such that $d(x_N,x_0)<\varepsilon$, we call it an $\varepsilon$-return.
\end{definition}
Irrational translations have no strict period, but for any $\varepsilon>0$ there are infinitely many $\varepsilon$-return times. Finite-resolution readout compresses such $\varepsilon$-returns into periodic symbolic structures, so that ``closed curves/periods'' become stable classes at the readout level rather than ontic periodicity.
