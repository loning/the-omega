\subsubsection{Principal singularity star clusters and fractal codimension of the ambiguity shell: analytic equivalence characterization via period modulation}\label{subsubsec:Ym-amb-singularity-star-codimension}
This subsection records two analytic-side corollaries that require no additional structural assumptions, building on the interface of Theorem \ref{thm:Ym-unsync-prefix-pf-sharp} that establishes ``unsynchronized prefix count \(=\) ambiguity subgraph path count'':
the first is a generating function characterization of ``period modulation \(\Longleftrightarrow\) principal singularity star cluster'';
the second provides closed-form expressions for the Hausdorff dimension and codimension of the ambiguity shell under the prefix ultrametric.

\begin{theorem}[Period modulation \texorpdfstring{$\Longleftrightarrow$}{<->} principal singularity star cluster: non-cancellability of equimodular poles]\label{thm:Ym-amb-main-singularity-star}
Assume \(Y_m^{\mathrm{amb}}\neq\varnothing\).
Following the notation of Definition \ref{def:Ym-unsync-prefix-count} and Theorem \ref{thm:Ym-unsync-prefix-pf-sharp}, let
\[
N_m^{\mathrm{unsync}}(n):=\#\mathcal{W}_{m}^{\mathrm{unsync}}(n),
\qquad
U_m(z):=\sum_{n\ge 0}N_m^{\mathrm{unsync}}(n)\,z^n,
\]
and let \(A_m^{\mathrm{amb}}\) denote the ambiguity adjacency matrix weighted by edge multiplicities (defined in Theorem \ref{thm:Ym-sync-tail-pf-sharp} and used in Theorem \ref{thm:Ym-unsync-prefix-pf-sharp} for unsynchronized prefix counting).
Then we have the strict identity
\[
U_m(z)=e_{V_m}^{\mathsf T}(I-zA_m^{\mathrm{amb}})^{-1}\mathbf 1,
\]
hence \(U_m\) is a rational function.
\par
Furthermore, suppose the maximal strongly connected component with spectral radius is unique and has period \(p\ge 1\).
Let \(\rho_{\mathrm{amb}}(m):=\rho(A_m^{\mathrm{amb}})\). Then the radius of convergence of \(U_m(z)\) is \(R=\rho_{\mathrm{amb}}(m)^{-1}\), and the solutions to \(\det(I-zA_m^{\mathrm{amb}})=0\) on the circle \(|z|=R\) form the equimodular point set
\[
z_r=\rho_{\mathrm{amb}}(m)^{-1}e^{2\pi i r/p},\qquad r=0,1,\dots,p-1.
\]
When \(p>1\), this point set yields the ``principal singularity candidate star cluster'': they are all solutions to \(\det(I-zA_m^{\mathrm{amb}})=0\) on \(|z|=R\).
Since \(U_m(z)\) is a matrix element of the resolvent, cancellations may occur; when no cancellation occurs, all \(p\) points become principal poles of \(U_m\), consistent with the \(n\bmod p\) modulation in Theorem \ref{thm:Ym-sync-tail-pf-sharp}.
\end{theorem}

\begin{proof}
By the matrix power formula
\(
N_m^{\mathrm{unsync}}(n)=e_{V_m}^{\mathsf T}(A_m^{\mathrm{amb}})^n\mathbf 1
\)
from the proof of Theorem \ref{thm:Ym-unsync-prefix-pf-sharp}, for \(|z|<\rho_{\mathrm{amb}}(m)^{-1}\) we have
\[
U_m(z)=\sum_{n\ge 0}e_{V_m}^{\mathsf T}(A_m^{\mathrm{amb}})^n\mathbf 1\,z^n
=e_{V_m}^{\mathsf T}\left(\sum_{n\ge 0}(zA_m^{\mathrm{amb}})^n\right)\mathbf 1
=e_{V_m}^{\mathsf T}(I-zA_m^{\mathrm{amb}})^{-1}\mathbf 1,
\]
thus \(U_m\) is rational and its singularities include the zeros of \(\det(I-zA_m^{\mathrm{amb}})=0\).
\par
If the maximal strongly connected component with spectral radius is unique and has period \(p\), then Perron--Frobenius theory yields that the eigenvalues on the spectral circle are precisely
\(
\rho_{\mathrm{amb}}(m)e^{2\pi i r/p}
\)
(\(r=0,\dots,p-1\)), hence
\(
\det(I-zA_m^{\mathrm{amb}})=0
\)
on \(|z|=R\) has solutions precisely at
\(
z_r=\rho_{\mathrm{amb}}(m)^{-1}e^{2\pi i r/p}
\).
Correspondingly, the resolvent \((I-zA_m^{\mathrm{amb}})^{-1}\) has singularities at these points; the principal singularities of \(U_m(z)=e_{V_m}^{\mathsf T}(I-zA_m^{\mathrm{amb}})^{-1}\mathbf 1\) can only come from these candidate points.
If the pairing of \(e_{V_m}^{\mathsf T}\) and \(\mathbf 1\) on the corresponding spectral projection is non-degenerate, then no cancellation occurs, and all candidate points become actual principal poles of \(U_m\); in this case, the equimodular pole cluster in the complex plane and the \(n\bmod p\) modulation in the time domain are analytically equivalent characterizations. This completes the proof.
\end{proof}

\begin{corollary}[Hausdorff dimension and codimension closed-form for the ambiguity shell: \texorpdfstring{$\mathrm{codim}=\varepsilon_m/\log 2$}{codim=eps/log2}]\label{cor:Ym-amb-hausdorff-codim}
Assume \(Y_m^{\mathrm{amb}}\neq\varnothing\), and let
\[
\varepsilon_m:=\log 2-\log\rho_{\mathrm{amb}}(m)>0
\]
denote the entropy gap (Theorem \ref{thm:Ym-sync-tail-pf-sharp}).
On \(X_m^{\NN}\), equip the prefix ultrametric
\[
d(y,y'):=2^{-\ell(y,y')},
\qquad
\ell(y,y'):=\sup\{k\ge 0:\ y_0\cdots y_{k-1}=y'_0\cdots y'_{k-1}\},
\]
and restrict it to the subsystem \(Y_m^{\mathrm{amb}}\subseteq X_m^{\NN}\).
Then
\[
\dim_{\mathrm H}(Y_m^{\mathrm{amb}})
=\frac{h_{\mathrm{top}}(Y_m^{\mathrm{amb}})}{\log 2}
=\frac{\log\rho_{\mathrm{amb}}(m)}{\log 2}
=1-\frac{\varepsilon_m}{\log 2}.
\]
Hence the fractal codimension of the ambiguity shell is strictly
\[
\mathrm{codim}(Y_m^{\mathrm{amb}})=\frac{\varepsilon_m}{\log 2}.
\]
\end{corollary}

\begin{proof}
Under the prefix ultrametric \(d\), cylinder sets of length \(n\) are precisely closed balls of radius \(2^{-n}\); and for any subshift \(Z\subseteq X_m^{\NN}\), its length-\(n\) word count satisfies
\(
N_Z(n)=\exp(h_{\mathrm{top}}(Z)\,n+o(n))
\).
By the standard dimension formula (in this ultrametric, the Hausdorff dimension equals entropy divided by \(\log 2\)), we obtain
\(
\dim_{\mathrm H}(Z)=h_{\mathrm{top}}(Z)/\log 2
\).
Taking \(Z=Y_m^{\mathrm{amb}}\) and using \(h_{\mathrm{top}}(Y_m^{\mathrm{amb}})=\log\rho_{\mathrm{amb}}(m)\) (Theorem \ref{thm:Ym-sync-tail-pf-sharp}) yields the stated closed-form; codimension is the direct rearrangement of \(1-\dim_{\mathrm H}\). This completes the proof.
\end{proof}
