\section{附录 C：Zeckendorf 合法语言计数与稳定化核构造}
\label{app:zeckendorf-fold}

无相邻 $1$ 的合法串计数满足 Fibonacci 递推并给出 $\card{X_m^{\mathrm Z}}=F_{m+2}$ 的组合证明见 \cite{PlanetMathZeckendorfCombinatorial}。
Zeckendorf 定理的形式化版本与唯一性表述见 \cite{AFPZeckendorf}。

\subsection{多尺度一致性：restriction 逆系统与逆极限同构}为严格化跨窗口长度 $m$ 的相容性，本附录给出 Zeckendorf 稳定类型空间的逆系统与其逆极限描述。\begin{definition}[位号标记的稳定类型空间]\label{def:app-XmZ-indexed}
对 $m\in\NN$，定义
\[
X_m^{\mathrm Z,\#}:=\Bigl\{(x_k)_{2\le k\le m+1}\in\{0,1\}^{\{2,3,\dots,m+1\}}:\ x_kx_{k+1}=0\ \text{对所有}\ 2\le k\le m\Bigr\}.
\]
其与词表示 $X_m^{\mathrm Z}\subset\{0,1\}^m$ 通过 $(x_k)_{2\le k\le m+1}\mapsto (x_{m+1},x_m,\dots,x_2)$ 自然同构。
\end{definition}

\begin{definition}[restriction 投影]\label{def:app-pi}
对 $m_2\ge m_1$，定义
\[
\pi_{m_2\to m_1}:X_{m_2}^{\mathrm Z,\#}\to X_{m_1}^{\mathrm Z,\#},\qquad
\bigl(\pi_{m_2\to m_1}(x)\bigr)_k:=x_k\ \ (2\le k\le m_1+1).
\]
\end{definition}\begin{lemma}[基本性质]\label{lem:app-pi-basic}
$\pi_{m_2\to m_1}$ 良定义、满射，并满足 $\pi_{m_3\to m_1}=\pi_{m_2\to m_1}\circ \pi_{m_3\to m_2}$（$m_3\ge m_2\ge m_1$）。
\end{lemma}

\begin{proof}
restriction 保持局部禁词约束，故良定义；任意元素可通过尾部补零扩展，故满射；一致性由逐坐标定义直接得到。
\end{proof}\begin{definition}[无穷 Zeckendorf 子移位]\label{def:app-XinfZ}
定义
\[
X_\infty^{\mathrm Z}:=\Bigl\{(x_k)_{k\ge 2}\in\{0,1\}^{\{2,3,\dots\}}:\ x_kx_{k+1}=0\ \text{对所有}\ k\ge 2\Bigr\},
\]
并定义 $\pi_m^\infty:X_\infty^{\mathrm Z}\to X_m^{\mathrm Z,\#}$ 为 restriction：$(\pi_m^\infty(x))_k:=x_k$（$2\le k\le m+1$）。
\end{definition}\begin{theorem}[逆极限同构]\label{thm:app-inverse-limit}
存在自然同构
\[
X_\infty^{\mathrm Z}\ \cong\ \varprojlim X_m^{\mathrm Z,\#},
\]
并满足交换关系 $\pi_{m_2\to m_1}\circ \pi_{m_2}^\infty=\pi_{m_1}^\infty$（$m_2\ge m_1$）。
\end{theorem}

\begin{proof}
对 $x\in X_\infty^{\mathrm Z}$ 取其 restriction 族 $(\pi_m^\infty(x))_{m\ge 1}$，得到一致族。反向，若给定一致族 $(x^{(m)})_{m\ge 1}$，令 $x_k:=x^{(m)}_k$（任取 $m\ge k-1$），一致性保证良定并保留禁词约束。两构造互逆且自然。
\end{proof}

\subsection{块编码双射：固定 \texorpdfstring{$m=6$}{m=6} 递归读出与 \texorpdfstring{$m=6K$}{m=6K} 的对齐}

令 $B:=X_6^{\mathrm Z}$ 视为 6-块空间。对 $b=(b_1,\dots,b_6)\in B$ 定义 $\mathrm{fst}(b)=b_1$、$\mathrm{lst}(b)=b_6$，并定义边界兼容关系
\[
b\to b'\iff \mathrm{lst}(b)\,\mathrm{fst}(b')\neq 11.
\]

\begin{proposition}[块编码双射]\label{prop:app-block-bijection}
对任意 $K\in\NN$，存在自然双射
\[
X_{6K}^{\mathrm Z}\ \longleftrightarrow\ \{(b_0,\dots,b_{K-1})\in B^K:\ b_i\to b_{i+1}\ \text{对所有}\ 0\le i<K-1\},
\]
其对应由“按 6 位切分”与“按块拼接”给出。
\end{proposition}

\begin{proof}
设 $w\in X_{6K}^{\mathrm Z}$。按连续 6 位切分得到 $K$ 个 6-块 $b_0,\dots,b_{K-1}$。块内合法由 $w\in X_{6K}^{\mathrm Z}$ 立即推出 $b_i\in X_6^{\mathrm Z}$。块间唯一可能产生禁词 $11$ 的位置是边界相邻位，因此 $w$ 的合法性等价于对所有 $i$ 满足 $b_i\to b_{i+1}$。反向，给定兼容路径 $(b_i)$，按块拼接得到长度 $6K$ 串；块内合法与边界兼容共同保证拼接后不含子词 $11$。切分与拼接互为逆，得双射。
\end{proof}
