\begin{theorem}[Optimal recovery time equals conditional entropy: unique measure of folding missing information]\label{thm:xi-optimal-recovery-time-equals-conditional-entropy}
Let \(\Fold_m:\Omega_m\to X_m\) be many-to-one folding, \(\mu\in\Delta(\Omega_m)\), and denote
\(
x:=\Fold_m(\omega)
\).
Consider any sequential readout protocol: given \(x\), read one alphabet \(\mathcal A\) symbol at each step (adaptive, possibly randomized), and recover \(\omega\) with probability \(1\);
denote stopping time as \(T\).
Then necessarily
\[
\mathbb E_\mu[T]\ \ge\ \frac{H_\mu(\omega\mid x)}{\log|\mathcal A|}.
\]
If conditional distribution \(\mu(\omega\mid x)\) is computable, then there exists a family of protocols such that
\[
\mathbb E_\mu[T]\ \le\ \frac{H_\mu(\omega\mid x)}{\log|\mathcal A|}+1.
\]
Thus in the optimal sense, time is exactly the conditional entropy scale of folding lost information.
\end{theorem}

\begin{proof}
Lower bound is given by at most \(\log|\mathcal A|\) bits per step and conditional entropy bound for optional stopping times.
Upper bound is given by prefix coding construction with side information (conditional code family by \(x\)), with redundancy not exceeding one codeword length unit.
\end{proof}

\begin{theorem}[Finite criterion for time additivity: cohomology obstruction completely generated by critical squares]\label{thm:xi-time-additivity-finite-critical-square-criterion}
Let \(\mathbf{PW}\) be given by finitely generated convergent rewrite system (terminating and confluent), and let
\[
\alpha(w_1,w_2):=\ell(w_1\circ w_2)-\ell(w_1)-\ell(w_2).
\]
Then cohomology class \([\alpha]\in H^2(\mathbf{PW};\mathbb Z)\) satisfies:
\[
[\alpha]=0
\quad\Longleftrightarrow\quad
\delta_{\square}=0\ \ \text{for all critical squares},
\]
where \(\delta_{\square}\) is the anomaly difference of two confluence filling paths for that square.
Therefore, strictly additive time can be completely determined by the finite critical pair set.
\end{theorem}

\begin{proof}
All higher-order overlaps in convergent systems are generated by critical pair completion.
If all critical square defects are zero, then anomaly is boundary at generating layer, thus globally cohomologically trivial; conversely, \([\alpha]=0\) directly implies any square defect is zero.
\end{proof}

\begin{theorem}[Uniqueness of canonical time: minimal energy gauge gives unique harmonic representative of cohomology class]\label{thm:xi-canonical-time-gauge-harmonic-representative}
Let generating graph be finite and connected, edge set \(E\), edge length function \(\ell:E\to\mathbb R\).
For any object function \(b:\mathrm{Ob}(\mathbf{PW})\to\mathbb R\), define
\[
\ell_b(e):=\ell(e)+b(t(e))-b(s(e)),
\qquad
\mathcal E(b):=\sum_{e\in E}\ell_b(e)^2.
\]
Then \(\mathcal E\) has a unique minimal element \(b^\ast\) on "add-constant equivalence classes," inducing canonical time
\(
\ell^\ast:=\ell_{b^\ast}
\).
This \(\ell^\ast\) satisfies the discrete harmonic equation (net flux at each point is zero), and depends only on cohomology class \([\ell]\in H^1(\mathbf{PW};\mathbb R)\).
\end{theorem}

\begin{proof}
\(\mathcal E\) is a strictly convex quadratic form, with kernel only constant translations.
Thus unique minimal element exists on quotient space.
Euler--Lagrange equation is exactly discrete divergence zero, giving harmonic representative; cohomological equivalence only changes gradient term in gauge, not the minimal representative class.
\end{proof}

\begin{theorem}[Cross-resolution time consistency obstruction: renormalization defect forms observable first-order invariant]\label{thm:xi-cross-resolution-time-renormalization-defect}
Let resolution tower \((X_m,p_{m+1,m})\) and strictly additive length \(\ell_m\) at each layer be given.
Define renormalization defect
\[
\Delta_m(w):=\ell_{m+1}(w)-\ell_m\!\bigl(p_{m+1,m}(w)\bigr).
\]
Then there exists constant sequence \(c_m\) such that
\[
\ell_m\!\bigl(p_{m+1,m}(w)\bigr)=\ell_{m+1}(w)-c_m\quad(\forall w)
\]
if and only if \(\Delta_m(w)\) is constant in \(w\) (depends only on \(m\)).
If \(\Delta_m\) has nontrivial \(w\)-dependence, then any global time consistency requires additional central extension; its minimal extension dimension equals the rank of abelian subgroup generated by all defect values.
\end{theorem}

\begin{proof}
If consistency holds, then \(\Delta_m\equiv c_m\) obviously.
Reverse direction: constant defect can be absorbed directly into inter-layer baseline rescaling.
When defect depends on \(w\), it cannot be canceled by single constant rescaling, can only externalize to central coordinate; minimal dimension given by number of independent defect generators.
\end{proof}

\begin{theorem}[Time-weighted spectral radius characterizes growth law: unique exponent of certificate counting determined by time pressure zero]\label{thm:xi-time-weighted-spectral-radius-growth-law}
On finite state representation, let each basic edge \(e\) be assigned duration \(\widetilde\ell(e)\in\mathbb N\), define time-weighted matrix
\[
(A_s)_{ij}:=\sum_{e:i\to j}e^{-s\widetilde\ell(e)}.
\]
Then there exists unique \(s_\ast>0\) such that \(\rho(A_{s_\ast})=1\).
Moreover, on reachable irreducible components,
\[
\lim_{t\to\infty}\frac1t\log \#\{w:\widetilde\ell(w)\le t\}=s_\ast.
\]
Thus exponential growth rate of realizable certificate counting is uniquely determined by single structural constant \(s_\ast\).
\end{theorem}

\begin{proof}
\(s\mapsto \rho(A_s)\) is continuous strictly monotone decreasing, exceeds \(1\) at \(s=0\), tends to \(0\) as \(s\to\infty\), thus unique zero-pressure point exists.
Counting growth rate is given by Perron--Frobenius and Tauber-type arguments for weighted transfer operator.
\end{proof}

\begin{theorem}[Multiscale CMI as time curvature: minimal extra time cost lower-bounded by curvature total amount]\label{thm:xi-multiscale-cmi-as-time-curvature}
Let \((X_m(t))_{t\ge 0}\) be multiscale readout process, assuming first-order Markov realizability at each fixed scale.
Define scale curvature
\[
\kappa_m:=I\!\bigl(X_{m+1};X_{m-1}\mid X_m\bigr).
\]
Then the following are equivalent:
\begin{enumerate}
  \item There exists global extended process making all scale chains simultaneously Markov factors of same underlying micro-process;
  \item For all \(m\), \(\kappa_m=0\).
\end{enumerate}
If there exists \(\kappa_m>0\), then any causality-preserving lossless scale-stitching protocol satisfies
\[
\mathbb E[T_{\mathrm{over}}]\ \ge\ \frac{\sum_m \kappa_m}{\log|\mathcal A|}.
\]
\end{theorem}

\begin{proof}
Zero conditional mutual information is equivalent to three-point Markov stitching condition; holding layer-wise yields global stitchability.
Nonzero \(\kappa_m\) indicates cross-scale additional dependency, requiring extra readout to disambiguate; per-step information upper limit gives time lower bound.
\end{proof}

\begin{theorem}[Central rigidity of time: semantic automorphisms on time register only allow sign and scale transformations]\label{thm:xi-time-center-semantic-rigidity}
In central extension \(\widetilde{\mathbf{PW}}_{\alpha}\), the time register central subgroup is denoted
\(
Z\cong\mathbb Z
\), with elements \((\mathrm{id},k)\).
For any automorphism \(\varphi\) preserving projection semantics and \(2\)-certificate equivalence classes, its action on \(Z\) must be
\[
\varphi(\mathrm{id},k)=(\mathrm{id},\pm k).
\]
If the center is relaxed to real linear unit rescaling, only overall scale transformation and sign inversion are allowed.
\end{theorem}

\begin{proof}
Central automorphism is determined by group automorphism of \(Z\); for integer center \(Z\cong\mathbb Z\), only \(\pm1\) exists.
Semantic preservation excludes mapping center into non-center components, thus conclusion holds.
\end{proof}

\begin{theorem}[Equivalence of address and stopping time: visibility set and prefix code set strictly isomorphic]\label{thm:xi-address-stopping-time-equivalence}
Let readout filtration be \((\mathcal F_t)_{t\ge 0}\).
Call address \(a\) visible if there exists stopping time \(\tau_a\) such that event "object falls in fiber subset corresponding to \(a\)" is decidable on \(\mathcal F_{\tau_a}\).
Then:
\begin{enumerate}
  \item Visible address set \(\mathcal A_{\mathrm{vis}}\) must be a prefix code;
  \item Any prefix code \(C\) uniquely induces a family of terminable stopping times \(\{\tau_c\}_{c\in C}\);
  \item Code length and minimal stopping time lower bound satisfy the same entropy law, constrained by Theorem \ref{thm:xi-optimal-recovery-time-equals-conditional-entropy}.
\end{enumerate}
Therefore, "visible existence" can be strictly restated as "existence of finite stopping time."
\end{theorem}

\begin{proof}
If not prefix-free, premature halting ambiguity arises, violating unique decidability; thus must be prefix conflict-free.
Prefix code conversely gives unique halting rule.
Length--time correspondence comes from per-step information upper limit and stopping time coding inequality.
\end{proof}

\begin{theorem}[Monotonicity of time under coarse-graining: coarse-graining cannot increase minimal realization time, gap controlled by compressible information amount lower bound]\label{thm:xi-time-monotone-under-coarse-graining}
Let \(C:X_{m+1}\to X_m\) be coarse-graining functor, \(\tau(\cdot)\) denote minimal realization time under fixed semantic constraint.
Then for any morphism \(w\),
\[
\tau(C(w))\le \tau(w).
\]
If under probabilistic model coarse-graining corresponds to discarding some observable sub-\(\sigma\)-algebra, then average time gap satisfies
\[
\mathbb E\!\bigl[\tau(w)-\tau(C(w))\bigr]
\ \ge\
\frac{I(\text{discarded information};\text{retained information})}{\log|\mathcal A|}.
\]
Thus coarse-graining is monotone in the time sense, with gap controlled by compressible information amount lower bound.
\end{theorem}

\begin{proof}
Any protocol realizing \(w\) can realize \(C(w)\) by post-processing, thus minimal time does not increase.
Second formula is obtained by combining data processing and per-step readout information upper limit.
\end{proof}
