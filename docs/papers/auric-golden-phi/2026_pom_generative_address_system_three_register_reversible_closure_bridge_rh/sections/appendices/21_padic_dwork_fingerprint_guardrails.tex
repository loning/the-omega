\section{\texorpdfstring{$p$}{p}-进 Dwork 同余的推论：稀疏化、\texorpdfstring{$\theta$}{theta}-导数滤波与二进制指纹}\label{app:padic-dwork-guardrails}

本附录把附录 \ref{app:bridge-rh-obligations} 的 Big-Witt/\(p\)-幂同余护栏（命题 \ref{prop:big-witt-guardrails}）进一步展开为一组更“硬”的可审计推论：它们把同余约束从“多项式整体同余”细化为
\textbf{(i)}~系数稀疏化（模 \(p^k\) 下只保留 \(u^p\) 子环），
\textbf{(ii)}~\(\theta\)-导数矩的逐阶 \(p\)-进滤波缩放律，
\textbf{(iii)}~在 \(p=2\) 下产生一族稳定二进制指纹。
这些对象均可在不求谱、不枚举轨道的口径下由迹多项式直接计算并核验，适合作为 BridgeRH 的零成本否证通道与回归指纹层。

\subsection{起点：Dwork 型同余（由 Big-Witt 护栏推出）}\label{subsec:dwork-congruence-start}
在命题 \ref{prop:big-witt-guardrails} 的设定下，对任意素数 \(p\) 与 \(k\ge 1\) 有
$$
\boxed{\ a_{p^k}(u)\equiv a_{p^{k-1}}(u^p)\pmod{p^k}\ }\qquad\text{于 }\ZZ[u].
$$
下文把该同余视为唯一输入，并推导若干系数级与导数级推论。

\subsection{\texorpdfstring{$p^k$}{p^k}-稀疏化：模 \texorpdfstring{$p^k$}{p^k} 下只保留 \texorpdfstring{$u^p$}{u^p}-子环}\label{subsec:pk-sparsification}

\begin{proposition}[\texorpdfstring{$p^k$}{p^k}-稀疏化：模 \(p^k\) 下只保留 \(u^p\)-子环]\label{prop:witt-pk-sparsification-bridge}
令 \(p\) 为素数、\(k\ge 1\)。将
$$
a_{p^k}(u)=\sum_{j\ge 0}a_{p^k,j}\,u^j\in\ZZ[u]
$$
按幂展开。若满足 Dwork 型同余
$
a_{p^k}(u)\equiv a_{p^{k-1}}(u^p)\pmod{p^k}
$，
则：
\begin{enumerate}
  \item 若 \(p\nmid j\)，则 \(a_{p^k,j}\equiv 0\pmod{p^k}\)；
  \item 若 \(j=p\ell\)，则 \(a_{p^k,p\ell}\equiv a_{p^{k-1},\ell}\pmod{p^k}\)。
\end{enumerate}
等价地，在 \(\ZZ[u]/(p^k)\) 中有
$$
a_{p^k}(u)\equiv \sum_{\ell\ge 0}a_{p^{k-1},\ell}\,u^{p\ell}\in \bigl(\ZZ/p^k\ZZ\bigr)[u^p].
$$
\end{proposition}
\begin{proof}
由同余
$
a_{p^k}(u)-a_{p^{k-1}}(u^p)\in p^k\ZZ[u]
$，
而
$
a_{p^{k-1}}(u^p)=\sum_{\ell\ge 0}a_{p^{k-1},\ell}\,u^{p\ell}
$
只含 \(u^{p\ell}\) 项。逐幂次比较系数即得两条同余。证毕。
\end{proof}

\subsection{\texorpdfstring{$p=2$}{p=2} 的 Chebyshev--Dwork 同余链：不变量坐标 \texorpdfstring{$t=u+u^{-1}$}{t=u+u^{-1}}}\label{subsec:chebyshev-dwork-chain}
本节指出：当偶次迹多项式在 \(u\mapsto 1/u\) 下满足回文对称时，Dwork 同余在不变量坐标 \(t=u+u^{-1}\) 上闭合，并在二幂塔上给出一个完全\textbf{矩阵自由}的模 \(2^k\) 递推护栏。

\begin{lemma}[偶次回文迹的 \texorpdfstring{$t=u+u^{-1}$}{t=u+u^{-1}} 不变量分解]\label{lem:trace-palindrome-even-invariant-bridge}
设 \(n\ge 2\) 为偶数，且 \(a_n(u)\in\ZZ[u]\) 满足回文对称
$$
a_n(u)=u^n a_n(1/u)
$$
（作为 \(\ZZ[u,u^{-1}]\) 中的恒等式）。
令 \(t:=u+u^{-1}\)。则存在唯一多项式 \(A_n(t)\in\ZZ[t]\) 使得在 Laurent 环 \(\ZZ[u,u^{-1}]\) 中成立
$$
\boxed{\ a_n(u)=u^{n/2}\,A_n(t)\ }.
$$
\end{lemma}
\begin{proof}
回文对称等价于 \(u^{-n/2}a_n(u)\) 在 involution \(u\mapsto 1/u\) 下不变，因此
$$
u^{-n/2}a_n(u)\in \ZZ[u,u^{-1}]^{\,u\mapsto 1/u}=\ZZ[u+u^{-1}]=\ZZ[t].
$$
于是存在 \(A_n(t)\in\ZZ[t]\) 使 \(u^{-n/2}a_n(u)=A_n(t)\)，即得所示分解。唯一性来自嵌入 \(\ZZ[t]\hookrightarrow \ZZ[u,u^{-1}]\) 的单射性（把 \(t\) 视为 \(u\) 的代数不变量）。证毕。
\end{proof}

\begin{theorem}[Chebyshev--Dwork 同余链：\texorpdfstring{$p=2$}{p=2} 的 Frobenius 在不变量坐标上闭合]\label{thm:chebyshev-dwork-congruence-chain-bridge}
令 \(a_n(u)\in\ZZ[u]\) 为迹多项式序列，并令 \(t:=u+u^{-1}\)。
对每个 \(k\ge 1\)，假设偶次迹 \(a_{2^k}(u)\) 满足引理 \ref{lem:trace-palindrome-even-invariant-bridge} 的回文对称，从而存在 \(A_{2^k}(t)\in\ZZ[t]\) 使
$$
a_{2^k}(u)=u^{2^{k-1}}A_{2^k}(t).
$$
则对所有 \(k\ge 2\)，由 Dwork 型同余（\(p=2\)）
$$
a_{2^k}(u)\equiv a_{2^{k-1}}(u^2)\pmod{2^k}
$$
推出在 \(\ZZ[t]\) 内闭合的同余递推：
$$
\boxed{\ A_{2^k}(t)\equiv A_{2^{k-1}}(t^2-2)\pmod{2^k}\qquad\text{于 }\ZZ[t]\ }.
$$
\end{theorem}
\begin{proof}
由 Dwork 同余（命题 \ref{prop:big-witt-guardrails} 在 \(p=2\) 的特例）知
$
a_{2^k}(u)\equiv a_{2^{k-1}}(u^2)\pmod{2^k}
$ 于 \(\ZZ[u]\)，亦可视为在 \(\ZZ[u,u^{-1}]\) 中成立。
对偶次 \(2^k\) 与 \(2^{k-1}\) 应用引理 \ref{lem:trace-palindrome-even-invariant-bridge}，写成
$$
a_{2^k}(u)=u^{2^{k-1}}A_{2^k}(t),\qquad
a_{2^{k-1}}(u^2)=(u^2)^{2^{k-2}}A_{2^{k-1}}(u^2+u^{-2}).
$$
注意恒等式
$
u^2+u^{-2}=(u+u^{-1})^2-2=t^2-2
$，
故
$
a_{2^{k-1}}(u^2)=u^{2^{k-1}}A_{2^{k-1}}(t^2-2)
$。
将其代入 Dwork 同余并在 \(\ZZ[u,u^{-1}]\) 中消去共同因子 \(u^{2^{k-1}}\)（\(u\) 为单位），得到
$
A_{2^k}(t)\equiv A_{2^{k-1}}(t^2-2)\pmod{2^k}
$
于 \(\ZZ[t]\)。证毕。
\end{proof}

\begin{corollary}[二幂塔上的矩阵自由同余递推：单变量代入替代 \texorpdfstring{$a_{2^k}(u)$}{a_{2^k}(u)}]\label{cor:chebyshev-dwork-matrixfree-bridge}
沿用定理 \ref{thm:chebyshev-dwork-congruence-chain-bridge} 的记号。对任意 \(k\ge 2\)，在环 \((\ZZ/2^k\ZZ)[t]\) 中有
$$
\boxed{\ A_{2^k}(t)\equiv A_{2^{k-1}}(t^2-2)\ }.
$$
因此，对固定层级 \(2^k\) 的“模 \(2^k\)”计算，可将对象从“矩阵幂读出迹多项式”归约为单变量代入 \(t\mapsto t^2-2\) 与取模运算；并且该归约在二幂塔上具有对数深度（以周期长度 \(n=2^k\) 为尺度，递推步数为 \(k\)）。
\end{corollary}
\begin{proof}
该结论是定理 \ref{thm:chebyshev-dwork-congruence-chain-bridge} 在模 \(2^k\) 商环中的直接重述。证毕。
\end{proof}

\subsection{\texorpdfstring{$\theta$}{theta}-导数的 \texorpdfstring{$p$}{p}-进滤波：逐阶可审计缩放律}\label{subsec:theta-derivative-filter}

\begin{proposition}[\texorpdfstring{$\theta$}{theta}-导数的 \(p\)-进滤波：逐阶可审计缩放律]\label{prop:witt-theta-derivative-filter-bridge}
令 \(p\) 为素数、\(k\ge 1\)。设 \(a_n(u)\in\ZZ[u]\) 为迹多项式，并令 \(u=e^\theta\)。
定义在 \(\theta=0\) 处的导数矩（等价于 Euler 导数矩）
$$
A_{n}^{(r)}:=\left.\frac{\mathrm{d}^r}{\mathrm{d}\theta^r}\,a_n(e^\theta)\right|_{\theta=0}\in\ZZ,\qquad r\ge 0.
$$
若满足 Dwork 型同余 \(a_{p^k}(u)\equiv a_{p^{k-1}}(u^p)\pmod{p^k}\)，则对任意 \(r\ge 0\) 有逐阶缩放同余
$$
\boxed{\ A_{p^k}^{(r)}\equiv p^r\,A_{p^{k-1}}^{(r)}\pmod{p^k}\ }.
$$
特别地，当 \(0\le r<k\) 时得到整除下界
$$
\boxed{\ p^{\,k-r}\ \big|\ A_{p^k}^{(r)}\ }.
$$
\end{proposition}
\begin{proof}
记 Euler 导数算子 \(D:=u\,\frac{\mathrm{d}}{\mathrm{d}u}\)，则
$
\frac{\mathrm{d}}{\mathrm{d}\theta}f(e^\theta)=(Df)(u)|_{u=e^\theta}
$
并归纳得
$
A_n^{(r)}=(D^r a_n)(1)\in\ZZ
$。
对 Dwork 同余在 \(\ZZ[u]\) 中施加 \(D^r\) 仍保持模 \(p^k\)（因为 \(D\) 为 \(\ZZ\)-线性微分算子）。
另一方面，对任意 \(f\) 有恒等式
$$
D\bigl(f(u^p)\bigr)=p\,(Df)(u^p),
$$
从而 \(D^r(f(u^p))=p^r(D^rf)(u^p)\)。
将其应用于 \(f=a_{p^{k-1}}\) 并在 \(u=1\) 处取值（此时 \(u^p=1\)），得
$$
(D^r a_{p^k})(1)\equiv p^r(D^r a_{p^{k-1}})(1)\pmod{p^k},
$$
即所示同余。最后一条整除下界由 \(r<k\) 时右端显含因子 \(p^r\) 直接推出。证毕。
\end{proof}

\begin{corollary}[稳定首位残差：\texorpdfstring{$\alpha_{p,r}\in\mathbb{F}_p$}{alpha in Fp}]\label{cor:witt-theta-stable-residue-bridge}
固定素数 \(p\) 与阶 \(r\ge 1\)。当 \(k>r\) 时，由命题 \ref{prop:witt-theta-derivative-filter-bridge} 有 \(p^{k-r}\mid A_{p^k}^{(r)}\)，从而可定义
$$
\alpha_{p,r}:=\left[p^{-(k-r)}A_{p^k}^{(r)}\right]\bmod p\ \in \ZZ/p\ZZ\cong\mathbb{F}_p.
$$
则 \(\alpha_{p,r}\) 与 \(k\) 的选择无关（只要 \(k>r\)）。
\end{corollary}
\begin{proof}
由命题 \ref{prop:witt-theta-derivative-filter-bridge} 的缩放同余，
$
A_{p^k}^{(r)}=p^rA_{p^{k-1}}^{(r)}+p^k t
$
对某个 \(t\in\ZZ\) 成立。两边除以 \(p^{k-r}\) 并模 \(p\) 取剩余，由 \(r\ge 1\) 知余项消失，得到
$
p^{-(k-r)}A_{p^k}^{(r)}\equiv p^{-(k-1-r)}A_{p^{k-1}}^{(r)}\pmod p
$，
迭代即得稳定性。证毕。
\end{proof}

\begin{corollary}[\texorpdfstring{$p=2$}{p=2} 的稳定奇偶指纹：由 \texorpdfstring{$\theta$}{theta}-导数生成一族无限二进制不变量]\label{cor:witt-theta-binary-fingerprint-bridge}
取 \(p=2\) 并固定阶 \(r\ge 1\)。对任意 \(k\ge r+1\)，由命题 \ref{prop:witt-theta-derivative-filter-bridge} 的估值下界 \(2^{k-r}\mid A_{2^k}^{(r)}\) 可定义整数
$$
B_k^{(r)}:=2^{-(k-r)}A_{2^k}^{(r)}\in\ZZ.
$$
则其奇偶性从层级 \(k=r+1\) 起永久稳定：
$$
\boxed{\ B_{k+1}^{(r)}\equiv B_{k}^{(r)}\pmod 2\qquad(\forall k\ge r+1)\ }.
$$
因此得到一族可审计的二进制指纹
$$
\varepsilon_r:=B_{k}^{(r)}\bmod 2\in\{0,1\}\qquad(r\ge 1),
$$
其与 \(k\) 的选择无关（只要 \(k\ge r+1\)）。
\end{corollary}
\begin{proof}
由推论 \ref{cor:witt-theta-stable-residue-bridge}（取 \(p=2\)），当 \(k>r\) 时
$
2^{-(k-r)}A_{2^k}^{(r)}\bmod 2
$
与 \(k\) 无关。取其整数代表即得奇偶稳定性与指纹定义。证毕。
\end{proof}

\subsection{可选：在 \texorpdfstring{$u^p=1$}{u^p=1} 下的塔式锁定}\label{subsec:root-of-unity-lock}

\begin{corollary}[\texorpdfstring{$u^p=1$}{u^p=1} 的塔式锁定：扭曲迹被推回无扭曲层]\label{cor:witt-root-of-unity-tower-lock-bridge}
令 \(p\) 为素数、\(k\ge 1\)，并令 \(R\) 为交换环。若 \(u\in R\) 满足 \(u^p=1\)，则
$$
a_{p^k}(u)\equiv a_{p^{k-1}}(1)\pmod{p^k}\qquad\text{于}\ R.
$$
更一般地，对任意 \(1\le \ell\le k\) 有
$$
\boxed{\ a_{p^k}(u)\equiv a_{p^{k-\ell}}(1)\pmod{p^\ell}\ }\qquad\text{于}\ R.
$$
\end{corollary}
\begin{proof}
由 Dwork 型同余
$
a_{p^k}(u)\equiv a_{p^{k-1}}(u^p)\pmod{p^k}
$
并用 \(u^p=1\) 得第一式。
对第二式只需在指数链上迭代 \(\ell\) 次：每一步把 \(u^{p^j}\) 推回到 \(1\)，同时模数降一阶。证毕。
\end{proof}

\begin{corollary}[阶 \texorpdfstring{$p$}{p} 圆分扭曲的完全不可见性]\label{cor:witt-cyclotomic-invisibility-bridge}
令 \(p\) 为素数、\(k\ge 1\)，并令 \(R\) 为交换环。若 \(u\in R\) 满足 \(u^p=1\)，则
$$
\boxed{\ a_{p^k}(u)\equiv a_{p^k}(1)\pmod{p^k}\qquad\text{于}\ R\ }.
$$
\end{corollary}
\begin{proof}
由推论 \ref{cor:witt-root-of-unity-tower-lock-bridge} 有
$
a_{p^k}(u)\equiv a_{p^{k-1}}(1)\pmod{p^k}
$。
另一方面，对 Dwork 同余取 \(u=1\) 得
$
a_{p^k}(1)\equiv a_{p^{k-1}}(1)\pmod{p^k}
$。
两式相减即得结论。证毕。
\end{proof}
