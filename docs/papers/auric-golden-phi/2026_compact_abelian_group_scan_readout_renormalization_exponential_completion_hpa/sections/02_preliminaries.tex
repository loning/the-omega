\section{Preliminaries: Dynamics, Discrepancy, and Symbolic Readout}

\subsection{Compact Abelian Groups and Translation Systems}

Let $G$ be a compact Abelian group and $\mu$ its Haar probability measure. For $g\in G$, define the translation
\[
T_g(x)=x+g,\qquad x\in G.
\]
Then $(G,\mu,T_g)$ is a measure-preserving system. Its ergodic and uniquely ergodic properties are controlled by algebraic properties of $g$.

\subsection{Density, Unique Ergodicity, and Window Frequencies}

Let $G=\TT^d=\RR^d/\ZZ^d$, and choose a step vector $\alpha\in\RR^d$. The orbit is
\[
x_n=x_0+n\alpha\pmod{1}.
\]
If $(1,\alpha_1,\dots,\alpha_d)$ is linearly independent over $\QQ$, the orbit is jointly equidistributed, and empirical measures converge to Haar measure. For window readout $E=\ind_W$,
\[
\lim_{N\to\infty}\frac{1}{N}\sum_{n=0}^{N-1}\ind_W(x_n)=\mu(W)
\]
holds under suitable regularity (e.g., zero-measure window boundary).

\subsection{Discrepancy and Prefix Uniformity at Finite Sample Length}

In one dimension, star discrepancy is defined by
\[
D_N^*(x_0,\dots,x_{N-1})=\sup_{u\in[0,1]}\left|\frac{1}{N}\#\{0\le n<N:x_n<u\}-u\right|.
\]
Discrepancy provides a deterministic certificate for finite-sample errors and connects to Diophantine approximation and continued-fraction structure.

\subsection{Rotation Coding, Sturmian Sequences, and Substitution Systems}

For $x_n=x_0+n\alpha \bmod 1$ and interval window $W=[0,\alpha)$, define binary readout
\[
u_n=\ind_{[0,\alpha)}(x_n)\in\{0,1\}.
\]
When $\alpha\notin\QQ$, the resulting sequence is Sturmian with minimal complexity $p(n)=n+1$. For $\alpha=\varphi^{-1}$, the sequence is equivalent (up to shift/complement) to the Fibonacci word and is generated by the Fibonacci substitution
\[
0\mapsto 01,\quad 1\mapsto 0,
\]
whose Perron--Frobenius dominant eigenvalue is $\varphi$.

\subsection{Pontryagin Duality and Fourier Expansion}

For a compact Abelian group $G$, its dual group $\widehat G=\mathrm{Hom}(G,S^1)$ provides a character basis. For $f\in L^2(G)$, the Fourier coefficients
\[
\widehat f(\chi)=\int_G f(x)\overline{\chi(x)}\,\dd\mu(x)
\]
carry spectral information, and they allow concept readout and discrepancy to be quantified in a common analytic representation.
