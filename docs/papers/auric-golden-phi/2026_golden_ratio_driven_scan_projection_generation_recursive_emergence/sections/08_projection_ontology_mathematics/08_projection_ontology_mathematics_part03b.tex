\subsubsection{二阶碰撞矩与 \texorpdfstring{$\mathrm{R\acute{e}nyi}$}{Renyi}--2 投影熵率}\label{subsec:pom-s2}
最大纤维多重度刻画“最坏空间缺陷”，但在统计意义上更稳健的指标来自碰撞矩（\(\ell^2\) 质量）。它给出一个可与在线延迟 \(\delta_{\min}=3\) 并列的、纯由折叠门 \(P_Z\) 决定的指数尺度常数。

\begin{lemma}[碰撞等价于模 \texorpdfstring{$F_{m+2}$}{F\_{m+2}} 同余]\label{lem:pom-fold-congruence}
设 \(\omega,\omega'\in\Omega_m=\{0,1\}^m\)，并定义窗口值
$$
N(\omega):=\sum_{k=1}^m \omega_k F_{k+1}.
$$
则有等价关系
$$
\Fold_m(\omega)=\Fold_m(\omega')
\quad\Longleftrightarrow\quad
N(\omega)\equiv N(\omega')\pmod{F_{m+2}}.
$$
\end{lemma}
\begin{proof}
记 \(x=\Fold_m(\omega)\in X_m\)、\(x'=\Fold_m(\omega')\in X_m\)。由引理 \ref{lem:pom-one-bit}（单隐藏位分解）存在唯一 \(b(\omega),b(\omega')\in\{0,1\}\) 使
$$
N(\omega)=V_m(x)+b(\omega)F_{m+2},\qquad
N(\omega')=V_m(x')+b(\omega')F_{m+2}.
$$
若 \(x=x'\)，则两式相减得 \(N(\omega)-N(\omega')=(b(\omega)-b(\omega'))F_{m+2}\)，从而同余成立。
反之，若 \(N(\omega)\equiv N(\omega')\ (\mathrm{mod}\ F_{m+2})\)，则两者在模 \(F_{m+2}\) 下的余数相同。又由于 \(V_m(\Fold_m(\cdot))\) 正是 Zeckendorf 规范形在前 \(m\) 位对应的值（定义 \ref{def:pom-window-value}），该余数唯一确定折叠后的稳定类型 \(x\in X_m\)，因此 \(x=x'\)，即 \(\Fold_m(\omega)=\Fold_m(\omega')\)。证毕。
\end{proof}

\begin{proposition}[折叠作为模同余投影的规范截面]\label{prop:pom-fold-as-section}
令 \(r(\omega):=N(\omega)\bmod F_{m+2}\in\{0,1,\dots,F_{m+2}-1\}\)。
对每个 \(0\le r\le F_{m+2}-1\)，令 \(Z(r)\in X_\infty\) 为 \(r\) 的 Zeckendorf 编码，并记截断 \(\pi_m: X_\infty\to X_m\)（第 6 节）。定义规范截面
\[
s_m(r):=\pi_m(Z(r))\in X_m.
\]
则对任意 \(\omega\in\Omega_m\) 都有
\[
\boxed{\;\Fold_m(\omega)=s_m\bigl(r(\omega)\bigr).\;}
\]
亦即 \(\Fold_m\) 因子化为“模 \(F_{m+2}\) 的线性投影”与一个规范截面的复合。
\end{proposition}
\begin{proof}
由引理 \ref{lem:pom-fold-congruence}，\(\Fold_m(\omega)\) 只依赖于 \(r(\omega)=N(\omega)\bmod F_{m+2}\)。
另一方面，\(V_m:X_m\to\{0,1,\dots,F_{m+2}-1\}\) 为双射（引理 \ref{lem:zeckendorf-range-bijection}），并且按定义 \(V_m(s_m(r))=r\)。
因此当 \(r(\omega)=r\) 时，\(\Fold_m(\omega)\) 必等于该唯一满足 \(V_m(x)=r\) 的稳定类型 \(x=s_m(r)\)。证毕。
\end{proof}

\begin{proposition}[\texorpdfstring{$q$}{q}-重碰撞的同余矩表达]\label{prop:pom-moment-congruence-q}
对任意整数 \(q\ge 2\)，定义
$$
S_q(m):=\sum_{x\in X_m} d_m(x)^q.
$$
并令
$$
c_m(r):=\#\{\omega\in\Omega_m:\ N(\omega)\equiv r\ (\mathrm{mod}\ F_{m+2})\},\qquad r\in\ZZ/F_{m+2}\ZZ.
$$
则有恒等式
$$
\boxed{\ 
S_q(m)=\sum_{r\in\ZZ/F_{m+2}\ZZ} c_m(r)^q.
\ }
$$
\end{proposition}
\begin{proof}
记
$$
R_m^{(q)}:=\{(\omega^{(1)},\dots,\omega^{(q)})\in\Omega_m^q:\ \Fold_m(\omega^{(1)})=\cdots=\Fold_m(\omega^{(q)})\}.
$$
由定义立得 \(|R_m^{(q)}|=\sum_{x\in X_m}d_m(x)^q=S_q(m)\)。
另一方面，由引理 \ref{lem:pom-fold-congruence}，对任意两条窗口 \(\omega,\omega'\) 有
\(
\Fold_m(\omega)=\Fold_m(\omega')\iff N(\omega)\equiv N(\omega')\ (\mathrm{mod}\ F_{m+2})
\)。
因此 \((\omega^{(1)},\dots,\omega^{(q)})\in R_m^{(q)}\) 当且仅当
$$
N(\omega^{(1)})\equiv \cdots \equiv N(\omega^{(q)})\pmod{F_{m+2}}.
$$
按共同余数类 \(r\in\ZZ/F_{m+2}\ZZ\) 分组计数，得到
$$
|R_m^{(q)}|
=\sum_{r\in\ZZ/F_{m+2}\ZZ}\#\{\omega^{(1)}:\ N(\omega^{(1)})\equiv r\}\cdots \#\{\omega^{(q)}:\ N(\omega^{(q)})\equiv r\}
=\sum_{r}c_m(r)^q.
$$
与 \(|R_m^{(q)}|=S_q(m)\) 合并即得结论。证毕。
\end{proof}

\begin{proposition}[任意 \texorpdfstring{$q$}{q} 的傅里叶 moment-kernel 闭式]\label{prop:pom-moment-fourier-q}
令 $F:=F_{m+2}$，并在加法群 $G:=\ZZ/F\ZZ$ 上定义同余分布
$$
c_m(r):=\#\{\omega\in\Omega_m:\ N(\omega)\equiv r\ (\mathrm{mod}\ F)\},\qquad r\in G.
$$
取 $\omega_F:=e^{2\pi i/F}$，定义离散傅里叶变换
$$
\widehat c_m(t):=\sum_{r\in G} c_m(r)\,\omega_F^{tr},\qquad t\in G.
$$
则对任意整数 $q\ge 2$ 有显式公式
$$
\boxed{\;
S_q(m)=\frac{1}{F^{\,q-1}}
\sum_{t_1,\dots,t_{q-1}\in G}
\widehat c_m(t_1)\cdots \widehat c_m(t_{q-1})\,
\widehat c_m\!\Bigl(-\sum_{j=1}^{q-1}t_j\Bigr)\;}
$$
并且 $\widehat c_m$ 具有完全因子化形式
$$
\boxed{\;
\widehat c_m(t)=\sum_{\omega\in\Omega_m}\omega_F^{tN(\omega)}
=\prod_{k=1}^m\bigl(1+\omega_F^{tF_{k+1}}\bigr)\; }.
$$
\end{proposition}
\begin{proof}
因子化式由
$$
\widehat c_m(t)=\sum_{\omega_1,\dots,\omega_m\in\{0,1\}}\ \prod_{k=1}^m \omega_F^{t\omega_kF_{k+1}}
=\prod_{k=1}^m\sum_{\omega_k\in\{0,1\}} \omega_F^{t\omega_kF_{k+1}}
$$
直接得到。

为得一般 $q$ 的闭式，先用傅里叶反演
$$
c_m(r)=\frac{1}{F}\sum_{t\in G}\widehat c_m(t)\,\omega_F^{-tr}.
$$
代入 $S_q(m)=\sum_{r\in G}c_m(r)^q$ 并展开：
$$
S_q(m)
=\frac{1}{F^q}\sum_{t_1,\dots,t_q\in G}\widehat c_m(t_1)\cdots\widehat c_m(t_q)\sum_{r\in G}\omega_F^{-(t_1+\cdots+t_q)r}.
$$
由角色正交性，
$$
\sum_{r\in G}\omega_F^{-(t_1+\cdots+t_q)r}
=
\begin{cases}
F,& t_1+\cdots+t_q\equiv 0\ (\mathrm{mod}\ F),\\
0,& \text{否则}.
\end{cases}
$$
因此
$$
S_q(m)=\frac{1}{F^{q-1}}\sum_{\substack{t_1,\dots,t_q\in G\\ t_1+\cdots+t_q\equiv 0}}
\widehat c_m(t_1)\cdots\widehat c_m(t_q).
$$
再以 $t_q=-(t_1+\cdots+t_{q-1})$ 消去约束即得所述公式。证毕。
\end{proof}

\begin{definition}[折叠碰撞矩]\label{def:pom-s2}
定义二阶碰撞和
$$
S_2(m):=\sum_{x\in X_m} d_m(x)^2,
\qquad d_m(x):=\bigl|\Fold_m^{-1}(x)\bigr|.
$$
等价地，从 \(\Omega_m\) 上均匀独立抽取两条窗口 \(\omega,\omega'\)，则
$$
\mathbb{P}\bigl[\Fold_m(\omega)=\Fold_m(\omega')\bigr]=\frac{S_2(m)}{4^m}.
$$
\end{definition}

\begin{proposition}[碰撞核的三态表示与三阶递推]\label{prop:pom-s2-recurrence}
令碰撞关系
$$
R_m:=\{(\omega,\omega')\in\Omega_m\times\Omega_m:\Fold_m(\omega)=\Fold_m(\omega')\}.
$$
则存在一个三态的非负整数矩阵（可视为“碰撞核”的邻接矩阵）
$$
A_2:=
\begin{pmatrix}
0&0&1\\
0&1&1\\
2&1&1
\end{pmatrix},
$$
使得对一切 \(m\ge 2\),
$$
S_2(m)=|R_m|
=2\,\mathbf{1}^{\mathsf T}A_2^{\,m-2}\mathbf{1},
\qquad \mathbf{1}:=(1,1,1)^{\mathsf T}.
$$
从而 \(S_2(m)\) 满足精确递推（\(m\ge 5\)）
$$
S_2(m)=2S_2(m-1)+2S_2(m-2)-2S_2(m-3),
$$
并具有初值 \(S_2(2)=6,S_2(3)=14,S_2(4)=36\)。
\end{proposition}
\begin{proof}
由引理 \ref{lem:pom-fold-congruence}，有
$$
|R_m|
=\#\{(\omega,\omega'):\ N(\omega)\equiv N(\omega')\ (\mathrm{mod}\ F_{m+2})\}
=\sum_{r\in\ZZ/F_{m+2}\ZZ} c_m(r)^2,
$$
其中 \(c_m(r):=\#\{\omega\in\Omega_m:\ N(\omega)\equiv r\ (\mathrm{mod}\ F_{m+2})\}\)。
因此 \(S_2(m)\) 完全由“模 \(F_{m+2}\) 的值同余计数”决定，可视为一个离散有限状态计数问题的平方和（碰撞核的 \(\ell^2\) 质量）。

接着给出一个\textbf{可审计构造}：三态非负 realization 可取上式所示的 \(A_2\)。该矩阵并非任意猜测——它可以从小 \(m\) 的枚举数据出发自动恢复：先由 \(S_2(2),\dots,S_2(7)\) 解出最小三阶递推系数，再在小整数矩阵类中搜索满足
\(
S_2(m)=2\,\mathbf{1}^{\mathsf T}A^{m-2}\mathbf{1}
\)
的非负 \(3\times 3\) realization，得到的规范解即为此 \(A_2\)（可复现脚本 \texttt{scripts/exp\_fold\_collision\_kernel\_a2\_search.py}）。
为核对其正确性，直接计算得
$$
2\,\mathbf{1}^{\mathsf T}A_2^{\,0}\mathbf{1}=6,\qquad
2\,\mathbf{1}^{\mathsf T}A_2^{\,1}\mathbf{1}=14,\qquad
2\,\mathbf{1}^{\mathsf T}A_2^{\,2}\mathbf{1}=36,
$$
与 \(S_2(2),S_2(3),S_2(4)\) 一致。另一方面，\(A_2\) 的特征多项式为
$$
\det(\lambda I-A_2)=\lambda^3-2\lambda^2-2\lambda+2,
$$
因此对任意向量序列 \(u_m:=\mathbf{1}^{\mathsf T}A_2^{m}\mathbf{1}\)（\(m\ge 0\)）满足由 Cayley--Hamilton 定理诱导的三阶线性递推。乘以常数 \(2\) 并作指标平移即可得到 \(S_2(m)\) 的递推式与初值。

最后，\(\mathbf{1}\) 与 \(A_2\) 的取法给出一个完全显式、可审计的三态“碰撞核”实现；其数值、递推与 \(A_2\) realization 的一致性可由脚本 \texttt{scripts/exp\_fold\_collision\_s2.py} 与 \texttt{scripts/exp\_fold\_collision\_kernel\_a2\_search.py} 逐项核验。证毕。
\end{proof}

\begin{proposition}[\texorpdfstring{$S_2(m)$}{S2(m)} 的 Plancherel 表达与角色因子化]\label{prop:pom-s2-plancherel}
令 \(G_m:=\ZZ/F_{m+2}\ZZ\)，并取其标准角色
\(
\chi_j(r):=\exp(2\pi i jr/F_{m+2})
\)
（\(j\in G_m\)）。对同余计数 \(c_m(r)=\#\{\omega\in\Omega_m:\ N(\omega)\equiv r\}\) 定义离散傅里叶变换
\[
\widehat c_m(j):=\sum_{r\in G_m} c_m(r)\,\chi_j(r).
\]
则二阶碰撞矩满足严格恒等式
\[
\boxed{\;
S_2(m)=\sum_{r\in G_m}c_m(r)^2=\frac{1}{|G_m|}\sum_{j\in G_m}\bigl|\widehat c_m(j)\bigr|^2\;}
\]
并且 \(\widehat c_m(j)\) 具有完全因子化形式
\[
\boxed{\;
\widehat c_m(j)=\sum_{\omega\in\Omega_m}\chi_j\!\bigl(N(\omega)\bigr)
=\prod_{k=1}^m\Bigl(1+\chi_j(F_{k+1})\Bigr)
=\prod_{k=1}^m\left(1+\exp\!\frac{2\pi i j F_{k+1}}{F_{m+2}}\right).\;}
\]
从而
\[
S_2(m)
=\frac{1}{F_{m+2}}\sum_{j=0}^{F_{m+2}-1}\ \prod_{k=1}^m\left|1+\exp\!\frac{2\pi i j F_{k+1}}{F_{m+2}}\right|^2.
\]
\end{proposition}
\begin{proof}
第一式是有限阿贝尔群上的 Plancherel 恒等式。这里给出与本节记号一致的直接推导：由反演公式
\(
c_m(r)=\frac{1}{|G_m|}\sum_{j}\widehat c_m(j)\,\overline{\chi_j(r)}
\)
与角色正交性
\(
\sum_{r}\chi_j(r)\overline{\chi_{j'}(r)}=|G_m|\,\mathbf{1}_{j=j'}
\)
立得
\[
\sum_{r}c_m(r)^2
=\frac{1}{|G_m|}\sum_{j}|\widehat c_m(j)|^2.
\]

对因子化：由定义
\[
\widehat c_m(j)
=\sum_{r}\sum_{\substack{\omega\in\Omega_m\\N(\omega)\equiv r}}\chi_j(r)
=\sum_{\omega\in\Omega_m}\chi_j\!\bigl(N(\omega)\bigr).
\]
再用窗口值的线性分解 \(N(\omega)=\sum_{k=1}^m \omega_k F_{k+1}\)（定义 \ref{def:pom-window-value}）与角色的乘法性，
\[
\chi_j\!\bigl(N(\omega)\bigr)
=\prod_{k=1}^m \chi_j(F_{k+1})^{\omega_k},
\]
对每一位 \(\omega_k\in\{0,1\}\) 求和即可得到
\(
\widehat c_m(j)=\prod_{k=1}^m(1+\chi_j(F_{k+1}))
\)。
最后一式由 \(|G_m|=F_{m+2}\) 与 \(\chi_j(F_{k+1})=\exp(2\pi i jF_{k+1}/F_{m+2})\) 代入得到。证毕。
\end{proof}

\begin{remark}[最大非平凡傅里叶系数证书与临界共振的阻碍]\label{rem:pom-max-fourier-obstruction}
把同余计数归一化为概率分布 \(p_m(r):=c_m(r)/2^m\)，并记 \(\widehat p_m(j):=\widehat c_m(j)/2^m\)。
则命题 \ref{prop:pom-s2-plancherel} 等价于
\[
\sum_{r\in G_m}p_m(r)^2=\frac{1}{F_{m+2}}\sum_{j\in G_m}\bigl|\widehat p_m(j)\bigr|^2.
\]
若令 \(A_m:=\max_{j\in G_m,\ j\neq 0}|\widehat p_m(j)|\)，则立刻得到
\[
\sum_{r\in G_m}p_m(r)^2\le \frac{1}{F_{m+2}}\Bigl(1+(F_{m+2}-1)A_m^2\Bigr).
\]
然而附录 \ref{app:fold-multiplicity} 的临界共振定理 \ref{thm:fold-critical-resonance-constant} 证明沿共振频率 \(j=F_m,F_{m+1}\) 有
\(
|\widehat p_m(j)|\to C_\varphi>0
\)
（并由推论 \ref{cor:fold-critical-resonance-gap} 给出相应的碰撞下界）。
因此在 raw Fold 上，\(A_m\) 不可能随 \(m\) 指数衰减：若要在傅里叶侧推进“碰撞趋均匀/谱收缩”，需要控制的是频率\emph{平均能量}或对 edge/bulk 分解后的 bulk 大偏差，而非仅控制最大系数（另见注 \ref{rem:fold-edge-nondecay}）。
\end{remark}

\begin{remark}[严格交换的“Fold--Fourier--Parseval”桥包（交换图）]\label{rem:fold-fourier-parseval-bridge}
在命题 \ref{prop:pom-moment-fourier-q}--\ref{prop:pom-s2-plancherel} 的记号下，令 \(F:=F_{m+2}\)、\(G_m:=\ZZ/F\ZZ\)。
由附录定理 \ref{thm:fold-congruence-mod-semirings}，有限分辨率下的 Fold 商对象 \(\Omega_m/\!\equiv_m\) 可被识别为显式循环环 \(G_m\)，从而同余计数 \(c_m\) 可视为群代数 \(\CC[G_m]\) 中的类函数。
记 \(\delta_a\in\CC[G_m]\) 为 Dirac 元，卷积记为 \(\ast\)。
则 \(c_m\) 具有“空间侧卷积叠加”的显式分解：
\[
c_m=(\delta_0+\delta_{F_2})\ast(\delta_0+\delta_{F_3})\ast\cdots\ast(\delta_0+\delta_{F_{m+1}})\qquad\text{in }\CC[G_m],
\]
而离散 Fourier 变换 \(\mathcal{F}:\CC[G_m]\to\CC[\widehat G_m]\) 将卷积严格对角化为点乘 \(\widehat{f\ast g}=\widehat f\cdot\widehat g\)，因此得到完全因子化（Fold 支路的 “Euler-product 形”）：
\[
\widehat c_m(t)=\prod_{k=1}^{m}\bigl(1+\omega_F^{tF_{k+1}}\bigr).
\]
另一方面，Plancherel/Parseval 恒等式把空间侧 \(\ell^2\) 质量与频域能量严格对齐；写成交换图即
\[
\begin{CD}
c_m=(\delta_0+\delta_{F_2})\ast(\delta_0+\delta_{F_3})\ast\cdots\ast(\delta_0+\delta_{F_{m+1}}) @>\mathcal{F}>> \widehat c_m(t)=\prod_{k=1}^{m}(1+\omega_F^{tF_{k+1}})\\
@V{\|\,\cdot\,\|_2^2}VV @VV{\frac{1}{F}\|\,\cdot\,\|_2^2}V\\
S_2(m)=\sum_{r\in G_m}c_m(r)^2 @= S_2(m)
\end{CD}
\]
其中 \(\|f\|_2^2:=\sum_{r\in G_m}|f(r)|^2\) 取计数测度。
特别地，若记
\[
\mathsf{Col}_m:=\mathbb{P}[\Fold_m(\omega)=\Fold_m(\omega')]=\frac{S_2(m)}{4^m}
\qquad(\text{见定义 \ref{def:pom-s2}}),
\]
则由 \(\widehat c_m(0)=\sum_r c_m(r)=2^m\) 得到一个“共同标尺”（非平凡频谱能量 \(\Longleftrightarrow\) 碰撞缺口）：
\[
\boxed{\;
F_{m+2}\,\mathsf{Col}_m-1=\frac{1}{4^m}\sum_{t\ne 0}\bigl|\widehat c_m(t)\bigr|^2\; }.
\]
因此在 Fold 支路上，“空间/时间对偶”可被严格落成：空间侧的卷积（串行叠加）\(\Longleftrightarrow\) 频域侧的点乘（可分解的乘法链），并由 Parseval 把二者钉死到同一个可审计标尺上。
\end{remark}

\begin{lemma}[三阶递推的最小性]\label{lem:pom-s2-minimal-order}
序列 \(\{S_2(m)\}_{m\ge 2}\) 不满足任何二阶常系数线性递推。因而命题 \ref{prop:pom-s2-recurrence} 的三阶递推在“线性递推阶数”意义下是最小的。
\end{lemma}
\begin{proof}
反设存在常数 \(\alpha,\beta\) 使对所有 \(m\) 有 \(S_2(m)=\alpha S_2(m-1)+\beta S_2(m-2)\)。
由初值 \(S_2(2)=6,S_2(3)=14,S_2(4)=36,S_2(5)=88\) 解线性方程组
$$
36=\alpha\cdot 14+\beta\cdot 6,\qquad
88=\alpha\cdot 36+\beta\cdot 14
$$
得 \(\alpha=6/5,\ \beta=16/5\)。于是
$$
S_2(6)=\alpha S_2(5)+\beta S_2(4)=\frac65\cdot 88+\frac{16}{5}\cdot 36=\frac{1104}{5}=220.8,
$$
与 \(S_2(6)=220\)（命题 \ref{prop:pom-s2-recurrence} 或脚本 \texttt{scripts/exp\_fold\_collision\_s2.py}）矛盾。故不存在二阶递推。证毕。
\end{proof}

\begin{remark}[最小 realization 维数与非负核]\label{rem:pom-s2-minimal-realization}
由引理 \ref{lem:pom-s2-minimal-order}，\(\{S_2(m)\}\) 的最小线性递推阶数为 \(3\)，因此任何线性状态空间表示 \(S_2(m)=u^{\mathsf T}B^{m}v\) 的最小维数至少为 \(3\)。命题 \ref{prop:pom-s2-recurrence} 给出的三态矩阵 \(A_2\) 因而在维数意义下是最小的；并且它是一个\emph{非负整数} realization（可视为一张三态有限图的邻接矩阵），从而将 \(S_2\) 的指数尺度完全纳入 Perron--Frobenius/\(\zeta\) 的统一接口。
\end{remark}

\begin{remark}[Hankel 最小实现与 \texorpdfstring{$A_2$}{A2} 的非负相似标准形]\label{rem:pom-s2-hankel}
将序列重标为 \(a_n:=S_2(n+2)/2\)（\(n\ge 0\)）。由递推阶数为 \(3\) 可知 Hankel 矩阵族的秩为 \(3\)，因此存在一个三维的\textbf{最小线性实现}（加权自动机/线性表示的标准口径）：取
$$
H_0=(a_{i+j})_{0\le i,j\le 2},\qquad H_1=(a_{i+j+1})_{0\le i,j\le 2},
$$
则在 \(H_0\) 可逆时可定义“移位实现矩阵”
$$
A:=H_0^{-1}H_1,
$$
从而 \(a_n=u^{\mathsf T}A^{n}v\)（某些 \(u,v\)）给出最小 realization。该 Hankel 标准形 \(A\) 一般包含符号（例如出现负系数），但它与非负整数核 \(A_2\) 相似：存在行列式为 \(\pm 1\) 的整矩阵 \(P\) 使
$$
P^{-1}AP=A_2.
$$
上述 Hankel 构造与 unimodular 相似变换可由脚本 \texttt{scripts/exp\_fold\_collision\_hankel\_realization.py} 一键复现，从而把“最小 realization（可证明）”与“非负核（可解释为有限图）”严格对齐。
\end{remark}

\begin{theorem}[Hankel 秩定理（加权自动机的线性 Myhill--Nerode 口径）]\label{thm:hankel-rank-minimal}
设 \(k\) 为域（例如 \(\mathbb{Q}\) 或 \(\mathbb{R}\)）。给定一个序列 \(a=(a_n)_{n\ge 0}\subset k\)，其 Hankel 矩阵定义为
$$
H(a):=(a_{i+j})_{i,j\ge 0}.
$$
称 \((u,A,v)\) 为 \(a\) 的一个线性表示（linear representation），若 \(u,v\in k^d\)、\(A\in k^{d\times d}\)，且对一切 \(n\ge 0\) 有
$$
a_n=u^{\mathsf T}A^n v.
$$
则有
$$
\mathrm{rank}\bigl(H(a)\bigr)=\min\{d:\exists\ (u,A,v)\text{ 给出维数为 }d\text{ 的线性表示}\}.
$$
等价地，\(\mathrm{rank}(H(a))\) 等于生成该序列的最小加权自动机（有理级数）的状态数（见 \cite{BerstelReutenauer1988RationalSeries}）。
\end{theorem}

\begin{lemma}[Hankel 窗口行列式非零：秩至少为 \texorpdfstring{$3$}{3}]\label{lem:pom-s2-hankel-det}

因此 Hankel 秩至少为 \(3\)，从而任何线性 realization 的最小维数至少为 \(3\)。
\end{lemma}
\begin{proof}

的前五项为 \((3,7,18,44,110)\)。直接代入计算 \(\det(H_0)\) 得 \(-2\)。证毕。
\end{proof}

\begin{theorem}[Hankel 秩恰为 \texorpdfstring{$3$}{3}，且三态为最小碰撞核]\label{thm:pom-s2-rank-exact}
令 \(a_n:=S_2(n+2)/2\) 并以 \(H=(a_{i+j})_{i,j\ge 0}\) 表示其 Hankel 矩阵。则：
\begin{enumerate}
  \item \(\mathrm{rank}(H)=3\)。
  \item \(a_n\) 的最小线性表示（等价于最小加权自动机）的状态维数为 \(3\)（由定理 \ref{thm:hankel-rank-minimal}）。
  \item 命题 \ref{prop:pom-s2-recurrence} 给出的三态非负整数核 \(A_2\) 在维数意义下是最小的碰撞核 realization。
\end{enumerate}
\end{theorem}
\begin{proof}
\textbf{(1)} 由引理 \ref{lem:pom-s2-hankel-det}，存在一个 \(3\times 3\) Hankel 主子式 \(\det(H_0)\neq 0\)，故 \(\mathrm{rank}(H)\ge 3\)。

$$
因此 Hankel 矩阵的行/列空间包含于 \(\mathbb{R}^3\) 的一个 3 维线性表示所张成的子空间，从而 \(\mathrm{rank}(H)\le 3\)。合并得 \(\mathrm{rank}(H)=3\)。

\textbf{(2)} 由定理 \ref{thm:hankel-rank-minimal}，\(\mathrm{rank}(H)\) 等于最小线性表示维数；结合 (1) 的 \(\mathrm{rank}(H)=3\)，得最小维数为 \(3\)。

\begin{corollary}[碰撞指数与 \texorpdfstring{$h_2$}{h2}]\label{cor:pom-s2-asymptotic}
令 \(r_2\) 为多项式
$$
x^3-2x^2-2x+2=0
$$
的最大实根（\(r_2\approx 2.481194304092\)）。则

S_2(m)=\Theta(r_2^m),
\qquad
\mathbb{P}\bigl[\Fold_m(\omega)=\Fold_m(\omega')\bigr]
=\Theta\!\left(\left(\frac{r_2}{4}\right)^m\right).
$$
进一步，令 \(\pi_m(x):=d_m(x)/2^m\) 为均匀微态经折叠后的输出分布，则其 \(\mathrm{R\acute{e}nyi}\)--2 投影熵率存在且为
$$
h_2:=\lim_{m\to\infty}\frac{1}{m}\Bigl(-\log\sum_{x\in X_m}\pi_m(x)^2\Bigr)
=\log\!\left(\frac{4}{r_2}\right)\approx 0.477554342632.
$$
\end{corollary}
\begin{proof}
由命题 \ref{prop:pom-s2-recurrence}，\(S_2(m)\) 的生成函数具有分母 \(1-2z-2z^2+2z^3\)，其收敛半径由该分母的最小模零点决定，等价于递推特征多项式的最大实根 \(r_2\)。因此 \(S_2(m)=\Theta(r_2^m)\)。又
$$
\sum_{x\in X_m}\pi_m(x)^2=\sum_x \frac{d_m(x)^2}{4^m}=\frac{S_2(m)}{4^m},
$$
代入渐近即可得 \(h_2=\log(4/r_2)\)。证毕。
\end{proof}

\begin{corollary}[二阶意义下的“近似均匀覆盖”：偏离常数的指数刻画]\label{cor:pom-renyi2-near-uniform}
令 \(\pi_m(x)=d_m(x)/2^m\) 为折叠输出分布，并令 \(u_m(x)\equiv 1/|X_m|=1/F_{m+2}\) 为 \(X_m\) 上的均匀分布。定义二阶碰撞比（亦即 Rényi--2 相对熵的指数化量）
\[
R_m
:=\frac{\sum_{x\in X_m}\pi_m(x)^2}{\sum_{x\in X_m}u_m(x)^2}
=|X_m|\sum_{x\in X_m}\pi_m(x)^2
=\frac{F_{m+2}\,S_2(m)}{4^m}.
\]
则有指数率
\[
\boxed{\;
\lim_{m\to\infty}\frac{1}{m}\log R_m
=\log\!\left(\frac{\varphi\,r_2}{4}\right)
\approx 3.657482\times 10^{-3}\; }.
\]
因此二阶意义下，\(\pi_m\) 偏离均匀分布的强度以基数 \(\varphi r_2/4\approx 1.003664\) 逐分辨率增长。
\end{corollary}
\begin{proof}
由定义，
\(
\sum_x u_m(x)^2=|X_m|\cdot |X_m|^{-2}=|X_m|^{-1}=F_{m+2}^{-1}
\)。
又由定义 \ref{def:pom-s2} 与推论 \ref{cor:pom-s2-asymptotic}，
\(
\sum_x\pi_m(x)^2=S_2(m)/4^m=\Theta((r_2/4)^m)
\)。
于是
\[
R_m=F_{m+2}\cdot \Theta\!\left(\left(\frac{r_2}{4}\right)^m\right)
=\Theta\!\left(\left(\frac{\varphi\,r_2}{4}\right)^m\right),
\]
其中使用 Fibonacci 渐近 \(F_{m+2}=\Theta(\varphi^m)\)。取对数除以 \(m\) 并令 \(m\to\infty\) 得结论。证毕。
\end{proof}

\begin{corollary}[熵谱夹逼：\texorpdfstring{$h_2\le h_1\le \log\varphi$}{h2<=h1<=log phi}]\label{cor:pom-shannon-entropy-squeeze}
令 \(\pi_m(x)=d_m(x)/2^m\) 为折叠输出分布，并定义其 Shannon 熵
\[
H_1(m):=-\sum_{x\in X_m}\pi_m(x)\log\pi_m(x).
\]
设其熵率上下极限为
\[
\underline h_1:=\liminf_{m\to\infty}\frac{1}{m}H_1(m),
\qquad
\overline h_1:=\limsup_{m\to\infty}\frac{1}{m}H_1(m).
\]
则有夹逼
\[
\boxed{\;
h_2\ \le\ \underline h_1\ \le\ \overline h_1\ \le\ \log\varphi\; }.
\]
从而 \(\overline h_1\) 与上界 \(\log\varphi\) 的间隙被二阶常数完全量化：
\[
0\ \le\ \log\varphi-\overline h_1\ \le\ \log\varphi-h_2
=\log\!\left(\frac{\varphi r_2}{4}\right),
\]
其中右端恰为推论 \ref{cor:pom-renyi2-near-uniform} 的偏离指数常数（约 \(3.657482\times 10^{-3}\)）。
\end{corollary}
\begin{proof}
对任意有限支持分布 \(\pi\)，Rényi 熵随阶数单调递减，故 \(H_2(\pi)\le H_1(\pi)\le H_0(\pi)\)；
这里 \(H_2(\pi)=-\log\sum_x\pi(x)^2\)、\(H_0(\pi)=\log|\mathrm{supp}(\pi)|\)。
应用于 \(\pi=\pi_m\) 得
\[
-\log\sum_x\pi_m(x)^2\ \le\ H_1(m)\ \le\ \log|X_m|=\log F_{m+2}.
\]
两端除以 \(m\) 并取 \(\liminf,\limsup\)，再用 \(\lim_{m\to\infty}\frac{1}{m}\log F_{m+2}=\log\varphi\) 与推论 \ref{cor:pom-s2-asymptotic} 的
\(
h_2=\lim_{m\to\infty}\frac{1}{m}\bigl(-\log\sum_x\pi_m(x)^2\bigr)
\)
即得第一组夹逼。不等式 \(\log\varphi-h_2=\log(\varphi r_2/4)\) 由 \(h_2=\log(4/r_2)\) 直接化简得到。证毕。
\end{proof}

\begin{proposition}[二阶碰撞核 \(A_2\) 的最小 5 态完备化]\label{prop:pom-a2-gut-completion}

\end{pmatrix}.

\[

\begin{pmatrix}
0&0\\
0&1\\
1&0
\end{pmatrix},
\qquad

\begin{pmatrix}
2&1&0\\
0&1&0
\end{pmatrix}.

\[

A_{\mathrm{GUT}}

\begin{pmatrix}

\begin{remark}[primitive 谱的快速计算器：\texorpdfstring{$O(N\log N)$}{O(N log N)} 生成 \texorpdfstring{$p_n(A_2)$}{p_n(A2)}]\label{rem:pom-a2-primitive-fast}
令 \(t_n:=\mathrm{Tr}(A_2^n)\)。
由 \(A_2\) 的特征多项式 \(\lambda^3-2\lambda^2-2\lambda+2=0\)（等价于 Cayley--Hamilton 恒等式 \(A_2^3=2A_2^2+2A_2-2I\)），两边乘以 \(A_2^{n-3}\) 并取迹得三阶递推
\[
\boxed{\;
t_n=2t_{n-1}+2t_{n-2}-2t_{n-3}\qquad(n\ge 4)\;}
\]
且初值为 \(t_1=2,t_2=8,t_3=14\)。
另一方面由 Witt/Möbius 反演（定义 \ref{def:pom-collision-zeta}）有
\[
p_n(A_2)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,t_{n/d}.
\]
因此对任意 \(N\)，可先用递推在 \(O(N)\) 时间生成 \(\{t_n\}_{n\le N}\)，再用约数和的筛法在 \(O(N\log N)\) 时间生成全部 \(\{p_n(A_2)\}_{n\le N}\)；
这为把 \(A_2\) 当作“碰撞 primitive 谱基准器”（用于对比更复杂核族的指数常数与有限部常数偏移）提供了一个直接可复现的算法口径。
\end{remark}

\begin{remark}[二阶碰撞核的严格 RH 裕量（\texorpdfstring{$A_2$}{A2}）]\label{rem:pom-collision-rh-margin}
记 \(\lambda:=\rho(A_2)=r_2\approx 2.481194304092\)。由 \(A_2\) 的特征多项式
\[
\det(\lambda I-A_2)=\lambda^3-2\lambda^2-2\lambda+2
\]
可知三条特征值均为实数。除 Perron 根外另外两根数值约为
\[
\mu_2\approx -1.170086486626,\qquad
\mu_3\approx 0.688892182534.
\]
因此
\[
\max\{|\mu_2|,|\mu_3|\}\approx 1.170086\ <\ \sqrt{r_2}\approx 1.575181,
\]
即二阶碰撞核在核版 RH 不等式上严格留有裕量：非主谱严格落在平方根圆盘内。
\end{remark}

\begin{proposition}[二阶碰撞核的严格 RH：区间证书]\label{prop:pom-a2-strict-rh-certificate}
记 \(r_2=\rho(A_2)\)，并令
\[
\Lambda_2:=\max_{\mu\in\mathrm{spec}(A_2)\setminus\{r_2\}}|\mu|
\]
为次谱半径（sub-spectral radius）。则有严格不等式
\[
\boxed{\;\Lambda_2<\sqrt{r_2}.\;}
\]
\end{proposition}
\begin{proof}
由命题 \ref{prop:pom-collision-det}（或直接计算）可知 \(A_2\) 的特征多项式为
\[
p(\lambda):=\lambda^3-2\lambda^2-2\lambda+2.
\]
直接计算得到
\[
p\!\left(\frac94\right)=-\frac{79}{64}<0,\qquad
p\!\left(\frac52\right)=\frac18>0,
\]
故存在根 \(r_2\in\left(\frac94,\frac52\right)\)，从而 \(\sqrt{r_2}>\frac32\)。
另一方面，
\[
p\!\left(-\frac32\right)=-\frac{23}{8}<0,\qquad p(-1)=1>0
\]
给出一条负根 \(\mu_-\in\left(-\frac32,-1\right)\)，因而 \(|\mu_-|<\frac32\)；并且
\[
p(0)=2>0,\qquad p(1)=-1<0
\]
给出一条正根 \(\mu_+\in(0,1)\)，因而 \(|\mu_+|<1\)。
因此 \(\Lambda_2=\max\{|\mu_-|,|\mu_+|\}<\frac32<\sqrt{r_2}\)。证毕。
\end{proof}

\begin{remark}[差值链扭曲核：三态 \texorpdfstring{$A_2$}{A2} 的“最小可审计”角色扭曲试验]\label{rem:pom-a2-diffchain-twist}
命题 \ref{prop:pom-s2-recurrence} 给出的 \(A_2\) 是一个\emph{无标签}的非负 realization；若要定义角色扭曲 \(A_{2,\chi}\)，需要先在边上放置一个可加的“差值荷”标签。一个最小的、完全可审计的选取是从成对窗口的差值
\(

\)
出发，令一步增量 \(\delta_k:=\omega_k-\omega'_k\in\{-1,0,1\}\)，并注意 \(\delta_k=0\) 有两种实现 \((0,0),(1,1)\) 而 \(\delta_k=\pm 1\) 各只有一种实现。
\par
引入差值生成函数
\[
P_m(z):=\prod_{k=1}^m\Bigl(2+z^{F_{k+1}}+z^{-F_{k+1}}\Bigr)=\sum_{D\in\ZZ}a_m(D)\,z^D,
\]
则 \(a_m(D)\) 计数满足 \(D_m=D\) 的窗口对数。追踪三系数向量
\(
v_m:=(a_m(0),a_m(F_{m+1}),a_m(F_{m+2}))^{\mathsf T}
\)
可得到一个三态整数转移 \(v_{m+1}=Mv_m\)，其中
\[
M=
\begin{pmatrix}
2&0&2\\
1&0&2\\
0&1&0
\end{pmatrix},
\qquad
\det(\lambda I-M)=\lambda^3-2\lambda^2-2\lambda+2.
\]
因此 \(\rho(M)=\rho(A_2)=r_2\)，且其非主谱也与注 \ref{rem:pom-collision-rh-margin} 一致（同一特征多项式）。
\par
进一步把每步增量 \(\delta_k\) 作为边标签并对角色 \(\chi(1)=u\) 做扭曲，可得一个显式的复矩阵族
\[
M(u)=M_0+uM_++u^{-1}M_-=
\begin{pmatrix}
2&0&u+u^{-1}\\
u&0&2\\
0&u&0
\end{pmatrix},
\]
注意此处 \(\chi\) 是对一步差值荷 \(\delta_k\in\{-1,0,1\}\) 的\emph{局部}扭曲，并不等同于命题 \ref{prop:pom-s2-plancherel} 中定义在 \(G_m=\ZZ/F_{m+2}\ZZ\) 上的\emph{全局}角色 \(\chi_j\)。因此它更适合作为“可控的最小扭曲平台”，用于测试 worst-mode 扫描与 \(\zeta\) 行列式分解的数值接口。
可用根单位 \(u=e^{2\pi i r/n}\) 做直接谱半径扫描，从而数值定位“最坏扭曲模态”（通常对应 \(u\) 近 \(1\) 的低频模态）。同时该例也提醒：\textbf{并非任意边标签扭曲都满足平方根圆盘界}；在此差值链扭曲下，存在大量非平凡 \(u\) 使 \(\rho(M(u))>\sqrt{r_2}\)，见表 \ref{tab:fold_collision_a2_twist_difference_chain_scan}（可复现脚本 \texttt{scripts/exp\_fold\_collision\_a2\_twist\_difference\_chain\_scan.py}）。
\end{remark}

\subsubsection{三阶碰撞矩与 \texorpdfstring{$\mathrm{R\acute{e}nyi}$}{Renyi}--3 投影熵率}\label{subsec:pom-s3}
二阶碰撞矩 \(S_2(m)\) 描述“成对折叠碰撞”的 \(\ell^2\) 质量；更高阶的碰撞矩把纤维分布的尖锐程度分解为一族 \(\mathrm{R\acute{e}nyi}\) 指纹。下面给出三阶（\(q=3\)）的最小三态实现，其结构与命题 \ref{prop:pom-s2-recurrence} 完全同型。

\begin{definition}[三阶折叠碰撞矩]\label{def:pom-s3}
定义三阶碰撞和
$$
S_3(m):=\sum_{x\in X_m} d_m(x)^3.
$$
等价地，从 \(\Omega_m\) 上均匀独立抽取三条窗口 \(\omega_1,\omega_2,\omega_3\)，则
$$
\mathbb{P}\bigl[\Fold_m(\omega_1)=\Fold_m(\omega_2)=\Fold_m(\omega_3)\bigr]=\frac{S_3(m)}{8^m}.
$$
\end{definition}

\begin{proposition}[三阶碰撞核的三态表示与三阶递推]\label{prop:pom-s3-recurrence}
存在一个三态的非负整数矩阵
$$
A_3:=
\begin{pmatrix}
0&2&2\\
1&0&2\\
0&1&2
\end{pmatrix},
$$
以及读出向量
$$
u_3:=(5,13,34)^{\mathsf T},\qquad v_3:=(1,0,0)^{\mathsf T},
$$
使得对一切 \(m\ge 2\),
$$
S_3(m)=2\,u_3^{\mathsf T}A_3^{\,m-2}v_3.
$$
从而 \(S_3(m)\) 满足精确递推（\(m\ge 5\)）
$$
S_3(m)=2S_3(m-1)+4S_3(m-2)-2S_3(m-3),
$$
并具有初值 \(S_3(2)=10,S_3(3)=26,S_3(4)=88\)。
\end{proposition}
\begin{proof}
与命题 \ref{prop:pom-s2-recurrence} 相同，本命题以\textbf{可审计构造}给出：三态非负 realization 可由小 \(m\) 的枚举数据恢复，并由脚本 \texttt{scripts/exp\_fold\_collision\_s3.py} 逐项核验（同时核验递推与读出公式）。给定 \(A_3\) 与读出向量，序列
\(
a_n:=u_3^{\mathsf T}A_3^{n}v_3
\)
自动满足由 Cayley--Hamilton 定理诱导的三阶线性递推，其系数由 \(A_3\) 的特征多项式决定。直接计算得
$$
\det(\lambda I-A_3)=\lambda^3-2\lambda^2-4\lambda+2,
$$
再乘以常数 \(2\) 并作指标平移即得所述递推式。证毕。
\end{proof}

\begin{corollary}[三阶碰撞指数与 \texorpdfstring{$h_3$}{h3}]\label{cor:pom-s3-asymptotic}
令 \(r_3\) 为多项式
$$
x^3-2x^2-4x+2=0
$$
的最大实根（\(r_3\approx 3.086130197651\)）。则
