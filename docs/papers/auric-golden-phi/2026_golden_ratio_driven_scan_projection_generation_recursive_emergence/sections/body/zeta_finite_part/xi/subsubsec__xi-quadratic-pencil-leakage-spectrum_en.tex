\subsubsection{Quadratic Eigenvalue Pencilization and the ``Off-Axis Leakage Spectrum'': Horizon Circle Reciprocal Symmetry and Modal Classification}\label{subsubsec:xi-quadratic-pencil-leakage-spectrum}
This paragraph reformulates the zero-point problem of \(\Xi_{\Delta,L}(s)=\Delta(L^{-s},L^{2s-1})\) as a quadratic eigenvalue pencil (QEP) problem, and thereby establishes a structural proposition that can be stated independently: the critical line \(\Re(s)=\tfrac12\) is strictly isomorphic to the horizon circle \(|z|=L^{-1/2}\) in the \(z\)-plane; the so-called ``off-axis zeros'' correspond precisely to quadratic eigenvalues falling inside or outside the horizon circle; if the horizon circle reciprocal symmetry is further satisfied, then zeros appear in pairs under the map \(s\mapsto 1-\overline{s}\).
In the commutative (simultaneously unitarily diagonalizable) bridge kernel case, the quadratic eigenvalue pencil decomposes into scalar quadratic modes, thereby compressing the off-axis leakage into a family of discrete invariants (leakage indices/leakage spectrum) and yielding a complete classification of the zero towers.

\paragraph{Quadratic Eigenvalue Pencilization of the Completed Bridge Kernel}
Fix \(L>1\).
Take a first-order matrix family \(B(u)=B_0+uB_1\in\CC^{n\times n}\) and define the determinantal bridge kernel
\[
\Delta(z,u):=\det\!\bigl(I-zB(u)\bigr),
\qquad
\Xi_{\Delta,L}(s):=\Delta\!\bigl(L^{-s},L^{2s-1}\bigr).
\]
Let \(z:=L^{-s}\). Then \(u=L^{2s-1}=L^{-1}z^{-2}\), so that
\begin{equation}\label{eq:xi-qep-determinant-rewrite}
\Xi_{\Delta,L}(s)=0
\quad\Longleftrightarrow\quad
\det\!\Bigl(I-zB_0-\frac{1}{Lz}B_1\Bigr)=0
\quad\Longleftrightarrow\quad
\det \mathcal{P}_L(z)=0,
\end{equation}
where the quadratic matrix polynomial (pencil) is defined as
\begin{equation}\label{eq:xi-qep-PL-def}
\mathcal{P}_L(z):=z^2B_0-zI+\frac1L B_1.
\end{equation}

\begin{theorem}[Zero--Quadratic Eigenvalue Equivalence]\label{thm:xi-qep-zero-equivalence}
Let \(s\in\CC\) and \(z=L^{-s}\).
Then \(\Xi_{\Delta,L}(s)=0\) if and only if \(z\) is an eigenvalue of the quadratic eigenvalue problem
\[
\mathcal{P}_L(z)v=0
\]
(i.e., there exists a nonzero vector \(v\neq 0\) satisfying this equation).
Therefore, the zero set of \(\Xi_{\Delta,L}\) is equivalent to the root set of the quadratic eigenvalue pencil \(\det \mathcal{P}_L(z)\) under the map \(s=-\log z/\log L\) (where the multivaluedness of \(\log\) corresponds to the zero tower with period \(2\pi/\log L\)).
\end{theorem}

\begin{proof}
By \eqref{eq:xi-qep-determinant-rewrite}, \(\Xi_{\Delta,L}(s)=0\) if and only if \(\det \mathcal{P}_L(z)=0\).
Moreover, \(\det \mathcal{P}_L(z)=0\) if and only if there exists a nonzero vector \(v\) such that \(\mathcal{P}_L(z)v=0\).
Finally, for a fixed root \(z\neq 0\), the relation \(z=L^{-s}\) is equivalent to the complex logarithmic relation
\[
\log z=-s\log L+2\pi i k,\qquad k\in\ZZ,
\]
so that
\(
s=-\frac{\log z}{\log L}+\frac{2\pi i}{\log L}k
\);
this precisely corresponds to the zero tower shifted by steps of \(2\pi/\log L\) in the imaginary direction. \qed
\end{proof}

\begin{corollary}[Critical Line--Horizon Circle Isomorphism]\label{cor:xi-critical-line-horizon-circle}
For any \(s\in\CC\) let \(z=L^{-s}\). Then
\[
\Re(s)=\tfrac12\quad\Longleftrightarrow\quad |z|=L^{-1/2}.
\]
Therefore, ``off-axis zeros'' are strictly equivalent to ``quadratic eigenvalues falling inside or outside the horizon circle \(\{ |z|=L^{-1/2}\}\).''
\end{corollary}

\begin{proof}
From \(z=L^{-s}\) we obtain \(|z|=L^{-\Re(s)}\).
Hence \(\Re(s)=\tfrac12\iff |z|=L^{-1/2}\). \qed
\end{proof}

\begin{corollary}[Additivity of Leakage Spectrum and Channel Superposition: Bridge Kernel Direct Sum as Disjoint Union of Zero Towers]\label{cor:xi-leakage-additivity-directsum}
If the bridge kernel decomposes as a block direct sum under the same parametrization
\[
B(u)=B^{(1)}(u)\oplus B^{(2)}(u),
\]
then the determinantal bridge kernel factorizes:
\[
\Delta(z,u)=\det(I-zB(u))=\det(I-zB^{(1)}(u))\cdot \det(I-zB^{(2)}(u)).
\]
Consequently, for any \(L>1\),
\[
\Xi_{\Delta,L}(s)=\Xi_{\Delta^{(1)},L}(s)\cdot \Xi_{\Delta^{(2)},L}(s),
\qquad
\Delta^{(i)}(z,u):=\det(I-zB^{(i)}(u)).
\]
Therefore, the zero set of \(\Xi_{\Delta,L}\) (counted with algebraic multiplicity) is the disjoint union of the zero towers of the two subsystems; in particular, any ``off-axis leakage index/leakage spectrum'' defined via the zero towers (e.g., characterized by the offset multiset from \(|z|=L^{-\Re(s)}\) or \(\Re(s)-\tfrac12\)) superimposes as a multiset union under this direct sum assembly.
\end{corollary}
\begin{proof}
For block-diagonal matrices, the determinant equals the product of the block determinants; substituting into the definitions of \(\Delta\) and \(\Xi_{\Delta,L}\) yields the factorization; the (multiplicity-counted) disjoint union of the zero sets is a direct consequence of the product of analytic functions. \qed
\end{proof}

\paragraph{\texorpdfstring{$L$}{L}-Reciprocal Symmetry and Critical Line Pairing}
Combining the circle inversion of the horizon circle \(|z|=L^{-1/2}\) with conjugation, define the reciprocal transformation
\begin{equation}\label{eq:xi-z-sharp}
z^\sharp:=\frac{1}{L\overline{z}}.
\end{equation}
Note that this transformation satisfies \((z^\sharp)^\sharp=z\), and aligns precisely with reflection in the \(s\)-plane:
\begin{equation}\label{eq:xi-z-sharp-s-map}
z^\sharp=L^{-(1-\overline{s})}\qquad\text{when }z=L^{-s}.
\end{equation}

\begin{definition}[\texorpdfstring{$L$}{L}-Self-Reciprocal Polynomial (Horizon Circle Reciprocal Symmetry)]\label{def:xi-L-self-reciprocal}
Let \(q_L(z)\in\CC[z]\) be a polynomial of degree \(2n\).
We call \(q_L\) \textbf{\(L\)-self-reciprocal} (or \(L\)-self-dual) if there exists a constant \(c\in\CC\) with \(|c|=1\) such that for all \(z\neq 0\),
\begin{equation}\label{eq:xi-L-self-reciprocal}
q_L(z)=c\,z^{2n}\,\overline{q_L(z^\sharp)}.
\end{equation}
\end{definition}

\begin{theorem}[Reciprocal Symmetry \(\Rightarrow\) Critical Line Symmetry]\label{thm:xi-L-reciprocal-implies-critical-pairing}
Let \(q_L(z):=\det \mathcal{P}_L(z)\) (where \(\mathcal{P}_L\) is defined by \eqref{eq:xi-qep-PL-def}), and assume that \(q_L\) is \(L\)-self-reciprocal (Definition \ref{def:xi-L-self-reciprocal}).
If \(s_0\in\CC\) satisfies \(\Xi_{\Delta,L}(s_0)=0\), then \(1-\overline{s_0}\) also satisfies \(\Xi_{\Delta,L}(1-\overline{s_0})=0\).
In particular, the zeros of \(\Xi_{\Delta,L}\) appear in pairs symmetric about the line \(\Re(s)=\tfrac12\).
\end{theorem}

\begin{proof}
Let \(z_0:=L^{-s_0}\). By Theorem \ref{thm:xi-qep-zero-equivalence}, \(q_L(z_0)=0\).
By the \(L\)-self-reciprocal property \eqref{eq:xi-L-self-reciprocal},
\(
0=q_L(z_0)=c\,z_0^{2n}\,\overline{q_L(z_0^\sharp)}
\), so \(q_L(z_0^\sharp)=0\).
By \eqref{eq:xi-z-sharp-s-map}, \(z_0^\sharp=L^{-(1-\overline{s_0})}\), and applying Theorem \ref{thm:xi-qep-zero-equivalence} again yields \(\Xi_{\Delta,L}(1-\overline{s_0})=0\).
Finally, since \(\Re(1-\overline{s_0})=1-\Re(s_0)\), the symmetry about \(\Re(s)=\tfrac12\) follows. \qed
\end{proof}

\paragraph{Modal Decomposition for Commutative Bridge Kernels: Spectral Decomposition of Zero Towers and Leakage Indices}
When \(B_0,B_1\) commute and are simultaneously unitarily diagonalizable, the quadratic eigenvalue pencil decomposes completely into scalar modes.

\begin{proposition}[Modal Decomposition for Commutative Bridge Kernels]\label{prop:xi-qep-commuting-modal-factorization}
Suppose \(B_0,B_1\) commute and are simultaneously unitarily diagonalizable: there exists a unitary matrix \(U\) such that
\[
U^\ast B_0U=\mathrm{diag}(\alpha_1,\dots,\alpha_n),
\qquad
U^\ast B_1U=\mathrm{diag}(\beta_1,\dots,\beta_n).
\]
Then
\[
q_L(z)=\det \mathcal{P}_L(z)=\prod_{j=1}^n\Bigl(\alpha_j z^2-z+\frac1L\beta_j\Bigr).
\]
Consequently, the zero set of \(\Xi_{\Delta,L}\) is the union of the zero towers of \(n\) scalar quadratic modes.
\end{proposition}

\begin{proof}
By unitary diagonalization and \eqref{eq:xi-qep-PL-def},
\(
U^\ast \mathcal{P}_L(z)U=\mathrm{diag}(\alpha_j z^2-z+\beta_j/L)_{j=1}^n
\).
Taking the determinant yields the stated decomposition. \qed
\end{proof}

To distill the ``magnitude of departure from the critical line'' into discrete invariants, introduce the \(r\)-parameter:
\begin{equation}\label{eq:xi-qep-r-def}
r:=L^{s-\tfrac12},
\qquad
z=L^{-s}=L^{-1/2}r^{-1}.
\end{equation}
Then for each mode \(j\), the equation \(\alpha_j z^2-z+\beta_j/L=0\) is equivalent to
\begin{equation}\label{eq:xi-qep-mode-r-quadratic}
\beta_j r^2-\sqrt L\,r+\alpha_j=0.
\end{equation}
Moreover,
\begin{equation}\label{eq:xi-qep-re-shift-by-r}
\Re(s)-\tfrac12=\frac{\log|r|}{\log L}.
\end{equation}

\begin{definition}[Modal Leakage Index]\label{def:xi-modal-leakage-index}
Under the notation of Proposition \ref{prop:xi-qep-commuting-modal-factorization}, if \(\alpha_j,\beta_j\in\RR_{>0}\), define the \textbf{leakage index} of the \(j\)-th mode as
\[
\lambda_j:=\frac12\log\frac{\beta_j}{\alpha_j}.
\]
\end{definition}

\begin{theorem}[Complete Classification of Scalar Mode Zero Towers]\label{thm:xi-qep-scalar-mode-tower-classification}
Fix \(L>1\), and take \(\alpha,\beta\in\RR_{>0}\) (allowing \(\alpha=\beta\)).
Let \(\Delta:=L-4\alpha\beta\).
Consider the zero set induced by the scalar mode equation \eqref{eq:xi-qep-mode-r-quadratic} (with \((\alpha_j,\beta_j)\) replaced by \((\alpha,\beta)\)):
\[
s=\frac12+\frac{\log r}{\log L},
\qquad r\ \text{is a root of \eqref{eq:xi-qep-mode-r-quadratic}}.
\]
Then the following classification holds.
\begin{enumerate}
  \item \textbf{(\(\Delta<0\): complex roots)}
  Let
  \[
  \varphi:=\arccos\!\Bigl(\frac{\sqrt L}{2\sqrt{\alpha\beta}}\Bigr)\in\Bigl(0,\frac{\pi}{2}\Bigr).
  \]
  Then the two roots of \eqref{eq:xi-qep-mode-r-quadratic} can be written as
  \(
  r_{\pm}=\sqrt{\alpha/\beta}\,e^{\pm i\varphi}
  \).
  The corresponding zeros form two arithmetic-progression towers on a vertical line:
  \[
  \Re(s)=\frac12+\frac{1}{2\log L}\log\frac{\alpha}{\beta},
  \qquad
  \Im(s)=\frac{\pm\varphi+2\pi k}{\log L},
  \quad k\in\ZZ.
  \]
  \item \textbf{(\(\Delta=0\): repeated root)}
  Equation \eqref{eq:xi-qep-mode-r-quadratic} has a repeated root \(r_0=\sqrt{\alpha/\beta}\).
  The corresponding zeros form a single vertical tower:
  \[
  \Re(s)=\frac12+\frac{1}{2\log L}\log\frac{\alpha}{\beta},
  \qquad
  \Im(s)=\frac{2\pi k}{\log L},
  \quad k\in\ZZ.
  \]
  \item \textbf{(\(\Delta>0\): two real roots)}
  Equation \eqref{eq:xi-qep-mode-r-quadratic} has two positive real roots
  \[
  r_{\pm}=\frac{\sqrt L\pm\sqrt{\Delta}}{2\beta}.
  \]
  The corresponding zeros form two (possibly coincident) vertical towers:
  \[
  \Re(s)=\frac12+\frac{\log r_{\pm}}{\log L},
  \qquad
  \Im(s)=\frac{2\pi k}{\log L},
  \quad k\in\ZZ.
  \]
\end{enumerate}
\end{theorem}

\begin{proof}
Applying the quadratic formula directly to \eqref{eq:xi-qep-mode-r-quadratic} yields the roots in the three cases.
When \(\Delta<0\),
\(
r_{\pm}=\frac{\sqrt L\pm i\sqrt{4\alpha\beta-L}}{2\beta}
\).
Writing these as \(\sqrt{\alpha/\beta}e^{\pm i\varphi}\) yields the stated \(\varphi\).
Substituting \(\log r=\log|r|+i(\arg r+2\pi k)\) into \(s=\frac12+\frac{\log r}{\log L}\) gives the corresponding zero tower expressions. \qed
\end{proof}

\begin{corollary}[Off-Axis Leakage Spectrum under Reciprocal Symmetry (Commutative Case)]\label{cor:xi-qep-leakage-spectrum-commuting}
Under the commutative setting of Proposition \ref{prop:xi-qep-commuting-modal-factorization}, further assume that \(q_L(z)=\det \mathcal{P}_L(z)\) is \(L\)-self-reciprocal (Definition \ref{def:xi-L-self-reciprocal}), and that \(\alpha_j,\beta_j\in\RR_{>0}\).
Let \(\lambda_j\) be the leakage index (Definition \ref{def:xi-modal-leakage-index}), and let \(\Delta_j:=L-4\alpha_j\beta_j\).
Then for each mode pair satisfying \(\lambda_j\neq 0\) (which must appear in pairs under \(L\)-self-reciprocity, so the leakage indices take values \(\pm\lambda_j\)), the contributed zero towers satisfy:
\begin{enumerate}
  \item If \(\Delta_j<0\), the zeros lie precisely on two vertical lines
  \[
  \Re(s)=\frac12\pm\frac{\lambda_j}{\log L},
  \]
  with imaginary parts on each line given by
  \[
  \Im(s)=\frac{\pm\varphi_j+2\pi k}{\log L},
  \qquad
  \varphi_j:=\arccos\!\Bigl(\frac{\sqrt L}{2\sqrt{\alpha_j\beta_j}}\Bigr)\in\Bigl(0,\frac{\pi}{2}\Bigr),
  \quad k\in\ZZ.
  \]
  \item If \(\Delta_j=0\), then (possibly coincident) vertical towers appear; their symmetric towers about \(\Re(s)=\tfrac12\) are enforced by Theorem \ref{thm:xi-L-reciprocal-implies-critical-pairing}.
  \item If \(\Delta_j>0\), let
  \[
  r_{j,\pm}:=\frac{\sqrt L\pm\sqrt{\Delta_j}}{2\beta_j}\in\RR_{>0}.
  \]
  Then mode \(j\) contributes two vertical towers
  \[
  \Re(s)=\frac12+\frac{\log r_{j,\pm}}{\log L},
  \qquad
  \Im(s)=\frac{2\pi k}{\log L},
  \quad k\in\ZZ,
  \]
  and under \(L\)-self-reciprocity, Theorem \ref{thm:xi-L-reciprocal-implies-critical-pairing} additionally yields their symmetric towers about \(\Re(s)=\tfrac12\):
  \[
  \Re(s)=\frac12-\frac{\log r_{j,\pm}}{\log L},
  \qquad
  \Im(s)=\frac{2\pi k}{\log L},
  \quad k\in\ZZ,
  \]
  resulting in four vertical zero towers symmetric about \(\Re(s)=\tfrac12\) (with possible coalescence in special alignment cases).
\end{enumerate}
\end{corollary}

\begin{proof}
By Theorem \ref{thm:xi-L-reciprocal-implies-critical-pairing}, under the \(L\)-self-reciprocal assumption, zeros appear in pairs about \(\Re(s)=\tfrac12\).
On the other hand, by Theorem \ref{thm:xi-qep-scalar-mode-tower-classification}, the zero tower of each scalar mode is completely determined by the discriminant \(\Delta_j\); when \(\Delta_j\le 0\), the real part of zeros in mode \(j\) is fixed at
\(
\frac12+\frac{1}{2\log L}\log(\alpha_j/\beta_j)=\frac12-\frac{\lambda_j}{\log L}
\).
Combining with the symmetric pairing yields two vertical lines \(\Re(s)=\frac12\pm\frac{\lambda_j}{\log L}\).
When \(\Delta_j>0\), each mode yields two vertical towers; after pairing, one obtains the four-tower symmetric structure about \(\Re(s)=\tfrac12\) (with possible coalescence). \qed
\end{proof}

\begin{corollary}[Irreducible Lower Bound on Leakage]\label{cor:xi-qep-leakage-lower-bound}
Under the assumptions of Corollary \ref{cor:xi-qep-leakage-spectrum-commuting}, if some paired mode satisfies \(\lambda_j\neq 0\), then there necessarily exist zeros \(s\) satisfying
\[
\Re(s)\le \frac12-\frac{|\lambda_j|}{\log L},
\qquad
\Re(s)\ge \frac12+\frac{|\lambda_j|}{\log L}.
\]
Moreover, equality holds if and only if \(\Delta_j\le 0\), in which case all zeros of this paired mode lie exactly on the two bounding lines.
\end{corollary}

\begin{proof}
By Corollary \ref{cor:xi-qep-leakage-spectrum-commuting}, when \(\Delta_j\le 0\) the real parts of the zeros are fixed at \(\frac12\pm\frac{|\lambda_j|}{\log L}\), so the conclusion holds with equality.
When \(\Delta_j>0\), Theorem \ref{thm:xi-qep-scalar-mode-tower-classification} yields two positive real roots \(r_{\pm}\); since \(r_{+}r_{-}=\alpha_j/\beta_j\), at least one root satisfies \(|r|\le \sqrt{\alpha_j/\beta_j}\), and by \eqref{eq:xi-qep-re-shift-by-r} the corresponding column of zeros satisfies \(\Re(s)\le \frac12-\frac{|\lambda_j|}{\log L}\).
By the critical line symmetry enforced by \(L\)-self-reciprocity (Theorem \ref{thm:xi-L-reciprocal-implies-critical-pairing}), the symmetric column yields zeros with \(\Re(s)\ge \frac12+\frac{|\lambda_j|}{\log L}\).
Finally, when \(\Delta_j>0\), there exist two distinct vertical towers (barring coalescence degeneracy), so it is not possible for ``all zeros to lie only on the two bounding lines.'' \qed
\end{proof}

\paragraph{Non-Commutative Leakage Spectrum Bounds: Generalized Eigenvalue Ratios and Condition Numbers for Positive Definite Hermitian Bridge Kernels}
The preceding paragraph provided a complete classification via ``leakage indices \(\lambda_j\)'' in the commutative/simultaneously unitarily diagonalizable regime.
When \(B_0,B_1\) do not commute, the modal decomposition no longer holds; however, when \(B_0,B_1\) are positive definite Hermitian, the off-axis leakage can still be compressed into a set of \emph{invariants of generalized eigenvalue ratios}:
these are controlled by the spectral interval and condition number of the operator \(M=B_1^{-1/2}B_0B_1^{-1/2}\), and yield auditable strip constraints in the ``complex root tower (underdamped)'' regime.

\begin{theorem}[Positive Definite Self-Dual Bridge Kernel Leakage Spectrum Bounds: Rayleigh Interval and Condition Number]\label{thm:xi-qep-noncommuting-cond-bound}
Continuing with the bridge kernel setup at the beginning of this paragraph, take \(B(u)=B_0+uB_1\) and let \(\Xi_{\Delta,L}(s)=\Delta(L^{-s},L^{2s-1})\).
Let \(r=L^{s-\frac12}\) and define the matrix quadratic pencil
\[
Q_L(r):=r^2B_1-\sqrt L\,r\,I+B_0,
\qquad
\Xi_{\Delta,L}(s)=0\ \Longleftrightarrow\ \det Q_L(r)=0.
\]
Further assume that \(B_0,B_1\) are positive definite Hermitian, and let
\[
M:=B_1^{-1/2}B_0B_1^{-1/2},
\qquad
\lambda_{\min}:=\lambda_{\min}(M),\ \lambda_{\max}:=\lambda_{\max}(M),\
\kappa(M):=\lambda_{\max}/\lambda_{\min}.
\]
If the sufficient condition
\[
L<4\,\lambda_{\min}(B_0)\,\lambda_{\min}(B_1)
\]
holds, then all quadratic eigenvalues \(r\) of \(Q_L(r)\) are non-real and appear in conjugate pairs; moreover, for any corresponding zero \(s\) (i.e., \(r=L^{s-\frac12}\)), the strict strip constraint
\[
\boxed{\;
\frac12+\frac{1}{2\log L}\log\lambda_{\min}
\ \le\
\Re(s)
\ \le\
\frac12+\frac{1}{2\log L}\log\lambda_{\max}\;}
\]
holds, so the width of this strip satisfies
\[
\mathrm{width}
\;=\;
\frac{1}{2\log L}\log\kappa(M).
\]
In the commutative and simultaneously unitarily diagonalizable case, the above inequality is attained at the extremal modes (reducing to the modal formula \(\Re(s)=\frac12+\frac{1}{2\log L}\log(\alpha_j/\beta_j)\)).
If additionally the spectral reciprocal symmetry \(\lambda_{\min}\lambda_{\max}=1\) holds (e.g., from self-dual similarity classes yielding pairwise reciprocal generalized eigenvalue ratios), then the strip is symmetric about the critical line and yields the half-width upper bound
\[
\boxed{\;
\bigl|\Re(s)-\tfrac12\bigr|
\le
\frac{1}{4\log L}\log\kappa(M)\; }.
\]
\end{theorem}

\begin{proof}
Let \(r\) be a quadratic eigenvalue of \(Q_L\), and take a nonzero vector \(v\neq 0\) such that \(Q_L(r)v=0\).
Taking the inner product with \(v\) yields the scalar quadratic equation
\[
(v^\ast B_1v)\,r^2-\sqrt L\,(v^\ast v)\,r+(v^\ast B_0v)=0.
\]
By positive definiteness, all three coefficients are positive real numbers.
From the lower bound \(v^\ast B_i v\ge \lambda_{\min}(B_i)\,v^\ast v\) (\(i=0,1\)), we obtain
\[
L(v^\ast v)^2-4(v^\ast B_1v)(v^\ast B_0v)
\le
L(v^\ast v)^2-4\lambda_{\min}(B_1)\lambda_{\min}(B_0)(v^\ast v)^2<0,
\]
so the discriminant is negative, and \(r\) must be non-real and appear as a conjugate pair.
In this case the two roots are conjugate, so their modulus satisfies
\[
|r|^2=\frac{v^\ast B_0v}{v^\ast B_1v}.
\]
Let \(w:=B_1^{1/2}v\neq 0\); then
\(
|r|^2=\frac{w^\ast Mw}{w^\ast w}
\)
is the Rayleigh quotient of \(M\), whence
\(
\lambda_{\min}\le |r|^2\le \lambda_{\max}
\).
Taking logarithms and using \(\Re(s)=\tfrac12+\frac{\log|r|}{\log L}\) yields the stated strip constraint and width formula.
In the commutative case, equality is attained at the extremal modes by choosing common eigenvectors.
If \(\lambda_{\min}\lambda_{\max}=1\), then \(\log\lambda_{\min}=-\log\lambda_{\max}=-\frac12\log\kappa(M)\); substituting into the strip endpoints yields the half-width upper bound. \qed
\end{proof}

\paragraph{Explicit Leakage Spectrum for the Minimal \texorpdfstring{$2\times2$}{2x2} Coupling Bridge Kernel: Hyperbolic Curvature Parametrization}
In the minimal self-dual coupling kernel (Theorem \ref{thm:xi-2x2-selfdual-offcritical-template}), \(B_0,B_1\) are simultaneously unitarily diagonalizable and yield a pair of commutative modal coefficients
\(
(\alpha_1,\beta_1)=(a-b,a+b)
\)
and
\(
(\alpha_2,\beta_2)=(a+b,a-b)
\).
The corresponding leakage indices are consistent and can be expressed as hyperbolic curvature parameters.

\begin{corollary}[Leakage Index and Zero Geometry for Minimal Coupling]\label{cor:xi-2x2-coupling-leakage-artanh}
Take integers \(a>|b|>0\), and let
\[
\lambda:=\frac12\log\frac{a+b}{a-b}=\operatorname{artanh}\!\Bigl(\frac{b}{a}\Bigr),
\qquad
\Delta:=L-4(a^2-b^2).
\]
Then the off-axis zeros of this minimal coupling bridge kernel appear in pairs about \(\Re(s)=\tfrac12\), and satisfy:
\begin{enumerate}
  \item If \(\Delta<0\), all off-axis zeros lie precisely on two vertical lines
  \[
  \Re(s)=\frac12\pm\frac{\lambda}{\log L},
  \]
  with imaginary parts forming arithmetic-progression towers with step size \(2\pi/\log L\) (phase offset given by \(\varphi\) from Theorem \ref{thm:xi-qep-scalar-mode-tower-classification}).
  \item If \(\Delta>0\), vertical zero towers symmetric about \(\Re(s)=\tfrac12\) appear (with possible coalescence in aligned cases); the corresponding \(r\)-roots are given by
  \[
  r_{\pm}=\frac{\sqrt L\pm\sqrt{\Delta}}{2(a+b)}
  \]
  and their reciprocal paired roots, so the real-part offset is completely determined by \(\log r_{\pm}/\log L\).
\end{enumerate}
Moreover, as \(b\to 0\), \(\lambda\sim b/a\), and the leakage vanishes to first order linearly.
\end{corollary}

\begin{proof}
The quadratic reduction of this coupling kernel was given in Theorem \ref{thm:xi-2x2-selfdual-offcritical-template}, where the two modal quadratic equations have coefficients \((\alpha,\beta)=(a-b,a+b)\) and the exchanged pair \((a+b,a-b)\).
The hyperbolic curvature expression for the leakage index follows directly from the identity
\(
\operatorname{artanh}(x)=\tfrac12\log\frac{1+x}{1-x}
\).
The remaining conclusions follow from Corollary \ref{cor:xi-qep-leakage-spectrum-commuting} (applied separately to the \(\Delta<0\) and \(\Delta>0\) cases); \(\lambda\sim b/a\) follows from \(\operatorname{artanh}(x)=x+O(x^3)\). \qed
\end{proof}

\begin{remark}[Conceptual Reduction: Externalization of the ``Leakage Spectrum'' and Type Recovery under Single-Slice Protocol]\label{rem:xi-qep-leakage-conceptual}
By Theorem \ref{thm:xi-qep-zero-equivalence} and Corollary \ref{cor:xi-critical-line-horizon-circle}, when the completion protocol is fixed to a single horizon slice (\(|z|=L^{-1/2}\) or equivalently \(\Re(s)=\tfrac12\)), any verifiable ``deviation'' can be equivalently stated as the radial departure of the spectrum of the quadratic eigenvalue pencil \(\mathcal{P}_L(z)\) from the horizon circle.
When horizon circle reciprocal symmetry holds (Theorem \ref{thm:xi-L-reciprocal-implies-critical-pairing}), such deviations are forced to appear in pairs symmetric about the critical line.
In finite-dimensional commutative bridge kernels, deviations are further compressed into discrete leakage indices/leakage spectrum (Definition \ref{def:xi-modal-leakage-index}; Corollary \ref{cor:xi-qep-leakage-spectrum-commuting}).
Therefore, under the single-slice visible domain regime of this paper (Definition \ref{def:xi-visible-read-null}, taking \(X_{\mathrm{vis}}=\mathsf{Phase}\)) and the type-safety closure (Theorem \ref{thm:xi-offset-null-type-safety}), the ``off-axis leakage spectrum'' can be understood as an \emph{orthogonal recording axis that must be externalized}: if the protocol permits externalization, it enters the certificate semantics as discrete spectral invariants; if the protocol forbids externalization, this degree of freedom is recovered by the type system in the closure and read out as \(\mathsf{NULL}\).
\end{remark}
