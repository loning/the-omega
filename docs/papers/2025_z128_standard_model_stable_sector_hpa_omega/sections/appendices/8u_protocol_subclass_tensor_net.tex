\subsection{A protocol subclass with tensor-product locality (T2 PT carrier)}
\label{app:protocol_subclass_tensor_net}

\AuditTag This appendix fixes an explicit protocol subclass in which microcausality becomes a PT-level theorem by construction.
The guiding principle is subsystem factorization: operationally disjoint readouts are organized as tensor-product subsystems (or commuting subalgebras) at the finite-readout level.

\subsubsection{Definition of the subclass}
\begin{definition}[Tensor-product readout subclass]
\label{def:tensor_product_readout_subclass}
\MathTag A protocol/readout organization belongs to the \emph{tensor-product readout subclass} if for any pair of operationally disjoint windows $W_1,W_2\in\mathcal{W}$, there exists a canonical identification of finite observable algebras
\[
\mathfrak{A}(W_1\cup W_2)\ \cong\ \mathfrak{A}(W_1)\otimes \mathfrak{A}(W_2),
\]
and the embeddings of local observables are the standard tensor embeddings:
\[
\iota_1:\mathfrak{A}(W_1)\hookrightarrow \mathfrak{A}(W_1)\otimes\mathfrak{A}(W_2),\quad A\mapsto A\otimes \ind,
\qquad
\iota_2:\mathfrak{A}(W_2)\hookrightarrow \mathfrak{A}(W_1)\otimes\mathfrak{A}(W_2),\quad B\mapsto \ind\otimes B.
\]
\end{definition}

\InterfaceTag Definition~\ref{def:tensor_product_readout_subclass} is a concrete structural specialization of the unified observable/state base (Appendix~\ref{app:observable_algebra_state_base}) and of composite-system tensor products (Appendix~\ref{app:composite_systems_tensor_products}).

\subsubsection{Microcausality as a theorem (PT in the subclass)}
\begin{proposition}[Causal separability holds in the tensor-product subclass]
\label{prop:causal_separability_from_tensor_subclass}
\MathTag Under Definition~\ref{def:tensor_product_readout_subclass}, Assumption~\ref{ass:causal_separability_readout_channels} holds.
\end{proposition}
\begin{proof}
In the tensor-product subclass, the union algebra is generated by $\iota_1(\mathfrak{A}(W_1))$ and $\iota_2(\mathfrak{A}(W_2))$ inside $\mathfrak{A}(W_1)\otimes\mathfrak{A}(W_2)$.
These subalgebras commute because $(A\otimes \ind)(\ind\otimes B)=(\ind\otimes B)(A\otimes \ind)$.
Passing to inductive limits yields the stated commuting-subalgebra organization.
\end{proof}

\begin{corollary}[Microcausality for the protocol-induced net in the tensor-product subclass]
\label{cor:microcausality_tensor_subclass}
\MathTag In the tensor-product readout subclass, the protocol-induced net satisfies microcausality for operationally disjoint regions.
\end{corollary}
\begin{proof}
Combine Proposition~\ref{prop:causal_separability_from_tensor_subclass} with Theorem~\ref{thm:microcausality_under_causal_separability}.
\end{proof}
