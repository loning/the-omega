where $p$ ranges over all Ihara primes. This definition eliminates ``immediately reversed'' spurious periods and better captures the irreducible patterns of reversible atomic operations.

\paragraph{Weighted Ihara--Bass: spectral fingerprint of carry energy}
Let $\kappa(e)$ denote the per-step carry activation energy and take the weight $u^{\kappa(e)}$. On the non-backtracking edge space, define the Hashimoto operator $B(u)$ (edge $e\to e'$ is assigned weight $u^{\kappa(e')}$ when connectable without backtracking). Then the weighted Ihara \(\zeta\) is
\[
Z_K(u,t):=\prod_{p}\left(1-t^{\ell(p)}u^{\kappa(p)}\right)^{-1}
=\frac{1}{\det(I-tB(u))},
\]
so ``atomic energy'' enters the spectral fingerprint while preserving determinant computability (the Bass--Hashimoto three-term determinant formula can be viewed as an equivalent formulation of the above definition on the undirected skeleton). This two-parameter object yields a family of \textbf{matrix-polynomial determinant invariants} that are closer to the ``prime spectrum'' of the ``reversible rewriting groupoid'' than the single periodic-point trace.

\begin{proposition}[Ihara--Witt composite bridge package: energy-resolved primitive spectrum of non-backtracking primes]\label{prop:ihara-witt-primitive-spectrum}
For the weighted Hashimoto operator \(B(u)\), define the Ihara trace polynomials
\[
b_n(u):=\Tr(B(u)^n)\in\ZZ[u],
\]
and define the Ihara-primitive polynomials (Witt coordinates)
\[
q_n(u):=\frac{1}{n}\sum_{d\mid n}\mu(d)\,b_{n/d}(u^d)\in\QQ[u].
\]
Then the weighted Ihara \(\zeta\) admits the strict Witt Euler product expansion:
\[
\boxed{\
Z_K(u,t)
=\exp\!\left(\sum_{n\ge 1}\frac{b_n(u)}{n}t^n\right)
=\prod_{n\ge 1}\bigl(1-t^n\bigr)^{-q_n(u)}
\ }.
\]
Writing \(q_n(u)=\sum_k q_{n,k}u^k\), the collection \(\{q_{n,k}\}\) provides the \textbf{energy-resolved primitive spectrum of non-backtracking Ihara primes}.
\end{proposition}
\begin{proof}
Apply the Big-Witt identity from \S\ref{app:witt-bridge-package} of this appendix for an arbitrary parametric matrix family \(T(u)\) to \(T(u)=B(u)\) (replacing the variable \(z\) by \(t\)). \qedhere
\end{proof}

\begin{corollary}[Ihara necklace divisibility law and Dwork-type congruence]\label{cor:ihara-witt-dwork}

\begin{enumerate}

\end{enumerate}
\end{corollary}
\begin{proof}

\begin{theorem}[Cyclotomic factor classification of the graph \texorpdfstring{$\zeta$}{zeta}-function: Ihara--Artin equivalent characterization of modular obstructions]\label{thm:conclusion75-ihara-cyclotomic-factor-classification}
Using the notation above and fixing \(u>0\).
For each irreducible representation \(\rho\in\widehat G\), the twisted Ihara--Artin \(\zeta/L\) factor of the twisted Hashimoto operator \(B_\rho(u)\) may be taken as
\[
Z_{K}(u,t;\rho):=L^{\mathrm{Ih}}_{K,\rho}(u,t)=\det(I-tB_\rho(u))^{-1}.
\]
Then a \textbf{nontrivial modular obstruction} exists if and only if there exists some \(\rho\neq\mathbf 1\) such that
\[
\boxed{\ \rho\!\bigl(B_\rho(u)\bigr)=\rho\!\bigl(B_{\mathbf 1}(u)\bigr)=:\lambda_1(u)\ }.
\]
Moreover, in this case, \(\det(I-tB_\rho(u))\) must contain a cyclotomic factor (equivalently, the normalized polynomial \(\det(I-(t/\lambda_1(u))B_\rho(u))\) has a root-of-unity factor on the unit circle), so that the denominator of \(Z_K(u,t;\rho)\) exhibits an enumerable cyclotomic factor structure.
Conversely, if \(\det(I-tB_\rho(u))\) for some \(\rho\neq\mathbf 1\) contains a cyclotomic factor, then \(\rho(B_\rho(u))=\lambda_1(u)\) necessarily holds.
\par
Therefore, ``finite modular obstruction/resonance sources'' can be equivalently reduced to: whether the determinants of certain nontrivial twisted channels contain cyclotomic factors; this criterion is entirely at the level of finite-dimensional algebraic factor classification, compressing the source of spectral gap failure into an enumerable unit-root structure.
\end{theorem}

\begin{proof}
If there exists \(\rho\neq\mathbf 1\) with \(\rho(B_\rho(u))=\lambda_1(u)\), then \(B_\rho(u)\) has an eigenvalue \(\lambda\) of modulus \(\lambda_1(u)\).
Since twisting only introduces unit-complex character values (roots of unity) from the finite group representation, and the matrix entries belong to a cyclotomic number field, any eigenvalue on the spectral boundary must correspond to a finite-order rotation; thus \(\lambda/\lambda_1(u)\) is a root of unity, which is equivalent to \(\det(I-tB_\rho(u))\) having a unit-root-type factor at \(t=\lambda^{-1}\), i.e., containing a cyclotomic factor.
The reverse implication is analogous: if a cyclotomic factor appears, there exists a root of unity \(\zeta\) such that \(\det(I-(\zeta/\lambda_1(u))B_\rho(u))=0\), so \(B_\rho(u)\) has an eigenvalue of modulus \(\lambda_1(u)\), giving \(\rho(B_\rho(u))=\lambda_1(u)\).
\qedhere
\end{proof}

\begin{theorem}[Ihara--Chebotarev: leading term and exponential error by Frobenius conjugacy class]\label{thm:ihara-chebotarev-exp}
Fix \(u>0\). Denote \(\lambda_1(u):=\rho(B(u))\) as the non-backtracking dominant spectral radius, and set
\[
\lambda_\star(u):=\max_{\mu\in\mathrm{spec}(B(u))\setminus\{\lambda_1(u)\}}|\mu|,
\qquad
\eta(u):=\max_{\rho\in\widehat G,\ \rho\neq\mathbf 1}\rho(B_\rho(u)),
\qquad
\Lambda(u):=\max\{\lambda_\star(u),\eta(u)\}.
\]
For any conjugacy class \(C\subseteq G\), define the class-stratified Ihara-primitive count
\[
q_{n,C}(u):=\sum_{\substack{p\ \mathrm{Ihara\ prime}\\ \ell(p)=n\\ \mathrm{Fr}(p)\in C}}u^{\kappa(p)}.
\]
Then as \(n\to\infty\),
\[
\boxed{\
q_{n,C}(u)=\frac{|C|}{|G|}\cdot\frac{\lambda_1(u)^n}{n}
+O\!\left(\frac{\Lambda(u)^n}{n}\right)\ }.
\]
The error exponent is explicitly controlled by \(\Lambda(u)/\lambda_1(u)\); in particular, if both
\(
\lambda_\star(u)\le \lambda_1(u)^{1/2}
\)
and
\(
\eta(u)\le \lambda_1(u)^{1/2}
\)
hold (Ihara--RH/GRH type spectral gap), then the error is at the square-root level
\(
O(\lambda_1(u)^{n/2}/n)
\).
\end{theorem}

\begin{proof}
The proof is entirely parallel to Theorem \ref{thm:kernel-chebotarev-exp} in the main text; one only needs to replace the operator \(M(u)\) by \(B(u)\) and the twisted family \(M_\rho(u)\) by \(B_\rho(u)\) throughout the chain ``periodic trace \(\Rightarrow\) Witt/M\"obius inversion \(\Rightarrow\) primitive.''
The trivial representation gives the leading term \(\lambda_1(u)^n/n\) with error \(O(\lambda_\star(u)^n/n)\); the contribution of nontrivial representations is controlled by the crude bound
\(
|\Tr(A^m)|\le (\dim A)\,\rho(A)^m
\)
as \(O(\eta(u)^n/n)\). Using character orthogonality to expand the conjugacy-class indicator function into representation components yields the conclusion.
\qedhere
\end{proof}

From this theorem one defines the auditable deviation rate indicator
\[
\theta_K(u):=\max_{\rho\neq\mathbf 1}\frac{\rho(B_\rho(u))}{\rho(B_{\mathbf 1}(u))}
=\frac{\eta(u)}{\lambda_1(u)},
\]
which quantifies the ``convergence rate of primes in congruence classes,'' forming a fully computable interface for the ``kernel-version Chebotarev/Artin-\(L\).''

\paragraph{Zero-temperature energy gap: computability of the minimum mean cycle}
Treating $\kappa(e)$ as energy, define the mean energy of a prime cycle as
\(
\bar\kappa(p)=\kappa(p)/\ell(p)
\)
and let
\[
\Delta_K:=\min_{p:\bar\kappa(p)>0}\bar\kappa(p).
\]
On the non-backtracking edge graph (Hashimoto graph), this problem is equivalent to the \textbf{minimum mean cycle}, which can be solved in polynomial time by Karp's algorithm (see \cite{Karp1978MinCycleMean}). $\Delta_K$ provides a comparable invariant for the ``ground state/first excited state energy gap'' and governs the leading scale of the first nontrivial correction as $u\to 0$.

\begin{proposition}[Degeneracy mechanism of the density gap and tunneling action]\label{prop:ihara-gap-degeneracy}
If the carry-free skeleton contains arbitrarily long non-backtracking closed loops (i.e., there exists a family of prime cycles with \(\kappa(p)=0\) and \(\ell(p)\to\infty\)), and if there exists some positive-energy non-backtracking closed loop \(q\) (\(\kappa(q)>0\)) that can be spliced with arbitrarily long zero-energy closed loops via local rewriting into non-backtracking closed loops (preserving the tailless condition), then the average density \(\bar\kappa=\kappa/\ell\) of positive-energy closed loops can be made arbitrarily small, so the \(\Delta_K\) defined as ``minimum positive density'' is either \(0\) or has only an infimum but no minimum.
\end{proposition}

\begin{definition}[Tunneling action \texorpdfstring{$A_K$}{A_K} (alternative zero-temperature first excited state fingerprint)]\label{def:ihara-tunneling-action}

i.e., the ``minimum number of carry events in a positive-energy Ihara prime.'' This quantity is unaffected by ``elongating zero-energy segments'' and is therefore more stable than \(\Delta_K\), serving as a structural fingerprint of the first excited state at zero temperature.
\end{definition}

\begin{proposition}[Support-cut (cut-flux) formula: first-order flux when \texorpdfstring{$\kappa=1$}{kappa=1} is absent from the core interior]\label{prop:c1-cut-flux}

\[

\]
Denote the support sets
\(
\mathrm{supp}(\ell):=\{i:\ell_i>0\}
\),
\(
\mathrm{supp}(r):=\{j:r_j>0\}
\).
If \(A_1\) vanishes on the ``core\(\to\)core'' block (the structural situation of Proposition \ref{prop:no-kappa1-in-core}), then the zero-temperature first-order response coefficient satisfies
\[
\boxed{\;
c_1=\frac{\ell^{\mathsf T}A_1r}{\ell^{\mathsf T}r}
=\frac{1}{\ell^{\mathsf T}r}\sum_{i\in\mathrm{supp}(\ell)}\sum_{j\in\mathrm{supp}(r)}\ell_i\,(A_1)_{ij}\,r_j\; }.
\]
Hence \(c_1\) is contributed only by those \(\kappa=1\) jumps that \textbf{cross} from the left support to the right support, interpretable as a ``weighted cut flux.''
\end{proposition}
\begin{proof}
In general, by Proposition \ref{prop:lambda-linear-response},
\(
c_1=\ell^{\mathsf T}A_1r/(\ell^{\mathsf T}r)

Expanding the bilinear form:
\[
\ell^{\mathsf T}A_1r=\sum_{i,j}\ell_i(A_1)_{ij}r_j.
\]
Since terms with \(\ell_i=0\) or \(r_j=0\) do not contribute, the sum can be restricted to \(i\in\mathrm{supp}(\ell)\), \(j\in\mathrm{supp}(r)\).
When \(A_1\) vanishes on the core\(\to\)core block, no internal transition contributes, and the remaining terms are exactly the ``cross-cut'' edge contributions; this yields the stated formula. \qedhere
\end{proof}
