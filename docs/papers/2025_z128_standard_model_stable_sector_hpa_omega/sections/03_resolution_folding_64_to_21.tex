
\section{Resolution folding core: \texorpdfstring{$\Omega_6\to X_6$}{Omega6->X6} and the \texorpdfstring{$64\to 21$}{64->21} stable sector}
\label{sec:folding_core}

This section records the finite-resolution mathematical layer of the construction, following the companion folding manuscript \cite{Ma2025RF}.

\subsection{Microstate readout space and balanced coupling}
\label{subsec:microstate_space}

Fix a window length $m\ge 1$ and define the microstate readout space
\[
\Omega_m:=\{0,1\}^m,\qquad |\Omega_m|=2^m,
\]
with linearization $\mathcal{H}_m:=\ell^2(\Omega_m)$.
We focus on the minimal balanced coupling $m=2n$ with $n=3$, hence $m=6$:
\[
\Omega_6=\{0,1\}^6,\qquad |\Omega_6|=64.
\]
At the same order $n=3$, the Hilbert-addressed grid has $4^n=64$ sites, yielding the minimal match between spatial addressing and local readout cardinality.

\subsection{Defect operators as commuting multiplication operators}
\label{subsec:defect_operators}

Although we introduce defects first as predicates on words, it is convenient to view them as genuine operators on the finite readout Hilbert space.

\begin{definition}[Defect operators on $\ell^2(\Omega_m)$]\label{def:defect_ops}
Given a nonnegative defect function $D:\Omega_m\to\RR_{\ge 0}$, define the associated defect operator (still denoted $D$) on $\mathcal{H}_m=\ell^2(\Omega_m)$ by pointwise multiplication:
\[
(D\psi)(w):=D(w)\,\psi(w),\qquad \psi\in\ell^2(\Omega_m),\ w\in\Omega_m.
\]
If $D_1,D_2$ are defect functions, then the corresponding operators commute.
\end{definition}

In particular, the $\varphi$- and $\pi$-defects introduced below define commuting multiplication operators $D_\varphi,D_\pi$.
The stable sector is the joint kernel of the defect operators (equivalently, the subspace supported on the admissible set).

\subsection{The \texorpdfstring{$\varphi$}{phi}-channel: Zeckendorf grammar and \texorpdfstring{$|X_6|=21$}{|X6|=21}}
\label{subsec:phi_channel}

Define the golden-mean admissible set (forbidden substring ``11'')
\begin{equation}\label{eq:Xm_def}
X_m:=\{w\in\{0,1\}^m:\ w_i w_{i+1}=0\ \text{for all }i=1,\dots,m-1\}.
\end{equation}
Equivalently, define the defect function
\[
D_\varphi(w):=\ind\{w\text{ contains the substring }11\},
\]
so that $\varphi$-stability is $D_\varphi(w)=0$ and the $\varphi$-stable sector is exactly $X_m$.

\begin{lemma}[Fibonacci count of admissible words]\label{lem:xm_fib}
For all $m\ge 1$, one has $|X_m|=F_{m+2}$. In particular, $|X_6|=F_8=21$.
\end{lemma}
\begin{proof}
Let $a_m:=|X_m|$.
An admissible length-$m$ word either starts with $0$ followed by an admissible length-$(m-1)$ word, or starts with $10$ followed by an admissible length-$(m-2)$ word.
Thus $a_m=a_{m-1}+a_{m-2}$ with initial values $a_1=2$, $a_2=3$, hence $a_m=F_{m+2}$.
\end{proof}

Define the $\varphi$-stable projection $P_\varphi:\mathcal{H}_m\to \mathcal{H}_m$ as the orthogonal projection onto $\ell^2(X_m)$.
At $m=6$, this is the first rigid compression:
\[
\rank(P_\varphi)=|X_6|=21.
\]

\subsection{The \texorpdfstring{$\pi$}{pi}-channel: cyclic closure and the \texorpdfstring{$18\oplus 3$}{18+3} split}
\label{subsec:pi_channel}

At finite window length, the $\pi$-channel refines stability by imposing a cyclic wrap-around admissibility.
Define the wrap-around defect
\[
D_\pi(w):=\ind\{w_1=w_m=1\}.
\]
At $m=6$, define
\[
X_6^{\mathrm{cyc}}:=\{w\in X_6:\ w_1 w_6=0\},
\qquad
X_6^{\mathrm{bdry}}:=\{w\in X_6:\ w_1=w_6=1\}.
\]

\begin{proposition}[Canonical cyclic/boundary split at $m=6$]\label{prop:cyc_bdry_6}
One has $|X_6^{\mathrm{cyc}}|=18$ and $|X_6^{\mathrm{bdry}}|=3$, and
\[
X_6^{\mathrm{bdry}}=\{100001,\ 100101,\ 101001\}.
\]
\end{proposition}
\begin{proof}
This is an explicit finite enumeration (Appendix~\ref{app:generated_tables}); see also \cite{Ma2025RF}.
\end{proof}

\subsection{The \texorpdfstring{$\e$}{e}-channel: Artin--Mazur zeta and Abel pole barrier}
\label{subsec:e_channel}

The $\e$-channel expresses analytic stability through zeta functions and Abel normalization \cite{ArtinMazur1965,ParryPollicott1990,Ruelle1978}.
For the golden mean shift, the Artin--Mazur zeta function admits the classical rational form
\begin{equation}\label{eq:golden_zeta}
\zeta(z)=\frac{1}{\det(I-zA)}=\frac{1}{1-z-z^2},
\end{equation}
where $A$ is the $2\times 2$ transition matrix of the shift of finite type.
Let $\varphi$ be the golden ratio and apply spectral normalization $z=r/\varphi$.
Then
\begin{equation}\label{eq:abel_normalized_zeta}
\zeta_\e(r):=\zeta(r/\varphi)=\frac{1}{(1-r)\bigl(1+\varphi^{-2}r\bigr)}.
\end{equation}
The principal pole at $r=1$ lies on the boundary of the unit disk and plays the role of an analytic ``pole barrier'' for Abel-type limits.

\begin{remark}[Finite window vs.\ higher-resolution distinction]\label{rem:e_channel_distinct}
At the minimal window length $m=6$, the analytic channel is used here primarily as an \emph{interpretation layer} for stability (holomorphy domain and boundary pole), while the admissible-word selection is already enforced by the $\varphi$-grammar.
In the companion folding manuscript \cite{Ma2025RF}, one records extensions (e.g.\ weighted/pressure-like variants) in which $\e$ becomes genuinely distinct from a single forbidden-word predicate at higher resolution.
\end{remark}

\subsection{The folding map \texorpdfstring{$\mathrm{Fold}_6$}{Fold6}: from 64 indices to 21 stable types}
\label{subsec:fold6_map}

Index the microstate words by integers $\{0,\dots,2^m-1\}$ via the usual binary identification.
For $N\in\{0,\dots,63\}$, let $(c_k)$ be the Zeckendorf digits of $N$ as in \eqref{eq:zeckendorf}.
Define the length-$6$ folding map by truncation:
\begin{equation}\label{eq:fold6_def}
\mathrm{Fold}_6(N):=(c_1,\dots,c_6)\in X_6,
\end{equation}
padding by zeros if the expansion length is $<6$.

\begin{definition}[Zeckendorf value of a stable word]\label{def:zeckendorf_value}
For $w=w_1\cdots w_6\in X_6$, define its Zeckendorf value
\begin{equation}\label{eq:Vw_def}
V(w):=\sum_{k=1}^{6} w_k F_{k+1}
=w_1\cdot 1+w_2\cdot 2+w_3\cdot 3+w_4\cdot 5+w_5\cdot 8+w_6\cdot 13.
\end{equation}
\end{definition}

\begin{proposition}[Value labeling of the $21$ stable types]\label{prop:V_bijection}
The map $V:X_6\to\{0,1,\dots,20\}$ is a bijection.
\end{proposition}
\begin{proof}
Every integer $0\le M\le 20$ admits a unique Zeckendorf expansion using Fibonacci weights up to $F_7=13$, hence yields a unique digit word $(w_1,\dots,w_6)\in X_6$ with value $M$.
Conversely, every admissible word $w\in X_6$ produces an integer $V(w)\le 1+3+8+13=25$, and the explicit table in Appendix~\ref{app:fold6_full_table} shows that the $21$ values realized are exactly $\{0,\dots,20\}$.
\end{proof}

\begin{theorem}[Finite folding statistics at $m=6$]\label{thm:fold6_stats}
The map $\mathrm{Fold}_6:\{0,\dots,63\}\twoheadrightarrow X_6$ is surjective.
Every stable type has preimage size $2$, $3$, or $4$, with degeneracy histogram
\[
(|V_2|,|V_3|,|V_4|)=(8,4,9),
\]
where $V_k:=\{w\in X_6:\ |\mathrm{Fold}_6^{-1}(w)|=k\}$.
Moreover, the boundary-sector preimages are
\[
\mathrm{Fold}_6^{-1}(100001)=\{14,48\},\quad
\mathrm{Fold}_6^{-1}(100101)=\{19,53\},\quad
\mathrm{Fold}_6^{-1}(101001)=\{17,51\}.
\]
\end{theorem}
\begin{proof}
This is a finite check with closed-form characterization in \cite{Ma2025RF}; Appendix~\ref{app:generated_tables} records the full table and a reproducible script.
\end{proof}

\paragraph{Summary at minimal resolution.}
At $(m,n)=(6,3)$, the three-channel stability mechanism yields an explicit, auditable chain:
\[
\Omega_6\ (64\ \text{microstates})
\xrightarrow[\text{$\varphi$-projection}]{\text{forbidden grammar}}
X_6\ (21\ \text{stable types})
\xrightarrow[\text{$\pi$-split}]{\text{cyclic closure}}
X_6^{\mathrm{cyc}}\oplus X_6^{\mathrm{bdry}}\ (18\oplus 3).
\]
This $64\to 21$ and $18\oplus 3$ structure is not a metaphor but a finite combinatorial theorem with explicit tables.
The remainder of the paper treats this rigid finite skeleton as the stable-sector substrate for Standard Model interface hypotheses.
