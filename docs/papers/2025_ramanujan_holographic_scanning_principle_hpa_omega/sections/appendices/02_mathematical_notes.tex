\section{Mathematical notes (proof sketches)}
\label{app:math_notes}

\subsection{Hecke prime-power recursion and prime generation}
\label{app:hecke_recursion}

Starting from the multiplication relation \eqref{eq:hecke_multiplication}, take $m=p^r$ and $n=p$ for a prime $p$. Using the divisors $d\mid (p^r,p)$, i.e.\ $d\in\{1,p\}$, one obtains
\begin{equation}
T_{p^r}T_p = T_{p^{r+1}} + p^{k-1}T_{p^{r-1}},
\end{equation}
which rearranges to the prime-power recursion \eqref{eq:hecke_prime_power}:
\begin{equation}
T_{p^{r+1}} = T_p T_{p^r} - p^{k-1}T_{p^{r-1}}.
\end{equation}
This shows all $T_{p^r}$ are generated by $T_p$, and together with \eqref{eq:hecke_multiplication} it implies that the full Hecke algebra is generated by $\{T_p\}_{p\ \mathrm{prime}}$.

\subsection{Weyl pairs and intrinsic incompatibility}
\label{app:weyl_incompatibility}

Suppose $UV=\e^{2\pi\iu\alpha}VU$ with $\alpha\notin\QQ$. If $\psi$ were a common eigenvector, $U\psi=\lambda\psi$ and $V\psi=\mu\psi$, then
\begin{equation}
UV\psi = \lambda\mu \psi,\qquad VU\psi=\mu\lambda\psi,
\end{equation}
but the Weyl relation forces $\lambda\mu = \e^{2\pi\iu\alpha}\mu\lambda$, hence $(1-\e^{2\pi\iu\alpha})\lambda\mu=0$. Since $\alpha\notin\QQ$, $\e^{2\pi\iu\alpha}\ne 1$, so $\lambda\mu=0$, impossible for unitary eigenvalues. Thus no nonzero vector can be a simultaneous eigenvector of $U$ and $V$.

\subsection{Why local admissibility constraints matter}
\label{app:local_constraints}

Ostrowski/Zeckendorf representations impose local admissibility rules (e.g.\ Zeckendorf's ``no adjacent $1$''). These rules turn a global integer representation into a locally checkable grammar. In finite-resolution and quasi-local settings, this locality is essential: a readout protocol that cannot be implemented by local checks is not physically actionable for a finite observer sector.

\subsection{Weyl equidistribution and induced frequencies}
\label{app:weyl_equidistribution}

Let $\alpha\notin\QQ$ and consider the irrational rotation on $\RR/\ZZ$,
\begin{equation}
x_t = x_0 + t\alpha \pmod 1.
\end{equation}
Weyl's equidistribution theorem states that the orbit is uniformly distributed modulo $1$ \cite{Weyl1916,KuipersNiederreiter1974}. In particular, for any interval $I\subset \RR/\ZZ$,
\begin{equation}
\frac{1}{N}\#\{0\le t\le N-1: x_t\in I\}\to |I|\qquad (N\to\infty),
\end{equation}
where $|I|$ denotes Lebesgue measure. This gives a protocol-level justification of induced frequencies: window readout asymptotically induces the Lebesgue measure.

\subsection{Denjoy--Koksma inequality (quantitative control at convergents)}
\label{app:denjoy_koksma}

Let $\alpha\notin\QQ$ and let $p_n/q_n$ be the convergents of its continued fraction. For a function $f$ of bounded variation on $\RR/\ZZ$, the Denjoy--Koksma inequality states that for all $x\in\RR/\ZZ$,
\begin{equation}
\left|\sum_{t=0}^{q_n-1} f(x+t\alpha) - q_n\int_{\RR/\ZZ} f\,\dd x\right|
\le \mathrm{Var}(f),
\end{equation}
see \cite{Denjoy1932,Koksma1935,KatokHasselblatt1995}. For an interval indicator $f=\mathbf{1}_I$, one has $\mathrm{Var}(f)=2$, hence the deviation of visit counts at length $q_n$ is uniformly bounded by $2$.

\subsection{Koksma inequality: discrepancy controls readout error}
\label{app:koksma_inequality}

Let $P=\{x_0,\ldots,x_{N-1}\}\subset[0,1)$ be a finite point set and let $f:[0,1]\to\RR$ be of bounded variation $\mathrm{Var}(f)$. The one-dimensional Koksma inequality states that
\begin{equation}
\left|\frac{1}{N}\sum_{i=0}^{N-1} f(x_i) - \int_0^1 f(x)\,\dd x\right|
\le \mathrm{Var}(f)\, D_N^*(P),
\end{equation}
where $D_N^*(P)$ is the star discrepancy of $P$ \cite{KuipersNiederreiter1974}.
For window readout $f=\mathbf{1}_I$ (an interval indicator), $\mathrm{Var}(f)=2$, hence
\begin{equation}
\left|\frac{1}{N}\sum_{i=0}^{N-1} \mathbf{1}_I(x_i) - |I|\right|\le 2\, D_N^*(P).
\end{equation}
This provides a direct quantitative bound that upgrades ``induced probability'' to a finite-$N$ error-controlled statement.

\subsection{Ostrowski block decomposition: discrepancy controlled by coding digits}
\label{app:ostrowski_discrepancy}

Let $\alpha\notin\QQ$ and let $p_n/q_n$ be the convergents of its continued fraction. Every integer $N\ge 1$ admits an Ostrowski expansion
\begin{equation}
N=\sum_{n=0}^{m} b_n q_n,
\end{equation}
with digits $b_n$ satisfying standard admissibility constraints (Section~\ref{sec:ostrowski}). This expansion yields a natural block decomposition of the first $N$ scan ticks into $\sum_n b_n$ blocks of convergent lengths.

\begin{proposition}[Bounded-variation error bound via Ostrowski digits]
\label{prop:ostrowski_birkhoff_bound}
Let $f$ have bounded variation $\mathrm{Var}(f)$ on $\RR/\ZZ$. Then for any $x\in\RR/\ZZ$,
\begin{equation}
\left|\sum_{t=0}^{N-1} f(x+t\alpha) - N\int_{\RR/\ZZ} f\,\dd x\right|
\le \mathrm{Var}(f)\sum_{n=0}^{m} b_n.
\end{equation}
\end{proposition}

\begin{proof}
Write $N=\sum_{n=0}^{m} b_n q_n$ and partition the interval of ticks $\{0,1,\ldots,N-1\}$ into consecutive blocks of lengths $q_n$, repeated $b_n$ times in any order consistent with the sum.
Since the Denjoy--Koksma inequality (Appendix~\ref{app:denjoy_koksma}) is uniform in the starting point, each block of length $q_n$ contributes an error of at most $\mathrm{Var}(f)$ relative to its expected value $q_n\int f$.
Summing over all blocks gives the stated bound.
\end{proof}

\paragraph{Consequence for star discrepancy.}
Taking $f=\mathbf{1}_I$ (interval indicator) gives $\mathrm{Var}(f)=2$, hence for every interval $I$,
\begin{equation}
\left|\frac{1}{N}\sum_{t=0}^{N-1}\mathbf{1}_I(x+t\alpha)-|I|\right|
\le \frac{2}{N}\sum_{n=0}^{m} b_n,
\end{equation}
so
\begin{equation}
D_N^*\!\left(P_N(\alpha)\right)\le \frac{2}{N}\sum_{n=0}^{m} b_n.
\end{equation}
Thus, at finite $N$, discrepancy is quantitatively controlled by the \emph{digit sum} of the canonical Ostrowski encoding of tick time.

\paragraph{Golden specialization.}
For $\alpha=\varphi^{-1}$, the Ostrowski system degenerates to Zeckendorf: $q_n=F_n$ and $b_n\in\{0,1\}$ with no adjacent $1$'s. Then $\sum b_n$ is exactly the Zeckendorf Hamming weight $w_Z(N)$, yielding
\begin{equation}
D_N^*\le \frac{2\,w_Z(N)}{N}.
\end{equation}
Since the largest Fibonacci index used in the Zeckendorf decomposition satisfies $F_k\le N< F_{k+1}$ and the no-adjacency constraint gives $w_Z(N)\le (k+1)/2$, a crude explicit bound is
\begin{equation}
D_N^*\le \frac{3+\log_\varphi N}{N},
\end{equation}
using $F_k\ge \varphi^{k-2}$ for $k\ge 2$.

\paragraph{Bounded partial quotients (constant type).}
More generally, if $\alpha$ has bounded continued-fraction coefficients $a_n\le A$, then Ostrowski digits satisfy $b_n\le a_{n+1}\le A$, hence $\sum_{n=0}^{m} b_n\le A(m+1)$. Since $q_{n+1}=a_{n+1}q_n+q_{n-1}\ge q_n+q_{n-1}$, one has $q_n\ge F_n$ and therefore $m\le 1+\log_\varphi N$ whenever $q_m\le N<q_{m+1}$. Combining these gives an explicit bound
\begin{equation}
D_N^*\le \frac{2A(2+\log_\varphi N)}{N},
\end{equation}
which recovers the standard $O((\log N)/N)$ discrepancy estimate for constant-type rotations (cf.\ \cite{KuipersNiederreiter1974}).

\subsection{The three-distance (three-gap) theorem}
\label{app:three_gap}

For $\alpha\notin\QQ$ and $N\ge 1$, the sorted set of points $\{t\alpha\}\ (t=0,\ldots,N-1)$ partitions the circle into $N$ gaps whose lengths take at most three distinct values (the three-distance theorem). A classical reference is \cite{Slater1967}.

\subsection{Exact star discrepancy at convergent lengths (closed form)}
\label{app:convergent_star_discrepancy}

For a point set $P=\{x_0,\ldots,x_{N-1}\}\subset[0,1)$, the one-dimensional star discrepancy is
\begin{equation}
D_N^*(P)=\sup_{t\in[0,1]}\left|\frac{1}{N}\#\{x_i<t\}-t\right|.
\end{equation}

\begin{proposition}[Convergent-length star discrepancy for a Kronecker orbit]
\label{prop:convergent_star_discrepancy}
Let $\alpha\in(0,1)\setminus\QQ$ and let $p/q$ be a reduced rational with
\begin{equation}
0<\left|\alpha-\frac{p}{q}\right|<\frac{1}{q^2}.
\end{equation}
Consider the Kronecker point set
\begin{equation}
P_q(\alpha):=\bigl\{\{t\alpha\}: t=0,1,\ldots,q-1\bigr\}\subset[0,1).
\end{equation}
Then
\begin{equation}
D_q^*\!\left(P_q(\alpha)\right)=
\begin{cases}
\dfrac{1}{q}, & \dfrac{p}{q}<\alpha,\\[6pt]
\dfrac{1}{q}+(q-1)\!\left(\dfrac{p}{q}-\alpha\right), & \dfrac{p}{q}>\alpha.
\end{cases}
\end{equation}
\end{proposition}

\begin{proof}
Write $\delta:=\alpha-\frac{p}{q}$ and note that for $0\le t\le q-1$ one has $|t\delta|<(q-1)/q^2<1/q$. Let $r_t\in\{0,1,\ldots,q-1\}$ be the residue $r_t\equiv tp\ (\mathrm{mod}\ q)$; since $\gcd(p,q)=1$, the map $t\mapsto r_t$ is a permutation.

\paragraph{Case 1: $\delta>0$ (i.e.\ $p/q<\alpha$).}
For each $t$,
\begin{equation}
\{t\alpha\}=\left\{\frac{tp}{q}+t\delta\right\}= \frac{r_t}{q}+t\delta \in \left[\frac{r_t}{q},\frac{r_t+1}{q}\right),
\end{equation}
where the last inclusion uses $0\le t\delta<1/q$ and $r_t=q-1$ is interpreted modulo $1$. Hence each interval $[j/q,(j+1)/q)$ contains exactly one point of $P_q(\alpha)$. This implies $D_q^*(P_q(\alpha))\le 1/q$. On the other hand, since $0\in P_q(\alpha)$, one has $D_q^*(P_q(\alpha))\ge 1/q$ (take $t\downarrow 0$), so equality holds.

\paragraph{Case 2: $\delta<0$ (i.e.\ $p/q>\alpha$).}
Let $\delta':=-\delta=\frac{p}{q}-\alpha>0$. For $t\ge 1$, $r_t\neq 0$ so $r_t/q\ge 1/q$, and
\begin{equation}
\{t\alpha\}=\left\{\frac{tp}{q}-t\delta'\right\}=\frac{r_t}{q}-t\delta' \in \left(\frac{r_t-1}{q},\frac{r_t}{q}\right),
\end{equation}
since $0<t\delta'<1/q$. Thus every interval $[j/q,(j+1)/q)$ with $j=0,\ldots,q-2$ contains exactly one point coming from the unique $t\in\{1,\ldots,q-1\}$ with $r_t=j+1$, while the point $0\in P_q(\alpha)$ lies in $[0,1/q)$.

Ordering points by increasing bin index $j$, we can write the sorted list as
\begin{equation}
x_0=0,\qquad x_{j+1}=\frac{j+1}{q}-t(j)\delta',\qquad j=0,1,\ldots,q-2,
\end{equation}
where $t(j)\in\{1,\ldots,q-1\}$ is the unique integer satisfying $t(j)p\equiv j+1\ (\mathrm{mod}\ q)$. Then
\begin{equation}
\frac{j+2}{q}-x_{j+1}=\frac{1}{q}+t(j)\delta'.
\end{equation}
Taking the maximum over $j$ and using $\max_j t(j)=q-1$, we obtain
\begin{equation}
D_q^*(P_q(\alpha))=\max_j\left(\frac{j+2}{q}-x_{j+1}\right)=\frac{1}{q}+(q-1)\delta'
\end{equation}
as claimed.
\end{proof}

\subsection{Golden branch specialization and the \texorpdfstring{$1+1/\sqrt{5}$}{1+1/sqrt(5)} constant}
\label{app:golden_constant}

Let $\varphi=(1+\sqrt{5})/2$ and $\alpha=\varphi^{-1}$. The convergents of $\alpha$ are ratios of Fibonacci numbers:
\begin{equation}
\frac{p_n}{q_n}=\frac{F_{n-1}}{F_n}.
\end{equation}
Using Binet's formula $F_n=(\varphi^n-\psi^n)/\sqrt{5}$ with $\psi=(1-\sqrt{5})/2=-\varphi^{-1}$, one obtains the exact approximation error
\begin{equation}
\alpha-\frac{F_{n-1}}{F_n}=\frac{(-1)^{n}}{\sqrt{5}\,F_n^2}.
\end{equation}
Combining this with Proposition~\ref{prop:convergent_star_discrepancy} yields an exact closed form for $D_{F_n}^*$ and, in particular, for even $n$,
\begin{equation}
F_n\,D_{F_n}^* = 1+\left(1-\frac{1}{F_n}\right)\frac{1}{\sqrt{5}}
\;\longrightarrow\;
1+\frac{1}{\sqrt{5}}.
\end{equation}

\subsection{Deligne's bound for Ramanujan \texorpdfstring{$\tau(p)$}{tau(p)}}
\label{app:deligne_bound}

For a normalized holomorphic Hecke eigenform of weight $k$, Deligne proved the Ramanujan--Petersson bound
\begin{equation}
|a_p|\le 2p^{(k-1)/2}\qquad (p\ \text{prime}),
\end{equation}
as a consequence of the Weil conjectures \cite{Deligne1974}. For the discriminant form $\Delta$ of weight $12$, this specializes to
\begin{equation}
|\tau(p)|\le 2p^{11/2}.
\end{equation}

\subsection{Prime powers and all integers: a global growth bound for \texorpdfstring{$\tau(n)$}{tau(n)}}
\label{app:tau_global_bound}

For $\Delta$ (weight $12$), the Hecke recursion at prime powers reads
\begin{equation}
\tau(p^{r+1})=\tau(p)\tau(p^r)-p^{11}\tau(p^{r-1}).
\end{equation}
Define the normalized sequence
\begin{equation}
b_r:=\frac{\tau(p^r)}{p^{11r/2}},\qquad r\ge 0.
\end{equation}
Then $b_0=1$, $b_1=\tau(p)/p^{11/2}$, and the recursion becomes
\begin{equation}
b_{r+1}=b_1 b_r-b_{r-1}.
\end{equation}
Under Deligne's bound $|b_1|\le 2$ (Appendix~\ref{app:deligne_bound}), we can make the growth control completely explicit via Chebyshev polynomials of the second kind. Let $U_r$ be defined by
\begin{equation}
U_0(t)=1,\qquad U_1(t)=2t,\qquad U_{r+1}(t)=2t\,U_r(t)-U_{r-1}(t).
\end{equation}
Setting $t=b_1/2$ gives the same recurrence and initial data as $\{b_r\}$, hence
\begin{equation}
b_r = U_r\!\left(\frac{b_1}{2}\right).
\end{equation}
If $|b_1|\le 2$, write $b_1=2\cos\theta$ for some $\theta\in[0,\pi]$. Then the standard identity
\begin{equation}
U_r(\cos\theta)=\frac{\sin((r+1)\theta)}{\sin\theta}
\end{equation}
implies $|U_r(\cos\theta)|\le r+1$ (use $|\sin((r+1)\theta)|\le (r+1)|\sin\theta|$ and continuity at $\theta=0,\pi$). Therefore
\begin{equation}
|b_r|\le r+1,
\end{equation}
and hence
\begin{equation}
|\tau(p^r)|\le (r+1)\,p^{11r/2}.
\end{equation}
Using multiplicativity of $\tau(n)$ on coprime indices and writing $n=\prod_i p_i^{r_i}$, we obtain the global bound
\begin{equation}
|\tau(n)|\le \prod_i (r_i+1)\,p_i^{11r_i/2} = d(n)\,n^{11/2},
\end{equation}
where $d(n)=\prod_i (r_i+1)$ is the divisor function. In particular, this yields the standard growth form $|\tau(n)|=O(n^{11/2+\varepsilon})$ for any $\varepsilon>0$.

\subsection{Tail bounds for Eisenstein \texorpdfstring{$q$}{q}-series (for numerical error control)}
\label{app:eisenstein_tail_bounds}

Let $q=\e^{2\pi\iu\tau}$ with $|q|=r<1$. The Eisenstein series have absolutely convergent $q$-expansions
\begin{equation}
E_4(\tau)=1+240\sum_{n\ge 1}\sigma_3(n)q^n,\qquad
E_6(\tau)=1-504\sum_{n\ge 1}\sigma_5(n)q^n.
\end{equation}
Using the bound $\sigma_k(n)\le \zeta(k)\,n^k$ for $k>1$ (indeed $\sigma_k(n)=\sum_{d\mid n}d^k\le n^k\sum_{m\ge 1}m^{-k}=\zeta(k)\,n^k$), one obtains explicit truncation bounds. For example, for the $N$-term truncation $E_4^{(N)}(\tau):=1+240\sum_{n=1}^{N}\sigma_3(n)q^n$,
\begin{equation}
|E_4(\tau)-E_4^{(N)}(\tau)|
\le 240\,\zeta(3)\sum_{n>N} n^3 r^n.
\end{equation}
The tail sum can be written as
\begin{equation}
\sum_{n>N} n^3 r^n
=r^{N+1}\sum_{m\ge 0}(m+N+1)^3 r^m,
\end{equation}
and expanded using the standard identities for $\sum_{m\ge 0} m^\ell r^m$ ($\ell=0,1,2,3$). Analogous bounds hold for $E_6$ with $\zeta(5)\sum_{n>N} n^5 r^n$.

These tail bounds justify the qualitative statement used in Section~\ref{sec:quant_j_invariant}: truncated $q$-series evaluations of $E_4,E_6$ exhibit essentially geometric decay in $N$ at fixed $\tau$, with constants depending on $r$ and polynomial factors in $N$.

\subsection{Certified truncation bounds for the \texorpdfstring{$j$}{j}-invariant}
\label{app:j_truncation_bounds}

Fix $\tau\in\HH$ and write $q=\e^{2\pi\iu\tau}$, $r=|q|<1$. Let $E_4^{(N)}(\tau),E_6^{(N)}(\tau)$ denote truncations at depth $N$ and let the corresponding tail bounds be
\begin{equation}
|E_4(\tau)-E_4^{(N)}(\tau)|\le \varepsilon_4(\tau;N),\qquad
|E_6(\tau)-E_6^{(N)}(\tau)|\le \varepsilon_6(\tau;N),
\end{equation}
where $\varepsilon_4,\varepsilon_6$ are obtained from the inequalities in Appendix~\ref{app:eisenstein_tail_bounds}.

Define
\begin{equation}
D(\tau):=E_4(\tau)^3-E_6(\tau)^2,\qquad
D^{(N)}(\tau):=E_4^{(N)}(\tau)^3-E_6^{(N)}(\tau)^2.
\end{equation}
Let $M_4:=|E_4^{(N)}(\tau)|+\varepsilon_4(\tau;N)$ and $M_6:=|E_6^{(N)}(\tau)|+\varepsilon_6(\tau;N)$. Then
\begin{equation}
|D(\tau)-D^{(N)}(\tau)|
\le 3M_4^2\,\varepsilon_4(\tau;N) + 2M_6\,\varepsilon_6(\tau;N)
\;=:\;\delta_D(\tau;N).
\end{equation}
Consequently, if $|D^{(N)}(\tau)|>\delta_D(\tau;N)$, then $D(\tau)\neq 0$ and
\begin{equation}
|D(\tau)|\ge |D^{(N)}(\tau)|-\delta_D(\tau;N)=:D_{\min}(\tau;N)>0.
\end{equation}

Since $j(\tau)=1728\,E_4(\tau)^3/D(\tau)$, one obtains the certified truncation bound
\begin{equation}
|j(\tau)-j^{(N)}(\tau)|
\le 1728\left(
\frac{3M_4^2\,\varepsilon_4(\tau;N)}{D_{\min}(\tau;N)}
\;+\;
|E_4^{(N)}(\tau)|^3\frac{\delta_D(\tau;N)}{D_{\min}(\tau;N)\,|D^{(N)}(\tau)|}
\right),
\end{equation}
where $j^{(N)}$ is computed from the truncated Eisenstein series.

\paragraph{Remark on floating-point saturation.}
The bound above controls truncation error. In finite-precision arithmetic, observed differences can plateau at a numerical floor once truncation becomes negligible; this is visible in Table~\ref{tab:j_invariant_truncation}.

\subsection{Petersson inner product and self-adjointness of Hecke operators}
\label{app:petersson_selfadjoint}

Let $S_k(\PSL_2(\ZZ))$ be the space of weight-$k$ cusp forms for $\PSL_2(\ZZ)$. The Petersson inner product is
\begin{equation}
\langle f,g\rangle
:=
\int_{\PSL_2(\ZZ)\backslash\HH} f(\tau)\,\overline{g(\tau)}\,(\Im\tau)^k\,\frac{\dd x\,\dd y}{y^2},
\qquad \tau=x+\iu y,
\end{equation}
which converges for cusp forms. For level~$1$, the Hecke operator $T_n$ admits the classical explicit formula
\begin{equation}
(T_n f)(\tau)
=
n^{k-1}\sum_{\substack{ad=n\\ d>0}}\;\sum_{b\!\!\!\pmod d}
d^{-k}\,
f\!\left(\frac{a\tau+b}{d}\right),
\end{equation}
see standard references such as \cite{Apostol1990,DiamondShurman2005}.

\paragraph{Self-adjointness (standard).}
With respect to the Petersson product, the Hecke operators are self-adjoint on cusp forms:
\begin{equation}
\langle T_n f, g\rangle = \langle f, T_n g\rangle,
\end{equation}
and the family $\{T_n\}$ commutes. A standard proof is by inserting the explicit formula for $T_n$ into the integral and applying a change of variables using the $\PSL_2(\ZZ)$-invariance of the hyperbolic measure $\dd x\,\dd y/y^2$ together with the automorphy factors (see \cite{DiamondShurman2005}).

\paragraph{Diagonalization consequence.}
As commuting self-adjoint operators on the finite-dimensional space $S_k(\PSL_2(\ZZ))$, the Hecke operators admit an orthogonal basis of simultaneous eigenforms. This justifies the ``diagonalization frame'' language used in Section~\ref{sec:eigenforms} and in the interpretation-layer discussion (Section~\ref{sec:diagonalization_perspective}).

\subsection{Abel regularization for rotation-orbit sums (R1)}
\label{app:abel_regularization}

R1 fixes a canonical regulated-to-continuum convention: \emph{Abel first, then limit}. In the rotation setting of O5, this can be stated in a clean, auditable form.

\paragraph{Abel means.}
Let $\alpha\notin\QQ$ and let $f:\RR/\ZZ\to\CC$ be continuous. For $0<r<1$, define the Abel-weighted orbit average
\begin{equation}
A_r(f;x)
:=
(1-r)\sum_{t\ge 0} r^t\, f(x+t\alpha),
\qquad x\in\RR/\ZZ.
\end{equation}
Then
\begin{equation}
\lim_{r\uparrow 1} A_r(f;x)=\int_{\RR/\ZZ} f(u)\,\dd u
\end{equation}
uniformly in $x$. A standard proof is by Fourier modes: for $f_m(x)=\e^{2\pi\iu m x}$ one has
\begin{equation}
A_r(f_m;x)
=(1-r)\e^{2\pi\iu m x}\sum_{t\ge 0}\left(r\,\e^{2\pi\iu m\alpha}\right)^t
=
(1-r)\e^{2\pi\iu m x}\frac{1}{1-r\,\e^{2\pi\iu m\alpha}},
\end{equation}
which tends to $0$ for $m\neq 0$ and equals $1$ for $m=0$. Trigonometric polynomials follow by linearity and density in $C(\RR/\ZZ)$. This is a classical Abel summability mechanism; see \cite{Hardy1949}.

\paragraph{Finite parts for orbit traces.}
If $\mu:=\int f$ is the mean, the unnormalized Abel sum has the canonical decomposition
\begin{equation}
\sum_{t\ge 0} r^t f(x+t\alpha)
\;=\;
\frac{\mu}{1-r}
\;+\;
\sum_{t\ge 0} r^t\bigl(f(x+t\alpha)-\mu\bigr),
\end{equation}
where the first term carries the universal divergence as $r\uparrow 1$. R1 selects the \emph{finite part} (when it exists) as
\begin{equation}
\mathrm{FP}(f;x)
:=
\lim_{r\uparrow 1}\sum_{t\ge 0} r^t\bigl(f(x+t\alpha)-\mu\bigr).
\end{equation}
For a trigonometric polynomial $f(x)=\sum_m \widehat f_m \e^{2\pi\iu m x}$, one obtains the explicit finite part
\begin{equation}
\mathrm{FP}(f;x)
=
\sum_{m\neq 0}\widehat f_m\,\frac{\e^{2\pi\iu m x}}{1-\e^{2\pi\iu m\alpha}},
\end{equation}
which shows how Abel regularization canonically resolves the constant-mode divergence while preserving the nontrivial phase dependence.

\subsection{Formal power-series integrality: why \texorpdfstring{$j(q)$}{j(q)} has integer coefficients}
\label{app:j_integrality_formal}

\begin{lemma}[unit-constant denominator implies integrality]
\label{lem:unit_constant_integrality}
Let $A(q),B(q)\in \ZZ[[q]]$ be formal power series with $B(0)=1$. Then there exists a unique $C(q)\in\ZZ[[q]]$ such that
\begin{equation}
A(q)=B(q)\,C(q),
\end{equation}
equivalently $C(q)=A(q)/B(q)$ in the ring $\ZZ[[q]]$.
\end{lemma}

\begin{proof}
Write $A(q)=\sum_{n\ge 0} a_n q^n$, $B(q)=\sum_{n\ge 0} b_n q^n$ with $b_0=1$, and seek $C(q)=\sum_{n\ge 0} c_n q^n$. The identity $A=BC$ is equivalent to the coefficient recursion
\begin{equation}
a_n=\sum_{k=0}^n b_k c_{n-k}.
\end{equation}
For $n=0$ this gives $c_0=a_0$. For $n\ge 1$, solve uniquely as
\begin{equation}
c_n=a_n-\sum_{k=1}^n b_k c_{n-k},
\end{equation}
which is an integer since all $a_n,b_k,c_{n-k}\in\ZZ$. Uniqueness follows from the recursion.
\end{proof}

\paragraph{Application to modular objects.}
For level~$1$, the Eisenstein series $E_4,E_6$ have $q$-expansions with integer coefficients and constant term $1$, and $\Delta=(E_4^3-E_6^2)/1728$ satisfies $\Delta(q)=q+O(q^2)$ with integer coefficients \cite{Serre1973,Apostol1990}. Write $\Delta(q)=q\,\Delta_1(q)$ with $\Delta_1(0)=1$ and $\Delta_1\in\ZZ[[q]]$. Then
\begin{equation}
j(q)=1728\,\frac{E_4(q)^3}{E_4(q)^3-E_6(q)^2}
=\frac{E_4(q)^3}{\Delta(q)}
=q^{-1}\,\frac{E_4(q)^3}{\Delta_1(q)},
\end{equation}
and Lemma~\ref{lem:unit_constant_integrality} implies $E_4^3/\Delta_1\in\ZZ[[q]]$, hence $j(q)\in q^{-1}\ZZ[[q]]$.
Appendix~\ref{app:quant_fits_qexp_integrality} records a reproducible coefficient sanity check (Experiment~16).
