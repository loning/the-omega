\subsubsection{加权核的压力与大偏差（可审计接口）}
令 $\phi$ 为同步图边上的可观测量（例如输出位 $e$ 的指示函数，或"到达含 $\overline{1}$ 状态"的指示函数）。定义加权矩阵
\[
(B_\theta)_{ij}:=\sum_{e:i\to j}\exp\bigl(\theta\,\phi(e)\bigr),
\]
其中求和遍历所有从 $i$ 到 $j$ 的边。记
\[
\lambda(\theta):=\rho(B_\theta),\qquad P(\theta):=\log\lambda(\theta).
\]
则 $P(\theta)$ 为有限型情形的压力函数，且 $P(0)=\log 3$。令
\[
I(\alpha):=\sup_{\theta\in\RR}\bigl(\theta\alpha-(P(\theta)-P(0))\bigr),
\]
则 $I(\alpha)$ 给出 $\phi$ 的长程平均的大偏差率函数（标准热力学形式口径，见 \cite{Ruelle1978Thermodynamic,ParryPollicott1990Zeta}）。

加权 $\zeta$ 仍由
\[
\zeta_{B,\theta}(z)=\frac{1}{\det(I-zB_\theta)}
\]
给出，并可定义加权 primitive 轨道谱
\[
a_n(\theta):=\mathrm{Tr}(B_\theta^n),\qquad
p_n(\theta)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}(d\theta),
\]
以及相应的加权 Mertens 常数族（对 $z\to 1/\lambda(\theta)$ 取 Abel 有限部）。
可审计数值由脚本 \texttt{scripts/exp\_sync\_kernel\_weighted\_pressure.py} 输出。

\subsubsection{加权 primitive--periodic 的 Witt 分解}\label{app:witt-weighted}
以下取输出位势 $\varphi(e)=e$，令
\[
u=e^\theta,\qquad B(u)=B_0+uB_1,\qquad \Delta(z,u):=\det(I-zB(u)).
\]
其中 $B_0$（resp.\ $B_1$）只计数输出位 $e=0$（resp.\ $e=1$）的边。
则
\[
\zeta(z,u)=\exp\!\left(\sum_{n\ge 1}\frac{\mathrm{Tr}(B(u)^n)}{n}z^n\right)=\Delta(z,u)^{-1}.
\]
令
\[
a_n(u):=\mathrm{Tr}(B(u)^n),\qquad
P(z,u):=\sum_{n\ge 1}p_n(u)z^n,
\]
其中
\(
p_n(u)=\sum_{\tau\ \mathrm{primitive},\,|\tau|=n}u^{S_\tau\varphi}
\)
为 primitive 轨道的加权谱多项式。Euler 乘积给出
\[
\zeta(z,u)=\prod_{\tau\ \mathrm{primitive}}\bigl(1-z^{|\tau|}u^{S_\tau\varphi}\bigr)^{-1}.
\]
取对数并展开幂重复得到 Witt 分解
\[
\log\zeta(z,u)=\sum_{m\ge 1}\frac{1}{m}P(z^m,u^m).
\]
由 Möbius 反演得到
\[
P(z,u)=\sum_{m\ge 1}\frac{\mu(m)}{m}\log\zeta(z^m,u^m),
\]
对系数比较即有
\[
p_n(u)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}(u^d)
\quad\Longleftrightarrow\quad
p_n(\theta)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}(d\theta).
\]

\subsubsection{Big-Witt/necklace 桥包与可检验同余}\label{app:witt-bridge-package}
上式不仅是“可计算公式”，而是 Big-Witt（necklace ring）上的标准坐标变换：$a_n(u)$ 为 ghost 分量，$p_n(u)$ 为 Witt 坐标，而 Euler 乘积是 plethystic exponential。等价地，对任意带参矩阵族 $T(u)$，有严格恒等式
\[
\boxed{\ \zeta_T(z,u)=\mathrm{Exp}_W\bigl(\mathrm{Mob}_W(\mathrm{TrSeq}(T))\bigr)\ }
\]
其中 $\mathrm{TrSeq}(T)=(a_n(u))_{n\ge 1}$、$\mathrm{Mob}_W(a)=p$ 给出 $p_n(u)=\frac1n\sum_{d\mid n}\mu(d)\,a_{n/d}(u^d)$，而 $\mathrm{Exp}_W$ 由
\(
\log\zeta=\sum_{m\ge 1}\frac{1}{m}P(z^m,u^m)
\)
定义。该“桥包”直接产出一批可检验算术约束：

\begin{proposition}[necklace 整除律]\label{prop:witt-necklace-div}
对任意 $n\ge 1$，有
\[
n\ \Big|\ \sum_{d\mid n}\mu(d)\,a_{n/d}(u^d)\qquad\text{于}\ \mathbb{Z}[u].
\]
\end{proposition}
\begin{proof}
由于 $p_n(u)\in\mathbb{Z}[u]$ 是 primitive 轨道的加权计数，多项式恒在整系数环内；由 $n\,p_n(u)=\sum_{d\mid n}\mu(d)a_{n/d}(u^d)$ 得整除结论。证毕。
\end{proof}

\begin{corollary}[$p$-幂 Dwork 型同余]\label{cor:witt-dwork-congruence}
取 $n=p^k$（素数 $p$），则
\[
a_{p^k}(u)\equiv a_{p^{k-1}}(u^p)\pmod{p^k}\qquad\text{于}\ \mathbb{Z}[u].
\]
\end{corollary}
\begin{proof}
由 $\mu(1)=1,\mu(p)=-1,\mu(p^2)=0$ 得
\(
p^k\,p_{p^k}(u)=a_{p^k}(u)-a_{p^{k-1}}(u^p)
\)。
利用命题 \ref{prop:witt-necklace-div} 即得结论。证毕。
\end{proof}

\subsubsection{同步核的自对偶对称}\label{app:kernel-self-dual}
对同步核存在状态对合 $\iota$，其作用为
\[
\iota:\quad
000\leftrightarrow 101,\quad
001\leftrightarrow 100,\quad
002\leftrightarrow 010,\quad
0\overline{1}2\leftrightarrow 11\overline{1},\quad
1\overline{1}2\leftrightarrow 01\overline{1}.
\]
令 $\Pi$ 为该置换的矩阵，则
\[
\Pi^{-1}B_0\Pi=B_1,\qquad \Pi^{-1}B_1\Pi=B_0,
\]
从而
\[
\Pi^{-1}B(u)\Pi=uB(1/u).
\]
于是
\[
\Delta(z,u)=\det(I-zB(u))=\det(I-uzB(1/u))=\Delta(uz,1/u).
\]
设 $z_\star(u)$ 为最小正根、$\lambda(u)=1/z_\star(u)$，则
\[
\lambda(u)=u\,\lambda(1/u),
\qquad
P(\theta)=\theta+P(-\theta).
\]
对按输出 1 次数分层的 primitive 轨道数 $p_{n,k}$，有
\[
p_{n,k}=p_{n,n-k}\quad(n\ge 2),
\]
等价地 $u^{-1}p_n(u)$ 为次数 $n-2$ 的回文多项式。

\begin{proposition}[周期迹多项式的回文对称]\label{prop:trace-palindrome}
记
\[
a_n(u):=\Tr(B(u)^n)=\sum_{k=0}^{n}a_{n,k}u^k.
\]
则对所有 $n\ge 1$ 有
\[
a_n(u)=u^n a_n(1/u)
\quad\Longleftrightarrow\quad
a_{n,k}=a_{n,n-k}.
\]
\end{proposition}

\begin{proof}
由自对偶 $\Pi^{-1}B(u)\Pi=uB(1/u)$ 与迹不变性，
\[
a_n(u)=\Tr(B(u)^n)=\Tr\!\bigl((uB(1/u))^n\bigr)=u^n\Tr(B(1/u)^n)=u^n a_n(1/u),
\]
比较系数即得结论。证毕。
\end{proof}

\begin{corollary}[奇阶消失律]\label{cor:trace-odd-vanish}
若 $n$ 为奇数，则
\[
a_n(-1)=0,
\qquad
(u+1)\mid a_n(u)\ \ \text{于}\ \mathbb{Z}[u].
\]
\end{corollary}
\begin{proof}
由命题 \ref{prop:trace-palindrome}，$a_n(u)=u^n a_n(1/u)$。取 $u=-1$ 得 $a_n(-1)=(-1)^n a_n(-1)$，奇数 $n$ 迫使 $a_n(-1)=0$，从而 $(u+1)\mid a_n(u)$。证毕。
\end{proof}

\begin{corollary}[$\Delta(z,u)$ 的系数回文结构]\label{cor:delta-coeff-palindrome}
写
\[
\Delta(z,u)=\det(I-zB(u))=\sum_{j=0}^{10}c_j(u)\,z^j.
\]
则对每个 $j$ 都有
\[
c_j(u)=u^j c_j(1/u),
\]
即 $c_j(u)$ 为次数 $j$ 的回文多项式（完成化后等价为 $u^{j/2}$ 乘以偶函数）。
\end{corollary}

\begin{remark}[不变量表示：系数仅依赖 $u+u^{-1}$]
取 $u=e^\theta$，则 $u^{-j/2}c_j(u)$ 为 $\theta$ 的偶函数，因而可写成 $u+u^{-1}$ 的多项式；等价地，完成化后 $\widehat{\Delta}(w,u)$ 的系数落在不变量子环 $\QQ[u+u^{-1}]$ 中。
\end{remark}

\begin{corollary}[完成化系数的“临界线实值性”]\label{cor:completion-real-line}
取 $u=e^{it}$（$|u|=1$），并定义
\[
d_j(u):=u^{-j/2}c_j(u).
\]
则
\[
\overline{d_j(u)}=d_j(u),
\]
即完成化后的系数在单位圆上为实数。
\end{corollary}

\begin{corollary}[有限长度的精确镜像对称]\label{cor:primitive-finite-sym}
令 $\pi_{n,k}:=p_{n,k}/\sum_j p_{n,j}$ 为长度 $n$ 的 primitive 轨道在 $k$-分层上的归一化分布，则
\[
\sum_k k\,\pi_{n,k}=\frac{n}{2},\qquad
\sum_k\left(k-\frac{n}{2}\right)^{2m+1}\pi_{n,k}=0\quad(m\ge 0).
\]
因此对每个有限 $n$，分布关于 $n/2$ 精确对称，所有奇阶中心矩严格为零。
\end{corollary}

此外由 $p_{n,k}=p_{n,n-k}$ 直接得到
\[
p_n(u)=u^n p_n(1/u).
\]

\begin{corollary}[奇长度 primitive 的整除律]\label{cor:primitive-odd-div}
若 $n\ge 3$ 为奇数，则 $u^{-1}p_n(u)$ 为奇次数回文多项式，因而
\[
(u+1)\mid u^{-1}p_n(u)\qquad\Longrightarrow\qquad u(u+1)\mid p_n(u).
\]
\end{corollary}

\subsubsection{完成化行列式与严格偶性}\label{app:kernel-completion}
令 $u=e^\theta$，定义完成化行列式
\[
\widetilde{\Delta}(z,\theta):=\Delta(e^{-\theta/2}z,e^\theta),
\qquad
\widetilde{\zeta}(z,\theta):=\widetilde{\Delta}(z,\theta)^{-1}.
\]
由 $\Delta(z,u)=\Delta(uz,1/u)$ 得
\[
\widetilde{\Delta}(z,-\theta)=\widetilde{\Delta}(z,\theta),
\qquad
\widetilde{\zeta}(z,-\theta)=\widetilde{\zeta}(z,\theta),
\]
即完成化后在参数 $\theta$ 上呈严格偶性。

\subsubsection{压力函数的解析性与显式导数}\label{app:pressure-analytic}
对 $u=e^\theta>0$，矩阵 $B(u)$ 仍为 primitive，故 Perron 根 $\lambda(u)$ 在 $u>0$ 上为单根并随 $u$ 实解析。记 $\lambda=\lambda(u)$，其满足六次方程
\[
\begin{aligned}
F(\lambda,u)
={}&\lambda^6-(1+u)\lambda^5-5u\lambda^4+3u(1+u)\lambda^3\\
&\ -u(u^2-3u+1)\lambda^2+u(u^3-3u^2-3u+1)\lambda+u^2(u^2+u+1)=0.
\end{aligned}
\]
由隐式微分
\[
\lambda_u=-\frac{F_u}{F_\lambda},\qquad
P'(\theta)=\frac{u\lambda_u}{\lambda},
\]
再微分一次得
\[
\lambda_{uu}=-\frac{F_{uu}+2F_{u\lambda}\lambda_u+F_{\lambda\lambda}\lambda_u^2}{F_\lambda},
\qquad
P''(\theta)=\frac{u\lambda_u}{\lambda}+u^2\left(\frac{\lambda_{uu}}{\lambda}-\Bigl(\frac{\lambda_u}{\lambda}\Bigr)^2\right).
\]
由对称性 $P(\theta)-\theta/2$ 为偶函数，故
\[
P^{(2k+1)}(0)=0\quad(k\ge 1),
\]
特别地 $P^{(3)}(0)=0$，且
\[
P(\theta)=\log 3+\frac12\theta+\frac{11}{204}\theta^2+\frac{1559}{1414944}\theta^4
-\frac{17123893}{1177686190080}\theta^6-\frac{122803509253}{25412897733672960}\theta^8
+O(\theta^{10}),
\]
从而
\[
P^{(4)}(0)=\frac{1559}{58956},\qquad
P^{(6)}(0)=-\frac{17123893}{1635675264},\qquad
P^{(8)}(0)=-\frac{122803509253}{630280201728}.
\]
此外
\[
P(\theta)\to 0\quad(\theta\to-\infty),\qquad
P(\theta)\sim \theta\quad(\theta\to+\infty),
\]
对应极端权重下的冻结极限。

\begin{corollary}[中心化后的四阶修正系数]\label{cor:edgeworth-fourth}
令 $X:=e-\frac12$，则所有奇阶累积量为 $0$，并且
\[
\sigma^2=P''(0)=\frac{11}{102},\qquad
\kappa_4=P^{(4)}(0)=\frac{1559}{58956}.
\]
因此标准化四阶比为
\[
\frac{\kappa_4}{\sigma^4}=\frac{4677}{2057}\approx 2.274,
\]
在 Edgeworth 型一阶修正中对应系数
\[
\frac{1}{24}\frac{\kappa_4}{\sigma^4}=\frac{4677}{49368}\approx 0.0947.
\]
\end{corollary}

\subsubsection{大偏差率函数的参数化闭式}\label{app:rate-param}
令
\[
\alpha(\theta):=P'(\theta)\in(0,1),
\]
则大偏差率函数满足
\[
I(\alpha(\theta))=\theta\alpha(\theta)-\bigl(P(\theta)-P(0)\bigr),\qquad P(0)=\log 3.
\]
由自对偶对称得
\[
I(\alpha)=I(1-\alpha),
\]
且在 $\alpha_\star=\frac12$ 附近
\[
I(\alpha)=\frac{(\alpha-\frac12)^2}{2P''(0)}+O\!\left((\alpha-\tfrac12)^3\right)
=\frac{51}{11}\Bigl(\alpha-\frac12\Bigr)^2+O(\cdot).
\]
由 $p_{n,k}\approx \frac{1}{n}\exp(n(\log 3-I(k/n)))$ 得到局部二次近似：若
\(
k=\frac{n}{2}+x\sqrt{n}
\)
（$x$ 有界），则
\[
p_{n,k}\approx \frac{3^n}{n}\exp\!\left(-\frac{x^2}{2P''(0)}\right)
=\frac{3^n}{n}\exp\!\left(-\frac{51}{11}x^2\right),
\]
即呈现方差尺度 $nP''(0)=\frac{11}{102}n$ 的高斯型指数近似。
\begin{corollary}[中等偏差展开（到 $\delta^6$）]\label{cor:moderate-deviation}
令 $\delta:=\alpha-\frac12$，则 Legendre 反演给出
\[
I\!\left(\tfrac12+\delta\right)
=\frac{51}{11}\delta^2
-\frac{238527}{29282}\delta^4
\frac{207314036559}{3117947360}\delta^6
O(\delta^8).
\]
\end{corollary}
端点值为
\[
I(0)=I(1)=\log 3.
\]
若记 $p_{n,k}=[u^k]p_n(u)$，则典型规模为
\[
p_{n,k}\approx \frac{1}{n}\exp\!\Bigl(n(\log 3-I(k/n))\Bigr).
\]

\subsubsection{加权 Mertens 常数与 Abel 有限部}\label{app:mertens-weighted}
令 $\lambda(u)$ 为 Perron 根，$\rho(u)$ 为其余特征值最大模，则
\[
a_n(u)=\mathrm{Tr}(B(u)^n)=\lambda(u)^n+O(\rho(u)^n),
\]
由 Witt 反演得
\[
p_n(u)=\frac{\lambda(u)^n}{n}+O\!\left(\frac{\rho(u)^n}{n}\right).
\]
因此
\[
\Pi_\theta(N):=\sum_{n\le N}p_n(\theta)\sim \frac{\lambda(\theta)}{\lambda(\theta)-1}\cdot\frac{\lambda(\theta)^N}{N}.
\]
记
\[
\mathcal{M}_\theta(N):=\sum_{n\le N}\frac{p_n(u)}{\lambda(u)^n},
\]
则
\[
\mathcal{M}_\theta(N)=\log N+M(\theta)+o(1),
\qquad
M(\theta)=\gamma+\sum_{n\ge 1}\left(\frac{p_n(u)}{\lambda(u)^n}-\frac{1}{n}\right),
\]

\begin{proposition}[Mertens 常数的偶性]\label{prop:mertens-even}
由 $p_n(u)=u^n p_n(1/u)$ 与 $\lambda(u)=u\lambda(1/u)$ 得
\[
\frac{p_n(u)}{\lambda(u)^n}=\frac{p_n(1/u)}{\lambda(1/u)^n},
\]
从而
\[
M(\theta)=M(-\theta),\qquad
\log\mathfrak{M}(\theta)=\log\mathfrak{M}(-\theta).
\]
\end{proposition}
其中 $\gamma$ 为 Euler 常数。设
\[
P_\theta(r):=\sum_{n\ge 1}\frac{p_n(u)}{\lambda(u)^n}r^n,\quad 0<r<1,
\]
则 Abel 有限部给出
\[
\log\mathfrak{M}(\theta):=\mathrm{fp}_{r=1}\bigl(P_\theta(r)+\log(1-r)\bigr)
=\sum_{n\ge 1}\left(\frac{p_n(u)}{\lambda(u)^n}-\frac{1}{n}\right),
\]
从而
\[
M(\theta)=\gamma+\log\mathfrak{M}(\theta).
\]
在 $\theta=0$ 处有
\[
M(0)\approx 0.2738887860704445,\qquad
\log\mathfrak{M}(0)\approx -0.3033268788310883.
\]

\subsubsection{仅用 \texorpdfstring{$\Delta$}{Delta} 的闭式计算}\label{app:delta-only}
令 $z_\star(u)=\lambda(u)^{-1}$ 为 $\Delta(z,u)$ 的最小正根，定义
\[
C(u):=\lim_{z\to z_\star}(1-z/z_\star)\zeta(z,u)=\frac{1}{-z_\star\,\partial_z\Delta(z_\star,u)}.
\]
由 Witt 反演与 $\zeta=1/\Delta$ 可得
\[
\log\mathfrak{M}(\theta)
=\log C(u)+\sum_{m\ge 2}\frac{\mu(m)}{m}\log\zeta(z_\star^m,u^m)
=\log C(u)-\sum_{m\ge 2}\frac{\mu(m)}{m}\log\Delta(z_\star^m,u^m),
\]
其中级数以指数速度收敛。
