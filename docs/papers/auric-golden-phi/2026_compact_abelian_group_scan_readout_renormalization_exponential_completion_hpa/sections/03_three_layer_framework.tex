\section{A Unified Three-Layer Framework: Ontic Scan, Finite Readout, and Renormalization Stability}

\subsection{Ontic Layer: Scan System}

\begin{definition}[Ontic scan system]
A triple $\mathcal{S}=(G,\mu,T_g)$ is called an ontic scan system, where $G$ is a compact Abelian group, $\mu$ is Haar probability measure, and $T_g$ is translation.
\end{definition}

The ontic layer specifies only invertible measure-preserving dynamics and the phase space; no discrete structure is presupposed.

\subsection{Readout Layer: Windows, Resolution, and Factors}

Let $\varepsilon>0$ be a resolution parameter.
\begin{definition}[Existence readout]
Given a measurable window $W\subset G$, define existence readout by
\[
E(x)=\ind_W(x)\in\{0,1\}.
\]
\end{definition}

\begin{definition}[Concept readout]
Let $\mathcal{C}$ be a concept index set, and let $\{f_c\}_{c\in\mathcal{C}}\subset L^2(G)$ be an observable family. For each $c$, define a finite-resolution quantization map
\[
\pi_\varepsilon^{(c)}:G\to \Sigma_\varepsilon,
\]
where $\Sigma_\varepsilon$ is a finite alphabet (or symbol space induced by a finite interval partition), yielding a factor system $(\Sigma_\varepsilon^\ZZ,\sigma)$.
\end{definition}

\begin{definition}[Readout sequence]
For an orbit $x_n=T_g^n(x_0)$, define the readout sequence by
\[
y_n=\pi_\varepsilon(x_n).
\]
\end{definition}

The readout layer maps continuous orbits to discrete sequences; ``stable structure'' is then stated through computable invariants of readout sequences.

\subsection{Renormalization Layer: Substitution, Blocking, and Fixed Points}

\begin{definition}[Renormalization operator]
Choose a block map or substitution map $R$ on symbol space, e.g.:
\begin{itemize}
  \item $R$ is a length-$k$ blocking map that treats each $k$-block as a new symbol;
  \item or $R$ is a primitive substitution system.
\end{itemize}
\end{definition}

\begin{definition}[Semantic stability]
If the readout-sequence family converges under $R$ to a fixed point or attractor (up to shift equivalence), the corresponding structure is called renormalization-stable.
\end{definition}

This layer provides criteria for cross-scale consistency: stability of growth rates (dominant eigenvalues), frequency vectors, and block statistics.
