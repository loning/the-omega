
\section{Audit closure checklist (this version)}
\label{app:audit_checklist}

This appendix summarizes, in audit form, what is assumed in the closed layer and what is derived.

\subsection{Closed-layer inputs (assumptions)}
\label{app:closed_inputs}

\paragraph{Omega axioms used.}
\begin{itemize}
\item O2: finite information / compact readout (Axiom~\ref{ax:O2}).
\item O5: scan--projection readout (Axiom~\ref{ax:O5}).
\item O6: unitary scan algebra / Weyl pair (Axiom~\ref{ax:O6}).
\item Canonical Abel regularization and finite-part admissibility along $r\uparrow 1$ (Section~\ref{sec:abel_finite_part}; Appendix~\ref{app:orbit_abel} records a sufficient existence criterion and uniqueness).
\end{itemize}

\paragraph{HTF structural bridge.}
\begin{itemize}
\item Holographic Trace Formula assumption (Assumption~\ref{assump:HTF}): the Abel-weighted scan trace matches a Weil-type explicit formula under the same regularization, and zeros contribute as sampled exponential modes $\e^{(\rho-\tfrac{1}{2})t}$, equivalently via meromorphic mode factors $(1-r\,\e^{(\rho-\tfrac{1}{2})})^{-1}$, in a way compatible with the geometric-side holomorphy on $|r|<1$.
\end{itemize}

\subsection{Closed-layer derivations (theorems)}
\label{app:closed_derivations}

\begin{itemize}
\item Lemma~\ref{lem:abel_obstruction}: if $\RePart(\rho)>\tfrac{1}{2}$ then the mode factor $(1-r\,\e^{(\rho-\tfrac{1}{2})})^{-1}$ has an interior pole at a point $r_\rho$ with $|r_\rho|<1$.
\item Theorem~\ref{thm:polar_rigidity}: Omega geometric-side holomorphy on the unit disk $|r|<1$ plus HTF mode structure implies $\RePart(\rho)=\tfrac{1}{2}$ for all nontrivial zeros.
\end{itemize}
