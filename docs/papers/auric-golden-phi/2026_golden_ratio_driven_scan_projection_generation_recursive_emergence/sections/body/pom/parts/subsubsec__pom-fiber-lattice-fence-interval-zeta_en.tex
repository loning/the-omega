\subsubsection{Distributive lattice—interval closure—multichain spectrum: lattice theory and spectral fingerprints with \texorpdfstring{\((\ell_i)\)}{(ℓ\_i)} as minimal sufficient statistic}\label{subsubsec:pom-fiber-lattice-fence-interval-zeta}
Under the \((011\leftrightarrow 100)\) local kernel, the preimage fiber \(F_x=\Fold_m^{-1}(x)\)'s combinatorial model has been identified as the independent set system of a path disjoint union
\(
F_x\cong \mathsf{Ind}(\Gamma_x)
\)
(Theorem \ref{thm:pom-fiber-indset-factorization}).
This subsection further demonstrates that, without introducing any external interpretation layer, \((\ell_i)\) not only determines the fiber cardinality and cubical geometry,
but also naturally generates a complete auditable framework of \emph{distributive lattice (Birkhoff representation)—interval closure—multichain spectrum (zeta/order polynomials)}.

\paragraph{Fence poset and ideal lattice}
\begin{definition}[Fence poset and ideal lattice]\label{def:pom-fence-poset}
For \(\ell\ge 0\), define the \textbf{fence poset} \(Z_\ell\) as the poset on ground set \(\{1,\dots,\ell\}\),
whose covering relation alternates direction along the linear order:
\[
1\prec 2 \succ 3\prec 4 \succ \cdots.
\]
For a finite poset \(P\), denote by \(J(P)\) the lattice of \textbf{order ideals} (down-closed sets) under inclusion (\(\wedge=\cap\), \(\vee=\) down-closure after union).
\end{definition}

\begin{theorem}[Birkhoff complete representation of fiber: fiber as fence ideal lattice]\label{thm:pom-fiber-birkhoff-fence-ideal-lattice}
Fix \(x\in X_m\), and suppose the invertible-site-induced graph decomposes into a path disjoint union
\[
\Gamma_x\ \cong\ \bigsqcup_{i=1}^r P_{\ell_i}.
\]
Then under the natural ``single inverse substitution switch reachability'' poset, the fiber \(F_x\) can be endowed with a finite distributive lattice structure \(\mathcal L_x\), and there exists a canonical isomorphism
\[
\mathcal L_x\ \cong\ J\!\Big(\bigsqcup_{i=1}^r Z_{\ell_i}\Big).
\]
In particular, the poset of join-irreducibles of \(\mathcal L_x\) (the base of the Birkhoff representation) is isomorphic to
\(\bigsqcup_i Z_{\ell_i}\); thus, the multiset \(\{\ell_i\}\) completely determines the internal join/meet structure and interval structure of the fiber.
\end{theorem}
\begin{proof}
For a single component \(P_\ell\), note that for any finite poset \(P\), the map
\[
\mathsf{Ant}(P)\ \longleftrightarrow\ J(P),
\qquad
A\mapsto \downarrow A,\ \ I\mapsto \max(I)
\]
gives a canonical bijection between antichains and down-closed sets (\(\downarrow A\) is the down-closure, \(\max(I)\) is the maximal element set).
For the fence poset \(Z_\ell\), comparability occurs only at adjacent points,
so the antichains \(\mathsf{Ant}(Z_\ell)\) are precisely equivalent to the independent sets on the path graph \(P_\ell\) (``cannot take adjacent points simultaneously'').
Thus, we obtain a canonical bijection \(\mathsf{Ind}(P_\ell)\cong J(Z_\ell)\).
\par
Through this bijection, pulling back the inclusion order and the \(\wedge,\vee\) operations of \(J(Z_\ell)\) to \(\mathsf{Ind}(P_\ell)\) (and then via Theorem \ref{thm:pom-fiber-indset-factorization} to \(F_x\)),
we can define a distributive lattice structure \(\mathcal L_x\) on the fiber.
\par
For a disjoint union \(\Gamma_x=\bigsqcup_i P_{\ell_i}\), the independent set system decomposes into a Cartesian product,
while the ideal lattice satisfies \(J(\bigsqcup_i P_i)\cong\prod_i J(P_i)\).
Combining both steps and using the fiber isomorphism given by Theorem \ref{thm:pom-fiber-indset-factorization}, we obtain the stated distributive lattice representation. \qed
\end{proof}

\begin{corollary}[Rank polynomial product factorization and Fibonacci normalization]\label{cor:pom-fiber-fence-rank-poly-factorization}
Taking the ideal size \(|I|\) as the rank function, define the fiber lattice rank polynomial
\[
R_x(q):=\sum_{I\in \mathcal L_x} q^{|I|}.
\]
Then
\[
R_x(q)=\prod_{i=1}^r R_{\ell_i}(q),
\qquad
R_\ell(q):=\sum_{I\in J(Z_\ell)} q^{|I|}.
\]
Moreover, at \(q=1\) this recovers the Fibonacci count: \(|J(Z_\ell)|=R_\ell(1)=F_{\ell+2}\), thus
\(
|F_x|=\prod_i F_{\ell_i+2}
\)
(consistent with the cardinality closed form of Theorem \ref{thm:pom-fiber-indset-factorization}).
\end{corollary}
\begin{proof}
By the direct product decomposition in Theorem \ref{thm:pom-fiber-birkhoff-fence-ideal-lattice},
\(\mathcal L_x\cong\prod_i J(Z_{\ell_i})\) and the rank function is additive, so the generating function factorizes.
Finally, \(|J(Z_\ell)|=|\mathsf{Ind}(P_\ell)|=F_{\ell+2}\) is the standard identity for path independent set counting. \qed
\end{proof}

\paragraph{Interval closure and Möbius rigidity}
\begin{theorem}[Interval closure: any interval remains a fence ideal lattice]\label{thm:pom-fence-interval-closure}
Suppose \(\mathcal L_x\cong J(P_x)\) and \(P_x=\bigsqcup_i Z_{\ell_i}\).
For any interval \([I,J]\subseteq \mathcal L_x\), let
\[
P_{I,J}:=J\setminus I\ \subseteq\ P_x
\]
with the induced poset from \(P_x\).
Then there exists a natural lattice isomorphism
\[
[I,J]\ \cong\ J(P_{I,J}).
\]
Moreover, since \(P_x\) is a fence disjoint union, the induced sub-poset \(P_{I,J}\) must decompose into finitely many fence posets in disjoint union (i.e., it remains in the fence category).
\end{theorem}
\begin{proof}
For general \(J(P)\), the map \(\Phi:[I,J]\to J(J\setminus I)\) given by \(\Phi(K)=K\setminus I\),
with inverse \(L\mapsto I\cup L\), can be directly verified to be a lattice isomorphism.
When \(P_x\) is a fence disjoint union, the Hasse diagram of the induced sub-poset is a path forest, so its connected components remain fences. \qed
\end{proof}

\begin{theorem}[Möbius rigidity: \texorpdfstring{\(\mu\in\{0,\pm1\}\)}{mu in {0,±1}}]\label{thm:pom-fence-mobius-rigidity}
Using the notation of Theorem \ref{thm:pom-fence-interval-closure},
the interval Möbius number satisfies
\[
\mu_{\mathcal L_x}(I,J)=
\begin{cases}
(-1)^{|P_{I,J}|},& P_{I,J}\ \text{is an antichain},\\
0,& \text{otherwise}.
\end{cases}
\]
Thus, for any \([I,J]\subseteq\mathcal L_x\), we have \(\mu_{\mathcal L_x}(I,J)\in\{0,\pm 1\}\).
This pattern gives a purely discrete interval certificate: \(\mu(I,J)\neq 0\) if and only if the interval is equivalent to a Boolean lattice (i.e., the join-irreducibles within the interval are pairwise incomparable).
\end{theorem}
\begin{proof}
By Theorem \ref{thm:pom-fence-interval-closure}, \(\mu_{\mathcal L_x}(I,J)=\mu_{J(P_{I,J})}(\hat0,\hat1)\).
For a general distributive lattice \(J(P)\), the interval \([\hat0,\hat1]\) is Boolean if and only if \(P\) is an antichain;
in this case, \(J(P)\cong \{0,1\}^{|P|}\) and \(\mu(\hat0,\hat1)=(-1)^{|P|}\).
If \(P\) contains a comparable pair, then \(J(P)\) contains a nontrivial three-element chain, so \(\mu(\hat0,\hat1)=0\).
Combining yields the result. \qed
\end{proof}

\begin{remark}[Comparison with fiber signed index: two types of \texorpdfstring{\(0/\pm 1\)}{0/±1} rigidity]\label{rem:pom-fence-mobius-vs-signed-index}
The Möbius rigidity of Theorem \ref{thm:pom-fence-mobius-rigidity} is a \(\{0,\pm 1\}\) pattern at the \emph{lattice interval} level,
whose nonvanishing is completely determined by whether the interval is Boolean (equivalent to the difference sub-poset being an antichain).
On the other hand, in the ``inverse substitution count'' statistic of \(\mathsf{Ind}(P_\ell)\), the fiber signed index invariant
\(\sigma_m(x)=Z_x(-1)\) (results in \S\ref{subsubsec:single-slice-reflector-completeness}; also see Proposition \ref{prop:pom-fiber-index-cgf})
also satisfies \(\{0,\pm 1\}\) rigidity, but its vanishing criterion comes from the 6-periodicity/mod 3 obstruction of the path independent set polynomial at \(-1\).
Thus, the two types of \(\{0,\pm 1\}\) invariants characterize ``interval Booleanity'' and ``path independent set cancellation phase'' respectively, complementarily serving the discrete auditing of fibers.
\end{remark}

\paragraph{Multichain spectrum: zeta polynomials, order polynomials, and the \texorpdfstring{\(\pi\)}{pi} constant}
\begin{definition}[Zeta polynomial and order polynomial]\label{def:pom-zeta-order-polynomial}
For a bounded finite lattice \(L\), define its zeta polynomial \(Z_L(k)\) as the number of \(k\)-step multichains from \(\hat0\) to \(\hat1\):
\[
Z_L(k):=\#\{\hat0=I_0\subseteq I_1\subseteq \cdots \subseteq I_k=\hat1\}.
\]
For a finite poset \(P\), denote its order polynomial by
\[
\Omega_P(k):=\#\{f:P\to\{1,\dots,k\}:\ f \ \text{is order-preserving}\}.
\]
\end{definition}

\begin{theorem}[Equivalence of zeta/order polynomials for distributive lattices]\label{thm:pom-zeta-equals-order-poly}
For any finite poset \(P\), its ideal lattice satisfies
\[
Z_{J(P)}(k)=\Omega_P(k)\qquad(k\in\NN).
\]
\end{theorem}
\begin{proof}
Given a multichain \(\hat0=I_0\subseteq\cdots\subseteq I_k=\hat1\), define \(f:P\to\{1,\dots,k\}\) by
\(
f(p):=\min\{j:\ p\in I_j\}
\).
By down-closure, \(f\) can be verified to be order-preserving; conversely, from \(f\) define \(I_j:=\{p:\ f(p)\le j\}\) to obtain a multichain.
The two constructions are mutually inverse, so the counts are equal. \qed
\end{proof}

\begin{remark}[Spectral rigidity of the factor 4: half-odd-integer angular quantization of the discrete Green kernel]\label{rem:pom-Kk-factor4-spectral-rigidity}
Let \(K_k=(K_k(i,j))_{1\le i,j\le k}\) with \(K_k(i,j)=\min(i,j)\), and write \(L_k:=K_k^{-1}\) for its inverse.

2 & -1 & 0 & \cdots & 0\\
-1 & 2 & -1 & \ddots & \vdots\\
0 & \ddots & \ddots & \ddots & 0\\
\vdots & \ddots & -1 & 2 & -1\\
0 & \cdots & 0 & -1 & 1
\end{pmatrix},
\]
which is a tridiagonal difference operator of a mixed-boundary discrete Laplacian (right endpoint Neumann, left endpoint Dirichlet).
Its spectrum satisfies half-odd-integer angular quantization
\[
\mathrm{Spec}(L_k)=
\left\{
2\bigl(1-\cos\theta_p\bigr):\
\theta_p=\frac{(2p-1)\pi}{2k+1},\ p=1,\dots,k
\right\},
\]
thus
\[
\mathrm{Spec}(K_k)=
\left\{
\frac{1}{2\bigl(1-\cos\theta_p\bigr)}
=
\frac{1}{4\sin^2(\theta_p/2)}:\ p=1,\dots,k
\right\},
\]

Therefore, the factor 4 in the spectral representation is not an adjustable normalization: once the mixed-boundary angular quantization \(\theta_p=(2p-1)\pi/(2k+1)\) is fixed,
it is structurally enforced by the half-angle identity \(2(1-\cos\theta)=4\sin^2(\theta/2)\);
equivalently, the forms \(4k+2\) and \(4\sin^2(\cdot)\) are half-angle rewrites of the same quantized angle.
\end{remark}

Substituting \(P_x=\bigsqcup_i Z_{\ell_i}\) yields the result. \qed

\paragraph{Maximal chain enumeration, continuum volume, and Ehrhart--Hilbert refinement spectrum}

\begin{definition}[Linear extension count and maximal chain count]\label{def:pom-fence-linext-maxchain}
For a finite poset \(P\), write \(e(P):=\#\mathrm{LinExt}(P)\) for its linear extension count.
For a finite lattice \(L\), write \(\mathrm{MC}(L)\) for its set of maximal chains (saturated chains from \(\hat0\) to \(\hat1\)).
\end{definition}

\begin{theorem}[Maximal chains \(\Longleftrightarrow\) linear extensions: scheduling equivalence for distributive lattices]\label{thm:pom-maxchains-equals-linext}
For any finite poset \(P\), there exists a canonical bijection between maximal chains of its ideal lattice \(J(P)\) and linear extensions of \(P\),
thus
\[
|\mathrm{MC}(J(P))|=e(P).
\]
In particular, for the fiber lattice \(\mathcal L_x\cong J(P_x)\) (Theorem \ref{thm:pom-fiber-birkhoff-fence-ideal-lattice}),
\[
|\mathrm{MC}(\mathcal L_x)|=e(P_x).
\]
\end{theorem}
\begin{proof}
Given a maximal chain \(\hat0=I_0\subset I_1\subset\cdots\subset I_N=\hat1\) (where \(N=|P|\)),
each step must add exactly one new element \(p_t\in P\), i.e., \(I_t=I_{t-1}\cup\{p_t\}\).
Down-closure ensures that the addition order satisfies the poset constraints, so \((p_1,\dots,p_N)\) is a linear extension of \(P\).
Conversely, any linear extension produces a maximal chain by taking prefix down-closures in order.
The two constructions are mutually inverse, yielding the bijection and count equality. \qed
\end{proof}

\begin{theorem}[Closed form for maximal chain count of fence components: Euler zigzag number control]\label{thm:pom-fence-maxchains-euler}
Suppose \(P_x=\bigsqcup_{i=1}^r Z_{\ell_i}\), and write \(N:=\sum_{i=1}^r \ell_i\).
Let \(E_n\) denote the Euler zigzag numbers (up/down numbers), defined by the exponential generating function:
\[
\sec z+\tan z=\sum_{n\ge 0} E_n\,\frac{z^n}{n!}.
\]
Then we have the strict identity
\[
|\mathrm{MC}(\mathcal L_x)|
\;=\;
e(P_x)
\;=\;
\binom{N}{\ell_1,\dots,\ell_r}\cdot \prod_{i=1}^r E_{\ell_i}.
\]
Thus, the fiber's ``scheduling complexity''
\[
\mathsf H_{\mathrm{sched}}(x):=\log |\mathrm{MC}(\mathcal L_x)|
\]
has an auditable decomposition
\[
\mathsf H_{\mathrm{sched}}(x)
=\log(N!) -\sum_{i=1}^r \log(\ell_i!) + \sum_{i=1}^r \log(E_{\ell_i}).
\]
\end{theorem}
\begin{proof}
By Theorem \ref{thm:pom-maxchains-equals-linext}, \(|\mathrm{MC}(\mathcal L_x)|=e(P_x)\).
For disjoint union posets \(P=\bigsqcup_i P_i\), linear extensions are equivalent to: first choosing a linear extension for each component, then shuffling these sequences in a multinomial coefficient \(\binom{N}{|P_1|,\dots,|P_r|}\) way.
Thus, \(e(P)=\binom{N}{|P_1|,\dots,|P_r|}\prod_i e(P_i)\).
\par
Finally, the linear extension count of the fence poset \(Z_n\) satisfies \(e(Z_n)=E_n\), which is the standard equivalent characterization of Euler zigzag numbers (from the recurrence/EGF of \(\sec z+\tan z\) or the classical bijection with alternating permutations).
Substituting \(P_i=Z_{\ell_i}\) yields the conclusion. \qed
\end{proof}

\begin{corollary}[Extremal structure of maximal chain count: all isolated components optimal]\label{cor:pom-fence-maxchains-all-isolated-optimal}
Using the notation of Theorem \ref{thm:pom-fence-maxchains-euler}, and writing \(N=\sum_i \ell_i\),
under the constraint of fixed \(N\), the maximal chain count satisfies the strict upper bound
\[
|\mathrm{MC}(\mathcal L_x)|
=
N!\prod_{i=1}^r \frac{E_{\ell_i}}{\ell_i!}
\le
N!,
\]
with equality if and only if \(\ell_1=\cdots=\ell_r=1\) (i.e., \(P_x=\bigsqcup_{i=1}^{N} Z_1\), equivalent to all components being isolated).
Thus, under the protocol of fixed total length \(N\), the fiber's ``scheduling freedom'' \(|\mathrm{MC}(\mathcal L_x)|\) is maximized by the \emph{most fragmented} component structure.
\end{corollary}
\begin{proof}
Since \(E_\ell\) counts alternating permutations, for any \(\ell\ge 1\) we have \(E_\ell\le \ell!\), with strict inequality for \(\ell\ge 2\) (alternating permutations are a proper subset of all permutations).
Substituting into the decomposition of Theorem \ref{thm:pom-fence-maxchains-euler} yields
\(
\prod_i(E_{\ell_i}/\ell_i!)\le 1
\), thus \(|\mathrm{MC}(\mathcal L_x)|\le N!\).
Equality holds if and only if each factor \(E_{\ell_i}/\ell_i!=1\), i.e., \(\ell_i=1\) for all \(i\). \qed
\end{proof}

\begin{definition}[Order polytope and continuum-time parametrization density]\label{def:pom-order-polytope-volume}
For a finite poset \(P\) with \(|P|=N\), define its \textbf{order polytope}

\begin{theorem}[Discrete--continuum identity: linear extension count = normalized volume]\label{thm:pom-order-polytope-volume-linext}
For any finite poset \(P\) with \(|P|=N\), the strict identity holds:
\[
\mathrm{vol}(\mathcal O(P))=\frac{e(P)}{N!}.
\]
In particular, for \(P_x=\bigsqcup_i Z_{\ell_i}\):
\[
\mathsf V(x)=\prod_{i=1}^r \frac{E_{\ell_i}}{\ell_i!},
\qquad
|\mathrm{MC}(\mathcal L_x)|=e(P_x)=N!\,\mathsf V(x).

\begin{proof}

\[
\Delta_\pi:=\{y\in[0,1]^P:\ 0\le y_{p_1}\le y_{p_2}\le\cdots\le y_{p_N}\le 1\}.
\]
If \(\pi\) is a linear extension, then \(\Delta_\pi\subseteq \mathcal O(P)\);
conversely, \(\mathcal O(P)\) decomposes by coordinate total order into a disjoint union of these simplices (ignoring boundaries).
Each \(\Delta_\pi\) has volume \(1/N!\), so \(\mathrm{vol}(\mathcal O(P))=e(P)/N!\).
For \(P_x\), substituting from Theorem \ref{thm:pom-fence-maxchains-euler} yields the result. \qed
\end{proof}

\begin{theorem}[Convex hull theorem for dual invariant feasible region: the Pareto geometry of \texorpdfstring{\((\log d,\log\mathrm{MC})\)}{(log d, log MC)} is generated by atomic lengths]\label{thm:pom-fiber-ab-convex-hull}
Fix total length \(N\ge 1\); for any component decomposition \(N=\sum_{i=1}^r \ell_i\), define normalized invariants
\[
A(\{\ell_i\})
:=\frac{1}{N}\sum_{i=1}^r \log F_{\ell_i+2}
=\frac{1}{N}\log d_m(x),
\]
\[
B(\{\ell_i\})
:=\frac{1}{N}\sum_{i=1}^r \log\frac{E_{\ell_i}}{\ell_i!}
=\frac{1}{N}\log \mathsf V(x)
=\frac{1}{N}\bigl(\log|\mathrm{MC}(\mathcal L_x)|-\log(N!)\bigr),
\]
where \(d_m(x)=\prod_i F_{\ell_i+2}\) is given by Theorem \ref{thm:pom-fiber-indset-factorization}, and \(\mathsf V(x)=\prod_i (E_{\ell_i}/\ell_i!)\) and \(|\mathrm{MC}(\mathcal L_x)|=N!\mathsf V(x)\) by Theorem \ref{thm:pom-order-polytope-volume-linext}.
Let the atomic point set be
\[
\mathcal P
:=
\left\{
\left(\frac{1}{\ell}\log F_{\ell+2},\ \frac{1}{\ell}\log\frac{E_\ell}{\ell!}\right):\ \ell\in\NN
\right\}
\subset\RR^2,
\]
and let \(\mathcal R\subset\RR^2\) be the feasible region formed by all points \((A(\{\ell_i\}),B(\{\ell_i\}))\) generated by decompositions \(\{\ell_i\}\) in the scale limit \(N\to\infty\) (i.e., the closed limit set of reachable points).
Then
\[
\mathcal R=\overline{\mathrm{conv}}(\mathcal P).
\]
Thus, the Pareto frontier of \((A,B)\) is equivalent to the supporting curve of \(\overline{\mathrm{conv}}(\mathcal P)\): for any supporting direction \((\alpha,\beta)\in\RR^2\), the extremal value \(\sup_{(A,B)\in\mathcal R}(\alpha A+\beta B)\) is realized by a supporting hyperplane spanned by finitely many atomic lengths \(\ell\).
\end{theorem}
\begin{proof}
For any decomposition \(N=\sum_i \ell_i\), write weights \(w_i:=\ell_i/N\); then \(\sum_i w_i=1\), and
\[
(A(\{\ell_i\}),B(\{\ell_i\}))
=\sum_{i=1}^r w_i\left(\frac{1}{\ell_i}\log F_{\ell_i+2},\ \frac{1}{\ell_i}\log\frac{E_{\ell_i}}{\ell_i!}\right),
\]
so any reachable point at finite \(N\) is a convex combination of \(\mathcal P\), thus \(\mathcal R\subseteq \overline{\mathrm{conv}}(\mathcal P)\).
Conversely, for any finite convex combination \(\sum_{j=1}^s \theta_j p_{\ell_j}\) (\(\theta_j\ge 0,\sum_j\theta_j=1\) and \(p_{\ell_j}\in\mathcal P\)), take large \(N\) and let \(\ell_j\) appear approximately \(\theta_j N/\ell_j\) times to construct a sequence of decompositions whose corresponding \((A,B)\) approximates this convex combination; by density of finite convex combinations in the closed convex hull, \(\overline{\mathrm{conv}}(\mathcal P)\subseteq \mathcal R\). \qed
\end{proof}

\begin{theorem}[Limiting density of the \texorpdfstring{\(\log(2/\pi)\)}{log(2/pi)} constant]\label{thm:pom-fence-volume-log2pi}
Euler zigzag numbers satisfy singularity-dominated asymptotics (the nearest singularity of \(\sec z+\tan z\) is at \(z=\pi/2\)), thus
\[
\lim_{n\to\infty}\frac{1}{n}\log\frac{E_n}{n!}=\log\frac{2}{\pi}.
\]
Therefore, at the ``bulk'' scale (e.g., when all \(\ell_i\to\infty\) or \(\sum_{\ell_i\le L}\ell_i=o(N)\) for any fixed \(L\)), there is density convergence
\[
\frac{1}{N}\log\mathsf V(x)
=\sum_{i=1}^r \frac{\ell_i}{N}\,\frac{1}{\ell_i}\log\frac{E_{\ell_i}}{\ell_i!}
\longrightarrow
\log\frac{2}{\pi}.
\]
\end{theorem}
\begin{proof}
The first formula is the classical asymptotic result for Euler zigzag numbers, obtained from the simple pole of \(\sec z+\tan z\) at \(z=\pm\pi/2\) and the standard coefficient asymptotic theorem.
The second formula follows from weighted averaging and the previous limit. \qed
\end{proof}

\begin{remark}[``Quarter-turn'' nearest-singularity law and the universal folding scale \texorpdfstring{\(\pi/2\)}{pi/2}]\label{rem:pom-quarter-turn-pi2-scale}
The constant \(\log(2/\pi)\) in Theorem \ref{thm:pom-fence-volume-log2pi} arises from nearest-singularity domination in analytic combinatorics: the nearest real singularity of \(\sec z+\tan z\) is at \(z=\pm\pi/2\).
Since the fundamental period of trigonometric functions is \(2\pi\), this singularity corresponds to ``one quarter of the full angle,'' i.e., \(\pi/2=(2\pi)/4\); thus \(\log(2/\pi)=-\log(\pi/2)\) can be understood as a strict ``quarter-turn singularity law.''
\par
On the other hand, from the closed form
\(
h_{\mathrm{ref}}(k)=-\log\!\bigl(2\sin(\pi/(4k+2))\bigr)
\)
in Corollary \ref{cor:pom-fence-multichain-entropy-pi}, setting \(N:=2k+1\) yields a fully explicit high-order asymptotic expansion
\[
h_{\mathrm{ref}}(k)
=\log N-\log\pi+\frac{\pi^2}{24N^2}+\frac{\pi^4}{2880N^4}+O(N^{-6}).
\]
Thus, the two types of entropy constants appearing in this paper
\[
-\log\pi,\qquad \log\frac{2}{\pi}=-\log\frac{\pi}{2}
\]
are not independent: they share a unique angular scale \(\pi/2\) (equivalently sharing the ratio \(2\pi/(\pi/2)=4\)), and appear on the spectral refinement side and the scheduling volume side in the forms of ``half-angle quantization'' and ``nearest-singularity domination,'' respectively.
Thus, the reappearance of the factor 4 can be unified under the same \(\pi/2\) mechanism: on the spectral side, it is enforced by the half-angle identity and half-odd-integer modes (Remark \ref{rem:pom-Kk-factor4-spectral-rigidity});
on the analytic side, it is locked by the ratio of the full-angle period to the nearest singularity position.
\par
At the level of pure spectral geometry, the above chain ``endpoint regularity \(\Rightarrow\) half-odd-integer modes \(\Rightarrow\) half-angle representation'' provides an implementation-independent structural prototype,
so when \(1/4\)-type proportionality constants appear under the horizon endpoint/conical-tip protocol (e.g., the \(1/4\) coefficient in black hole entropy formulas), they can be viewed as strict projections of the same angular scale under two equivalent presentations ``half-angle/quarter-turn,'' without relying on additional physical input.
\end{remark}

\paragraph{Ehrhart--Hilbert refinement spectrum: auditable counting of integer refinement times}

\begin{definition}[Ehrhart polynomial]\label{def:pom-ehrhart-order-polytope}
For a lattice polytope \(Q\subset\RR^N\), denote its Ehrhart polynomial by
\(
L_Q(t):=\#(tQ\cap\ZZ^N)
\)
(\(t\in\ZZ_{\ge 0}\)).
\end{definition}

\begin{theorem}[Ehrhart count of order polytope = order polynomial (with shift)]\label{thm:pom-ehrhart-equals-orderpoly-shift}
For any finite poset \(P\),
\[
L_{\mathcal O(P)}(t)=\Omega_P(t+1)\qquad(t\in\ZZ_{\ge 0}),
\]
where \(\Omega_P(k)\) is the order polynomial (order-preserving map count) of Definition \ref{def:pom-zeta-order-polynomial}.
In particular,
\[
L_{\mathcal O(P)}(1)=\Omega_P(2)=|J(P)|.
\]
\end{theorem}
\begin{proof}
Points in \(t\mathcal O(P)\cap\ZZ^P\) are equivalent to integer assignments \(a:P\to\{0,1,\dots,t\}\) satisfying \(a(p)\le a(q)\) (when \(p\le_P q\)).
Shifting overall by \(a\mapsto a+1\) yields order-preserving maps \(P\to\{1,\dots,t+1\}\), so the count is \(\Omega_P(t+1)\).
When \(t=1\), \(\{0,1\}\)-valued order-preserving maps are equivalent to indicator functions of order ideals, so \(\Omega_P(2)=|J(P)|\). \qed
\end{proof}

\begin{corollary}[Ehrhart--Hilbert refinement spectrum product closure for general fiber]\label{cor:pom-fiber-ehrhart-product}
For \(P_x=\bigsqcup_i Z_{\ell_i}\), define the fiber refinement spectrum
\[
\mathsf{Ehr}_x(t):=L_{\mathcal O(P_x)}(t)\qquad(t\in\ZZ_{\ge 0}).
\]
Then
\[
\mathsf{Ehr}_x(t)=\Omega_{P_x}(t+1)=\prod_{i=1}^r \Omega_{Z_{\ell_i}}(t+1)
=\prod_{i=1}^r L_{\mathcal O(Z_{\ell_i})}(t).
\]
Moreover, at \(t=1\) it recovers the fiber cardinality: \(\mathsf{Ehr}_x(1)=|J(P_x)|=|\mathcal L_x|=|F_x|\).
\end{corollary}
\begin{proof}
By Theorem \ref{thm:pom-ehrhart-equals-orderpoly-shift}, \(\mathsf{Ehr}_x(t)=\Omega_{P_x}(t+1)\).
Order polynomials strictly factorize multiplicatively for disjoint union posets: \(\Omega_{\bigsqcup_i P_i}(k)=\prod_i\Omega_{P_i}(k)\);
substituting \(P_i=Z_{\ell_i}\) yields the result. The final formula follows immediately from the \(t=1\) indicator function characterization. \qed
\end{proof}

\begin{definition}[\texorpdfstring{\(h^*\)}{h*}-polynomial: Hilbert refinement numerator]\label{def:pom-hstar-order-polytope}
For a lattice polytope \(Q\) of dimension \(N\), its Ehrhart series can be written as
\[
\sum_{t\ge 0} L_Q(t)\,z^t=\frac{h_Q^*(z)}{(1-z)^{N+1}},
\]
where \(h_Q^*(z)\in\ZZ[z]\) is the \(h^*\)-polynomial.
For a poset \(P\), write \(h_P^*(z):=h_{\mathcal O(P)}^*(z)\).
\end{definition}

\begin{theorem}[Descent distribution representation: \texorpdfstring{\(h^*\)}{h*} as descent generating function of linear extensions]\label{thm:pom-hstar-descent-linext}
Suppose \(P\) is a \emph{naturally labeled} poset (i.e., \(p\le_P q \Rightarrow p\le q\) as integer labels).
Then its order polytope's \(h^*\)-polynomial satisfies
\[
h_P^*(z)=\sum_{\pi\in\mathrm{LinExt}(P)} z^{\mathrm{des}(\pi)},
\qquad
\mathrm{des}(\pi):=\#\{1\le j<N:\ \pi_j>\pi_{j+1}\}.
\]
In particular, \(h_P^*(1)=e(P)=N!\,\mathrm{vol}(\mathcal O(P))\).
\end{theorem}
\begin{proof}
The standard unimodular triangulation of the order polytope is indexed by linear extensions (see the simplex family \(\{\Delta_\pi\}\) in Theorem \ref{thm:pom-order-polytope-volume-linext}),
whose shelling \(h\)-vector is given by descent statistics; for unimodular triangulations, the \(h\)-polynomial coincides with the \(h^*\)-polynomial.
Thus, \(h_P^*(z)\) is the descent generating function of linear extensions.
Substituting \(z=1\) and using Theorem \ref{thm:pom-order-polytope-volume-linext} yields \(h_P^*(1)=e(P)=N!\mathrm{vol}(\mathcal O(P))\). \qed
\end{proof}

\begin{remark}[Refined invariant of \texorpdfstring{\(Z_n\)}{Z_n}: descent structure refines the Euler number]\label{rem:pom-fence-hstar-refinement}
For the fence poset \(Z_n\), although its standard labeling \(1\prec 2\succ 3\prec\cdots\) is not naturally labeled, one can relabel via any linear extension to obtain an isomorphic naturally labeled poset;
since \(\mathcal O(P)\) preserves lattice equivalence under coordinate permutation, the \(h^*\)-polynomial and volume do not depend on this choice.
Thus, \(h_{Z_n}^*(z)\) gives an auditable refinement of \(E_n=e(Z_n)\): its coefficient sum is \(E_n\), while the coefficient distribution records the descent structure of scheduling sequences.
\par
It is worth emphasizing that the Euler zigzag number \(E_n\) simultaneously counts alternating permutations and has the same cardinality as the linear extension count of \(Z_n\),
but the two sets are not identical element-by-element; the descent structure recorded by \(h_{Z_n}^*(z)\) is a refined statistic with reference to ``linear extensions under natural labeling,''
thereby avoiding the misreading of the fixed up-down pattern of alternating permutations as the (degenerate) descent distribution of \(h^*\).
\end{remark}
