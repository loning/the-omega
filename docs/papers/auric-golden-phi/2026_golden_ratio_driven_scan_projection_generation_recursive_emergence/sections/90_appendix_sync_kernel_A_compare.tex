\subsubsection{核对比：不同字母表/不同折叠核的统一接口（\texorpdfstring{$(p)$}{p}-local $\Rightarrow$ 在线延迟）}\label{app:sync-kernel-A-compare}
本附录的目标是把“更换字母表/更换归一化核”统一压成同一类可比对象，并把对比量落到可计算的同步图不变量（$\zeta$、primitive 轨道谱、“有限部常数”等）上。

\paragraph{把核视为 digit-set conversion}
固定基数 $\beta=\varphi$。给定输出字母表 $A$，记输入字母表为
\[
B:=A+A.
\]
所谓“折叠核/归一化核”在算法层面就是一个保持值的不变换（digit-set conversion）：
\[
K:\ B^{\ZZ}\to A^{\ZZ},
\qquad
\sum_{j\in\ZZ}(Ku)_j\beta^j=\sum_{j\in\ZZ}u_j\beta^j,
\]
并且要求有限支撑输入映到有限支撑输出（以对应有限表示的加法）。

\begin{definition}[\texorpdfstring{$(p)$}{p}-local、memory 与 anticipation]\label{def:plocal}
设 $\varphi:A^{\ZZ}\to B^{\ZZ}$。称 $\varphi$ 为 $(p)$-local（滑动块码），若存在 $r,t\in\NN$ 与局部规则 $\Phi:A^{p}\to B$（其中 $p=r+t+1$），使得对任意 $u\in A^{\ZZ}$ 与 $v=\varphi(u)$，
\[
v_j=\Phi(u_{j+t},\dots,u_{j-r})\qquad(\forall j\in\ZZ).
\]
其中 $r$ 称为 memory，$t$ 称为 anticipation（见 \cite{FrougnyHellerPelantovaSvobodovaParallelAlgorithmsSlides} 的 Definition~1；亦参见 \cite{FrougnyPelantovaSvobodova2013MinimalDigitSets} 的定义口径）。
\end{definition}

\begin{corollary}[单趟在线延迟]\label{cor:online-delay-from-plocal}
若 $\varphi$ 为 $(t,r)$-local，则将其按 MSDF 顺序单趟在线执行时，输出 $v_j$ 至少需等待读到 $u_{j+t}$，因此其单趟在线延迟为
\[
\delta(\varphi)=t.
\]
\end{corollary}

\paragraph{同步图与可比的谱指纹}
一旦把 $K$ 实现为确定性在线有限状态转导器，其同步部分给出有限有向图 $\Gamma_K$（状态集 $Q_K$，边标记为输入字母表符号），并可在边上定义“原子活动度”势（见后文 $\kappa$）。据此可把核的差异压缩为一组可比的谱指纹。

\begin{definition}[核的谱指纹（A 路线）]\label{def:kernel-spectral-fingerprint}
对给定核 $K$，定义其谱指纹为
\[
\mathfrak{F}(K):=\bigl(\delta(K),\ |Q_K|,\ \zeta_K(z,0),\ \lambda_K(u),\ \{B_{K,n}(u)\}_{n\le N}\bigr),
\]
其中 $\delta(K)$ 为单趟在线延迟（推论 \ref{cor:online-delay-from-plocal}），$|Q_K|$ 为同步状态数，$\zeta_K(z,0)$ 为 carry-free 骨架的 $\zeta$，$\lambda_K(u)=\rho(M_K(u))$ 为带权压力曲线，$B_{K,n}(u)$ 为 primitive 分层谱（其严格定义见后文）。
\end{definition}

\paragraph{与“最小冗余字母表并行核”的关系（外部基准）}
在同一基数 $\beta=\varphi$ 上，\cite{FrougnyPelantovaSvobodova2013MinimalDigitSets} 的 Corollary~38 证明：并行加法可在
\[
A=\{-1,0,1\}\quad\text{或}\quad A=\{0,1,2\}
\]
上完成，且两者均为\textbf{最小}字母表（$|A|=3$）。此外，针对 $A=\{-1,0,1\}$ 的一个显式并行算法（Algorithm~G）在讲义中给出，并注明该加法在该字母表上为 \textbf{21-local}（因此在在线口径下粗对应 $\delta\approx 10$；见 \cite{FrougnyHellerPelantovaSvobodovaParallelAlgorithmsSlides}，以及 Leiden 讲稿中的 21/13/9-local 对照 \cite{Frougny2012NumbersAsStreamsLeiden}）。

这些并行核一旦被编译为确定性在线转导器，其同步图同样给出 $\zeta$ 与 primitive 谱，从而与表 \ref{tab:sync-kernel-A-compare} 的“机制核”完全共享同一比较接口；本文把它作为后续可直接对接的外部基准，而不在当前版本中展开完整同步图编译。
\par
另一方面，\cite{Frougny1999OnlineAddition} 证明：黄金比例基下的加法可由在线有限自动机实现，但在非冗余表示口径下并不满足并行（$(p)$-local）可计算性；这为本文的 “在线核 vs.\ 并行核” 提供一个外部参照边界。

\paragraph{三核在文献中的精确锚点：零式与 locality}
为避免“同一核在不同讲义/实现中口径漂移”，本文对三类并行核的引用统一锚定在同一份讲义 \cite{FrougnyHellerPelantovaSvobodovaParallelAlgorithmsSlides}（以及其在 AAMP VIII 的摘要表述 \cite{SvobodovaAAMPVIIIParallelAddition}）：
\begin{itemize}
  \item \textbf{强零式 + $9$-local（黄金比例基）}：以 $-\beta^4+7-\beta^{-4}=0$ 为注入零式，在 $A=\{-5,\dots,5\}$ 上得到加法为 $9$-local；
  \item \textbf{弱零式 + $13$-local（黄金比例基）}：以 $-\beta^2+3-\beta^{-2}=0$ 为注入零式，在 $A=\{-3,\dots,3\}$ 上得到加法为 $13$-local；
  \item \textbf{最小字母表 + $21$-local（Algorithm G）}：在 $A=\{-1,0,1\}$ 上给出两相消元算法 $G=A\circ B$，并明确加法为 $21$-local；其两相消元均使用弱零式 $-\beta^2+3-\beta^{-2}=0$。
\end{itemize}
上述三点正是本附录后续把“不同字母表/不同折叠核”压缩成同一类同步图指纹对象的共同输入。

\begin{definition}[零式注入的 \texorpdfstring{$(+;q)$}{(+;q)}-场更新]\label{def:zero-injection}
给定 Laurent 多项式
$$
Z(X)=\sum_{i=-h}^{k}b_iX^i,\qquad b_i\in\ZZ,
$$
以及满足 $Z(\tau)=0$ 的基数 $\tau$。取一个有限取值的整数场 $q=(q_j)_{j\in\ZZ}$，并定义一次更新
$$
z'_j:=z_j-\sum_{i=-h}^{k}b_i\,q_{j-i}.
$$
\end{definition}

\begin{proposition}[值保持]\label{prop:zero-injection-value-preserving}
在定义 \ref{def:zero-injection} 的条件下（并假设 $q$ 仅有限支撑，或 $z,q$ 皆为有限支撑序列），有
\[
\sum_{j\in\ZZ} z'_j\tau^j=\sum_{j\in\ZZ} z_j\tau^j,
\]
即该更新保持 $\tau$-加权值不变。
\end{proposition}

\begin{proof}
把定义 \ref{def:zero-injection} 代入并交换求和（有限支撑保证合法）：
\[
\sum_j z'_j\tau^j=\sum_j z_j\tau^j-\sum_j\sum_{i=-h}^{k}b_i q_{j-i}\tau^j
=\sum_j z_j\tau^j-\sum_{i=-h}^{k}b_i\tau^i\sum_\ell q_{\ell}\tau^{\ell}.
\]
而 $\sum_{i=-h}^{k}b_i\tau^i=Z(\tau)=0$，故第二项为 $0$，证毕。
\end{proof}

\paragraph{两条稀疏零式（黄金比例基）}
在黄金比例基的两条稀疏零式特例中（见 \cite{FrougnyHellerPelantovaSvobodovaParallelAlgorithmsSlides} 的 Example/Algorithm 段落），它退化为：
\begin{itemize}
  \item \textbf{强零式（$9$-local）}：
  $$
  z'_j=z_j-7q_j+q_{j-4}+q_{j+4}.
  $$
  \item \textbf{弱零式（$13$-local 与 $21$-local 的底层）}：
  $$
  z'_j=z_j-3q_j+q_{j-2}+q_{j+2}.
  $$
\end{itemize}

\paragraph{把三类并行核编译到可 BFS 的在线机（单流）}
为把“核的局部规则”落实到可计算对象，我们按 stencil 的同余类解耦，把核限制到单一同余流（$\pm 4$ stencil 对应模 $4$ 的 4 条流，$\pm 2$ stencil 对应模 $2$ 的 2 条流）。在单流上，核成为一个确定性在线机：每一步读取一个输入符号 $v_n$（逐位和/中间字母表符号），更新有限状态并输出固定延迟后的输出符号。由此可直接用 BFS 枚举全部可达状态，得到单流同步图 $\Gamma^{\mathrm{flow}}_K$。

在比较层面，\textbf{无权计数}会退化：确定性完备读入使得每状态出边数固定为 $|B|$，从而 $\zeta$ 只反映 $|B|$。因此我们引入边势
\[
\kappa(e):=\#\{\text{本步新决定的非零 }q\text{ 事件}\},
\]
并以 $u^{\kappa(e)}$ 加权得到带权邻接矩阵族 $M_K(u)$，其谱半径 $\lambda_K(u)=\rho(M_K(u))$ 形成“carry 活跃度”的可比曲线。进一步取 $u=0$ 得到 carry-free 骨架（仅保留 $\kappa=0$ 的边），其 $\zeta$ 与 primitive 谱给出一个极硬的结构指纹。

\paragraph{两变量 \texorpdfstring{$\zeta_K(z,u)$}{zeta(z,u)} 与 primitive 分层谱 \texorpdfstring{$B_{K,n}(u)$}{B\_{K,n}(u)}（严格定义）}
令 $\Gamma_K$ 为单流同步图，其顶点集为状态集 $Q_K$，边集记为 $E_K$。对每条边 $e:i\to j$ 的活动度沿用上式 $\kappa(e)\in\ZZ_{\ge 0}$。定义带权邻接矩阵族（把同一对端点的多条边按权重累加）
\[
(M_K(u))_{ij}:=\sum_{e:i\to j}u^{\kappa(e)}\in\ZZ[u].
\]
对该有限图对应的边移位（SFT），定义两变量动力学 $\zeta$（\cite{BrinStuck2002,ParryPollicott1990Zeta}）
\[
\zeta_K(z,u)
:=\exp\!\left(\sum_{n\ge 1}\frac{\Tr(M_K(u)^n)}{n}z^n\right)
=\frac{1}{\det(I-zM_K(u))}.
\]
对每条 primitive 周期轨道 $\gamma$ 定义
\[
\kappa(\gamma):=\sum_{e\in\gamma}\kappa(e),
\]
并定义 primitive 分层谱多项式
\[
B_{K,n}(u):=\sum_{\substack{\gamma\ \mathrm{primitive}\\ |\gamma|=n}}u^{\kappa(\gamma)}\in\ZZ[u].
\]
记周期迹多项式 $a_{K,n}(u):=\Tr(M_K(u)^n)$。则 Witt/Möbius 反演给出从周期迹到 primitive 指纹的完全显式公式：
\[
B_{K,n}(u)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{K,n/d}(u^d)
\qquad\Bigl(\Leftrightarrow\ a_{K,n}(u)=\sum_{d\mid n} d\,B_{K,d}(u^{n/d})\Bigr),
\]
其中 $\mu$ 为 Möbius 函数。这组对象即为“核指纹”：不同核的差异被压缩为同一组可比序列 $\{B_{K,n}(u)\}_{n\le N}$ 与（等价地）$\zeta_K(z,u)$ 的系数结构。

\paragraph{全局块级在线机的组装：直积状态与张量积矩阵}
在“块时间” $n$ 下，把所有同余流同步喂入。对 $\pm 4$ stencil（强核）有 $m=4$ 条流；对 $\pm 2$ stencil（弱核）有 $m=2$ 条流。全局状态是流状态的直积
\[
S_n^{\mathrm{glob}}=\bigl(S_n^{(0)},\dots,S_n^{(m-1)}\bigr),
\]
因此全局可达状态数满足 $|Q_{\mathrm{glob}}|=|Q_{\mathrm{flow}}|^m$。

对带权矩阵族 $M(u)$（边权为 $u^{\kappa(e)}$），由于全局一步等价于 $m$ 个独立流的一步并行，且 $\kappa$ 在流之间可加，故矩阵满足张量积结构
\[
M_{\mathrm{glob}}(u)=\bigotimes_{r=0}^{m-1} M_{\mathrm{flow}}(u),
\qquad
\lambda_{\mathrm{glob}}(u)=\rho(M_{\mathrm{glob}}(u))=\lambda_{\mathrm{flow}}(u)^m.
\]
同理，均匀输入下的活动度均值满足 $\mathbb{E}[\kappa_{\mathrm{glob}}]=m\,\mathbb{E}[\kappa_{\mathrm{flow}}]$。

\begin{proposition}[压力曲线的端点、单调性与全局提升]\label{prop:lambda-endpoints-monotone-uplift}
对任一单流同步图的带权邻接矩阵族 $M_K(u)$（定义见上文），其谱半径
\[
\lambda_K(u):=\rho(M_K(u))\qquad (u\in[0,1])
\]
满足：
\begin{enumerate}
  \item \textbf{端点（$u=0$）}：$\lambda_K(0)=\rho(A_0)$，其中 $A_0$ 为 carry-free 骨架邻接矩阵（保留且仅保留 $\kappa=0$ 的边）。
  \item \textbf{端点（$u=1$）}：$\lambda_K(1)=|B|$，其中 $|B|$ 为单步输入字母表大小（确定性完备读入导致每个状态恰有 $|B|$ 条出边）。
  \item \textbf{单调性}：$u\mapsto \lambda_K(u)$ 在 $[0,1]$ 上非减且连续。
  \item \textbf{全局提升}：若全局由 $m$ 个同构流的直积装配，则
  \[
  \lambda_{\mathrm{glob}}(u)=\lambda_{\mathrm{flow}}(u)^m,
  \qquad
  \log\lambda_{\mathrm{glob}}(u)=m\,\log\lambda_{\mathrm{flow}}(u).
  \]
\end{enumerate}
\end{proposition}

\begin{proof}
\textbf{(1)} $u=0$ 时 $u^{\kappa(e)}$ 仅在 $\kappa(e)=0$ 时为 $1$，其余为 $0$，故 $M_K(0)$ 等于 carry-free 骨架邻接矩阵 $A_0$，从而 $\lambda_K(0)=\rho(A_0)$。

\textbf{(2)} $u=1$ 时每条边权均为 $1$，于是 $M_K(1)$ 的每行元素和等于该状态的出边数；对确定性完备读入机，每状态恰有 $|B|$ 条出边，所以 $M_K(1)\mathbf{1}=|B|\mathbf{1}$，即 $|B|$ 为特征值。另一方面 $\rho(M_K(1))\le \|M_K(1)\|_{\infty}=\max_i\sum_j (M_K(1))_{ij}=|B|$，故 $\rho(M_K(1))=|B|$。

\textbf{(3)} 对 $u\in[0,1]$，矩阵元素 $(M_K(u))_{ij}$ 是 $u$ 的非负系数多项式，因此随 $u$ 单调不减且连续。谱半径对非负矩阵满足单调性：若 $0\le A\le B$（逐项），则 $\rho(A)\le \rho(B)$；并且由连续依赖性（有限维特征值连续）可得 $u\mapsto \rho(M_K(u))$ 连续。

\textbf{(4)} 全局装配下 $M_{\mathrm{glob}}(u)=\bigotimes_{r=0}^{m-1} M_{\mathrm{flow}}(u)$，而 Kronecker 乘积满足谱半径可乘：$\rho(A\otimes B)=\rho(A)\rho(B)$，由归纳即得 $\rho(M_{\mathrm{glob}}(u))=\rho(M_{\mathrm{flow}}(u))^m$，取对数即得结论。
\end{proof}

\begin{proposition}[零温附近的线性响应不变量]\label{prop:lambda-linear-response}
写
\[
M_K(u)=A_0+uA_1+u^2A_2+\cdots,
\]
其中 $A_k$ 由 $\kappa(e)=k$ 的边贡献组成。若 $A_0$ 的 Perron 根 $\lambda_0$ 为简单特征值（carry-free 骨架为 primitive SFT 时成立），则 $\lambda_K(u)$ 在 $u=0$ 附近解析，并满足
\[
\lambda_K(u)=\lambda_0+c_1u+O(u^2),
\qquad
c_1=\frac{\ell^{\mathsf T}A_1 r}{\ell^{\mathsf T}r},
\]
其中 $r,\ell$ 分别是 $A_0$ 的右/左 Perron 特征向量。
\end{proposition}

\begin{corollary}[三核的精确一阶系数]\label{cor:lambda-linear-response-kernels}
对单流编译的 $K_9/K_{13}/K_{21}$，carry-free 骨架满足
\(
\lambda_0=7,\ 3,\ \varphi
\)；据此得到
\[
\lambda_{9}(u)=7+14u\quad(\text{严格线性}),
\]
\[
\lambda_{13}(u)=3+\frac{8}{3}u+O(u^2),
\qquad
\lambda_{21}(u)=\varphi+\sqrt5\,u+O(u^2).
\]
其中 $\sqrt5=\varphi+\varphi^{-1}$，并且一般有 $c_1\in\QQ(\lambda_0)$。
\end{corollary}

\begin{corollary}[零温 carry 密度的线性律与排序]\label{cor:kappa-linear-response-order}
在权重 $u^{\kappa}$ 下，平衡态的平均 carry 密度满足
\[
\mathbb{E}_u[\kappa]=u\frac{\mathrm{d}}{\mathrm{d}u}\log\lambda_K(u),
\]
因此
\[
\mathbb{E}_u[\kappa]=\frac{c_1}{\lambda_0}u+O(u^2).
\]
对三核得到
\[
\mathbb{E}_u[\kappa]=2u+O(u^2)\ (K_9),\qquad
\mathbb{E}_u[\kappa]=\frac{8}{9}u+O(u^2)\ (K_{13}),\qquad
\mathbb{E}_u[\kappa]=(3-\varphi)u+O(u^2)\ (K_{21}),
\]
从而零温附近的“抗 carry”排序为 $K_{13}<K_{21}<K_{9}$。
\end{corollary}

此外，流级延迟 $\delta_{\mathrm{flow}}$（按块时间计）对应的数字级在线延迟为

\begin{remark}[近正则缺陷 \texorpdfstring{$\Delta_{\mathrm{reg}}$}{Delta_reg}：把 \texorpdfstring{$(\rho(A),\rho(B))$}{rhoA,rhoB} 转成可比较的几何先验]\label{rem:ihara-delta-reg}
在 $d$-正则无向图上有精确关系 \(\rho(B)=d-1\)、\(\rho(A)=d\)，从而 \(\rho(A)-\rho(B)-1=0\)。
因此可定义
$$
\Delta_{\mathrm{reg}}:=\rho(A)-\rho(B)-1
$$
作为“近正则缺陷”：\(\Delta_{\mathrm{reg}}\) 越小，表示 ordinary walk 与 non-backtracking walk 的增长率关系越接近正则情形（差值锁定为 \(1\)）；\(\Delta_{\mathrm{reg}}\) 越大则表示度分布与局部结构更不均匀，non-backtracking 的增长率相对 ordinary walk 更受抑制。

该量与核版 RH/Ramanujan 指标 \(R_K(u)=\lambda_\star(u)/\sqrt{\lambda_1(u)}\) 互补：前者是骨架几何的“先验不变量”，后者是带权 Hashimoto 算子的“谱隙质量”指标；两者共同决定无回溯 prime cycle 口径下的可达误差尺度。
\end{remark}

\begin{remark}[回溯惩罚 \texorpdfstring{$\pi_{\mathrm{bt}}$}{pi_bt}：ordinary walk 到 non-backtracking 的增长率损失]\label{rem:ihara-pi-bt}
除近正则缺陷外，还可定义无量纲的回溯惩罚
$$
\pi_{\mathrm{bt}}:=1-\frac{\rho(B)}{\rho(A)}\in[0,1).
$$
它直接量化“禁止回溯”带来的指数增长率损失比例。表 \ref{tab:parallel-addition-kernels-ihara-fingerprint} 的输出显示：在三核中 \(21\)-local 的 \(\pi_{\mathrm{bt}}\) 显著更大，表明其原子操作骨架中“可逆撤销/短回溯”结构占比更高，因此以 Ihara primes（无回溯）对素数影子做正规化的必要性也更强。
\end{remark}

\paragraph{Ihara primes：无回溯原子环的规范化素数}
在“原子操作图”（边为原子操作、反向边为其逆）上，把 prime-like 对象定义为\textbf{无回溯、无尾（tailless）的 primitive 环}：即闭环 $p=e_1\cdots e_\ell$ 满足 $e_{i+1}\neq e_i^{-1}$ 且 $e_1\neq e_\ell^{-1}$，并且不为更短环的幂；此外按循环平移视为同一环。其 Euler 乘积为
\[
Z_K(t)=\prod_{p}\left(1-t^{\ell(p)}\right)^{-1},
\]
其中 $p$ 遍历 Ihara primes。该定义剔除了“立即撤销”的伪周期，更贴合可逆原子操作的不可约模式。

\paragraph{带权 Ihara--Bass：carry 能量的谱指纹}
令 $\kappa(e)$ 为每步 carry 触发能量，取权重 $u^{\kappa(e)}$。在无回溯边空间上定义 Hashimoto 算子 $B(u)$（边 $e\to e'$ 可接且不回溯时赋权 $u^{\kappa(e')}$）。则带权 Ihara \(\zeta\) 为
\[
Z_K(u,t):=\prod_{p}\left(1-t^{\ell(p)}u^{\kappa(p)}\right)^{-1}
=\frac{1}{\det(I-tB(u))},
\]
从而“原子能量”进入谱指纹并保持行列式可计算性（Bass--Hashimoto 三项行列式公式可视为上述定义在无向骨架上的等价表述）。该双参数对象给出一族\textbf{矩阵多项式行列式不变量}，比单一的周期点迹更贴近“可逆重写群胚”的素数谱。

\paragraph{核版 RH/Ramanujan 指标}
记 $\lambda_1(u)=\rho(B(u))$ 为无回溯主谱半径，$\lambda_\star(u)$ 为最大非平凡特征值模。定义
\[
R_K(u):=\frac{\lambda_\star(u)}{\sqrt{\lambda_1(u)}}.
\]
当 $R_K(u)\le 1$ 时，prime cycle 的长度计数可达平方根级误差：对对应的 primitive 计数 $p_n(u)$，谱分解给出主项
\[
p_n(u)=\frac{\lambda_1(u)^n}{n}+O\!\left(\frac{\lambda_1(u)^{n/2}}{n}\right),
\]
这在有限矩阵框架下无需额外 Tauberian 工具。该指标为“素数影子质量”的可检验谱界（Ihara--RH/Ramanujan 类比；参见 \cite{TerrasZetaGraphs,StarkTerras1996GraphCoverings} 的图版 RH 综述）。

\paragraph{核版 Chebotarev：群标签的同余类分布}
给每条边赋有限群标签 $g(e)\in G$（隐含属性），对 prime 环定义 Frobenius 类 $\mathrm{Fr}(p)=\prod_{e\in p}g(e)$。对每个不可约表示 $\rho$，定义扭曲 Hashimoto 算子 $B_{\rho}(u)$ 与
\(
L_{K,\rho}(u,t)=\det(I-tB_\rho(u))^{-1}
\)。
由定理 \ref{thm:kernel-chebotarev-exp}（见主文同余类分层）可得同余类等分布的误差由 $\max_{\rho\neq\mathbf 1}\rho(B_\rho(u))$ 控制。于是可定义可审计的偏差率指标
\[
\theta_K(u):=\max_{\rho\neq\mathbf 1}\frac{\rho(B_\rho(u))}{\rho(B_{\mathbf 1}(u))},
\]
并据此量化“素数在同余类中的收敛速度”，形成“核版 Chebotarev/Artin-$L$”的完全可计算接口。

\paragraph{零温能隙：最小平均回路的可计算性}
把 $\kappa(e)$ 视为能量，定义 prime 环平均能量
\(
\bar\kappa(p)=\kappa(p)/\ell(p)
\)
并令
\[
\Delta_K:=\min_{p:\bar\kappa(p)>0}\bar\kappa(p).
\]
在无回溯边图（Hashimoto 图）上该问题等价于\textbf{最小平均回路}，可由 Karp 算法在多项式时间内求出（见 \cite{Karp1978MinCycleMean}）。$\Delta_K$ 给出“基态/第一激发态能隙”的可比不变量，并控制 $u\to 0$ 时第一个非平凡修正的主尺度。

\begin{proposition}[密度能隙的退化机制与隧穿作用量]\label{prop:ihara-gap-degeneracy}
若 carry-free 骨架含有任意长的无回溯闭环（即存在 \(\kappa(p)=0\) 的 prime 环族且 \(\ell(p)\to\infty\)），并且存在某个正能量无回溯闭环 \(q\)（\(\kappa(q)>0\)）能与任意长的零能量闭环通过局部改写拼接成无回溯闭环（保持 tailless 条件），则正能量闭环的平均密度 \(\bar\kappa=\kappa/\ell\) 可做到任意小，从而以“最小正密度”定义的 \(\Delta_K\) 要么为 \(0\)，要么仅有下确界而无最小值。
\end{proposition}

$$
A_K:=\min\{\kappa(p):\ p\ \text{为无回溯 prime 环且}\ \kappa(p)>0\}\in\ZZ_{>0},

\end{definition}

\paragraph{大规模素数谱推进：Hutch++ 的 PSD 化轨道}

\paragraph{做“素数影子”对比的最省力接口}
对任意核（无权或带权），最省力的计算接口是“先算周期迹，再用反演提 primitive”。具体地：
\[
a_n=\Tr(M^n),\qquad
p_n=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}.
\]
对带权情形（边权记作 $u^{\kappa}$，得到矩阵 $M(u)$），primitive 的加权谱满足（Witt 分解）
\[
p_n(u)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}(u^d),
\qquad a_n(u)=\Tr(M(u)^n),
\]
与本附录 \S\ref{app:witt-weighted} 的推导一致。
而在全局块级直积装配下，若各流同构，则
\[
a_n^{\mathrm{glob}}(u)=a_n^{\mathrm{flow}}(u)^m,
\]
从而只需单流的 $a_n(u)$ 即可得到全局的 primitive 谱（再做一次 Möbius 反演即可）。

\begin{proposition}[carry-free 骨架的三种极限 SFT 与 \texorpdfstring{$\zeta_K(z,0)$}{zeta(z,0)}]\label{prop:carry-free-skeleton-three}
对三类 $\beta=\varphi$ 的并行核（$9/13/21$-local），取 $u=0$ 等价于在同步图上仅保留 $\kappa=0$ 的边（完全不触发任何非零 $q$ 事件），得到 carry-free 骨架 SFT。其不可约骨架分别为：
\begin{enumerate}
  \item \textbf{$K_9$（$9$-local，SRZ）}：骨架为 $7$-全移位，因此
  \[
  \zeta_{9}(z,0)=\frac{1}{1-7z}.
  \]
  \item \textbf{$K_{13}$（$13$-local，WRZ 迭代）}：骨架为 $3$-全移位，因此
  \[
  \zeta_{13}(z,0)=\frac{1}{1-3z}.
  \]
  \item \textbf{$K_{21}$（$21$-local，Algorithm G）}：骨架为黄金均值移位（禁止相邻 $11$），因此
  \[
  \zeta_{21}(z,0)=\frac{1}{1-z-z^2}.
  \]
\end{enumerate}
这给出 A-对比的硬分化点：仅最小字母表核在 $u=0$ 层面就强制黄金均值约束。
\end{proposition}

\begin{proof}
\textbf{(1)} 对 $K_9$（Algorithm S / SRZ），强核一步的 $q_j$ 选择以“把 $z_j-q_jB$ 拉回内层字母表 $A'$”为准则（\cite{FrougnyHellerPelantovaSvobodovaParallelAlgorithmsSlides}，Example~9）。carry-free 即 $q\equiv 0$，等价于所有位置都已落在内层 $A'=\{-3,\dots,3\}$，因此每步有 $7$ 个自由输入符号且无额外约束，骨架为 $7$-全移位，$\zeta=1/(1-7z)$。

\textbf{(2)} 对 $K_{13}$（Algorithm W / WRZ），每次迭代采用“若 $z_j\in A'$ 则 $q_j=0$，否则 $q_j=\mathrm{sgn}(z_j)$”的 neighbour-free 规则（\cite{FrougnyHellerPelantovaSvobodovaParallelAlgorithmsSlides}，Algorithm~W 与 Example~12）。carry-free 要求每次迭代均不触发非零 $q$，因此 $z_j$ 始终处于内层 $A'=\{-1,0,1\}$，得到 $3$-全移位骨架，$\zeta=1/(1-3z)$。

\textbf{(3)} 对 $K_{21}$（Algorithm G），先经 Algorithm A 去除 $-2$，再经 Algorithm B 去除 $2$（\cite{FrougnyHellerPelantovaSvobodovaParallelAlgorithmsSlides}，Algorithm~A/B/G）。carry-free 要求两相均不触发：A 相的触发条件包含 $z_j=-2,-1$ 等，故排除这些；B 相首先排除 $z_j=2$，从而剩余符号为 $\{0,1\}$；并且 B 相的邻敏条件“$z_j=1$ 且 $(z_{j+2}\ge 1$ 或 $z_{j-2}\ge 1)$”在单流坐标上恰为“禁止相邻 $11$”。因此骨架为黄金均值移位，其邻接矩阵为 $\begin{pmatrix}1&1\\1&0\end{pmatrix}$，故 $\zeta=1/(1-z-z^2)$（\cite{LindMarcus1995}）。
\end{proof}

\paragraph{两变量 \texorpdfstring{$\zeta_K(z,u)$}{zeta(z,u)} 指纹（有限图版本）}
上文已给出 $\zeta_K(z,u)$ 与 $B_{K,n}(u)$ 的严格定义；此处仅重申“有限图 $\Rightarrow$ 行列式 $\Rightarrow$ Euler 乘积”的封闭链条。对单流同步图的带权邻接矩阵族 $M_K(u)$，定义
\[
\zeta_K(z,u)
:=\exp\!\left(\sum_{n\ge 1}\frac{\Tr(M_K(u)^n)}{n}z^n\right)
=\frac{1}{\det(I-zM_K(u))},
\]
并以 Euler 乘积定义 primitive 分层谱多项式
\(
B_n(u)=\sum_{|\gamma|=n}u^{\kappa(\gamma)}
\)。
这些对象把“核的差异”压缩为同一组可比不变量：$\lambda(u)=\rho(M_K(u))$、$\zeta_K(z,u)$、以及 $\{B_n(u)\}_{n\le N}$（见 \cite{Ruelle1976ZetaExpanding,ParryPollicott1990Zeta,BrinStuck2002}）。

\subsubsection{A 分支：核指纹实验协议（端点闭式 \texorpdfstring{$+$}{+} 中间 \texorpdfstring{$u\in(0,1)$}{u in (0,1)} 近似；一页版）}\label{app:kernel-fingerprint-protocol}
本附录把 $9/13/21$-local 三核统一压到同一对象组
\[
\bigl(M_K(u),\ \zeta_K(z,u),\ \lambda_K(u),\ B_{K,n}(u)\bigr),
\]
并给出“端点闭式基准 + 中间 $u$ 可扩展近似”的统一可复现实验协议。

\begin{algorithm}[A 分支核指纹实验协议（不爆内存）]\label{alg:kernel-fingerprint-protocol}
给定核 $K\in\{K_9,K_{13},K_{21}\}$，以及默认参数
\[
U:=\{0,0.25,0.5,0.75,1\},\quad N_{\mathrm{end}}:=20,\quad \mathcal{N}_{\mathrm{mid}}:=\{3,5,7\},\quad S:=200.
\]
执行：
\begin{enumerate}
  \item \textbf{单流编译（BFS）}：按同余流解耦（$K_9$ 取 $m=4$，$K_{13},K_{21}$ 取 $m=2$），对单流构造同步图 $\Gamma^{\mathrm{flow}}_K$，并在边上定义活动度势 $\kappa(e)$。得到单流带权邻接矩阵族 $M_K(u)$ 以及基本结构量（$\delta$、$|Q|$、$\mathbb{E}[\kappa]$）。
  \item \textbf{端点闭式（零误差）}：
  \begin{enumerate}
    \item $u=0$：取 carry-free 骨架 $A_0:=M_K(0)$，计算 $\zeta_K(z,0)=\det(I-zA_0)^{-1}$，并由 Möbius 反演得到 primitive 轨道数 $p_n(0)$（$n\le N_{\mathrm{end}}$）。
    \item $u=1$：取无权基线 $M_K(1)$，其 $\zeta_K(z,1)$ 与 $p_n(1)$ 退化为 full shift 的闭式（仅依赖输入字母表大小）。
  \end{enumerate}
  \item \textbf{多流提升（避免直积大图）}：用张量积闭式
  \[
  M_{\mathrm{glob}}(u)=M_{\mathrm{flow}}(u)^{\otimes m},\qquad
  \Tr(M_{\mathrm{glob}}(u)^n)=\Tr(M_{\mathrm{flow}}(u)^n)^m,\qquad
  \lambda_{\mathrm{glob}}(u)=\lambda_{\mathrm{flow}}(u)^m,
  \]
  并由一次 Möbius 反演得到全局 primitive 端点谱。
  \item \textbf{中间 $u$（矩阵自由近似）}：对每个 $u\in U\cap(0,1)$：
  \begin{enumerate}
    \item 计算压力样本 $\lambda_K(u)=\rho(M_K(u))$（幂法，仅需稀疏乘向量）。
    \item 对每个 $n\in\mathcal{N}_{\mathrm{mid}}$，用 Hutchinson 估计近似 $a_n(u)=\Tr(M_K(u)^n)$，并以 Möbius 反演得到 $\widehat B_{K,n}(u)$。为降低 Möbius 组合中的相消方差，默认使用 common random numbers（在所有 $u$ 与幂次间复用同一组 Rademacher 探针）。
  \end{enumerate}
  \item \textbf{输出（论文可直接 \texorpdfstring{$\backslash$}{\textbackslash}input）}：写出三类表格片段：
  \[
  \texttt{tab\_parallel\_addition\_kernels\_fingerprint\_main.tex},\quad
  \texttt{tab\_parallel\_addition\_kernels\_endpoint\_primitive.tex},\quad
  \texttt{tab\_parallel\_addition\_kernels\_fingerprint\_u\_samples.tex},
  \]
  并将全部数值与采样参数写入 \texttt{artifacts/export/*.json} 以供审计复算。
\end{enumerate}
\end{algorithm}

\subsubsection{中间权重 \texorpdfstring{$u\in(0,1)$}{u in (0,1)} 的可控近似：随机迹估计 \texorpdfstring{$+$}{+} Möbius 反演（协议）}\label{app:kernel-fingerprint-hutchinson}
上一小节给出了端点 $u\in\{0,1\}$ 的闭式“零误差”谱表与指纹；但在 $u\in(0,1)$ 时，$M_K(u)^n$ 的非零结构会迅速变得致密，因而“显式形成 $M^n$ 再取迹/系数”的精确计算不可扩展。为把核指纹推进到连续谱对比（压力曲线 $\lambda_K(u)$ 与分层 prime spectrum $\{B_{K,n}(u)\}$），本文采用一个只依赖\textbf{稀疏矩阵乘向量}的可控近似协议。

\begin{definition}[Hutchinson 随机迹估计]\label{def:hutchinson-trace}
令 $A\in\RR^{N\times N}$ 为任意矩阵。取随机向量 $\mathbf r\in\{\pm 1\}^N$，其坐标独立同分布且 $\mathbb{P}(r_i=1)=\mathbb{P}(r_i=-1)=1/2$。定义单样本估计量
\[
T(\mathbf r;A):=\mathbf r^{\mathsf T}A\mathbf r.
\]
对 $S$ 个独立样本 $\mathbf r_1,\dots,\mathbf r_S$，定义 Hutchinson 迹估计
\[
\widehat{\mathrm{Tr}}_S(A):=\frac{1}{S}\sum_{s=1}^{S}\mathbf r_s^{\mathsf T}A\mathbf r_s.
\]
\end{definition}

\begin{proposition}[无偏性与矩阵自由实现]\label{prop:hutchinson-unbiased-mfree}
对任意 $A\in\RR^{N\times N}$，若 $\mathbf r$ 为定义 \ref{def:hutchinson-trace} 的 Rademacher 向量，则
\[
\mathbb{E}\bigl[\mathbf r^{\mathsf T}A\mathbf r\bigr]=\mathrm{Tr}(A),
\qquad
\mathbb{E}\bigl[\widehat{\mathrm{Tr}}_S(A)\bigr]=\mathrm{Tr}(A).
\]
并且当 $A=M_K(u)^n$ 时，$\mathbf r^{\mathsf T}A\mathbf r$ 可通过 $n$ 次稀疏乘向量递推计算：
\[
\mathbf v_0=\mathbf r,\qquad \mathbf v_{t+1}=M_K(u)\mathbf v_t,\qquad
\mathbf r^{\mathsf T}M_K(u)^n\mathbf r=\mathbf r^{\mathsf T}\mathbf v_n.
\]
\end{proposition}

\begin{proof}
由于 $\mathbb{E}[r_ir_j]=\delta_{ij}$，有
\(
\mathbb{E}[\mathbf r^{\mathsf T}A\mathbf r]=\sum_{i,j}A_{ij}\mathbb{E}[r_ir_j]=\sum_i A_{ii}=\mathrm{Tr}(A)
\)。
矩阵自由实现只需展开定义即可。
\end{proof}

\begin{remark}[误差口径与样本数选择]\label{rem:hutchinson-error}
对一般矩阵 $A$，Hutchinson 估计的集中界可用标准 Monte Carlo 方式处理；在 $A$ 为\textbf{对称半正定}的常见情形，可使用更强的相对误差样本复杂度界（见 \cite{Hutchinson1989ASE,AvronToledo2011Trace}）。在本文的核指纹设定中，我们采用两层可审计策略：
\begin{itemize}
  \item \textbf{固定预算 + 标准误差：}给定样本数 $S$，同时输出 $\widehat{\mathrm{Tr}}_S(M(u)^n)$ 的样本标准误差 $\mathrm{se}=s/\sqrt{S}$（$s$ 为样本标准差），把误差控制写成 $O(S^{-1/2})$ 的可见量；
  \item \textbf{pilot 自适应：}先用小 $S_0$ 估计方差与尺度，再按目标相对误差阈值放大到所需 $S$（该步骤只改变采样次数，不改变任何数学定义）。
\end{itemize}
由于本节的目的是给出一个“可落地、可复现”的协议口径，正文表格默认采用前一种固定预算输出，并把样本数 $S$ 写入表注以便审计。
\end{remark}

\begin{proposition}[由单流避免全局直积大图：\texorpdfstring{$m$}{m}-流提升]\label{prop:hutchinson-global-lift}
若全局块级核由 $m$ 条同构同余流张量装配，则
\[
M_{\mathrm{glob}}(u)=M_{\mathrm{flow}}(u)^{\otimes m},
\qquad
\mathrm{Tr}\!\left(M_{\mathrm{glob}}(u)^n\right)=\left(\mathrm{Tr}\!\left(M_{\mathrm{flow}}(u)^n\right)\right)^m.
\]
因此对任意 $n$，只需估计单流的 $\mathrm{Tr}(M_{\mathrm{flow}}(u)^n)$ 即可得到全局迹，并进一步由 Möbius 反演得到全局 primitive 分层谱：
\[
B^{\mathrm{glob}}_{n}(u)=\frac{1}{n}\sum_{d\mid n}\mu(d)\left(\mathrm{Tr}\!\left(M_{\mathrm{flow}}(u^d)^{n/d}\right)\right)^m.
\]
\end{proposition}
