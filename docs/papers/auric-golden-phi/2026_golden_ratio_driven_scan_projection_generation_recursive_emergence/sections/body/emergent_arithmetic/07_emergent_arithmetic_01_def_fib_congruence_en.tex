\subsection{From the Scale-Count Monoid Presented by Fibonacci Relations to the Natural Numbers (Quotients and Invariants)}\label{subsec:emergent-arithmetic-fib-congruence}

The key point of this section is that \textbf{the arithmetic structure is not pre-installed}; rather, it is forced to emerge by taking the quotient (or extracting the invariant) of the ``free structure of scale counts'' under the Fibonacci relations. The Zeckendorf normal form is merely a \textbf{computable canonical representative system} on this quotient structure.

Let
\[
\mathcal{D}:=\Bigl\{a=(a_1,a_2,\dots): a_k\in\NN,\ \text{only finitely many }a_k\neq 0\Bigr\}.
\]
We regard $a\in\mathcal{D}$ as a counting configuration ``occurring $a_k$ times at each scale $k$''; under componentwise addition, $\mathcal{D}$ is a free commutative monoid.
Let $e_k\in\mathcal{D}$ denote the $k$-th standard basis element ($e_k$ has a $1$ in position $k$ and $0$ elsewhere).

\begin{definition}[Congruence generated by Fibonacci relations]\label{def:fib-congruence}
Let $\equiv$ denote the smallest commutative monoid congruence on $\mathcal{D}$ such that
\begin{enumerate}
  \item \textbf{(Baseline congruence)} $e_2\equiv 2e_1$ (folding a ``repeated occurrence at the same weight'' into a single occurrence at the next scale);
  \item \textbf{(Recursive congruence)} for all $k\ge 1$,
\[
e_{k+2}\ \equiv\ e_{k+1}+e_k.
\]
\end{enumerate}
Equivalently, $\equiv$ is generated by ``baseline folding + Fibonacci recursion'' at the level of scale counts and is closed under addition.
\end{definition}

\begin{definition}[Value homomorphism (invariant)]\label{def:val-on-D}
Define the map $\mathrm{Val}:\mathcal{D}\to\NN$ by
\[
\mathrm{Val}(a):=\sum_{k\ge 1} a_k F_{k+1}.
\]
This is a commutative monoid homomorphism and is consistent with the previously defined $\mathrm{Val}$ on $X_\infty$ (by viewing $c\in X_\infty$ as the special case $a_k=c_k$).
\end{definition}

\begin{proposition}[Completeness of the Fibonacci congruence]\label{prop:val-invariant}
If $a\equiv b$, then $\mathrm{Val}(a)=\mathrm{Val}(b)$. Consequently, $\mathrm{Val}$ induces a well-defined homomorphism on the quotient monoid $\mathcal{D}/\!\equiv$:
\[
\overline{\mathrm{Val}}:\mathcal{D}/\!\equiv\ \to\ \NN.
\]
\end{proposition}

\textbf{Proof:}
It suffices to verify that the generating relations preserve the value. For the recursive congruence $e_{k+2}\equiv e_{k+1}+e_k$,
\[
\mathrm{Val}(e_{k+2})=F_{k+3}=F_{k+2}+F_{k+1}=\mathrm{Val}(e_{k+1}+e_k),
\]
and for the baseline congruence $e_2\equiv 2e_1$,
\[
\mathrm{Val}(e_2)=F_3=2=2F_2=\mathrm{Val}(2e_1).
\]
The remaining cases follow from the closure of the congruence under addition and the additive homomorphism property of $\mathrm{Val}$.

\begin{theorem}[Emergence of the natural numbers: isomorphism of the quotient monoid]\label{thm:monoid-quotient-is-N}
The map $\overline{\mathrm{Val}}:\mathcal{D}/\!\equiv \to \NN$ is a commutative monoid isomorphism. Moreover, $X_\infty$ (the set of Zeckendorf-legal sequences) provides a unique canonical representative for each congruence class: for any $a\in\mathcal{D}$, there exists a unique $c\in X_\infty$ such that $a\equiv c$ and $\mathrm{Val}(a)=\mathrm{Val}(c)$.
\end{theorem}

\textbf{Proof:}
\begin{itemize}
  \item \textbf{Surjectivity:} For any $N\in\NN$, take its Zeckendorf digits $Z(N)=(c_1,c_2,\dots)\in X_\infty$ (\cite{Zeckendorf1972}). Then $\mathrm{Val}(Z(N))=N$, so $\overline{\mathrm{Val}}$ is surjective.
  \item \textbf{Isomorphism (using the presentation to collapse all classes to a single generator):}
  In the quotient monoid $\mathcal{D}/\!\equiv$, let $\varepsilon:=[e_1]$. By the baseline congruence, $[e_2]=2\varepsilon$. Then by induction using the recursive congruence, for all $k\ge 1$,
  \[
  [e_{k+2}]=[e_{k+1}]+[e_k],
  \qquad
  \Rightarrow\qquad
  [e_k]=F_{k+1}\,\varepsilon.
  \]
  Therefore, for any $a=\sum_{k\ge 1} a_k e_k\in\mathcal{D}$,
  \[
  [a]=\sum_{k\ge 1} a_k [e_k]=\left(\sum_{k\ge 1} a_k F_{k+1}\right)\varepsilon
  =\mathrm{Val}(a)\,\varepsilon.
  \]
This shows that every congruence class is completely characterized by its value $\mathrm{Val}(a)$, and $\overline{\mathrm{Val}}([a])=\mathrm{Val}(a)$ is precisely the monotone isomorphism sending $n\varepsilon$ to $n$. Hence $\overline{\mathrm{Val}}$ is a commutative monoid isomorphism.
  \item \textbf{Canonical representative:} For any $a\in\mathcal{D}$, let $N=\mathrm{Val}(a)$ and take $c=Z(N)\in X_\infty$. The preceding step gives
  \(
  [a]=N\varepsilon=[c],
  \)
  i.e., $a\equiv c$. Uniqueness follows from Zeckendorf uniqueness: if $c,c'\in X_\infty$ and $c\equiv c'$, then $\mathrm{Val}(c)=\mathrm{Val}(c')$, whence $c=c'$.
\end{itemize}

\begin{corollary}[Folding as a computable section of the quotient map]\label{cor:fold-as-section}
The conclusion of the preceding theorem can be summarized as:
\[
\mathcal{D}\ \twoheadrightarrow\ \mathcal{D}/\!\equiv\ \cong\ \NN
\quad\text{and}\quad
X_\infty \hookrightarrow \mathcal{D}
\]
is a computable section (selecting canonical representatives). This is precisely the concrete realization, in the Fibonacci case, of the abstract mechanism ``folding = extracting invariant/taking quotient + selecting the normal form.''
\end{corollary}

\textbf{Proof:}
This follows directly from the isomorphism and unique canonical representative conclusions of Theorem~\ref{thm:monoid-quotient-is-N}.
