\subsubsection{Pole residue and multipole moment growth: quantitative form of Ramanujan--horizon dichotomy}\label{subsubsec:xi-horizon-residue-multipole-temperedness}
Appendix \ref{subsubsec:app-horizon-weyl-carath-toeplitz} has already equivalently formulated RH as the Carath\'eodory positivity of the horizon logarithmic derivative
\(\mathscr{C}_{\zeta}\), and further equivalent to the Toeplitz--PSD hierarchy.
Meanwhile, the quadratic radial compression law of off-critical-line offset in disk radius is given by \eqref{eq:app-offcritical-radius-closed} (also see Theorem \ref{thm:xi-offcritical-quadratic-radial-compression}).
This subsection gives three \emph{quantitative} results strictly beyond the above "equivalence + position compression law": off-line zeros are not only compressed into the endpoint thin layer, but also induce a residue strength compression law of the same order on \(\mathscr{C}_{\zeta}\); and this in-disk pole forces exponential-type growth tail terms on the horizon multipole moments \(\{c_n\}\), thus giving a dichotomy criterion isomorphic to Ramanujan temperedness in the moment spectral radius sense.

\paragraph{Horizon logarithmic derivative and multipole moments}
We continue to use the definition from \eqref{eq:app-horizon-carath-def}:
\[
\mathscr{C}_{\zeta}(w):=-\frac{\Xi_{\zeta}'(s(w))}{\Xi_{\zeta}(s(w))},\qquad
s(w)=\tfrac12+\mathrm{i}\,\mathcal{C}^{-1}(w)\qquad(w\in\mathbb{D}),
\]
and define the horizon multipole moments (also "visible hair coefficients") by the normalized power series
\begin{equation}\label{eq:xi-horizon-carath-series-recall}
\mathscr{C}_{\zeta}(w)=c_0+2\sum_{n\ge 1}c_n w^n
\end{equation}

\begin{theorem}[Pole residue formula and "depth--strength" compression law]\label{thm:xi-horizon-pole-residue-depth-strength}
Suppose \(\Xi_{\zeta}\) has a zero \(\rho=\beta+\mathrm{i}\gamma\) satisfying \(0<\beta<\tfrac12\), and let
\[
\delta:=\tfrac12-\beta>0,\qquad
t_{\rho}:=\gamma+\mathrm{i}\delta\in\mathbb{H},\qquad
w_{\rho}:=\mathcal{C}(t_{\rho})\in\mathbb{D}.
\]
Denote the multiplicity of this zero by \(m_{\rho}\ge 1\).
Then \(\mathscr{C}_{\zeta}\) has a simple pole at \(w=w_{\rho}\), and its residue satisfies the complete closed form
\begin{equation}\label{eq:xi-horizon-pole-residue-closed}
\boxed{\;
\operatorname*{Res}_{w=w_{\rho}}\mathscr{C}_{\zeta}(w)
=-\;m_{\rho}\,\frac{(1+w_{\rho})^{2}}{2}
=-\;\frac{2m_{\rho}}{(1-\mathrm{i}t_{\rho})^{2}}\,.
\;}
\end{equation}
Moreover, let the endpoint depth (horizon depth)
\(
h(\rho):=1-|w_{\rho}|^{2}
\)
(see \eqref{eq:app-offcritical-radius-closed}), then the residue modulus satisfies the "depth--strength" compression law
\begin{equation}\label{eq:xi-horizon-depth-strength-law}
\boxed{\;
\Bigl|\operatorname*{Res}_{w=w_{\rho}}\mathscr{C}_{\zeta}(w)\Bigr|
=\frac{2m_{\rho}}{\gamma^{2}+(1+\delta)^{2}}
=\frac{m_{\rho}}{2\delta}\,h(\rho)\; }.
\end{equation}
\end{theorem}

\begin{proof}
Let \(F_{\zeta}(w):=\Xi_{\zeta}(s(w))\) (see \eqref{eq:app-Fzeta-disk}), then \(F_{\zeta}\) has a zero of order \(m_{\rho}\) at \(w_{\rho}\).
Therefore the logarithmic derivative \(\frac{\dd}{\dd w}\log F_{\zeta}(w)\) has a simple pole of residue \(m_{\rho}\) at \(w_{\rho}\).
Also by the chain rule
\[
\frac{\dd}{\dd w}\log F_{\zeta}(w)
=\frac{\Xi_{\zeta}'(s(w))}{\Xi_{\zeta}(s(w))}\,s'(w),
\qquad
\mathscr{C}_{\zeta}(w)=-\frac{1}{s'(w)}\frac{\dd}{\dd w}\log F_{\zeta}(w),
\]
thus
\[
\operatorname*{Res}_{w=w_{\rho}}\mathscr{C}_{\zeta}(w)
=-\frac{m_{\rho}}{s'(w_{\rho})}.
\]
By \eqref{eq:app-s-of-w} we have \(s'(w)=\frac{2}{(1+w)^2}\), so we get the first equality.
On the other hand, by \eqref{eq:app-cayley-disk} we have \(1+w=\frac{2}{1-\mathrm{i}t}\), substituting gives the second equality.
\par
For the modulus, note that
\(
|1-\mathrm{i}t_{\rho}|^2=\gamma^2+(1+\delta)^2
\)
thus we get \(\bigl|\operatorname*{Res}\bigr|=\frac{2m_{\rho}}{\gamma^2+(1+\delta)^2}\).
Again by the identity from Corollary \ref{cor:app-offcritical-horocycle-busemann}
\(
\delta=\frac{1-|w_{\rho}|^2}{|1+w_{\rho}|^2}
\)
we get \(|1+w_{\rho}|^2=h(\rho)/\delta\), combining with
\(
\bigl|\operatorname*{Res}\bigr|=\frac{m_{\rho}}{2}|1+w_{\rho}|^2
\)
gives \eqref{eq:xi-horizon-depth-strength-law}. This completes the proof.
\end{proof}

\begin{corollary}[Redshift suppression: residue strength decays with depth on the same order at high altitude]\label{cor:xi-horizon-residue-redshift}
Under the setting of Theorem \ref{thm:xi-horizon-pole-residue-depth-strength}, if \(|\gamma|\to\infty\) and \(\delta\) is bounded, then
\[
h(\rho)\sim \frac{4\delta}{\gamma^2},
\qquad
\Bigl|\operatorname*{Res}_{w=w_{\rho}}\mathscr{C}_{\zeta}(w)\Bigr|\sim \frac{2m_{\rho}}{\gamma^2}.
\]
Therefore off-line zeros are not only compressed into the endpoint thin layer (position redshift), their coupling strength to the horizon observable function \(\mathscr{C}_{\zeta}\) is also suppressed on the same \(\gamma^{-2}\) scale (strength redshift).
\end{corollary}

\begin{theorem}[Ramanujan temperedness dichotomy of horizon multipole moments]\label{thm:xi-horizon-multipole-temperedness-dichotomy}
Let \(\mathscr{C}_{\zeta}(w)=c_0+2\sum_{n\ge 1}c_n w^n\) as in \eqref{eq:xi-horizon-carath-series-recall}.
Define the moment spectral radius (multipole moment spectral radius)
\[
\mathfrak{r}_{\mathrm{mom}}
:=\limsup_{n\to\infty}|c_n|^{1/n}\in[0,\infty],
\qquad
\Lambda_{\mathrm{mom}}:=\log \mathfrak{r}_{\mathrm{mom}}\in[-\infty,\infty].
\]
Then the following dichotomy holds:
\begin{enumerate}
  \item \textbf{(tempered)} If RH holds, then \(\mathscr{C}_{\zeta}\) is a Carath\'eodory function (Corollary \ref{cor:app-horizon-carath-rh}).
  Hence there exists a finite positive measure \(\mu\) such that \(c_n=\int_{\TT}\xi^{-n}\,\dd\mu(\xi)\) (see \eqref{eq:app-horizon-herglotz-disk}--\eqref{eq:app-horizon-carath-series}), thus
  \begin{equation}\label{eq:xi-horizon-cn-bounded}
  \boxed{\;
  |c_n|\le c_0=\mu(\TT)\qquad(\forall n\ge 0)\;}
  \end{equation}
  and \(\Lambda_{\mathrm{mom}}\le 0\).
  \item \textbf{(non-tempered)} If RH does not hold, then \(\mathscr{C}_{\zeta}\) has at least one pole in \(\mathbb{D}\).
  Let
  \begin{equation}\label{eq:xi-horizon-rmin-pole}
  r_{\min}:=\min\Bigl\{|a|:\ a\in\mathbb{D},\ \mathscr{C}_{\zeta}\ \text{has a pole at}\ a\Bigr\}\in(0,1).
  \end{equation}
  Then the convergence radius of the power series of \(\mathscr{C}_{\zeta}\) at \(w=0\) equals \(r_{\min}\), and
  \begin{equation}\label{eq:xi-horizon-moment-spectral-radius}
  \boxed{\;
  \mathfrak{r}_{\mathrm{mom}}=\frac{1}{r_{\min}}>1,
  \qquad
  \Lambda_{\mathrm{mom}}=-\log r_{\min}>0\; }.
  \end{equation}
  \item \textbf{(Closed-form exponent: given by deepest in-disk pole)} If \(r_{\min}\) is realized by the pole \(w_{\rho}\) induced by some off-line zero \(\rho\), then
  \begin{equation}\label{eq:xi-horizon-Lambda-closed}
  \boxed{\;
  \Lambda_{\mathrm{mom}}
  =-\log|w_{\rho}|
  =\frac12\log\!\left(\frac{\gamma^{2}+(1+\delta)^{2}}{\gamma^{2}+(1-\delta)^{2}}\right)\; }.
  \end{equation}
  and in the high-altitude regime \(|\gamma|\to\infty\) with \(\delta\) bounded,
  \begin{equation}\label{eq:xi-horizon-Lambda-asymptotic}
  \boxed{\;
  \Lambda_{\mathrm{mom}}\sim \frac{2\delta}{\gamma^{2}}\sim \frac{h(\rho)}{2}\; }.
  \end{equation}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) By the Carath\'eodory/Herglotz representation, \(c_n\) are Fourier moments of a finite positive measure \(\mu\), thus
\(
|c_n|\le \int_{\TT}|\xi^{-n}|\,\dd\mu=\mu(\TT)=c_0
\),
yielding \eqref{eq:xi-horizon-cn-bounded}.
Since \(\{c_n\}\) is bounded, we have \(\limsup |c_n|^{1/n}\le 1\), thus \(\Lambda_{\mathrm{mom}}\le 0\).
\par
(2) When RH does not hold, \(\mathscr{C}_{\zeta}\) has poles in \(\mathbb{D}\) (see the end of the proof of Corollary \ref{cor:app-horizon-carath-rh} and Proposition \ref{prop:app-horizon-pole-imprint}).
By general theory of analytic functions, the convergence radius of the power series at \(w=0\) equals the distance to the nearest singularity; hence the convergence radius equals \(r_{\min}\).
Again by the Cauchy--Hadamard theorem, \(\frac{1}{r_{\min}}=\limsup |2c_n|^{1/n}=\limsup |c_n|^{1/n}\), yielding \eqref{eq:xi-horizon-moment-spectral-radius}.
\par
(3) By \eqref{eq:app-offcritical-radius-closed} we have
\(
|w_{\rho}|^2=\frac{\gamma^2+(1-\delta)^2}{\gamma^2+(1+\delta)^2}
\),
so \(-\log|w_{\rho}|\) has the closed form \eqref{eq:xi-horizon-Lambda-closed}.
Also by \(h(\rho)=1-|w_{\rho}|^2\to 0\) (\(|\gamma|\to\infty\), \(\delta\) bounded) and the identity
\(
-\log|w_{\rho}|=-\tfrac12\log(1-h(\rho))
=\tfrac12 h(\rho)+O(h(\rho)^2)
\),
and using \(h(\rho)\sim 4\delta/\gamma^2\) (from \eqref{eq:app-offcritical-radius-closed}) we obtain \eqref{eq:xi-horizon-Lambda-asymptotic}. This completes the proof.
\end{proof}

\begin{corollary}[Sub-exponential growth certificate: Ramanujan reduction of multipole moments]\label{cor:xi-horizon-subexp-growth-implies-rh}
If
\[
\limsup_{n\to\infty}\frac{1}{n}\log|c_n|\le 0,
\]
then RH holds.
Equivalently, any bound \( |c_n|\le e^{o(n)} \) (sub-exponential growth/no positive exponential growth) implies RH.
\end{corollary}

\begin{proof}
If RH does not hold, then by Theorem \ref{thm:xi-horizon-multipole-temperedness-dichotomy} part (2) we have
\(
\limsup_{n\to\infty}\frac{1}{n}\log|c_n|
=\Lambda_{\mathrm{mom}}>0
\),
contradicting the assumption; thus RH holds. This completes the proof.
\end{proof}

\begin{theorem}[Forced geometric tail of a single off-line pole on multipole moments]\label{thm:xi-horizon-forced-geometric-tail}
Under the setting of Theorem \ref{thm:xi-horizon-pole-residue-depth-strength}, denote
\(
R_{\rho}:=\operatorname*{Res}_{w=w_{\rho}}\mathscr{C}_{\zeta}(w)
\neq 0
\).
Then there exists a function \(H_{\rho}\) analytic in \(\mathbb{D}\) such that
\begin{equation}\label{eq:xi-horizon-pole-decomposition}
\boxed{\;
\mathscr{C}_{\zeta}(w)=\frac{R_{\rho}}{w-w_{\rho}}+H_{\rho}(w)\qquad(w\in\mathbb{D}).\;}
\end{equation}
Expand \(H_{\rho}\) at \(w=0\) as \(H_{\rho}(w)=h_0+2\sum_{n\ge 1}h_n w^n\), then for all \(n\ge 1\) the coefficient identity holds
\begin{equation}\label{eq:xi-horizon-cn-forced-tail}
\boxed{\;
c_n=-\frac{R_{\rho}}{2\,w_{\rho}^{n+1}}+h_n\qquad(n\ge 1).\;}
\end{equation}
In particular,
\[
\limsup_{n\to\infty}|c_n|^{1/n}\ \ge\ \frac{1}{|w_{\rho}|}.
\]
\end{theorem}

\begin{proof}
Since \(\mathscr{C}_{\zeta}\) is analytic on \(\mathbb{D}\setminus\{w_{\rho}\}\) and has only one principal part at \(w_{\rho}\), we can define
\(
H_{\rho}(w):=\mathscr{C}_{\zeta}(w)-\frac{R_{\rho}}{w-w_{\rho}}
\),
then \(H_{\rho}\) is analytic in \(\mathbb{D}\), yielding \eqref{eq:xi-horizon-pole-decomposition}.
For \(\frac{R_{\rho}}{w-w_{\rho}}=-\frac{R_{\rho}}{w_{\rho}}\cdot\frac{1}{1-w/w_{\rho}}
=-\sum_{n\ge 0}R_{\rho}\,w_{\rho}^{-(n+1)}w^n\), comparing the geometric series expansion with \eqref{eq:xi-horizon-carath-series-recall} gives \eqref{eq:xi-horizon-cn-forced-tail}; the last statement follows directly from Cauchy--Hadamard. This completes the proof.
\end{proof}

\begin{corollary}[Strength redshift \(\times\) multipole order amplification: an explicit threshold law]\label{cor:xi-horizon-redshift-growth-threshold}
Under the setting of Theorem \ref{thm:xi-horizon-forced-geometric-tail}, denote
\(\Lambda(\rho):=-\log|w_{\rho}|>0\).
Then the amplitude of the forced geometric tail term is
\begin{equation}\label{eq:xi-horizon-forced-term-amplitude}
\boxed{\;
\frac{|R_{\rho}|}{2}\,|w_{\rho}|^{-(n+1)}
=\frac{m_{\rho}}{\gamma^{2}+(1+\delta)^{2}}\,
\exp\!\bigl((n+1)\Lambda(\rho)\bigr)\; }.
\end{equation}
In the high-altitude regime \(|\gamma|\to\infty\) with \(\delta\) bounded, we have
\[
\frac{|R_{\rho}|}{2}\,|w_{\rho}|^{-(n+1)}
\asymp
\gamma^{-2}\exp\!\bigl(n\,\Lambda(\rho)\bigr),
\qquad
\Lambda(\rho)\asymp \delta/\gamma^{2},
\]
thus off-line zeros simultaneously manifest in the horizon multipole moments as: redshift suppression \(\asymp\gamma^{-2}\) at the strength level, and exponential threshold amplification in the multipole order \(n\) direction (whose exponent is also compressed by the endpoint thin layer).
\end{corollary}

\begin{remark}[Horizon semantics of non-temperedness parameter]\label{rem:xi-horizon-nontempered-parameter-semantics}
Theorem \ref{thm:xi-horizon-multipole-temperedness-dichotomy} compresses the RH falsification mechanism to a computable "moment spectral radius" parameter:
\(\mathfrak{r}_{\mathrm{mom}}>1\) is equivalent to the entry of in-disk poles, and equivalent to unavoidable exponential tails of multipole moments;
in contrast, under RH \(\mathfrak{r}_{\mathrm{mom}}\le 1\) and multipole moments are uniformly bounded, thus corresponding to "tempered/unitary horizon spectral objects" in the semantics of this paper.
In this sense, \(\Lambda_{\mathrm{mom}}\) can be viewed as a canonical non-temperedness exponent, whose closed-form compression \(\Lambda_{\mathrm{mom}}\sim h(\rho)/2\) (high altitude) directly aligns "non-temperedness" with endpoint thin layer geometry.
\end{remark}
