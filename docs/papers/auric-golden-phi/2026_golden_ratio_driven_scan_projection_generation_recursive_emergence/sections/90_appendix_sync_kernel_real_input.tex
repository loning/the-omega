\subsubsection{真实算术输入约束下的同步核（40 状态）}
令 $X\subset\{0,1\}^{\ZZ}$ 为 Zeckendorf（黄金均值）移位：禁止相邻子词 $11$。它是 $1$-step SFT，邻接矩阵
\[
A=\begin{pmatrix}1&1\\[2pt]1&0\end{pmatrix},\qquad \rho(A)=\varphi,
\]

在线延迟核的输入字母表虽为 $\{0,1,2\}$，但真实算术输入来自两条合法 Zeckendorf 轨道
$x,y\in X$ 的逐位和 $d_k=x_k+y_k$，其中 $x_kx_{k+1}=0,\ y_ky_{k+1}=0$。
因此把同步核状态 $s\in Q$ 与输入记忆 $(p_x,p_y)\in\{0,1\}^2$ 合并，得到
\[
\Sigma:=Q\times\{0,1\}^2,\qquad |\Sigma|=40.
\]
从 $(s,p_x,p_y)$ 出发，允许选择 $(x,y)\in\{0,1\}^2$ 且满足 $p_xx=0,\ p_yy=0$，
令 $d=x+y$，并由同步核确定性转移更新 $s$，下一个记忆为 $(x,y)$。
由此得到 40 状态 SFT，其邻接矩阵记为 $M$。

\begin{proposition}[40 状态核的本质核心：唯一 essential SCC]\label{prop:real-input-40-essential-20}

\end{itemize}
并且该 $20$ 状态核心可按记忆 $m=(x_{k-1},y_{k-1})$ 分组为
\[
\begin{array}{ll}
m=00:& s\in\{000,001,010,100,01\overline{1},11\overline{1}\}\quad(6\ \text{个}),\\
m=01:& s\in\{000,001,002,100,101\}\quad(5\ \text{个}),\\

\end{array}

\end{proposition}

\begin{proposition}[把 40 状态压缩为 20 状态而不改变 \texorpdfstring{$\zeta$}{zeta}/周期轨道对象]\label{prop:real-input-40-essential-reduction}
设 $M$ 为 40 状态邻接矩阵，$M_{\mathrm{ess}}$ 为其唯一 essential 强连通分量诱导的 $20\times 20$ 邻接矩阵（按命题 \ref{prop:real-input-40-essential-20} 的分量选取）。则存在一个置换矩阵 $P$ 与某个 $20\times 20$ 矩阵 $N$ 使得
\[
PMP^{-1}=
\begin{pmatrix}
M_{\mathrm{ess}} & *\\
0 & N
\end{pmatrix},
\qquad
N\ \text{幂零（nilpotent）}.
\]
因此对任意 $n\ge 1$，
\[
\Tr(M^n)=\Tr(M_{\mathrm{ess}}^n),
\qquad
\det(I-zM)=\det(I-zM_{\mathrm{ess}}),
\qquad
\zeta_M(z)=\zeta_{M_{\mathrm{ess}}}(z),
\]
从而所有依赖周期点/primitive 轨道的对象（例如 $a_n=\Tr(M^n)$、primitive 轨道数 $p_n$、Abel/Mertens 常数与其扭曲版本）都可在 $M_{\mathrm{ess}}$ 上无损计算。
\end{proposition}

\begin{proof}
令 $G$ 为由 $M$ 定义的有向图。把顶点按“essential 分量在前、其余顶点在后”的顺序重排，即得某个置换矩阵 $P$ 使得 $PMP^{-1}$ 为块上三角矩阵，左上块即为 $M_{\mathrm{ess}}$，右下块记为 $N$；左下块为 $0$ 的原因是：从瞬态顶点不可能回到 essential 分量（否则瞬态顶点也会落入同一强连通分量）。

接着证明 $N$ 幂零。由定义，瞬态顶点不属于任何有向环；特别地，瞬态子图中不存在有向环。因此瞬态子图是一个 DAG，可按拓扑序排序使得其邻接矩阵严格上三角，从而 $N$ 幂零（等价地：存在 $L$ 使得 $N^L=0$）。

由块上三角结构，$M^n$ 同样为块上三角，其迹满足
\(
\Tr(M^n)=\Tr(M_{\mathrm{ess}}^n)+\Tr(N^n)
\)；而 $N$ 幂零给出对所有 $n\ge 1$ 有 $\Tr(N^n)=0$，故 $\Tr(M^n)=\Tr(M_{\mathrm{ess}}^n)$。
同理，对任意 $z$，有
\[
\det(I-zM)=\det(I-zM_{\mathrm{ess}})\det(I-zN).
\]
但 $N$ 幂零 $\Rightarrow$ $I-zN$ 为单位上三角矩阵（在某基下）且其全部特征值为 $1$，故 $\det(I-zN)=1$，于是 $\det(I-zM)=\det(I-zM_{\mathrm{ess}})$。
最后由有限图的 zeta--行列式口径 $\zeta(z)=1/\det(I-zM)$ 得 $\zeta_M=\zeta_{M_{\mathrm{ess}}}$。证毕。
\end{proof}

\subsubsection{Abel--Mertens 接口：primitive 谱、主极点与有限部（通用闭式）}\label{app:abel-mertens-universal}
令 $A$ 为有限有向图（或 SFT）的邻接矩阵，并记
\[
a_n:=\Tr(A^n),\qquad
\zeta_A(z):=\exp\!\left(\sum_{n\ge 1}\frac{a_n}{n}z^n\right)=\frac{1}{\det(I-zA)}.
\]
令 $p_n$ 为长度为 $n$ 的 primitive 闭轨道数（按轨道计），并记 primitive 生成函数
\[
P(z):=\sum_{n\ge 1}p_n z^n.
\]

\begin{theorem}[primitive 对数接口与 Abel 有限部常数]\label{thm:abel-mertens-universal}
有恒等式
\[
P(z)=\sum_{k\ge 1}\frac{\mu(k)}{k}\log\zeta_A(z^k)
\qquad(|z|\ \text{充分小}).
\]
进一步假设 $\zeta_A(z)$ 在 $z_\star$ 处有唯一主极点且为简单极点，并记
\[
\lambda:=\rho(A)=\frac{1}{z_\star},
\qquad
C:=\lim_{z\to z_\star}(1-\lambda z)\zeta_A(z).
\]
则存在 Abel/Hadamard 有限部常数 $\log\mathfrak{M}$ 使得
\[
P(z)=-\log(1-\lambda z)+\log\mathfrak{M}+o(1)\qquad(z\to z_\star),
\]
并且常数项满足闭式分解
\[
\log\mathfrak{M}
=\log C+\sum_{k\ge 2}\frac{\mu(k)}{k}\log\zeta_A(z_\star^{\,k}).
\]
若写 $\Delta_A(z):=\det(I-zA)$，则当 $z_\star$ 为 $\Delta_A$ 的简单零点时有
\[
C=\frac{1}{-z_\star\,\Delta_A'(z_\star)}=-\frac{\lambda}{\Delta_A'(z_\star)}.
\]
\end{theorem}

\begin{proof}
由 $\zeta_A(z)=\prod_{\tau\ \mathrm{primitive}}(1-z^{|\tau|})^{-1}$（闭轨道欧拉乘积）取对数得
\[
\log\zeta_A(z)=\sum_{\tau\ \mathrm{primitive}}\sum_{m\ge 1}\frac{z^{m|\tau|}}{m}
=\sum_{m\ge 1}\frac{P(z^m)}{m}.
\]
对上式作 Möbius 反演即得
\(
P(z)=\sum_{k\ge 1}\frac{\mu(k)}{k}\log\zeta_A(z^k)
\)。

在 $z\to z_\star$ 时，$\zeta_A(z)=\frac{C}{1-\lambda z}+O(1)$，故
\[
\log\zeta_A(z)=-\log(1-\lambda z)+\log C+o(1).
\]
将该展开代入 primitive 对数接口，并将 $k=1$ 项单独提出：当 $k\ge 2$ 时 $z^k\to z_\star^k$ 且 $|z_\star^k|<|z_\star|$，
因此 $\log\zeta_A(z^k)$ 在 $z=z_\star$ 处解析并收敛到 $\log\zeta_A(z_\star^k)$。于是得到
\[
P(z)=-\log(1-\lambda z)+\log C+\sum_{k\ge 2}\frac{\mu(k)}{k}\log\zeta_A(z_\star^{\,k})+o(1),
\]
从而有限部常数为所示的闭式分解。
最后，当 $\Delta_A(z)=\det(I-zA)$ 在 $z_\star$ 为简单零点时，
\(
\zeta_A(z)=1/\Delta_A(z)
\)
线性化即可得
\(
C=\lim_{z\to z_\star}(1-\lambda z)/\Delta_A(z)=1/(-z_\star\Delta_A'(z_\star))
\)。
\end{proof}

\begin{corollary}[Mertens 常数与 Abel 有限部常数]\label{cor:abel-mertens-constant-universal}
令
\[
\mathcal{M}(N):=\sum_{n\le N}\frac{p_n}{\lambda^n}.
\]
则存在常数 $\mathsf{M}$ 使得
\[
\mathcal{M}(N)=\log N+\mathsf{M}+o(1),
\qquad
\boxed{\ \mathsf{M}=\gamma+\log\mathfrak{M}\ }.
\]
\end{corollary}

\begin{proof}
由主极点谱隙（次大特征值模 $<\lambda$）可得 $p_n=\lambda^n/n+O(\lambda_2^n/n)$，从而
\(
b_n:=p_n/\lambda^n-1/n
\)
绝对可和。于是
\[
\sum_{n\le N}\frac{p_n}{\lambda^n}
=\sum_{n\le N}\frac{1}{n}+\sum_{n\le N}b_n
=\log N+\gamma+\sum_{n\ge 1}b_n+o(1).
\]
另一方面，
\[
P(z)+\log(1-\lambda z)=\sum_{n\ge 1}\left(\frac{p_n}{\lambda^n}-\frac{1}{n}\right)(\lambda z)^n,
\]
故在 $z\to z_\star$（即 $\lambda z\to 1$）时，其 Abel 有限部常数正是 $\sum_{n\ge 1}b_n=\log\mathfrak{M}$。
\end{proof}

\subsubsection{有限 RH：\texorpdfstring{$s$}{s}-变量 $\zeta$ 的临界带极点与 primitive 的平方根振荡}\label{app:real-input-40-finite-rh}

\begin{proposition}[临界带中的非平凡极点都在 \texorpdfstring{$\Re(s)=\tfrac12$}{Re(s)=1/2}]\label{prop:finite-rh-40}

\Re(s)=\frac12.
\]
更具体地，唯一落在 $0<\Re(s)<1$ 的极点列来自特征值 $-\varphi=-\sqrt{\lambda}$，其位置为
\[
s=\frac12+\frac{(2k+1)\pi i}{\log\lambda},\qquad k\in\ZZ.
\]
\end{proposition}

\begin{proof}
由上文的谱分解，$M$ 的非零特征值集合为
\[
\{\lambda,\ -\sqrt{\lambda},\ 1^{(2)},\ (-1)^{(3)},\ \lambda^{-1},\ \lambda^{-1/2}\},
\]

\(
s=\tfrac12+\frac{(2k+1)\pi i}{\log\lambda}

\]
\[
p_n=\frac{\lambda^n}{n}+O\!\left(\frac{\lambda^{n/3}}{n}\right)\qquad(n\ \text{为偶数}),
\]
其中偶数情形的 $\lambda^{n/2}$ 级项被 Möbius 反演中的 $d=1$ 与 $d=2$ 两项\textbf{精确抵消}。
\[
\boxed{
\;p_n=\frac{\lambda^n}{n}+\frac{\bigl((-1)^n-\mathbf{1}_{2\mid n}\bigr)\lambda^{n/2}}{n}
+O\!\left(\frac{\lambda^{n/3}}{n}\right)\; }
\]
并且具有可核验的“奇偶共振 signature”：
\[
\lim_{\substack{n\to\infty\\ n\ \mathrm{odd}}}\ \frac{n}{\lambda^{n/2}}\left(p_n-\frac{\lambda^n}{n}\right)=-1,
\qquad
\lim_{\substack{n\to\infty\\ n\ \mathrm{even}}}\ \frac{n}{\lambda^{n/2}}\left(p_n-\frac{\lambda^n}{n}\right)=0.
\]
\end{corollary}

\begin{proof}
由上文的闭式
\(
a_n=\varphi^{2n}+(-\varphi)^n+2+3(-1)^n+\varphi^{-2n}+\varphi^{-n}
\)
可得
\(
a_n=\lambda^n+(-1)^n\lambda^{n/2}+O(1)
\)。

由 Möbius 反演 $p_n=\frac1n\sum_{d\mid n}\mu(d)a_{n/d}$：
若 $n$ 为奇数，则所有 $d\ge 2$ 满足 $d\ge 3$，故 $a_{n/d}=O(\lambda^{n/3})$。于是
\[
np_n=a_n+\sum_{\substack{d\mid n\\ d\ge 3}}\mu(d)a_{n/d}
=\lambda^n-\lambda^{n/2}+O(\lambda^{n/3}),
\]
得到第一式。

若 $n$ 为偶数，则写出 $d=1,2$ 两项：
\[
np_n=a_n-\;a_{n/2}+\sum_{\substack{d\mid n\\ d\ge 3}}\mu(d)a_{n/d}.
\]
此时 $a_n=\lambda^n+\lambda^{n/2}+O(1)$（因为 $(-1)^n=1$），而 $a_{n/2}=\lambda^{n/2}+O(\lambda^{n/4})$，故
\[
a_n-a_{n/2}=\lambda^n+O(\lambda^{n/4}),
\]
并且余项由 $d\ge 3$ 给出 $O(\lambda^{n/3})$，从而得到第二式。这里 $\lambda^{n/2}$ 的抵消正是 $a_n$ 的平方根项（$d=1$）与 $a_{n/2}$ 的主项（$d=2$）之间的精确抵消。
\end{proof}

\begin{theorem}[primitive--Dirichlet 级数的自然边界：\texorpdfstring{$\Re(s)=0$}{Re(s)=0}]\label{thm:real-input-40-primedirichlet-dense-branch}
令
\[
\mathcal{P}(s):=P(\lambda^{-s})=\sum_{n\ge 1}p_n\,\lambda^{-ns}\qquad(\Re(s)>1),
\]
其中 $P(z)=\sum_{n\ge 1}p_n z^n$ 为 primitive 生成函数，$\lambda=\rho(M)=\varphi^2$，并定义
\[
\widehat{\zeta}(s):=\zeta_M(\lambda^{-s}).
\]
则：
\begin{enumerate}
  \item \textbf{（解析延拓到右半平面）} 对任意 $\sigma_0>0$，级数
  \[
  \sum_{k\ge 1}\frac{\mu(k)}{k}\log \widehat{\zeta}(ks)
  \]
  在集合
  \(
  \{s:\Re(s)\ge \sigma_0\}
  \)
  的任意紧子集上（避开下述极点列）一致收敛，从而定义出 $\mathcal{P}(s)$ 在开半平面 $\Re(s)>0$ 上的一个解析延拓（其在 $\Re(s)>1$ 与原 Dirichlet 级数一致）。
  \item \textbf{（稠密奇点列与自然边界）} 对每个奇素数 $p$ 与任意 $m\in\ZZ$，点
  \[
  s_{p,m}:=\frac{1}{p}+\frac{2\pi i m}{p\log\lambda}
  \]
  是该解析延拓的一个对数型分支奇点。并且集合 $\{s_{p,m}\}_{p\ \mathrm{odd\ prime},\,m\in\ZZ}$ 在直线 $\Re(s)=0$ 上稠密。
\end{enumerate}
因此 \(\Re(s)=0\) 是 \(\mathcal{P}(s)\) 的\textbf{自然边界}：\(\mathcal{P}(s)\) 不可能在任何虚轴点的邻域内作解析延拓穿过该直线。
\end{theorem}

\begin{proof}
\emph{第 1 点（收敛与解析性）}：
当 $|z|$ 足够小时有 $\log\zeta_M(z)=O(z)$（因为 $\zeta_M(0)=1$ 且 $\zeta_M$ 在 $0$ 的邻域解析），因此对任意 $\sigma_0>0$，当 $\Re(s)\ge \sigma_0$ 时
\(
|\log\widehat{\zeta}(ks)|=|\log\zeta_M(\lambda^{-ks})|=O(\lambda^{-k\sigma_0})
\)
随 $k$ 指数衰减。于是 $\sum_{k\ge 1}|\mu(k)|\,|\log\widehat{\zeta}(ks)|/k$ 在任意避开极点列的紧集上一致收敛；逐项可导与一致收敛给出其和函数在 $\Re(s)>0$（去掉极点列）上解析，从而得到 $\mathcal{P}(s)$ 的解析延拓。

\emph{第 2 点（显式奇点与稠密性）}：
由本核的行列式分解
\[
\det(I-zM)=(1-z)^2(1+z)^3(1-3z+z^2)(1+z-z^2),
\]
可见 $z_\star=\lambda^{-1}$ 是 $\zeta_M(z)$ 的一个\textbf{简单极点}（对应因子 $1-3z+z^2$ 的简单零点），故 $\log\zeta_M(z)$ 在 $z=z_\star$ 具有对数型分支奇性。
当 $s=s_{p,m}$ 时有
\[
\lambda^{-ps}=\lambda^{-1}e^{-2\pi i m}=z_\star,
\]
因此 $\log\widehat{\zeta}(ps)=\log\zeta_M(\lambda^{-ps})$ 在 $s=s_{p,m}$ 处为对数型分支奇点，从而项
\(
\frac{\mu(p)}{p}\log\widehat{\zeta}(ps)
\)
在该点奇异。对任意 $k\neq p$，若 $\log\widehat{\zeta}(ks)$ 在 $s=s_{p,m}$ 奇异，则必有 $\lambda^{-ks_{p,m}}=z_\star$，等价于 $ks_{p,m}=1+\frac{2\pi i \ell}{\log\lambda}$（某个 $\ell\in\ZZ$）。比较实部得 $k/p=1$，故 $k=p$。因此除 $k=p$ 一项外，其余各项在 $s_{p,m}$ 的某邻域解析，且由第 1 点一致收敛可知其和亦解析，于是 $\mathcal{P}(s)$ 在 $s_{p,m}$ 必为对数型分支奇点。

最后证明稠密性：给定任意 $t\in\RR$ 与 $\varepsilon>0$，取足够大的奇素数 $p$，令
\(
m:=\left\lfloor \frac{t\,p\log\lambda}{2\pi}\right\rceil
\)
为最近整数，则
\[
\left|\Im(s_{p,m})-t\right|
=\left|\frac{2\pi m}{p\log\lambda}-t\right|
\le \frac{\pi}{p\log\lambda}<\varepsilon
\]
当 $p$ 足够大时成立，同时 $\Re(s_{p,m})=1/p<\varepsilon$。故 $s_{p,m}\to it$，从而 $\{s_{p,m}\}$ 在虚轴上稠密。由于任意穿过虚轴的邻域都包含这些奇点，故无法跨越解析延拓，\(\Re(s)=0\) 为自然边界。
\end{proof}

\paragraph{primitive 轨道计数曲线与 \texorpdfstring{$x/\log x$}{x/log x} 的常数因子}
令
\[
\Pi(N):=\sum_{n\le N}p_n
\]
为按长度累计的 primitive 轨道数（按轨道计）。由 $p_n\sim \lambda^n/n$ 的“末项主导 + 几何尾”可得
\[
\Pi(N)\sim \frac{\lambda}{\lambda-1}\cdot \frac{\lambda^N}{N}.
\]
在本核上 $\lambda=\varphi^2$ 且 $\lambda-1=\varphi$，故
\[
\Pi(N)\sim \varphi\cdot\frac{\varphi^{2N}}{N}=\frac{\varphi^{2N+1}}{N}.
\]
若定义“轨道范数” $N(\tau):=\lambda^{|\tau|}$ 并令
\[
\pi(x):=\#\{\tau\ \mathrm{primitive}:\ N(\tau)\le x\},
\]
则 $\pi(\lambda^N)=\Pi(N)$，从而
\[
\pi(x)\sim \frac{\lambda\log\lambda}{\lambda-1}\cdot \frac{x}{\log x}.
\]
对 $\lambda=\varphi^2$，该常数因子化简为
\[
\frac{\lambda\log\lambda}{\lambda-1}
=\frac{\varphi^2\log(\varphi^2)}{\varphi}
=2\varphi\log\varphi.
\]

定义轨道 Mertens 和
\[
\mathcal{M}(N):=\sum_{n\le N}\frac{p_n}{\lambda^n}
=\log N+\mathsf{M}_{\varphi^2}+o(1),
\]
其中常数项可写为
\[
\mathsf{M}_{\varphi^2}
=\gamma+\sum_{n\ge 1}\left(\frac{p_n}{\lambda^n}-\frac{1}{n}\right),
\]
$\gamma$ 为 Euler--Mascheroni 常数。数值上
\[
\sum_{n\ge 1}\left(\frac{p_n}{\lambda^n}-\frac{1}{n}\right)\approx -0.254312661901716,
\qquad
\mathsf{M}_{\varphi^2}\approx 0.322903002999817.
\]
另外记 $P(z)=\sum_{n\ge 1}p_n z^n$，则在 $z\to z_\star=\lambda^{-1}$ 时
\[
P(z)=-\log(1-\lambda z)+\log\mathfrak{M}+o(1),
\]
其中
\[
\log\mathfrak{M}\approx -0.254312661901716,\qquad
\mathfrak{M}\approx 0.7754493106833835.
\]
由因式分解可得主极点 $z_\star=\lambda^{-1}=\varphi^{-2}$，并且
\[
\zeta_M(z)=\frac{C}{1-\lambda z}+O(1),\qquad
C=\lim_{z\to z_\star}(1-\lambda z)\,\zeta_M(z)=\frac{47+21\sqrt5}{100}\ (\approx 0.9395742753).
\]
进一步由 Witt 分解/ Möbius--log 反演（与 \S\ref{subsec:zeta-online-kernel} 相同口径）可得一个直接由 $\zeta_M$ 主极点读出的 Abel/有限部闭式：
\[
\log\mathfrak{M}
=\log C+\sum_{m\ge 2}\frac{\mu(m)}{m}\log\zeta_M(z_\star^{\,m}),
\qquad z_\star=\lambda^{-1}.
\]
右端把有理函数 $\zeta_M$ 代入代数数 $z_\star^{\,m}$，由于 $z_\star<1$，该级数以指数速度收敛。

\paragraph{显式 Möbius--$\zeta$ 级数与对数乘积（仅用 \texorpdfstring{$\det(I-zM)$}{det} 的因式分解）}
由
\[
\zeta_M(z)=\frac{1}{\Delta(z)},\qquad
\Delta(z)=\det(I-zM)=(1-z)^2(1+z)^3(1-3z+z^2)(1+z-z^2),
\]
以及 $z_\star=\lambda^{-1}\in(0,1)$，可将上式等价写成只含 $\Delta$ 的形式
\[
\log\mathfrak{M}
\;=\;
\log C-\sum_{m\ge 2}\frac{\mu(m)}{m}\log \Delta(z_\star^{\,m}).
\]
进一步把 $\Delta$ 的因子逐项展开，可得一个完全“正实数对数”的闭式：
\[
\log\mathfrak{M}
=\log C
-\sum_{m\ge 2}\frac{\mu(m)}{m}\Bigl[
2\log(1-z_\star^{\,m})
3\log(1+z_\star^{\,m})
\log(1-3z_\star^{\,m}+z_\star^{\,2m})
\log(1+z_\star^{\,m}-z_\star^{\,2m})
\Bigr].
\]
该式在数值上极稳定：因为 $z_\star^{\,m}$ 指数趋零，且每项只需对正数取对数。
等价地，可写成 Euler--Möbius 乘积（将“有限部常数”完全压缩为一个可比标量）：
\[
\mathfrak{M}
=C\prod_{m\ge 2}\zeta_M(z_\star^{\,m})^{\mu(m)/m}
=C\prod_{m\ge 2}\Delta(z_\star^{\,m})^{-\mu(m)/m}.
\]
此外，由 $\sqrt5=2\varphi-1$，留数常数亦可写成
\[
C=\frac{47+21\sqrt5}{100}=\frac{13+21\varphi}{50}.
\]

\subsubsection{Abel 归一化：把主极点锁到 $r\uparrow 1$ 的接口坐标}\label{app:real-input-40-abel}
主文第 9 节把"离散化—正则化—有限部"作为稳定接口：对 SFT/sofic，离散化把动力学压到有限矩阵；
正则化把发散量变成可控解析对象；有限部抽取"模型不变常数"作为可比较标量（定义 \ref{def:abel-finite-part}）。
对真实输入 40 状态核，该闭环已在本附录完全闭合：$M$ 为可审计有限矩阵，$\zeta_M$ 为显式有理函数，
而 $\log\mathfrak{M}$ 为 Abel/有限部输出。

\paragraph{Abel 坐标}
记主特征值 $\lambda=\rho(M)=\varphi^2$，主极点 $z_\star=\lambda^{-1}$。引入 Abel 归一化坐标
\[
r:=\lambda z\quad(\Leftrightarrow z=r/\lambda),\qquad
\widetilde{\zeta}_M(r):=\zeta_M(r/\lambda).
\]
则主极点从 $z=z_\star$ 被搬到 $r=1$，并且由 $\zeta_M(z)=1/\det(I-zM)$ 得到一个同样显式的有理式：
\[
\widetilde{\zeta}_M(r)
=\frac{1}{\det\!\bigl(I-(r/\lambda)M\bigr)}
=\frac{1}{(1-r/\lambda)^2(1+r/\lambda)^3\bigl(1-3r/\lambda+r^2/\lambda^2\bigr)\bigl(1+r/\lambda-r^2/\lambda^2\bigr)}.
\]
利用 $\lambda^2-3\lambda+1=0$ 与 $\lambda=\varphi^2$，两段二次因子可在 $r$-变量中进一步分解为线性因子：
\[
1-3r/\lambda+r^2/\lambda^2=(1-r)\bigl(1-r/\lambda^2\bigr),\qquad
1+r/\lambda-r^2/\lambda^2=\bigl(1+r/\varphi\bigr)\bigl(1-r/\varphi^3\bigr).
\]
因此归一化行列式可写成"极点位置一眼可见"的完全分解：
\[
\det\!\bigl(I-(r/\lambda)M\bigr)
=-\frac{(r-\lambda)^2(r+\lambda)^3(r-1)(r-\lambda^2)(r-\varphi^3)(r+\varphi)}{\lambda^9}.
\]
由此可直接读出 $\widetilde{\zeta}_M(r)$ 的全部奇点位置为
\[
r\in\{1,\ \pm\lambda,\ \lambda^2,\ \varphi^3,\ -\varphi\},
\]
其中除 $r=1$ 外其余均满足 $|r|>1$。因此在 $|r|<1$ 内，$\widetilde{\zeta}_M(r)$ 无任何极点；唯一的临界障碍位于 $r=1$。
并且主极点留数常数在 Abel 坐标下同样可读为
\[
\widetilde{\zeta}_M(r)=\frac{C}{1-r}+O(1),
\qquad
C=\lim_{r\to 1}(1-r)\,\widetilde{\zeta}_M(r)=\frac{47+21\sqrt5}{100}.
\]

\subsubsection{Abel 有限部常数 $\log\mathfrak{M}$ 的接口式}\label{app:real-input-40-finite-part}
令 primitive 轨道生成函数

\boxed{\ \log\mathfrak{M}=\mathrm{fp}_{r=1}\Bigl(\mathcal{M}_{\mathrm{Abel}}(r)+\log(1-r)\Bigr)\ },

这正是主文定义 \ref{def:abel-finite-part}（定义 9.5）“Abel 正则化 + Laurent 常数项有限部”的同型读出：区别仅在于此处的临界 $1/n$ 发散来自 primitive 轨道权 $p_n/\lambda^n\sim 1/n$，而非轨道 Dirichlet 级数的 $n^{-s}$ 临界。

\begin{corollary}[prime-orbit Mertens 常数与 Abel 有限部常数的恒等式]\label{cor:real-input-40-Mertens-constant}
令
\[
\mathcal{M}(N):=\sum_{n\le N}\frac{p_n}{\lambda^n}.
\]
则存在常数 $\mathsf{M}$ 使得
\[
\mathcal{M}(N)=\log N+\mathsf{M}+o(1),
\qquad
\boxed{\ \mathsf{M}=\gamma+\log\mathfrak{M}\ }.
\]
\end{corollary}

\begin{proof}
写
\(
\frac{p_n}{\lambda^n}=\frac{1}{n}+b_n
\)
其中 $b_n:=\frac{p_n}{\lambda^n}-\frac{1}{n}$。
由 $p_n=\lambda^n/n+O(\varphi^n/n)$（见上文 $|\mathrm{spec}_2|=\varphi<\lambda$）可知 $\sum_{n\ge 1}b_n$ 绝对收敛。
于是
\[
\sum_{n\le N}\frac{p_n}{\lambda^n}
\;=\;
\sum_{n\le N}\frac{1}{n}
\;+\;
\sum_{n\le N}b_n
\;=\;
\log N+\gamma+\sum_{n\ge 1}b_n+o(1).
\]
另一方面，由备注 \ref{rem:real-input-40-mertens-abel} 中的 Abel 有限部读出，
\(
\sum_{n\ge 1}b_n=\log\mathfrak{M}
\)
（等价于 $\mathrm{fp}_{r=1}\bigl(\sum b_n r^n\bigr)=\sum b_n$），从而得到 $\mathsf{M}=\gamma+\log\mathfrak{M}$。
\end{proof}

\subsubsection{Dirichlet 变量口径（与 Abel 单位圆盘完全等价）}\label{app:real-input-40-dirichlet}

\mathcal{P}(s):=\sum_{n\ge 1}p_n\,\lambda^{-ns}=P(\lambda^{-s}),
\qquad
\mathcal{Z}(s):=\zeta_M(\lambda^{-s}).

注意到 Abel 归一化的 $r=\lambda z$ 与 Dirichlet 变量满足
\[
z=\lambda^{-s},\qquad r=\lambda^{1-s}\in(0,1)\quad(s>1),

1-\lambda^{1-s}=1-e^{-(s-1)\log\lambda}\sim (s-1)\log\lambda\qquad(s\to 1^+),

\mathcal{P}(r)=-\log(1-r)+\log\mathfrak{M}+o(1)

\mathcal{P}(s)=-\log(s-1)-\log\log\lambda+\log\mathfrak{M}+o(1)\qquad(s\to 1^+).

\mathrm{fp}_{s=1}\bigl(\mathcal{P}(s)+\log(s-1)\bigr)=\log\mathfrak{M}-\log\log\lambda.

另一方面，由 $\widetilde{\zeta}_M(r)=\frac{C}{1-r}+O(1)$ 与 $1-r\sim (s-1)\log\lambda$ 可得

这给出 Abel 单位圆盘变量与 Dirichlet 变量两套口径的完全等价闭式接口。

并由 Abel 有限部定义
\[
\log\mathfrak{M}(\theta)=\mathrm{fp}_{r=1}\Bigl(\sum_{n\ge 1}\frac{p_n(\theta)}{\lambda(\theta)^n}r^n+\log(1-r)\Bigr).
\]
在 $z_\star(\theta)$ 为简单极点时，上式亦可写为一条“主留数 + Möbius 尾”的单级数（与 \S\ref{app:real-input-40-finite-part} 同口径）：
\[
\log\mathfrak{M}(\theta)
=\log C(\theta)+\sum_{m\ge 2}\frac{\mu(m)}{m}\log\zeta_\theta\bigl(z_\star(\theta)^{\,m}\bigr)
=\log C(\theta)+\sum_{m\ge 2}\frac{\mu(m)}{m}\log\zeta_\theta\bigl(\lambda(\theta)^{-m}\bigr).
\]
同理，若定义加权 primitive 轨道的累计计数
\[
\Pi_\theta(N):=\sum_{n\le N}p_n(\theta),
\]
则由 $p_n(\theta)\sim \lambda(\theta)^n/n$ 得到离散“素数定理形状”的常数修正：
\[
\Pi_\theta(N)\sim \frac{\lambda(\theta)}{\lambda(\theta)-1}\cdot\frac{\lambda(\theta)^N}{N}.
\]
若以轨道范数 $N(\tau)=\lambda(\theta)^{|\tau|}$ 定义 $\pi_\theta(x):=\#\{\tau\ \mathrm{primitive}:N(\tau)\le x\}$，则
\[
\pi_\theta(x)\sim \frac{\lambda(\theta)\log\lambda(\theta)}{\lambda(\theta)-1}\cdot\frac{x}{\log x}.
\]
进一步由 primitive 分解（欧拉乘积/幂展开口径）与 Möbius 反演得到一组可直接数值评估的接口式：
\[
\log\mathfrak{M}(\theta)
:=\log C(\theta)-\sum_{m\ge 2}\frac{P_\theta(\lambda(\theta)^{-m})}{m},
\]
\[
P_\theta(z)=\sum_{k\ge 1}\frac{\mu(k)}{k}\log\zeta_\theta(z^k),
\]
从而得到双重 Möbius--Abel 级数；由于 $m\ge 2$，故 $\lambda(\theta)^{-mk}$ 以指数速度趋零，从而该双和在数值上高度稳定。
\[
\log\mathfrak{M}(\theta)
=\log C(\theta)-\sum_{m\ge 2}\frac{1}{m}\sum_{k\ge 1}\frac{\mu(k)}{k}\log\zeta_\theta\bigl(\lambda(\theta)^{-mk}\bigr).
\]
其中 $C(\theta)$ 为主极点留数常数
\[
C(\theta):=\lim_{z\to z_\star(\theta)}(1-\lambda(\theta)z)\,\zeta_\theta(z).
\]
当 $z_\star(\theta)$ 为简单极点时，还可用行列式导数把 $C(\theta)$ 写成只依赖 $\det(I-zM_\theta)$ 的闭式：
\[
C(\theta)=\frac{1}{-z_\star(\theta)\,\partial_z\det(I-zM_\theta)\big|_{z=z_\star(\theta)}}.
\]
数值相图复现见脚本 \texttt{scripts/exp\_sync\_kernel\_real\_input\_40\_logM\_theta.py}。

\paragraph{arity-charge 的线性并入（无需重建核）}
取最小 arity-charge
\[
\chi:=\varphi_e-\varphi_2\in\{-1,0,1\},
\]
则对 $\chi$ 的线性加权 $\exp(\theta_\chi\chi)$ 等价于参数替换
\[
(\theta_e,\theta_2)\mapsto(\theta_e+\theta_\chi,\ \theta_2-\theta_\chi),
\]
等价地，在三参数压力 $P(\theta_e,\theta_-,\theta_2)=\log\rho(M_\theta)$ 上这是一维切线
\[
P_\chi(t):=P(t,0,-t).
\]
因此 $\chi$ 的均值/协方差与 $\log\mathfrak{M}(\theta)$ 沿该方向的切片均可直接从同一族加权核 $M_\theta$ 读出，而无需引入新的状态扩张。

\begin{definition}[四维观测向量与 $\chi$ 维度]\label{def:real-input-40-observables-4d}
在真实输入 40 状态核的边上，取三条基础可加观测量
\[
\varphi_e=e,\qquad
\varphi_-=\mathbf{1}_{\{t\in Q_-\}},\qquad
\varphi_2=\mathbf{1}_{\{d=2\}},
\]
并定义最小 arity-charge
\[
\chi:=\varphi_e-\varphi_2\in\{-1,0,1\}.
\]
记四维观测向量为
\(
\Psi:=(\varphi_e,\varphi_-,\varphi_2,\chi)
\)
并对参数 $\vartheta=(\theta_e,\theta_-,\theta_2,\theta_\chi)\in\RR^4$ 定义加权矩阵族
\[
(M_\vartheta)_{ij}:=\sum_{i\to j}\exp\bigl(\theta_e\varphi_e+\theta_-\varphi_-+\theta_2\varphi_2+\theta_\chi\chi\bigr).
\]
\end{definition}

\begin{proposition}[$\chi$ 作为第四维并不引入新自由度（线性重参数化）]\label{prop:chi-reparam-4d}
对任意 $\vartheta=(\theta_e,\theta_-,\theta_2,\theta_\chi)$，有恒等式
\[
M_{(\theta_e,\theta_-,\theta_2,\theta_\chi)}\;=\;M_{(\theta_e+\theta_\chi,\ \theta_-,\ \theta_2-\theta_\chi,\ 0)},
\]
从而对所有“素数影子”接口量（压力、$\zeta$、Abel 有限部常数）都有同一重参数化：
\[
\lambda(\theta_e,\theta_-,\theta_2,\theta_\chi)=\lambda(\theta_e+\theta_\chi,\theta_-,\theta_2-\theta_\chi),
\]
\[
\zeta_{\vartheta}(z)=\frac{1}{\det(I-zM_\vartheta)}=\zeta_{(\theta_e+\theta_\chi,\theta_-,\theta_2-\theta_\chi)}(z),
\]
\[
\log\mathfrak{M}(\vartheta)=\log\mathfrak{M}(\theta_e+\theta_\chi,\theta_-,\theta_2-\theta_\chi).
\]
\end{proposition}

\begin{proof}
由定义 \ref{def:real-input-40-observables-4d}，$\chi=\varphi_e-\varphi_2$，故
\[
\theta_e\varphi_e+\theta_2\varphi_2+\theta_\chi\chi
=(\theta_e+\theta_\chi)\varphi_e+(\theta_2-\theta_\chi)\varphi_2.
\]
把该恒等式代入每条边的权重定义即可得 $M_\vartheta$ 的逐项相等；其余结论由 $\lambda=\rho(M)$、$\zeta=1/\det(I-zM)$ 与 Abel 有限部定义在同一矩阵族上的函子性直接推出。
\end{proof}

\begin{corollary}[$\chi$ 的均值与协方差密度（由三维 Hessian 线性组合读出）]\label{cor:chi-moments-from-hessian}
令 $P(\theta_e,\theta_-,\theta_2)=\log\rho(M_\theta)$。则
\[
\mathbb{E}[\chi]=\partial_{\theta_e}P(0)-\partial_{\theta_2}P(0)=\frac{17-5\sqrt5}{20},
\]
并且协方差密度满足
\[
\mathrm{Var}_\infty(\chi)=\partial_{\theta_e\theta_e}^2P(0)+\partial_{\theta_2\theta_2}^2P(0)-2\partial_{\theta_e\theta_2}^2P(0),
\qquad
\mathrm{Cov}_\infty(\chi,\varphi_-)=\partial_{\theta_e\theta_-}^2P(0)-\partial_{\theta_2\theta_-}^2P(0).
\]
\end{corollary}

\]
则离散傅里叶反演给出
\[
p_{n,(r_1,r_2)}^{(m_1,m_2)}
=\frac{1}{m_1m_2}\sum_{j_1=0}^{m_1-1}\sum_{j_2=0}^{m_2-1}
\omega_{m_1}^{-j_1r_1}\omega_{m_2}^{-j_2r_2}\,P_n(\omega_{m_1}^{j_1},\omega_{m_2}^{j_2}).

\[

\paragraph{定理（扭曲谱半径下降与指数等分布）}
记
\[

\qquad \omega_m=e^{2\pi i/m},
\]
则对任意同余类 $(r_1,r_2)$ 有
\[

定义

C_{r_1,r_2}^{(m_1,m_2)}:=\frac{\mathsf{M}_{\varphi^2}}{m_1m_2}
+\frac{1}{m_1m_2}\sum_{(j_1,j_2)\neq(0,0)}\omega_{m_1}^{-j_1r_1}\omega_{m_2}^{-j_2r_2}
C(\omega_{m_1}^{j_1},\omega_{m_2}^{j_2}),
\]
则
\[
\sum_{n\le N}\frac{p_{n,(r_1,r_2)}^{(m_1,m_2)}}{\lambda^n}
=\frac{1}{m_1m_2}\log N+C_{r_1,r_2}^{(m_1,m_2)}+o(1).
\]

在 $(m_1,m_2)=(2,2)$ 下，二维 Mertens 和满足
\[
\sum_{n\le N}\frac{p_{n,(r_1,r_2)}^{(2,2)}}{\lambda^n}
=\frac{1}{4}\log N+C_{r_1,r_2}+o(1),
\]
其中

\subsubsection{真实输入核的一参数加权 $\zeta(z,u)$（输出位势）}\label{app:real-input-40-zeta-u}
只对输出位势 $\varphi_e=e$ 加权，令 $u=e^{\theta_e}$，记相应的加权矩阵为 $M(u)$，并定义
\[
\zeta(z,u)=\frac{1}{\Delta(z,u)},\qquad \Delta(z,u):=\det(I-zM(u)).
\]
在真实输入 40 状态核上，$\Delta(z,u)$ 仍可压缩为低阶多项式并显式因子化：
\[
\Delta(z,u)=(1-uz^2)\Bigl(
1-z-6uz^2+2uz^3+(9u^2-u)z^4+(u-3u^2)z^5-(u^3+2u^2)z^6
+(4u^3-3u^2)z^7+(u^3-u^4)z^8
\Bigr).
\]
其中因子 $1-uz^2$ 并非“偶然代数分解”：它对应一条可在 primitive 层面被\textbf{精确定位}的原子闭轨道，并由此给出 Abel 有限部常数的精确剥离。

\begin{proposition}[原子 \texorpdfstring{$(2,1)$}{(2,1)}-prime orbit 与有限部常数的精确剥离]\label{prop:atomic-2-orbit-split-logM}
对加权矩阵族 $M(u)$，令每条 primitive 周期轨道 $\tau$ 的长度为 $|\tau|$，并记其一圈累计输出 $1$ 的次数为
\[
E(\tau):=\sum_{e\in\tau} e\in\ZZ_{\ge 0}.
\]
则有且仅有一条 primitive 轨道 $\tau_2$ 满足
\[
|\tau_2|=2,\qquad E(\tau_2)=1,
\]
并且该轨道在加权 zeta 的 Euler 乘积中贡献出因子 $(1-uz^2)^{-1}$。

进一步，令未加权 primitive 生成函数
\(
P(z)=\sum_{n\ge 1}p_n z^n
\)（$p_n$ 为未加权 primitive 轨道数，按轨道计），并定义“去掉原子 $2$-轨道后的核心部分”
\[
P_{\mathrm{core}}(z):=P(z)-z^2.
\]
则在 $z\to z_\star=\lambda^{-1}$ 时
\[
P_{\mathrm{core}}(z)=-\log(1-\lambda z)+\log\mathfrak{M}_{\mathrm{core}}+o(1),
\]
并满足常数的精确分解
\[
\boxed{\ \log\mathfrak{M}=\log\mathfrak{M}_{\mathrm{core}}+z_\star^2\ }.
\]
在本核上 $z_\star=\varphi^{-2}$，故
\[
z_\star^2=\varphi^{-4}=\frac{7-3\sqrt5}{2}\approx 0.145898033750315.
\]
因此由上文数值 $\log\mathfrak{M}\approx -0.254312661901716$ 可得
\[
\log\mathfrak{M}_{\mathrm{core}}\approx -0.400210695652031,
\qquad
\mathsf{M}_{\mathrm{core}}:=\gamma+\log\mathfrak{M}_{\mathrm{core}}\approx 0.177004969249501.
\]
\end{proposition}

\begin{proof}
对带权 SFT，闭轨道 Euler 乘积给出
\[
\zeta(z,u)=\prod_{\tau\ \mathrm{primitive}}\left(1-u^{E(\tau)}z^{|\tau|}\right)^{-1}
\qquad\text{（参见 \cite{BrinStuck2002,ParryPollicott1990Zeta} 的同型推导）}.
\]
与 $\Delta(z,u)=\det(I-zM(u))$ 的显式分解相比，因子 $(1-uz^2)^{-1}$ 只能由满足 $(|\tau|,E(\tau))=(2,1)$ 的 primitive 轨道贡献。
若存在两条不同的此类轨道，则该因子在乘积中应出现二次幂；与行列式分解矛盾，故此类轨道存在且唯一。

另一方面，$u=1$ 时该轨道在未加权 primitive 生成函数中贡献 $z^{|\tau_2|}=z^2$，从而
$P(z)=z^2+P_{\mathrm{core}}(z)$。由于 $z^2$ 在 $z=z_\star$ 处解析，故 Abel 有限部常数相差恰为
$z_\star^2$，即 $\log\mathfrak{M}=\log\mathfrak{M}_{\mathrm{core}}+z_\star^2$。
\end{proof}

\begin{proposition}[输出压力的冻结端点与 \texorpdfstring{$\sqrt u$}{sqrt(u)} 标度]\label{prop:real-input-40-pressure-freezing}
令 $u=e^{\theta}$，记
\[
\lambda(u):=\rho(M(u)),\qquad P(\theta):=\log\lambda(e^{\theta}).
\]
把 $\Delta(z,u)=(1-uz^2)F(z,u)$ 中的 $F$ 写为 $\lambda$-多项式
\[
G(\lambda,u):=\lambda^8\,F(\lambda^{-1},u)
=\lambda^8-\lambda^7-6u\lambda^6+2u\lambda^5+(9u^2-u)\lambda^4+(u-3u^2)\lambda^3-(u^3+2u^2)\lambda^2
+(4u^3-3u^2)\lambda+(u^3-u^4).
\]
则 $\lambda(u)$ 为 $G(\lambda,u)=0$ 的最大正实根，并且：
\begin{enumerate}
  \item \textbf{$u\to 0$（$\theta\to-\infty$）冻结：}有 $\lambda(u)\to 1$，并且在 $u=0$ 的正规微扰给出
  \[
  \lambda(u)=1+4u-5u^2-115u^3+2317u^4+O(u^5),
  \]
  从而
  \(
  P(\theta)=4e^{\theta}+O(e^{2\theta})
  \)。
  \item \textbf{$u\to\infty$（$\theta\to+\infty$）饱和：}存在常数 $c>0,d\in\RR$ 使得
  \[
  \lambda(u)=c\sqrt u+d+O(u^{-1/2}),
  \]
  其中 $y:=c^2$ 为四次方程
  \[
  y^4-6y^3+9y^2-y-1=0
  \]
  的最大正根；并且
  \[
  d=\frac{y^3-2y^2+3y-4}{8y^3-36y^2+36y-2}\Bigg|_{y=c^2}.
  \]
  数值上 $c\approx 1.89635299351377,\ d\approx 0.807943317323508$，从而
  \[
  P(\theta)=\frac{\theta}{2}+\log c+\frac{d}{c}e^{-\theta/2}+O(e^{-\theta}).
  \]
\end{enumerate}
因此
\[
\lim_{\theta\to-\infty}P'(\theta)=0,\qquad \lim_{\theta\to+\infty}P'(\theta)=\frac12,
\]
即输出 $1$ 的典型密度在极端偏置下冻结在区间 $[0,\tfrac12]$。
\end{proposition}
