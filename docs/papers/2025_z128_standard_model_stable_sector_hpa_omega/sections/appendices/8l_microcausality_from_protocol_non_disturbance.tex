\subsection{Microcausality from protocol-level non-disturbance (T2-2)}
\label{app:microcausality_from_protocol_non_disturbance}

\AuditTag This appendix records a theorem-facing route from protocol-level non-disturbance/no-signaling constraints to microcausality of the induced net.
The result is conditional: it upgrades microcausality from a free-standing interface axiom into a theorem under a declared class of admissible protocol organizations.

\subsubsection{Non-disturbance as a composability constraint}
\paragraph{Protocol separability hypothesis.}
Let $\mathcal{O}_1,\mathcal{O}_2\in\mathfrak{R}_{\mathrm{prot}}$ be two regions induced by windows (Appendix~\ref{app:construct_local_net_from_protocol}).
We record an admissibility hypothesis capturing ``constructible non-disturbance'':

\begin{assumption}[Causal separability of readout channels]
\label{ass:causal_separability_readout_channels}
For any two operationally disjoint regions $\mathcal{O}_1,\mathcal{O}_2$ (intended to be spacelike separated under the interface map), the global observable algebra on their union is generated by commuting subalgebras:
\[
\mathcal{A}_{\mathrm{prot}}(\mathcal{O}_1\cup\mathcal{O}_2)
=
\mathcal{A}_{\mathrm{prot}}(\mathcal{O}_1)\,\vee\,\mathcal{A}_{\mathrm{prot}}(\mathcal{O}_2),
\qquad
[\mathcal{A}_{\mathrm{prot}}(\mathcal{O}_1),\mathcal{A}_{\mathrm{prot}}(\mathcal{O}_2)]=0,
\]
where $\vee$ denotes the unital $^\ast$-algebra generated by the two subalgebras.
\end{assumption}

\AuditTag Assumption~\ref{ass:causal_separability_readout_channels} is the explicit protocol-level content of ``no instantaneous influence'' in the algebraic language.
It can be audited by verifying that the declared readout construction is realized as a tensor-product organization of subsystems (Appendix~\ref{app:composite_systems_tensor_products}) or as a commuting-subalgebra organization in the GNS representation.

\paragraph{PT carrier subclass.}
\AuditTag In the tensor-product readout subclass (Appendix~\ref{app:protocol_subclass_tensor_net}), Assumption~\ref{ass:causal_separability_readout_channels} is discharged by construction (Proposition~\ref{prop:causal_separability_from_tensor_subclass}).

\subsubsection{Microcausality theorem for the protocol-induced net}
\begin{theorem}[Microcausality under causal separability]
\label{thm:microcausality_under_causal_separability}
\MathTag Under Assumption~\ref{ass:causal_separability_readout_channels}, for any pair of operationally disjoint regions $\mathcal{O}_1,\mathcal{O}_2$ the protocol-induced net satisfies microcausality:
\[
[A,B]=0,\qquad \forall\,A\in\mathcal{A}_{\mathrm{prot}}(\mathcal{O}_1),\ B\in\mathcal{A}_{\mathrm{prot}}(\mathcal{O}_2).
\]
\end{theorem}
\begin{proof}
This is immediate from the commutativity clause in Assumption~\ref{ass:causal_separability_readout_channels}.
\end{proof}

\paragraph{Interface reading.}
\InterfaceTag Given an explicit interface map identifying ``operationally disjoint'' with ``spacelike separated'' in Minkowski space, Theorem~\ref{thm:microcausality_under_causal_separability} provides the microcausality axiom of an AQFT net as a theorem from protocol composability.

\subsubsection{Failure points}
\AuditTag If Assumption~\ref{ass:causal_separability_readout_channels} fails, then microcausality cannot be asserted as a theorem.
In that case, the manuscript must treat microcausality as an explicit interface axiom (Appendix~\ref{app:aqft_axioms_local_nets}) and record the mismatch as a failure-point instance (Appendix~\ref{app:minimal_failure_point_templates}).
