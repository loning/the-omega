\subsection{Covariance and spectrum from a declared protocol dynamics class (T2-3)}
\label{app:covariance_spectrum_from_protocol_dynamics}

\AuditTag This appendix records a theorem-facing version of covariance and the spectrum condition under a declared dynamics class on the protocol-induced net.
As with microcausality, this is conditional: we isolate the minimal hypotheses and make the failure modes explicit.

\subsubsection{Dynamics as an automorphism action}
\paragraph{Protocol dynamics class.}
Let $\mathcal{A}_{\mathrm{prot}}$ be the quasi-local algebra constructed in Appendix~\ref{app:construct_local_net_from_protocol}.
We declare a dynamics class by specifying a group (or semigroup) action by $^\ast$-automorphisms:
\[
\alpha_g:\mathcal{A}_{\mathrm{prot}}\to\mathcal{A}_{\mathrm{prot}},
\qquad g\in \mathcal{G},
\]
such that $\alpha_g(\mathcal{A}_{\mathrm{prot}}(\mathcal{O}))=\mathcal{A}_{\mathrm{prot}}(g\mathcal{O})$ for the induced action on the region family.

\begin{assumption}[Covariant dynamics class]
\label{ass:covariant_dynamics_class}
There exists a subgroup $\mathcal{G}$ (intended as a realization of translations or the Poincar\'e group, depending on scope) and an action $\alpha$ by $^\ast$-automorphisms on $\mathcal{A}_{\mathrm{prot}}$ satisfying net covariance:
\[
\alpha_g\bigl(\mathcal{A}_{\mathrm{prot}}(\mathcal{O})\bigr)=\mathcal{A}_{\mathrm{prot}}(g\mathcal{O})
\quad\text{for all }g\in\mathcal{G},\ \mathcal{O}\in\mathfrak{R}_{\mathrm{prot}}.
\]
\end{assumption}

\paragraph{PT carrier for covariance.}
\AuditTag In the presence of a window/region action compatible with refinement, covariance can be discharged as a PT statement by induction on the directed system; see Appendix~\ref{app:covariance_from_window_action}.

\subsubsection{Vacuum state and spectral positivity}
\begin{assumption}[Vacuum state and implementability]
\label{ass:vacuum_state_implementability}
There exists a state $\omega_0$ on $\mathcal{A}_{\mathrm{prot}}$ invariant under $\alpha$.
In the GNS representation $(\pi_0,\mathcal{H}_0,\Omega_0)$, $\alpha$ is implemented by a unitary representation $U(g)$ with $U(g)\Omega_0=\Omega_0$.
\end{assumption}

\begin{assumption}[Spectrum condition (template)]
\label{ass:spectrum_condition_template}
For the translation subgroup (when included in $\mathcal{G}$), the joint spectrum of the self-adjoint generators lies in the closed forward light cone.
\end{assumption}

\subsubsection{Theorem layer}
\begin{theorem}[Covariance/spectrum for the protocol-induced net (conditional)]
\label{thm:covariance_spectrum_conditional}
\MathTag Under Assumptions~\ref{ass:covariant_dynamics_class}--\ref{ass:spectrum_condition_template}, the protocol-induced net can be viewed as a covariant AQFT net satisfying the spectrum condition in the GNS representation.
\end{theorem}
\begin{proof}
This is a direct unpacking of the assumptions and matches Definitions~\ref{def:z128_poincare_covariance_net}--\ref{def:z128_vacuum_spectrum_condition} once the interface identification of region families is fixed.
\end{proof}

\subsubsection{Failure points}
\AuditTag If covariance implementability or the spectrum condition cannot be discharged in the declared dynamics class, then scattering and Wightman-bridge steps cannot be used as theorem-level consequences.
In that case, those statements must be ledger-scoped as interface assumptions, and their failure recorded using the normalized template in Appendix~\ref{app:minimal_failure_point_templates}.
