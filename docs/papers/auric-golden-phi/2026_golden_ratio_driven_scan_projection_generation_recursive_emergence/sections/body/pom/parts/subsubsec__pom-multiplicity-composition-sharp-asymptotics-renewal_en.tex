\subsubsection{Simple-Pole Constants, Renewal Hitting, and Two-Parameter Pressure: Sharp Leading Terms, Limit Theorems, and Exponential Mixing for \texorpdfstring{$Z_L^{(q)}$}{ZLq}}\label{subsec:pom-multiplicity-composition-sharp-asymptotics-renewal}
This section, building upon the ``exact conditional i.i.d.'' renewal representation and transfer matrix equivalence of \S\ref{subsec:pom-multiplicity-composition-exact-update-softcore},
sharpens the integer-parameter (and indeed all real-parameter) moment pressure \(\lambda(q)\) to the level of \emph{leading-term constants} and \emph{exponential convergence rates},
and accordingly provides a unified template of limit theorems for \(q\)-tilted compositions regarding part counts, entropy rates, extreme values, and path fluctuations.
For consistency with the preceding text, we use
\(
A_q(t):=\sum_{k\ge 1}F_{k+2}^{\,q}t^k
\)
as a synonymous notation for \(W_q(t)\) from Definition \ref{def:pom-multiplicity-real-q-pressure} (to emphasize its role as a ``renewal atom'').

\paragraph{Sharp leading term: the simple-pole constant \texorpdfstring{$C(q)=1/\mu_q$}{C(q)=1/mu\_q}}
\begin{proposition}[Simple-pole constant and sharp leading term for the integer-parameter moment partition function]\label{prop:pom-multiplicity-composition-sharp-main-term-constant}
Take a real number \(q\ge 0\) and let
\[
A_q(t):=\sum_{k\ge 1}F_{k+2}^{\,q}t^k,\qquad |t|<\varphi^{-q},
\qquad \varphi=\frac{1+\sqrt5}{2}.
\]
Let \(\rho(q)\in(0,\varphi^{-q})\) be the unique solution of \(A_q(\rho(q))=1\), and let \(\lambda(q):=\rho(q)^{-1}\) (Definition \ref{def:pom-multiplicity-real-q-pressure}).
For \(L\ge 0\) define
\[
Z_L^{(q)}:=\sum_{\boldsymbol{\ell}\in\mathcal{C}_L}\prod_i F_{\ell_i+2}^{\,q}.
\]
Then the generating function satisfies the rigorous identity
\begin{equation}\label{eq:pom-multiplicity-composition-gf-simple-pole}
\boxed{\;
\sum_{L\ge 0}Z_L^{(q)}t^L=\frac{1}{1-A_q(t)}\;}
\end{equation}
and has a simple pole at \(t=\rho(q)\).
Let the renewal distribution \(p_q\) and its mean \(\mu_q\) be
\[
p_q(k):=F_{k+2}^{\,q}\rho(q)^k\quad(k\ge 1),
\qquad
\mu_q:=\sum_{k\ge 1}k\,p_q(k)=\rho(q)A_q'(\rho(q)).
\]
Then there exists \(\theta(q)\in(0,1)\) such that the sharp asymptotics hold:
\begin{equation}\label{eq:pom-multiplicity-composition-sharp-asymptotic-ZLq}
\boxed{\;
Z_L^{(q)}=\frac{1}{\rho(q)A_q'(\rho(q))}\,\lambda(q)^{L}\;+\;O\!\bigl(\lambda(q)^L\,\theta(q)^{L}\bigr)
=\frac{1}{\mu_q}\,\lambda(q)^{L}\;+\;O\!\bigl(\lambda(q)^L\,\theta(q)^{L}\bigr).}
\end{equation}
Equivalently, the leading-term constant is \(C(q)=1/\mu_q\).
\end{proposition}
\begin{proof}[Proof outline]
The generating function identity \eqref{eq:pom-multiplicity-composition-gf-simple-pole} follows from the multiplicative concatenation property of ordered compositions: each part of length \(k\) contributes the atomic weight \(F_{k+2}^{\,q}t^k\), so the total over all compositions is \(\sum_{r\ge 0}A_q(t)^r=1/(1-A_q(t))\).
\par
Since \(A_q\) is strictly increasing on \((0,\varphi^{-q})\) and \(A_q(\rho(q))=1\), we have \(A_q'(\rho(q))>0\), so \(1-A_q(t)\) has a simple zero at \(t=\rho(q)\), making \(t=\rho(q)\) a simple pole.
By the positivity of coefficients and Pringsheim's theorem, the radius of convergence is \(\rho(q)\), and the dominant singularity occurs only on the positive real axis.
Taking \(R(q)>\rho(q)\) as the modulus of the next-nearest singularity (or any \(R\in(\rho(q),R(q))\)), the function is analytic in \(|t|<R\) except for this simple pole;
by the standard residue/transfer theorem, the leading coefficient is given by \(\mathrm{Res}_{t=\rho(q)}(1/(1-A_q(t)))\), namely
\(
1/(\rho(q)A_q'(\rho(q)))
\),
and the error is controlled by \(\theta(q)=\rho(q)/R\in(0,1)\), yielding \eqref{eq:pom-multiplicity-composition-sharp-asymptotic-ZLq}. \qed
\end{proof}

\paragraph{Exponential convergence of hitting probabilities: renewal interpretation}
\begin{proposition}[Renewal hitting probability and rigorous factorization of \texorpdfstring{$Z_L^{(q)}$}{ZLq}; exponential convergence \texorpdfstring{$u_q(L)\to 1/\mu_q$}{u\_q(L)->1/mu\_q}]\label{prop:pom-multiplicity-composition-renewal-hit-prob}
Take \(q\ge 0\), let \(p_q\) be as in Proposition \ref{prop:pom-multiplicity-composition-sharp-main-term-constant}, and define the renewal hitting probability
\[
u_q(L):=\sum_{r\ge 1}p_q^{*r}(L)\qquad(L\ge 1),
\]
where \(p_q^{*r}\) denotes the \(r\)-fold convolution.
Then for each \(L\ge 1\) the rigorous identity holds:
\begin{equation}\label{eq:pom-multiplicity-composition-ZLq-renewal-factor}
\boxed{\;
Z_L^{(q)}=\lambda(q)^{L}\,u_q(L).\;}
\end{equation}
Moreover, since \(p_q(k)>0\) for all \(k\ge 1\) (hence \(\gcd\{k:p_q(k)>0\}=1\)), there exists a constant \(r(q)>1\) such that
\begin{equation}\label{eq:pom-multiplicity-composition-renewal-exp-conv}
\boxed{\;
u_q(L)=\frac{1}{\mu_q}+O\!\bigl(r(q)^{-L}\bigr)\qquad(L\to\infty).\;}
\end{equation}
Here \(r(q)>1\) can be taken as the reciprocal of the error factor \(\theta(q)\) from Proposition \ref{prop:pom-multiplicity-composition-sharp-main-term-constant}:
from \(u_q(L)=\rho(q)^L Z_L^{(q)}\) and \eqref{eq:pom-multiplicity-composition-sharp-asymptotic-ZLq} one immediately obtains
\(
u_q(L)=\frac{1}{\mu_q}+O(\theta(q)^L)
\),
so one may set
\(
r(q)=\theta(q)^{-1}
\).
When \(A_q\) admits an auditable meromorphic continuation (for example, in the integer \(q\) case, by the rationality of Proposition \ref{prop:pom-multiplicity-composition-moment-hierarchy-rational-growth}),
\(r(q)\) can also be equivalently characterized as the modulus of the nearest singularity of the renewal generating function
\(
U_q(z)=\sum_{L\ge 1}u_q(L)z^L
\)
beyond \(z=1\).
\end{proposition}
\begin{proof}[Proof outline]
From \(p_q(k)=F_{k+2}^{\,q}\rho(q)^k\), for any \(\boldsymbol{\ell}\in\mathcal{C}_L\) we have
\(
\prod_i p_q(\ell_i)=\rho(q)^{L}\prod_i F_{\ell_i+2}^{\,q}
\).
Summing over a given \(r\) yields
\[
p_q^{*r}(L)=\rho(q)^L\sum_{\substack{\boldsymbol{\ell}\in\mathcal{C}_L\\ R(\boldsymbol{\ell})=r}}\ \prod_i F_{\ell_i+2}^{\,q}.
\]
Summing over \(r\ge 1\) gives
\(
u_q(L)=\rho(q)^L Z_L^{(q)}
\),
i.e., \(Z_L^{(q)}=\rho(q)^{-L}u_q(L)=\lambda(q)^L u_q(L)\), establishing \eqref{eq:pom-multiplicity-composition-ZLq-renewal-factor}.
\par
From \(u_q(L)=\rho(q)^L Z_L^{(q)}\) and the sharp asymptotics \eqref{eq:pom-multiplicity-composition-sharp-asymptotic-ZLq} of Proposition \ref{prop:pom-multiplicity-composition-sharp-main-term-constant},
one immediately obtains
\(
u_q(L)=\frac{1}{\mu_q}+O(\theta(q)^L)
\),
where \(\theta(q)\in(0,1)\).
Rewriting as \(O(r(q)^{-L})\) (with \(r(q)=\theta(q)^{-1}\)) gives \eqref{eq:pom-multiplicity-composition-renewal-exp-conv}. \qed
\end{proof}

\begin{corollary}[Renewal interpretation of Proposition \ref{prop:pom-multiplicity-composition-sharp-main-term-constant}]\label{cor:pom-multiplicity-composition-sharp-main-term-from-renewal}
Under the setting of Proposition \ref{prop:pom-multiplicity-composition-renewal-hit-prob},
\[
Z_L^{(q)}=\frac{1}{\mu_q}\lambda(q)^L+O\!\left(\Bigl(\frac{\lambda(q)}{r(q)}\Bigr)^{\!L}\right),
\]
so \(\theta(q)\) can be taken as \(1/r(q)\).
\end{corollary}

\paragraph{Limit laws for part counts: LLN/CLT/local limit theorem for general \texorpdfstring{$q$}{q}}
\begin{theorem}[Limit laws for part counts of \texorpdfstring{$q$}{q}-tilted compositions]\label{thm:pom-multiplicity-composition-partcount-generalq-llt}
Fix \(q\ge 0\) and let \(R_L\) denote the number of parts under \(P_L^{(q)}(\boldsymbol{\ell})\propto \prod_i F_{\ell_i+2}^{\,q}\).
Let \(\ell\sim p_q\) be the renewal increment of Proposition \ref{prop:pom-multiplicity-composition-sharp-main-term-constant},
and write
\[
\mu_q:=\mathbb{E}[\ell],\qquad
\sigma_q^2:=\mathrm{Var}(\ell)=\Bigl(\rho(q)^2A_q''(\rho(q))+\rho(q)A_q'(\rho(q))\Bigr)-\mu_q^2\in(0,\infty).
\]
Then the law of large numbers and central limit theorem hold:
\begin{equation}\label{eq:pom-multiplicity-composition-partcount-lln-clt-generalq}
\boxed{\;
\frac{R_L}{L}\xrightarrow[L\to\infty]{}\frac{1}{\mu_q}\qquad(\text{in probability}),\qquad
\frac{R_L-\frac{L}{\mu_q}}{\sqrt{L}}\Longrightarrow \mathcal{N}\!\left(0,\ \frac{\sigma_q^2}{\mu_q^{3}}\right).}
\end{equation}
Moreover, a local limit theorem holds: for integers \(r\) satisfying \(\bigl|r-\frac{L}{\mu_q}\bigr|=O(\sqrt{L})\),
\begin{equation}\label{eq:pom-multiplicity-composition-partcount-local-limit-generalq}
\mathbb{P}_{P_L^{(q)}}(R_L=r)
=\frac{1}{\sqrt{2\pi(\sigma_q^2/\mu_q^3)L}}
\exp\!\left(-\frac{\bigl(r-\frac{L}{\mu_q}\bigr)^2}{2(\sigma_q^2/\mu_q^3)L}\right)
\;+\;O(L^{-1/2}),
\end{equation}
where the error term can be made uniform over this central window.
\end{theorem}
\begin{proof}[Proof outline]
From the two-parameter partition function \(Z_L^{(q)}(y)\) (see Definition \ref{def:pom-multiplicity-composition-two-parameter-partition} below) and
\(
\mathbb{E}_{P_L^{(q)}}[e^{tR_L}]=Z_L^{(q)}(e^t)/Z_L^{(q)}(1)
\),
combined with the simple pole and analytic perturbation of Proposition \ref{prop:pom-multiplicity-composition-sharp-main-term-constant},
one can write \(\log \mathbb{E}[e^{tR_L}]\) as \(L\) times an analytic function plus a bounded remainder (quasi-power form).
From this, the LLN/CLT and local limit theorem follow;
the mean and variance are given by the first and second derivatives of \(\log\lambda(q,e^t)\) at \(t=0\),
which can be explicitly aligned with the renewal distribution moments (\(\mu_q,\sigma_q^2\)), yielding \eqref{eq:pom-multiplicity-composition-partcount-lln-clt-generalq}--\eqref{eq:pom-multiplicity-composition-partcount-local-limit-generalq}. \qed
\end{proof}

\paragraph{Two-parameter pressure and joint large deviations: auditable rate function for \texorpdfstring{$(R_L/L,E_L)$}{(R\_L/L,E\_L)}}
\begin{definition}[Two-parameter partition function, root equation, and two-parameter pressure]\label{def:pom-multiplicity-composition-two-parameter-partition}
For \(q\ge 0\) and \(y>0\), define the two-parameter partition function
\[
Z_L^{(q)}(y):=\sum_{\boldsymbol{\ell}\in\mathcal{C}_L}y^{R(\boldsymbol{\ell})}\prod_i F_{\ell_i+2}^{\,q}.
\]
Define \(\rho(q,y)\in(0,\varphi^{-q})\) as the unique solution of
\(
yA_q(\rho(q,y))=1
\), and write
\[
\boxed{\;\lambda(q,y):=\rho(q,y)^{-1}.\;}
\]
\end{definition}

\begin{theorem}[Two-parameter pressure and joint large deviations: G\"artner--Ellis rate function for \texorpdfstring{$(R_L/L,E_L)$}{(R\_L/L,E\_L)}]\label{thm:pom-multiplicity-composition-joint-ldp-partcount-energy}
Fix \(q\ge 0\) and define the energy density under \(P_L^{(q)}\) as
\[
E_L(\boldsymbol{\ell}):=\frac{1}{L}\log w(\boldsymbol{\ell}),
\qquad
w(\boldsymbol{\ell})=\prod_i F_{\ell_i+2}.
\]
Then the vector \(\bigl(\frac{R_L}{L},E_L\bigr)\) satisfies a G\"artner--Ellis-type large deviation principle with normalized cumulant generating function
\begin{equation}\label{eq:pom-multiplicity-composition-joint-cgf}
\boxed{\;
\Psi_q(t,s)
:=\lim_{L\to\infty}\frac{1}{L}\log \mathbb{E}_{P_L^{(q)}}\!\big[\exp(tR_L+sLE_L)\big]
=\log \lambda(q+s,e^{t})-\log\lambda(q,1),\qquad s>-q.}
\end{equation}
The corresponding rate function is the bivariate Legendre--Fenchel transform
\begin{equation}\label{eq:pom-multiplicity-composition-joint-rate}
\boxed{\;
I_q(r,e)=\sup_{t\in\RR,\ s>-q}\bigl\{tr+se-\Psi_q(t,s)\bigr\}.}
\end{equation}
On the interior of its range, \(I_q\) is strictly convex and real-analytic; its gradient parametrization is given by the moments (\(\mathbb{E}[\ell]\), \(\mathbb{E}[\log F_{\ell+2}]\)) of the tilted renewal distribution
\[
p_{q+s,e^t}(k):=e^{t}F_{k+2}^{\,q+s}\rho(q+s,e^t)^{k}\qquad(k\ge 1),
\]
thereby achieving parametrized auditability.
\end{theorem}
\begin{proof}[Proof outline]
From the definition and \(P_L^{(q)}(\boldsymbol{\ell})=w(\boldsymbol{\ell})^q/Z_L^{(q)}(1)\),
\[
\mathbb{E}_{P_L^{(q)}}\!\big[e^{tR_L+sLE_L}\big]
=\frac{1}{Z_L^{(q)}(1)}\sum_{\boldsymbol{\ell}\in\mathcal{C}_L}e^{tR(\boldsymbol{\ell})}w(\boldsymbol{\ell})^{q+s}
=\frac{Z_L^{(q+s)}(e^t)}{Z_L^{(q)}(1)}.
\]
By the concatenation structure, the two-parameter generating function satisfies
\(
\sum_{L\ge 0}Z_L^{(q)}(y)z^L=1/(1-yA_q(z))
\),
so the dominant singularity is given by the unique positive root \(\rho(q,y)\) of \(yA_q(z)=1\).
Therefore
\(
L^{-1}\log Z_L^{(q)}(y)\to \log\lambda(q,y)
\).
Substituting back and taking the limit yields \eqref{eq:pom-multiplicity-composition-joint-cgf}.
Strict convexity and analyticity come from the strict convex analyticity of \((q,t)\mapsto\log\lambda(q,e^t)\) (obtainable via H\"older's inequality and the implicit function theorem; see also the proof strategy of Theorem \ref{thm:pom-multiplicity-real-q-pressure} for the one-parameter strict convexity).
The Legendre--Fenchel transform gives \eqref{eq:pom-multiplicity-composition-joint-rate}. \qed
\end{proof}

\begin{proposition}[Strict convexity of the two-parameter pressure: positive definiteness of the Hessian of \texorpdfstring{$(q,t)\mapsto\log\lambda(q,e^t)$}{(q,t)->log lambda(q,e^t)}]\label{prop:pom-multiplicity-composition-two-parameter-hessian-pd}
Let \(f(q,t):=\log\lambda(q,e^t)\) (\(q>0,t\in\RR\)), and let \(\rho(q,e^t)=\lambda(q,e^t)^{-1}\) as in Definition \ref{def:pom-multiplicity-composition-two-parameter-partition}.
Define the tilted renewal distribution
\[
p_{q,t}(k):=e^t F_{k+2}^{\,q}\rho(q,e^t)^{k}\qquad(k\ge 1),
\qquad \sum_{k\ge 1}p_{q,t}(k)=1,
\]
and under \(p_{q,t}\) write the random variable \(K:=k\), \(X:=\log F_{k+2}\), with means
\[
\mu(q,t):=\mathbb{E}[K],\qquad m(q,t):=\mathbb{E}[X],\qquad e(q,t):=\frac{m(q,t)}{\mu(q,t)}.
\]
Then the gradient has the closed form
\[
\boxed{\;
\partial_t f(q,t)=\frac{1}{\mu(q,t)},\qquad
\partial_q f(q,t)=e(q,t)=\frac{\mathbb{E}[X]}{\mathbb{E}[K]}\;}
\]
and the Hessian can be explicitly written as
\[
\boxed{\;
\nabla^2 f(q,t)=
\begin{pmatrix}
\dfrac{\mathrm{Var}(K)}{\mu(q,t)^3} & -\dfrac{\mathrm{Cov}(K,\,X-e(q,t)K)}{\mu(q,t)^2}\\[1.2ex]
-\dfrac{\mathrm{Cov}(K,\,X-e(q,t)K)}{\mu(q,t)^2} & \dfrac{\mathrm{Var}(X-e(q,t)K)}{\mu(q,t)}
\end{pmatrix}.}
\]
In particular, since \(k\mapsto \log F_{k+2}\) is not an affine function of \(k\), the covariance matrix is non-degenerate, so \(\nabla^2 f(q,t)\) is positive definite at every point; therefore \((q,t)\mapsto \log\lambda(q,e^t)\) is strictly convex and real-analytic on \((0,\infty)\times\RR\).
\end{proposition}
\begin{proof}[Proof outline]
From the root equation \(e^tA_q(\rho(q,e^t))=1\), implicit differentiation with respect to \(t\) gives
\(
\rho_t=-1/(e^tA_q'(\rho))
\);
then from \(\mu(q,t)=\sum k p_{q,t}(k)=e^t\rho A_q'(\rho)\), one obtains \(\partial_t f=-\rho_t/\rho=1/\mu(q,t)\).
The ratio expression for \(\partial_q f\) follows similarly from implicit differentiation with respect to \(q\) (consistent with the \(t=0\) case in Proposition \ref{prop:pom-multiplicity-lambdaq-derivative-gibbs}).
\par
For the second-order derivatives, note that \(p_{q,t}(k)=\exp(t+qX+k\log\rho)\) and \(\sum_k p_{q,t}(k)\equiv 1\);
from this, for any integrable \(g\), \(\partial_t\mathbb{E}[g]=\mathbb{E}[g(1-K/\mu)]\) and \(\partial_q\mathbb{E}[g]=\mathbb{E}[g(X-eK)]\).
Applying these to \(g=K\) and \(g=X\) and substituting into the chain rule for \(f_t=1/\mu\) and \(f_q=e\) yields the stated Hessian formula.
Positive definiteness is equivalent to the strict positivity of the determinant of the two-dimensional covariance matrix; if \(\mathrm{Var}(X-eK)=0\), then \(X\) would be an almost-sure affine function of \(K\), contradicting the strictly nonlinear growth of Fibonacci numbers. \qed
\end{proof}

\paragraph{Entropy rates and multifractal: closed-form Shannon/R\'enyi hierarchy}
\begin{proposition}[Closed-form Shannon entropy rate for parameter \(q\) and two-endpoint asymptotics]\label{prop:pom-multiplicity-composition-shannon-entropy-rate-generalq}
Let
\(
h(q):=\lim_{L\to\infty}\frac{1}{L}H(P_L^{(q)})
\)
be the Shannon entropy rate (bits/length) of \(P_L^{(q)}\).
Then the limit exists and satisfies the fully closed form
\begin{equation}\label{eq:pom-multiplicity-composition-shannon-rate-closed}
\boxed{\;
h(q)=\log_2\lambda(q)-q\,\frac{d}{dq}\log_2\lambda(q).\;}
\end{equation}
Moreover, \(h(0)=1\), \(\lim_{q\to\infty}h(q)=0\), and \(h\) is real-analytic and strictly decreasing on \((0,\infty)\).
Furthermore, if \(\log_2\lambda(q)=1+\frac{S_1}{2}q+\frac12 c_2 q^2+O(q^3)\) (Proposition \ref{prop:pom-multiplicity-lambdaq-taylor-q0}), then
\[
h(q)=1-\frac{c_2}{2}q^2+O(q^3)\qquad(q\downarrow 0).
\]
If \(\lambda(q)=2^q(1+(3/4)^q+o((3/4)^q))\) (Proposition \ref{prop:pom-multiplicity-lambdaq-large-q-asymptotic}), then
\[
h(q)=\frac{1+q\ln(4/3)}{\ln 2}\Bigl(\frac34\Bigr)^{q}+o\!\Bigl(\Bigl(\frac34\Bigr)^{q}\Bigr)\qquad(q\to\infty).
\]
\end{proposition}
\begin{proof}[Proof outline]
From \(P_L^{(q)}(\boldsymbol{\ell})=w(\boldsymbol{\ell})^q/Z_L^{(q)}(1)\),
\[
H(P_L^{(q)})
=-\sum_{\boldsymbol{\ell}}P_L^{(q)}(\boldsymbol{\ell})\log_2 P_L^{(q)}(\boldsymbol{\ell})
=\log_2 Z_L^{(q)}(1)-q\,\mathbb{E}_{P_L^{(q)}}[\log_2 w(\boldsymbol{\ell})].
\]
Here
\(
\mathbb{E}_{P_L^{(q)}}[\log w]=\partial_q \log Z_L^{(q)}(1)
\)
follows from direct differentiation.
Dividing the above identity by \(L\) and using
\(
L^{-1}\log Z_L^{(q)}(1)\to \log\lambda(q)
\)
together with interchangeable differentiation (guaranteed by analyticity and the simple-pole consistency of \(\log\lambda\)) yields \eqref{eq:pom-multiplicity-composition-shannon-rate-closed}.
The endpoint \(h(0)=1\) follows from the uniform composition \(P_L^{(0)}\) on \(|\mathcal{C}_L|=2^{L-1}\);
\(\lim_{q\to\infty}h(q)=0\) follows from the low-temperature concentration of \(P_L^{(q)}\) on the maximum-weight atom (Proposition \ref{prop:pom-multiplicity-low-temperature-gibbs}).
The two-endpoint asymptotics are obtained by substituting the corresponding expansions of \(\log_2\lambda\). \qed
\end{proof}

\begin{proposition}[R\'enyi entropy rate and self-collision index: fully closed forms via \texorpdfstring{$\lambda(\cdot)$}{lambda} two-point comparison]\label{prop:pom-multiplicity-composition-renyi-collision-generalq}
For any \(\alpha>1\), let
\(
h^{(\alpha)}(q):=\lim_{L\to\infty}\frac{1}{L}H_\alpha(P_L^{(q)})
\)
be the \(\alpha\)-th order R\'enyi entropy rate, where
\(
H_\alpha(P)=\frac{1}{1-\alpha}\log_2\sum_x P(x)^\alpha
\).
Then the limit exists and satisfies the closed form
\begin{equation}\label{eq:pom-multiplicity-composition-renyi-rate-generalq}
\boxed{\;
h^{(\alpha)}(q)=\frac{1}{1-\alpha}\Bigl(\log_2\lambda(\alpha q)-\alpha\log_2\lambda(q)\Bigr).\;}
\end{equation}
In particular, the collision probability
\(
C_L(q):=\sum_{\boldsymbol{\ell}}P_L^{(q)}(\boldsymbol{\ell})^2
\)
has the exponential law
\[
\boxed{\;
-\lim_{L\to\infty}\frac{1}{L}\log_2 C_L(q)=\log_2\frac{\lambda(q)^2}{\lambda(2q)}.\;}
\]
\end{proposition}
\begin{proof}
From the definition,
\(
\sum P_L^{(q)}(\boldsymbol{\ell})^\alpha
=Z_L^{(\alpha q)}(1)/(Z_L^{(q)}(1))^\alpha
\).
Taking logarithms, dividing by \(L\), and using
\(
L^{-1}\log Z_L^{(q)}(1)\to \log\lambda(q)
\)
yields \eqref{eq:pom-multiplicity-composition-renyi-rate-generalq};
setting \(\alpha=2\) gives the collision index. \qed
\end{proof}

\paragraph{Extreme values and path fluctuations: Gumbel and Brownian bridge for general \texorpdfstring{$q$}{q}}
\begin{theorem}[Universal extreme value law for the maximum part length: Gumbel scaling for general \(q\)]\label{thm:pom-multiplicity-composition-maxpart-gumbel-generalq}
Fix \(q\ge 0\) and let \(M_L:=\max_i \ell_i\) be the maximum part length under \(P_L^{(q)}\).
Let
\[
a(q):=\varphi^{q}\rho(q)=\frac{\varphi^{q}}{\lambda(q)}\in(0,1).
\]
Then the renewal distribution \(p_q(k)=F_{k+2}^{\,q}\rho(q)^k\) has a geometric tail: there exists a constant \(c_q>0\) such that
\[
\mathbb{P}(\ell\ge k)\sim c_q\,a(q)^k\qquad(k\to\infty),
\qquad
c_q=\frac{\varphi^{2q}}{5^{q/2}(1-a(q))}.
\]
Define the centering function
\[
m_L(x):=\left\lfloor \frac{\log\!\bigl((L/\mu_q)c_q\bigr)+x}{-\log a(q)}\right\rfloor .
\]
Then the extreme value limit holds:
\[
\lim_{L\to\infty}\mathbb{P}_{P_L^{(q)}}\bigl(M_L\le m_L(x)\bigr)=\exp(-e^{-x}),
\]
hence
\(
M_L=\frac{\log L}{-\log a(q)}+O_{\mathbb{P}}(1)
\), with fluctuations belonging to the Gumbel universality class, and the scaling constant fully determined by \(a(q)=\varphi^{q}/\lambda(q)\).
\end{theorem}
\begin{proof}[Proof outline]
The tail asymptotics follow from the Binet formula \(F_{k+2}=\varphi^{k+2}/\sqrt5\cdot(1+O(\varphi^{-2k}))\), giving
\(
p_q(k)=F_{k+2}^{\,q}\rho(q)^k\sim (\varphi^{2q}/5^{q/2})a(q)^k
\), and summing the geometric series yields \(c_q\).
\par
By Theorem \ref{thm:pom-multiplicity-composition-partcount-generalq-llt},
the typical part count \(R_L\) satisfies \(R_L=L/\mu_q+O_{\mathbb{P}}(\sqrt{L})\);
and by the exponential mixing/renewal representation of Proposition \ref{prop:pom-multiplicity-composition-renewal-hit-prob},
the dependence between distant windows is exponentially small, so the extreme value of \(M_L\) can be aligned with the maximum of \(n\approx L/\mu_q\) i.i.d.\ geometric-tailed samples.
Standard extreme value theory for the latter gives the Gumbel limit under the centering \(m_L(x)\); absorbing the fluctuation of the random sample size into the logarithmic centering term yields the stated limit. \qed
\end{proof}

\begin{theorem}[Path-level invariance principle under hitting: from the renewal process to the Brownian bridge]\label{thm:pom-multiplicity-composition-brown-bridge-invariance}
Fix \(q>0\). Let \((\ell_i)_{i\ge 1}\) be i.i.d.\ renewal increments with \(\ell_i\sim p_q\), and let the partial sums be \(S_n=\sum_{i=1}^n\ell_i\).
Let \(\tau_L=\inf\{n\ge 1:S_n=L\}\).
Under the conditional distribution \(\mathbb{P}(\cdot\mid\tau_L<\infty)\) (which is rigorously equivalent to \(P_L^{(q)}\) by Theorem \ref{thm:pom-multiplicity-composition-exact-conditional-iid}), define the continuous interpolation process
\[
X_L(t):=\frac{1}{\sqrt{L}}\Bigl(S_{\lfloor t\,\tau_L\rfloor}-tL\Bigr),\qquad t\in[0,1].
\]
Then as \(L\to\infty\), the functional limit theorem holds:
\[
X_L(\cdot)\Longrightarrow \sqrt{\frac{\sigma_q^2}{\mu_q}}\,B^\circ(\cdot),
\]
where \(B^\circ\) is a standard Brownian bridge.
\end{theorem}
\begin{proof}[Proof outline]
Applying Donsker's invariance principle to the unconditional random walk \((S_n)_{n\ge 0}\) yields a Brownian motion limit;
then conditioning on the event \(\{S_{\tau_L}=L\}\) (equivalent to \(\{\tau_L<\infty\}\)) converts the path distribution to that of a bridge (via Doob-\(h\) transform), yielding the Brownian bridge limit.
The fluctuation of the random stopping time \(\tau_L\) is controlled by Theorem \ref{thm:pom-multiplicity-composition-partcount-generalq-llt},
so the perturbation from the discrete-time interpolation \(\lfloor t\tau_L\rfloor\) is absorbed into the same tightness framework. \qed
\end{proof}

\paragraph{Exponential mixing rate: fully controlled by the next-nearest root \texorpdfstring{$r(q)$}{r(q)}}
\begin{theorem}[Spectral-gap-determined exponential mixing: boundary decoupling rate for general \(q\)]\label{thm:pom-multiplicity-composition-exponential-mixing-generalq}
Fix \(q\ge 0\) and let \(r(q)>1\) be as in Proposition \ref{prop:pom-multiplicity-composition-renewal-hit-prob} (given by the renewal exponential convergence rate).
Let \(A,B\) be arbitrary cylinder events depending only on the composition within two finite windows separated by at least \(n\) vertices.
Then in the bulk limit \(L\to\infty\), there exists a constant \(C(A,B;q)<\infty\) such that
\[
\boxed{\;
\bigl|\mathbb{P}(A\cap B)-\mathbb{P}(A)\mathbb{P}(B)\bigr|
\le C(A,B;q)\,r(q)^{-n}.\;}
\]
Therefore, the exponential mixing rate of the \(q\)-tilted composition is fully determined by the next-nearest root structure of the analytic equation \(A_q(\rho(q)z)=1\).
\end{theorem}
\begin{proof}[Proof outline]
By Proposition \ref{prop:pom-multiplicity-composition-renewal-hit-prob},
the renewal hitting probability \(u_q(L)\) converges to \(1/\mu_q\) at rate \(r(q)^{-L}\).
For fixed window events \(A,B\), viewing the composition on the gap interval as a connecting segment of the renewal process,
and using ``multiplicative approximation of hitting probabilities + exponential control of the error'' compresses the long-range correlation into an error term dominated by the tail of \(u_q\);
the exponential rate of this error term is precisely \(r(q)^{-n}\).
A formal proof can be given via generating function decomposition or the spectral radius of the solution operator for the renewal equation. \qed
\end{proof}
