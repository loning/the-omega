\section{Appendix D: Internal Reference Correspondence and Reproducibility Index (for Verification of Definitions, Counting, and Experiment Interfaces)}

\begin{quote}
This appendix provides only the localization of formal objects in this paper to internal references, facilitating experiment reproduction and counting verification; the main text is independent of external citation chains.
\end{quote}

\begin{enumerate}
  \item \textbf{Weyl Pair and No Common Eigenvector:} Provides the proposition and proof that when $U_{\mathrm{scan}}V=e^{2\pi i\alpha}VU_{\mathrm{scan}}$ with $\alpha\notin\QQ$, no common eigenvector exists.
  \item \textbf{Hurwitz Extremality and Phase-Locking Resistance Interpretation of Golden Branches:} Worst approximation property of golden slope and its delay effect on phase locking.
  \item \textbf{Sturmian Readout, Marginal Frequencies, and Discrepancy Control:} Sturmian structure of window readout under irrational rotation, limit frequency of symbol $1$, and discrepancy bounds for finite $N$ deviation.
  \item \textbf{$X_m$ Counting, $\varphi$-Stable Projection, and Entropy Rate Limit:} Counting of no-adjacent-$1$'s language $|X_m|=F_{m+2}$, dimension of projection operator $P_\varphi$, and $\log\varphi$ entropy rate.
  \item \textbf{$\Fold_m$ Average Degeneracy Scaling and $\Fold_6$ Explicit Structure Entry:} Quantitative table of $d_m=2^m/F_{m+2}$ and chapter entry for $\Fold_6$ fiber classification.
  \item \textbf{$\Fold_6$: 64→21 and Degenerate Spectrum (2/3/4):} The 64→21 compression of $\Fold_6$, degenerate spectrum, and explicit preimage list.
  \item \textbf{$\pi$-Channel: Cyclic Closure and $18\oplus 3$ Split:} Definition of cyclic admissibility, boundary state set, and counting.
  \item \textbf{$\zeta$ Factorization and Pole Position; $e$-Channel Regularization:} Factorization of $1-z-z^2$, pole locations, and regularization explanation.
  \item \textbf{Hilbert Addressing: Definition of $\Sigma_n$ and $H_n$ and Locality Requirement:} Structure of addressing mapping tick indices to screen points and axiomatization of locality.
  \item \textbf{Discrepancy Certificate Implementation and Experiment Table Interface:} Implementation with golden slope achieving $\log 2$, $\pi$ etc. average experiment table and Koksma certificate interface.
  \item \textbf{Hecke/Prime Skeleton and End-to-End Toy Instance:} Cusp evaluation, $q$-expansion, Hecke verification, and end-to-end example explanation of the encoding chain.
  \item \textbf{Zeckendorf Normalization Algorithm and $\Fold_6$ Computation Workflow:} Greedy Zeckendorf algorithm and implementation steps for $\Fold_6$.
\end{enumerate}
