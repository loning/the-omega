\subsection{The full admissible set \texorpdfstring{$X_6$}{X6}}
The admissible set
\[
X_6=\{w\in\{0,1\}^6:\ \text{$w$ contains no occurrence of $11$}\}

\subsection{Hamming-weight distribution}
Let $|w|_1:=\sum_{i=1}^6 w_i$ be the Hamming weight.
The weight distribution of $X_6$ is
\[
\#\{w\in X_6:|w|_1=k\}=
\begin{cases}
1, & k=0,\\
6, & k=1,\\
10, & k=2,\\
4, & k=3,\\
0, & k\ge 4.
\end{cases}
\]
Indeed, choosing $k$ ones with no adjacency is equivalent to choosing a $k$-subset of $\{1,\dots,6\}$ with no consecutive integers, whose count is $\binom{6-k+1}{k}$.

\subsection{Boundary states and cyclic states}
The cyclic admissibility condition is the wrap-around constraint $w_6w_1\neq 11$.
Thus the boundary set $X_6^{\mathrm{bdry}}$ consists of admissible words with $w_1=w_6=1$:
\[
X_6^{\mathrm{bdry}}=\{\texttt{100001},\ \texttt{100101},\ \texttt{101001}\},
\qquad |X_6^{\mathrm{bdry}}|=3,
\]
and the cyclic admissible set has size $|X_6^{\mathrm{cyc}}|=18$ (Proposition~\ref{prop:boundary_3}).

\subsection{Degeneracy classes under \texorpdfstring{$\mathrm{Fold}_6$}{Fold\_6} on \texorpdfstring{$\{0,\dots,63\}$}{0..63}}
