\section{A modular toolbox: singularities, divergences, and lattice path sums}
\label{sec:modules}

\subsection{Topologizing singularities on the Riemann sphere}
On the extended complex plane $\widehat{\CC}:=\CC\cup\{\infty\}$, the inversion map
$$
\mathcal{I}(z):=\frac{1}{z}
$$
is a biholomorphic automorphism satisfying $\mathcal{I}(0)=\infty$ and $\mathcal{I}(\infty)=0$. Thus ``division by zero'' is replaced by a compactified pole within the object domain, shifting singularities from ``undefined'' to ``boundary points'' of a compactified geometry.

\subsection{Finite-part extraction: an Abel prototype}
Consider the Abel family
$$
S(\varepsilon):=\sum_{n=1}^{\infty} n\,\e^{-n\varepsilon},\qquad \varepsilon>0.
$$
A direct computation yields
$$
S(\varepsilon)=\frac{\e^{-\varepsilon}}{\bigl(1-\e^{-\varepsilon}\bigr)^2}.
$$
As $\varepsilon\to 0^+$,
$$
S(\varepsilon)=\frac{1}{\varepsilon^2}-\frac{1}{12}+O(\varepsilon^2).
$$
Therefore, in the finite-part sense (subtracting the pole terms and taking the limit),
$$
\FP_{\varepsilon\to 0^+} \sum_{n=1}^{\infty} n\,\e^{-n\varepsilon}=-\frac{1}{12}.
$$
This Abel-type finite-part extraction (and the appearance of $-1/12$ as the constant term) is classical; see e.g.~\cite{Hardy1949}.
More generally, any divergent object that admits a controlled Abel asymptotic can be treated within the orbit finite-part template of Section~\ref{sec:cyclic_calculus}.

\subsection{Lattice path sums over Zeckendorf-coded configuration spaces}
Suppose a class of configurations is coded by Zeckendorf bitstrings $(c_k)$ subject to the local constraint $c_k c_{k+1}=0$. Then a formally ill-posed continuum path integral may be replaced by a controlled sum over a discrete, locally constrained code space:
$$
\int \mathcal{D}[\mathrm{path}]\,\e^{\iu S[\mathrm{path}]}
\quad\longrightarrow\quad
\sum_{(c_k)\in\mathcal{C}} W(c)\,\e^{\iu S(c)},
$$
where $\mathcal{C}$ is the set of legal bitstrings and $W(c)$ is a weight induced by the projection density or an application-specific measure.
Because the constraint is local, such sums admit a transfer-matrix organization; in particular, the periodic and primitive-event statistics of the underlying Zeckendorf shift are encoded by the trace and zeta formulas of Subsection~\ref{subsec:zeckendorf_primitive_trace}.

\subsection{Prime--spectrum module: an adelic completion of the scan}
The Zeckendorf shift provides a finite-state, locally constrained readout grammar. However, finite-state zeta functions are rational (cf.~Proposition~\ref{prop:zeckendorf_zeta}) and cannot support the analytic complexity of the completed Riemann zeta function and its infinite set of nontrivial zeros. A concrete way to couple multiplicative irreducibility to dynamical primitives, while retaining a canonical trace/regularization mechanism, is to enlarge the scan from the real place to an adelic space that simultaneously carries the real coordinate and all $p$-adic coordinates.

\begin{definition}[Adeles and the adele class space]
Let $\mathbb{A}_{\QQ}$ be the ring of adeles of $\QQ$, and $\mathbb{A}_{\QQ}^\times$ its group of ideles. The multiplicative group $\QQ^\times$ embeds diagonally in $\mathbb{A}_{\QQ}^\times$. The \emph{adele class space} is the quotient
$$
X_{\QQ}:=\mathbb{A}_{\QQ}/\QQ^\times,
$$
understood as a noncommutative quotient in the sense of measured groupoids/$C^*$-algebras; see e.g.~\cite{Connes1999,Connes1994}.
The multiplicative group $\RR_{>0}$ acts on $\mathbb{A}_{\QQ}$ by scaling in the real component (and trivially on the $p$-adic components), inducing an $\RR_{>0}$-action on $X_{\QQ}$ (the ``scaling flow'').
\end{definition}

\begin{proposition}[Primes as primitive periodic orbits of the scaling flow]
\label{prop:adelic_prime_orbits}
In the adelic module, the scaling flow on $X_{\QQ}$ has periodic orbits with periods $\log p$ indexed by rational primes $p\in\PP$, and the corresponding prime powers arise from iterates of these periodic trajectories. In particular, there is a canonical internal prime predicate
$$
P(\mathcal{O})\quad\Longleftrightarrow\quad \mathcal{O}\ \text{is a primitive periodic orbit of the scaling flow},
$$
whose truth value does not require an external primality test on integers.
\end{proposition}

\begin{proof}
This is part of the periodic-orbit analysis of the scaling action on the adele class space and its interpretation of the explicit formulas as a trace formula; see~\cite{Connes1999}.
\end{proof}

\begin{corollary}[A prime--primitive orbit product and the Euler factorization]
\label{cor:prime_orbit_product}
Let $\mathcal{O}_p$ denote the primitive periodic orbit associated to a prime $p$ and set its orbit length to be
$$
L(\mathcal{O}_p):=\log p.
$$
Then the formal dynamical orbit product
$$
Z(s):=\prod_{\mathcal{O}\ \mathrm{primitive}}\frac{1}{1-\exp(-sL(\mathcal{O}))}
$$
reduces to the Euler product
$$
Z(s)=\prod_{p\in\PP}\frac{1}{1-p^{-s}}=\zeta(s)
$$
in the half-plane $\Re(s)>1$, and one obtains the von Mangoldt expansion
$$
-\frac{Z'(s)}{Z(s)}=\sum_{n\ge 1}\frac{\Lambda(n)}{n^s},
$$
where $\Lambda(p^k)=\log p$ and $\Lambda(n)=0$ otherwise.
\end{corollary}

\begin{proof}
The first identity is the standard Euler factorization of $\zeta(s)$, with the identification $\exp(-sL(\mathcal{O}_p))=p^{-s}$. Differentiating $\log Z(s)=-\sum_p\log(1-p^{-s})$ and expanding $-\log(1-x)=\sum_{k\ge 1}x^k/k$ yields the Dirichlet series with von Mangoldt weights.
\end{proof}

\begin{remark}[Completion and the functional equation as a duality constraint]
The completed zeta function
$$
\xi(s):=\pi^{-s/2}\Gamma\!\left(\frac{s}{2}\right)\zeta(s)
$$
satisfies $\xi(s)=\xi(1-s)$. In an adelic formulation this symmetry is not an external trick but a duality constraint: it follows from self-duality of $\mathbb{A}_{\QQ}$ under Fourier transform together with Poisson summation (Tate's thesis viewpoint); see e.g.~\cite{TateThesis1967}.
\end{remark}

\begin{remark}[An operator-algebraic realization of the Euler product]
Beyond orbit products, there is a $C^*$-dynamical system (the Bost--Connes system) whose equilibrium partition function is $\zeta(\beta)$ and whose structure factorizes over primes in a canonical way. This provides a complementary realization of the principle that composites should not be new primitive generators: the prime degrees of freedom are local factors, while composite contributions arise from tensor-product combinations. See~\cite{BostConnes1995}.
\end{remark}

\begin{remark}[Positivity and a Hilbert--P\'olya-type interface]
The Euler product organizes the \emph{prime end} of the explicit formula, while the zero set organizes the \emph{spectral end}. In noncommutative geometry the explicit formulas can be expressed as a trace formula for the scaling action on $X_{\QQ}$, and the Riemann hypothesis can be reformulated as a positivity statement for a canonical trace pairing (spectral realization as an ``absorption spectrum''); see~\cite{Connes1999}. In the present toolkit, the role of the regulator is played by the orbit finite-part machinery of Section~\ref{sec:cyclic_calculus}, in particular Subsection~\ref{subsec:fp_determinants}; the adelic module supplies a canonical prime--orbit length assignment $L(\mathcal{O}_p)=\log p$ and hence a natural target for a self-adjoint spectral operator.
\end{remark}
