\section{二值读出：机械词与 Sturmian 复杂度不变量}

\subsection{二分割读出与符号化}

\begin{definition}[二分割读出]\label{def:binary-partition-readout}
取 $\TT^1$ 的可测分割
\[
\mathcal{P}=\{I_0,I_1\},\qquad I_0\cap I_1=\varnothing,\ I_0\cup I_1=\TT^1,
\]
并定义二值读出序列
\[
w_t:=\ind_{I_1}(\theta_t)\in\{0,1\}.
\]
称 $w=(w_t)_{t\ge 0}$ 为由 $(\alpha,\mathcal{P},\theta_0)$ 诱导的机械词。
\end{definition}

\subsection{Sturmian 性质与最小复杂度}

\begin{theorem}[Sturmian 最小复杂度]\label{thm:sturmian-min-complexity}
若 $\alpha$ 为无理数且 $\mathcal{P}$ 为两区间分割（端点不落在轨道的退化情形除外），则机械词 $w$ 为 Sturmian 序列；其因子复杂度满足
\[
p(n)=n+1,\qquad \forall n\ge 1,
\]
其中 $p(n)$ 为长度为 $n$ 的不同子词数量。
\end{theorem}

\begin{remark}
证明要点见附录 \ref{app:sturmian-outline}。核心为：无理旋转的符号化给出平衡序列；两字母非周期且复杂度最小等价于 Sturmian。
\end{remark}

\subsection{黄金分支的实现（可选模型）}

\begin{assumption}[黄金斜率选择]\label{assump:golden-slope}
取 $\alpha$ 的连分数部分商均为 $1$（等价于 $\alpha=\varph^{-1}$ 或其线性分式变换像），则诱导的 Sturmian 序列为 Fibonacci 型特例；该选择具有极小近似系数意义下的反共振性质，并在后续 Zeckendorf 稳定化中给出最简的二值规范形结构。
\end{assumption}
