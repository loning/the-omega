\subsubsection{Trace-class realization of complex-weight atomic Euler products: a constructive reverse section from Euler factors to cyclic-block Fredholm determinants}\label{subsec:cyclic-traceclass-lift-complex}
The preceding \S\ref{subsec:cyclic-traceclass-lift} provided the cyclic-block trace-class lift "atomic spectral data \((\beta_j,n_j,m_j)\mapsto L_{\mathcal D}\)" under the positivity condition \(\beta_j\in(0,1)\).
That construction does not depend on the real positivity of \(\beta_j\): in the complex-weight setting, it suffices to control the countable direct sum in the trace-norm sense in order to obtain an equally strict reverse section, thereby lifting parameter families of atomic Euler products to directly compilable trace-class operator certificates.

\begin{corollary}[Reverse section from Euler products to trace-class Fredholm \texorpdfstring{$\zeta$}{zeta}: constructive realization of arbitrary atomic Euler products (complex-weight version)]\label{cor:cyclic-euler-product-complex-section}
Suppose a countable data family \(\mathcal D=\{(\beta_j,n_j,m_j)\}_{j\in J}\subset \CC\times \NN_{\ge 1}\times \NN\) is given, satisfying
\begin{equation}\label{eq:tc-executability-complex}
\sum_{j\in J} m_j\,n_j\,|\beta_j|^{1/n_j}<\infty,
\qquad
|\beta_j|<1\ \ (j\in J).
\end{equation}
For each \(j\in J\) choose an \(n_j\)-th root \(\alpha_j\in\CC\) with \(\alpha_j^{n_j}=\beta_j\) (so that \(|\alpha_j|=|\beta_j|^{1/n_j}\)).
Set
\[
H_{\mathcal D}:=\bigoplus_{j\in J}\bigl(\CC^{n_j}\otimes \CC^{m_j}\bigr),
\qquad
L_{\mathcal D}:=\bigoplus_{j\in J}\bigl(C(n_j,\alpha_j)\otimes I_{m_j}\bigr)\in \mathcal{S}_1(H_{\mathcal D}),
\]
where \(C(n,\alpha)=\alpha\Pi_n\) is the cyclic block of Definition \ref{def:cyclic-block}.
Define
\[
\zeta_{\mathcal D}(r):=\det(I-rL_{\mathcal D})^{-1}.
\]
Then on \(|r|<1\) the following strict identity holds:
\[
\boxed{\;
\zeta_{\mathcal D}(r)=\prod_{j\in J}\bigl(1-\beta_j r^{n_j}\bigr)^{-m_j}\; }.
\]
\end{corollary}

\begin{proof}
From \eqref{eq:tc-executability-complex} and \(|\alpha_j|=|\beta_j|^{1/n_j}\) one obtains
\(
\sum_{j\in J}m_jn_j|\alpha_j|<\infty
\), hence \(L_{\mathcal D}\in\mathcal{S}_1(H_{\mathcal D})\).
Moreover, \(|\alpha_j|<1\) together with \(\alpha_j\to 0\) (from summability) implies \(\|L_{\mathcal D}\|=\sup_j|\alpha_j|<1\), so that \((I-rL_{\mathcal D})\) is invertible on \(|r|<1\).
\par
For each cyclic block, Proposition \ref{prop:cycle-permutation-determinant} gives
\[
\det\bigl(I-rC(n_j,\alpha_j)\bigr)=1-(\alpha_j r)^{n_j}=1-\beta_j r^{n_j}.
\]
By the multiplicativity of Fredholm determinants over direct sums and tensor products,
\[
\det(I-rL_{\mathcal D})
=\prod_{j\in J}\det\bigl(I-r(C(n_j,\alpha_j)\otimes I_{m_j})\bigr)
=\prod_{j\in J}\bigl(1-\beta_j r^{n_j}\bigr)^{m_j}.
\]
Taking reciprocals on both sides yields the conclusion. \qed
\end{proof}
