\paragraph{总圆维预算、素数秩障碍与无穷对象的圆维剖面}
本段在定义 \ref{def:circle-dimension}--\ref{def:half-circle-dimension} 与命题 \ref{prop:circle-dimension-laws} 的口径下，将实现层的相位--账本编码预算与结构层的秩障碍统一为可审计下界。有限纤维标签只进入残差寄存器，不计入结构圆维增量；对应的最小化问题由规模指数、核秩与共终剖面共同控制。

\begin{theorem}[总圆维预算下界与相位--账本守恒律]\label{thm:xi-total-cdim-budget-phase-ledger-conservation}
令 \(k\in\mathbb N\) 且
\[
B_N:=\{-N,\dots,N\}^k\subset\mathbb Z^k,\qquad |B_N|=(2N+1)^k.
\]
设存在常数 \(C_{\mathcal L}>0\) 及编码族
\[
\mathrm{Enc}_N:B_N\hookrightarrow \mathcal A_N\times \mathcal L_N
\]
满足：
\begin{enumerate}
  \item \(\mathcal A_N\) 由 \(p\) 个相位通道组成，且每通道使用至多 \(b(N)\) 比特，从而
  \[
  |\mathcal A_N|\le 2^{p\,b(N)};
  \]
  \item \(\mathcal L_N\) 为 \(m\) 个计数型账本寄存器的截断状态集合，且
  \[
  |\mathcal L_N|\le C_{\mathcal L}(N+1)^m;
  \]
  \item
  \(
  b(N)=\alpha\log_2N+o(\log N)
  \)
  （\(N\to\infty\)）对某个 \(\alpha\ge 0\) 成立。
\end{enumerate}
则必有
\begin{equation}\label{eq:xi-total-cdim-budget-general}
p\alpha+m\ge k.
\end{equation}
特别地，当 \(\alpha=1\) 时，
\begin{equation}\label{eq:xi-total-cdim-budget-alpha-one}
p+m\ge k.
\end{equation}
定义总圆维预算
\[
\mathsf B(p,m):=\cdim(\mathbb Z^p)+2\cdim^+(\mathbb N^m)=p+m,
\]
则在 \(\alpha=1\) 的审计制度下有
\[
\mathsf B(p,m)\ge k.
\]
并且对任意分解 \(k=p+m\) 该下界可达到。
\end{theorem}
\begin{proof}
由单射性，
\[
(2N+1)^k\le |\mathcal A_N|\,|\mathcal L_N|
\le 2^{p\,b(N)}\,C_{\mathcal L}(N+1)^m.
\]
取 \(\log_2\)：
\[
k\log_2(2N+1)
\le p\,b(N)+m\log_2(N+1)+\log_2C_{\mathcal L}.
\]
除以 \(\log_2N\) 并令 \(N\to\infty\)，使用
\[
\frac{\log_2(2N+1)}{\log_2N}\to 1,\qquad
\frac{\log_2(N+1)}{\log_2N}\to 1,\qquad
\frac{\log_2C_{\mathcal L}}{\log_2N}\to 0,
\]
以及 \(b(N)=\alpha\log_2N+o(\log N)\)，得到
\eqref{eq:xi-total-cdim-budget-general}。
\eqref{eq:xi-total-cdim-budget-alpha-one}
为 \(\alpha=1\) 的直接特例。

对可达性，给定任意 \(k=p+m\)，定义
\[
\mathrm{Enc}_N(x_1,\dots,x_k)
=\Big((x_1,\dots,x_p),\,(x_{p+1}+N,\dots,x_k+N)\Big).
\]
相位部分可用每坐标 \(b(N)=\lceil\log_2(2N+1)\rceil\) 比特编码；账本部分落在 \(\{0,\dots,2N\}^m\)，因此
\[
|\mathcal L_N|=(2N+1)^m\le 2^m(N+1)^m
\]
（取 \(C_{\mathcal L}=2^m\)）。
此时 \(b(N)=\log_2N+O(1)\)，故 \(\alpha=1\)，且 \(p+m=k\) 取等号。证毕。
\end{proof}

\begin{corollary}[基本数系的实现层圆维签名极小化]\label{cor:xi-basic-number-systems-cdim-signature-minimization}
在定理 \ref{thm:xi-total-cdim-budget-phase-ledger-conservation} 的 \(\alpha=1\) 口径下，
对具有有限近似族 \(S_N\) 的对象 \(S\)，定义实现层签名集合
\[
\Sigma(S):=
\left\{
\left(p,\frac m2\right):
\exists\,\mathrm{Enc}_N:S_N\hookrightarrow \mathcal A_N\times \mathcal L_N
\ \text{满足定理条件并对充分大 }N\text{ 成立}
\right\}.
\]
则有以下极小预算：
\begin{enumerate}
  \item \(S=\mathbb N\)（\(S_N=\{0,\dots,N\}\)）：
  \[
  \min\left\{p+2\ell:\ (p,\ell)\in\Sigma(\mathbb N)\right\}=1,
  \]
  且可取 \((p,\ell)=(0,\tfrac12)\)；
  \item \(S=\mathbb Z\)（\(S_N=\{-N,\dots,N\}\)）：
  \[
  \min\left\{p+2\ell:\ (p,\ell)\in\Sigma(\mathbb Z)\right\}=1,
  \]
  且可取 \((p,\ell)=(1,0)\)；
  \item \(S=\mathbb Q\)，取
  \[
  S_N:=\left\{\frac ab:\ |a|\le N,\ 1\le b\le N,\ \gcd(a,b)=1\right\},
  \]
  则 \(|S_N|\asymp N^2\)，并有
  \[
  \min\left\{p+2\ell:\ (p,\ell)\in\Sigma(\mathbb Q)\right\}=2,
  \]
  且可取 \((p,\ell)=(1,\tfrac12)\)；
  \item \(S=\mathbb Z^k\)（\(S_N=B_N\)）：
  \[
  \left(p,\frac{k-p}{2}\right)\in\Sigma(\mathbb Z^k),\qquad p=0,1,\dots,k,
  \]
  且全部满足 \(p+2\cdot\frac{k-p}{2}=k\)。
\end{enumerate}
\end{corollary}
\begin{proof}
先给统一下界：在 \(\alpha=1\) 下，
\[
|\mathcal A_N|\,|\mathcal L_N|
\le
2^{p(\log_2N+o(\log N))}\,C_{\mathcal L}(N+1)^m
=N^{p+m+o(1)}.
\]
因此凡满足 \(|S_N|\ge cN^{d}\)（某 \(c>0\)）的对象，都必须有 \(p+m\ge d\)。
对 \(\mathbb N,\mathbb Z,\mathbb Q,\mathbb Z^k\) 分别取 \(d=1,1,2,k\) 即得下界。

对 \(\mathbb Q\) 的下界，由 \(|S_N|\asymp N^2\) 与
\[
|S_N|\le |\mathcal A_N|\,|\mathcal L_N|
\le 2^{p\,b(N)}C_{\mathcal L}(N+1)^m
=N^{p+m+o(1)}
\]
得 \(p+m\ge 2\)。
对可达性，取既约表示 \(\frac ab\)（\(b>0\)）的唯一参数对 \((a,b)\)，令相位通道编码 \(a\in[-N,N]\)，账本通道编码 \(b\in[1,N]\)，得 \((p,m)=(1,1)\)，即 \((p,\ell)=(1,\tfrac12)\)。
\(\mathbb N,\mathbb Z,\mathbb Z^k\) 的可达性分别由自然编码、整数单通道编码与定理 \ref{thm:xi-total-cdim-budget-phase-ledger-conservation} 的等号构造给出。
于是最小总预算如述。证毕。
\end{proof}

\begin{theorem}[有界秩但非有限生成对象的半圆维深度轴下界]\label{thm:xi-nonfinitely-generated-bounded-rank-halfaxis}
设 \(G\) 为可数无挠交换群，且
\[
\operatorname{rank}_{\mathbb Q}(G):=
\dim_{\mathbb Q}(G\otimes_{\mathbb Z}\mathbb Q)=k<\infty.
\]
若 \(G\) 非有限生成，则任何满足下述扩展保存分解的数据
\[
\Phi:P\times M\to G
\]
都强迫
\[
\cdim^+(M)\ge \frac12.
\]
其中：
\begin{enumerate}
  \item \(P\) 为有限生成交换群，\(\cdim(P)=k\)；
  \item \(M\) 为有限生成可消去交换幺半群；
  \item 对每个 \(m\in M\)，切片 \(\Phi(\cdot,m):P\to G\) 为群同态；
  \item \(\Phi\) 满射，且每个纤维 \(\Phi^{-1}(g)\) 有限。
\end{enumerate}
\end{theorem}
\begin{proof}
先证 \(M\) 必无限。
反设 \(M\) 有限，记 \(M=\{m_1,\dots,m_r\}\)。
则
\[
G=\Phi(P\times M)=\bigcup_{i=1}^{r}\Phi(P,m_i).
\]
每个 \(\Phi(P,m_i)\) 是 \(G\) 的有限生成子群（\(P\) 有限生成且切片为同态像）。
取各子群有限生成集并并，得到 \(G\) 的有限生成集，故 \(G\) 有限生成，矛盾。
因此 \(M\) 无限。

又 \(M\) 为有限生成可消去交换幺半群，故其群完成 \(G(M)\) 为有限生成交换群。
若 \(\cdim(G(M))=0\)，则 \(G(M)\) 有限。
由于 \(M\subseteq G(M)\)，遂 \(M\) 也有限，与上步矛盾。
故 \(\cdim(G(M))\ge 1\)。
按定义 \(\cdim^+(M)=\frac12\cdim(G(M))\)，结论成立。证毕。
\end{proof}

\begin{corollary}[对 \texorpdfstring{$(\mathbb Q,+)$}{(Q,+)} 的直接约束]\label{cor:xi-additive-q-halfaxis-lower-bound}
设 \(G=(\mathbb Q,+)\)。
在定理 \ref{thm:xi-nonfinitely-generated-bounded-rank-halfaxis} 的扩展保存框架下，任意可审计分解都满足
\[
\cdim^+(M)\ge \frac12.
\]
因此实现层至少需要一条半圆维深度轴。
\end{corollary}
\begin{proof}
\((\mathbb Q,+)\) 为可数无挠交换群，\(\operatorname{rank}_{\mathbb Q}(\mathbb Q)=1\)，且非有限生成。直接套用定理
\ref{thm:xi-nonfinitely-generated-bounded-rank-halfaxis}
即得结论。证毕。
\end{proof}

\begin{theorem}[素数寄存器的不可替换性与半圆维下界]\label{thm:xi-prime-register-irreplaceability-halfcdim-lower-bound}
令 \(S\) 为有限素数集，\(|S|=n\)，并记
\[
\mathbb Q_S^{\times}:=
\left\{\prod_{p\in S}p^{e_p}:e_p\in\mathbb Z\right\}
\cong \mathbb Z^n.
\]
设 \(P\) 为有限生成交换群且 \(\cdim(P)=d\)。
则：
\begin{enumerate}
  \item 对任意群同态 \(\varphi:\mathbb Q_S^{\times}\to P\)，有
  \[
  \cdim(\ker\varphi)\ge n-d.
  \]
  \item 若存在有限生成可消去交换幺半群 \(M\) 及群同态
  \[
  \psi:\mathbb Q_S^{\times}\to G(M)
  \]
  使得 \((\varphi,\psi):\mathbb Q_S^{\times}\to P\times G(M)\) 为单射，则
  \[
  \cdim^+(M)\ge \frac{n-d}{2}.
  \]
\end{enumerate}
特别地，若 \(S=\{p\le B\}\)（\(n=\pi(B)\)），则任意固定 \(d\) 的可见相位系统都满足
\[
\cdim^+(M)\ge \frac{\pi(B)-d}{2}.
\]
\end{theorem}
\begin{proof}
由 \(\mathbb Q_S^{\times}\cong\mathbb Z^n\)，其自由秩为 \(n\)。
同态 \(\varphi\) 的像自由秩至多 \(d\)，故
\[
\operatorname{rank}(\ker\varphi)
=n-\operatorname{rank}(\operatorname{im}\varphi)\ge n-d.
\]
对有限生成交换群，\(\cdim\) 等于自由秩，得第 (1) 条。

对第 (2) 条，\((\varphi,\psi)\) 单射意味着
\[
n
=\operatorname{rank}\bigl(\mathbb Q_S^\times\bigr)
=\operatorname{rank}\bigl(\operatorname{im}(\varphi,\psi)\bigr)
\le \operatorname{rank}(P)+\operatorname{rank}(G(M))
=d+\cdim(G(M)).
\]
因此 \(\cdim(G(M))\ge n-d\)，即
\[
\cdim^+(M)=\frac12\cdim(G(M))\ge \frac{n-d}{2}.
\]
最后把 \(n\) 替换为 \(\pi(B)\) 即得截断素数预算式。证毕。
\end{proof}

\begin{theorem}[无穷对象的圆维剖面与共终不变性]\label{thm:xi-circle-dimension-profile-cofinal}
设 \(G\) 为可数交换群。
定义其圆维剖面为
\[
\mathcal P(G):=\bigl(\mathcal F(G),\le,\cdim\bigr),
\]
其中 \(\mathcal F(G)\) 为所有有限生成子群构成的偏序集（按包含关系）。
则：
\begin{enumerate}
  \item \(\mathcal F(G)\) 是有向集：任意 \(H_1,H_2\in\mathcal F(G)\)，存在 \(H_3\in\mathcal F(G)\) 使 \(H_1,H_2\le H_3\)；
  \item 若 \((H_i)_{i\in I}\)、\((K_j)_{j\in J}\) 为两条穷尽链（\(\bigcup_iH_i=G=\bigcup_jK_j\)），则
  \[
  L_{(i,j)}:=H_i+K_j
  \]
  给出同时共终于两者的共同细化链；因此“圆维随深度增长的剖面”在共终意义下良定义；
  \item 例：
  \begin{enumerate}
    \item 对 \((\mathbb Q,+)\)，任意非平凡有限生成子群 \(H\) 皆同构于 \(\mathbb Z\)，故 \(\cdim(H)=1\)；
    \item 对 \(\mathbb Q_{>0}^{\times}\)，剖面无界：可取
    \(
    H_n:=\langle 2,3,\dots,p_n\rangle\cong\mathbb Z^n
    \)
    使 \(\cdim(H_n)=n\to\infty\)。
  \end{enumerate}
\end{enumerate}
\end{theorem}
\begin{proof}
对 (1)，取 \(H_3:=H_1+H_2\)。
有限生成子群之和仍有限生成，且显然包含 \(H_1,H_2\)。

对 (2)，定义 \(L_{(i,j)}:=H_i+K_j\)。
每个 \(L_{(i,j)}\) 有限生成，且
\[
H_i\le L_{(i,j)},\qquad K_j\le L_{(i,j)}.
\]
又
\[
\bigcup_{i,j}L_{(i,j)}
=\left(\bigcup_iH_i\right)+\left(\bigcup_jK_j\right)=G.
\]
故 \((L_{(i,j)})\) 同时共终于两条链，得到共终不变性。

对 (3a)，设 \(H=\langle q_1,\dots,q_r\rangle\le (\mathbb Q,+)\)。
令 \(D\) 为所有 \(q_\ell\) 的公分母，则 \(H\subseteq \frac1D\mathbb Z\cong\mathbb Z\)，从而 \(H\) 循环，非平凡时 \(\cdim(H)=1\)。

对 (3b)，\(\mathbb Q_{>0}^{\times}\) 的有限生成子群由有限素数支撑决定，且同构于有限秩自由阿贝尔群。
取前 \(n\) 个素数生成子群即得 \(\mathbb Z^n\) 子群，故剖面无界。证毕。
\end{proof}

\paragraph{Lee--Yang 判别式证书与对数坐标下的数论分层}
沿用 \(2026\)-\(02\)-\(22\) 文档的谱核记号，记 \(\Pi(\lambda,y)\in\overline{\mathbb Q}[\lambda,y]\) 为主谱核的最小多项式，并满足
\[
\mathrm{Disc}_{\lambda}\bigl(\Pi(\lambda,y)\bigr)=-y(y-1)P_{\mathrm{LY}}(y).
\]
在该可编译判别式口径下，分歧值、诱导奇点与零点凝聚曲线均由显式代数对象给出；进一步在 \(s=\log y\) 坐标可得到代数量与对数超越量之间的严格分层。

\begin{theorem}[简单特征值解析分支的代数喷射封闭性]\label{thm:xi-leyang-simple-root-algebraic-jet-closure}
设 \(K\subset\mathbb C\) 为数域，\(M(t)\in\mathrm{Mat}_{n}(K[t])\)，并记
\[
\chi(\lambda,t):=\det(\lambda I_n-M(t))\in K[\lambda,t].
\]
若存在 \((\lambda_0,t_0)\in\overline K\times K\) 使
\[
\chi(\lambda_0,t_0)=0,\qquad
\partial_{\lambda}\chi(\lambda_0,t_0)\neq 0,
\]
则在 \(t_0\) 的某邻域内存在唯一解析函数 \(\lambda(t)\) 满足
\[
\lambda(t_0)=\lambda_0,\qquad
\chi(\lambda(t),t)\equiv 0.
\]
并且对任意整数 \(m\ge 0\)，皆有
\[
\lambda^{(m)}(t_0)\in K(\lambda_0)\subset\overline K.
\]
\end{theorem}
\begin{proof}
由隐函数定理，\(\partial_{\lambda}\chi(\lambda_0,t_0)\neq0\) 蕴含上述解析支存在且唯一。
对恒等式 \(\chi(\lambda(t),t)\equiv0\) 求导并在 \(t=t_0\) 处代入，得
\[
\chi_{\lambda}(\lambda_0,t_0)\lambda'(t_0)+\chi_t(\lambda_0,t_0)=0,
\]
从而
\[
\lambda'(t_0)
=-\frac{\chi_t(\lambda_0,t_0)}{\chi_{\lambda}(\lambda_0,t_0)}
\in K(\lambda_0).
\]
对 \(m\ge2\) 重复求导，\(\lambda^{(m)}(t_0)\) 的线性系数始终为 \(\chi_{\lambda}(\lambda_0,t_0)\neq0\)，右端仅含 \(\chi\) 在 \((\lambda_0,t_0)\) 的偏导值与低阶导数 \(\lambda^{(j)}(t_0)\)（\(j<m\)）的有理组合。
以 \(m\) 归纳即得 \(\lambda^{(m)}(t_0)\in K(\lambda_0)\)。证毕。
\end{proof}

\begin{corollary}[有限核热力学的导数代数性与熵项超越性]\label{cor:xi-leyang-finite-kernel-algebraic-vs-transcendental-layering}
设 \(\lambda_+(y)\) 为某有限核转移矩阵的主特征值解析支，且满足代数方程 \(\Pi(\lambda_+(y),y)=0\)。
若在 \(y=1\) 处
\[
\partial_{\lambda}\Pi\!\bigl(\lambda_+(1),1\bigr)\neq0,
\]
并定义
\[
\psi(s):=\log\lambda_+(e^s)-\log2,
\]
则对任意 \(m\ge1\)，有 \(\psi^{(m)}(0)\in\overline{\mathbb Q}\)。
此外，若
\[
\frac{\lambda_+(1)}{2}\in\overline{\mathbb Q}_{>0}\setminus\{1\},
\]
则
\[
\psi(0)=\log\!\frac{\lambda_+(1)}{2}
\]
为超越数。
特别地，记
\[
\kappa_{\infty}:=\frac{6\sqrt5-5}{250},
\qquad
c_1:=\log\!\frac{2}{\varphi},
\qquad
\varphi:=\frac{1+\sqrt5}{2},
\]
则 \(\kappa_{\infty}\in\overline{\mathbb Q}\)、\(c_1\) 为超越数，且对任意 \(r\in\overline{\mathbb Q}^{\times}\) 均有
\[
\kappa_{\infty}\neq r\,c_1^{2},
\qquad
\frac{\kappa_{\infty}}{c_1^{2}}\ \text{为超越数}.
\]
\end{corollary}
\begin{proof}
把定理 \ref{thm:xi-leyang-simple-root-algebraic-jet-closure} 应用于 \(\chi(\lambda,y)=\Pi(\lambda,y)\) 在 \(y=1\) 的简单根分支，得 \(\lambda_+^{(m)}(1)\in\overline{\mathbb Q}\)（\(m\ge0\)）。
与 \(y=e^s\) 复合后，\(\lambda_+(e^s)\) 在 \(s=0\) 的各阶导数仍由代数数经有理运算给出，故 \(\psi^{(m)}(0)\in\overline{\mathbb Q}\)（\(m\ge1\)）。
若 \(\lambda_+(1)/2\in\overline{\mathbb Q}_{>0}\setminus\{1\}\)，则 \(\log(\lambda_+(1)/2)\) 由 Lindemann--Weierstrass 定理为超越数 \cite{Baker1975TranscendentalNumberTheory}。
又 \(\kappa_{\infty}\) 显然代数，而 \(2/\varphi\in\overline{\mathbb Q}_{>0}\setminus\{1\}\) 使 \(c_1\) 超越，故 \(c_1^2\) 亦超越。
于是 \(r c_1^2\)（\(r\in\overline{\mathbb Q}^{\times}\)）超越，不可能等于代数数 \(\kappa_{\infty}\)。
若 \(\kappa_{\infty}/c_1^2\) 代数，则 \(c_1^2=\kappa_{\infty}/(\kappa_{\infty}/c_1^2)\) 代数，矛盾。证毕。
\end{proof}

设 \(y_{\star}\in\overline{\mathbb Q}\setminus\{0,1\}\) 为 \(\lambda_+(y)\) 的任一分歧值（等价地，\(\mathrm{Disc}_{\lambda}(\Pi)\) 的零点之一）。

处，任一

\end{proof}

\begin{proposition}[Lee--Yang 振荡角频率与 \texorpdfstring{$\pi$}{pi} 的不可公度性]\label{prop:xi-leyang-angle-over-pi-irrational}
令
\[
y_{\mathrm c}^{+}:=\frac{3+i\sqrt{111}}{96},
\qquad
\theta:=\arg(y_{\mathrm c}^{+})\in(0,\pi).
\]
则
\[
\frac{\theta}{\pi}\notin\mathbb Q.
\]
\end{proposition}
\begin{proof}
反设 \(\theta/\pi\in\mathbb Q\)。
则 \(\zeta:=e^{i\theta}\) 为单位根，且
\[
\zeta=\frac{y_{\mathrm c}^{+}}{|y_{\mathrm c}^{+}|}
=\frac{3+i\sqrt{111}}{2\sqrt{30}}
\in F:=\mathbb Q(\sqrt{-111},\sqrt{30}).
\]
域 \(F/\mathbb Q\) 为双二次扩张，\(\mathrm{Gal}(F/\mathbb Q)\cong V_4\)。
因此若 \(\zeta\) 的阶为 \(n\)，则 \(\mathbb Q(\zeta_n)\subseteq F\)，其伽罗瓦群 \((\mathbb Z/n\mathbb Z)^{\times}\) 必是 \(V_4\) 的子商，故指数至多为 \(2\) 且次数 \(\varphi(n)\le4\)。
可得 \(n\in\{1,2,3,4,6,8,12\}\)。
然而 \(F\) 的三个二次子域仅为
\[
\mathbb Q(\sqrt{-111}),\qquad
\mathbb Q(\sqrt{30}),\qquad
\mathbb Q(\sqrt{-3330}),
\]
其中不含 \(\mathbb Q(i)\) 与 \(\mathbb Q(\sqrt{-3})\)。
故 \(n\notin\{3,4,6,8,12\}\)，只能 \(n\in\{1,2\}\)，即 \(\zeta=\pm1\)。
但 \(y_{\mathrm c}^{+}\) 非实，故 \(\zeta\neq\pm1\)，矛盾。
因此 \(\theta/\pi\notin\mathbb Q\)。证毕。
\end{proof}

\begin{theorem}[平方耦合关系的代数--超越障碍]\label{thm:xi-leyang-kappa-vs-c1-square-obstruction}
令
\[
\kappa_{\infty}=\frac{6\sqrt5-5}{250},
\qquad
c_1=\log\!\frac{2}{\varphi},
\qquad
\varphi=\frac{1+\sqrt5}{2}.
\]
则对任意 \(r\in\overline{\mathbb Q}^{\times}\) 均有
\[
\kappa_{\infty}\neq r\,c_1^2.
\]
并且
\[
\frac{\kappa_{\infty}}{c_1^2}
\]
为超越数。
\end{theorem}
\begin{proof}
\(\kappa_{\infty}\) 为代数数。
由 \(2/\varphi\in\overline{\mathbb Q}_{>0}\setminus\{1\}\)，\(c_1=\log(2/\varphi)\) 为超越数，故 \(c_1^2\) 超越。
于是任意 \(r\in\overline{\mathbb Q}^{\times}\) 的乘积 \(r c_1^2\) 仍超越，不能等于代数数 \(\kappa_{\infty}\)。
若 \(\kappa_{\infty}/c_1^2\) 代数，则 \(c_1^2=\kappa_{\infty}/(\kappa_{\infty}/c_1^2)\) 代数，矛盾。证毕。
\end{proof}

\begin{corollary}[代数输运常数与对数熵常数之间不存在闭合平方律]\label{cor:xi-leyang-algebraic-transport-vs-log-entropy-no-square-law}
在有限核谱多项式框架中，凡可写为“简单特征支局部喷射经有理运算与有限代数消元”所得的输运常数，均为代数数。
另一方面，对任意
\[
\alpha\in\overline{\mathbb Q}_{>0}\setminus\{1\},
\qquad
c:=\log\alpha,
\]
有 \(c^2\) 超越。
因此对任意代数常数 \(T\in\overline{\mathbb Q}\) 与任意 \(r\in\overline{\mathbb Q}^{\times}\)，恒有
\[
T\neq r\,c^2.
\]
\end{corollary}
\begin{proof}
第一部分由定理 \ref{thm:xi-leyang-simple-root-algebraic-jet-closure} 的导数闭合性与代数消元稳定性给出。
第二部分由 Lindemann--Weierstrass 定理给出 \(c\) 超越，进而 \(c^2\) 超越。
代数数不可能等于上述超越数，结论成立。证毕。
\end{proof}
