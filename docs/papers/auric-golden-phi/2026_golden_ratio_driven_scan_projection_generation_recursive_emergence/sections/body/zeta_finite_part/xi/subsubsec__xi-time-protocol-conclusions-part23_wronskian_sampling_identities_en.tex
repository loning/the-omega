\begin{conclusion}[Resultant–discriminant product formula for diagonal sampling weights: closed forms for \texorpdfstring{$\prod_m K(b_m,b_m)$}{prod K(bm,bm)} and \texorpdfstring{$\prod_j K(x_j,x_j)$}{prod K(xj,xj)}]\label{con:xi-terminal-wronskian-diagonal-sampling-resultant-discriminant-product}
Following the notation of Conclusion~\ref{con:xi-terminal-wronskian-kernel-discrete-mercer-parseval}, let \(x_1,\dots,x_\kappa\) denote the distinct real zeros of \(y\) and \(b_0,\dots,b_\kappa\) the distinct real zeros of \(y_{+}\).
Then the following strict product identities hold:
\[
\boxed{\;
\prod_{m=0}^{\kappa}K(b_m,b_m)
=\pi^{-(\kappa+1)}\operatorname{Res}(A,y_{+})
=\pi^{-(\kappa+1)}(-1)^{\frac{\kappa(\kappa+1)}{2}}\operatorname{Res}(y,y_{+})\,\mathrm{Disc}(y_{+})\;},
\]
and
\[
\boxed{\;
\prod_{j=1}^{\kappa}K(x_j,x_j)
=\pi^{-\kappa}\operatorname{Res}(A,y)
=\pi^{-\kappa}(-1)^{\frac{\kappa(\kappa+1)}{2}}\operatorname{Res}(y,y_{+})\,\mathrm{Disc}(y)\;},
\]
where
\(
\mathrm{Disc}(y)=\prod_{1\le i<j\le \kappa}(x_i-x_j)^2
\)
and
\(
\mathrm{Disc}(y_{+})=\prod_{0\le i<j\le \kappa}(b_i-b_j)^2
\).
\end{conclusion}
\begin{proof}
By \(K(x,x)=A(x)/\pi\) (Conclusion~\ref{con:xi-terminal-wronskian-kernel-discrete-mercer-parseval}), we have
\[
\prod_{m=0}^{\kappa}K(b_m,b_m)=\pi^{-(\kappa+1)}\prod_{m=0}^{\kappa}A(b_m)
=\pi^{-(\kappa+1)}\operatorname{Res}(A,y_{+}).
\]
By Conclusion~\ref{con:xi-terminal-wronskian-resultant-discriminant-factorization},
\(
\operatorname{Res}(A,y_{+})
=(-1)^{\frac{\kappa(\kappa+1)}{2}}\operatorname{Res}(y,y_{+})\,\mathrm{Disc}(y_{+})
\), substituting yields the first identity.
\par
Similarly,
\[
\prod_{j=1}^{\kappa}K(x_j,x_j)=\pi^{-\kappa}\prod_{j=1}^{\kappa}A(x_j)
=\pi^{-\kappa}\operatorname{Res}(A,y).
\]
Since \(y(x_j)=0\), substituting into the Wronskian identity \(A=yy_{+}'-y'y_{+}\) yields
\(
A(x_j)=-y'(x_j)\,y_{+}(x_j)
\).
Thus
\[
\operatorname{Res}(A,y)
=\prod_{j=1}^{\kappa}A(x_j)
=(-1)^{\kappa}\Bigl(\prod_{j=1}^{\kappa}y'(x_j)\Bigr)\Bigl(\prod_{j=1}^{\kappa}y_{+}(x_j)\Bigr)
=(-1)^{\kappa}\Bigl(\prod_{j=1}^{\kappa}y'(x_j)\Bigr)\operatorname{Res}(y,y_{+}).
\]
For a monic polynomial \(y\), we have
\(
\prod_{j=1}^{\kappa}y'(x_j)=(-1)^{\frac{\kappa(\kappa-1)}{2}}\mathrm{Disc}(y)
\);
combining the sign \( (-1)^{\kappa+\frac{\kappa(\kappa-1)}{2}}=(-1)^{\frac{\kappa(\kappa+1)}{2}} \) yields
\(
\operatorname{Res}(A,y)
=(-1)^{\frac{\kappa(\kappa+1)}{2}}\operatorname{Res}(y,y_{+})\,\mathrm{Disc}(y)
\).
Substituting yields the second identity.
\end{proof}

\begin{conclusion}[Purely Lagrange form kernel identity: explicit sampling expansion of \texorpdfstring{$K$}{K} and bi-variable algebraic identity]\label{con:xi-terminal-wronskian-kernel-lagrange-sampling-identity}
Following the setting of Conclusion~\ref{con:xi-terminal-wronskian-kernel-discrete-mercer-parseval}, let \(b_0,\dots,b_\kappa\) be the distinct real zeros of \(y_{+}\).
Define the Lagrange basis polynomials
\[
L_m(z):=\frac{y_{+}(z)}{(z-b_m)\,y_{+}'(b_m)}\qquad (m=0,\dots,\kappa).
\]
Then for all \(x,y\in\RR\), the explicit kernel expansion holds:
\[
\boxed{\;
K(x,y)=\sum_{m=0}^{\kappa}K(b_m,b_m)\,L_m(x)\,L_m(y)
=\sum_{m=0}^{\kappa}\frac{A(b_m)}{\pi}\,L_m(x)\,L_m(y)\;}
\]
and equivalently the purely algebraic bi-variable identity holds:
\[
\boxed{\;
\frac{y_{+}(x)y(y)-y(x)y_{+}(y)}{x-y}
=y_{+}(x)\,y_{+}(y)\sum_{m=0}^{\kappa}\frac{y(b_m)}{(x-b_m)(y-b_m)\,y_{+}'(b_m)}\; }.
\]
\end{conclusion}
\begin{proof}
Since \(y_{+}(b_m)=0\), we have
\[
K(x,b_m)=\frac{y_{+}(x)y(b_m)}{\pi(x-b_m)},
\qquad
K(b_m,b_m)=\frac{A(b_m)}{\pi}=\frac{y(b_m)\,y_{+}'(b_m)}{\pi},
\]
hence \(K(x,b_m)=K(b_m,b_m)\,L_m(x)\).
Substituting into the discrete Mercer decomposition of Conclusion~\ref{con:xi-terminal-wronskian-kernel-discrete-mercer-parseval} yields
\(
K(x,y)=\sum_m K(b_m,b_m)L_m(x)L_m(y)
\).
The bi-variable form follows by rearranging the definition of \(K\) and \(K(b_m,b_m)=y(b_m)y_{+}'(b_m)/\pi\).
\end{proof}

\begin{conclusion}[Top coefficient extraction formula: Lagrange–derivative weight representation of \texorpdfstring{$[z^{\kappa}]$}{[z^k]}]\label{con:xi-terminal-wronskian-top-coefficient-extraction}
Following the notation of Conclusion~\ref{con:xi-terminal-wronskian-kernel-lagrange-sampling-identity}, let \(q\in\CC[z]\) with \(\deg q\le \kappa\).
Then the top coefficient extraction identity holds:
\[
\boxed{\;[z^{\kappa}]\,q(z)=\sum_{m=0}^{\kappa}\frac{q(b_m)}{y_{+}'(b_m)}\; }.
\]
In particular,
\[
\sum_{m=0}^{\kappa}\frac{1}{y_{+}'(b_m)}=0,\qquad
\sum_{m=0}^{\kappa}\frac{b_m}{y_{+}'(b_m)}=0,\qquad
\sum_{m=0}^{\kappa}\frac{y(b_m)}{y_{+}'(b_m)}=1.
\]
\end{conclusion}
\begin{proof}
For \(\deg q\le\kappa\), perform Lagrange interpolation:
\[
q(z)=\sum_{m=0}^{\kappa} q(b_m)\,L_m(z)
=\sum_{m=0}^{\kappa}q(b_m)\,\frac{y_{+}(z)}{(z-b_m)\,y_{+}'(b_m)}.
\]
Note that \(y_{+}(z)/(z-b_m)\) is a monic polynomial of degree \(\kappa\), hence the \(z^\kappa\) coefficient of \(L_m\) is \(1/y_{+}'(b_m)\).
Comparing the \(z^\kappa\) coefficients on both sides yields the main identity.
Applying to \(q(z)=1\), \(q(z)=z\), \(q(z)=y(z)\) and using \([z^\kappa]1=0\), \([z^\kappa]z=0\), \([z^\kappa]y(z)=1\) yields the three corollaries.
\end{proof}
