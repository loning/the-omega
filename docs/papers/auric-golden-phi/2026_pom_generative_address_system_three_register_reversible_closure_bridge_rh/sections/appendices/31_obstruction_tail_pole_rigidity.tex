\section{粘合障碍的四分量分解、端点尾律与极点格点刚性：可证伪的结构下界}\label{app:obstruction-tail-pole-rigidity}

本附录把三类“不可消去性”组织为统一的可证伪结构律：
\textbf{(i)} 粘合障碍在四因子记录轴下的正交分解与最小连续扩维维数判别；
\textbf{(ii)} 端点紧化的普适尾律与对数深度账本坐标的范畴唯一性；
\textbf{(iii)} 有限维行列式的极点格点刚性，及由此导出的自由度下界与二阶差集梳齿结构。

\subsection{四因子记录轴下的粘合障碍正交分解}\label{subsec:app31-h2-four-components}

设 $\mathcal{S}^\times$ 为“非 $\NULL$”证书层，其全局截面存在性刻画全局可寻址性（附录 \ref{app:adelic-sheaf-null-witness} 定理 \ref{thm:future-null-cohomology}）。设 $\underline{\mathcal{A}}$ 为异常标签的常值系数层，并以
$$
[\alpha]\in H^2(\Addr,\underline{\mathcal{A}})
$$
表示粘合障碍类（命题 \ref{prop:future-asc-torsion}）。
\par
在扩展保存制度下，最小记录轴的规范四因子分解已在附录 \ref{app:structural-synthesis-phase-diagram} \S\ref{subsubsec:app28-record-axis-four-factor} 给出：
$$
r\ \cong\ r_{\mathrm{fin}}\times r_{\widehat{\ZZ}}\times r_{\mathrm{wind}}\times r_{\mathrm{end}}.
$$

\begin{assumption}[互不耦合与系数层直和分解]\label{ass:app31-orthogonal-sum}
假设“互不耦合”在层论上可提升为系数层的直和分解：
$$
\underline{\mathcal{A}}
\ \cong\
\underline{\mathcal{A}}_{\mathrm{fin}}
\oplus \underline{\mathcal{A}}_{\widehat{\ZZ}}
\oplus \underline{\mathcal{A}}_{\mathrm{wind}}
\oplus \underline{\mathcal{A}}_{\mathrm{end}},
$$
其中四个子层分别对应有限缺陷、profinite 同余、绕数折叠与端点载荷四类正交缺陷来源（与四因子记录轴一一对应；见附录 \ref{app:structural-synthesis-phase-diagram} \S\ref{subsubsec:app28-record-axis-four-factor}）。
\end{assumption}

\begin{theorem}[四分量障碍的正交分解]\label{thm:app31-h2-splitting}
在假设 \ref{ass:app31-orthogonal-sum} 下，存在自然同构
$$
H^2(\Addr,\underline{\mathcal{A}})
\ \cong\
H^2(\Addr,\underline{\mathcal{A}}_{\mathrm{fin}})
\oplus H^2(\Addr,\underline{\mathcal{A}}_{\widehat{\ZZ}})
\oplus H^2(\Addr,\underline{\mathcal{A}}_{\mathrm{wind}})
\oplus H^2(\Addr,\underline{\mathcal{A}}_{\mathrm{end}}),
$$
并且
$$
[\alpha]=0
\quad\Longleftrightarrow\quad
[\alpha_{\mathrm{fin}}]=[\alpha_{\widehat{\ZZ}}]=[\alpha_{\mathrm{wind}}]=[\alpha_{\mathrm{end}}]=0.
$$
更进一步，若仅允许\textbf{连续扩维}（即附录 \ref{app:adelic-sheaf-null-witness} 定理 \ref{thm:future-register-minimality} 的 $\Addr\times\RR^d$ 型扩展），则最小连续扩维维数满足
$$
d_{\min}([\alpha])=d_{\min}\!\bigl([\alpha_{\mathrm{end}}]\bigr),
$$
因为前三类分量可由离散记录轴实现吸收，而不可被离散化吸收的连续负担严格落在端点精度载荷方向。
\end{theorem}

\begin{proof}
在阿贝尔层范畴中，直和函子与导函子意义下的上同调相容：对直和分解可逐分量取 $H^2$ 并得到直和分解。零性判别由直和分解的分量零性等价得出。连续最小维数结论与“端点载荷的连续精度账本严格正交于离散端点分支与其余离散因子”（附录 \ref{app:structural-synthesis-phase-diagram} \S\ref{subsubsec:app28-record-axis-four-factor} 与 \S\ref{subsubsec:app28-resource-phase}）一致：连续扩维只对端点连续负担敏感。证毕。
\end{proof}

\begin{corollary}[第四寄存器的唯一连续性]\label{cor:app31-fourth-register-unique-cont}
若研究目标要求证书化乘积公式闭合项（$\mathrm{BR}_{\mathrm{PF}}$），则其缺口必表现为 $[\alpha_{\mathrm{end}}]$ 的非平凡连续分量，从而迫使 $d_{\min}([\alpha])\ge 1$；其最小实现由无穷位径向寄存器 $\rho$ 给出（定理 \ref{thm:future-bridge-decomposition}）。
\end{corollary}

\subsection{端点紧化的普适尾律与对数深度账本坐标}\label{subsec:app31-tail-law}

端点紧化在有限精度下的硬阈值（附录 \ref{app:structural-synthesis-phase-diagram} \S\ref{subsubsec:app28-resource-phase}）并非面积相位门的偶然性质，而是紧化映射尾部行为的普适后果。

\begin{theorem}[端点尾指数定理与可核对动态范围]\label{thm:app31-tail-exponent}
设 $\upsilon:[0,\infty)\to(0,u_\ast)$ 为严格单调 $C^1$ 紧化，且存在常数 $c>0$ 与指数 $\alpha>0$ 使得
$$
u_\ast-\upsilon(x)\sim c\,x^{-\alpha}
\qquad(x\to\infty).
$$
对任意量化步长 $\Delta u>0$，定义端点分辨动态范围
$$
X_{\max}(\Delta u):=\sup\Bigl\{X>0:\ u_\ast-\upsilon(X)\ge \Delta u\Bigr\}.
$$
则当 $\Delta u\downarrow 0$ 时有渐近律
$$
X_{\max}(\Delta u)\sim\Bigl(\frac{c}{\Delta u}\Bigr)^{1/\alpha},
\qquad\text{等价地}\qquad
\Delta u\sim c\,X_{\max}^{-\alpha}.
$$
\end{theorem}

\begin{proof}
由单调性，$u_\ast-\upsilon(X)\ge\Delta u$ 与 $X\le X_{\max}(\Delta u)$ 等价。尾渐近给出
$$
u_\ast-\upsilon(X)=cX^{-\alpha}(1+o(1)),
$$
因此解得 $X=(c/\Delta u)^{1/\alpha}(1+o(1))$。证毕。
\end{proof}

\begin{corollary}[对数深度坐标的加性账本性]\label{cor:app31-log-depth}
令端点深度变量
$$
y(x):=-\log\bigl(u_\ast-\upsilon(x)\bigr).
$$
在定理 \ref{thm:app31-tail-exponent} 的条件下，
$$
y(x)=\alpha\log x-\log c+o(1)
\qquad(x\to\infty).
$$
并且在伸缩半群作用 $x\mapsto \lambda x$ 下有
$$
y(\lambda x)-y(x)=\alpha\log\lambda+o(1)
\qquad(x\to\infty).
$$
因此 $y$ 给出端点压缩的\textbf{加性账本坐标}，其角色正是最小记录轴端点因子 $r_{\mathrm{end}}$ 的连续精度载荷（附录 \ref{app:structural-synthesis-phase-diagram} \S\ref{subsubsec:app28-record-axis-four-factor}）。
\end{corollary}

\subsection{有限维极点格点刚性：计数上界、自由度下界与差集梳齿}\label{subsec:app31-pole-rigidity}

附录 \ref{app:pole-lattice-transfer-operator-upgrade} 已给出有限维极点格点屏障（命题 \ref{prop:finite-pole-lattice-geometry}）。本小节抽取两个直接用于“否证器/下界”叙述的推论：极点计数的线性上界与差集离散梳齿结构。

\begin{theorem}[极点计数的线性上界与维数下界]\label{thm:app31-pole-count-linear}
沿用命题 \ref{prop:finite-pole-lattice-geometry} 的记号。记 $N_M(T)$ 为 $\widehat{\zeta}_M$ 在带状区域
$$
\{s\in\CC:\ |\Im(s)|\le T\}
$$
内的极点总数（按重数计）。则存在绝对常数 $C_0>0$ 使对一切 $T\ge 1$ 有
$$
N_M(T)\le \frac{\log\lambda}{\pi}\,d\,T + C_0 d.
$$
因而对任意一族有限维模型 $(M_T)_{T\ge 1}$，若期望在高度 $T$ 上实现至少
$$
N_{M_T}(T)\ge c\,T\log T
$$
型的密度增长（$c>0$ 常数），则必有维数下界
$$
d_T\ge \frac{\pi c}{\log\lambda_T}\log T - O(1).
$$
\end{theorem}

\begin{proof}
由命题 \ref{prop:finite-pole-lattice-geometry}，每个非零特征值 $\mu$ 贡献一条虚部步长为 $2\pi/\log\lambda$ 的等差列；区间 $[-T,T]$ 中至多落入
$$
\frac{2T}{2\pi/\log\lambda}+O(1)=\frac{\log\lambda}{\pi}T+O(1)
$$
个格点。对全部非零特征值按代数重数求和，且非零特征值总数不超过 $d$，即得线性上界；反解即得维数下界。证毕。
\end{proof}

\begin{theorem}[差集离散性与二阶梳齿刚性]\label{thm:app31-difference-set-comb}
令 $\Gamma_M\subset\RR$ 为 $\widehat{\zeta}_M$ 全部极点的虚部多重集。则存在有限个 $a_1,\dots,a_J\in\RR$ 使
$$
\Gamma_M-\Gamma_M\subset \bigcup_{j=1}^{J}\Bigl(a_j+\frac{2\pi}{\log\lambda}\ZZ\Bigr).
$$
因此任何以“差分统计/二点结构”作为识别护栏的强化版本都会把固定有限核模型判为结构不相容：其二阶结构被迫为有限并的离散梳齿陪集，无法通过局部修补获得连续型极限统计。
\end{theorem}

\begin{proof}
由命题 \ref{prop:finite-pole-lattice-geometry}，$\Gamma_M$ 为有限条等差列（公差 $2\pi/\log\lambda$）之并；两条等差列的差为同一步长格的一个陪集；有限并仍为有限陪集并。证毕。
\end{proof}
