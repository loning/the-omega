\subsection{金属平均族的 gap（常数型连分数）}\label{app:kronecker-discrepancy-metallic-gap}

本小节给出一个与本文"反共振 $\Rightarrow$ 有限样本证书"主线更紧密的定量比较：在常数型连分数族（亦即金属平均族）中，黄金分支在 $O((\log N)/N)$ 证书的对数系数上达到严格最小，从而提供一个可审计的 gap。

\begin{definition}[常数型与金属平均族]\label{def:const-type}
对整数 $A\ge 1$，令
\[
\alpha_A:=[0;A,A,A,\dots]
\]
为常数型连分数。其收敛分母满足递推
\[
q_{n+1}=Aq_n+q_{n-1},\qquad q_0=1,\ q_1=A.
\]
当 $A=1$ 时 $\alpha_1=\varphi^{-1}$（黄金分支）；$A=2$ 对应白银族的典型代表；一般 $A$ 对应金属平均族的一个标准分支。
\end{definition}

\begin{lemma}[指数增长率]\label{lem:q-growth-const-type}
令
\[
\lambda_A:=\frac{A+\sqrt{A^2+4}}{2}>1.
\]
则 $q_n$ 具有指数增长率 $\lambda_A$：存在常数 $c_A,C_A>0$ 使得对所有 $n\ge 0$，
\[
|c_A\,\lambda_A^n\ \le\ q_n\ \le\ C_A\,\lambda_A^n.
\]
\end{lemma}

\begin{proof}
递推 $q_{n+1}=Aq_n+q_{n-1}$ 的特征方程为 $x^2=Ax+1$，其两个根为 $\lambda_A$ 与 $\lambda_A^{-1}$。因此 $q_n$ 有闭式表达 $q_n=u_A\lambda_A^n+v_A\lambda_A^{-n}$，其中系数由初值确定且 $u_A>0$。于是存在常数 $c_A,C_A$ 使得 $q_n$ 与 $\lambda_A^n$ 等价，得到所述不等式。
\end{proof}

\begin{proposition}[常数型的显式 \texorpdfstring{$O((\log N)/N)$}{O(logN/N)} 络线]\label{prop:const-type-envelope}
对 $\alpha=\alpha_A$（定义 \ref{def:const-type}），定理 \ref{thm:kronecker-discrepancy} 给出：对任意 $N\ge 2$，
\[
D_N^*(\alpha_A)\ \le\ \frac{C_A'}{N}+\frac{A}{\log\lambda_A}\cdot\frac{\log N}{N},
\]
其中 $C_A'>0$ 为仅依赖于 $A$ 的常数。
\end{proposition}

\begin{proof}
对常数型有 $a_j\equiv A$，故定理 \ref{thm:kronecker-discrepancy} 的分子为 $1+(n+1)A$。由引理 \ref{lem:q-growth-const-type}，若 $q_n\le N<q_{n+1}$，则存在仅依赖于 $A$ 的常数 $C$ 使得
\[
n\le \frac{\log N}{\log \lambda_A}+C.
\]
代入 $\frac{1+(n+1)A}{N}$ 即得所述形式（吸收常数项为 $C_A'/N$）。
\end{proof}

\begin{theorem}[金属平均族的严格 gap：黄金分支最小化对数系数]\label{thm:metallic-gap}
设 $A\ge 1$，记
\[
\kappa(A):=\frac{A}{\log\lambda_A},\qquad \lambda_A=\frac{A+\sqrt{A^2+4}}{2}.
\]
则函数 $\kappa(A)$ 在 $[1,\infty)$ 上严格递增。特别地，对所有整数 $A\ge 2$，
\[
\kappa(A)\ >\ \kappa(1)=\frac{1}{\log\varphi}.
\]
因此在金属平均族（常数型连分数）中，黄金分支 $\alpha=\varphi^{-1}$ 使得差异度证书的主导项 \(\kappa(A)\,(\log N)/N\) 的系数达到严格最小。
\end{theorem}

\begin{proof}
把 $\kappa$ 视为连续变量 $A>0$ 的函数。注意到 $\lambda_A$ 满足隐式方程
\[
\lambda_A^2=A\lambda_A+1
\quad\Longleftrightarrow\quad
A=\lambda_A-\lambda_A^{-1}.
\]
因此可将 $\kappa$ 写成
\[
\kappa(A)=\frac{\lambda-\lambda^{-1}}{\log\lambda}
\quad(\lambda=\lambda_A>1).
\]
令 $g(\lambda):=(\lambda-\lambda^{-1})/\log\lambda$。对 $\lambda>1$ 求导：
\[
g'(\lambda)=\frac{(1+\lambda^{-2})\log\lambda-(\lambda-\lambda^{-1})\cdot(1/\lambda)}{(\log\lambda)^2}
=\frac{(1+\lambda^{-2})\log\lambda-(1-\lambda^{-2})}{(\log\lambda)^2}.
\]
设 $h(\lambda):=(1+\lambda^{-2})\log\lambda-(1-\lambda^{-2})$。有 $h(1)=0$，并且
\[
h'(\lambda)=\frac{1+\lambda^{-2}}{\lambda}-2\lambda^{-3}\log\lambda-2\lambda^{-3}
=\frac{1}{\lambda}-\frac{1}{\lambda^3}-\frac{2\log\lambda}{\lambda^3}.
\]
再令 $\tilde h(\lambda):=\lambda^3 h'(\lambda)=\lambda^2-1-2\log\lambda$，则
\[
\tilde h'(\lambda)=2\lambda-\frac{2}{\lambda}=2\frac{\lambda^2-1}{\lambda}>0\quad(\lambda>1),
\qquad \tilde h(1)=0.
\]
故对 $\lambda>1$ 有 $\tilde h(\lambda)>0$，从而 $h'(\lambda)>0$，进而 $h(\lambda)>0$。于是 $g'(\lambda)>0$，即 $g$ 在 $(1,\infty)$ 上严格递增。由于 $A\mapsto\lambda_A$ 严格递增，复合得到 $\kappa(A)$ 在 $[1,\infty)$ 上严格递增。证毕。
\end{proof}
