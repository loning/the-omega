\subsubsection{Fifth-order collision moment and \texorpdfstring{$\mathrm{R\acute{e}nyi}$}{Renyi}--5 projection entropy rate}\label{subsec:pom-s5}
\begin{definition}[Fifth-order fold collision moment]\label{def:pom-s5}
Define the fifth-order collision sum
$$
S_5(m):=\sum_{x\in X_m} d_m(x)^5.
$$
Equivalently, drawing five windows \(\omega_1,\dots,\omega_5\) uniformly and independently from \(\Omega_m\), we have
$$
\mathbb{P}\bigl[\Fold_m(\omega_1)=\cdots=\Fold_m(\omega_5)\bigr]=\frac{S_5(m)}{32^m}.
$$
\end{definition}

\begin{proposition}[Fifth-order recurrence, dominant root and \texorpdfstring{$h_5$}{h5}]\label{prop:pom-s5-recurrence}
For all \(m\ge 13\) there exists an exact recurrence
$$
S_5(m)=2S_5(m-1)+11S_5(m-2)+8S_5(m-3)+20S_5(m-4)-10S_5(m-5).
$$
Let \(r_5\) denote the largest real root of the characteristic polynomial
$$
x^5-2x^4-11x^3-8x^2-20x+10=0
$$
(numerically \(r_5\approx 4.800521005045\)). Then
$$
S_5(m)=\Theta(r_5^m),
\qquad
h_5:=\lim_{m\to\infty}\frac{1}{m}\Bigl(-\log\sum_{x\in X_m}w_m(x)^5\Bigr)
:=\log\!\left(\frac{32}{r_5}\right).
$$
\end{proposition}
\begin{proof}
The recurrence is verified term-by-term on enumerated values by script \texttt{scripts/exp\_fold\_collision\_moment\_recursions.py} (which also determines the earliest valid starting point). The Perron root of its characteristic polynomial is \(r_5\), yielding \(S_5(m)=\Theta(r_5^m)\). Using
$$
\sum_{x\in X_m}w_m(x)^5=\sum_x\frac{d_m(x)^5}{32^m}=\frac{S_5(m)}{32^m},
$$
we obtain \(h_5=\log(32/r_5)\). This completes the proof.
\end{proof}

\begin{proposition}[\texorpdfstring{$h_q$}{hq} geometrically: Rényi dimension spectrum of the projected output measure]\label{prop:pom-renyi-dimension-spectrum}
Let \(w_m(x)=d_m(x)/2^m\) denote the push-forward of the uniform microstate. Identify resolution \(m\) with geometric scale via
\[
\varepsilon_m:=\varphi^{-m}.
\]
Define the \(q\)-th order Rényi dimension approximation at scale \(\varepsilon_m\) as
\[
D_q(m):=\frac{1}{q-1}\cdot \frac{\log\sum_{x\in X_m}w_m(x)^q}{\log \varepsilon_m}
\qquad(q\ge 2).
\]
If \(h_q=\lim_{m\to\infty}\frac{1}{m}H_q(m)\) exists (for instance when \(S_q(m)=\Theta(r_q^m)\)), then the limit
\[
D_q:=\lim_{m\to\infty}D_q(m)
\]
exists and satisfies the closed form
\[
\boxed{\;
D_q=\frac{h_q}{(q-1)\log\varphi}\; }.
\]
\end{proposition}
\begin{proof}
By definition \(H_q(m)=-\log\sum_x w_m(x)^q\) and \(\log\varepsilon_m=-m\log\varphi\) we have
\[
D_q(m)=\frac{1}{q-1}\cdot \frac{-H_q(m)}{-m\log\varphi}
:=\frac{1}{(q-1)\log\varphi}\cdot \frac{1}{m}H_q(m).
\]
Taking \(m\to\infty\) yields the result. This completes the proof.
\end{proof}

\begin{proposition}[Universal sandwich: high-order collision dominant exponent and projection entropy rate locked by \texorpdfstring{$|X_m|$}{|Xm|} and maximal fiber]\label{prop:pom-rq-universal-bounds}
For integer \(q\ge 2\), let
$$
S_q(m):=\sum_{x\in X_m} d_m(x)^q,\qquad d_m(x):=\bigl|\Fold_m^{-1}(x)\bigr|.
$$
Let \(r_q\) denote the exponential growth rate of \(S_q(m)\) (e.g., the Perron root given by Theorem \ref{thm:pom-collision-kernel-ak}; equivalently \(r_q=\lim_{m\to\infty}S_q(m)^{1/m}\)), and define the unnormalized \(q\)-th order projection entropy rate
$$
h_q:=\log\!\left(\frac{2^q}{r_q}\right)=q\log 2-\log r_q.
$$
Then we have two-sided bounds
$$
\frac{2^q}{\varphi^{q-1}}\ \le\ r_q\ \le\ 2\,\varphi^{(q-1)/2},
$$
equivalently
$$
(q-1)\log\!\left(\frac{2}{\sqrt{\varphi}}\right)\ \le\ h_q\ \le\ (q-1)\log\varphi.
$$
\end{proposition}
\begin{proof}
For any fixed \(m\) and \(q\ge 2\), by power-mean/Hölder inequality,
$$
S_q(m)=\sum_{x\in X_m}d_m(x)^q\ \ge\ \frac{\bigl(\sum_x d_m(x)\bigr)^q}{|X_m|^{\,q-1}}
:=\frac{2^{qm}}{F_{m+2}^{\,q-1}},
$$
where \(\sum_x d_m(x)=|\Omega_m|=2^m\) and \(|X_m|=F_{m+2}\).
On the other hand, letting \(D_m:=\max_{x\in X_m} d_m(x)\), we have
$$
S_q(m)=\sum_x d_m(x)^q\ \le\ D_m^{\,q-1}\sum_x d_m(x)=2^m D_m^{\,q-1}.
$$
Taking logarithm and dividing by \(m\) in both inequalities, then letting \(m\to\infty\), and combining Fibonacci asymptotics \(F_{m+2}\sim \varphi^{m}/\sqrt5\) with maximal fiber asymptotics \(D_m=\Theta(\varphi^{m/2})\) (Theorem \ref{thm:pom-max-fiber}), we obtain
$$
\log r_q=\lim_{m\to\infty}\frac{1}{m}\log S_q(m)\in\Bigl[q\log 2-(q-1)\log\varphi,\ \log 2+\frac{q-1}{2}\log\varphi\Bigr],
$$
whence the bounds on \(r_q\) hold. Substituting into \(h_q=q\log 2-\log r_q\) yields the equivalent bounds on \(h_q\). This completes the proof.
\end{proof}

\begin{proposition}[Extreme-order endpoint: \texorpdfstring{$r_q^{1/q}\to\sqrt{\varphi}$}{rq^(1/q)->sqrt(phi)} and normalized entropy endpoint constant]\label{prop:pom-rq-qinfty-endpoint}
Let \(r_q\) denote the exponential growth rate of \(S_q(m)\) (e.g., given by the Perron root of the moment-kernel), and let \(\varphi\) denote the golden ratio.
Then as \(q\to\infty\) we have the limit
\[
\boxed{\;
\lim_{q\to\infty} r_q^{1/q}=\sqrt{\varphi}\; }.
\]
Equivalently,
\[
\lim_{q\to\infty}\frac{1}{q}\,h_q
:=\log\!\left(\frac{2}{\sqrt{\varphi}}\right),
\qquad
\lim_{q\to\infty}\frac{1}{q-1}\,h_q
:=\log\!\left(\frac{2}{\sqrt{\varphi}}\right),
\]
so the Rényi dimension spectrum endpoint satisfies
\[
\lim_{q\to\infty}D_q
:=\frac{\log\!\left(2/\sqrt{\varphi}\right)}{\log\varphi}
\approx 0.94042009
\]
(where \(D_q\) is defined in Proposition \ref{prop:pom-renyi-dimension-spectrum}).
\par
Moreover for any integer \(q\ge 2\) we have explicit error bound
\[
\boxed{\;
0\ \le\ \log\!\left(\frac{2}{\sqrt{\varphi}}\right)-\frac{h_q}{q}\ \le\ \frac{\log\varphi}{q}\; }.
\]
\end{proposition}
\begin{proof}
For any \(m,q\ge 2\), letting
\(
D_m:=\max_x d_m(x)
\)
we have the sandwich
\[
D_m^q\ \le\ S_q(m)\ \le\ |X_m|\,D_m^q.
\]
Taking \(1/(mq)\)-th power of both sides yields
\[
D_m^{1/m}\ \le\ S_q(m)^{1/(mq)}\ \le\ D_m^{1/m}\,|X_m|^{1/(mq)}.
\]
Taking \(m\to\infty\), the left side by Theorem \ref{thm:pom-max-fiber} gives
\(
\lim_{m\to\infty}D_m^{1/m}=\varphi^{1/2}
\);
the factor on the right satisfies
\(
\lim_{m\to\infty}|X_m|^{1/(mq)}=\varphi^{1/q}
\)
(since \(|X_m|=F_{m+2}\sim \varphi^m/\sqrt5\)).
Thus for each fixed \(q\ge 2\) we have
\[
\sqrt{\varphi}\ \le\ r_q^{1/q}\ \le\ \sqrt{\varphi}\,\varphi^{1/q}.
\]
Taking \(q\to\infty\) yields \(\lim_{q\to\infty}r_q^{1/q}=\sqrt{\varphi}\).
Finally the various endpoint limits follow directly from \(h_q=q\log 2-\log r_q\) and the closed form in Proposition \ref{prop:pom-renyi-dimension-spectrum}.
\par
From \(\sqrt{\varphi}\le r_q^{1/q}\le \sqrt{\varphi}\,\varphi^{1/q}\) we obtain
\[
\frac{1}{2}\log\varphi\ \le\ \frac{1}{q}\log r_q\ \le\ \frac{1}{2}\log\varphi+\frac{1}{q}\log\varphi.
\]
Substituting into \(\frac{h_q}{q}=\log 2-\frac{1}{q}\log r_q\) immediately yields
\[
0\ \le\ \log\!\left(\frac{2}{\sqrt{\varphi}}\right)-\frac{h_q}{q}\ \le\ \frac{\log\varphi}{q}.
\]
This completes the proof.
\end{proof}

\begin{proposition}[Convexity of pressure and concavity of fingerprint]\label{prop:pom-pressure-convexity}
For any \(m\ge 2\), the function \(q\mapsto \log S_q(m)\) is convex on \(q\ge 0\).

\[
\boxed{\;

\[
\boxed{\;
h_{q_1}\ \ge\ \frac{q_2-q_1}{q_2-q_0}\,h_{q_0}\ +\ \frac{q_1-q_0}{q_2-q_0}\,h_{q_2}\; }.
\]
\end{proposition}
\begin{proof}
Fix \(m\). Take any \(q_0,q_2\ge 0\) and \(\theta\in(0,1)\), and let \(q_1=\theta q_0+(1-\theta)q_2\).
By Hölder's inequality (or convexity of log-sum-exp) we have
\[
S_{q_1}(m)=\sum_{x\in X_m}d_m(x)^{q_1}
\le
\left(\sum_{x\in X_m}d_m(x)^{q_0}\right)^{\theta}
\left(\sum_{x\in X_m}d_m(x)^{q_2}\right)^{1-\theta}
:=S_{q_0}(m)^{\theta}S_{q_2}(m)^{1-\theta}.
\]
Taking logarithm yields convexity of \(q\mapsto \log S_q(m)\).
\par
Let \(q_0<q_1<q_2\) be integers and take
\(
\theta=\frac{q_2-q_1}{q_2-q_0}
\)
so that \(q_1=\theta q_0+(1-\theta)q_2\).
Taking \(\frac{1}{m}\log\) of the above and letting \(m\to\infty\) yields
\[
\log r_{q_1}\le \theta\log r_{q_0}+(1-\theta)\log r_{q_2},
\]
namely discrete convexity of \(P\). Finally \(h_q=q\log 2-P(q)\) is a linear function minus a convex function, hence concave at integer points. This completes the proof.
\end{proof}

\begin{proposition}[Cross-\texorpdfstring{$q$}{q} inequality: monotonicity of \texorpdfstring{$g(q)$}{g(q)}]\label{prop:pom-crossq-g-monotone}
For any integer \(m\ge 2\) and \(2\le p<q\), let \(d_m=(d_m(x))_{x\in X_m}\) and write \(S_k(m)=\sum_x d_m(x)^k=\|d_m\|_k^k\).
Then finite-dimensional norm inequality gives
\[
\|d_m\|_p\ \le\ |X_m|^{\frac{1}{p}-\frac{1}{q}}\ \|d_m\|_q,
\]
whence
\[
S_q(m)\ \ge\ S_p(m)^{q/p}\,|X_m|^{1-q/p}.
\]
If there exist \(r_k\) with \(S_k(m)=\Theta(r_k^m)\), and \(|X_m|\sim \varphi^m\), then we have cross-order constraint
\[
\boxed{\;
\log r_q \ \ge\ \frac{q}{p}\log r_p + \left(1-\frac{q}{p}\right)\log\varphi\; }.
\]
Equivalently, defining
\[
g(q):=\frac{\log r_q-\log\varphi}{q},
\]
we have \(g(q)\) nondecreasing on \(q\ge 2\).
Table \ref{tab:fold_collision_crossq_g_profile_q2_23} provides auditable readings for \(q=2,\dots,23\), serving as regression consistency checks for subsequent extensions/reimplementations (one-click generation via script \texttt{scripts/exp\_fold\_collision\_crossq\_g\_profile\_q2\_23.py}).
\end{proposition}
\begin{proof}
For nonnegative vector \(d_m\) of length \(n=|X_m|\), when \(2\le p<q\) we have \(\|d_m\|_p\le n^{1/p-1/q}\|d_m\|_q\).
Taking \(q\)-th power of both sides yields
\(
\|d_m\|_q^q \ge \|d_m\|_p^q\,n^{1-q/p}
\),
i.e.,
\(
S_q(m)\ge S_p(m)^{q/p}|X_m|^{1-q/p}
\).
Taking \(\frac{1}{m}\log\) of both sides and letting \(m\to\infty\), using
\(
\lim_{m\to\infty}\frac{1}{m}\log S_k(m)=\log r_k
\)
and
\(
\lim_{m\to\infty}\frac{1}{m}\log|X_m|=\log\varphi
\)
yields the first inequality; substituting back into the definition gives monotonicity of \(g(q)\). This completes the proof.
\end{proof}

\begin{remark}[Structural origin of closure dimension \texorpdfstring{$2\lfloor q/2\rfloor+1$}{2 floor(q/2)+1}: finite truncation of congruence-role modes]\label{rem:pom-closure-dim-mode-trunc}

\end{remark}
