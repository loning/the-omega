\section{Variation details}
\label{app:variation_details}

This appendix records the core variational identities behind Section~\ref{sec:variational_derivation}.
We emphasize that the purpose is auditability: all assumptions are isolated in the action ansatz \eqref{eq:omega_action_cap2}, while the derivation is standard.

\subsection{Metric variation}
\label{app:metric_variation}

For a covariant action $S=\int\sqrt{-g}\,\mathcal{L}$, variation of the Einstein--Hilbert term yields
\begin{equation}
  \delta\left(\sqrt{-g}\,R\right)
  = \sqrt{-g}\,G_{\mu\nu}\,\delta g^{\mu\nu} + \text{(boundary term)}.
\end{equation}
For the Fisher-amplitude term one uses $\delta\sqrt{-g}=-\frac12\sqrt{-g}\,g_{\mu\nu}\delta g^{\mu\nu}$ and the definition $(\nabla\chi)^2=g^{\mu\nu}\nabla_\mu\chi\nabla_\nu\chi$ to obtain the stress tensor \eqref{eq:info_stress_cap2}.

\subsection{Scalar variation}
\label{app:scalar_variation}

Varying with respect to $\chi$ gives the Euler--Lagrange equation
\begin{equation}
  2\lambda_F\,\square\chi-\frac{\dd V}{\dd\chi}=0,
\end{equation}
where $\square=\nabla^\mu\nabla_\mu$.
When the scalar equation holds, the divergence of $T^{\mathrm{info}}_{\mu\nu}$ reduces to a term proportional to the scalar equation, yielding the consistency statement in \eqref{eq:stress_conservation_cap2}.

\subsection{\texorpdfstring{$u=\log(\kappa/\kappa_0)$}{u=log(kappa/kappa0)} chain rule identity}
\label{app:u_chain_rule}

This subsection records the elementary chain rule behind Proposition~\ref{prop:u_equation}.
Under Assumption~\ref{ass:cost_to_density}, write $u=\log(\kappa/\kappa_0)$ so that
\begin{equation}
  \chi=\sqrt{\rho}=\sqrt{\rho_0}\,\e^{\frac{p}{2}u}.
\end{equation}
Then
\begin{equation}
  \nabla_\mu\chi=\frac{p}{2}\,\chi\,\nabla_\mu u,
  \qquad
  \square\chi=\nabla^\mu\nabla_\mu\chi
  =\frac{p}{2}\,\chi\,\square u+\frac{p^2}{4}\,\chi\,(\nabla u)^2,
\end{equation}
where $(\nabla u)^2=g^{\mu\nu}\nabla_\mu u\nabla_\nu u$.
Substituting into the scalar equation $2\lambda_F\square\chi-\dd V/\dd\chi=0$ and dividing by $\chi>0$ yields
\begin{equation}
  \lambda_F\,p\,\square u+\lambda_F\,\frac{p^2}{2}\,(\nabla u)^2
  =\frac{1}{\chi}\,\frac{\dd V}{\dd\chi},
\end{equation}
which is \eqref{eq:u_equation}.
