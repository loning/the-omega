\subsection{Disk Completion and the Disk Zero-Free Equivalence of RH: Compressing \texorpdfstring{$t\to\infty$}{t->infty} to the Boundary Point \texorpdfstring{$w=-1$}{w=-1}}\label{subsec:unit-disk-rh-equivalence}
This subsection provides a \textbf{parameter-free diskification restatement} complementary to the unitary slice caliber of the main text: the completed function \(\Xi_{\zeta}\) (whose zeros coincide with the nontrivial zeros of \(\zeta\)) is transported via the Cayley transform to an analytic function inside the unit disk, thereby reformulating RH equivalently as two types of criteria: ``disk-interior zero exclusion'' and ``vanishing of the finite circumferential average defect at the boundary.''
By contrast, in the main text \S\ref{subsec:xi-from-self-dual}, the Dirichlet cross-section \(u_L(s)=L^{2s-1}\) identifies the critical line as the unit circle slice on the parameter side (Remark~\ref{rem:xi-critical-line-unit-circle}); the present subsection goes further by compactifying the critical-line parameter \(t\in\RR\) itself onto a single arc on the unit circle, merging \(\pm\infty\) into the single boundary point \(w=-1\).

\paragraph{Non-periodic phase parametrization of the self-dual univariate \texorpdfstring{$\Xi$}{Xi}: rewriting the ``single-point infinity'' of the critical line}
Let \(\Xi_{\Delta,L}(s)=\Delta(L^{-s},L^{2s-1})\) as in \S\ref{subsec:xi-from-self-dual}.
Under the completion condition, the critical-line real-valuedness
\(
\Xi_{\Delta,L}(\tfrac12+\mathrm{i}t)\in\RR
\)
is given by Proposition~\ref{prop:xi-delta-L-critical-line-real}.
Accordingly one introduces a \textbf{non-periodic critical-line parametrization} consistent with \eqref{eq:dirichlet-cayley-phase-axis}: the Cayley phase axis compactifies \(t\in\RR\) to a unit circle parameter on \(\TT\setminus\{-1\}\), merging \(t\to\pm\infty\) into the endpoint \(-1\).

\begin{definition}[Non-periodic critical-line phase parametrization]\label{def:xi-nonperiodic-phase}
Define
\begin{equation}\label{eq:xi-nonperiodic-phase}
\boxed{\;
\widetilde{\Xi}_{\Delta,L}(u_{\mathrm C})
:=\Xi_{\Delta,L}\!\Bigl(\frac12+\mathrm{i}\,t(u_{\mathrm C})\Bigr),
\qquad
t(u_{\mathrm C})=-\mathrm{i}\,\frac{u_{\mathrm C}-1}{u_{\mathrm C}+1}
\qquad
(u_{\mathrm C}\in\TT\setminus\{-1\})\; }.
\end{equation}
\end{definition}

\begin{proposition}[Unit circle symmetry: rewriting the functional equation on \texorpdfstring{$\TT$}{T}]\label{prop:xi-nonperiodic-phase-symmetry}
Under the conditions of Proposition~\ref{prop:xi-delta-L-critical-line-real}, \(\widetilde{\Xi}_{\Delta,L}\) is a real-valued function defined on \(\TT\setminus\{-1\}\).
Moreover, it satisfies the pure unit circle symmetry
\begin{equation}\label{eq:xi-unit-circle-inversion-symmetry}
\boxed{\;
\widetilde{\Xi}_{\Delta,L}(u_{\mathrm C})
=\widetilde{\Xi}_{\Delta,L}(1/u_{\mathrm C})\; }.
\end{equation}
The endpoint limit \(u_{\mathrm C}\to-1\) corresponds to \(t(u_{\mathrm C})\to\pm\infty\), so that the infinite height of the critical line is strictly rewritten as a single-point boundary layer on the unit circle.
\end{proposition}
\begin{proof}
When \(u_{\mathrm C}\in\TT\setminus\{-1\}\), one has \(t(u_{\mathrm C})\in\RR\), so by Proposition~\ref{prop:xi-delta-L-critical-line-real} one obtains \(\widetilde{\Xi}_{\Delta,L}(u_{\mathrm C})\in\RR\).
Furthermore, from the functional equation \(\Xi_{\Delta,L}(s)=\Xi_{\Delta,L}(1-s)\) (Proposition~\ref{prop:xi-delta-L-functional-equation}), its equivalent form on the critical line is
\(
\Xi_{\Delta,L}(\tfrac12+\mathrm{i}t)=\Xi_{\Delta,L}(\tfrac12-\mathrm{i}t)
\),
so \(\Xi_{\Delta,L}\) is an even function on the critical line.
The Cayley inverse satisfies \(t(1/u_{\mathrm C})=-t(u_{\mathrm C})\) (for instance, taking \(u_{\mathrm C}=e^{\mathrm{i}\theta}\) gives \(t(u_{\mathrm C})=\tan(\theta/2)\)),
hence
\[
\widetilde{\Xi}_{\Delta,L}(1/u_{\mathrm C})
=\Xi_{\Delta,L}\!\Bigl(\tfrac12+\mathrm{i}\,t(1/u_{\mathrm C})\Bigr)
=\Xi_{\Delta,L}\!\Bigl(\tfrac12-\mathrm{i}\,t(u_{\mathrm C})\Bigr)
=\widetilde{\Xi}_{\Delta,L}(u_{\mathrm C}),
\]
yielding \eqref{eq:xi-unit-circle-inversion-symmetry}.
The endpoint correspondence follows immediately from \(\mathcal{C}(t)=\frac{1+\mathrm{i}t}{1-\mathrm{i}t}\to-1\ (t\to\pm\infty)\) (Proposition~\ref{prop:app-cayley-upperhalf-disk}). \qed
\end{proof}

\paragraph{Completed object}
Write
\begin{equation}\label{eq:app-xi-classical}
\Xi_{\zeta}(s):=\frac12\,s(s-1)\,\pi^{-s/2}\Gamma\!\left(\frac{s}{2}\right)\zeta(s),
\end{equation}
so that \(\Xi_{\zeta}\) is an entire function satisfying the strict functional equation \(\Xi_{\zeta}(s)=\Xi_{\zeta}(1-s)\), whose zero set coincides exactly (including multiplicities) with the nontrivial zeros of the Riemann \(\zeta\) function (see \cite{Apostol1976}).

\paragraph{Upper half-plane \texorpdfstring{$\leftrightarrow$}{<->} disk: parameter-free Cayley compactification}
Let the upper half-plane and the unit disk be
\[
\mathbb{H}:=\{t\in\CC:\Im(t)>0\},
\qquad
\mathbb{D}:=\{w\in\CC:|w|<1\}.
\]
Define the M\"obius transformation (consistent with \eqref{eq:unit-circle-cayley-embed} on the real axis)
\begin{equation}\label{eq:app-cayley-disk}
\boxed{\;
w=\mathcal{C}(t):=\frac{1+\mathrm{i}t}{1-\mathrm{i}t},
\qquad
t=\mathcal{C}^{-1}(w):=-\mathrm{i}\,\frac{w-1}{w+1}\; }.
\end{equation}

\begin{proposition}[Disk geometry of the Cayley compactification]\label{prop:app-cayley-upperhalf-disk}
\(\mathcal{C}\) gives a biholomorphic isomorphism \(\mathcal{C}:\mathbb{H}\to\mathbb{D}\), with inverse given by \(\mathcal{C}^{-1}\) in \eqref{eq:app-cayley-disk}.
Moreover, when \(t\in\RR\) one has \(|\mathcal{C}(t)|=1\), and as \(t\to\pm\infty\) one has \(\mathcal{C}(t)\to-1\).
\end{proposition}
\begin{proof}
Direct computation gives
\[
|\mathcal{C}(t)|^2
=\frac{|1+\mathrm{i}t|^2}{|1-\mathrm{i}t|^2}
=\frac{(1+\mathrm{i}t)(1-\mathrm{i}\overline t)}{(1-\mathrm{i}t)(1+\mathrm{i}\overline t)}
=\frac{1+|t|^2-2\Im(t)}{1+|t|^2+2\Im(t)}.
\]
Therefore \(\Im(t)>0\Rightarrow |\mathcal{C}(t)|<1\) and \(\Im(t)=0\Rightarrow |\mathcal{C}(t)|=1\).
The inverse is obtained by algebraic solving, and one can verify that it maps \(|w|<1\) to \(\Im(t)>0\).
Finally, when \(t\in\RR\) one has \(|\mathcal{C}(t)|=1\), and as \(t\to\pm\infty\) one has \(\mathcal{C}(t)=\frac{1+\mathrm{i}t}{1-\mathrm{i}t}\to-1\). \qed
\end{proof}

\paragraph{Diskified completed function}
Let
\begin{equation}\label{eq:app-s-of-w}
\boxed{\;
s(w):=\frac12+\mathrm{i}\,\mathcal{C}^{-1}(w)
=\frac12+\frac{w-1}{w+1}\; },
\qquad (|w|<1),
\end{equation}
and define the analytic function inside the disk
\begin{equation}\label{eq:app-Fzeta-disk}
\boxed{\;
F_{\zeta}(w):=\Xi_{\zeta}\!\bigl(s(w)\bigr)\qquad(|w|<1)\; }.
\end{equation}
Since \(\Xi_{\zeta}\) is entire and \(s(w)\) is holomorphic in \(|w|<1\) (its only pole \(w=-1\) lies on the boundary), \(F_{\zeta}\) is analytic in the open unit disk.
When \(w=e^{\mathrm{i}\theta}\in\TT\setminus\{-1\}\), from \eqref{eq:app-cayley-disk} one obtains \(\mathcal{C}^{-1}(e^{\mathrm{i}\theta})=\tan(\theta/2)\in\RR\), whence
\begin{equation}\label{eq:app-Fzeta-boundary}
F_{\zeta}(e^{\mathrm{i}\theta})=\Xi_{\zeta}\!\left(\tfrac12+\mathrm{i}\tan(\theta/2)\right),
\end{equation}
i.e., ``evaluation on the critical line'' is rewritten as ``phase parametrization on the unit circle boundary.''

\begin{theorem}[Disk equivalence of RH: disk-interior zero exclusion]\label{thm:app-rh-iff-disk-zero-free}
RH is equivalent to
\[
\boxed{\ \forall\,w\in\mathbb{D},\quad F_{\zeta}(w)\neq 0\ }.
\]
\end{theorem}
\begin{proof}
(\(\Rightarrow\)) If RH does not hold, then there exists a zero \(\rho=\beta+\mathrm{i}\gamma\) of \(\Xi_{\zeta}\) with \(\beta\neq \tfrac12\) and \(0<\beta<1\).
By the symmetry \(\Xi_{\zeta}(s)=\Xi_{\zeta}(1-s)\), one may assume without loss of generality that \(\beta<\tfrac12\).
Let
\[
t_{\rho}:=\gamma-\mathrm{i}\left(\beta-\tfrac12\right),
\qquad
\Im(t_{\rho})=\tfrac12-\beta>0,
\]
so \(t_{\rho}\in\mathbb{H}\) and \( \tfrac12+\mathrm{i}t_{\rho}=\rho\).
By Proposition~\ref{prop:app-cayley-upperhalf-disk}, take \(w_{\rho}:=\mathcal{C}(t_{\rho})\in\mathbb{D}\) with \(\mathcal{C}^{-1}(w_{\rho})=t_{\rho}\).
Therefore
\[
F_{\zeta}(w_{\rho})
=\Xi_{\zeta}\!\left(\tfrac12+\mathrm{i}\,\mathcal{C}^{-1}(w_{\rho})\right)
=\Xi_{\zeta}\!\left(\tfrac12+\mathrm{i}t_{\rho}\right)
=\Xi_{\zeta}(\rho)
=0,
\]
so a zero appears inside the disk.

(\(\Leftarrow\)) Conversely, if there exists \(w_0\in\mathbb{D}\) with \(F_{\zeta}(w_0)=0\), let \(t_0:=\mathcal{C}^{-1}(w_0)\in\mathbb{H}\) and set \(s_0:=\tfrac12+\mathrm{i}t_0\).
Since \(\Im(t_0)>0\), one has \(\Re(s_0)=\tfrac12-\Im(t_0)<\tfrac12\), and
\(
\Xi_{\zeta}(s_0)=F_{\zeta}(w_0)=0
\).
By the functional equation and conjugate symmetry, there immediately exists a zero off the critical line, so RH does not hold. \qed
\end{proof}

\paragraph{Precise quantitative compression of off-critical-line shift in disk radius}
In the proof of Theorem~\ref{thm:app-rh-iff-disk-zero-free}, the off-critical-line zero event is transported to a disk-interior zero event.
The following proposition gives the completely closed-form compression law of this transport in disk radius: the off-line shift \(\tfrac12-\beta\) is compressed into the boundary layer at the scale \(\Theta(\gamma^{-2})\) in disk coordinates.

\begin{proposition}[Closed-form compression of off-critical-line shift in disk radius]\label{prop:app-offcritical-radius-compression}
Suppose \(\Xi_{\zeta}\) has a zero \(\rho=\beta+\mathrm{i}\gamma\) with \(\beta\neq\tfrac12\) and \(0<\beta<1\).
By the symmetry \(\Xi_{\zeta}(s)=\Xi_{\zeta}(1-s)\), assume without loss of generality that \(\beta<\tfrac12\), and let
\[
\delta:=\tfrac12-\beta>0,
\qquad
t_{\rho}:=\gamma+\mathrm{i}\delta\in\mathbb{H},
\qquad
w_{\rho}:=\mathcal{C}(t_{\rho})\in\mathbb{D}.
\]
Then \(F_{\zeta}(w_{\rho})=0\).
Moreover, the disk radius of \(w_{\rho}\) satisfies the completely closed form
\begin{equation}\label{eq:app-offcritical-radius-closed}
\boxed{\;
|w_{\rho}|^{2}
=\frac{\gamma^{2}+(\delta-1)^{2}}{\gamma^{2}+(\delta+1)^{2}},
\qquad
1-|w_{\rho}|^{2}
=\frac{4\delta}{\gamma^{2}+(\delta+1)^{2}}\; }.
\end{equation}
\end{proposition}
\begin{proof}
By the construction in Theorem~\ref{thm:app-rh-iff-disk-zero-free}, \(F_{\zeta}(w_{\rho})=\Xi_{\zeta}(\rho)=0\).
The closed form for the modulus follows from the identity in Proposition~\ref{prop:app-cayley-upperhalf-disk},
\(
|\mathcal{C}(t)|^{2}=\frac{1+|t|^{2}-2\Im(t)}{1+|t|^{2}+2\Im(t)}
\),
by taking \(t=t_{\rho}=\gamma+\mathrm{i}\delta\) and simplifying directly.
\end{proof}

\begin{corollary}[High-height compression scale]\label{cor:app-offcritical-radius-high-height}
When \(|\gamma|\to\infty\) with \(\delta\) bounded,
\[
1-|w_{\rho}|^{2}\sim \frac{4\delta}{\gamma^{2}},
\qquad
1-|w_{\rho}|\sim \frac{2\delta}{\gamma^{2}}.
\]
Therefore, for a fixed off-critical-line shift \(\delta\), the resolvable scale of the shift in disk coordinates is \(\Theta(\gamma^{-2})\): the larger the height, the stronger the radial compression.
\end{corollary}
\begin{proof}
From \eqref{eq:app-offcritical-radius-closed} one obtains
\(
1-|w_{\rho}|^{2}=\frac{4\delta}{\gamma^{2}+(\delta+1)^{2}}
\sim \frac{4\delta}{\gamma^{2}}
\).
By the identity
\(
1-|w_{\rho}|=\frac{1-|w_{\rho}|^{2}}{1+|w_{\rho}|}
\)
and noting \(|w_{\rho}|\to 1\), the second expression follows. \qed
\end{proof}

\begin{corollary}[``Second-order radial channel'' in area-phase endpoint depth]\label{cor:app-offcritical-radius-second-order-radial-channel}
Along the area-phase coordinate \(\upsilon(x)=\pi^{-1}\arctan(x)\) (equation~\eqref{eq:unit-circle-phase-upsilon}), for height \(\gamma\in\RR\) define the endpoint depth
\[
\delta_{\gamma}:=\frac12-\frac{1}{\pi}\arctan|\gamma|\in\Bigl(0,\frac12\Bigr),
\qquad\text{so that}\qquad
|\gamma|=\cot(\pi\delta_{\gamma}),\quad \delta_{\gamma}\sim\frac{1}{\pi|\gamma|}\ (|\gamma|\to\infty).
\]
Suppose \(\rho=\beta+\mathrm{i}\gamma\) is a zero of \(\Xi_{\zeta}\) with \(0<\beta<\frac12\), and let \(w_{\rho}\in\mathbb{D}\) be as in Proposition~\ref{prop:app-offcritical-radius-compression}.
Then when \(\delta_{\gamma}\downarrow 0\) (equivalently \(|\gamma|\to\infty\)) with \(\beta\) bounded,
\[
\boxed{\;
1-|w_{\rho}|^{2}
=
4\pi^{2}\Bigl(\tfrac12-\beta\Bigr)\,\delta_{\gamma}^{2}\bigl(1+o(1)\bigr),
\qquad
1-|w_{\rho}|
=
2\pi^{2}\Bigl(\tfrac12-\beta\Bigr)\,\delta_{\gamma}^{2}\bigl(1+o(1)\bigr)\; }.
\]
\end{corollary}
\begin{proof}
By Corollary~\ref{cor:app-offcritical-radius-high-height},
\(
1-|w_{\rho}|^{2}=\frac{4(\tfrac12-\beta)}{\gamma^{2}}(1+o(1))
\)
and \(1-|w_{\rho}|=\frac{2(\tfrac12-\beta)}{\gamma^{2}}(1+o(1))\).
On the other hand, from \(\delta_{\gamma}=\frac12-\pi^{-1}\arctan|\gamma|\) one obtains \(|\gamma|=\cot(\pi\delta_{\gamma})\sim(\pi\delta_{\gamma})^{-1}\), whence \(\gamma^{-2}=\pi^{2}\delta_{\gamma}^{2}(1+o(1))\).
Substituting back yields the stated result. \qed
\end{proof}

\begin{remark}[Second-order radial suppression of ``off-critical-line'' information]\label{rem:app-offcritical-second-order-radial-suppression}
Corollary~\ref{cor:app-offcritical-radius-second-order-radial-channel} establishes a strict structural fact: under ``unit circle area phase'' compactification, the height endpoint depth \(\delta_{\gamma}\) is a first-order visible scale (\(|\gamma|\sim(\pi\delta_{\gamma})^{-1}\)), whereas the off-critical-line shift \(\tfrac12-\beta\) does not appear as a first-order phase perturbation but is instead compressed into the \emph{second-order radial term} \(\delta_{\gamma}^{2}\) of the endpoint depth.
Therefore, within a closure that admits only phase slices (unitary slice) and refuses radial degrees of freedom, the information about \(\beta-\tfrac12\) is structurally pushed into an invisible channel; this phenomenon will be manifested as the ``unaddressable'' hard rejection in the null-value (\(\mathsf{NULL}\)) semantics of \S\ref{subsec:xi-pw-type-safety-null}.
\end{remark}

\begin{remark}[Precision potential formulation: exponential compression factor of height]\label{rem:app-offcritical-radius-precision-potential}
By Definition~\ref{def:unit-circle-phase-precision-potential}, the precision potential satisfies
\(
e^{-\mathcal{P}(\gamma)}=\bigl(\pi(1+\gamma^{2})\bigr)^{-1}
\).
Therefore \eqref{eq:app-offcritical-radius-closed} can be rewritten as
\begin{equation}\label{eq:app-offcritical-radius-via-precision-potential}
\boxed{\;
1-|w_{\rho}|^{2}
=4\pi\,\delta\,e^{-\mathcal{P}(\gamma)}\cdot\frac{1+\gamma^{2}}{\gamma^{2}+(\delta+1)^{2}}\; }.
\end{equation}
In particular, when \(|\gamma|\to\infty\) with \(\delta\) bounded,
\(
1-|w_{\rho}|^{2}\sim 4\pi\,\delta\,e^{-\mathcal{P}(\gamma)}
\).
This equivalence precisely attributes the ``compression of disk visibility of the off-critical-line shift'' to the exponential factor \(e^{-\mathcal{P}(\gamma)}\) of the precision potential.
\end{remark}

\begin{remark}[Universal constant of the endpoint layer: logarithmic weak divergence of the \texorpdfstring{$T\log T$}{T log T} density under phase compactification]\label{rem:app-rvm-endpoint-constant}
Let \(N(T)\) be the counting function of nontrivial zeros of \(\zeta\) in the height window \(0<\Im(\rho)\le T\).
The Riemann--von Mangoldt formula gives the main term
\[
N(T)=\frac{T}{2\pi}\log\!\Bigl(\frac{T}{2\pi}\Bigr)-\frac{T}{2\pi}+O(\log T)\qquad(T\to\infty)
\]
(see \cite{Apostol1976}).
Write the main term as
\(
M(T):=\frac{T}{2\pi}\log(\frac{T}{2\pi})-\frac{T}{2\pi}
\), and using the parameter-free compactification \(u=\upsilon(T)=\frac{1}{\pi}\arctan(T)\) define the endpoint-compressed main term
\(
M_{\Phi}(u):=M(\tan(\pi u))
\).
Then as \(\delta:=\tfrac12-u\downarrow 0\) one has the limit
\begin{equation}\label{eq:app-rvm-endpoint-constant}
\boxed{\;
\lim_{\delta\downarrow 0}\;
\frac{\delta^{2}}{\log(1/\delta)}\,
\frac{\dd}{\dd u}M_{\Phi}\!\left(\tfrac12-\delta\right)
\;=\;\frac{1}{2\pi^{2}}\; }.
\end{equation}
This constant shows that under the phase compactification coordinate, the \(T\log T\) growth at ``infinite height'' is rewritten as a logarithmic weak divergence at the endpoint layer \(\delta\downarrow 0\), and its leading scale is stripped to a universal constant independent of sub-leading oscillations.
Geometrically, \(\delta\downarrow 0\) is equivalent to \(\vartheta(T)\to\pi\), i.e., the boundary point \(w=e^{\mathrm{i}\vartheta(T)}\to-1\) (Proposition~\ref{prop:app-cayley-upperhalf-disk}).
\end{remark}
\begin{proof}
From \(M'(T)=\frac{1}{2\pi}\log(\frac{T}{2\pi})\) and \(\frac{\dd}{\dd u}\tan(\pi u)=\pi(1+\tan^2(\pi u))\), one obtains
\[
\frac{\dd}{\dd u}M_{\Phi}(u)
=\frac{1}{2\pi}\log\!\Bigl(\frac{\tan(\pi u)}{2\pi}\Bigr)\cdot \pi\bigl(1+\tan^2(\pi u)\bigr).
\]
Taking \(u=\tfrac12-\delta\) and using the asymptotics \(\tan(\pi(\tfrac12-\delta))=\cot(\pi\delta)\sim \frac{1}{\pi\delta}\), one has
\(
1+\tan^2(\pi u)\sim \frac{1}{\pi^2\delta^2}
\)
and
\(
\log(\tan(\pi u))=\log(1/\delta)+O(1)
\).
Substituting back yields
\(
\frac{\dd}{\dd u}M_{\Phi}(\tfrac12-\delta)
=\frac{1}{2\pi^{2}\delta^{2}}\log(1/\delta)\,(1+o(1))
\), and hence \eqref{eq:app-rvm-endpoint-constant} holds. \qed
\end{proof}

\begin{proposition}[Average resolvable scale under endpoint compactification: dyadic bit depth of adjacent zeros]\label{prop:app-adjacent-zero-dyadic-resolution}
Let \(\rho_n=\beta_n+\mathrm{i}\gamma_n\) be the nontrivial zeros of \(\zeta\) (counted with multiplicity), ordered by increasing imaginary part \(\gamma_1\le \gamma_2\le\cdots\).
Take the endpoint phase compactification coordinate
\(
u_n:=\upsilon(\gamma_n)=\pi^{-1}\arctan(\gamma_n)\in(-\tfrac12,\tfrac12)
\)
(equation~\eqref{eq:unit-circle-phase-derivative}).
Then on the average scale as \(\gamma_n\sim T\to\infty\) one has
\begin{equation}\label{eq:app-adjacent-zero-u-gap-scale}
\boxed{\;
|u_{n+1}-u_n|\ \asymp\ \frac{1}{T^{2}\log T}\; }.
\end{equation}
Therefore, if dyadic box width \(2^{-b}\) is used as the endpoint resolution budget (i.e., giving the dyadic bracket of \(u\) with \(b\) bits of precision), then the minimum bit depth required to distinguish adjacent zeros satisfies
\begin{equation}\label{eq:app-adjacent-zero-dyadic-depth}
\boxed{\;
b_{n}:=\Bigl\lceil-\log_{2}|u_{n+1}-u_n|\Bigr\rceil
\;=\;2\log_{2}T+\log_{2}\log T+O(1)\qquad(T\to\infty)\; }.
\end{equation}
\end{proposition}

\begin{proof}
By the Riemann--von Mangoldt main term (Remark~\ref{rem:app-rvm-endpoint-constant}), the zero density at height \(T\) is of order
\(
N(T+H)-N(T)\asymp \frac{H}{2\pi}\log T
\)
(for \(1\ll H\ll T\) local windows), so the ``mean gap scale'' satisfies
\begin{equation}\label{eq:app-adjacent-zero-mean-gap-scale}
\gamma_{n+1}-\gamma_n\ \asymp\ \frac{2\pi}{\log T}
\qquad(\gamma_n\sim T\to\infty).
\end{equation}
On the other hand, \(\upsilon'(t)=\bigl(\pi(1+t^2)\bigr)^{-1}\) (equation~\eqref{eq:unit-circle-phase-derivative}), so by the mean value theorem there exists \(\xi_n\in[\gamma_n,\gamma_{n+1}]\) such that
\[
u_{n+1}-u_n=\upsilon'(\xi_n)\,(\gamma_{n+1}-\gamma_n)
=\frac{\gamma_{n+1}-\gamma_n}{\pi(1+\xi_n^{2})}.
\]
When \(\gamma_n\sim T\) and \(\gamma_{n+1}-\gamma_n\) is at the above average scale, one also has \(\xi_n\sim T\), whence \((1+\xi_n^2)^{-1}\asymp T^{-2}\).
Combining with \eqref{eq:app-adjacent-zero-mean-gap-scale} yields \eqref{eq:app-adjacent-zero-u-gap-scale}.
Finally, taking \(-\log_2\) of \eqref{eq:app-adjacent-zero-u-gap-scale} and using \(-\log_2(T^{-2}(\log T)^{-1})=2\log_2 T+\log_2\log T+O(1)\) yields \eqref{eq:app-adjacent-zero-dyadic-depth}. \qed
\end{proof}
