\section{Worked motivic examples: periods with Frobenius-trace data}
\label{app:motivic_worked_examples}

Section~\ref{sec:langlands_compatibility} formulates a compatibility target in which the closed functor $\HSP:\ScanAlg\to\PerDatum$ is followed by (programmatic) motive and realization arrows. The goal of this appendix is to anchor that target by recording two classical, fully worked test cases in which:
\begin{itemize}
\item a period datum is canonically associated to a standard geometric source;
\item the same source carries $\ell$-adic realizations with Frobenius traces (hence an Euler product and an $L$-function);
\item the corresponding automorphic object is explicit.
\end{itemize}
Nothing in this appendix is used in the closed proof chain of the paper; it serves only as a controlled reference point for what the arrows in \eqref{eq:diagram_objective} mean in standard arithmetic geometry.

\subsection{Mixed Tate test case: \texorpdfstring{$\mathbb{P}^1\setminus\{0,1,\infty\}$}{P1 minus 0,1,infty} and iterated integrals}
\label{app:mixed_tate_test_case}

Let $X:=\mathbb{P}^1\setminus\{0,1,\infty\}$. A standard period calculus on $X$ is generated by the logarithmic $1$-forms
\[
\omega_0:=\frac{\dd t}{t},
\qquad
\omega_1:=\frac{\dd t}{1-t},
\]
and their iterated integrals along paths between rational points. These iterated integrals are periods of the (unipotent) fundamental group of $X$ and lie in the period algebra of mixed Tate motives over $\ZZ$ (and more generally over $\ZZ[1/N]$ when endpoints involve $N$) \cite{KontsevichZagier2001,Brown2012,DeligneGoncharov2005}.

\paragraph{A rational-endpoint period for $\log 2$.}
The basic scan-period example $\int_0^1 (1+x)^{-1}\dd x=\log 2$ can be rewritten as an iterated-integral period on $X$ with rational endpoints. Indeed, with the change of variables $x=t/(1-t)$ one has $\dd x/(1+x)=\dd t/(1-t)$ and
\begin{equation}
\label{eq:log2_iterated_integral}
\log 2
=\int_0^1 \frac{\dd x}{1+x}
=\int_0^{1/2}\frac{\dd t}{1-t}
=\int_{0}^{1/2}\omega_1.
\end{equation}
Thus $\log 2$ is realized as a period datum $(D,f,\mathrm{Id})$ with $D=[0,1/2]\subset\RR$ and $f(t)=1/(1-t)\in\QQ(t)$, which lies in the controlled cubical setting of this paper.

\paragraph{Multiple zeta values as periods on $X$.}
Multiple zeta values admit a well-known iterated integral representation on $X$ (see, e.g., \cite{KontsevichZagier2001,Brown2012}): for integers $n\ge 2$,
\begin{equation}
\label{eq:zeta_n_iterated_integral}
\zeta(n)=\int_0^1 \omega_1\,\omega_0^{n-1}.
\end{equation}
In particular,
\begin{equation}
\label{eq:zeta2_iterated_integral}
\zeta(2)=\int_0^1 \omega_1\,\omega_0
=\int_{0<t_1<t_2<1}\frac{\dd t_1}{1-t_1}\frac{\dd t_2}{t_2},
\end{equation}
which is a Kontsevich--Zagier period with a $\QQ$-semialgebraic domain (a simplex) and a rational integrand.

\paragraph{Interpretation for the compatibility target.}
In this test case, the ``motive'' arrow in \eqref{eq:diagram_objective} can be instantiated within the standard mixed Tate framework: the motivic source is a mixed Tate object attached to $X$ (e.g.\ via the unipotent fundamental group), and its periods include the iterated integrals above. The point of recording \eqref{eq:log2_iterated_integral}--\eqref{eq:zeta2_iterated_integral} is that they provide a concrete bridge from our cubical period data to a motivic organization where period relations are governed by a well-studied structure \cite{Brown2012,DeligneGoncharov2005}.

\subsection{Modular/elliptic test case: weight-2 newforms, periods, and Frobenius traces}
\label{app:modular_elliptic_test_case}

Let $f(z)=\sum_{n\ge 1} a_n\,q^n$ be a normalized weight-$2$ newform on $\Gamma_0(N)$, with $q=e^{2\pi \iu z}$. Its $L$-series is
\[
L(f,s):=\sum_{n=1}^\infty \frac{a_n}{n^s},
\qquad \Re(s)\gg 1.
\]
The Mellin transform gives the classical period integral expression (see, e.g., \cite{Bump1997,DiamondShurman2005})
\begin{equation}
\label{eq:mellin_weight2}
(2\pi)^{-s}\Gamma(s)\,L(f,s)
=\int_0^\infty f(\iu t)\,t^{s-1}\,\dd t,
\end{equation}
and in particular at $s=1$,
\begin{equation}
\label{eq:lf1_period}
L(f,1)
=2\pi\int_0^\infty f(\iu t)\,\dd t.
\end{equation}
Equivalently, the holomorphic differential $\omega_f:=2\pi \iu f(z)\,\dd z$ on the modular curve $X_0(N)$ has periods given by integrals of $\omega_f$ along $1$-cycles (modular symbols), and these periods control special values such as $L(f,1)$.

\paragraph{Frobenius traces from $\ell$-adic realizations.}
Deligne attaches to $f$ an $\ell$-adic Galois representation $\rho_{f,\ell}$ whose Frobenius traces recover the Hecke eigenvalues \cite{Deligne1971Bourbaki,Deligne1974}. Concretely, for primes $p\nmid N\ell$,
\begin{equation}
\label{eq:frob_trace_ap}
\mathrm{Tr}\big(\rho_{f,\ell}(\Frob_p)\big)=a_p,
\end{equation}
and the Euler factors of $L(f,s)$ are determined by these traces.

\paragraph{Elliptic curve specialization.}
When the Hecke field of $f$ is $\QQ$, the modular abelian variety attached to $f$ is an elliptic curve $E/\QQ$ and the motive may be identified with $H^1(E)$; in this case \eqref{eq:frob_trace_ap} specializes to the familiar point-counting identity
\[
a_p=p+1-\#E(\mathbb{F}_p),
\]
and the period integrals of the N\'eron differential on $E$ yield the real/imaginary periods of $E$. The equality $L(E,s)=L(f,s)$ provides a concrete instance of the ``period--motive--Frobenius'' chain in \eqref{eq:diagram_objective}.

\paragraph{Interpretation for the compatibility target.}
This modular test case is a regime in which the arrows ``period data $\to$ motive $\to$ $\ell$-adic realization'' and ``motive $\to$ automorphic representation'' are classical. It therefore provides a canonical benchmark for any future attempt to connect scan-realized period data (via $\HSP$) with Hecke/Frobenius trace data extracted from protocol-level constructions.
