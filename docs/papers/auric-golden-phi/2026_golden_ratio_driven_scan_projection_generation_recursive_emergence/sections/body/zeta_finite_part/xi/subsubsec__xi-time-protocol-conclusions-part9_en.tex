\paragraph{Ap\'ery Reduction of the Tail-Additive Closure: Two-Generator Normal Form, Congruence Truncation Ledger, and Reset Clock Moment Invariants}
This paragraph, building on the tail-additive mechanism of Theorem \ref{thm:xi-window6-tail-semigroup-frobenius-threshold}, further reduces reachability from a "generating set" to a family of congruence thresholds, and provides the equivalent normal form, a finite truncation partition of the unreachable set, and moment invariants of the reset interval clock.
\par

\begin{theorem}[Two-Generator Reduction and Canonical Solution of the Tail-Additive Mechanism]\label{thm:xi-window6-tail-semigroup-two-generator-normal-form}
Under the tail offset additive closure framework of the window-\(6\) uplift, let the additive generating set be \(\{21,34,55\}\).
Since \(55=21+34\), the corresponding reachable semigroup satisfies
\[
S_6=\langle 21,34,55\rangle=\langle 21,34\rangle\subset\mathbb N.
\]
For any \(n\in\mathbb N\), let \(r\equiv n\ (\mathrm{mod}\ 21)\).
Define
\[
b_r:=13r\ (\mathrm{mod}\ 21)\in\{0,1,\dots,20\},
\qquad
w_r:=34\,b_r,
\]
then \(n\in\langle 21,34\rangle\) if and only if \(n\ge w_r\).
Moreover, in the reachable case, there exists a unique pair
\(
(a,b)\in\mathbb N\times\{0,1,\dots,20\}
\)
satisfying
\[
n=21a+34b,\qquad b=b_r,\qquad a=\frac{n-w_r}{21}.
\]
Therefore, reachability admits a constant-time criterion on each congruence class \(r\), and induces a unique "mod \(21\) normal form" (reducing \(b\) to \(\{0,\dots,20\}\)).
\end{theorem}

\begin{proof}
The identity \(55=21+34\) implies \(\langle 21,34,55\rangle=\langle 21,34\rangle\).
Since \(34\equiv 13\ (\mathrm{mod}\ 21)\) and \(13^2\equiv 1\ (\mathrm{mod}\ 21)\), \(13\) is the modular inverse of \(34\) modulo \(21\), so
\(
34\,b_r\equiv r\ (\mathrm{mod}\ 21)
\)
holds, and \(b_r\) is unique within \(\{0,\dots,20\}\).
\par
If \(n=21a+34b\) is reachable, then necessarily \(b\equiv b_r\ (\mathrm{mod}\ 21)\), so \(b=b_r+21k\) (\(k\in\mathbb N\)), which gives
\(
n=w_r+21(a+34k)\ge w_r
\).
Conversely, if \(n\equiv r\ (\mathrm{mod}\ 21)\) and \(n\ge w_r\), then \(n-w_r\) is a nonnegative multiple of \(21\); setting \(a=(n-w_r)/21\ge 0\) and \(b=b_r\) yields the decomposition.
Uniqueness is forced by the reduction condition \(b\in\{0,\dots,20\}\): if \(21a+34b=21a'+34b'\) with \(b,b'\in\{0,\dots,20\}\), then \(34(b-b')\) is a multiple of \(21\), and since \(\gcd(34,21)=1\), we get \(b=b'\), hence \(a=a'\). This completes the proof.
\end{proof}

\paragraph{Bounded Drift and Constant-Time Computability of the Sturmian Reset Clock}

\begin{proposition}[Bounded Drift of the Two-Interval Sturmian Reset Clock: Jitter Upper Bound Equals Gap Difference]\label{prop:xi-window6-reset-clock-bounded-drift}
Let the reset event sequence of window-\(6\) be as in Theorem \ref{thm:terminal-reset-events-sturmian}, and denote the consecutive reset intervals as
\[
G_n:=r_{n+1}-r_n\in\{34,55\}\qquad(n\ge 1).
\]
Define the partial sum (reset time increment)
\[
T_n:=\sum_{j=1}^{n}G_j\qquad(n\ge 1),
\]
and let the mean drift constant
\[
\mu:=\mathbb E[G]=\frac{55}{\varphi}+\frac{34}{\varphi^2}=13+21\varphi
\]
be as in Proposition \ref{prop:xi-window6-reset-gap-mgf-moments}.
Then there exists an intercept parameter \(\theta\in[0,1)\) such that the long-interval count
\[
S_n:=\#\{1\le j\le n:\ G_j=55\}
\]
satisfies \(S_n=\lfloor n/\varphi+\theta\rfloor\), and the drift deviation has uniform boundedness
\[
T_n-n\mu=21\Bigl(S_n-\frac{n}{\varphi}\Bigr),
\qquad
\bigl|T_n-n\mu\bigr|\le 21
\quad(\forall n\ge 1).
\]
Hence this reset clock has deterministic "zero-diffusion" jitter at all scales: the error deviating from the mean linear drift does not grow with \(n\), and its worst-case amplitude equals precisely the gap difference \(55-34=21\).
\end{proposition}

\begin{proof}
By Theorem \ref{thm:terminal-reset-events-sturmian}, the indicator sequence \(\ind_{\{G_j=55\}}\) is a Sturmian sequence of slope \(\varphi^{-1}\) (equivalent to reading out an interval window from an irrational rotation).
The balance property of Sturmian sequences gives: there exists \(\theta\in[0,1)\) such that
\(
S_n=\lfloor n/\varphi+\theta\rfloor
\)
and \(|S_n-n/\varphi|\le 1\).
On the other hand, the identity \(T_n=55S_n+34(n-S_n)=34n+21S_n\) and \(\mu=34+21/\varphi\) imply
\(
T_n-n\mu=21(S_n-n/\varphi)
\);
from \(|S_n-n/\varphi|\le 1\), we get \(|T_n-n\mu|\le 21\). This completes the proof.
\end{proof}

\begin{corollary}[Constant-Time Computability of the Reset Clock: Beatty Rounding and Scan-Free Update]\label{cor:xi-window6-reset-clock-O1-computable}
Under the notation of Proposition \ref{prop:xi-window6-reset-clock-bounded-drift}, given \(n\), one can compute \(T_n\) in constant time:
let \(S_n=\lfloor n/\varphi+\theta\rfloor\), then
\[
T_n=34n+21S_n.
\]
Thus the reset time set \(\{T_n\}\) can be realized by a single Beatty rounding, without enumerating \((G_1,\dots,G_n)\).
\end{corollary}

\begin{remark}[Rotation Representation and Congruence Invariants of the Drift Residual]\label{rem:xi-window6-reset-clock-residual-rotation-congruence}
From \(S_n=\lfloor n/\varphi+\theta\rfloor\), we have
\[
S_n-\frac{n}{\varphi}=\theta-\Bigl\{\frac{n}{\varphi}+\theta\Bigr\}\in(\theta-1,\theta],
\qquad
T_n-n\mu=21\Bigl(\theta-\Bigl\{\frac{n}{\varphi}+\theta\Bigr\}\Bigr),
\]
where \(\{\cdot\}\) denotes the fractional part.
Since \(1/\varphi\) is irrational, \(\{\frac{n}{\varphi}+\theta\}\) is dense in \([0,1)\), so the drift residual \(\{T_n-n\mu\}_{n\ge 1}\) is dense in some interval of length \(21\).
On the other hand, the sign of \(\theta-\{\frac{n}{\varphi}+\theta\}\) is determined solely by whether
\(\{\frac{n}{\varphi}+\theta\}\in[0,\theta]\) or \((\theta,1)\);
hence the \emph{sign semantics} of the drift residual can be compressed under a binary budget (corresponding to the local pattern dichotomy under binary window readout in the Sturmian sequence).
Moreover, since \(34\equiv 55\equiv 13\ (\mathrm{mod}\ 21)\), we have
\[
T_n\equiv 13n\ (\mathrm{mod}\ 21),
\]
so the reset times exhibit a fully deterministic finite residue-class structure modulo \(21\).
\end{remark}

\begin{corollary}[Reachability Coupling of the Reset Clock and Tail Semigroup: Finite Obstruction Only at Shallow Scales]\label{cor:xi-window6-reset-clock-semigroup-coupling}
Let \(S=\langle 21,34\rangle\).
Then for any integer \(t\ge 715\), there exist \(n\in\mathbb N\) and \(s\in S\) such that
\[
t=T_n+s
\qquad\text{and}\qquad
\bigl|t-(n\mu+s)\bigr|\le 21,
\]
where \(\mu\) and \(\{T_n\}\) are as in Proposition \ref{prop:xi-window6-reset-clock-bounded-drift}.
Hence at deep scales, "the reset clock (Sturmian)" and "the tail reachability semigroup (Ap\'ery)" form a deterministic near-cofinal structure; true unreachability obstructions can only appear at finitely many shallow scales.
\end{corollary}

\begin{proof}
By Theorem \ref{thm:xi-window6-tail-semigroup-two-generator-normal-form} (or equivalently by cofiniteness of \(\gcd(21,34)=1\) from Frobenius), we have \(n\ge 660\Rightarrow n\in S\).
Take any \(t\ge 715\).
Note that \(T_1\in\{34,55\}\), so \(s:=t-T_1\ge 660\), from which \(s\in S\) and \(t=T_1+s\).
On the other hand, by Proposition \ref{prop:xi-window6-reset-clock-bounded-drift} (taking \(n=1\)),
\(
|T_1-\mu|\le 21
\), so
\(
|t-(\mu+s)|=|T_1-\mu|\le 21
\).
This completes the proof.
\end{proof}

\begin{corollary}[Parity Closed Form of the Maximum Fiber and Finite-State Spectrum of the Maximum-Achiever Hidden-Bit Splitting]\label{cor:xi-max-fiber-parity-hiddenbit-finite-spectrum}
Let \(D_m:=\max_{x\in X_m}d_m(x)\).

\[
D_{2k}=F_{k+2},\qquad D_{2k+1}=2F_{k+1}

For any maximum achiever \(x\in X_m\) (i.e., \(d_m(x)=D_m\)), define the hidden-bit splitting \((d_{m,0}(x),d_{m,1}(x))\) (Remark \ref{rem:pom-max-fiber-achievers-bsplit}); then within the stable threshold interval, the splitting pair satisfies a closed-form finite-state structure:
\begin{itemize}
  \item \textbf{Even \(m=2k\ge 10\):}
  \[
  (d_{2k,0}(x),d_{2k,1}(x))\in\{(F_{k+1},F_k),(F_k,F_{k+1})\}.
  \]
  \item \textbf{Odd \(m=2k+1\ge 13\):}
  \[
  (d_{2k+1,0}(x),d_{2k+1,1}(x))\in\{(F_{k+1},F_{k+1}),\ (F_{k+1}\pm F_{k-2},\,F_{k+1}\mp F_{k-2})\}.

p_1(x)\in\left\{\frac12,\ \frac12\pm\frac{F_{k-2}}{2F_{k+1}}\right\},

\begin{proof}
The parity closed form of the maximum fiber is given by Theorem \ref{thm:pom-max-fiber}.
The closed-form finite-state structure of achiever splitting pairs is from Remark \ref{rem:pom-max-fiber-achievers-bsplit}, and the three-state spectrum and limiting spectrum are from Corollary \ref{cor:pom-max-fiber-achievers-hidden-bit-prob-spectrum}.
The asymptotics follow from Binet's formula: \(F_{k-2}/F_{k+1}=\varphi^{-3}+O(\varphi^{-2k})\). This completes the proof.
