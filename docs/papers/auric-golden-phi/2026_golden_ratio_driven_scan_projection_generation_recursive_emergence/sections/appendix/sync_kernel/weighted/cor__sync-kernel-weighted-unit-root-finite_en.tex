\begin{corollary}[Finite conjugacy of unit-root twists: reducing "worst-case mode" to finite algebraic comparison]\label{cor:sync-kernel-weighted-unit-root-finite}

\end{corollary}

\begin{proposition}[Functional equation of the completion determinant]\label{prop:kernel-xi-functional-equation}
Assume the sync kernel satisfies the self-dual identity $\Delta(z,u)=\Delta(uz,1/u)$ (see \S\ref{app:kernel-self-dual}), and take
\[

\]
Define the univariate function along the curve
\[

\]
by
\[
\Xi(s):=\Delta(z(s),u(s))=\Delta(\lambda^{-s},\lambda^{2s-1}).
\]
Then the strict functional equation holds:
\[
\boxed{\ \Xi(s)=\Xi(1-s)\ }.
\]
Equivalently, define the same section of $\zeta$:
\[
\zeta_{\mathrm{sd}}(s):=\zeta(z(s),u(s))=\Delta(\lambda^{-s},\lambda^{2s-1})^{-1},
\]
then $\zeta_{\mathrm{sd}}(s)=\zeta_{\mathrm{sd}}(1-s)$.
\end{proposition}
\begin{proof}

\[

=\Delta(\lambda^{2s-1}\lambda^{-s},\lambda^{1-2s})

\(
z(1-s)=\lambda^{-(1-s)}=\lambda^{s-1}
\)
and
\(

so $\Xi(s)=\Delta(z(1-s),u(1-s))=\Xi(1-s)$. Proof complete.
\end{proof}

\subsubsection{Weighted Mertens constant and Abel finite part}\label{app:mertens-weighted}
Let $\lambda(u)$ be the Perron root, $\rho(u)$ be the maximum modulus of the remaining eigenvalues, then
\[
a_n(u)=\mathrm{Tr}(B(u)^n)=\lambda(u)^n+O(\rho(u)^n),
\]
by Witt inversion we get
\[
p_n(u)=\frac{\lambda(u)^n}{n}+O\!\left(\frac{\rho(u)^n}{n}\right).
\]
Thus
\[
\Pi_\theta(N):=\sum_{n\le N}p_n(\theta)\sim \frac{\lambda(\theta)}{\lambda(\theta)-1}\cdot\frac{\lambda(\theta)^N}{N}.
\]
Write
\[
\mathcal{M}_\theta(N):=\sum_{n\le N}\frac{p_n(u)}{\lambda(u)^n},
\]
then
\[
\mathcal{M}_\theta(N)=\log N+M(\theta)+o(1),
\qquad
M(\theta)=\gamma+\sum_{n\ge 1}\left(\frac{p_n(u)}{\lambda(u)^n}-\frac{1}{n}\right),
\]

\begin{proposition}[Evenness of Mertens constant]\label{prop:mertens-even}
From $p_n(u)=u^n p_n(1/u)$ and $\lambda(u)=u\lambda(1/u)$ we get
\[
\frac{p_n(u)}{\lambda(u)^n}=\frac{p_n(1/u)}{\lambda(1/u)^n},
\]
so
\[
M(\theta)=M(-\theta),\qquad
\log\mathfrak{M}(\theta)=\log\mathfrak{M}(-\theta).
\]
\end{proposition}

\[
P_\theta(r):=\sum_{n\ge 1}\frac{p_n(u)}{\lambda(u)^n}r^n,\quad 0<r<1,
\]
then the Abel finite part gives
\[
\log\mathfrak{M}(\theta):=\mathrm{fp}_{r=1}\bigl(P_\theta(r)+\log(1-r)\bigr)
=\sum_{n\ge 1}\left(\frac{p_n(u)}{\lambda(u)^n}-\frac{1}{n}\right),
\]

\[
M(\theta)=\gamma+\log\mathfrak{M}(\theta).
\]

\[
M(0)\approx 0.2738887860704445,\qquad
\log\mathfrak{M}(0)\approx -0.3033268788310883.
