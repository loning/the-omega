\subsection{Holo3: regional protocol capacity}
\label{subsec:holo3_regional_capacity}

\paragraph{Per-site capacity convention.}
\InterfaceTag At fixed resolution $m$, a local readout alphabet has size $|\Omega_m|=2^m$.
When the boundary is modeled quantum mechanically, the corresponding local Hilbert register is taken to have dimension
\[
d_{\mathrm{site}}(m)=2^m.
\]
This is the same finite-cardinality convention used throughout the manuscript when attaching screen statistics
to window-length $m$ (Section~\ref{sec:introduction} and Section~\ref{sec:hilbert_addressing}).

\begin{definition}[Regional protocol capacity]
\label{def:regional_protocol_capacity}
\InterfaceTag For a screen region $R\subseteq S_n$, define its protocol capacity (in bits) as
\[
I_{\mathrm{prot}}(m,n;R):=m\,|R|.
\]
For the full screen $S_n$ (with $|S_n|=4^n$ for the 2D Hilbert screen), this reduces to the global capacity
\[
I_{\mathrm{prot}}(m,n):=m4^n,
\]
as used in the capacity-first uplift staging module (Appendix~\ref{app:resolution_uplift_fusion_horizon_unification}).
\end{definition}

\subsection{Holo3: entropy upper bounds from finite boundary dimension (PT)}
\label{subsec:holo3_entropy_bounds}

\paragraph{Classical and quantum entropy conventions.}
In this appendix, all entropies are measured in \emph{bits}, i.e.\ with $\log_2$.

\begin{definition}[Boundary entropy of a region (finite resolution)]
\label{def:holo_boundary_entropy}
\MathTag Let $R\subseteq S_n$.
\begin{itemize}
\item \textbf{Classical readout:} if the boundary state on $R$ is a probability mass function $p$ on a finite outcome set
$\Omega_R$, define the Shannon entropy
\[
H_{\partial}(R):=-\sum_{\omega\in\Omega_R} p(\omega)\log_2 p(\omega).
\]
\item \textbf{Quantum readout:} if the boundary state on $R$ is a density operator $\rho_R$ on a finite-dimensional Hilbert space
$\mathcal{H}_R$, define the von Neumann entropy
\[
S_{\partial}(R):=-\Tr(\rho_R\log_2\rho_R).
\]
\end{itemize}
\end{definition}

\begin{proposition}[Finite-dimensional entropy bounds (bits)]
\label{prop:holo_entropy_upper_bound_finite_dim}
\MathTag In the setting of Definition~\ref{def:holo_boundary_entropy} one has:
\begin{enumerate}
\item \textbf{Classical:} $H_{\partial}(R)\le \log_2|\Omega_R|$.
\item \textbf{Quantum:} $S_{\partial}(R)\le \log_2\dim(\mathcal{H}_R)$.
\end{enumerate}
\end{proposition}
\begin{proof}
Both inequalities are standard finite-dimensional facts.
For the classical case, the Shannon entropy is maximized by the uniform distribution.
For the quantum case, the von Neumann entropy is maximized by the maximally mixed state $\rho_R=\ind/\dim(\mathcal{H}_R)$.
\end{proof}

\begin{corollary}[Protocol capacity bound (area-law surrogate)]
\label{cor:holo_protocol_capacity_bound}
\InterfaceTag Under the per-site capacity convention $d_{\mathrm{site}}(m)=2^m$ and the tensor-product locality subclass
(Appendix~\ref{app:protocol_subclass_tensor_net}), a region $R$ carries a Hilbert space
$\mathcal{H}_R\cong (\CC^{2^m})^{\otimes |R|}$ and therefore
\[
S_{\partial}(R)\ \le\ \log_2\dim(\mathcal{H}_R)\ =\ \log_2\Bigl((2^m)^{|R|}\Bigr)\ =\ m|R|\ =\ I_{\mathrm{prot}}(m,n;R).
\]
\end{corollary}
\begin{proof}
Combine Proposition~\ref{prop:holo_entropy_upper_bound_finite_dim} with Definition~\ref{def:regional_protocol_capacity}.
\end{proof}
