\subsection{Explicit Algorithms and Complexity Bounds}\label{subsec:zeckendorf-algorithms-complexity}

This subsection promotes the ``computable, terminating'' statement to \textbf{explicit algorithms} and \textbf{complexity bounds}. To keep dependencies minimal, the algorithms adopt the shortest closed loop: \textbf{value-level operations} (integer addition/multiplication) + \textbf{Zeckendorf normalization} (greedy).

\subsubsection{Zeckendorf normalization: from integers to legal digits}

Given $N\in\NN$, the goal is to compute its unique Zeckendorf digits $Z(N)=(c_1,c_2,\dots)$ (only finitely many $1$'s and no adjacent $1$'s) such that $N=\sum_{k\ge 1}c_kF_{k+1}$.

\begin{itemize}
  \item \textbf{Input:} Integer $N$.
  \item \textbf{Output:} Zeckendorf-legal digit sequence $Z(N)$.
\end{itemize}

\textbf{Algorithm (greedy + adjacent-position skip):}
\begin{enumerate}
  \item Pre-generate Fibonacci weights $F_{2},F_{3},\dots,F_{M+1}$, where $M$ is large enough so that $F_{M+1}\le N<F_{M+2}$;
  \item From $k=M$ decreasing to $1$: if $F_{k+1}\le N$, set $c_k:=1$, update $N\leftarrow N-F_{k+1}$, and skip the adjacent position (set $k\leftarrow k-1$ and continue decreasing); otherwise set $c_k:=0$;
  \item Output $c=(c_1,c_2,\dots,c_M)$ (all subsequent positions are $0$).
\end{enumerate}

The correctness and uniqueness of this algorithm follow from the Zeckendorf theorem (\cite{Zeckendorf1972}; see also Appendix~C of this paper).

\subsubsection{Stable addition and stable multiplication: two-step implementation}

For inputs $c,d\in X_\infty$ (only finitely many $1$'s), let the effective resolution be
\[
m:=\max\{k:\ c_k=1\text{ or }d_k=1\}.
\]
Then:
\begin{itemize}
  \item \textbf{Stable addition} $c\oplus d$: compute $N=\mathrm{Val}(c)+\mathrm{Val}(d)$, then output $Z(N)$;
  \item \textbf{Stable multiplication} $c\otimes d$: compute $N=\mathrm{Val}(c)\mathrm{Val}(d)$, then output $Z(N)$.
\end{itemize}
This is precisely the ``value-level operation + normal form'' shortest closed-loop implementation from Appendix~E.

\subsubsection{Complexity (two calibers: by resolution \texorpdfstring{$m$}{m} and by bit length)}

To avoid model controversies, Appendix~E$'$ simultaneously gives complexity bounds in two equivalent calibers: resolution caliber (treating $m$ as the input length) and bit caliber (measuring cost in terms of binary length of the integers). The conclusions can be summarized as follows (detailed proofs in Propositions~\ref{prop:fold-complexity} and~\ref{prop:stable-arith-complexity} of Appendix~E$'$):

\begin{itemize}
  \item \textbf{Folding/normalization:} For an input of length $m$, Zeckendorf normalization takes $O(m)$ steps in the resolution caliber; $O(m^2)$ bit operations in the bit caliber.
  \item \textbf{Stable addition:} Resolution caliber $O(m)$; bit caliber $O(m^2)$.
  \item \textbf{Stable multiplication:} Resolution caliber is ``value-level multiplication cost + $O(m)$''; bit caliber is $O(M(m)+m^2)$, where $M(m)$ is the cost of multiplying $\Theta(m)$-bit integers (schoolbook multiplication gives $O(m^2)$).
\end{itemize}

\subsubsection{Single-pass on-line delay \texorpdfstring{$3$}{3}: unified transducer}\label{subsec:online-delay}
There exists a \textbf{single-pass on-line} transducer (delay $3$) that directly implements the normalization; its synchronization part can be given via ``remainder conservation'' (\cite{Frougny1999OnlineAddition}).

Let $\varphi=(1+\sqrt5)/2$, and define the remainder
\[
R(s):=s_1\varphi^{-1}+s_2\varphi^{-2}+s_3\varphi^{-3}\in[0,1),
\]
with input $d\in\{0,1,2\}$ and output $e\in\{0,1\}$. The synchronization edge $(s\to t)$ satisfies the conservation equation
\[
R(s)+d\varphi^{-4}=e\varphi^{-1}+t_1\varphi^{-2}+t_2\varphi^{-3}+t_3\varphi^{-4},
\]
equivalently
\[
R(t)=\varphi R(s)+d\varphi^{-3}-e,\qquad R(t)\in[0,1).
\]
\begin{remark}[Expansion--rounding representation of the delay-$3$ normalization]
From $R(t)\in[0,1)$ one can write
\[
e=\Bigl\lfloor \varphi R(s)+d\varphi^{-3}\Bigr\rfloor,\qquad
R(t)=\Bigl\{\varphi R(s)+d\varphi^{-3}\Bigr\},
\]
where $\{\cdot\}$ denotes the fractional part. Since the synchronization kernel only allows $e\in\{0,1\}$ (the transition table in Appendix~B contains no $e=2$ edges), the above on-line kernel is equivalent to a dynamical system on $[0,1)$ generated by the expanding map $r\mapsto \varphi r+d\varphi^{-3}$, projected back by the rounding cocycle. This representation directly encodes the ``on-line delay'' as the serial structure of multiplicative expansion and branch selection.
\end{remark}
The state set for the synchronization part can be taken as
\[
Q=\{000,001,002,010,100,101,0\overline{1}2,1\overline{1}2,01\overline{1},11\overline{1}\},
\]
with outgoing edges (labeled $d/e$) given by:
\[
\begin{aligned}
000&\xrightarrow{d/0}00d,\qquad 100\xrightarrow{d/1}00d\quad(d=0,1,2),\\
001&\xrightarrow{0/0}010,\ \xrightarrow{1/0}100,\ \xrightarrow{2/0}101,\\
002&\xrightarrow{0/0}11\overline{1},\ \xrightarrow{1/1}000,\ \xrightarrow{2/1}001,\\
010&\xrightarrow{0/0}100,\ \xrightarrow{1/0}101,\ \xrightarrow{2/1}0\overline{1}2,\\
101&\xrightarrow{0/1}010,\ \xrightarrow{1/1}100,\ \xrightarrow{2/1}101,\\
0\overline{1}2&\xrightarrow{0/0}01\overline{1},\ \xrightarrow{1/0}010,\ \xrightarrow{2/0}100,\\
1\overline{1}2&\xrightarrow{0/1}01\overline{1},\ \xrightarrow{1/1}010,\ \xrightarrow{2/1}100,\\
01\overline{1}&\xrightarrow{0/0}001,\ \xrightarrow{1/0}002,\ \xrightarrow{2/0}1\overline{1}2,\\
11\overline{1}&\xrightarrow{0/1}001,\ \xrightarrow{1/1}002,\ \xrightarrow{2/1}1\overline{1}2.
\end{aligned}
\]
The terminal function can be taken as
\[
\omega(s_1s_2s_3):=Z(s_1F_4+s_2F_3+s_3F_2),
\]
to flush the residual of the delay window and land back in Zeckendorf normal form (\cite{Frougny1999OnlineAddition}).

\begin{theorem}[Single-pass on-line delay-$3$ implementation of $\Fold_\infty$]\label{thm:online-delay-fold}
The above on-line transducer (including the terminal function) outputs $\Fold_\infty(s)$ for any $s\in\{0,1,2\}^m$; its synchronization graph is the 10-state synchronization kernel of Appendix~B.
\end{theorem}

\textbf{Proof:}
The conservation equation ensures value invariance; the terminal function outputs elements of $X_m$; uniqueness of $\Fold_\infty(s)$ follows from the transversal uniqueness of Theorem~\ref{thm:zeckendorf-transversal}. The synchronization graph matches the states/transitions of Appendix~B. \qed

\begin{remark}[Primitive orbits and ``prime shadows'']\label{rem:primitive-shadow}
The synchronization graph of the above on-line transducer is a finite directed graph whose primitive periodic orbits give the ``irreducible cyclic patterns.'' Let $B$ be its adjacency matrix; then the periodic point count $a_n=\mathrm{Tr}(B^n)$ and the primitive orbit count $p_n$ are related by

a_n=\sum_{d\mid n} d\,p_d,\qquad
p_n=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d},

consistent with the $\zeta$--Euler product interface of Section~\ref{sec:zeta-finite-part} (see \cite{LindMarcus1995} and the explicit decomposition and numerical spectrum in Appendix~B).

\begin{remark}[Prime shadows vs.\ arithmetic primes: hierarchical difference]\label{rem:prime-shadow-vs-primes}
The primitive orbit spectrum of a finite kernel is a prime-like object at the dynamical system level, strictly determined by the kernel's $\zeta$-function and spectrum; it is not identical to the set of integer primes. Any fixed-resolution / finite-state readout can carry only finite information and hence cannot be semantically equivalent to ``integer primes.'' To introduce ``primes'' as ontological structure, one would need to import primes as atomic labels of an operator family at a higher level: for example, in the Hecke correspondence framework, taking $T_p$ as a generating operator and reading off $a_p$ on the eigenbasis; in that case, the prime register is an operator register rather than a factorization register. The synchronization kernel spectrum and $\zeta$-computation of this paper provide the dynamical template layer, serving as a pedestal for such upgrading interfaces.
\end{remark}

\subsubsection{Commuting diagram of time/space duality: strict consistency of trace--iterate and Euler product}\label{subsec:time-space-commuting-diagram}
This section formalizes the duality between the time side (iteration/trace) and the space side (primitive/Euler product) as a strictly commutative diagram. The key point is: at the level of finite matrices (or trace-class operators), $\det(I-zT)$, $\mathrm{Tr}(T^n)$, and primitive extraction (M\"obius/Witt) are all conjugacy invariants, and hence can be functorially organized into an auditable commutative diagram.

\paragraph{Three categories (retaining only conjugacy invariants)}
Take the following three types of objects:
\begin{itemize}
  \item \textbf{Endomorphism category $\mathbf{End}^{\simeq}$}: objects are pairs $(V,T)$ (finite-dimensional linear space and operator $T:V\to V$), morphisms are similarity transformations (conjugation);
  \item \textbf{Sequence category $\mathbf{Seq}$}: objects are sequences $a=(a_n)_{n\ge 1}$ (valued in $\CC$), taken with the discrete category caliber;
  \item \textbf{Euler category $\mathbf{Euler}$}: objects are formal power series $F(z)\in 1+z\CC[[z]]$ (taken with the discrete category caliber).
\end{itemize}

\paragraph{Four functors (time side, atomization, Euler product, and determinant $\zeta$)}
Define:
\[
\mathcal{T}:\mathbf{End}^{\simeq}\to\mathbf{Seq},\qquad
\mathcal{T}(T)=(a_n)_{n\ge 1},\ a_n:=\mathrm{Tr}(T^n),
\]
\[
\mathcal{M}:\mathbf{Seq}\to\mathbf{Seq},\qquad
(\mathcal{M}a)_n:=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d},
\]
\[
\mathcal{E}:\mathbf{Seq}\to\mathbf{Euler},\qquad
\mathcal{E}(p):=\prod_{n\ge 1}(1-z^n)^{-p_n},
\]
\[
\mathcal{Z}:\mathbf{End}^{\simeq}\to\mathbf{Euler},\qquad
\mathcal{Z}(T):=\frac{1}{\det(I-zT)}=\exp\!\left(\sum_{n\ge 1}\frac{\mathrm{Tr}(T^n)}{n}z^n\right).
\]

\begin{proposition}[Commuting diagram (strictly commutative)]\label{prop:time-space-commuting-diagram}
Under the above notation, the identity
\[
\boxed{\ \mathcal{Z}=\mathcal{E}\circ\mathcal{M}\circ\mathcal{T}\ }
\]
holds. Equivalently, letting $a_n=\mathrm{Tr}(T^n)$ and $p=\mathcal{M}(a)$,
\[
\frac{1}{\det(I-zT)}=\prod_{n\ge 1}(1-z^n)^{-p_n}\in 1+z\CC[[z]].
\]
\end{proposition}

\begin{proof}
By definition, $\log\mathcal{E}(p)=-\sum_{n\ge 1}p_n\log(1-z^n)$; expanding gives
\[
\log\mathcal{E}(p)=\sum_{n\ge 1}p_n\sum_{k\ge 1}\frac{z^{nk}}{k}
=\sum_{m\ge 1}\left(\sum_{n\mid m}\frac{n\,p_n}{m}\right)z^m.
\]
On the other hand, $p=\mathcal{M}(a)$ is equivalent to $a_m=\sum_{n\mid m}n\,p_n$ (decomposition of periodic closed loops into powers of primitive ones), so
\(
\log\mathcal{E}(p)=\sum_{m\ge 1}\frac{a_m}{m}z^m
\).
Exponentiating gives $\mathcal{E}(p)=\exp(\sum_{m\ge 1}\frac{\mathrm{Tr}(T^m)}{m}z^m)=\mathcal{Z}(T)$. \qed
\end{proof}

\begin{remark}[Self-duality of the synchronization kernel and two-variable functional equation]
Taking output bit potential $\varphi(e)=e$, let $B(u)=B_0+uB_1$, $\Delta(z,u)=\det(I-zB(u))$, and $\zeta(z,u)=\Delta(z,u)^{-1}$. Appendix~\ref{app:kernel-self-dual} gives a permutation matrix $\Pi$ such that $\Pi^{-1}B(u)\Pi=uB(1/u)$, and hence
\[
\Delta(z,u)=\Delta(uz,1/u),\qquad
\zeta(z,u)=\zeta(uz,1/u).
\]
This can be viewed as a strict commutation between the endomorphism-side duality functor $D_{\mathbf{End}}:B(u)\mapsto uB(1/u)$ and the Euler-side duality functor $D_{\mathbf{Euler}}:F(z,u)\mapsto F(uz,1/u)$, thereby aligning ``time length $z$'' and ``event weight $u$'' as verifiable dual transformations within the same $\zeta$-interface.
\end{remark}

\begin{remark}[Completion and \texorpdfstring{$\ZZ/2$}{Z2} action]
Viewing $D_{\mathbf{End}}$ and $D_{\mathbf{Euler}}$ as duality actions on the parameter $u$, $D^2=\mathrm{Id}$ gives a $\ZZ/2$-symmetry. If $F(z,u)$ satisfies $F(z,u)=F(uz,1/u)$, define the completion
\[
\widetilde{F}(z,\theta):=F(e^{-\theta/2}z,e^\theta),\qquad u=e^\theta,
\]
then $\widetilde{F}(z,-\theta)=\widetilde{F}(z,\theta)$ is strictly even. The corresponding strictly commutative diagram
\[
\begin{CD}
\mathbf{End}^{\simeq} @>D_{\mathbf{End}}>> \mathbf{End}^{\simeq}\\
@V\mathcal{Z}VV @VV\mathcal{Z}V\\
\mathbf{Euler} @>D_{\mathbf{Euler}}>> \mathbf{Euler}
\end{CD}
\]
can be written via the completion operator
\[
\mathcal{C}(F)(z,\theta):=F(e^{-\theta/2}z,e^\theta),
\]
so that $\mathcal{C}\circ D_{\mathbf{Euler}}=\mathcal{C}$, and
\[
\widetilde{\mathcal{Z}}:=\mathcal{C}\circ\mathcal{Z}
\]
satisfies $\widetilde{\mathcal{Z}}=\widetilde{\mathcal{Z}}\circ D_{\mathbf{End}}$. Therefore, for the synchronization kernel's $\Delta$ and $\zeta$, the completed forms exhibit strict evenness in $\theta$ (see Appendix~\ref{app:kernel-completion}).
\par
Furthermore, one may rewrite \(u=e^\theta\) as \(u=r^2\) and introduce the invariant \(s=r+r^{-1}\), \(w=zr\).
This is equivalent to taking the invariant subring \(\ZZ[r,r^{-1}]^{\,r\mapsto r^{-1}}=\ZZ[s]\) of the Laurent ring under the involution \(r\mapsto r^{-1}\), thereby compressing the multiplicative root-of-unity twist \(u\in\mu_m\) to the real trace coordinate \(s_m=2\cos(\pi/m)\) (Remark~\ref{rem:completion-invariant-ring}).

\subsubsection{Rationality of compiled linear constraint counting: synchronous transducer and characteristic polynomial elimination}\label{subsec:compiled-linear-constraint-counting}

\begin{proposition}[Rationality of the generating function for linear constraint counting (matrix elimination form)]\label{prop:conclusion61-rational-gf-linear-constraints}
In the sofic--Zeckendorf framework, fix an integer matrix \(Q\in\ZZ^{r\times k}\) and a vector \(b\in\ZZ^r\) (with dimensions \(r,k\) fixed).
For each \(m\ge 0\), let
\[
\Omega_m(Q,b):=\Bigl\{(c^{(1)},\dots,c^{(k)})\in X_m^k:\ Q\bigl(\mathrm{Val}(c^{(1)}),\dots,\mathrm{Val}(c^{(k)})\bigr)=b\Bigr\}.
\]
Then the length generating function
\[
F_{Q,b}(z):=\sum_{m\ge 0} \#\Omega_m(Q,b)\,z^m
\]
is a rational function and can be written as
\[
\boxed{\;
F_{Q,b}(z)=u^\top(I-zB_{Q,b})^{-1}v\;}
\]
where \(B_{Q,b}\) is a nonneg\-ative integer matrix explicitly constructed from the synchronous transducer, and \(u,v\ge 0\) are endpoint vectors.
In particular, all poles of \(F_{Q,b}\) are contained in the root set of \(\det(I-zB_{Q,b})=0\); consequently, \(m\mapsto \#\Omega_m(Q,b)\) satisfies a finite-order linear recurrence and can be expressed as a sum of exponential polynomials involving finitely many characteristic roots (with polynomial factors given by Jordan decomposition).
\end{proposition}

\begin{proof}
Read the \(k\) input streams \(c^{(1)},\dots,c^{(k)}\) in parallel at the same time step, and take the synchronous direct product of ``legality constraint automaton \(\times\) on-line normalization kernel \(\times\) linear constraint carry/remainder register'' to obtain a finite-state transducer: its state space depends only on the fixed \(Q,b\) and the base kernel, not on \(m\).
The number of accepting paths of length \(m\) equals \(\#\Omega_m(Q,b)\), so there exist \(u,v\) with \(\#\Omega_m(Q,b)=u^\top B_{Q,b}^m v\).
For any finite matrix \(B\),
\(
\sum_{m\ge 0}u^\top B^m v\,z^m=u^\top(I-zB)^{-1}v
\),
and the denominator is given by \(\det(I-zB)\); the pole and recurrence structures follow. \qed
\end{proof}
