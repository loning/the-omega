\begin{definition}[independence complex]\label{def:pom-independence-complex}
For a finite simple graph \(G=(V,E)\), define its \textbf{independent set complex} as an abstract simplicial complex
\[
\mathrm{Ind}^{\Delta}(G):=\{S\subseteq V:\ S\ \text{is an independent set of}\ G\ \text{}\},
\]
Its geometric implementation is denoted \(|\mathrm{Ind}^{\Delta}(G)|\).
Note: By definition, the face set of \(\mathrm{Ind}^{\Delta}(G)\) is the same as the independent set family \(\mathsf{Ind}(G)\),
However, this section will discuss homotopy types and homology as simple complexes (rather than pure sets).
\end{definition}

\begin{lemma}[Independence set complex of path: \(\bmod 3\) Three-period bisection law]\label{lem:pom-independence-complex-path}

\[
|\mathrm{Ind}^{\Delta}(P_{\ell})|
\ \simeq\
\begin{cases}
\ast, & \ell\equiv 1\ (\mathrm{mod}\ 3),\\[2pt]
S^{\lfloor(\ell+1)/3\rfloor-1}, & \ell\not\equiv 1\ (\mathrm{mod}\ 3),
\end{cases}
\]
Among them, \(\ast\) represents contractible space, and it is agreed that \(S^{-1}\) represents the \((-1)\)-dimensional spherical endpoint mark of the empty complex (only for unified writing).
\end{lemma}

\begin{proof}
This conclusion can be obtained by the following recursive reduction; an auditable recursive proof idea is given here (the proof details can be completed by discrete Morse matching).
\par
\textbf{(1) Recursive reduction. }
For the path \(P_{\ell}\), take the endpoint \(v_1\) and its neighbor point \(v_2\). Decompose the independent set complex according to whether it contains endpoints to get the decomposition:
\[
\mathrm{Ind}^{\Delta}(P_{\ell})
\ =\
\mathrm{Ind}^{\Delta}(P_{\ell}\setminus v_1)\ \cup\
\bigl(v_1\ast \mathrm{Ind}^{\Delta}(P_{\ell}\setminus\{v_1,v_2\})\bigr),

\begin{lemma}[Disjoint union and join: multiplicative structure of independent set complex]\label{lem:pom-independence-complex-join}
If \(G=\bigsqcup_{i=1}^r G_i\) is a disjoint union of graphs, then
\[
\mathrm{Ind}^{\Delta}(G)\ \cong\ \mathrm{Ind}^{\Delta}(G_1)\ast\cdots\ast \mathrm{Ind}^{\Delta}(G_r),
\]
Among them, \(\ast\) is a simple join.
\end{lemma}
\begin{proof}
For disjoint union \(G=\bigsqcup_i G_i\), the independent set \(S\subseteq V(G)\) is equivalent to selecting the independent set \(S_i\subseteq V(G_i)\) for each component and taking the union \(S=\bigsqcup_i S_i\).
This is exactly the description of join: \(\sigma\in K_1\ast\cdots\ast K_r\) if and only if \(\sigma=\bigsqcup_i \sigma_i\) and \(\sigma_i\in K_i\).
Therefore, it is isomorphic. Certification completed.
\end{proof}

\)。

Only the first bit is different between them, so the Hamming distance is \(1\).

\end{proof}

\subsubsection{Hypercube random walk fold factor chain: reversibility, zero self-loops and lower bounds on spectral constants}\label{subsubsec:pom-fold-factor-chain}

This section gives a probability theory interface decoupled from "folding details": taking a simple random walk on \(\Omega_m=\{0,1\}^m\) as the microscopic baseline,
Treat the fold \(\Fold_m\) as an observation map and define its factor chain on the stable layer \(X_m\).
The conclusion shows that under the uniform condition of natural fibers, the factor chain is not only reversible and self-loop-free,
And its Poincar\'e/logarithmic Sobolev constant is strictly \emph{not inferior to} the hypercube random walk itself.

\begin{definition}[Hypercube simple random walk and fold factor chain]\label{def:pom-fold-factor-chain}
Take the simple random walk kernel without self-loop \(P^{\square}_m\) on the microstate space \(\Omega_m=\{0,1\}^m\):
For \(\omega,\omega'\in\Omega_m\), let
\[
P^{\square}_m(\omega,\omega')
:=
\begin{cases}
1/m,& \mathrm{Ham}(\omega,\omega')=1,\\
0,& \text{otherwise}.
\end{cases}
\]
The chain is stationary and reversible with a uniform distribution \(\mu_m(\omega)=2^{-m}\).
\par
Define the \textbf{folding factor chain} on the stable layer as the push-forward compression core where the fibers are uniformly lifted
\[
\overline P^{\square}_m\ :=\ P_m\circ P^{\square}_m\circ Q_m:\Delta(X_m)\to\Delta(X_m),
\]
Among them, \(P_m=(\Fold_m)_*\) and \(Q_m\) are uniform lifting of fibers (see \S\ref{subsec:pom-information-ledger}).
At the level of transition probability, this is equivalent to: first taking the uniform microstate \(\omega\in\Fold_m^{-1}(x)\) on the fiber of the current stable type \(x\),
Do one step of hypercube flipping, and then read out the stability type of the next step.
\end{definition}

\begin{proposition}[Reversibility and zero self-loop]\label{prop:pom-fold-factor-chain-reversible-noloop}
make
\(
\pi_m(x):=d_m(x)/2^m
\)
Fold forward distribution for uniform microstates.
Then \(\pi_m\) is a stationary distribution of \(\overline P^{\square}_m\) and satisfies detailed equilibrium (reversibility).
Additionally, for any \(x\in X_m\) there is
\[
\overline P^{\square}_m(x,x)=0.
\]
\end{proposition}
\begin{proof}
By definition \ref{def:pom-fold-factor-chain},
\(
Q_m\pi_m
\)
is exactly a uniform distribution on \(\Omega_m\): for any \(\omega\),
\(
(Q_m\pi_m)(\omega)=\pi_m(\Fold_m(\omega))/d_m(\Fold_m(\omega))=(d_m/2^m)/d_m=2^{-m}
\)。
therefore
\[
\pi_m\overline P^{\square}_m
=
P_m\bigl((Q_m\pi_m)P^{\square}_m\bigr)
=
P_m(Q_m\pi_m)
=\pi_m,
\]
That is, \(\pi_m\) is stable.
In the same way, the reversibility and compression form of \(P^{\square}_m\) under \(\mu_m\) is \(\overline P^{\square}_m=P_mP^{\square}_mQ_m\),
For any \(x,y\) the flow rate \(\pi_m(x)\overline P^{\square}_m(x,y)\) can be written as an undirected edge count across the fiber and exploit symmetry,
Get a fine balance.
\par
For zero self-loop: if \(\omega,\omega'\in\Omega_m\) is a hypercube neighbor (Hamming distance \(1\)), then
\(
N(\omega')-N(\omega)=\pm F_{k+1}
\)
To a certain \(1\le k\le m\).
According to Lemma \ref{lem:pom-one-bit} (single hidden bit decomposition), the value difference between two points in the fiber of the same stable type \(x\) can only be \(0\) or \(\pm F_{m+2}\).
\(\Fold_m(\omega')=\Fold_m(\omega)\) cannot occur because \(F_{k+1}<F_{m+2}\).
Therefore, a one-step hypercube flip must change the stability type, which is equivalent to \(\overline P^{\square}_m(x,x)=0\). Certification completed.
\end{proof}

\begin{definition}[spectral gap and log Sobolev constant]\label{def:pom-gap-lsi}
For the finite state reversible Markov kernel \(P\) (stationary distribution \(\pi\)), define the Dirichlet form
\[
\mathcal E_P(f,f):=\frac12\sum_{x,y}\pi(x)P(x,y)\bigl(f(x)-f(y)\bigr)^2.
\]
Its \textbf{spectral gap} and \textbf{logarithmic Sobolev constant} are defined as
\[
\mathrm{gap}(P):=\inf_{f\not\equiv \mathrm{const}}\frac{\mathcal E_P(f,f)}{\mathrm{Var}_{\pi}(f)},
\qquad
\alpha(P):=\inf_{f\not\equiv \mathrm{const}}\frac{\mathcal E_P(f,f)}{\mathrm{Ent}_{\pi}(f^2)},
\]
in
\(
\mathrm{Ent}_{\pi}(g):=\mathbb E_{\pi}[g\log g]-\mathbb E_{\pi}[g]\log\mathbb E_{\pi}[g]
\)。
\end{definition}

\begin{theorem}[Spectral constant dominance of factor chain: no worse than hypercube]\label{thm:pom-fold-factor-chain-gap-lsi}
Under the settings defined \ref{def:pom-fold-factor-chain},
\[
\mathrm{gap}(\overline P^{\square}_m)\ \ge\ \mathrm{gap}(P^{\square}_m)=\frac{2}{m},
\qquad
\alpha(\overline P^{\square}_m)\ \ge\ \alpha(P^{\square}_m)=\frac{1}{m}.
\]

\end{theorem}
\begin{proof}
Take any \(f:X_m\to\mathbb R\) and define its fiber constant value improvement \(\widetilde f:=f\circ \Fold_m:\Omega_m\to\mathbb R\).
By the definition of \(\mu_m=Q_m\pi_m\) and the compression core \(\overline P^{\square}_m=P_mP^{\square}_mQ_m\),
Can be checked directly (expand and sum by fiber group):
\[
\mathrm{Var}_{\pi_m}(f)=\mathrm{Var}_{\mu_m}(\widetilde f),\qquad
\mathrm{Ent}_{\pi_m}(f^2)=\mathrm{Ent}_{\mu_m}(\widetilde f^{\,2}),
\qquad
\mathcal E_{\overline P^{\square}_m}(f,f)=\mathcal E_{P^{\square}_m}(\widetilde f,\widetilde f).
\]
