\begin{theorem}[endpoint正then化幂链的二符号喷射：\texorpdfstring{$\zeta$}{zeta} 型三态constantlayer与 \texorpdfstring{$\chi_4$}{chi4} 型线性毛发layer]\label{thm:xi-endpoint-two-symbol-jet}
let \(g\) 在包含区间 \([-2,2]\) 的某开neighborhood上解析。
take完成化trace坐标 \(S\) 与 Chebyshev--Adams 多项式 \(C_m(S)\)（Proposition \ref{prop:half-angle-z4-residue}），并take \(\sigma>1\)。
沿用附录中endpoint正then化幂链operator \(\mathcal W_\sigma\)（式 \eqref{eq:endpoint-Wsigma-even}），并将其natural扩张到局部解析测试function \(g\)：
\[
(\mathcal W_\sigma g)(S):=\sum_{m\ge 1}\frac{1}{m^{1+\sigma}}\,g\!\bigl(C_m(S)\bigr).
\]
then当 \(S\to 0\) 时成立endpoint喷射展开
\begin{equation}\label{eq:xi-endpoint-two-symbol-jet}
\boxed{\;
(\mathcal W_\sigma g)(S)
:=A_\sigma(g)\;+\;L(\sigma,\chi_4)\,g'(0)\,S\;+\;O(S^2)\;}
\end{equation}
其中constantlayer仅依赖endpoint三态值 \(g(0),g(2),g(-2)\)：
\[
A_\sigma(g):=\mathsf H_{\mathrm{odd}}(\sigma)\,g(0)+\mathsf H_{0}(\sigma)\,g(2)+\mathsf H_{2}(\sigma)\,g(-2),
\]
且三通道权kernel为
\[
\mathsf H_{\mathrm{odd}}(\sigma):=\sum_{\substack{m\ge 1\\ m\ \text{为奇数}}}\frac{1}{m^{1+\sigma}},\qquad
\mathsf H_{0}(\sigma):=\sum_{m\equiv 0(4)}\frac{1}{m^{1+\sigma}},\qquad
\mathsf H_{2}(\sigma):=\sum_{m\equiv 2(4)}\frac{1}{m^{1+\sigma}}
\]
（其闭式 \(\zeta\)-符号表示见Proposition \ref{prop:endpoint-dirichlet-symbol-weights}）。
一阶喷射coefficient由模 \(4\) 角色 \(\chi_4\) 控制：let
\[
\chi_4(m):=\sin(m\pi/2)\in\{-1,0,1\},
\qquad
L(\sigma,\chi_4):=\sum_{m\ge 1}\frac{\chi_4(m)}{m^\sigma},
\]
then \(\chi_4\) equivalent to模 \(4\) 的非平凡 Dirichlet 角色（Theorem \ref{thm:endpoint-four-channel-chi4}），and线性项完全由奇通道的一阶喷射 \(C_m'(0)=m\sin(m\pi/2)\) 诱导。
该一阶喷射引入额外因子 \(m\)，thus把 \(\mathcal W_\sigma\) 的权指数 \(1+\sigma\) 在符号layer面平移为 \(L(\sigma,\chi_4)\)。
\end{theorem}

\begin{proof}
由endpoint三态compression \(C_m(0)\in\{0,\pm 2\}\)（Proposition \ref{prop:half-angle-z4-residue} 与Corollary \ref{cor:endpoint-tristate-compression}），并由Corollary \ref{cor:chebyshev-endpoint-jet-regularization} 的奇偶喷射展开，
当 \(S\to 0\) 时
\[
g(C_m(S))=
\begin{cases}
g(C_m(0))+O(S^2), & m\ \text{为偶数},\\
g(0)+g'(0)\,C_m'(0)\,S+O(S^2), & m\ \text{为奇数}.
\end{cases}
\]
对偶通道按 \(m\bmod 4\) 分组，constantlayer即为三态权kernel的线性组合；由Proposition \ref{prop:endpoint-dirichlet-symbol-weights} 得其闭式 \(\zeta\)-符号表示。
对线性项，偶 \(m\) 无一阶贡献，而奇 \(m\) 由Proposition \ref{prop:chebyshev-endpoint-derivative-splitting} 给出
\(
C_m'(0)=m\sin(m\pi/2)=m\chi_4(m)
\)，
hence线性coefficient为
\[
\sum_{m\ge 1}\frac{1}{m^{1+\sigma}}\,C_m'(0)
:=\sum_{m\ge 1}\frac{\chi_4(m)}{m^\sigma}
:=L(\sigma,\chi_4).
\]
综上得到 \eqref{eq:xi-endpoint-two-symbol-jet}。\qed
\end{proof}

\begin{corollary}[endpointorthogonal解耦（喷射layer面）：logarithmic主scaleconstantlayer \(\perp\) \texorpdfstring{$\chi_4$}{chi4} 线性记忆]\label{cor:xi-endpoint-jet-orthogonal-decoupling}
在Theorem \ref{thm:xi-endpoint-two-symbol-jet} 的let定下，endpoint喷射的两类信息严格分离：
\begin{enumerate}
  \item constantlayer \(A_\sigma(g)\) 仅由endpoint三态 \(C_m(0)\in\{0,\pm 2\}\) 的模 \(4\) 分组权kernel决定，因而属于 \(\zeta\)-型主scale通道（Proposition \ref{prop:endpoint-dirichlet-symbol-weights}）。
  \item 一阶喷射完全由 \(\chi_4\) 通道 \(L(\sigma,\chi_4)\) 控制，且不与endpointprecision势ledger \(y\) 的加性平移律耦合（Proposition \ref{prop:endpoint-orthogonal-decomposition-z4-Rplus}）。
\end{enumerate}
therefore，在endpoint喷射layer面，“连续无穷的logarithmicledger”与“离散分支毛发的线性记忆”呈严格orthogonal解耦：前者只进入三态constantlayer的 \(\zeta\)-符号权kernel，后者以一维 \(\chi_4\) 本征通道的形式占据unique的一阶可见分支。
\end{corollary}

\begin{proof}
由Theorem \ref{thm:xi-endpoint-two-symbol-jet}，constantlayer仅依赖endpoint三态值且其权kernel由 \(\zeta(1+\sigma)\) 型 Dirichlet 正then化给出；而线性layer仅由 \(\chi_4\) 通道给出。
Proposition \ref{prop:endpoint-orthogonal-decomposition-z4-Rplus} 指出endpoint处“模 \(4\) 离散载荷 \(\oplus\) precision势ledger”的orthogonal分解，therefore两类喷射分量不发生耦合。\qed
\end{proof}

\begin{theorem}[Jensen defectcertificate的二次resolution壁垒：有限height带的不可绕过半径threshold]\label{thm:xi-jensen-quadratic-resolution-barrier}
fix \(T\ge 1\) 与 \(\delta_0\in(0,\tfrac12]\)，并记
\[
R(T,\delta_0)^2:=\frac{T^2+(\delta_0-1)^2}{T^2+(\delta_0+1)^2}\in(0,1)
\]
（Theorem \ref{thm:xi-jensen-single-radius-band-exclusion}）。
if某一 RH certificate链仅以defectupper bound链 \(D_\zeta(\varrho_n)\le \varepsilon_n\to 0\)（等价地 \(r_\ast(\varrho_n)=\varrho_n e^{-D_\zeta(\varrho_n)}\uparrow 1\)，见Theorem \ref{thm:dephys-defect-repulsion-radius}）作为全部输入，
并要求在height带 \(|\gamma|\le T\) 上排除all满足偏移lower bound \(\tfrac12-\beta\ge\delta_0\) 的离slicezero，
then在asymptotic意义下其半径sequence必须进入不可绕过的二次resolutionlayer：
\begin{equation}\label{eq:xi-jensen-quadratic-resolution-barrier}
\boxed{\;
1-\varrho_n\ \lesssim\ \frac{\delta_0}{T^2}\;}
\end{equation}
（constant仅依赖 \(\delta_0\) 的有界性约束）。
\end{theorem}

\begin{proof}
由Theorem \ref{thm:xi-offcritical-quadratic-radial-compression}，
for any满足 \(|\gamma|\le T\) 且 \(\delta=\tfrac12-\beta\ge\delta_0\) 的离slicezero \(\rho=\beta+\mathrm{i}\gamma\) 的disk像 \(w_\rho\) 满足
\(|w_\rho|\le R(T,\delta_0)\)；
thereforeif \(\varrho\le R(T,\delta_0)\)，then该类离线zero未必进入 Jensen 求和式的“盘内贡献”项（Proposition \ref{prop:app-jensen-formula-defect}），thus单凭 \(D_\zeta(\varrho)\) 的upper bound无法形成带状排除结论。
要在该height带内implementation结构性排除，certificate链必须在某一layer级达到 \(\varrho_n>R(T,\delta_0)\)（Theorem \ref{thm:xi-jensen-single-radius-band-exclusion} 的certificate形式）。
\par
另一方面，直接计算得
\[
1-R(T,\delta_0)^2
:=\frac{4\delta_0}{T^2+(\delta_0+1)^2}
\asymp \frac{\delta_0}{T^2},
\]
thus一旦 \(\varrho_n>R(T,\delta_0)\) 便必有
\(
1-\varrho_n\le 1-R(T,\delta_0)\lesssim 1-R(T,\delta_0)^2\lesssim \delta_0/T^2
\)，
得到 \eqref{eq:xi-jensen-quadratic-resolution-barrier}。\qed
\end{proof}

\begin{corollary}[endpoint角窗的不可替代性：高height贡献强制局部化于 \texorpdfstring{$\theta\approx\pi$}{theta≈pi}]\label{cor:xi-jensen-endpoint-angle-window-rigidity}

Jensen 圆平均的高虚部贡献被强制集中于endpoint角窗 \(\theta\approx \pi\)（即 \(w=\varrho e^{\mathrm{i}\theta}\approx -1\)）：

therefore，任何基于defect圆平均/certificateupper bound链的有效带状信息，必然通过endpoint角窗与endpoint喷射（Theorem \ref{thm:xi-endpoint-two-symbol-jet}）进入closed-loopinterface，而不能被远离endpoint的均匀角采样所等价替代。
\end{corollary}

\begin{proof}
由Theorem \ref{thm:dephys-timefiber-two-ended}，Cayley 逆像的虚部为
\(
\Im t(\varrho,\theta)=\frac{1-\varrho^2}{1+\varrho^2+2\varrho\cos\theta}
\)，
并在 \(\theta=\pi\) 处take得最大值 \(y_+(\varrho)=\frac{1+\varrho}{1-\varrho}\to\infty\)（\(\varrho\uparrow 1\)）。
therefore，当 \(\varrho\) 进入endpointresolutionlayer时，任何涉及height窗的主贡献必须来自 \(\theta\approx\pi\) 的neighborhood；将该endpoint局部化与endpoint喷射符号分解合并即得结论。\qed
\end{proof}

\begin{theorem}[三重一致性审计原理：\texorpdfstring{$p$}{p}-进闭合 \(\times\) Hankel rigidity \(\times\) 语义保守]\label{thm:xi-triple-consistency-audit}

\end{enumerate}

\begin{proof}

\paragraph{扩展保存category：规范记录轴与unique连续横截方向}
本节所用的“扩展保存”语言，旨在将“不可逆读出”视为\emph{记录轴不足}的结构性现象，并以externalized记录轴把信息守恒严格化为扩展空间上的单射性（Proposition \ref{prop:unit-circle-phase-extension-preserving}；并见附录 \ref{subsec:unit-circle-phase-information}）。
在verifiable语义下，这一思想可被组织为一个因子category：object为可使扩展读出单射化的记录轴，morphism为保持certificateverifiable等价的compression/projection。

\begin{definition}[扩展保存category与因子morphism]\label{def:xi-extension-preserving-category}
fix一个读出（允许为多对一compression）\(\pi:\Omega\to Y\)，并把“可kernel对的单值读出”作为typelayer前提（Proposition \ref{prop:xi-null-three-way}）。
称map \(r:\Omega\to R\) 为一个\textbf{记录轴}，并称扩展读出
\[
\pi^{\mathrm e}:=(\pi,r):\Omega\to Y\times R
\]
为\textbf{扩展保存}，if \(\pi^{\mathrm e}\) 在objectlayer为单射（等价地：\(\omega\neq\omega'\Rightarrow (\pi(\omega),r(\omega))\neq(\pi(\omega'),r(\omega'))\)）。
\par
Definition \(\mathrm{Ext}(\pi)\) 为as followscategory：object为使 \(\pi^{\mathrm e}\) 单射的记录轴 \(r\)；
morphism \(r\to r'\) 为一个map \(\phi:R\to R'\) 使得 \(r'=\phi\circ r\)，并要求该因子化在certificateverifiable等价意义下保持一致（即：\(\phi\) 不引入新的不可离线kernel验区分；在具体implementation中可take“由有限object编码并可被确定性检查器复kernel”的因子map）。
\end{definition}

\begin{theorem}[扩展保存category的初object与规范记录轴]\label{thm:xi-extension-preserving-initial-object}
在 unitary slice 可见域约束与 \(\mathbf{PW}\) 闭包口径下，进一步假let：
\begin{enumerate}
  \item \textbf{（orthogonal三分律）}全局 \(\mathsf{NULL}\) 的来源在体系内分解为互不替代的orthogonalwitness（Theorem \ref{thm:xi-null-orthogonal-trichotomy-resource-nonsubstitutable}）；
  \item \textbf{（最小externalized的rigidity）}同像碰撞的可逆化必须externalized residue/逆limit轴并满足defectbudgetlower bound（Theorem \ref{thm:address-defect-law}），虚向折叠的可逆化必须externalized绕数certificate（Theorem \ref{thm:dephys-dirichlet-winding-minimal}），而endpointdilation半群的连续累product仅能沿unique Koenigs depthledger发生（Theorem \ref{thm:dephys-endpoint-koenigs-rigidity}）；
  \item \textbf{（endpoint有限部的闭式封口）}endpoint偶通道的全部divergence在重整化后compression为universal \(\zeta\)-极与显式有限部，并与反变 \(\chi_4\) 本征通道orthogonal（Theorem \ref{thm:endpoint-even-finite-part-renormalization} 与注 \ref{rem:endpoint-renormalization-universal-pole-discrete-hair}）。
\end{enumerate}
then在扩展保存category \(\mathrm{Ext}(\pi)\)（Definition \ref{def:xi-extension-preserving-category}）中there exists一个规范object
\[
r_0
:=
r_{\mathrm{fin}}\times r_{\widehat{\ZZ}}\times r_{\mathrm{wind}}\times r_{\mathrm{end}},
\]
其中 \(r_{\mathrm{fin}}\) 记录endpoint偶通道的universal极/有限部，\(r_{\widehat{\ZZ}}\) 为 residue/逆limit（prime-register）记录轴（congruence轴 \(\widehat{\ZZ}\) 的有限支撑截断口径见注 \ref{rem:pom-mod-support-count}），\(r_{\mathrm{wind}}\) 为绕数certificate轴（\(\kappa_\lambda\)，式 \eqref{eq:dirichlet-winding-certificate}），而 \(r_{\mathrm{end}}\) 为endpoint Koenigs depthledger（Theorem \ref{thm:dephys-endpoint-koenigs-rigidity}）。
and对for anyobject \(r\in\mathrm{Ext}(\pi)\)，there existsunique（在certificateverifiable等价意义下）morphism \(r_0\to r\)。
换言之：\(r_0\) 为 \(\mathrm{Ext}(\pi)\) 的初object，任何其它扩展保存记录轴都可视为对 \(r_0\) 的因子化/compression。
\end{theorem}

\begin{proof}
由Theorem \ref{thm:xi-null-orthogonal-trichotomy-resource-nonsubstitutable}，全局失败witness在orthogonal分解下不可相互替代，therefore任何扩展保存记录轴if要在typelayer消除相应的 \(\mathsf{NULL}\)，就必须分别提供与三类witness相配套的externalized承载。

\end{proof}

\begin{corollary}[unique连续横截方向：radialregister的因子unique性]\label{cor:xi-unique-continuous-transverse-register}

\]

\begin{proof}

最后一句由Theorem \ref{thm:xi-pw-no-continuous-hair} 的“额外连续坐标register不可出现”给出：if拒绝externalizedthe aboveradial轴，thenradial自由度在typelayer被编译为 \(\mathsf{NULL}\)。\qed
\end{proof}

\begin{theorem}[全局 \texorpdfstring{$\mathsf{NULL}$}{NULL} 的完备三分律与可解释性（离线verifiable）]\label{thm:xi-null-complete-trichotomy-offline}
在Definition \ref{def:xi-visible-read-null} 的可见域约束、Proposition \ref{prop:xi-null-three-way} 的单射化原then与 \(\mathbf{PW}\) 闭包type安全（Theorem \ref{thm:xi-offset-null-type-safety}）下，

\begin{enumerate}
  \item \textbf{指称/域外拒绝（语义或protocol空值）}：地址未给定，或object的解释需要超出可见域 \(X_{\mathrm{vis}}\) 的自由度（典型地：需要 \(|u|\neq 1\) 的radial模式，而可见域锁定为 \(\mathsf{Phase}\)），thus在闭包内无地址并按protocol读出为 \(\mathsf{NULL}\)；
  \item \textbf{粘合扭结（二阶相容性障碍）}：局部certificateobject在cover上逐片there exists，但因二阶一致性约束失败而无法全球化；其失败可由一个verifiable的 Čech \(2\)-余圈障碍类 \([\alpha]\neq 0\) witness（Definition \ref{def:prefix-site-h2-gluing-obstruction}，Proposition \ref{prop:null-as-h2-obstruction}）；
  \item \textbf{硬资源不足（阈值型budget违背）}：为把多对一compression读出提升为单值verifiable读出，residue/precisionbudget未达硬阈值，典型地包括：
  \begin{itemize}
    \item prime-register 的模productbudget未越过分位阈值（Proposition \ref{prop:xi-null-three-way}）；
    \item endpointlayerresolution/precisionbudget不足以承载离slice信号的二次compression缺口（Theorem \ref{thm:xi-offcritical-endpoint-resolution-lower-bound}；并见Theorem \ref{thm:xi-radial-information-projection-lower-bound}）。
  \end{itemize}
\end{enumerate}
and三类witness在体系内相互orthogonal、不可替代：增加某一轴的budget不能替代修复另一轴的defect。
\end{theorem}

\begin{proof}
该三分律与不可替代性即为Theorem \ref{thm:xi-null-orthogonal-trichotomy-resource-nonsubstitutable} 的结论在本节口径下的重述。
其中“解释必须可离线kernel验”来自this paper的certificate语义：任何可接受断言必须以有限object编码并由确定性验证器复kernel（见Definition \ref{def:xi-cert-refinement-equivalence} 与Theorem \ref{thm:xi-us-deterministic-tristate} 的确定性读出），therefore失败witness亦必须以同一类有限object形式输出，否then在protocol语义中不构成可解释断言。
\end{proof}

\begin{remark}[拒绝radialregister时的闭合residuewitness：有限prime集合与区间certificate]\label{rem:xi-null-closure-defect-finite-prime-witness}
Corollary \ref{cor:xi-unique-continuous-transverse-register} 表明：if希望以扩展保存方式在连续方向承载离slice信息，then模式轴的连续部分在因子意义下只能由radialregister \(\varrho(u)=\log|u|\) 提供。

\[
\varrho_{\mathrm{vis}}(q)=\sum_{p\in S} v_p(q)\,\log p,
\]
其中 \(\varrho_{\mathrm{vis}}(q)\) 为radial模式轴的可见读出（ifprotocol拒绝externalized该轴，then按纯phase制度强制take \(\varrho_{\mathrm{vis}}(q)=0\)；见Theorem \ref{thm:xi-pw-no-continuous-hair}）。
therefore拒绝radialregister时，闭合defect被强制转写为一个\textbf{有限}residue
\[
\varrho_{\mathrm{def}}(q):=\varrho_{\mathrm{vis}}(q)-\sum_{p\in S} v_p(q)\,\log p,
\]
其非零性可由一个 dyadic 区间certificate \([L_{\mathrm{def}},U_{\mathrm{def}}]\ni \varrho_{\mathrm{def}}(q)\) 且 \(0\notin [L_{\mathrm{def}},U_{\mathrm{def}}]\) 离线复kernel。

\(
\varrho_{\mathrm{def}}(q)=-\log p\neq 0
\)，其自身即构成闭合defect的witness。
该“闭合residue”是Theorem \ref{thm:xi-null-complete-trichotomy-offline} 第 (3) 类（硬资源/闭合defect）的一种标准auditable实Example。
\end{remark}

\paragraph{precision一致性与implementation中立性：certificateobject的逆limit同一性}

（Proposition \ref{prop:xi-unitary-slice-spectral-crossing}）。

\begin{definition}[certificateobject、细化与verifiable等价]\label{def:xi-cert-refinement-equivalence}
fix一个 unitary slice 上的verifiable实值function \(F:\RR\to\RR\)，并fix一个确定性验证器 \(\mathrm{Ver}\)；
验证器的输入是一个有限object \(C\)（由有限整数/有理数据编码），输出为 \(\mathsf{pass}/\mathsf{fail}\)。
\par
对each dyadic depth \(b\in\NN\)，let \(\mathsf{Cert}_b(F)\) 表示all满足下述条件的certificateobject集合：
\[
C=(I,\sigma),\qquad I=[\theta_{-},\theta_{+}]\subset\RR,\ \ \theta_{-},\theta_{+}\in 2^{-b}\ZZ,
\]
其中 \(I\) 为区间括号，\(\sigma\) 为辅助数据（Example如root区间certificate、单root性certificate或其它与protocol一致的unique性/收缩certificate），并要求
\(
\mathrm{Ver}(F;C)=\mathsf{pass}
\)。
\par
Definition\textbf{细化}关系 \(C'\preceq C\)（允许 \(b'\ge b\)）if and only if：
\begin{enumerate}
  \item \(I(C')\subseteq I(C)\)；
  \item \(C'\) 在同一验证器下通过，即 \(\mathrm{Ver}(F;C')=\mathsf{pass}\)。
\end{enumerate}
并Definition\textbf{verifiable等价} \(C\sim C'\) if and only ifthere exists一个共同细化 \(D\) 使 \(D\preceq C\) 且 \(D\preceq C'\)。
\end{definition}

\begin{theorem}[precision一致性的逆limit归并与implementation中立性]\label{thm:xi-cert-inverse-limit-coherence}
沿用Definition \ref{def:xi-cert-refinement-equivalence} 的记号。
letthere existssome \(\theta_\star\in\RR\) 与一条certificate链 \((C_b)_{b\ge b_0}\) 满足：
\[
C_{b}\in\mathsf{Cert}_b(F),\qquad C_{b+1}\preceq C_{b},\qquad \bigl|I(C_b)\bigr|\xrightarrow[b\to\infty]{}0,
\]
andeach \(C_b\) 的断言都以“至多一root”的意义在 \(I(C_b)\) 内隔离出同一zero（Example如 Sturm root隔离或等价的unique性certificate口径，见注 \ref{rem:xi-hard-zero-certificate}）。
then：
\begin{enumerate}
  \item \textbf{（certificate同一性）}任何指称 \(\theta_\star\) 的有效certificateobject在 \(\sim\) 下归并为单一等价类；
  该等价类可由逆系统 \((\mathsf{Cert}_b(F),\preceq)\) 上的一条嵌套链的\emph{逆limit}表示，其区间交为单点 \(\bigcap_{b\ge b_0} I(C_b)=\{\theta_\star\}\)。
  \item \textbf{（implementation中立性/无侧信息通道）}
  if验证器 \(\mathrm{Ver}\) 仅依赖certificateobject的有限编码（并按fix规then在整数/有理运算layer面运行），
  then对for any两种implementation环境，只要输入同一certificateobject \(C\)，验证输出严格一致；
  and对fix \(F\)，\(\theta_\star\) 的寻址仅take决于其 \(\sim\)-等价类，而与certificate生成时可能出现的浮点轨trace、采样路径等非certificate数据无关。
\end{enumerate}
\end{theorem}

\begin{proof}
\begin{enumerate}
  \item
  由假let，“至多一root”的certificate语义保证：if \(C,C'\) 都指称同一zero \(\theta_\star\)，then对足够大的 \(b\) 可选take细化到 \(I(C)\cap I(C')\) 内的更小certificate括号，thusthere exists共同细化 \(D\)；
  thereforefor any两份有效certificate在 \(\sim\) 下等价，归并为单一等价类。
  嵌套区间长度趋于 \(0\) 推出交集为单点，并与逆limit的标准表示一致。
  \item
  if \(\mathrm{Ver}\) 是对有限编码的确定性计算，then其输出是输入certificateobject的function值；
  therefore对同一 \(C\) 的输出在不同implementation环境下不变。
  同时，由Definition \ref{def:xi-cert-refinement-equivalence}，等价类只依赖verifiable的“可共同细化”关系，
  任何不进入certificate编码的生成细节在语义layer面被主动丢弃，因而不影响寻址。
\end{enumerate}
\qed
\end{proof}

\begin{corollary}[type否定：zero不是地址]\label{cor:xi-zero-not-address}
在this paper的verifiable语义中（Definition \ref{def:xi-cert-refinement-equivalence}），所谓“地址object”是一个\emph{有限}certificateobject
\(
C=(I,\sigma)\in\mathsf{Cert}_b(F)
\)
（或其在可共同细化等价 \(\sim\) 下的等价类），其作用是以有限编码指定一个可离线复kernel的断言。
与之相对，Theorem \ref{thm:xi-cert-inverse-limit-coherence} 表明：zero \(\theta_\star\) 属于逆系统 \((\mathsf{Cert}_b(F),\preceq)\) 的逆limit意义下的\emph{点object}，其指称由一条无限细化链的区间交
\(
\bigcap_{b\ge b_0} I(C_b)=\{\theta_\star\}
\)
给出。
therefore，“一个zero是另一个zero的地址”在type上不成立：点object不能充当有限certificateobject。
在this paper制度下可成立且具verifiable含义的陈述是：\emph{两zero在给定 dyadic depth \(b\) 下共享同一可见前缀}（thus在该resolution下不可区分），其形式化由下述 dyadic 前缀相容depth给出。
\end{corollary}

\begin{definition}[dyadic 前缀相容depth与地址超度量]\label{def:xi-dyadic-prefix-depth-ultrametric}
对for any \(b\in\NN\) 与 \(\theta\in\RR\)，Definition dyadic 盒
\[
J_b(\theta):=\Bigl[\frac{\lfloor 2^b\theta\rfloor}{2^b},\ \frac{\lfloor 2^b\theta\rfloor+1}{2^b}\Bigr),
\qquad |J_b(\theta)|=2^{-b}.
\]
对 \(\theta,\theta'\in\RR\) Definition其 dyadic 前缀相容depth
\[
m(\theta,\theta'):=\sup\{\,b\in\NN:\ J_b(\theta)=J_b(\theta')\,\}\in\NN\cup\{\infty\},
\]
并约定 \(2^{-\infty}=0\)。
由此Definition\textbf{地址超度量}
\[
d_{\mathrm{addr}}(\theta,\theta'):=2^{-m(\theta,\theta')}.
\]
在该约定下，\(d_{\mathrm{addr}}\) 为超度量（ultrametric），且其半径 \(2^{-b}\) 的闭球恰为 dyadic 盒：对for any \(\theta\in\RR\)，
\(
\overline{B}(\theta,2^{-b})=J_b(\theta)
\)。
\end{definition}

\begin{remark}[共享前缀与certificate细化depth]\label{rem:xi-shared-prefix-cert-depth}
Definition \ref{def:xi-dyadic-prefix-depth-ultrametric} 的 \(m(\theta,\theta')\) 给出一个完全verifiable的“共享前缀”刻画：\(m(\theta,\theta')\ge b\) if and only if \(\theta,\theta'\) 落在同一 dyadic 盒 \(J_b\) 中。
therefore，对两zero \(\theta_\star,\theta_\star'\) 而言，“共享 \(b\) 位前缀”应理解为：在certificatetype索引 \(b\) 的layer级上（即endpoint落在 \(2^{-b}\ZZ\) 的区间括号），它们尚不可被任何depth为 \(b\) 的 dyadic certificate分离；而一旦细化到 \(b>m(\theta_\star,\theta_\star')\)，二者必落入不同 dyadic 盒，thus在该depth上可被区分。
\par
该超度量结构与正文的“球即柱集”口径同型：在 \(\Omega_\infty\) 上由共同前缀长度诱导的前缀超度量 \(d_{\mathrm{pre}}\)（段落 \ref{par:spg-prefix-ultrametric}）与此处的 \(d_{\mathrm{addr}}\) 仅是同一“前缀细化 \(=\) 半径收缩”的 dyadic implementation。
therefore，在this paper的地址语义下，“zero之间的分辨关系”应被理解为共享前缀depth（或等价的超度量距离），而非把some点object误识为另一个点object的地址。
\end{remark}

\begin{remark}[resolution标度与有限态密度barrier]\label{rem:xi-resolution-height-finite-state-barrier}
在this paper的endpoint紧化/前缀语义下，“清晰度”首先表现为height \(T\) 与 dyadic depthbudget \(b\) 的定量关系：在phase紧化坐标 \(u=\upsilon(T)\) 中，相邻zero的平均可分辨scale为 \(|u_{n+1}-u_n|\asymp (T^{2}\log T)^{-1}\)，因而区分相邻zero所需位depth满足
\[
b=2\log_{2}T+\log_{2}\log T+O(1)
\qquad(T\to\infty)
\]
（附录Proposition \ref{prop:app-adjacent-zero-dyadic-resolution}）。
\par
另一方面，“zero互为地址”的任何fix有限态/有限语法版本都遭遇密度阶barrier：有限维determinant模板的pole几何被锁定为有限条竖直直线与有限phase梳齿（注 \ref{rem:finite-pole-lattice-barrier}），且在有限频带/有限scale窗的scale叠加制度中critical linezero计数至多为 \(N(T)=O(T)\)（Proposition \ref{prop:xi-scale-mellin-zero-density-upper}）。
therefore，要在密度阶上接近 \(T\log T\) 行为，protocol必须在某一方向释放随 \(T\) 增长的外生自由度budget（最小机制下为 \(\Theta(\log T)\)，见注 \ref{rem:xi-scale-mellin-logT-budget}），而不可能由zero之间的自指寻址在fix规then下内生地产生。
\end{remark}

\begin{theorem}[验证程序layer的可重复性与precision一致性：certificatethere exists性偏function]\label{thm:xi-us-deterministic-tristate}
在 unitary slice 审计口径下，fix：
\begin{enumerate}
  \item 一个可见域约束与检查器（Definition \ref{def:xi-visible-read-null}，take \(X_{\mathrm{vis}}=\mathsf{Phase}\)）；
  \item 一个确定性验证器 \(\mathrm{Ver}\) 与其certificate后端（Definition \ref{def:xi-cert-refinement-equivalence}），并约定certificateresolution/precision索引由 dyadic depth \(b\in\NN\) type化（即certificateendpoint落在 \(2^{-b}\ZZ\)）；
  \item 一套fix的implementation约定（algorithm标识、外向舍入与规范化输入约定），使得任何“候选certificateobject”皆可被第三方离线确定性复kernel其 \(\mathsf{pass}/\mathsf{fail}\)。
\end{enumerate}
thenthere exists一个确定性的三值偏function（按type化地址的读出）
\[
\mathrm{Read}_{\mathrm{US}}:\ \mathrm{Addr}_{\mathrm{US}}\ \rightharpoonup\ \{\mathsf{NonZero},\mathsf{RootInterval},\mathsf{NULL}\},
\qquad
\mathrm{Addr}_{\mathrm{US}}:=\{(L,\theta,b):\ L>1,\ \theta\in\RR,\ b\in\NN\},
\]
其语义为：对each输入地址 \((L,\theta,b)\)，要么输出一份在depth \(b\) 上通过kernel验的有限区间certificate（Example如“非零certificate/root区间certificate”两类之一；参见注 \ref{rem:xi-hard-zero-certificate} 与Proposition \ref{prop:xi-rh-interval-criterion}），要么输出空值 \(\mathsf{NULL}\)。
and：
\begin{enumerate}
  \item \textbf{（跨implementation可复现）}在fiximplementation约定下，\(\mathrm{Read}_{\mathrm{US}}\) 为确定性偏function；同一输入地址在不同平台/implementation环境中给出同一输出。
  \item \textbf{（\(\mathsf{NULL}\) 的object语义）}\(\mathsf{NULL}\) 不表达数值不稳定，而表达certificateobject的结构性缺失：在站点/（预）layer口径下它equivalent to“该地址处不there existsprotocol允许的局部截面/certificatefiber为空”（Proposition \ref{prop:null-as-local-section-obstruction}）；在具体 unitary certificate后端中，它equivalent to“输入位串与certificate后端不相容thus无可接受certificate”（Proposition \ref{prop:unitary-two-integer-reproducible}）。
\end{enumerate}
\end{theorem}

\begin{proof}
由于 \(\mathrm{Ver}\) 对有限编码object的运行是确定性的，任何“读出为非 \(\mathsf{NULL}\)”的断言都必须由some有限certificateobject \(C\) witness，且其verifiable性完全由 \(\mathrm{Ver}(F;C)\) 的真值决定（Definition \ref{def:xi-cert-refinement-equivalence}）。
therefore在fiximplementation约定下，读出输出仅依赖输入地址与certificateobject的有限编码，thus跨implementation一致；这正是Theorem \ref{thm:xi-cert-inverse-limit-coherence} 的“implementation中立性/无侧信息通道”在读出layer的直接Corollary。
\par
另一方面，Proposition \ref{prop:null-as-local-section-obstruction} 将 \(\mathsf{NULL}\) 内禀化为certificatefiber \(\mathscr F(a)\) 的空性；而Proposition \ref{prop:unitary-two-integer-reproducible} 给出在一个具体、可执行的 unitary certificate后端中“同一位串 \(\Rightarrow\) uniquecertificate或unique空值”的可回放性。
二者共同保证：\(\mathsf{NULL}\) 的来源是objectlayer的“不可指称/不可构造”（certificate不there exists），而非浮点轨trace或数值噪声。\qed
\end{proof}

\begin{enumerate}
  \item \textbf{（闭式identity/horocycle implementation）}
  \[
  \delta=\mathcal{B}_{-1}(w_\rho)=\frac{1-|w_\rho|^2}{|1+w_\rho|^2},
  \qquad
  |w_\rho|^2=\frac{\gamma^2+(1-\delta)^2}{\gamma^2+(1+\delta)^2},
  \qquad
  1-|w_\rho|^2=\frac{4\delta}{\gamma^2+(1+\delta)^2}=\delta\,|1+w_\rho|^2
  \]
  （Corollary \ref{cor:app-offcritical-horocycle-busemann}）。
  therefore当 \(|\gamma|\to\infty\) 且 \(\delta\) 有界时，
  \[
  1-|w_\rho|^{2}=\frac{4\delta}{\gamma^{2}}\bigl(1+o(1)\bigr),
  \qquad
  1-|w_\rho|=\frac{2\delta}{\gamma^{2}}\bigl(1+o(1)\bigr),
  \]
  thus在fix偏移 \(\delta\) 下，离slice信息被压入diskboundarylayer的可分辨scale \(\Theta(\gamma^{-2})\)，其compression强度随height二次增强。
  该规律并非“对称轴”意义下的经验几何，而是“involutionfixslice \(|u|=1\) + 紧化门”在可见化口径中的结构后果。
  \item \textbf{（endpoint二阶radial通道）}
  记endpointdepth \(\delta_\gamma:=\tfrac12-\pi^{-1}\arctan|\gamma|\downarrow 0\)，then
  \[
  1-|w_\rho|^2
  =4\pi^2\delta\,\delta_\gamma^2\bigl(1+o(1)\bigr)\qquad(\delta_\gamma\downarrow 0)
  \]
  （Corollary \ref{cor:app-offcritical-radius-second-order-radial-channel}）。
  因而在endpointneighborhood的naturalscale \(|1+w|\ll 1\) 下，\(\delta\) 以 chordal 距离 \(|1+w|^2\) 的\emph{二次}项进入。
  \item \textbf{（precision势归因）}
  letprecision势 \(\mathcal{P}(x)=\log\pi+\log(1+x^2)\)（Definition \ref{def:unit-circle-phase-precision-potential}），then
  \[
  1-|w_\rho|^2\sim 4\pi\,\delta\,e^{-\mathcal{P}(\gamma)}\qquad(|\gamma|\to\infty,\ \delta\ \text{有界})
  \]
  （注 \ref{rem:app-offcritical-radius-precision-potential}）。
  这将“离critical line偏移的disk可见性compression”精确归因于precision势的指数因子 \(e^{-\mathcal{P}(\gamma)}\)。
\end{enumerate}
\end{theorem}

\begin{proof}
三条断言分别为Corollary \ref{cor:app-offcritical-horocycle-busemann}、Corollary \ref{cor:app-offcritical-radius-second-order-radial-channel} 与注 \ref{rem:app-offcritical-radius-precision-potential} 的直接重述。\qed
\end{proof}

\begin{corollary}[precision势分解与“偏移负载”：endpointlayerdepth的logarithmic可加结构]\label{cor:xi-offcritical-endpoint-depth-log-decomposition}
在Theorem \ref{thm:xi-offcritical-quadratic-radial-compression} 的let定下，记endpointlayer余量
\(
h(\rho):=1-|w_\rho|^2
\)。
then当 \(|\gamma|\to\infty\) 且 \(\delta\) 有界时，有乘product型asymptotic
\begin{equation}\label{eq:xi-offcritical-endpoint-product-law}
\boxed{\;
\frac{h(\rho)}{\delta}\,e^{\mathcal{P}(\gamma)}\to 4\pi\;}
\end{equation}
以及等价的logarithmicdepth分解
\begin{equation}\label{eq:xi-offcritical-endpoint-depth-load-decomposition}
\boxed{\;
-\log h(\rho)=\mathcal{P}(\gamma)+\log\frac{1}{\delta}-\log(4\pi)+o(1)\qquad(|\gamma|\to\infty)\; }.
\end{equation}
\end{corollary}

\begin{proof}
由Theorem \ref{thm:xi-offcritical-quadratic-radial-compression} 第 (3) 条的asymptotic式
\(
h(\rho)\sim 4\pi\,\delta\,e^{-\mathcal{P}(\gamma)}
\)
立得 \eqref{eq:xi-offcritical-endpoint-product-law}。
两侧take负logarithmic并整理即得 \eqref{eq:xi-offcritical-endpoint-depth-load-decomposition}。\qed
\end{proof}

\begin{remark}[共动horizon局部化：线性显影与地址前缀的制度成本]\label{rem:xi-comoving-horizon-localization-address-null}
式 \eqref{eq:xi-offcritical-endpoint-product-law}--\eqref{eq:xi-offcritical-endpoint-depth-load-decomposition} 的endpointdepth分解是在fix Cayley 图表 \(\mathcal{C}\)（即fix基点的radial审计）下得到的：在 \(|\gamma|\to\infty\) 且 \(\delta\) 有界时，\(\delta\) 通过因子 \(e^{-\mathcal{P}(\gamma)}\asymp \gamma^{-2}\) 被结构性压入endpointneighborhood的二次radial通道（Theorem \ref{thm:xi-offcritical-quadratic-radial-compression}）。
\par
然而该“二阶compression”并非 horocycle 结构的内禀障碍。
upper half-plane中的水平平移 \(t\mapsto t-x_0\)（\(x_0\in\RR\)）保持 \(\Im(t)\) 不变，hence在diskendpoint Busemann 坐标 \(\delta=\mathcal{B}_{-1}(w)\)（Corollary \ref{cor:app-offcritical-horocycle-busemann}）上表现为沿同一 horocycle 的disk自isomorphism重选。
附录Theorem \ref{thm:app-comoving-horizon-localization} 给出其闭式效应：take共动参数 \(x_0=\gamma\) 时，
\[
1-|w_{\rho,\gamma}|^2=\frac{4\delta}{(1+\delta)^2}\sim 4\delta\qquad(\delta\downarrow 0),
\]
thus在共动局部化图表中离critical line偏移以线性scale显现，而不再随 \(|\gamma|\) 产生二阶compression。
\par
该局部化在几何上消解二次compression，但在verifiable制度中并非无代价：它与this paper的“地址先于take值”制度相容，且平移参数 \(x_0\) 必须作为可寻址前缀被显式给定（\S\ref{sec:recursive-addressing}）；地址未给定时相应评价function在objectlayer不Definition并读出为 \(\mathsf{NULL}\)，且递归地址化只在既有可见域内重选与组织事件而不扩张可见 \(\sigma\)-代数（Corollary \ref{cor:recursive-addressing-visible-sigma-nonexpansion}）。
therefore，后续关于endpointresolution与位复杂度的lower bound（Example如Theorem \ref{thm:xi-offcritical-endpoint-resolution-lower-bound}）仍刻画fix可见interface下的不可绕过budget：if不externalized该共动前缀，then离线偏移在高height区间必通过二次radial通道进入并触发相应的capacitythreshold。
\end{remark}

\begin{remark}[endpointdepth的可加负载]\label{rem:xi-offcritical-endpoint-depth-additive-load}
式 \eqref{eq:xi-offcritical-endpoint-depth-load-decomposition} 表明：在endpointcompressionscale下，endpointlayerdepth \(-\log(1-|w_\rho|^2)\) 在asymptotic上分解为两项可加负载：由无参紧化 Jacobian 诱导的heightprecision负载 \(\mathcal{P}(\gamma)\)，以及由离critical line偏移贡献的偏移负载 \(\log(1/\delta)\)。
在this paper的verifiable语义中，这一分解与“orthogonalslice”结论相容：偏离slice的radial信息并非作为可自由调节的额外连续坐标进入，而是以endpointlayer的auditableledger负载形式出现。
\end{remark}

\begin{theorem}[endpointresolution—位复杂度barrier：二次compression导出的不可约budgetlower bound（显式最坏情形）]\label{thm:xi-offcritical-endpoint-resolution-lower-bound}
fix \(\delta_0\in(0,\tfrac12)\) 与 \(T>0\)。
ifthere exists离critical linezero \(\rho=\beta+\mathrm{i}\gamma\) 满足
\[
|\gamma|\le T,\qquad \beta\le \tfrac12-\delta_0,
\]
并仍记其disk像为 \(w_\rho=\mathcal{C}(\gamma+\mathrm{i}(\tfrac12-\beta))\)，thenendpointlayer余量满足显式lower bound
\begin{equation}\label{eq:xi-offcritical-endpoint-margin-min}
\boxed{\;
1-|w_\rho|^2\ \ge\ \frac{4\delta_0}{T^2+(1+\delta_0)^2}\; }.
\end{equation}
therefore，任何试图仅以endpointlayerresolution阈值控制来排除the aboveheight窗内全部离线zero的kernel验策略，if其对余量 \(1-|w|^2\) 的最小可分辨scale为 \(2^{-p}\)，then必须满足
\begin{equation}\label{eq:xi-offcritical-endpoint-bit-lower-bound}
\boxed{\;
p\ \ge\ \log_2\!\Bigl(\frac{T^2+(1+\delta_0)^2}{4\delta_0}\Bigr)\; }.
\end{equation}
and当 \(T\to\infty\) 时有asymptoticlower bound
\begin{equation}\label{eq:xi-offcritical-endpoint-bit-asymptotic}
\boxed{\;
p\ \ge\ 2\log_2 T+\log_2\!\frac{1}{\delta_0}+O(1)\; }.
\end{equation}
利用precision势 \(\mathcal{P}(x)=\log\pi+\log(1+x^2)\)，该asymptotic亦可写为
\begin{equation}\label{eq:xi-offcritical-endpoint-bit-precision-potential-form}
\boxed{\;
p\ \ge\ \frac{\mathcal{P}(T)+\log(1/\delta_0)-\log(4\pi)}{\log 2}+o(1)\qquad(T\to\infty)\; }.
\end{equation}
\end{theorem}

\begin{proof}
由Theorem \ref{thm:xi-offcritical-quadratic-radial-compression} 第 (1) 条的闭式identity，
\[
1-|w_\rho|^2=\frac{4\delta}{\gamma^2+(1+\delta)^2},
\qquad \delta=\tfrac12-\beta\in[\delta_0,\tfrac12).
\]
在 \(\delta\in(0,1)\) 上，function \(\delta\mapsto \frac{4\delta}{\gamma^2+(1+\delta)^2}\) 单调递增；and对 \(|\gamma|\le T\) 其右端随 \(|\gamma|\) 增大而减小。
therefore在约束 \(\delta\ge\delta_0,|\gamma|\le T\) 下，余量的最小值由 \(\delta=\delta_0\) 与 \(|\gamma|=T\) take得，thus得到 \eqref{eq:xi-offcritical-endpoint-margin-min}。
\par
ifkernel验机制对endpoint余量的最小可分辨scale为 \(2^{-p}\)，then要在该height窗内implementation对全部可能离线zero的endpointlayer可分辨排除，必须满足 \(2^{-p}<\inf(1-|w_\rho|^2)\)，结合 \eqref{eq:xi-offcritical-endpoint-margin-min} 即得 \eqref{eq:xi-offcritical-endpoint-bit-lower-bound}。
\eqref{eq:xi-offcritical-endpoint-bit-asymptotic} 由 \(\log_2(T^2+(1+\delta_0)^2)=2\log_2 T+O(1)\) 得到。
最后，由 \(\mathcal{P}(T)=\log\pi+\log(1+T^2)\) 可将主项写成precision势形式，得到 \eqref{eq:xi-offcritical-endpoint-bit-precision-potential-form}。\qed
\end{proof}

\begin{corollary}[共动前缀的位budget收益律：二次compression \texorpdfstring{$\Rightarrow$}{=>} constantbudget]\label{cor:xi-comoving-prefix-bit-budget-gain}
沿用Theorem \ref{thm:xi-offcritical-endpoint-resolution-lower-bound} 的let定并fix \(\delta_0\in(0,\tfrac12)\)。
if允许在objectlayerexternalized可寻址前缀 \(x_0\) 并take共动局部化 \(x_0=\gamma\)（附录Theorem \ref{thm:app-comoving-horizon-localization}；并见注 \ref{rem:xi-comoving-horizon-localization-address-null} 对制度成本的说明），then离线偏移在共动图表中的可检测余量满足
\[
h_{\gamma}(\rho)=1-|w_{\rho,\gamma}|^2=\frac{4\delta}{(1+\delta)^2}\ \ge\ \frac{4\delta_0}{(1+\delta_0)^2}\qquad(\delta\ge\delta_0).
\]
therefore，ifkernel验机制以阈值 \(2^{-p}\) 作为“可检测余量”（即要求 \(h_{\gamma}(\rho)\ge 2^{-p}\) 对all \(\delta\ge\delta_0\) 成立），then必须满足height无关的constantlower bound
\begin{equation}\label{eq:xi-comoving-bit-threshold}
\boxed{\;
p\ \ge\ \log_2\!\Bigl(\frac{(1+\delta_0)^2}{4\delta_0}\Bigr)\; }.
\end{equation}
与 \eqref{eq:xi-offcritical-endpoint-bit-asymptotic} 对照可见：在height窗 \(|\gamma|\le T\) 上，共动前缀所“回收”的主阶位budget为 \(2\log_2T+O(1)\)。
\end{corollary}

\begin{proof}
由附录Theorem \ref{thm:app-comoving-horizon-localization}，当take \(x_0=\gamma\) 时
\(
h_{\gamma}(\rho)=\frac{4\delta}{(1+\delta)^2}
\)。
在 \(\delta\in(0,1)\) 上该function单调递增，hence对all \(\delta\ge\delta_0\) 有
\(
h_{\gamma}(\rho)\ge \frac{4\delta_0}{(1+\delta_0)^2}
\)。
if要求 \(h_{\gamma}(\rho)\ge 2^{-p}\)，then必有 \(2^{-p}\le \frac{4\delta_0}{(1+\delta_0)^2}\)，equivalent to \eqref{eq:xi-comoving-bit-threshold}。
最后一句由将 \eqref{eq:xi-comoving-bit-threshold} 与 \eqref{eq:xi-offcritical-endpoint-bit-asymptotic} 相减得到。\qed
\end{proof}
