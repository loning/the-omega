\section{Symbols and minimal object table}
\label{app:symbols}

\begin{itemize}
  \item $U_{\mathrm{scan}}$: unitary scan operator (ontic layer).
  \item $V$: pointer-phase unitary; $(U_{\mathrm{scan}},V)$ form a Weyl pair.
  \item $\alpha\in(0,1)\setminus\QQ$: scan slope for irrational rotation.
  \item $x_n=x_0+n\alpha\pmod 1$: Kronecker orbit on $[0,1)$.
  \item $D_N^\ast$: star discrepancy of $\{x_1,\dots,x_N\}$.
  \item $E_N := N D_N^\ast$: accumulated mismatch (audit certificate).
  \item $S_{\mathrm{pf}}(N) := \kB E_N$: phase-friction entropy certificate on a length-$N$ window.
  \item $\tau$: intrinsic scan time (iteration count).
  \item $\sigma(x,\tau) := \frac{1}{\kB}\frac{\dd S_{\mathrm{pf}}(x;\tau)}{\dd\tau}$: mismatch density (entropy production density in scan time).
  \item $t$: external time.
  \item $\kappa(x)$: routing overhead (implementation cost).
  \item $\lapse(x) := \kappa_0/\kappa(x)$: computational lapse.
  \item $T_c$: computational temperature (operational temperature at the interface).
  \item $W_{\mathrm{erase}}$: work required for erasure/re-encoding.
  \item $Z_{\mathrm{geom}}$: geometric impedance term in geometric Landauer principle.
  \item $\dot F_{\mathrm{pred}}$: predictive free-energy gain rate achievable via measurement--feedback control.
  \item $\dot I_{\mathrm{pred}}$: predictive mutual information rate (internal state vs future readout).
  \item $\dot W_{\mathrm{diss}}$: dissipation rate required for maintenance/repair against phase friction.
  \item $\eta_{\mathrm{pred}} := \dot F_{\mathrm{pred}}/\dot W_{\mathrm{diss}}$: predictive efficiency.
  \item $Q_\delta(\alpha)$: resonance susceptibility / anti-locking index at tolerance $\delta$.
\end{itemize}
