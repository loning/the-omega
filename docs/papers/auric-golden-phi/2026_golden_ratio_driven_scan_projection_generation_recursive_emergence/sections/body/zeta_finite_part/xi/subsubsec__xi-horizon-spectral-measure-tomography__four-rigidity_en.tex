\paragraph{Rigidity characterization of the constant \texorpdfstring{$4$}{4}: triple consistency of the residue gauge, depth weight, and saturation gauge}
In the preceding horizon interface, ``\(4\)'' is not introduced as a new dynamical constant but is jointly locked by three independent consistency requirements of the regime closure:
the first is the curvature gauge on the geometric side (Proposition \ref{prop:dephys-hyperbolic-curvature-normalization-four});
the second is the residue gauge on the analytic side (Proposition \ref{prop:xi-four-rigidity-residue-depth-normalization} below);
the third is the saturation gauge on the counting side (Proposition \ref{prop:xi-fourpi-entropy-normalization-uniqueness}).
This paragraph makes these three requirements rigorous in the scalar certificate language of scan tomography.

\begin{proposition}[Rigidity characterization of the coefficient \texorpdfstring{$4$}{4}: the normalization constant is uniquely determined by consistency of spectral residues and geometric depth]\label{prop:xi-four-rigidity-residue-depth-normalization}
Under the finite pointwise defect aperture, let the comoving profile density take a family of undetermined normalizations
\[
H_c(x)
:=\sum_{j=1}^{\kappa} m_j\,\frac{c\,\delta_j}{(x-\gamma_j)^2+(1+\delta_j)^2},
\qquad c>0.
\]
Let its Cauchy transform (\(\Im z>0\))
\[
M_{H_c}(z):=\int_{\RR}\frac{H_c(x)}{x-z}\,\dd x,
\]
and retain the conventions for the inner function \(S_{\mathrm{HP}}\) and
\(
M_P(z):=-2\mathrm{i}\,\frac{d}{dz}\log S_{\mathrm{HP}}(z)
\)
(see Theorem \ref{thm:xi-double-herglotz-residue-ratio}).
Then the residues of both at the lower half-plane poles \(z_j^-=\gamma_j-\mathrm{i}(1+\delta_j)\) satisfy
\[
\operatorname*{Res}_{z=z_j^-} M_{H_c}(z)=-\pi c\,m_j\,\frac{\delta_j}{1+\delta_j},
\qquad
\operatorname*{Res}_{z=z_j^-} M_{P}(z)=2\mathrm{i}\,m_j,
\]
whence
\[
\frac{1}{2\pi}\left|\frac{\operatorname*{Res}_{z=z_j^-} M_{H_c}(z)}{\operatorname*{Res}_{z=z_j^-} M_P(z)}\right|
=\frac{c}{4}\,\frac{\delta_j}{1+\delta_j}.
\]
If one requires that the ``dimensionless depth weight recovered from the residue ratio'' coincides exactly with the geometric Busemann depth weight \(\delta_j/(1+\delta_j)\), i.e.,
\[
\frac{\delta_j}{1+\delta_j}
=\frac{1}{2\pi}\left|\frac{\operatorname*{Res}_{z=z_j^-} M_{H_c}(z)}{\operatorname*{Res}_{z=z_j^-} M_P(z)}\right|
\quad(\forall j),
\]
then necessarily and uniquely \(c=4\).
\end{proposition}

\begin{proof}
For a single atom \(H_{c,(\gamma,\delta,m)}(x):=m\,\frac{c\delta}{(x-\gamma)^2+w^2}\) (\(w=1+\delta\)), repeat the residue computation in the proof of Theorem \ref{thm:xi-double-herglotz-residue-ratio} (replacing \(4\) by \(c\)) to obtain
\[
M_{H_c}(z)
=-\pi c\,m\,\frac{\delta}{w}\cdot\frac{1}{z-(\gamma-\mathrm{i}w)},
\]
so the residue at \(z^-=\gamma-\mathrm{i}(1+\delta)\) is \(-\pi c\,m\,\frac{\delta}{1+\delta}\); the multi-atom case follows by linearity.
The second formula is established in Theorem \ref{thm:xi-double-herglotz-residue-ratio}.
Substituting both and taking absolute values yields \(\frac{c}{4}\frac{\delta}{1+\delta}\); requiring this to hold for all \(j\) forces \(c=4\). \qedhere
\end{proof}

\begin{proposition}[Uniqueness of the depth weight: M\"obius linearization of the multiplicative dilation semigroup forces \texorpdfstring{$p(\delta)=\delta/(1+\delta)$}{p(delta)=delta/(1+delta)}]\label{prop:xi-depth-weight-uniqueness-mobius-linearization}
Consider the depth dilation action \(\delta\mapsto m\delta\) (\(m>0\)).
Suppose there exists a continuous strictly monotone function \(p:(0,\infty)\to(0,1)\) such that for all \(m>0\) and \(\delta>0\),
\[
p(m\delta)=\Phi_m\!\bigl(p(\delta)\bigr),
\qquad
\Phi_m(p):=\frac{m p}{1+(m-1)p},
\]
where \(\Phi_m\) is a fractional linear transformation preserving the endpoints \(0,1\).
Then there exists a constant \(a>0\) such that
\[
\boxed{\;
p(\delta)=\frac{\delta}{\delta+a}\;}
\]
and under the unitary-slice normalization of this paper (fixing \(a=1\)), one obtains the unique form
\[
\boxed{\;
p(\delta)=\frac{\delta}{1+\delta}\; }.
\]
\end{proposition}

\begin{proof}
Set \(q(\delta):=\frac{p(\delta)}{1-p(\delta)}\in(0,\infty)\).
Direct computation from the functional equation gives
\[
q(m\delta)=\frac{p(m\delta)}{1-p(m\delta)}=\frac{\Phi_m(p(\delta))}{1-\Phi_m(p(\delta))}=m\,q(\delta).
\]
By continuity, \(q(\delta)=\delta/a\) (\(a:=q(1)^{-1}>0\)), whence
\(
p(\delta)=\frac{q(\delta)}{1+q(\delta)}=\frac{\delta}{\delta+a}
\).
The convention of the Cayley/unitary-slice gauge adopted in this paper selects ``unit depth'' as the symmetry point when \(\delta=1\), so that \(p(1)=\tfrac12\) and \(a=1\) is fixed. \qedhere
\end{proof}

\begin{proposition}[Boundary action formulation: defect entropy is the weighted Neumann boundary term of the Green potential, with the normalization constant uniquely locked]\label{prop:xi-defect-entropy-green-neumann-boundary-action}
Let \(t_j:=\gamma_j+\mathrm{i}(1+\delta_j)\in\mathbb H\), and define the Green-type potential
\[
U_j(t):=\log\left|\frac{t-\overline{t_j}}{t-t_j}\right|
\qquad(t\in\mathbb H).
\]
Then its normal derivative at the boundary \(y=0\) satisfies
\[
\partial_y U_j(x)=\frac{2(1+\delta_j)}{(x-\gamma_j)^2+(1+\delta_j)^2}.
\]
Further define the weighted boundary action
\[
\mathcal S
:=\frac{1}{4\pi}\int_{\RR}\sum_{j=1}^{\kappa}m_j\,\frac{2\delta_j}{1+\delta_j}\,\partial_y U_j(x)\,\dd x,
\]
then the closed-form identity holds:
\[
\boxed{\;
\mathcal S=\sum_{j=1}^{\kappa}m_j\,\frac{\delta_j}{1+\delta_j}=S_{\mathrm{def}}.\;}
\]
Moreover, the integrand can be rewritten as
\(
\frac{2\delta_j}{1+\delta_j}\partial_y U_j(x)=\frac{4\delta_j}{(x-\gamma_j)^2+(1+\delta_j)^2}
\), which is fully consistent with the \(c=4\) locked in Proposition \ref{prop:xi-four-rigidity-residue-depth-normalization}.
\end{proposition}

\begin{proof}
From
\(
U_j(x+\mathrm{i}y)
=\frac12\log\!\bigl((x-\gamma_j)^2+(y+1+\delta_j)^2\bigr)
-\frac12\log\!\bigl((x-\gamma_j)^2+(y-1-\delta_j)^2\bigr)
\),
differentiate directly and set \(y=0\) to obtain the explicit expression for \(\partial_y U_j(x)\).
Then, using the identity
\[
\int_{\RR}\frac{2a}{(x-\gamma)^2+a^2}\,\dd x=2\pi\qquad(a>0),
\]
integrate term by term for each \(j\) to obtain
\(
\mathcal S=\sum_j m_j\frac{2\delta_j}{1+\delta_j}\cdot\frac{1}{4\pi}\cdot 2\pi
=\sum_j m_j\frac{\delta_j}{1+\delta_j}
\),
and the last step is consistent with \eqref{eq:xi-defect-entropy-closed-form} (Proposition \ref{prop:xi-defect-entropy-closed-form-invariant}). \qedhere
\end{proof}

\begin{proposition}[Monotone contraction of information distinguishability: Poisson coarse-graining strictly contracts any \texorpdfstring{$f$}{f}-divergence]\label{prop:xi-poisson-coarsegraining-f-divergence-contraction}
Let two defect configurations induce two nonnegative profiles \(H,\widetilde H\in L^1(\RR)\), and normalize them to probability densities
\(
h:=H/\|H\|_{L^1},\ \tilde h:=\widetilde H/\|\widetilde H\|_{L^1}.
\)
For any \(t>0\), let the Poisson kernel \(P_t(x):=\frac{1}{\pi}\frac{t}{x^2+t^2}\), and define the coarse-grained versions \(h_t:=P_t*h\), \(\tilde h_t:=P_t*\tilde h\).
For any convex function \(f\) (with \(f(1)=0\)), the corresponding \(f\)-divergence
\[
D_f(h\mid \tilde h):=\int_{\RR}\tilde h(x)\,f\!\left(\frac{h(x)}{\tilde h(x)}\right)\dd x
\]
satisfies the data processing monotonicity
\[
\boxed{\;
D_f(h_t\mid \tilde h_t)\le D_f(h\mid \tilde h)\qquad(\forall t>0).\;}
\]
If \(f\) is strictly convex, then the inequality is strict unless \(h=\tilde h\) (indistinguishable within the regime visible domain).
\end{proposition}

\begin{proof}
Write the convolution as a Markov kernel \(K_t(y\mid x):=P_t(y-x)\), so that
\(
h_t(y)=\int K_t(y\mid x)h(x)\dd x
\) and \(
\tilde h_t(y)=\int K_t(y\mid x)\tilde h(x)\dd x
\).
For each \(y\), define the probability kernel
\[
\pi_y(\dd x):=\frac{K_t(y\mid x)\tilde h(x)}{\tilde h_t(y)}\,\dd x,
\]
so that
\(
\frac{h_t(y)}{\tilde h_t(y)}=\int \frac{h(x)}{\tilde h(x)}\,\pi_y(\dd x)
\)
is the conditional expectation of \(\frac{h}{\tilde h}\) under \(\pi_y\).
By Jensen's inequality and the convexity of \(f\),
\[
f\!\left(\frac{h_t(y)}{\tilde h_t(y)}\right)
\le
\int f\!\left(\frac{h(x)}{\tilde h(x)}\right)\pi_y(\dd x).
\]
Multiplying both sides by \(\tilde h_t(y)\) and integrating over \(y\), then exchanging the order of integration yields
\(
D_f(h_t\mid \tilde h_t)\le D_f(h\mid \tilde h)
\).
If \(f\) is strictly convex, then Jensen's inequality is an equality if and only if \(\frac{h}{\tilde h}\) is almost surely constant under \(\pi_y\); since \(K_t>0\) everywhere on \(\RR\times\RR\), this forces \(h=\tilde h\). \qedhere
\end{proof}
\begin{proposition}[Decidable version of the black hole entropy analogy: the unique normalization making the ``deep defect saturation limit'' strictly equal to the topological count]\label{prop:xi-fourpi-entropy-normalization-uniqueness}
For any nonnegative profile \(H\in L^1(\RR)\), define the ``area-type'' quantity \(A(H):=\int_{\RR}H(x)\dd x\) and the ``entropy-type'' normalization \(S_\lambda:=A(H)/\lambda\) (\(\lambda>0\) constant).
Require that for any single-atom defect \((\gamma,\delta,m)\):
\begin{enumerate}
  \item \(S_\lambda\) is invariant under Cayley gauge rescaling (see Remark \ref{rem:dephys-cayley-scale-gauge-four}, i.e., only permitted to appear in log-difference/ratio invariants);
  \item \(S_\lambda\) is additive for mutually uncoupled defects;
  \item In the deep defect limit \(\delta\to\infty\), \(S_\lambda\) saturates strictly to the topological count \(m\);
\end{enumerate}
then under the regime convention of this paper, necessarily and uniquely
\[
\boxed{\;
\lambda=4\pi,\qquad
H(x)=\sum_{j}m_j\,\frac{4\delta_j}{(x-\gamma_j)^2+(1+\delta_j)^2},\qquad
S_\lambda(H)=\sum_{j}m_j\,\frac{\delta_j}{1+\delta_j}=S_{\mathrm{def}}.\;}
\]
\end{proposition}
\begin{proof}
By Proposition \ref{prop:xi-four-rigidity-residue-depth-normalization}, residue consistency uniquely locks the profile normalization to \(c=4\); hence the single-atom area is
\(
A(H)=\pi\cdot 4m\,\frac{\delta}{1+\delta}
\).
The deep defect limit requirement \(\lim_{\delta\to\infty}S_\lambda(H)=m\) forces \(\lambda=4\pi\).
On the other hand, the M\"obius linearization of the multiplicative dilation semigroup in bounded coordinates uniquely locks the depth weight to
\(
p(\delta)=\delta/(1+\delta)
\)
(Proposition \ref{prop:xi-depth-weight-uniqueness-mobius-linearization}), consistent with the closed-form expression \eqref{eq:xi-defect-entropy-closed-form} of \(S_{\mathrm{def}}\).
Therefore the three requirements jointly force the unique normalization. \qedhere
\end{proof}
\begin{remark}[Ultimate explanation of the coefficient \texorpdfstring{$4$}{4}: triple consistency of curvature gauge, residue gauge, and saturation gauge]\label{rem:xi-four-triple-consistency-curvature-residue-saturation}
In the closed regime of this paper, the constant \(4\) simultaneously satisfies three mutually independent consistency conditions:
\begin{enumerate}
  \item \textbf{Geometric side (curvature gauge):} The normalization of the disk hyperbolic metric \(ds^2_{\mathbb D}=\frac{4|dw|^2}{(1-|w|^2)^2}\) fixes the Gaussian curvature to \(-1\) (Proposition \ref{prop:dephys-hyperbolic-curvature-normalization-four});
  \item \textbf{Analytic side (residue gauge):} Ensures that the residue ratio of \(M_{H_c}\) and \(M_P\) exactly recovers the dimensionless depth weight (Proposition \ref{prop:xi-four-rigidity-residue-depth-normalization});
  \item \textbf{Counting side (saturation gauge):} Ensures that the area-normalized entropy saturates exactly to the topological count in the deep defect limit (Proposition \ref{prop:xi-fourpi-entropy-normalization-uniqueness}).
\end{enumerate}
Therefore, the ``division by \(4\)'' structure is not an arithmetic coincidence but the unique mechanism that normalizes boundary geometric quantities to dimensionless counting quantities within the regime-admissible observable combinations; its formal resemblance to the black hole entropy formula arises rigorously from the same class of boundary normalization and saturation alignment mathematical structures, rather than from introducing any new dynamical interpretation.
\end{remark}
