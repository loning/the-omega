\section{Explicit 1D Fibonacci DQCA model and numerical protocol}
\label{app:fib_dqca}

This appendix specifies a concrete 1D Fibonacci (cut-and-project) Dirac quantum cellular automaton (DQCA) used as a checkable subsystem model for the quasicrystal program in Part~III. It also fixes the diagnostics (dispersion extraction, group velocities, and norm errors) used for the numerical evidence reported in Part~VI.\par
Throughout, $\phi=\tfrac{1+\sqrt5}{2}$ and $\alpha=\phi^{-2}$.

\subsection{Fibonacci chain as a cut-and-project set}
Fix an intercept (phason shift) $\rho\in[0,1)$ and define the internal coordinate
\begin{equation}
    u_j \equiv \{j\alpha+\rho\}\in[0,1),\qquad j\in\mathbb{Z},
\end{equation}
where $\{\cdot\}$ denotes the fractional part. Define the tile-type word $\sigma(j)\in\{L,S\}$ by
\begin{equation}
    \sigma(j)=
    \begin{cases}
    L,& u_j\in[0,1-\alpha),\\
    S,& u_j\in[1-\alpha,1).
    \end{cases}
\label{eq:fib_sigma_appendix}
\end{equation}
This is the standard Sturmian coding of an irrational rotation and yields the Fibonacci word for $\alpha=\phi^{-2}$ \cite{MorseHedlund1940,Lothaire2002}. The geometric embedding is obtained by assigning two edge lengths
\begin{equation}
    \ell_L\equiv \phi\,\varepsilon,\qquad \ell_S\equiv \varepsilon,
\end{equation}
and defining vertex positions recursively by
\begin{equation}
    r_0=0,\qquad r_{j+1}=r_j+\ell_{\sigma(j)}.
\end{equation}
The embedded set $\{r_j\}\subset\mathbb{R}$ is a 1D cut-and-project quasicrystal (Fibonacci chain).

\subsection{Local gates and the one-step unitary}
For continuum-limit diagnostics we work in the single-particle sector
\begin{equation}
    \mathcal{H}_{1p}=\ell^2(\mathbb{Z})\otimes\mathbb{C}^2,
\end{equation}
with internal basis states $\{|R\rangle,|L\rangle\}$.
Let the on-site (Long/Short) coin rotations be
\begin{equation}
    C_L\equiv e^{-i\theta_L\sigma_y},\qquad C_S\equiv e^{-i\theta_S\sigma_y}.
\end{equation}
Define the position-dependent coin operator
\begin{equation}
    C\equiv \sum_{j\in\mathbb{Z}} |j\rangle\langle j|\otimes C_{\sigma(j)}.
\end{equation}
Define the conditional nearest-neighbor shift
\begin{equation}
    S\equiv \sum_{j\in\mathbb{Z}}\Big(|j+1\rangle\langle j|\otimes|R\rangle\langle R| + |j-1\rangle\langle j|\otimes|L\rangle\langle L|\Big).
\end{equation}
The one-step Floquet operator is the coined quantum walk
\begin{equation}
    U \equiv S\,C.
\end{equation}
Unitarity is immediate from unitarity of $S$ (a permutation on the basis) and of $C$ (block-diagonal with unitary blocks).

\subsection{Phason shift and phason strain}
The intercept $\rho$ shifts the acceptance window in internal space and generates a family of locally indistinguishable Fibonacci chains (different global phason choices). A convenient strain model lets $\rho$ vary slowly with the site index $j$:
\begin{equation}
    \rho\to \rho(j),\qquad u_j\equiv \{j\alpha+\rho(j)\}.
\end{equation}
Define the discrete phason strain by the incremental change
\begin{equation}
    \eta_j \equiv \rho(j+1)-\rho(j),
\end{equation}
and its norms (used in the numerics) by
\begin{equation}
    \|\eta\|_{\infty}\equiv \sup_j |\eta_j|,\qquad
    \|\eta\|_{2;N}\equiv \left(\frac{1}{N}\sum_{j=0}^{N-1}\eta_j^2\right)^{1/2}.
\end{equation}
The unstrained Fibonacci chain corresponds to $\eta_j\equiv 0$. Small $\|\eta\|$ generates a controlled density of local phason flips (local changes of the L/S pattern).

\subsection{Periodic approximants and dispersion extraction}
To define a momentum-space diagnostic on an aperiodic texture we use periodic Fibonacci approximants. Let $F_0=0$, $F_1=1$, $F_{n+2}=F_{n+1}+F_n$ be Fibonacci numbers and set
\begin{equation}
    \alpha_n \equiv \frac{F_{n-1}}{F_n}\to \alpha.
\end{equation}
Define a length-$N$ periodic texture by taking $N\equiv F_n$ and replacing $\alpha$ with $\alpha_n$ in Eq.~(\ref{eq:fib_sigma_appendix}), so $\sigma_n(j+N)=\sigma_n(j)$. The finite-volume Floquet operator $U_N$ acts on $\mathbb{C}^{2N}$ and is block-circulant up to the coin texture.\par
For each quasi-momentum $k\in[-\pi,\pi)$, implement the Bloch boundary condition
\begin{equation}
    \psi(j+N)=e^{ikN}\psi(j),
\end{equation}
and compute the $2\times 2$ quasienergy branches $E^{(n)}_{\pm}(k)$ from the eigenphases of the resulting $2N\times 2N$ matrix representation of $U_N(k)$. The group velocity is defined by
\begin{equation}
    v^{(n)}_{\pm}(k)\equiv \frac{\partial E^{(n)}_{\pm}(k)}{\partial k},
\end{equation}
computed numerically by finite differences on a $k$ grid.

\subsection{Error norms}
Let $E_D(k)=\sqrt{(c_{\mathrm{eff}}k)^2+m_{\mathrm{eff}}^2}$ be the target Dirac dispersion in the long-wavelength regime, with $(c_{\mathrm{eff}},m_{\mathrm{eff}})$ fit from small-$k$ data. We define the dispersion error at cutoff $k_{\max}$ by
\begin{equation}
    \delta_E^{(n)}(k_{\max})\equiv \max_{|k|\le k_{\max}} \big|E^{(n)}_{+}(k)-E_D(k)\big|.
\end{equation}
For wavepacket propagation, given an initial normalized state $|\psi_0\rangle\in\mathcal{H}_{1p}$ and integer time $T$, define
\begin{equation}
    \epsilon_{\mathrm{prop}}(T)\equiv \|U^T|\psi_0\rangle-|\psi_D(t)\rangle\|,\qquad t\equiv T\varepsilon,
\end{equation}
where $|\psi_D(t)\rangle$ is the corresponding Dirac evolution under a chosen continuum Hamiltonian $H_D$ (with the same fitted parameters and the same initial envelope). In the numerical implementation we report both $\epsilon_{\mathrm{prop}}(T)$ and its scaling with $(\varepsilon,k_{\max},\|\eta\|)$.

\section{Projection filtering and a doubler gap bound (1D Fibonacci case)}
\label{app:doubling_filter}
This appendix spells out the acceptance-window Fourier transform and a simple mass-gap bound for the projection-filter mechanism discussed in Section~\ref{sec:doubling_filter}, specializing to the standard 1D Fibonacci cut-and-project construction from $\mathbb{Z}^2$.\par

\subsection{Interval window and explicit Fourier transform}
For the Fibonacci chain, the internal space is one-dimensional and the acceptance window $W\subset\mathbb{R}$ can be taken as an interval of length $R>0$ (up to an affine change of coordinates). Let $W=[-R/2,R/2]$ and define the normalized Fourier transform
\begin{equation}
    \widehat{W}(q)\equiv \frac{1}{R}\int_{-R/2}^{R/2} e^{-iqy}\,dy
    = e^{-iq\cdot 0}\,\frac{\sin(qR/2)}{qR/2}
    \equiv \mathrm{sinc}\!\left(\frac{qR}{2}\right),
\end{equation}
with $\mathrm{sinc}(x)\equiv \sin(x)/x$ and $\widehat{W}(0)=1$.
The elementary bound $|\sin x|\le 1$ implies
\begin{equation}
    |\widehat{W}(q)| \le \min\left\{1,\ \frac{2}{|q|R}\right\}.
\label{eq:window_sinc_bound}
\end{equation}

\subsection{A concrete lower bound for would-be doubler momenta}
In the cut-and-project embedding picture one decomposes embedding momentum $k\in(\mathbb{R}/(2\pi/\varepsilon)\mathbb{Z})^2$ into components $(k_\parallel,k_\perp)$ along the physical and internal directions. For a ``mirror'' shift by $\pi/\varepsilon$ in one embedding coordinate, Diophantine irrationality of the slope implies that the induced internal component is not arbitrarily small; in the noble (golden) case one can take a bound of the form
\begin{equation}
    |q_{\mathrm{doubler}}|\equiv |k_\perp|\ge \frac{c_0}{\varepsilon}
\end{equation}
for some constant $c_0>0$ that depends on the embedding convention and on the chosen family of mirror shifts. (For golden slopes, this ultimately reduces to the same bad-approximability property that underlies the Roth-type bound quoted in Appendix~\ref{app:homogenization}.)
\par

\subsection{Mass gap from the window bound}
With the projection-filter mass term $\mathcal{M}(q)=\mu(1-\widehat{W}(q))$ from Eq.~(\ref{eq:proj_filter_mass}), the sinc bound (\ref{eq:window_sinc_bound}) implies that for $|q|\ge c_0/\varepsilon$ one has $|\widehat{W}(q)|\le 2\varepsilon/(c_0R)$. Therefore
\begin{equation}
    |\mathcal{M}(q_{\mathrm{doubler}})| \ge \mu\left(1-\frac{2\varepsilon}{c_0R}\right).
\end{equation}
Choosing the UV normalization $\mu=\mu_0/\varepsilon$ yields
\begin{equation}
    |\mathcal{M}(q_{\mathrm{doubler}})| \ge \frac{\mu_0}{\varepsilon}\left(1-\frac{2\varepsilon}{c_0R}\right),
\end{equation}
which is of order $1/\varepsilon$ for small $\varepsilon$, as claimed.\par

\subsection{Numerical cross-check on periodic approximants}
For the periodic approximants used in Appendix~\ref{app:fib_dqca}, a would-be doubler would appear as a near-zero quasienergy branch at Brillouin-edge momenta. The fixed-parameter script \texttt{scripts/generate\_fib\_dqca\_figures.py} regenerates the dispersion figures and can be extended to directly evaluate the smallest quasienergy magnitude at $k=\pi$ as a diagnostic for the doubler sector.

\section{Rigorous Proof of Quasicrystal Homogenization}
\label{app:homogenization}

We provide a formal proof of Theorem \ref{prop:dirac_limit}, establishing the emergence of the Dirac propagator from the Quantum Walk on a cut-and-project quasicrystal.

\subsection{Setup and Definitions}
Let $\Lambda \subset \mathbb{R}^d$ be a cut-and-project set defined by the projection $P: \mathbb{R}^n \to \mathbb{R}^d$ ($n > d$) of the strip $S = \mathbb{R}^d \times W$, where $W \subset \mathbb{R}^{n-d}$ is a compact window.
Sites are $x \in \Lambda$. The unitary evolution is $\hat{U} = S \cdot C$, with shift $S = \sum_x \sum_j |x+e_j\rangle\langle x| \otimes P_j$ and coin $C = \bigoplus_x C_x$.

\begin{definition}[Phason Strain]
The phason coordinate of a site $x$ is $x_\perp = P_\perp(\vec{n})$ where $x = P_{||}(\vec{n})$. The \textbf{Phason Strain Field} $\phi(x)$ describes the local deviation of the point set from a perfect linear cut. For a standard cut-and-project, $||\phi(x)||$ is strictly bounded by the diameter of the window $W$:
\begin{equation}
    ||\phi(x)|| \le \text{diam}(W) < \infty.
\end{equation}
\end{definition}

\begin{lemma}[Ergodic Averaging]
\label{lemma:ergodic}
Let $f: \mathbb{R}^{n-d} \to \mathbb{C}$ be a Riemann-integrable function on the internal space. For an irrational cut (where the projection of $\mathbb{Z}^n$ is dense in $W$), the lattice sum converges to the integral:
\begin{equation}
    \lim_{R \to \infty} \frac{1}{Vol(B_R)} \sum_{x \in \Lambda \cap B_R} f(x_\perp) = \rho \int_W f(y) dy.
\end{equation}
This follows from Weyl-type equidistribution for cut-and-project sets; see, e.g., \cite{Hof1995}.
\end{lemma}

\subsection{Homogenization Theorem}
We analyze the propagator in momentum space. The effective Hamiltonian $H_{eff}(k)$ is defined by $e^{-i H_{eff} \tau} = \hat{U}(k)$.

\begin{theorem}
Let $\Lambda$ be an icosahedral quasicrystal. The effective Hamiltonian expands as:
\begin{equation}
    H_{eff}(k) = \vec{\sigma} \cdot \vec{k} + \mathcal{E}(k),
\end{equation}
where the error term is bounded by:
\begin{equation}
    ||\mathcal{E}(k)|| \le C_1 |k|^2 + C_2 \frac{|k|}{\Lambda_{UV}} ||\nabla \phi||.
\end{equation}
\end{theorem}

\begin{proof}
1. \textbf{Expansion}: We expand the shift operator $S = e^{-i \vec{k} \cdot \hat{x}}$. For small $k$:
\begin{equation}
    \hat{U}(k) \approx I - i \sum_j (\vec{k} \cdot \vec{e}_j) P_j - i H_{coin}.
\end{equation}
2. \textbf{Isotropy of sums (icosahedral symmetry)}: The linear term depends on the sum of projectors $\sum_j \vec{e}_j P_j$. For the icosahedral set, the sum of rank-1 tensors $\sum \vec{e}_j \otimes \vec{e}_j^T$ is irreducible under $SO(3)$ and thus proportional to the identity $\delta_{ab}$ (Schur's Lemma). This forces the velocity tensor $v_{ab}$ to be $c \delta_{ab}$.
3. \textbf{Phason Error Control}: The key difficulty in quasicrystals is the non-periodic modulation. We invoke the \textbf{Ergodic Path Integral Lemma}:
\begin{equation}
    \left| \sum_{x \in \Lambda_L} e^{i \vec{q} \cdot x} \phi(x) \right| < C L^{d/2} ||\phi||_{W^{1,2}}.
\end{equation}
The error term $\mathcal{E}(k)$ is dominated by the Bragg peaks of the phason strain. For the Golden Mean tiling, these peaks are hierarchically suppressed by powers of $\phi^{-3}$.
4. \textbf{Result}: The first non-vanishing anisotropic term is rank-4 (since rank-3 vanishes by inversion). This scales as $k^2$, giving the $O(\varepsilon^2)$ bound.
\end{proof}

\subsection{Error Bounds and Diophantine Conditions}
The rate of convergence depends on the smoothness of the window boundary $\partial W$.
If $\partial W$ is fractal (canonical cut-and-project), the error decays as $N^{-\alpha}$. If we "smooth" the window (fuzzy boundary), the error decays exponentially $e^{-\gamma N}$, but this introduces non-local hopping tails.
For the "Omega Network" with physical finite information density, $W$ is sharp, so we rely on the \textbf{Diophantine condition} on the projection slope $\tau$. Since $\tau = \phi$ (Golden Ratio) is bad-diophantine (hardest to approximate by rationals), the distribution is maximally uniform ("Noble Number").
\begin{proposition}[Diophantine Bound]
For the Golden Mean $\phi$, Roth's Theorem guarantees that for any rational $p/q$:
\begin{equation}
    \left| \phi - \frac{p}{q} \right| > \frac{1}{\sqrt{5} q^2}.
\end{equation}
This implies that the Fourier coefficients of the lattice density $\hat{\rho}(k)$ decay as:
\begin{equation}
    |\hat{\rho}(k)| \le C \frac{1}{|k|}.
\end{equation}
Substituting this into the ergodic integral, the error term in the Hamiltonian scales as:
\begin{equation}
    ||\mathcal{E}|| \le \int_{|q| > \Lambda} |\hat{\phi}(q)| dq \sim \mathcal{O}(\varepsilon^2).
\end{equation}
The constant $C$ is determined by the surface area of the projection window, $C \approx \text{Vol}(\partial W)$.
\end{proposition}

\subsection{Three-gap structure and Fibonacci return times}
\label{app:three_gap}
The same bad-approximability property that controls the Diophantine bounds above also governs the arithmetic structure of intrinsic phase-orbit readouts (Part~II), modeled by an irrational rotation $x_{t+1}=x_t+\alpha\ (\mathrm{mod}\ 1)$ with $\alpha=\phi^{-2}$. A classical result (the Three-Gap/Three-Distance Theorem) states that for any irrational $\alpha$, the ordered set of points $\{n\alpha\}_{n=0}^{N-1}$ partitions the circle into at most three distinct gap lengths \cite{Slater1967}.
Moreover, when $N$ equals a denominator $q_k$ of a continued-fraction convergent of $\alpha$, the partition exhibits a particularly rigid two-gap structure, and the closest returns occur at such $q_k$.

For the golden case $\alpha=\phi^{-2}$, the continued fraction is eventually constant, so the convergent denominators are Fibonacci numbers (up to an index shift). Consequently, the strongest near-returns and the most uniform finite sampling of the phase orbit occur on Fibonacci time scales. This provides a concrete number-theoretic underpinning for the Fibonacci ``resonant eras'' discussed in Part~VI.

\subsection{Golden-mean constraint, capacity, and Zeckendorf coding}
\label{app:golden_mean_capacity}
Finite-information locality suggests that operational discretizations of intrinsic readouts should obey simple admissibility constraints. A canonical example is the \emph{golden-mean constraint} on binary strings (no adjacent excitations): for $\mathbf{b}=(b_1,\dots,b_L)\in\{0,1\}^L$ require $b_i b_{i+1}=0$. The number of admissible strings satisfies a Fibonacci recursion and grows as
\begin{equation}
    N(L)=F_{L+2}\sim \frac{\phi^{L+2}}{\sqrt{5}},
\end{equation}
so the topological entropy is $h_{\mathrm{top}}=\lim_{L\to\infty}\frac{1}{L}\ln N(L)=\ln\phi$, and the corresponding capacity is
\begin{equation}
    C=\log_2\phi \approx 0.6942\ \text{bits/site}.
\end{equation}
Zeckendorf's theorem gives the matching uniqueness statement in arithmetic form: every integer admits a unique representation as a sum of non-consecutive Fibonacci numbers, which is equivalent to the golden-mean admissibility condition on digits \cite{Zeckendorf1972,AlloucheShallit2003}. In this sense, the Zeckendorf/Ostrowski representation is the canonical discretization compatible with Fibonacci combinatorics and finite-resolution constraints in the golden construction.

\section{Variational Derivation of Informational Gravity}
\label{app:action_proof}

We rigorously derive the field equations from the Principle of Least Action.

\subsection{Assumptions and Definitions}
\begin{enumerate}
    \item \textbf{Metric Signature}: We use the "mostly plus" convention $(-+++)$.
    \item \textbf{Connection}: The connection $\Gamma^\lambda_{\mu\nu}$ is the Levi-Civita connection (torsion-free).
    \item \textbf{Scalar Identification}: Let $\varrho$ denote a coarse-grained information/probability density and define the Fisher-amplitude scalar $\chi\equiv \sqrt{\varrho}$ (equivalently $\varphi\equiv\ln\chi$).
    \item \textbf{Matter Coupling}: Standard Model fields $\psi_m$ couple to the metric via the standard Lagrangian $\mathcal{L}_m$.
\end{enumerate}

\paragraph{Compilation-cost identification.}
Under the cost--density ansatz $\varrho(x)=\varrho_0\,\kappa(x)^p$ (Section \ref{chap:info_budget}), we have
\begin{equation}
    \chi=\sqrt{\varrho}=\sqrt{\varrho_0}\,\kappa^{p/2},
\end{equation}
so the Fisher-amplitude kinetic term becomes a routing-gradient term proportional to $\varrho\,(\nabla\ln\kappa)^2$.

\subsection{The Omega Action}
The total action is the sum of the Einstein--Hilbert term, the Fisher-amplitude term (proportional to Fisher information of $\varrho$), and the Matter term:
\begin{equation}
    S = \int d^4x \sqrt{-g} \left[ \frac{R}{16\pi G} - \lambda_F g^{\mu\nu} \nabla_\mu \varphi \nabla_\nu \varphi - V(\varrho) + \mathcal{L}_m(g_{\mu\nu}, \psi_m) \right].
\end{equation}
\noindent
In the minimal sector used in this manuscript we equivalently write the Fisher term in terms of $\chi$:
\begin{equation}
    S = \int d^4x \sqrt{-g} \left[ \frac{R}{16\pi G} - \lambda_F g^{\mu\nu} \nabla_\mu \chi \nabla_\nu \chi - V(\chi^2) + \mathcal{L}_m(g_{\mu\nu}, \psi_m) \right],
\label{eq:omega_action_appendix_chi}
\end{equation}
which differs from the $\varphi$-form only by the identity $\nabla_\mu\chi=\chi\,\nabla_\mu\varphi$ and by a rescaling of $\lambda_F$ consistent with Eq.~(\ref{eq:omega_action_chi}) in Part~V.

\subsection{Variation with respect to the Metric}
We vary $S$ with respect to $g^{\mu\nu}$.
\begin{enumerate}
    \item \textbf{Gravity}: $\delta S_{EH} = \int \sqrt{-g} \frac{1}{16\pi G} (G_{\mu\nu}) \delta g^{\mu\nu}$.
    \item \textbf{Information}: Using $\delta \sqrt{-g} = -\frac{1}{2}\sqrt{-g} g_{\mu\nu} \delta g^{\mu\nu}$, the variation of the Fisher-amplitude term is:
    \begin{equation}
        \delta S_{info} = -\lambda_F \int \sqrt{-g} \left[ \nabla_\mu \chi \nabla_\nu \chi \,\delta g^{\mu\nu} - \frac{1}{2} g_{\rho\sigma} (\nabla \chi)^2 \delta g^{\rho\sigma} \right].
    \end{equation}
    This yields the \textbf{Information Stress-Energy Tensor}:
    \begin{equation}
        T_{\mu\nu}^{info} \equiv -\frac{2}{\sqrt{-g}} \frac{\delta (S_{info}+S_V)}{\delta g^{\mu\nu}}
        = 2\lambda_F \left( \nabla_\mu \chi \nabla_\nu \chi - \frac{1}{2} g_{\mu\nu} (\nabla \chi)^2 \right) - g_{\mu\nu} V(\chi^2),
    \end{equation}
    where $S_V = -\int d^4x \sqrt{-g}\, V(\varrho)$.
    \item \textbf{Matter}: By definition, $T_{\mu\nu}^{(m)} \equiv -\frac{2}{\sqrt{-g}} \frac{\delta S_m}{\delta g^{\mu\nu}}$.
\end{enumerate}
Collecting terms, we obtain the field equations:
\begin{equation}
    G_{\mu\nu} = 8\pi G \left( T_{\mu\nu}^{(m)} + T_{\mu\nu}^{info} \right).
\end{equation}

\subsection{Variation with respect to the Scalar Field}
Varying with respect to $\chi$ yields the equation of motion for the information field.
\begin{equation}
    \delta S_{info} = -2\lambda_F \int d^4x \sqrt{-g} \nabla^\mu \chi \nabla_\mu (\delta \chi)
    = 2\lambda_F \int d^4x \sqrt{-g} (\Box \chi)\, \delta \chi.
\end{equation}
Including the potential term, $\delta S_V = -\int d^4x \sqrt{-g}\, \frac{dV}{d\chi}\, \delta\chi$, the Euler--Lagrange equation is:
\begin{equation}
    2\lambda_F \Box \chi - \frac{dV}{d\chi} = 0,
\end{equation}
with $\frac{dV}{d\chi} = 2\chi\,\frac{dV}{d\varrho}$ when $V=V(\varrho)$ and $\varrho=\chi^2$.

\subsection{Conservation Laws}
Diffeomorphism invariance ensures $\nabla^\mu G_{\mu\nu} = 0$. Thus, the total stress tensor must be conserved:
\begin{equation}
    \nabla^\mu (T_{\mu\nu}^{(m)} + T_{\mu\nu}^{info}) = 0.
\end{equation}
Checking the divergence of the information tensor explicitly:
\begin{equation}
    \nabla^\mu T_{\mu\nu}^{info} = 2\lambda_F [ (\Box \chi) \nabla_\nu \chi + \nabla^\mu \chi \nabla_\mu \nabla_\nu \chi - \frac{1}{2} \nabla_\nu (\nabla_\alpha \chi \nabla^\alpha \chi) ] - \nabla_\nu V(\chi^2).
\end{equation}
Using $\nabla_\nu (\nabla \chi)^2 = 2 \nabla^\mu \chi \nabla_\nu \nabla_\mu \chi$, the last two terms cancel. We are left with:
\begin{equation}
    \nabla^\mu T_{\mu\nu}^{info} = \left( 2\lambda_F \Box \chi - \frac{dV}{d\chi} \right) \nabla_\nu \chi.
\end{equation}
Thus, provided the scalar field satisfies its equation of motion, the information stress tensor is separately conserved.

\subsection{The Information Current}
We define a natural covariant ``information current'' by
\begin{equation}
    J^\mu \equiv \chi\, \nabla^\mu \chi = \frac{1}{2}\nabla^\mu \varrho.
\end{equation}
In the minimal case $V\equiv 0$, the scalar equation reduces to $\Box \chi = 0$.

\section{Holographic Isometry and Error Correction}
\label{app:isometry}

We formalize the construction of the holographic mapping $\Phi$ as a Tensor Network Quantum Error Correcting Code (TN-QECC).

\subsection{Tensor Network Construction}
The "Golden MERA" is constructed from a hierarchy of tensors $\{T^{(s)}\}$ acting on the dual graph of the quasicrystal inflation.
\begin{definition}[Perfect Tensor Class]
We assume the local tensors $W$ (isometry) and $D$ (disentangler) are \textbf{Perfect Tensors} (or $\epsilon$-perfect). A tensor $T_{i_1 \dots i_n}$ is perfect if any bipartition of its indices into sets $A$ and $B$ with $|A| \le |B|$ defines an isometry from $A$ to $B$.
\end{definition}
The global isometry $\Phi: \mathcal{H}_{bulk} \to \mathcal{H}_{boundary}$ is the limit of the layer composition:
\begin{equation}
    \Phi = \prod_{s=0}^\infty \left( \bigotimes_{x} D^{(s)}_x \cdot \bigotimes_{y} W^{(s)}_y \right).

\subsection{Finite-information localization and approximate factorization}
We formalize the finite-information postulate in a way compatible with operator-algebraic locality. In continuum AQFT, local von Neumann algebras are typically type~III and do not admit an exact tensor factorization $\mathcal{H}\simeq\mathcal{H}_R\otimes\mathcal{H}_{R^c}$; this is tied to ultraviolet entanglement across sharp boundaries (see, e.g., \cite{Haag1996}). The finite-information axiom is therefore interpreted as a statement about a \emph{regulated/operational} description: at finite resolution (or on a finite-dimensional code subspace), the relevant effective algebra is type~I and admits an approximate subsystem structure.\par

\begin{theorem}[Finite-information effective dimension]
\label{thm:factorization}
Let $V$ be a finite causal region with boundary area $A(\partial V)$.
Assume that the set of physically distinguishable states accessible to observers localized in $V$ obeys a Bekenstein--Hawking entropy bound
\begin{equation}
    S_{\max}(V)\ \le\ \frac{A(\partial V)}{4 l_P^2}.
\end{equation}
Then any family of pairwise orthogonal operational states $\{|\psi_i\rangle\}_{i=1}^N$ localized in $V$ has cardinality bounded by
\begin{equation}
    N\ \le\ \exp\!\left(\frac{A(\partial V)}{4 l_P^2}\right).
\end{equation}
Equivalently, the operational state space in $V$ can be supported on an effective finite-dimensional Hilbert space $\mathcal{H}^{\mathrm{eff}}_V$ of dimension at most $\exp(A(\partial V)/4l_P^2)$.
\end{theorem}
\begin{proof}
Let $\{|\psi_i\rangle\}_{i=1}^N$ be orthonormal. Consider the maximally mixed state on their span,
\begin{equation}
    \rho \equiv \frac{1}{N}\sum_{i=1}^{N}|\psi_i\rangle\langle\psi_i|.
\end{equation}
Its von Neumann entropy is $S(\rho)=\ln N$. By definition of $S_{\max}(V)$, one has $\ln N\le S_{\max}(V)$, and the stated bound follows.
\end{proof}

\paragraph{Remark (split property and the continuum limit).}
In theories satisfying the split property, for nested regions $V\subset V_{\epsilon}$ there exists an intermediate type~I factor $\mathfrak{N}$ with $\mathfrak{A}(V)\subset \mathfrak{N}\subset \mathfrak{A}(V_{\epsilon})$ \cite{Buchholz1987}. This provides a controlled notion of localization at finite resolution and an approximate tensor-product structure on the associated state spaces, while the sharp-boundary limit recovers type~III behavior.
In Omega Theory, statements such as $\mathcal{H}\cong\bigotimes_x\mathcal{H}_x$ are to be read in this regulated/code-subspace sense (cf.\ Part~II), not as a claim about exact factorization in continuum QFT.\par

\paragraph{Corollary (area-limited independent degrees of freedom).}
If a microscopic model admits a local tensor-factor description on a graph, but the physically realized states in a region $V$ are restricted to an operational/code subspace obeying the bound above, then the number of independent degrees of freedom in $V$ is limited by area rather than volume. This is the sense in which the holographic constraint enforces redundancy of bulk degrees of freedom in a regulated discrete description.
