\paragraph{Restricted Fenchel Duality and Spectral Tomography: Inverting \texorpdfstring{$\tau|_{[0,1]}$}{tau on [0,1]} from Success Exponent}
Under the setting of Theorem \ref{thm:pom-oracle-success-variational-laplace}--\ref{thm:pom-oracle-capacity-thermal-cut}, the success exponent \(\Sigma_{\mathrm{orc}}\) and capacity exponent \(\Gamma_{\mathrm{orc}}\) are parameterized by the same budget scale \(a\in[0,\log 2]\);
and by \(S_1(m)=2^m\) we have the baseline pressure endpoint \(\tau(1)=\log 2\).
This paragraph shows that under the interval constraint \(q\in[0,1]\), \(\Sigma_{\mathrm{orc}}\) and the "negative-temperature" pressure difference \(\tau(1+\theta)-\tau(1)\) are Fenchel conjugates of each other;
thus the full curve of \(\Sigma_{\mathrm{orc}}\) on \([0,\log 2]\) is sufficient to uniquely recover the values of \(\tau\) on \([0,1]\), and under analytic continuation regime determines the entire integer moment spectrum and kernel spectrum.

\begin{theorem}[Restricted Fenchel Duality: \texorpdfstring{$\Sigma_{\mathrm{orc}}$}{Sigma\_orc} and "Negative-Temperature" Pressure Difference]\label{thm:pom-oracle-restricted-fenchel-duality-negative-temp}
Assume \(\tau(q)\) exists and is convex on \(q\in[0,1]\), and \(\Gamma_{\mathrm{orc}}(a)\), \(\Sigma_{\mathrm{orc}}(a)\) exist on \(a\in[0,\log 2]\).
Let
\[
\Lambda_{-}(\theta):=\tau(1+\theta)-\tau(1),\qquad \theta\in[-1,0].
\]
Then for any \(a\in[0,\log 2]\) the strict equality holds
\[
\boxed{\;
-\Sigma_{\mathrm{orc}}(a)=\sup_{\theta\in[-1,0]}\bigl(\theta a-\Lambda_{-}(\theta)\bigr)\;}
\]
and the inverse transform also holds:
\[
\boxed{\;
\Lambda_{-}(\theta)=\sup_{a\in[0,\log 2]}\bigl(\theta a+\Sigma_{\mathrm{orc}}(a)\bigr),\qquad \theta\in[-1,0]\; }.
\]
Therefore on their respective natural domains, \(\Sigma_{\mathrm{orc}}\) and \(\Lambda_{-}\) are Fenchel conjugates with interval constraints.
\end{theorem}
\begin{proof}
By definition \(\mathrm{Succ}^{\mathrm{orc}}_m(B)=\mathcal{C}^{\mathrm{orc}}_m(B)/2^m\) we have
\[
\Sigma_{\mathrm{orc}}(a)=\Gamma_{\mathrm{orc}}(a)-\tau(1)\qquad\bigl(\tau(1)=\log 2\bigr).
\]
On the other hand, Theorem \ref{thm:pom-oracle-capacity-thermal-cut} gives
\(
\Gamma_{\mathrm{orc}}(a)=\sup_{\alpha\in[0,\log 2]}\bigl(f(\alpha)+\min(\alpha,a)\bigr)
\);
and by the Fenchel--Moreau duality of \(f(\alpha)=\inf_{q\ge 0}(\tau(q)-q\alpha)\), on \(q\in[0,1]\) we have
\(
\tau(q)=\sup_{\alpha\in[0,\log 2]}(f(\alpha)+q\alpha)
\).
Note that
\[
\min(\alpha,a)=\inf_{q\in[0,1]}\bigl(q\alpha+(1-q)a\bigr),
\]
and using linearity in the \(q\) variable with interval compactness (the minimax exchange theorem guarantees \(\sup_\alpha\inf_q=\inf_q\sup_\alpha\)), we obtain the restricted variational representation of the capacity curve
\begin{equation}\label{eq:pom-oracle-gamma-restricted-variational-q01}
\boxed{\;
\Gamma_{\mathrm{orc}}(a)=\inf_{q\in[0,1]}\bigl(\tau(q)+(1-q)a\bigr)\qquad(a\in[0,\log 2]).\;}
\end{equation}
Let \(\theta=q-1\in[-1,0]\), and substitute \(\Sigma_{\mathrm{orc}}=\Gamma_{\mathrm{orc}}-\tau(1)\) and \eqref{eq:pom-oracle-gamma-restricted-variational-q01} to obtain
\[
\begin{aligned}
-\Sigma_{\mathrm{orc}}(a)
&=\tau(1)-\Gamma_{\mathrm{orc}}(a)
=-\inf_{q\in[0,1]}\bigl(\tau(q)-\tau(1)+(1-q)a\bigr)\\
&=\sup_{q\in[0,1]}\bigl((q-1)a-(\tau(q)-\tau(1))\bigr)
=\sup_{\theta\in[-1,0]}\bigl(\theta a-\Lambda_{-}(\theta)\bigr),
\end{aligned}
\]
which is the first formula.
Applying the restricted Fenchel--Moreau inverse transform to the right side yields the second formula. \qed
\end{proof}

\begin{corollary}[Interval Uniqueness: \texorpdfstring{$\Sigma_{\mathrm{orc}}$}{Sigma\_orc} Uniquely Determines \texorpdfstring{$\tau|_{[0,1]}$}{tau on [0,1]}]\label{cor:pom-oracle-sigma-determines-tau-on-01}
Under the setting of Theorem \ref{thm:pom-oracle-restricted-fenchel-duality-negative-temp}, the values of \(\Sigma_{\mathrm{orc}}\) on \([0,\log 2]\) uniquely determine the values of \(\Lambda_{-}\) on \([-1,0]\);
thus uniquely determine the values of \(\tau(q)\) on \(q\in[0,1]\), satisfying the explicit inversion
\[
\boxed{\;
\tau(q)=\tau(1)+\Lambda_{-}(q-1)=\log 2+\Lambda_{-}(q-1),\qquad q\in[0,1].\;}
\]
\end{corollary}
\begin{proof}
By the inverse transform of Theorem \ref{thm:pom-oracle-restricted-fenchel-duality-negative-temp}, \(\Sigma_{\mathrm{orc}}\) on \([0,\log 2]\) uniquely determines \(\Lambda_{-}\) on \([-1,0]\).
Then by \(\Lambda_{-}(\theta)=\tau(1+\theta)-\tau(1)\) and \(\tau(1)=\log 2\) we obtain the shown inversion and uniqueness. \qed
\end{proof}

\begin{corollary}[Analytic Extension Determines All Spectra]\label{cor:pom-oracle-analytic-extension-all-spectrum}
Under the setting of Corollary \ref{cor:pom-oracle-sigma-determines-tau-on-01}, further assume \(\tau\) can be real-analytically extended to some connected open neighborhood \(U\supset[0,1]\) (equivalent to the existence of complex analytic extension).
Then the \(\tau|_{[0,1]}\) already determined by \(\Sigma_{\mathrm{orc}}\) uniquely implies \(\tau|_{U}\);
thus for all integers \(q\in U\cap\NN\) the pressure values \(\tau(q)\) are uniquely determined, and therefore uniquely determine the Perron roots \(\rho_q=\exp(\tau(q))\) and their induced moment-kernel principal spectral family.
\end{corollary}
\begin{proof}
An analytic function on a connected domain is uniquely determined by its values on any subset with an accumulation point (such as the interval \([0,1]\)).
By Corollary \ref{cor:pom-oracle-sigma-determines-tau-on-01} we know \(\tau\) on \([0,1]\) is uniquely determined by \(\Sigma_{\mathrm{orc}}\), so its analytic extension on \(U\) is also unique;
thus for \(q\in U\cap\NN\), \(\tau(q)\) and \(\rho_q=\exp(\tau(q))\) are all uniquely determined by \(\Sigma_{\mathrm{orc}}\). \qed
\end{proof}

\paragraph{Area Identity of Conjugate Map and Oracle Gap Area Law}
The following two identities compress the "global geometry on the budget axis" into the quadratic energy of the pressure slope \(\tau'\).

\begin{theorem}[Area Identity of Conjugate Map: Integral of \texorpdfstring{$q_*(a)$}{q\_*(a)} Determined Only by Endpoint Data]\label{thm:pom-oracle-qstar-area-identity}
Assume \(\tau\) is \(C^2\) and strictly convex on \((0,1)\), and let
\[
a_0:=\tau'(0^+),\qquad a_1:=\tau'(1^-).
\]
Inside the thermal phase interval \((a_0,a_1)\), let \(q_*(a)\in(0,1)\) be the unique conjugate parameter satisfying \(\tau'(q_*(a))=a\) (same notation as Theorem \ref{thm:pom-oracle-capacity-thermal-cut}).
Then the strict identity holds
\[
\boxed{\;
\int_{a_0}^{a_1} q_*(a)\,\dd a \;=\; a_1-\bigl(\tau(1)-\tau(0)\bigr).\;}
\]
In particular, under the endpoint normalization of this paper \(\tau(0)=\log\varphi,\ \tau(1)=\log 2\),
\[
\int_{a_0}^{a_1} q_*(a)\,\dd a \;=\; a_1-(\log 2-\log\varphi).
\]
\end{theorem}
\begin{proof}
By strict convexity, \(\tau'\) is strictly increasing on \((0,1)\), so \(q_*(a)=(\tau')^{-1}(a)\) is well-defined.
Make the variable substitution \(a=\tau'(q)\), then \(\dd a=\tau''(q)\dd q\), and when \(a\) increases from \(a_0\) to \(a_1\), \(q\) increases from \(0\) to \(1\).
Therefore
\[
\int_{a_0}^{a_1} q_*(a)\,\dd a
=\int_{0}^{1} q\,\tau''(q)\,\dd q
=\bigl[q\tau'(q)\bigr]_{0}^{1}-\int_{0}^{1}\tau'(q)\,\dd q
=a_1-\bigl(\tau(1)-\tau(0)\bigr).
\]
\qed
\end{proof}

\begin{theorem}[Oracle Gap Area Law: Full-Interval Capacity Integral Equals \texorpdfstring{$L^2$}{L2} Energy of Pressure Slope]\label{thm:pom-oracle-gap-area-law-energy}
Under the setting of Theorem \ref{thm:pom-oracle-qstar-area-identity}, define the oracle gap area
\[
\mathcal{A}^{\mathrm{orc}}_{\mathrm{gap}}
:=\int_{0}^{\log 2}\bigl(\log 2-\Gamma_{\mathrm{orc}}(a)\bigr)\,\dd a.
\]
Then the exact identity holds
\[
\boxed{\;
\mathcal{A}^{\mathrm{orc}}_{\mathrm{gap}}
=\frac12\int_{0}^{1}\bigl(\tau'(q)\bigr)^2\,\dd q.\;}
\]
Equivalently,
\[
\boxed{\;
\int_{0}^{\log 2}\Gamma_{\mathrm{orc}}(a)\,\dd a
=(\log 2)^2-\frac12\int_{0}^{1}\bigl(\tau'(q)\bigr)^2\,\dd q.\;}
\]
The right side is the \(L^2\) energy of the pressure slope on \(q\in[0,1]\); the left-side deficit \(\mathcal{A}^{\mathrm{orc}}_{\mathrm{gap}}\) is the cumulative area of the "success rate gap."
\end{theorem}
\begin{proof}
By \eqref{eq:pom-oracle-gamma-restricted-variational-q01}, for each \(a\in[0,\log 2]\) there exists an optimal point \(q(a)\in[0,1]\) such that
\(
\Gamma_{\mathrm{orc}}(a)=\tau(q(a))+(1-q(a))a
\).
Let \(a_0=\tau'(0^+)\), \(a_1=\tau'(1^-)\).
Strict convexity gives a three-phase structure: when \(a\in[0,a_0]\) the optimum is at \(q(a)=0\), thus \(\Gamma_{\mathrm{orc}}(a)=\tau(0)+a\);
when \(a\in[a_1,\log 2]\) the optimum is at \(q(a)=1\), thus \(\Gamma_{\mathrm{orc}}(a)=\tau(1)=\log 2\);
when \(a\in(a_0,a_1)\) the optimum is at the unique interior point \(q(a)=q_*(a)\in(0,1)\), satisfying \(\tau'(q_*(a))=a\).
Thus
\[
\mathcal{A}^{\mathrm{orc}}_{\mathrm{gap}}
=\int_{0}^{a_0}\bigl(\tau(1)-\tau(0)-a\bigr)\,\dd a
\;+\;
\int_{a_0}^{a_1}\Bigl(\tau(1)-\tau(q_*(a))-(1-q_*(a))a\Bigr)\,\dd a.
\]
For the thermal phase segment make the substitution \(a=\tau'(q)\), \(\dd a=\tau''(q)\dd q\), to obtain
\begin{equation}\label{eq:pom-oracle-gap-area-law-hotphase}
\mathcal{A}^{\mathrm{orc}}_{\mathrm{gap}}
=\bigl(\tau(1)-\tau(0)\bigr)a_0-\frac12a_0^2
\;+\;
\int_{0}^{1}\Bigl(\tau(1)-\tau(q)-(1-q)\tau'(q)\Bigr)\tau''(q)\,\dd q.
\end{equation}
For the last term integrate by parts: let
\(
G(q):=\tau(1)-\tau(q)-(1-q)\tau'(q)
\)
then \(G(1)=0\), \(G(0)=\tau(1)-\tau(0)-a_0\), and \(G'(q)=-(1-q)\tau''(q)\).
Therefore
\[
\int_{0}^{1}G(q)\tau''(q)\,\dd q
=\bigl[G(q)\tau'(q)\bigr]_{0}^{1}-\int_{0}^{1}\tau'(q)G'(q)\,\dd q
=a_0(a_0-\tau(1)+\tau(0))+\int_{0}^{1}(1-q)\tau'(q)\tau''(q)\,\dd q.
\]
For \(\int_{0}^{1}(1-q)\tau'(q)\tau''(q)\,\dd q\) use
\(
\frac{\dd}{\dd q}\bigl((1-q)(\tau'(q))^2/2\bigr)=-(\tau'(q))^2/2+(1-q)\tau'(q)\tau''(q)
\)
to simplify, yielding
\[
\int_{0}^{1}(1-q)\tau'(q)\tau''(q)\,\dd q
=-\frac12a_0^2+\frac12\int_{0}^{1}\bigl(\tau'(q)\bigr)^2\,\dd q.
\]
Substituting these two formulas back into \eqref{eq:pom-oracle-gap-area-law-hotphase}, the \(\bigl(\tau(1)-\tau(0)\bigr)a_0-\frac12a_0^2\) and boundary terms precisely cancel, yielding
\(
\mathcal{A}^{\mathrm{orc}}_{\mathrm{gap}}=\frac12\int_{0}^{1}(\tau')^2
\).
The second formula follows from
\(
\int_0^{\log 2}\Gamma_{\mathrm{orc}}=\int_0^{\log 2}\log 2-\mathcal{A}^{\mathrm{orc}}_{\mathrm{gap}}
=(\log 2)^2-\mathcal{A}^{\mathrm{orc}}_{\mathrm{gap}}
\)
immediately. \qed
\end{proof}

\begin{corollary}[Universal Bilateral Bounds and Universal Lower Bound of Critical Budget]\label{cor:pom-oracle-gap-area-universal-bounds-beta-crit}
Under the setting of Theorem \ref{thm:pom-oracle-gap-area-law-energy}, denote the endpoint difference
\(
\tau(1)-\tau(0)=\log 2-\log\varphi
\).
Then the gap area satisfies the absolute bounds
\[
\boxed{\;
\frac12(\log 2-\log\varphi)^2\ \le\ \mathcal{A}^{\mathrm{orc}}_{\mathrm{gap}}\ \le\ \frac12\,\log 2\,(\log 2-\log\varphi).\;}
\]
And denoting \(a_1=\tau'(1^-)\) (same as the critical slope in Theorem \ref{thm:pom-oracle-critical-window-gaussian}, \(\beta_{\mathrm{crit}}=a_1/\log 2\)), we have
\[
\boxed{\;
a_1\ \ge\ \log 2-\log\varphi,
\qquad
\beta_{\mathrm{crit}}\ \ge\ 1-\frac{\log\varphi}{\log 2}\; }.
\]
\end{corollary}
\begin{proof}
By Theorem \ref{thm:pom-oracle-gap-area-law-energy},
\(
2\mathcal{A}^{\mathrm{orc}}_{\mathrm{gap}}=\int_0^1(\tau')^2
\).
Also
\(
\int_0^1\tau'(q)\,\dd q=\tau(1)-\tau(0)=\log 2-\log\varphi
\).
By Cauchy--Schwarz,
\(
\int_0^1(\tau')^2\dd q\ge \bigl(\int_0^1\tau'\dd q\bigr)^2
\), giving the lower bound.
On the other hand, by Corollary \ref{cor:pom-pressure-derivatives-from-escort} and \(0\le Y_m\le \log 2\) we have \(0\le \tau'(q)\le \log 2\), so
\(
\int_0^1(\tau')^2\dd q\le (\sup\tau')\int_0^1\tau'\dd q\le \log 2\,(\log 2-\log\varphi)
\), giving the upper bound.
\par
Finally, by convexity \(\tau'\) is monotone non-decreasing, so \(a_1=\tau'(1^-)\ge \int_0^1\tau'(q)\dd q=\tau(1)-\tau(0)\), thus yielding the lower bound on \(\beta_{\mathrm{crit}}=a_1/\log 2\). \qed
\end{proof}

\begin{theorem}[Extremal Rigidity: Area Bound Equality Holds If and Only If System Degenerates to Single-Point or Two-Layer Spectrum on Exponential Scale]\label{thm:pom-oracle-gap-area-rigidity}
Under the setting of Corollary \ref{cor:pom-oracle-gap-area-universal-bounds-beta-crit}:
\begin{enumerate}
  \item If \(\mathcal{A}^{\mathrm{orc}}_{\mathrm{gap}}=\frac12(\log 2-\log\varphi)^2\), then \(\tau'(q)\) is almost everywhere constant \(\log 2-\log\varphi\) on \([0,1]\);
  equivalently,
  \[
  \tau(q)=\log\varphi+q(\log 2-\log\varphi)\qquad(q\in[0,1]),
  \]
  and the thermal phase interval length collapses to zero (\(a_0=a_1=\log 2-\log\varphi\)).
  \item If \(\mathcal{A}^{\mathrm{orc}}_{\mathrm{gap}}=\frac12\,\log 2\,(\log 2-\log\varphi)\), then \(\tau'(q)\) takes almost everywhere only the endpoint values \(0\) and \(\log 2\) on \([0,1]\), satisfying the mean constraint
  \(
  \int_0^1\tau'(q)\dd q=\log 2-\log\varphi
  \);
  under this rigidity, \(\tau\) is piecewise affine on \([0,1]\), and the main support of fiber exponents on the exponential scale degenerates to two endpoint spectral layers (aside from exponentially sparse transition layers, only two types of extreme fiber exponents carry weight).
\end{enumerate}
\end{theorem}
\begin{proof}
The first condition is the equality case of Cauchy--Schwarz: \(\int(\tau')^2=(\int\tau')^2\) if and only if \(\tau'\) is almost everywhere constant;
integrating to recover \(\tau\) and using endpoints \(\tau(0)=\log\varphi,\tau(1)=\log 2\) gives the linear form.
\par
The second condition is the equality case of the upper bound \(\int(\tau')^2\le (\sup\tau')\int\tau'\);
since \(0\le\tau'\le\log 2\), the pointwise inequality \((\tau')^2\le(\log 2)\tau'\) achieves equality forcing \(\tau'\in\{0,\log 2\}\) almost everywhere.
The mean constraint is given by \(\int_0^1\tau'=\tau(1)-\tau(0)\).
The final spectral tomography statement follows from the piecewise affine nature of \(\tau\) and the degenerate support of its Legendre dual spectrum. \qed
\end{proof}
