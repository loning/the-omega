\subsubsection{Large deviation rate function of the arity-charge (\texorpdfstring{$\chi$}{chi} slice)}\label{subsec:chi-ldp}
Along the $\chi$-slice $\theta(t)=(t,0,-t)$, the pressure function $P_\chi(t)=\log\rho(M_{\theta(t)})$ provides an exponential tilting family for $\chi$; the corresponding large deviation rate function can be read off directly via Legendre duality.

\begin{remark}[Standard theorem interface: G\"{a}rtner--Ellis]\label{thm:chi-gartner-ellis}
For the \(\chi\)-slice pressure \(P_\chi(t)=\log\rho(M_{\theta(t)})\) of the original matrix family, its Legendre dual
\(
I_\chi(\alpha)=\sup_{t\in\RR}(t\alpha-P_\chi(t))
\)
gives the large deviation rate function of the empirical mean; at \(\bar\alpha=P_\chi'(0)\) one has the quadratic approximation \(I_\chi(\bar\alpha+\delta)=\delta^2/(2P_\chi''(0))+o(\delta^2)\). See \cite{DemboZeitouni2010LargeDeviations}.
\end{remark}

\begin{remark}[Support endpoints of \texorpdfstring{$\chi$}{chi} and endpoint pressures (real input 40-state kernel)]\label{rem:chi-ldp-endpoints}
For the essential \(20\)-state core of the real input \(40\)-state kernel, Theorem
\ref{thm:real-input-40-arity-charge-density-bound}
in Appendix \ref{app:real-input-40-arity-charge}
gives the hard constraint for any primitive periodic orbit $\gamma$ ($n=|\gamma|$):
\(
0\le \chi(\gamma)\le \lfloor n/2\rfloor
\), so that any empirical average
\(
\bar\chi_n=\chi(\gamma)/n
\)
must lie in the interval $[0,1/2]$.
Furthermore, Proposition \ref{prop:real-input-40-arity-charge-endpoints} in the same appendix section gives the endpoint pressures:
as $t\to-\infty$ (i.e., $q=e^t\to 0^+$),
\[
P_\chi(t)\to \log\kappa,
\qquad
\kappa^4-\kappa^3-1=0,\ \kappa>1;
\]
as $t\to+\infty$ (i.e., $q\to+\infty$),
\[
P_\chi(t)=\frac12\,t+\frac12\log 3+o(1).
\]
Therefore the Legendre dual rate function $I_\chi(\alpha)=\sup_t(t\alpha-P_\chi(t))$ must take $+\infty$ for $\alpha\notin[0,1/2]$ (following directly from the endpoint linear slopes), and its effective support endpoints are pinned precisely by
\(
\lim_{t\to-\infty}P_\chi'(t)=0
\)
and
\(
\lim_{t\to+\infty}P_\chi'(t)=1/2
\).
\end{remark}

\paragraph{Intrinsic time scales (no external input): correlation time of the \texorpdfstring{$\chi$}{chi} slice}
The parameter $t$ in this subsection not only controls the static cost of exponential tilting ($P_\chi(t)$) and the constant-level drift ($\log\mathfrak{M}_\chi(t)$), but also simultaneously induces a \textbf{purely kernel-intrinsic} time scale: under the equilibrium state at $\theta(t)=(t,0,-t)$, the temporal correlation of the arity-charge observable decays exponentially at the spectral gap rate.
To avoid confusion with the character projection mixing time $\tau_{\mathrm{mix}}$ of $P_\chi$ (Remark \ref{rem:pom-chi-mixing}), we denote this scale by $\tau_{\mathrm{corr}}$.

\begin{remark}[Correlation time of the \texorpdfstring{$\chi$}{chi} slice (spectral gap formulation)]\label{prop:chi-slice-time-correlation}
Let the weighted matrix of the real input 40-state kernel at $\theta(t)=(t,0,-t)$ be $M_{\theta(t)}$, and assume the matrix family is primitive in a neighborhood of the given $t$.
Define
\[
\lambda_1(t):=\rho\bigl(M_{\theta(t)}\bigr),
\qquad
\Lambda(t):=\max\{ |\lambda|:\ \lambda\in\mathrm{spec}(M_{\theta(t)}),\ |\lambda|<\lambda_1(t)\},
\qquad
\rho_{\mathrm{corr}}(t):=\frac{\Lambda(t)}{\lambda_1(t)}\in[0,1),
\]
and the correlation time
\[
\tau_{\mathrm{corr}}(t):=\frac{1}{-\log \rho_{\mathrm{corr}}(t)}.
\]
For a finite-state primitive Markov chain, the temporal correlation of a centered finite-window observable satisfies the exponential decay envelope
\(
|\mathrm{Cov}(\chi_0,\chi_n)|\le C(t)\,\rho_{\mathrm{corr}}(t)^n
\), where $C(t)<\infty$ (standard spectral gap result; see \cite{LevinPeresWilmer2009MarkovMixing}).
This paper uses only the auditable interface of ``reading off the exponential envelope from the finite-dimensional spectrum''.
\end{remark}

\begin{remark}[Finite-sample deviation certificate of the event clock (second-moment interface)]\label{prop:chi-event-clock-deviation-certificate}
Under the Parry equilibrium state at $t=0$, let $e_t\in\{0,1\}$ be the output bit, $T_n:=\sum_{t=0}^{n-1}e_t$, and $\mu_e:=\mathbb{E}[e_0]$.
Viewing the one-step output bit as a bounded observable on a finite-state Parry Markov chain, the covariance sequence
\(
\mathrm{Cov}(e_0,e_k)
\)
and the partial sum variance
\(
\mathrm{Var}(T_n)
\)
can both be \emph{exactly} computed from powers and inner products of the finite-dimensional transfer matrix $P$.
Furthermore, let
$$
S_n:=\sum_{t=1}^{n}\mathrm{Var}(T_t),\qquad b_n(\delta):=\sqrt{\frac{S_n}{\delta}}.
$$
Then by a union bound and Chebyshev's inequality, one obtains an auditable uniform deviation certificate: for any $\delta\in(0,1)$ and any $n\ge 1$,
$$
\mathbb{P}\Bigl(\max_{1\le t\le n}|T_t-\mu_e t|\ge b_n(\delta)\Bigr)\le \delta.
$$
Table \ref{tab:real-input-40-event-clock-deviation-certificate} provides the numerical readout of $(\mu_e,\mathrm{Var}(T_n),S_n,b_n)$ for the real input 40-state kernel; since its inputs consist solely of finite-state linear algebra objects of the kernel, this certificate is fully reproducible within the framework of this paper.
\end{remark}

\begin{remark}[External theorem interface for stronger exponential-type certificates]
Proposition \ref{prop:chi-event-clock-deviation-certificate} uses second moments and a union bound to give a uniform bound, the advantage being that no additional assumptions (such as reversibility) are needed and all quantities admit closed-form auditing via finite-dimensional linear algebra.
If one further desires \emph{exponential-type} tail bounds related to the mixing strength, one can treat $T_n$ as an additive functional of a finite-state Markov chain and invoke Markov chain concentration inequalities (e.g., the McDiarmid-type bound of \cite{Paulin2015MarkovConcentration}; in the reversible chain case, the optimal Hoeffding-type bound of \cite{LeonPerron2004OptimalHoeffding} is also available).
These external theorems can uniformly express the deviations of $|T_n-\mu_e n|$ and the corresponding hitting time $\tau_k:=\inf\{n:\,T_n\ge k\}$ as explicit exponential tails controlled by spectral gap/mixing parameters, thereby tightening the error budget of ``event clock rescaling'' to a form comparable in scale with $\tau_{\mathrm{mix}}$.
\end{remark}

\begin{remark}[Coarse time change for event time: $\mu_e$ rescaling of exponential decay rates]\label{lem:chi-event-time-rescaling}
Under the setting of Proposition \ref{prop:chi-slice-time-correlation}, suppose some centered finite-window observable has a step-count time correlation satisfying the exponential envelope
$$
\bigl|\mathrm{Cov}(F,\,G\circ\sigma^n)\bigr|\le C\,\rho^n\qquad (n\ge 0),
$$
where $\rho\in(0,1)$.
Under the Parry equilibrium state at $t=0$, introduce the event clock $T_n=\sum_{s=0}^{n-1}e_s$ and write $\mu_e=\mathbb{E}[e_0]>0$.
Define an ``event time approximate time change'' depending only on the mean density:
$$
\widetilde\tau(k):=\left\lfloor\frac{k}{\mu_e}\right\rfloor\qquad (k\in\mathbb{N}).
$$
Then the corresponding event-time correlation envelope satisfies
$$
\bigl|\mathrm{Cov}(F,\,G\circ\sigma^{\widetilde\tau(k)})\bigr|
\le C\,\rho^{\lfloor k/\mu_e\rfloor}
\le C\,\exp\!\left(-\frac{-\log\rho}{\mu_e}\,k\right).
$$
This result uses only the ``step-count time exponential envelope'' and the event density $\mu_e$, thereby purely rescaling the step-count time scale $\tau_{\mathrm{corr}}=1/(-\log\rho)$ to the event time scale $\tau_{\mathrm{corr}}^{(e)}=\mu_e\tau_{\mathrm{corr}}$.
For background: for finite-state reversible Markov chains, the above exponential envelope can be derived directly from spectral gap covariance decay (see \cite[Chapter~12]{LevinPeresWilmer2009MarkovMixing}).
\end{remark}

\begin{remark}[Auditable deviation certificate for the hitting time $\tau(k)$ (variance type)]
\label{prop:chi-event-clock-hitting-time-certificate}
Under the Parry equilibrium state at $t=0$, let $e_t\in\{0,1\}$ be the output bit, $T_n:=\sum_{t=0}^{n-1}e_t$, $\mu_e:=\mathbb{E}[e_0]>0$, and define the hitting time
$$
\tau(k):=\inf\{n\ge 1:\,T_n\ge k\}\qquad (k\in\mathbb{N}).
$$
For any $\delta\in(0,1)$ and any fixed $k\in\mathbb{N}$, define the two-sided certificate points
$$
n_-(k,\delta):=\max\Bigl\{n\ge 1:\ k-\mu_e n>0\ \text{and}\ \frac{\mathrm{Var}(T_n)}{(k-\mu_e n)^2}\le\delta\Bigr\},
$$
$$
n_+(k,\delta):=\min\Bigl\{n\ge 1:\ \mu_e n-k>0\ \text{and}\ \frac{\mathrm{Var}(T_n)}{(\mu_e n-k)^2}\le\delta\Bigr\}.
$$
Then by Chebyshev's inequality and the identity $\mathbb{P}(\tau(k)>n)=\mathbb{P}(T_n<k)$,
$$
\mathbb{P}\bigl(\tau(k)\le n_-(k,\delta)\bigr)\le\delta,
\qquad
\mathbb{P}\bigl(\tau(k)>n_+(k,\delta)\bigr)\le\delta,
$$
and hence by a union bound
$$
\mathbb{P}\bigl(n_-(k,\delta)<\tau(k)\le n_+(k,\delta)\bigr)\ge 1-2\delta.
$$
Table \ref{tab:real-input-40-hitting-time-deviation-certificate} provides reproducible numerical certificates for several values of $k$.
\end{remark}
