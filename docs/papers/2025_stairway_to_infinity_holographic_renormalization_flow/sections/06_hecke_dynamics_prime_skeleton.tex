
\section{Hecke dynamics: the prime skeleton of the discrete layer}
\label{sec:hecke_dynamics_prime}

\subsection{Hecke operators and their algebraic closure}
\label{sec:hecke_operators}

On spaces of modular forms of level~1 and weight~$k$, Hecke operators $T_n$ form a commutative algebra encoding arithmetic symmetry; standard references include \cite{Apostol1990,DiamondShurman2005}. The defining structure is that Hecke operators are indexed by all $n\ge 1$, but the algebra is generated by the operators at primes.

The fundamental multiplication law is
\begin{equation}
\label{eq:hecke_multiplication_law}
T_m T_n = \sum_{d\mid (m,n)} d^{k-1}\,T_{mn/d^2}.
\end{equation}
In particular, if $(m,n)=1$ then $T_{mn}=T_mT_n$, and for prime powers one obtains the recursion
\begin{equation}
\label{eq:hecke_prime_power}
T_{p^{r+1}} = T_p T_{p^r} - p^{k-1} T_{p^{r-1}},\qquad r\ge 1.
\end{equation}

\subsection{Geometric meaning: Hecke correspondences rescale height}
\label{sec:hecke_geometric_scaling}

Beyond the abstract algebra, Hecke operators have a concrete geometric action on functions on $\HH$ via finite sums of explicit fractional-linear maps (double-coset correspondences). For level~$1$ holomorphic modular forms of weight $k$, a standard explicit formula is (see, e.g., \cite{Apostol1990,DiamondShurman2005})
\begin{equation}
\label{eq:hecke_explicit_formula}
(T_n f)(\tau)=n^{k-1}\sum_{\substack{ad=n\\0\le b<d}} d^{-k}\, f\!\left(\frac{a\tau+b}{d}\right)
=\sum_{\substack{ad=n\\0\le b<d}} \frac{a^{k-1}}{d}\, f\!\left(\frac{a\tau+b}{d}\right),
\end{equation}
In particular, for a prime $p$,
\begin{equation}
\label{eq:hecke_prime_formula}
(T_p f)(\tau)=p^{k-1}f(p\tau)+\frac{1}{p}\sum_{b=0}^{p-1} f\!\left(\frac{\tau+b}{p}\right).
\end{equation}
These maps rescale the slice height $y=\Im\tau$ by rational factors:
\begin{equation}
\Im(p\tau)=p\,y,\qquad \Im\!\left(\frac{\tau+b}{p}\right)=\frac{y}{p}.
\end{equation}
Thus primes act as a discrete cross-scale skeleton: one term pushes deeper into the cusp ($y\mapsto p y$) while the other pulls back toward the boundary ($y\mapsto y/p$).

\begin{remark}[action on the cusp coordinate]
Under $\tau\mapsto p\tau$ one has $q\mapsto q^p$. Under $\tau\mapsto (\tau+b)/p$, one has $q\mapsto \e^{2\pi \iu b/p}\,q^{1/p}$. In the full Hecke sum~\eqref{eq:hecke_prime_formula} the fractional powers cancel, and $T_p$ preserves the usual $q$-expansion space, which is another way to see why the prime skeleton closes on integer-indexed coefficients.
\end{remark}

\subsection{From double cosets to \texorpdfstring{$q$}{q}-coefficients (explicit closure)}
\label{sec:hecke_on_qexp}

The statement that ``fractional powers cancel'' can be made completely explicit at the coefficient level.

\begin{proposition}[Hecke action on Fourier coefficients]
\label{prop:hecke_on_qexp}
Let $f$ be a holomorphic modular form of weight $k$ with $q$-expansion
\begin{equation}
f(\tau)=\sum_{m\ge 0} a_m q^m,\qquad q=\e^{2\pi\iu\tau}.
\end{equation}
Then $T_n f$ has the $q$-expansion
\begin{equation}
(T_n f)(\tau)=\sum_{\ell\ge 0}\left(\sum_{d\mid(n,\ell)} d^{k-1}\,a_{n\ell/d^2}\right) q^\ell.
\end{equation}
In particular, for a prime $p$,
\begin{equation}
(T_p f)(\tau)=\sum_{\ell\ge 0}\left(a_{p\ell}+p^{k-1}a_{\ell/p}\right)q^\ell,
\end{equation}
where $a_{\ell/p}=0$ if $p\nmid \ell$.
\end{proposition}
\begin{proof}
Insert the $q$-expansion of $f$ into the explicit formula~\eqref{eq:hecke_explicit_formula} and use the root-of-unity sum
\begin{equation}
\sum_{b=0}^{d-1} \e^{2\pi\iu m b/d}
=
\begin{cases}
d, & d\mid m,\\
0, & d\nmid m.
\end{cases}
\end{equation}
Writing $m=d m'$ and collecting coefficients of $q^\ell$ yields the stated divisor sum; see \cite{Apostol1990,DiamondShurman2005} for standard derivations.
\end{proof}

\subsection{Eigenforms: stable directions and coefficient rigidity}
\label{sec:eigenforms_coeff_rigidity}

If $f$ is a normalized Hecke eigenform with $q$-expansion
\begin{equation}
f(\tau)=\sum_{n\ge 1} a_n q^n,
\end{equation}
then
\begin{equation}
T_n f = a_n f,
\end{equation}
so the $q$-coefficients coincide with Hecke eigenvalues (up to normalization conventions). The Hecke relations immediately imply two rigid constraints on the coefficient spectrum:
\begin{enumerate}
\item \textbf{Coprime multiplicativity.} If $(m,n)=1$, then
\begin{equation}
a_{mn}=a_m a_n.
\end{equation}
\item \textbf{Prime-power recursion.} For any prime $p$ and $r\ge 1$,
\begin{equation}
a_{p^{r+1}} = a_p a_{p^r} - p^{k-1} a_{p^{r-1}}.
\end{equation}
\end{enumerate}
Thus the discrete spectrum is not an arbitrary integer sequence: it is generated and constrained by the prime skeleton.

\subsection{Example: Ramanujan's discriminant and \texorpdfstring{$\tau(n)$}{tau(n)}}
\label{sec:ramanujan_tau_example}

The classical example is the weight-$12$ cusp form
\begin{equation}
\Delta(\tau)=q\prod_{n\ge 1}(1-q^n)^{24}=\sum_{n\ge 1}\tau(n)q^n,
\end{equation}
whose coefficients $\tau(n)$ satisfy the Hecke constraints above (with $k=12$). Quantitative checks of multiplicativity and prime-power recursion for $\tau(n)$ serve as a clean audit point for the prime-skeleton claim (Section~\ref{sec:quantitative_closure}).
Moreover, the slice--sampling pipeline closes directly on the same eigenform: Appendix~\ref{app:quant_fits_slice_sampling_delta} recovers $\tau(1),\tau(2),\tau(4)$ from scan-orbit samples of the slice $\Delta(x+\iu y)$ with an explicit discrepancy-controlled bound, and Appendix~\ref{app:quant_fits_tau_hecke_closure} certifies the $p=2$ Hecke recursion from the recovered coefficients under explicit error propagation.

\subsection{Euler products and local-to-global closure}
\label{sec:euler_product}

For normalized eigenforms, the Dirichlet series
\begin{equation}
L(s,f)=\sum_{n\ge 1}\frac{a_n}{n^s}
\end{equation}
admits an Euler product factorization
\begin{equation}
L(s,f)=\prod_{p}\left(1-a_p p^{-s}+p^{k-1}p^{-2s}\right)^{-1}.
\end{equation}
This expresses local-to-global closure explicitly: prime-indexed data determine the full spectrum.
