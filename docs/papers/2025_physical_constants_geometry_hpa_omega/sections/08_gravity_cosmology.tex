\section{Generalization III: \texorpdfstring{$c$}{c}, gravity, and cosmology---from compilation lapse to dimensionless combinations}
\label{sec:gravity_cosmology}

\subsection{The status of \texorpdfstring{$c$}{c}: defining constant and baseline propagation}
In the SI, $c$ belongs to the defining-constants layer \cite{BIPMSI2019}. In the HPA program, $c$ is the baseline propagation speed in a regime where both gravitational stretching (delays) and high-energy ``shortcuts'' are negligible. We keep these two layers explicitly separated: metrologically, $c$ is a fixed interface; physically, effective propagation is controlled by geometry and readout mechanisms.

\subsection{Gravitational time delay as computational slowdown: lapse in the \texorpdfstring{$\Omega$}{Omega} framework}
Define a computable ``computational lapse'' from one-dimensional nearest-neighbor compilation/routing costs. This provides an explicit interface with the standard relativistic lapse function that relates coordinate time to proper time in a $3+1$ decomposition \cite{Wald1984}.
\begin{definition}[Computational lapse from routing costs]\label{def:computational_lapse}
Let $\kappa(x)$ denote the local routing overhead and define
\begin{equation}
\lapse(x)=\frac{\kappa_0}{\kappa(x)},\qquad \diff\tau_{\mathrm{loc}}(x)=\lapse(x)\,\diff t.
\end{equation}
\end{definition}
Operationally: larger routing costs imply fewer intrinsic updates per unit background depth, corresponding to gravitational time dilation as computational slowdown.

\begin{proposition}[Interface identification with a GR lapse]\label{prop:omega_gr_lapse_interface}
In a static spacetime with relativistic lapse $N(x)$ (so $\diff\tau=N(x)\,\diff t$ for static observers), define the routing-cost field by $\kappa(x):=\kappa_0/N(x)$. Then the computational lapse $\lapse(x)$ of Definition~\ref{def:computational_lapse} equals the GR lapse $N(x)$. In Schwarzschild coordinates, $N(r)=\sqrt{1-2GM/(rc^2)}$, hence
\begin{equation}
\frac{\kappa(r)}{\kappa_0}=\frac{1}{\sqrt{1-2GM/(rc^2)}}=1+\frac{GM}{rc^2}+O\!\left(\frac{G^2M^2}{r^2c^4}\right).
\end{equation}
\end{proposition}
\begin{proof}
The first statement is immediate from the definitions. The Schwarzschild lapse is the standard $g_{tt}$ relation \cite{Wald1984}, and the expansion uses $(1-x)^{-1/2}=1+x/2+O(x^2)$.
\end{proof}

\subsection{\texorpdfstring{$G$}{G} and cosmological parameters: geometricize only dimensionless combinations}
CODATA provides a recommended value for $G$ with a comparatively large uncertainty \cite{CODATA2022RMP}. We encode the interface discipline as an axiom.

\begin{axiom}[Interface-only treatment of $G$]\label{ax:G_interface}
The Newton constant $G$ is treated as a metrological interface constant that converts geometric/spectral costs into SI readouts. Consequently, the primary geometricization targets are dimensionless combinations such as $Gm^2/(\hbar c)$, $\Lambda\ell_P^2$, and $H_0 t_P$ rather than the decimal value of $G$ itself.
\end{axiom}

\subsection{A worked example: gravitational fine-structure constant at the proton scale}
Define the dimensionless Newton coupling (gravitational ``fine-structure constant'') for a particle of mass $m$ by
\begin{equation}\label{eq:alpha_G_def}
\alpha_G(m):=\frac{Gm^2}{\hbar c},
\end{equation}
and in particular $\alpha_G(p)=Gm_p^2/(\hbar c)$ for the proton. The associated logarithmic inverse
\begin{equation}\label{eq:I_G_def}
I_G(m):=\log\!\big(\alpha_G(m)^{-1}\big)
\end{equation}
is the scan--readout ``impedance'' variable: $\alpha_G(m)=\exp(-I_G(m))$, so $I_G$ is the log-cost of the multiplicative weight $\alpha_G$. This matches the role of logarithms in renormalization and dimensional transmutation \cite{PeskinSchroeder1995,PDGRPP2024}.

\begin{axiom}[Gravitational three-stratum impedance in log form]\label{ax:gravity_log_impedance}
At the proton scale, assign each stratum $j$ a multiplicative weight $w^{(G)}_j:=\exp(-V^{(G)}_j)$ and define the geometric Newton coupling by
\begin{equation}
\alpha_{G,\mathrm{geo}}(p):=\prod_{j} w^{(G)}_j.
\end{equation}
The logarithmic impedance is the log-cost
\begin{equation}
I_G(p):=-\log\alpha_{G,\mathrm{geo}}(p)=V^{(G)}_{\mathrm{bulk}}+V^{(G)}_{\mathrm{boundary}}+V^{(G)}_{\mathrm{line}},
\end{equation}
where the costs $V^{(G)}_j$ are canonical phase volumes with phase spaces
\begin{equation}
\mathcal{M}^{(G)}_{\mathrm{bulk}}\cong U(1)\times SO(3),\qquad
\mathcal{M}^{(G)}_{\mathrm{boundary}}\cong SU(2),\qquad
\mathcal{M}^{(G)}_{\mathrm{line}}\cong U(1),
\end{equation}
and costs $V^{(G)}_j=\mathrm{Vol}(\mathcal{M}^{(G)}_j)$ under the canonical volume normalization.
\end{axiom}

\begin{remark}[Primitive factorization equivalences at fixed volumes]\label{rem:gravity_phase_factorizations}
Under the primitive set $\mathfrak{P}=\{U(1),SU(2),SO(3),\RR P^1\}$ (Axiom~\ref{ax:primitive_phase_spaces}), the canonical-volume identities in Appendix~\ref{app:group_volumes} imply the exchange
\[
\mathrm{Vol}(U(1)\times SO(3))=\mathrm{Vol}(SU(2)\times \RR P^1)=2\pi^3,
\qquad
\mathrm{Vol}(SU(2))=\mathrm{Vol}(U(1)\times \RR P^1)=2\pi^2.
\]
Thus the bulk and boundary volumes in Axiom~\ref{ax:gravity_log_impedance} admit equivalent primitive factorizations related by $\ZZ_2$ quotients; the choice records a projective convention, as in the proton phase-space assignment.
\end{remark}

\begin{proposition}[Primitive factorization rigidity for the gravitational strata]\label{prop:gravity_phase_factorization_rigidity}
Fix the primitive set $\mathfrak{P}$ from Axiom~\ref{ax:primitive_phase_spaces} with canonical volumes (Appendix~\ref{app:group_volumes}). Impose the minimal factor counts compatible with the $\pi$-powers of the stratum costs: realize $V^{(G)}_{\mathrm{line}}=2\pi$ and $V^{(G)}_{\mathrm{boundary}}=2\pi^2$ each by a single primitive, and realize $V^{(G)}_{\mathrm{bulk}}=2\pi^3$ by a product of two primitives. Then
\[
\mathcal{M}^{(G)}_{\mathrm{line}}\cong U(1),\qquad
\mathcal{M}^{(G)}_{\mathrm{boundary}}\cong SU(2),
\]
and the bulk admits exactly the two equivalent factorizations
\[
\mathcal{M}^{(G)}_{\mathrm{bulk}}\cong U(1)\times SO(3)
\qquad\text{or}\qquad
\mathcal{M}^{(G)}_{\mathrm{bulk}}\cong SU(2)\times \mathbb{R}P^1,
\]
which have the same canonical volume (Remark~\ref{rem:gravity_phase_factorizations}).
\end{proposition}
\begin{proof}
By Appendix~\ref{app:group_volumes}, the primitive volumes are $2\pi$, $2\pi^2$, $\pi^2$, $\pi$. For one-factor realizations, $2\pi$ occurs only for $U(1)$ and $2\pi^2$ occurs only for $SU(2)$ among primitives.

For two-factor products, the exponent condition $\pi^3$ forces exponent pattern $(1,2)$ and the coefficient $2$ forces exactly one factor to contribute a coefficient $2$. Hence the two possibilities are $U(1)\times SO(3)$ and $SU(2)\times \mathbb{R}P^1$, which have equal volume by the displayed identities in Remark~\ref{rem:gravity_phase_factorizations}.
\end{proof}

\begin{remark}[Numerical isolation of serial composition in the logarithmic variable]\label{rem:IG_aggregation_isolation}
Let $V_1:=V^{(G)}_{\mathrm{bulk}}=2\pi^3$, $V_2:=V^{(G)}_{\mathrm{boundary}}=2\pi^2$, and $V_3:=V^{(G)}_{\mathrm{line}}=2\pi$. For the CODATA reference value $I_G(p)=\log(\alpha_G(p)^{-1})\approx 88.024824446$, standard alternatives to serial addition give
\begin{equation}
\begin{aligned}
\sqrt{V_1^2+V_2^2+V_3^2} &= 65.3809724311 &&(\text{rel.\ }-2.57\times 10^{-1}),\\
\max\{V_1,V_2,V_3\} &= 62.0125533606 &&(\text{rel.\ }-2.96\times 10^{-1}),\\
(V_1^{-1}+V_2^{-1}+V_3^{-1})^{-1} &= 4.4259282857 &&(\text{rel.\ }-9.50\times 10^{-1}),
\end{aligned}
\end{equation}
where ``rel.'' denotes the relative deviation from $I_G(p)$. By contrast, the serial sum gives $V_1+V_2+V_3=2\pi^3+2\pi^2+2\pi=88.0349474700$ (rel.\ $+1.15\times 10^{-4}$).

In the multiplicative variable $\alpha_G=\exp(-I_G)$, these alternatives correspond to ratios $\alpha_G/\allowbreak \alpha_G(p)$ of order $6.82\times 10^{9}$ (Euclidean), $1.98\times 10^{11}$ (max), and $2.03\times 10^{36}$ (parallel). The serial sum gives $\alpha_{G,\mathrm{geo}}(p)/\allowbreak \alpha_G(p)\approx 9.90\times 10^{-1}$.
\end{remark}

\begin{theorem}[Proton Newton coupling from canonical volumes]\label{thm:alpha_G_proton}
Under Axiom~\ref{ax:gravity_log_impedance},
\begin{equation}
I_G(p)=2\pi^3+2\pi^2+2\pi\approx 88.034947470,
\qquad
\alpha_{G,\mathrm{geo}}(p)=\exp\!\big(-I_G(p)\big)\approx 5.84666\times 10^{-39}.
\end{equation}
\end{theorem}
\begin{proof}
Using Appendix~\ref{app:group_volumes}, $\mathrm{Vol}(U(1))=2\pi$, $\mathrm{Vol}(SU(2))=2\pi^2$, and $\mathrm{Vol}(SO(3))=\pi^2$. Therefore
\begin{equation}
\mathrm{Vol}(U(1)\times SO(3))=(2\pi)(\pi^2)=2\pi^3,
\end{equation}
and the sum gives $I_G(p)=2\pi^3+2\pi^2+2\pi$. Then $\alpha_{G,\mathrm{geo}}(p)=\exp(-I_G(p))$ by the definition~\eqref{eq:I_G_def}.
\end{proof}

\paragraph{Numerical comparison.}
CODATA 2022 gives $\alpha_G(p)=Gm_p^2/(\hbar c)\approx 5.90615\times 10^{-39}$ \cite{CODATA2022RMP}, so the relative deviation of $\alpha_{G,\mathrm{geo}}(p)$ is $-1.01\times 10^{-2}$.
Equivalently,
\begin{equation}
\log\!\left(\frac{\alpha_{G,\mathrm{geo}}(p)}{\alpha_G(p)}\right)\approx -1.01\times 10^{-2}.
\end{equation}
Equivalently, define the multiplicative matching input
\begin{equation}
s_G:=\frac{\alpha_G(p)}{\alpha_{G,\mathrm{geo}}(p)}=\exp\!\left(-\log\!\frac{\alpha_{G,\mathrm{geo}}(p)}{\alpha_G(p)}\right)\approx 1+1.01\times 10^{-2}.
\end{equation}

Equivalently, in the logarithmic impedance variable, $I_G(p)=\log(\alpha_G(p)^{-1})\approx 88.024824446$ (CODATA) versus $I_{G,\mathrm{geo}}(p)=2\pi^3+2\pi^2+2\pi\approx 88.034947470$, i.e.\ $\Delta I_G:=I_{G,\mathrm{geo}}(p)-I_G(p)\approx +1.01\times 10^{-2}$ and $\Delta I_G/I_G(p)\approx +1.15\times 10^{-4}$.

\begin{proposition}[Low-complexity rigidity for $I_G(p)$]\label{prop:alpha_G_rigidity}
Let $I_G(p)$ be computed from CODATA via~\eqref{eq:I_G_def}. Among all triples $(a,b,c)\in\ZZ_{\ge 0}^3$ with $a+b+c\le 10$, the unique minimizer of
\begin{equation}
\big|a\pi^3+b\pi^2+c\pi-I_G(p)\big|
\end{equation}
is $(a,b,c)=(2,2,2)$. The next-best triple in the same coefficient-sum domain has relative error at least $4.94\times 10^{-3}$.
\end{proposition}
\begin{proof}
This is a finite check: enumerate all triples $(a,b,c)\in\ZZ_{\ge 0}^3$ with $a+b+c\le 10$ and minimize the absolute error.
\end{proof}
