\section{Symbols and objects quick reference}
\label{app:symbols}

\begin{itemize}
  \item $\omega_{\Omega}$: global state (Axiom O1).
  \item $\alpha$: scan slope for irrational rotation; golden branch $\alpha=\varphi^{-1}$.
  \item $U,V$: Weyl pair; $UV=\e^{2\pi\iu\alpha}VU$.
  \item $W$: readout window; $s_k=\ind_W(x_k)$ mechanical word.
  \item $E_k^{(\varepsilon)}$: readout effects/POVM elements at resolution $\varepsilon$; $P_k^{(\varepsilon)}=\Tr(\rho_{\mathrm{eff}}E_k^{(\varepsilon)})$.
  \item $D_N^\ast$: star discrepancy; $E_N=ND_N^\ast$ accumulated mismatch.
  \item $\kappa(x)$: routing overhead (implementation depth); $\lapse(x)=\kappa_0/\kappa(x)$ computational lapse.
  \item $\varrho=\chi^2$: information density; $\chi$ Fisher-information amplitude.
  \item $\Phi$: phase potential sourced by mismatch; $\mathbf{P}_{\Phi}=-\nabla\Phi$ phase pressure.
\end{itemize}
