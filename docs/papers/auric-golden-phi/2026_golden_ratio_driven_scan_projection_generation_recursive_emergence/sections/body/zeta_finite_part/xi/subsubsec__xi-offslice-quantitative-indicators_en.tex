\subsubsection{Quantitative indicators of off-critical-slice deviation: exact correspondence of Blaschke origin modulus, real-part deviation, and residual dimension}\label{subsubsec:xi-offslice-quantitative-indicators}
Section \ref{subsubsec:selfdual-blaschke-defect-fixed-slice} has indexed "discrete orthogonal residual axis deviation from fixed slice" as Blaschke defect dimension \(\mathfrak d_{\mathrm{Bl}}\), and interpreted it as point spectral residual and model space complementary dimension of minimal realization.
This section provides a more \emph{quantitative} version under the same caliber: for off-slice zeros (point-wise Blaschke data) we can not only count (residual dimension), but compress their "deviation load" to a single scalar \(|B(0)|\) (origin modulus encapsulation), satisfying strict summation law with zero real-part deviation.
This paragraph will serve as one of the auditable interface coordinates for off-slice defects in subsequent "certificate chain closure" (\S\ref{subsubsec:xi-fixed-slice-audit-chain}).

\paragraph{Self-dual completion and disk coordinates}
Let \(F(s)\) be analytic in critical strip \(0<\Re(s)<1\), satisfying antilinear self-duality
\begin{equation}\label{eq:xi-offslice-selfdual-on-strip}
F(s)=\overline{F(1-\overline s)}.
\end{equation}
Take any scale parameter \(L>0\), define completion disk coordinate
\begin{equation}\label{eq:xi-offslice-exp-disk-coordinate}
u(s):=\exp\!\bigl(-(2s-1)L\bigr)\in\CC^\times.
\end{equation}
Then duality transformation \(s\mapsto 1-\overline s\) in \(u\)-coordinate becomes
\begin{equation}\label{eq:xi-offslice-exp-disk-duality}
u(1-\overline s)=\frac{1}{\overline{u(s)}},
\end{equation}
and critical line \(\Re(s)=\tfrac12\) precisely corresponds to fixed slice \(|u|=1\).
Therefore, symmetry no longer presents as "axis exchange" in this coordinate, but manifests as involution fixed-slice structure of "interior-exterior domains are mirror images" (consistent with semantics in Remarks \ref{rem:xi-critical-line-unit-circle}, \ref{rem:xi-symmetry-object-autread}).

\begin{remark}[Notation alignment with main text Dirichlet section]\label{rem:xi-offslice-exp-vs-dirichlet}
Main text commonly uses completion parameter \(u_{L_0}(s)=L_0^{2s-1}\) (\(L_0>1\), Remark \ref{rem:xi-critical-line-unit-circle}), which likewise satisfies \(|u_{L_0}(s)|=1\iff \Re(s)=\tfrac12\).
Formula \eqref{eq:xi-offslice-exp-disk-coordinate} can be viewed as equivalent choice of this general framework: taking \(L_0=e^{L}\) and composing inversion \(u\mapsto u^{-1}\), can map \(\Re(s)>\tfrac12\) to interior of unit disk \(\mathbb{D}\) (instead of exterior), thus convenient for directly encoding off-slice point-wise data via Blaschke product under Hardy/Smirnov factorization caliber (see \S\ref{subsubsec:selfdual-blaschke-defect-fixed-slice}).
\end{remark}

\paragraph{Blaschke encoding of off-slice zeros and two residual indicators}
Let \(\mathcal{Z}\) be zero multiset of \(F\) in \(0<\Re(s)<1\) (counting by self-dual orbit closure), and denote
\[
\mathcal Z_{+}:=\{s\in\mathcal Z:\ \Re(s)>\tfrac12\}.
\]
For each \(s\in\mathcal Z_{+}\) define
\begin{equation}\label{eq:xi-offslice-a-of-s}
a(s):=u(s),
\end{equation}
then from \(\Re(s)>\tfrac12\) obtain \(|a(s)|<1\).
If \(\mathcal Z_{+}\) is finite multiset (e.g., truncation to height window \(|\Im(s)|\le T\), see below), then corresponding finite Blaschke product can be constructed
\begin{equation}\label{eq:xi-offslice-bz-def}
B_{\mathcal Z}(u):=\prod_{s\in\mathcal Z_{+}}\frac{u-a(s)}{1-\overline{a(s)}\,u}.
\end{equation}
This introduces two types of residual indicators.

\begin{definition}[Residual dimension and residual weight]\label{def:xi-offslice-ind-weight}
Under finite caliber of \eqref{eq:xi-offslice-bz-def}, define:
\begin{enumerate}
  \item \textbf{Residual dimension (integer)}
  \[
  \mathrm{Ind}(\mathcal Z):=\deg B_{\mathcal Z}= \#\mathcal Z_{+}\qquad(\text{counting multiplicity});
  \]
  \item \textbf{Residual weight (real number)}
  \[
  \mathrm{W}(\mathcal Z):=\sum_{s\in\mathcal Z_{+}}\log\frac{1}{|a(s)|}.
  \]
\end{enumerate}
\end{definition}

\paragraph{Origin modulus identity: single-scalar encapsulation of \texorpdfstring{$\mathrm{W}(\mathcal Z)$}{W(Z)}}
Residual weight can be compressed to modulus of Blaschke product at origin.

\begin{theorem}[Blaschke origin modulus identity]\label{thm:xi-offslice-blaschke-origin-modulus}
Let \(B_{\mathcal Z}\) be given by \eqref{eq:xi-offslice-bz-def}, then there holds strict identity
\[
|B_{\mathcal Z}(0)|=\prod_{s\in\mathcal Z_{+}}|a(s)|,
\qquad
\mathrm{W}(\mathcal Z)=-\log|B_{\mathcal Z}(0)|.
\]
\end{theorem}

\begin{proof}
From \eqref{eq:xi-offslice-bz-def},
\(
B_{\mathcal Z}(0)=\prod_{s\in\mathcal Z_{+}}(-a(s))
\);
taking modulus yields \(|B_{\mathcal Z}(0)|=\prod|a(s)|\).
Taking negative logarithm yields \(\mathrm{W}(\mathcal Z)=-\log|B_{\mathcal Z}(0)|\).
\end{proof}

\begin{remark}[Single-scalar interface and "congruence loss" interpretation]\label{rem:xi-offslice-origin-modulus-contract-loss}
In fixed-slice audit, \(|B_{\mathcal Z}(0)|\in(0,1]\) compresses "total deviation load" of off-slice point-wise data to single scalar: \(|B_{\mathcal Z}(0)|=1\) equivalent to no off-slice zeros, while \(|B_{\mathcal Z}(0)|<1\) provides strict residual load.
Under auditable semantics of \(\mathbf{PW}\), this single scalar can be viewed as an addressable audit coordinate: if rejecting externalized carrying of off-slice data via orthogonal recording axis, this load can only enter through failure witness/defect ledger (see interface dichotomy in Proposition \ref{prop:xi-offcritical-falsifiable-restatement} and Theorem \ref{thm:xi-null-complete-trichotomy-offline}).
\end{remark}

\paragraph{Exact summation law of real-part deviation: strict correspondence of \texorpdfstring{$\Re(s)-\tfrac12$}{Re(s)-1/2} and \texorpdfstring{$|B_{\mathcal Z}(0)|$}{|B(0)|}}
From \eqref{eq:xi-offslice-exp-disk-coordinate} obtain
\(
|a(s)|=\exp\!\bigl(-(2\Re(s)-1)L\bigr)
\), thus residual weight strictly equals \(L\)-multiple of sum of real-part deviations.

\begin{theorem}[Real-part deviation summation law]\label{thm:xi-offslice-realpart-sum-law}
Under disk coordinate \eqref{eq:xi-offslice-exp-disk-coordinate}, there holds exact identity
\[
\mathrm{W}(\mathcal Z)
= L\sum_{s\in\mathcal Z_{+}}\bigl(2\Re(s)-1\bigr),
\qquad
|B_{\mathcal Z}(0)|
=\exp\!\Bigl(-L\sum_{s\in\mathcal Z_{+}}\bigl(2\Re(s)-1\bigr)\Bigr).
\]
Let total off-critical-line deviation
\[
\mathcal A(\mathcal Z):=\sum_{s\in\mathcal Z_{+}}\Bigl(\Re(s)-\tfrac12\Bigr),
\]
then we have
\[
\mathcal A(\mathcal Z)=\frac{1}{2L}\,\mathrm{W}(\mathcal Z)
=-\frac{1}{2L}\log|B_{\mathcal Z}(0)|.
\]
\end{theorem}

\begin{proof}
For each \(s\in\mathcal Z_{+}\), from \(|a(s)|=\exp(-(2\Re(s)-1)L)\) obtain
\(
\log(1/|a(s)|)=L(2\Re(s)-1)
\).
Summing over \(s\in\mathcal Z_{+}\) yields first formula; second follows from Theorem \ref{thm:xi-offslice-blaschke-origin-modulus} and exponential rewriting.
Last line merely factors out coefficient \(2\) and rearranges.
\end{proof}

\paragraph{Height truncation and stability corollary: small residual weight \texorpdfstring{$\Rightarrow$}{->} near critical line}
For \(T>0\) let
\[
\mathcal Z_T:=\{s\in\mathcal Z:\ |\Im(s)|\le T\},
\qquad
(\mathcal Z_T)_{+}:=\mathcal Z_T\cap\mathcal Z_{+}.
\]
Since \(F\) is analytic with discrete zeros, \(\mathcal Z_T\) is finite multiset within bounded height window, thus above construction applies.

\begin{corollary}[Near-critical-line stability]\label{cor:xi-offslice-near-critical-stability}
If some truncated zero subset satisfies \(\mathrm{W}(\mathcal Z_T)\le \varepsilon\), then
\[
\max_{s\in(\mathcal Z_T)_{+}}\Bigl(\Re(s)-\tfrac12\Bigr)\le \frac{\varepsilon}{2L},
\qquad
\sum_{s\in(\mathcal Z_T)_{+}}\Bigl(\Re(s)-\tfrac12\Bigr)\le \frac{\varepsilon}{2L}.
\]
\end{corollary}

\begin{proof}
From Theorem \ref{thm:xi-offslice-realpart-sum-law} obtain
\(
\sum_{s\in(\mathcal Z_T)_{+}}(\Re(s)-\tfrac12)=\mathrm{W}(\mathcal Z_T)/(2L)\le \varepsilon/(2L)
\).
Since each term is nonnegative, maximum does not exceed total sum, thus first inequality follows.
\end{proof}

\paragraph{Geometric realization of residual dimension: minimal complementary dimension of model space}
Let \(H^2(\mathbb{D})\) be Hardy space, and denote model space \(K_B:=H^2(\mathbb{D})\ominus BH^2(\mathbb{D})\).

\begin{proposition}[Geometric realization of residual dimension]\label{prop:xi-offslice-model-space-dimension}
If \(B_{\mathcal Z}\) is finite Blaschke product, then
\[
\dim K_{B_{\mathcal Z}}=\deg B_{\mathcal Z}=\mathrm{Ind}(\mathcal Z).
\]
\end{proposition}

\begin{proof}
In finite Blaschke case \(\dim K_B=\deg B\) is standard fact of model space, already recorded as defect dimension caliber in Remark \ref{rem:blaschke-defect-finite}; substituting \(B=B_{\mathcal Z}\) and using Definition \ref{def:xi-offslice-ind-weight} yields result.
\end{proof}

\begin{remark}[Alignment with Blaschke defect dimension \texorpdfstring{$\mathfrak d_{\mathrm{Bl}}$}{d_Bl}]\label{rem:xi-offslice-ind-equals-dbl}
If some disk readout object \(F\)'s disk interior zero set (counting multiplicity) is exactly \(\{a(s):s\in\mathcal Z_{+}\}\), then its Blaschke factor is \(B_F=B_{\mathcal Z}\).
Therefore under finite defect caliber,
\(
\mathrm{Ind}(\mathcal Z)=\deg(B_F)=\mathfrak d_{\mathrm{Bl}}(F)
\)
(Definition \ref{def:blaschke-defect-dimension} and Remark \ref{rem:blaschke-defect-finite}).
This alignment transcribes "off-slice zero counting" to "complementary dimension of minimal external orthogonal residual axis".
\end{remark}

\paragraph{Local-global additivity: residual conservation law under product decomposition}
Suppose there exists product decomposition (e.g., analytic decomposition induced by local correspondence factors)
\[
F(s)=\prod_{\nu}F_{\nu}(s),
\]
and corresponding zero multisets on \(\Re(s)>\tfrac12\) side satisfy disjoint union
\(
\mathcal Z_{+}=\bigsqcup_{\nu}\mathcal Z_{+,\nu}
\)
(counting multiplicity).
Then by construction, \(B_{\mathcal Z}(u)=\prod_{\nu}B_{\mathcal Z_\nu}(u)\).

\begin{theorem}[Residual indicator additivity]\label{thm:xi-offslice-indicators-additivity}
Under above conditions, there hold strict identities
\[
\mathrm{Ind}(\mathcal Z)=\sum_{\nu}\mathrm{Ind}(\mathcal Z_{\nu}),
\qquad
\mathrm{W}(\mathcal Z)=\sum_{\nu}\mathrm{W}(\mathcal Z_{\nu}),
\qquad
|B_{\mathcal Z}(0)|=\prod_{\nu}|B_{\mathcal Z_{\nu}}(0)|.
\]
\end{theorem}

\begin{proof}
Since \(B_{\mathcal Z}=\prod_{\nu}B_{\mathcal Z_\nu}\) and each factor is finite Blaschke product, degree and zero counting by multiplicity are additive, thus \(\mathrm{Ind}\) is additive.
By Definition \ref{def:xi-offslice-ind-weight}, \(\mathrm{W}\) is point-wise summation over \(\mathcal Z_{+}\), therefore strictly additive under disjoint union.
Last formula follows directly from multiplicativity of \(|B_{\mathcal Z}(0)|\) and Theorem \ref{thm:xi-offslice-blaschke-origin-modulus}.
\end{proof}

\paragraph{Criterion form of fixed-slice carrying: no residual \texorpdfstring{$\Leftrightarrow$}{<->} indicator annihilation}
Above quantitative indicators provide a criterion directly isomorphic with "fixed-slice carrying": existence of off-slice zeros equivalent to nonzero residual indicator.

\begin{proposition}[Indicator criterion of fixed-slice carrying]\label{prop:xi-offslice-fixed-slice-indicator-criterion}
Under disk coordinate \eqref{eq:xi-offslice-exp-disk-coordinate} and finite caliber of Definition \ref{def:xi-offslice-ind-weight},
\[
\mathcal Z_{+}=\varnothing
\quad\Longleftrightarrow\quad
\mathrm{Ind}(\mathcal Z)=0
\quad\Longleftrightarrow\quad
\mathrm{W}(\mathcal Z)=0
\quad\Longleftrightarrow\quad
|B_{\mathcal Z}(0)|=1.
\]
\end{proposition}

\begin{proof}
\(\mathcal Z_{+}=\varnothing\) if and only if \(\deg B_{\mathcal Z}=0\), i.e., \(\mathrm{Ind}(\mathcal Z)=0\), which is definition of Blaschke product.
If \(\mathcal Z_{+}\neq\varnothing\), then for each \(s\in\mathcal Z_{+}\) we have \(|a(s)|<1\), thus \(\log(1/|a(s)|)>0\), hence \(\mathrm{W}(\mathcal Z)>0\); conversely \(\mathcal Z_{+}=\varnothing\) gives empty sum \(0\).
Last equivalence group given by Theorem \ref{thm:xi-offslice-blaschke-origin-modulus}: \(\mathrm{W}(\mathcal Z)=0\iff -\log|B_{\mathcal Z}(0)|=0\iff |B_{\mathcal Z}(0)|=1\).
\end{proof}

\begin{remark}[Consequence in protocol semantics: minimal residual axis and verifiable addressing]\label{rem:xi-offslice-protocol-consequence}
Proposition \ref{prop:xi-offslice-fixed-slice-indicator-criterion} shows: if "verifiable readout" is axiomatized at protocol layer as "can be made injective without additional residual register", must enforce \(\mathrm{Ind}(\mathcal Z)=0\), thus \(|B_{\mathcal Z}(0)|=1\), excluding all point-wise zero events with \(\Re(s)\neq\tfrac12\).
This conclusion agrees with multi-axis address caliber of \S\ref{subsec:xi-certificate-coherence-orthogonal-axis-capacity}: off-slice information, if not externalized via orthogonal recording axis, can only trigger \(\mathsf{NULL}\) failure witness in visible semantics, and cannot be bypassed by "implicit mixed precision/implicit continuous coordinate" (Theorem \ref{thm:xi-us-deterministic-tristate} and Corollary \ref{cor:xi-precision-type-no-leakage}).
\end{remark}
