\subsection{Conclusion cluster II: quotient-remainder as a transversal of the translation groupoid}\label{subsec:pom-division-groupoid}
\begin{definition}[Translation groupoid and fundamental domain]\label{def:pom-translation-groupoid}
Fix an integer $b>0$, and let $\mathbb{Z}$ act on $\mathbb{Z}$ by
\[
k\cdot a:=a-kb.
\]
The orbits are precisely the congruence classes $a\bmod b$. Take the fundamental domain
\[
\mathcal F_b:=\{0,1,\dots,b-1\},
\]
and define the projection
\[
\Pi_b(a):=\text{the unique }r\in\mathcal F_b\text{ satisfying }a\equiv r\pmod b.
\]
\end{definition}

\begin{theorem}[Quotient-remainder as a transversal projection]\label{thm:pom-division-section}
For any $a\in\mathbb{Z}$, there exists a unique $(q,r)\in\mathbb{Z}\times\mathcal F_b$ such that
\[
a=qb+r,\qquad r=\Pi_b(a).
\]
Here $q$ is the orbit coordinate, and $r$ is the image of the projection along the translation orbit onto the transversal.
\end{theorem}
