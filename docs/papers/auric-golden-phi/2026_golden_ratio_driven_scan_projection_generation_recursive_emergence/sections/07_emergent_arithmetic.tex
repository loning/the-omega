\section{算术的涌现：在稳定地址空间上的归一化运算}

本节证明：在稳定类型空间（Zeckendorf 语法）上可定义加法与乘法，使其与经典自然数算术同构；运算本身由规范形与可计算归一化过程给出，从而符合“运算为统计稳定读出规律”的要求。

\subsection{从 Fibonacci 关系呈示的尺度计数单子到自然数（商与不变量）}

本节的关键点是：\textbf{算术结构并非预置}，而是从“尺度计数的自由结构”在 Fibonacci 关系下取商（或取不变量）后被强制出来；Zeckendorf 规范形只是该商结构上的一个\textbf{可计算的规范代表系}。

令
\[
\mathcal{D}:=\Bigl\{a=(a_1,a_2,\dots): a_k\in\NN,\ \text{仅有限个 }a_k\neq 0\Bigr\}.
\]
将 $a\in\mathcal{D}$ 视为“在各尺度 $k$ 上出现 $a_k$ 次”的计数配置；$\mathcal{D}$ 在逐点加法下是自由交换幺半群（自由交换单子）。
记 $e_k\in\mathcal{D}$ 为第 $k$ 个标准基（$e_k$ 在第 $k$ 位为 $1$，其余为 $0$）。

\begin{definition}[Fibonacci 关系生成的同余]\label{def:fib-congruence}
记 $\equiv$ 为 $\mathcal{D}$ 上最小的交换幺半群同余，使得
\begin{enumerate}
  \item \textbf{（基准同余）}$e_2\equiv 2e_1$（把“同一权重的重复”折叠为下一尺度的单次出现）；
  \item \textbf{（递推同余）}对所有 $k\ge 1$ 都有
\[
e_{k+2}\ \equiv\ e_{k+1}+e_k.
\]
\end{enumerate}
等价地，$\equiv$ 由“基准折叠 + Fibonacci 递推”在“尺度计数”层面生成，并对加法封闭。
\end{definition}

\begin{definition}[值同态（不变量）]\label{def:val-on-D}
定义映射 $\mathrm{Val}:\mathcal{D}\to\NN$ 为
\[
\mathrm{Val}(a):=\sum_{k\ge 1} a_k F_{k+1}.
\]
这是交换幺半群同态，并与前述对 $X_\infty$ 的 $\mathrm{Val}$ 定义一致（将 $c\in X_\infty$ 视为 $a_k=c_k$ 的特殊情形）。
\end{definition}

\begin{proposition}[Fibonacci 同余的完备性]\label{prop:val-invariant}
若 $a\equiv b$，则 $\mathrm{Val}(a)=\mathrm{Val}(b)$。因此 $\mathrm{Val}$ 在商幺半群 $\mathcal{D}/\!\equiv$ 上诱导良定义的同态
\[
\overline{\mathrm{Val}}:\mathcal{D}/\!\equiv\ \to\ \NN.
\]
\end{proposition}

\textbf{证明：}
只需验证生成关系保持值。对递推同余 $e_{k+2}\equiv e_{k+1}+e_k$，
\[
\mathrm{Val}(e_{k+2})=F_{k+3}=F_{k+2}+F_{k+1}=\mathrm{Val}(e_{k+1}+e_k),
\]
对基准同余 $e_2\equiv 2e_1$，
\[
\mathrm{Val}(e_2)=F_3=2=2F_2=\mathrm{Val}(2e_1).
\]
其余由同余对加法封闭、以及 $\mathrm{Val}$ 的加法同态性推出。

\begin{theorem}[自然数的涌现：商幺半群的同构]\label{thm:monoid-quotient-is-N}
映射 $\overline{\mathrm{Val}}:\mathcal{D}/\!\equiv \to \NN$ 是交换幺半群同构。并且 $X_\infty$（Zeckendorf 合法序列）给出每个同余类的唯一规范代表：对任意 $a\in\mathcal{D}$，存在唯一 $c\in X_\infty$ 使得 $a\equiv c$ 且 $\mathrm{Val}(a)=\mathrm{Val}(c)$。
\end{theorem}

\textbf{证明：}
\begin{itemize}
  \item \textbf{满射性：}对任意 $N\in\NN$，取其 Zeckendorf 数字 $Z(N)=(c_1,c_2,\dots)\in X_\infty$（\cite{Zeckendorf1972}），则 $\mathrm{Val}(Z(N))=N$，故 $\overline{\mathrm{Val}}$ 满射。
  \item \textbf{同构性（用呈示把一切类压到单生成元）：}
  在商幺半群 $\mathcal{D}/\!\equiv$ 中，令 $\varepsilon:=[e_1]$。由基准同余有 $[e_2]=2\varepsilon$。再由递推同余归纳得对所有 $k\ge 1$，
  \[
  [e_{k+2}]=[e_{k+1}]+[e_k],
  \qquad
  \Rightarrow\qquad
  [e_k]=F_{k+1}\,\varepsilon.
  \]
  因此对任意 $a=\sum_{k\ge 1} a_k e_k\in\mathcal{D}$，
  \[
  [a]=\sum_{k\ge 1} a_k [e_k]=\left(\sum_{k\ge 1} a_k F_{k+1}\right)\varepsilon
  =\mathrm{Val}(a)\,\varepsilon.
  \]
  这说明每个同余类都由其值 $\mathrm{Val}(a)$ 完全刻画，且 $\overline{\mathrm{Val}}([a])=\mathrm{Val}(a)$ 正是把 $n\varepsilon$ 送到 $n$ 的单调同构。于是 $\overline{\mathrm{Val}}$ 为交换幺半群同构。
  \item \textbf{规范代表：}对任意 $a\in\mathcal{D}$，令 $N=\mathrm{Val}(a)$，取 $c=Z(N)\in X_\infty$。上一步给出
  \(
  [a]=N\varepsilon=[c],
  \)
  即 $a\equiv c$。唯一性由 Zeckendorf 唯一性推出：若 $c,c'\in X_\infty$ 且 $c\equiv c'$，则 $\mathrm{Val}(c)=\mathrm{Val}(c')$，从而 $c=c'$。
\end{itemize}

\begin{remark}[折叠作为商映射的可计算截面]\label{rem:fold-as-section}
上一定理给出的结论可概括为：
\[
\mathcal{D}\ \twoheadrightarrow\ \mathcal{D}/\!\equiv\ \cong\ \NN
\quad\text{且}\quad
X_\infty \hookrightarrow \mathcal{D}
\]
是一个可计算的截面（选择规范代表）。这正是“折叠（folding）= 取不变量/取商 + 选规范形”的抽象机制在 Fibonacci 情形下的具体实现。
\end{remark}

\subsection{群胚视角：值保持重写的轨道与 Zeckendorf 横截面}\label{sec:groupoid-zeckendorf}

为了把“折叠为何是涌现而非注入”说得更具体，我们把 Fibonacci 关系实现为一族\textbf{可逆局部变换}（群胚生成元），并把 Zeckendorf 合法序列理解为其轨道横截面。

\begin{definition}[Fibonacci 局部变换（可逆重写）]\label{def:fib-moves}
定义以下作用于 $\mathcal{D}$ 的部分可逆变换（在各自可行域上）：
\begin{itemize}
  \item \textbf{基准变换：}
  \[
  E_0(a)=a+2e_1-e_2\quad\text{当 }a_{2}\ge 1,
  \qquad
  E_0^{-1}(a)=a-2e_1+e_2\quad\text{当 }a_{1}\ge 2.
  \]
  \item \textbf{递推变换（$k\ge 1$）：}
  \[
  E_k(a)=a+e_k+e_{k+1}-e_{k+2}\quad\text{当 }a_{k+2}\ge 1,
  \]
  其逆变换 $E_k^{-1}$ 在 $a_k,a_{k+1}\ge 1$ 时定义为
  \[
  E_k^{-1}(a)=a-e_k-e_{k+1}+e_{k+2}.
  \]
\end{itemize}
令 $\mathcal{G}$ 为由 $\{E_k,E_k^{-1}\}_{k\ge 0}$ 生成的群胚（对象为 $\mathcal{D}$，态射为这些部分双射的复合）。
\end{definition}

\begin{proposition}[轨道不变量]\label{prop:orbits-preserve-val}
对任意群胚态射（由 $E_k$ 与 $E_k^{-1}$ 复合得到的可逆变换）$a\mapsto a'$，都有 $\mathrm{Val}(a)=\mathrm{Val}(a')$。
\end{proposition}

\textbf{证明：}
对生成元 $E_k$，
\[
\mathrm{Val}(E_k(a))-\mathrm{Val}(a)
=F_{k+1}+F_{k+2}-F_{k+3}=0,
\]
逆变换同理。复合保持不变量。
对基准生成元 $E_0$ 亦有 $\mathrm{Val}(E_0(a))-\mathrm{Val}(a)=2F_2-F_3=0$，故结论成立。

\begin{theorem}[轨道横截面与折叠的群论解释]\label{thm:zeckendorf-transversal}
对任意 $a\in\mathcal{D}$，其 $\mathcal{G}$-轨道与其值纤维一致：
\[
\mathcal{G}\cdot a \ =\ \{b\in\mathcal{D}:\mathrm{Val}(b)=\mathrm{Val}(a)\}.
\]
并且该轨道与 $X_\infty$ 恰好在一点相交，交点为 $Z(\mathrm{Val}(a))$。因此定义
\[
\Fold_\infty(a):=Z(\mathrm{Val}(a))\in X_\infty
\]
给出“沿轨道投影到横截面”的折叠算子；它把每个不可观测的值保持对称轨道压缩为唯一稳定代表。
\end{theorem}

\textbf{证明：}
由命题 \ref{prop:orbits-preserve-val} 得 $\mathcal{G}\cdot a$ 包含于值纤维。反向包含由定理 \ref{thm:monoid-quotient-is-N} 给出：若 $\mathrm{Val}(b)=\mathrm{Val}(a)$，则 $[b]=[a]$，即 $b\equiv a$。而 $\equiv$ 正是由基准同余 $e_2\equiv 2e_1$ 与递推同余 $e_{k+2}\equiv e_{k+1}+e_k$ 在配置空间上的闭包；相应的局部可逆重写由 $E_0$ 与 $E_k$（$k\ge 1$）实现，故存在群胚态射把 $a$ 送到 $b$。最后，值纤维中落在 $X_\infty$ 的元素按 Zeckendorf 唯一性至多一个，而 $Z(\mathrm{Val}(a))$ 显然属于该纤维，故交点唯一且为所述。

\subsection{Zeckendorf 值与稳定地址的数值化}

对 $w=w_1\cdots w_m\in X_m$，定义其 Zeckendorf 值
\[
V_m(w)=\sum_{k=1}^m w_k F_{k+1}.
\]
当 $0\le N\le F_{m+2}-1$ 时，$N$ 的 Zeckendorf 数字 $Z(N)=(c_1,c_2,\dots)$ 在第 $m+1$ 位之后全为 $0$（引理 \ref{lem:zeckendorf-range-bijection}），故令
\[
w(N):=\pi_m(Z(N))=(c_1,\dots,c_m)\in X_m,
\]
则
\[
N=V_m\bigl(w(N)\bigr).
\]
因此 $X_m$ 在该范围上可作为自然数的规范“地址表示”。

\subsection{加法作为归一化：从规范形到交换幺半群}

考虑无限 Zeckendorf 地址空间
\[
X_\infty=\{(c_1,c_2,\dots): c_k\in\{0,1\},\ c_kc_{k+1}=0,\ \text{仅有限个 }1\}.
\]
定义值映射
\[
\mathrm{Val}(c)=\sum_{k\ge1}c_kF_{k+1}\in\NN.
\]

为避免将“合流/终止”作为额外假设，本文首先用 Zeckendorf 唯一性给出运算的数学定义（\cite{Zeckendorf1972}），随后再说明其可计算实现（附录 E）。

\begin{definition}[稳定加法]\label{def:stable-add}
对 $c,d\in X_\infty$，定义
\[
c\oplus d:=\mathrm{Val}^{-1}\bigl(\mathrm{Val}(c)+\mathrm{Val}(d)\bigr)\in X_\infty.
\]
\end{definition}

\begin{remark}[关于“同义反复”的澄清：定义性同构 vs. 可观测实现]\label{rem:arith-not-novel-as-abstract}
需要承认：定义 \ref{def:stable-add} 将运算直接拉回到 $(\NN,+)$ 上，因此“\(\mathrm{Val}\) 给出同构”这一结论在抽象代数层面是\textbf{定义性的}，并不产生一个与 \((\NN,+)\) 非同构的新幺半群结构。

本文把该同构写成定理的目的仅是：固定记号并强调“无需把 \(+\) 作为原语输入”。本节真正承担的数学内容是另外两类、与“扫描—投影读出”接口对齐的结论：
\begin{itemize}
  \item \textbf{（轨道/折叠的内容）}由群胚重写生成的轨道与值纤维一致，并存在唯一横截面（定理 \ref{thm:zeckendorf-transversal}）。这给出“折叠 = 沿值保持对称轨道投影到唯一稳定代表”的严格形式。
  \item \textbf{（纯数字层的内容）}运算可等价表述为“逐位叠加/卷积 $\Rightarrow$ 归一化折叠”（命题 \ref{prop:add-as-fold}），从而把抽象的 \(\mathrm{Val}^{-1}\) 定义转成一个可实现、可审计的归一化规则；其复杂度界见附录 E$'$。
\end{itemize}
\end{remark}

\begin{theorem}[Zeckendorf 加法的纯数字层实现（线性时间/有限状态）]\label{thm:zeckendorf-add-fst}
设 $c,d\in X_\infty$ 仅有限个 $1$，并令
\(
m:=\max\{k:\ c_k=1\text{ 或 }d_k=1\}
\)
为有效分辨率。则存在一个仅对 Zeckendorf 数字进行局部有限操作的算法（等价地：一个确定的有限状态换能器/有限延迟顺序换能器），在时间 $O(m)$ 内直接从 $(c_1,\dots,c_m)$ 与 $(d_1,\dots,d_m)$ 计算出 $c\oplus d$ 的 Zeckendorf 数字输出，而不需要显式构造整数 $\mathrm{Val}(c),\mathrm{Val}(d)$。
\end{theorem}

\begin{proof}
这是 Zeckendorf/黄金比例 Ostrowski 数制的标准“归一化可由有限自动机实现”结论在加法特例下的表述：加法可视为先逐位相加得到字母表 $\{0,1,2\}$ 上的非规范数字，再执行一个保持值不变的规范化（消去 $2$ 与相邻 $11$）得到唯一 Zeckendorf 形式。Pisot 基（特别是黄金比例）情形下，这类规范化映射可由有限自动机/顺序换能器实现，从而给出线性时间算法（见 \cite{Frougny1992NumbersAutomata,Frougny1999OnlineAddition}；对 Zeckendorf 情形的显式高效算法与实现细节可参见 \cite{AhlbachUsatineFrougnyPippenger2013}；Ostrowski 三遍扫描算法的黄金比例特例亦见 \cite{HieronymiTerry2018OstrowskiAddition}）。
\end{proof}

\begin{remark}[on-line 与不可并行化：时间读出语义的实现侧约束]\label{rem:online-not-parallel}
在“扫描—读出”的语义里，运算规则若能由一个 on-line 换能器实现，意味着读出可以沿时间单向流式推进并维持有限记忆状态，这与第 6 节的滑动块算子视角一致。另一方面，黄金比例基上的加法虽可 on-line 实现，但一般不可并行化（即不存在把输入划分为并行块并在常数轮数内完成的加法电路口径），这一现象在本文语境中可解释为：\textbf{稳定读出虽是有限记忆的，但仍携带跨尺度的进位/归一化依赖}（参见 \cite{Frougny1999OnlineAddition} 的精确结论与讨论）。
\end{remark}

\begin{proposition}[折叠公式：加法即“叠加后归一化”]\label{prop:add-as-fold}
将 $c,d\in X_\infty$ 视为 $\mathcal{D}$ 中的 $0/1$ 配置，令其逐点和为
\[
(c+d)_k:=c_k+d_k\in\{0,1,2\},
\]
则
\[
c\oplus d=\Fold_\infty(c+d).
\]
\end{proposition}

\textbf{证明：}
由定义 \ref{def:val-on-D}，$\mathrm{Val}(c+d)=\mathrm{Val}(c)+\mathrm{Val}(d)$，再由定理 \ref{thm:zeckendorf-transversal} 的定义 $\Fold_\infty(a)=Z(\mathrm{Val}(a))=\mathrm{Val}^{-1}(\mathrm{Val}(a))$ 即得。

\begin{theorem}[加法的定义性同构（记号固定）]\label{thm:add-definitional}
$(X_\infty,\oplus)$ 为交换幺半群，且 $\mathrm{Val}:(X_\infty,\oplus)\to(\NN,+)$ 为同构。
\end{theorem}

\textbf{证明：}
由定义 \ref{def:stable-add} 直接有 $\mathrm{Val}(c\oplus d)=\mathrm{Val}(c)+\mathrm{Val}(d)$。又 $\mathrm{Val}$ 为双射（Zeckendorf 唯一性），故 $(X_\infty,\oplus)$ 的结合律、交换律与单位元均由 $(\NN,+)$ 传递得到，且 $\mathrm{Val}$ 给出同构。

该定理将加法解释为“从非规范叠加到稳定地址的投影读出”：原始叠加（逐位相加）一般不保持稳定语法，需要通过规范化投影到稳定地址空间。

\subsection{乘法作为二层归一化：卷积与规范形}

\begin{definition}[稳定乘法]\label{def:stable-mul}
对 $c,d\in X_\infty$，定义
\[
c\otimes d:=\mathrm{Val}^{-1}\bigl(\mathrm{Val}(c)\cdot\mathrm{Val}(d)\bigr)\in X_\infty.
\]
\end{definition}

\begin{theorem}[乘法的定义性同构（记号固定）]\label{thm:mul-definitional}
$(X_\infty,\oplus,\otimes)$ 为交换半环，且 $\mathrm{Val}$ 给出与 $(\NN,+,\times)$ 的半环同构。
\end{theorem}

\textbf{证明：}
由定义 \ref{def:stable-add} 与 \ref{def:stable-mul}，$\mathrm{Val}$ 同时保持加法与乘法。由于 $\mathrm{Val}$ 为双射，交换律、结合律与分配律均从 $(\NN,+,\times)$ 传递，因而得到交换半环同构。

由此，自然数、加法、乘法在稳定地址空间中完全由\textbf{语法约束 + 规范形归一化}定义，而不需将 $(+,\times)$ 作为原语预置。

\begin{proposition}[可计算实现]\label{prop:zeckendorf-arith-computable}
对任意 $c,d\in X_\infty$，$c\oplus d$ 与 $c\otimes d$ 可由有限步算法计算：先在值层执行整数加法/乘法得到 $\mathrm{Val}(c)\pm\mathrm{Val}(d)$ 或 $\mathrm{Val}(c)\mathrm{Val}(d)$，再以 Zeckendorf 贪心算法取唯一规范形（附录 C）。
\end{proposition}

\subsection{显式算法与复杂度界（审稿级）}\label{subsec:zeckendorf-algorithms-complexity}

本小节把“可计算、可终止”的表述提升为\textbf{显式算法}与\textbf{复杂度界}。为保持依赖最小，算法采用最短闭环：\textbf{值层运算}（整数加法/乘法）+\textbf{Zeckendorf 规范化}（贪心）。

\subsubsection{Zeckendorf 规范化：从整数到合法数字}

给定 $N\in\NN$，目标是计算其唯一 Zeckendorf 数字 $Z(N)=(c_1,c_2,\dots)$（仅有限个 $1$ 且无相邻 $1$）使得 $N=\sum_{k\ge 1}c_kF_{k+1}$。

\begin{itemize}
  \item \textbf{输入：}整数 $N$。
  \item \textbf{输出：}Zeckendorf 合法数字序列 $Z(N)$。
\end{itemize}

\textbf{算法（贪心 + 跳邻位）：}
\begin{enumerate}
  \item 预生成 Fibonacci 权重 $F_{2},F_{3},\dots,F_{M+1}$，其中 $M$ 足够大使得 $F_{M+1}\le N<F_{M+2}$；
  \item 从 $k=M$ 递减到 $1$：若 $F_{k+1}\le N$，则置 $c_k:=1$、令 $N\leftarrow N-F_{k+1}$，并跳过相邻位（将 $k\leftarrow k-1$ 再继续递减）；否则置 $c_k:=0$；
  \item 输出 $c=(c_1,c_2,\dots,c_M)$（其后全为 $0$）。
\end{enumerate}

该算法的正确性与唯一性由 Zeckendorf 定理给出（\cite{Zeckendorf1972}；本稿中见附录 C）。

\subsubsection{稳定加法与稳定乘法：两步实现}

对输入 $c,d\in X_\infty$（仅有限个 $1$），令其有效分辨率为
\[
m:=\max\{k:\ c_k=1\text{ 或 }d_k=1\}.
\]
则
\begin{itemize}
  \item \textbf{稳定加法} $c\oplus d$：计算 $N=\mathrm{Val}(c)+\mathrm{Val}(d)$，再输出 $Z(N)$；
  \item \textbf{稳定乘法} $c\otimes d$：计算 $N=\mathrm{Val}(c)\mathrm{Val}(d)$，再输出 $Z(N)$。
\end{itemize}
这正是附录 E 的“值层运算 + 规范形”的最短闭环实现。

\subsubsection{复杂度（按分辨率 \texorpdfstring{$m$}{m} 与按比特两种口径）}

为了避免模型争议，本文在附录 E$'$ 中同时给出两种等价口径的复杂度界：分辨率口径（把 $m$ 视为输入长度）与比特口径（把整数按二进制长度计成本）。结论可概括如下（详证见附录 E$'$ 中的命题 \ref{prop:fold-complexity} 与 \ref{prop:stable-arith-complexity}）：

\begin{itemize}
  \item \textbf{折叠/规范化：}对长度 $m$ 的输入，Zeckendorf 规范化在分辨率口径为 $O(m)$ 步；在比特口径为 $O(m^2)$ 比特运算。
  \item \textbf{稳定加法：}分辨率口径 $O(m)$；比特口径 $O(m^2)$。
  \item \textbf{稳定乘法：}分辨率口径为“值层乘法成本 + $O(m)$”；比特口径为 $O(M(m)+m^2)$，其中 $M(m)$ 为 $\\Theta(m)$ 比特整数乘法成本（学校乘法为 $O(m^2)$）。
\end{itemize}

\begin{remark}[直接在 Zeckendorf 数字上归一化]
除“值层运算 + 贪心规范化”外，也可在 Zeckendorf 数字上给出纯局部重写的 carry/normalize 算法，从而避免显式构造大整数值层。该方向可给出更强的实现层贡献（例如线性时间归一化与平均复杂度分析），但需要额外的合流/终止与复杂度证明，本文在最小输入原则下暂将其作为可选扩展。
\end{remark}
