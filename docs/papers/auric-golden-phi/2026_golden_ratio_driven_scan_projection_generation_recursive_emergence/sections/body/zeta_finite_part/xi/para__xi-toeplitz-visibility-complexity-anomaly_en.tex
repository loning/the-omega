\begin{corollary}[Visibility Complexity "Anomaly" Covariance Law Under Uniform Rescaling: \texorpdfstring{$\mathscr{C}_{\partial}(m)=\mathscr{C}_{\partial}(1)+\log m$}{C(m)=C(1)+log m}]\label{cor:xi-toeplitz-visibility-complexity-anomaly}
Under the point-defect setting of Corollary \ref{cor:xi-toeplitz-visibility-complexity-threshold}, let \(\mathsf P\) be the corresponding Pick--Poisson Gram matrix, and define the boundary entropy potential and boundary volume potential
\[
S_{\partial}:=-\log\det\mathsf P,
\qquad
V_{\partial}:=\log\mathrm{tr}(\mathsf P)=\log\Bigl(\sum_{j=1}^{\kappa}p_j\Bigr),
\]
together with the visibility complexity functional
\[
\mathscr{C}_{\partial}:=S_{\partial}+(\kappa-1)V_{\partial}.
\]
Then the threshold bracket in Corollary \ref{cor:xi-toeplitz-visibility-complexity-threshold} can be written as
\[
\log(1+C_0)+\mathscr{C}_{\partial}.
\]
Moreover, under the uniform rescaling \(\phi_m\) (Proposition \ref{prop:xi-pick-poisson-det-scaling-law}), if we let \(\mathsf P(m)\) be the rescaled Gram matrix and correspondingly define
\(
S_{\partial}(m):=-\log\det\mathsf P(m)
\), \(V_{\partial}(m):=\log\mathrm{tr}(\mathsf P(m))\), \(\mathscr{C}_{\partial}(m):=S_{\partial}(m)+(\kappa-1)V_{\partial}(m)\),
then for all \(m>0\) the strict covariance law holds
\[
\boxed{\;
\mathscr{C}_{\partial}(m)=\mathscr{C}_{\partial}(1)+\log m\; }.
\]
\end{corollary}
\begin{proof}
By the factorization of Proposition \ref{prop:xi-pick-poisson-det-factorization}
\(
\det\mathsf P=(\prod_j p_j)(\prod_{j<k}\rho_{jk}^2)
\),
we have
\(
S_{\partial}=-\sum_j\log p_j-2\sum_{j<k}\log\rho_{jk}
\),
thus the logarithmic potential inside the bracket in Corollary \ref{cor:xi-toeplitz-visibility-complexity-threshold} is precisely
\(
\log(1+C_0)+S_{\partial}+(\kappa-1)V_{\partial}
\).
\par
On the other hand, by Proposition \ref{prop:xi-pick-poisson-det-scaling-law},
\(
\det\mathsf P(m)=m^{-\kappa}\det\mathsf P
\)
and \(p_j(m)=m^{-1}p_j\), hence
\[
S_{\partial}(m)=S_{\partial}+\kappa\log m,
\qquad
V_{\partial}(m)=\log\Bigl(\sum_j p_j(m)\Bigr)=\log\Bigl(m^{-1}\sum_j p_j\Bigr)=V_{\partial}-\log m.
\]
Substituting into \(\mathscr{C}_{\partial}(m)=S_{\partial}(m)+(\kappa-1)V_{\partial}(m)\) yields
\(
\mathscr{C}_{\partial}(m)=\mathscr{C}_{\partial}(1)+\log m
\).
QED.
\end{proof}
