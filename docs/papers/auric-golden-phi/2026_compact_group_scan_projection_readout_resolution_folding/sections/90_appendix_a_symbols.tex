\section{附录A　符号与基本对象}

\begin{enumerate}
  \item $\TT=\RR/\ZZ$，$\TT^d$ 为 $d$ 维环面。
  \item $\varphi=(1+\sqrt5)/2$ 为黄金比；常用斜率 $\alpha=\varphi^{-1}$ 或其等价变换。
  \item 扫描轨道：$x_n=x_0+n\alpha\pmod1$。
  \item 二值读出：$s_n=\ind_W(x_n)$。
  \item $D_N^*$：一维 star discrepancy。
  \item Fibonacci 数列：$F_1=F_2=1$，$F_{n+2}=F_{n+1}+F_n$。
  \item 黄金均值语言：$X_m=\{w\in\{0,1\}^m:\text{无子词 }11\}$，$|X_m|=F_{m+2}$。
  \item 分辨率折叠：$\Fold_m:\{0,1\}^m\to X_m$（由 Zeckendorf 规范化诱导）。
  \item 循环可容许：
  \[
  X_m^{\mathrm{cyc}}=\{w\in X_m:w_mw_1\neq 11\};
  \qquad
  X_m^{\mathrm{bdry}}=X_m\setminus X_m^{\mathrm{cyc}}.
  \]
  \item Weyl 对：$U_{\mathrm{scan}}V=e^{2\pi i\alpha}VU_{\mathrm{scan}}$。
\end{enumerate}
