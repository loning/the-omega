\begin{corollary}[Strict dissipation of bulk-boundary holographic gap: monotonicity and limit of \texorpdfstring{$\Xi(m)$}{Xi(m)}]\label{cor:xi-bulk-boundary-holographic-gap-dissipation}
Using finite atomic notation from Proposition \ref{prop:xi-hankel-det-scaling-covariance}, take boundary defect entropy family
\(
S_{\mathrm{gen}}(m):=-\log\det\mathsf P(m)
\)
as unified dilation family defined in Proposition \ref{prop:xi-pick-poisson-det-scaling-law} (equivalently Corollary \ref{cor:xi-pick-poisson-entropy-derivative-visible-dimension}).
Define bulk-boundary holographic gap
\[
\Xi(m):=S_{\mathrm{bulk}}(m)-\kappa\,S_{\mathrm{gen}}(m)\qquad(m>0).
\]
Let \(s=\log m\).
Then \(\Xi(e^s)\) is differentiable on \(s\in\RR\) and strictly monotone decreasing, satisfying explicit derivative formula
\[
\boxed{\;
\frac{\dd}{\dd s}\,\Xi(e^s)
=-(2\kappa-1)\sum_{j=1}^{\kappa}\frac{1}{1+e^s\delta_j}\;<\;0\;}
\]
hence there exists a finite limit
\[
\boxed{\;
\lim_{m\to\infty}\Xi(m)
=\Xi(1)-(2\kappa-1)\sum_{j=1}^{\kappa}\log\frac{1+\delta_j}{\delta_j}\; }.
\]
\end{corollary}
\begin{proof}
By Proposition \ref{prop:xi-hankel-det-scaling-covariance},
\(
S_{\mathrm{bulk}}(m)=S_{\mathrm{bulk}}(1)-\log J_{\kappa-1}(m;\delta)
\);
while by Proposition \ref{prop:xi-pick-poisson-det-scaling-law},
\(
S_{\mathrm{gen}}(m)=S_{\mathrm{gen}}(1)+\kappa\log m
\).
Therefore
\[
\Xi(m)
=\Xi(1)-\log J_{\kappa-1}(m;\delta)-\kappa^2\log m.
\]
Substituting
\(
\log J_{\kappa-1}(m;\delta)
=\kappa(\kappa-1)\log m+(2\kappa-1)\sum_{j=1}^{\kappa}\bigl(\log(1+\delta_j)-\log(1+m\delta_j)\bigr)
\)
and simplifying yields
\[
\Xi(m)=\Xi(1)+(2\kappa-1)\sum_{j=1}^{\kappa}\log\frac{1+m\delta_j}{m(1+\delta_j)}.
\]
Setting \(m=e^s\) and differentiating with respect to \(s\) gives
\[
\frac{\dd}{\dd s}\,\Xi(e^s)
=(2\kappa-1)\sum_{j=1}^{\kappa}\left(\frac{e^s\delta_j}{1+e^s\delta_j}-1\right)
=-(2\kappa-1)\sum_{j=1}^{\kappa}\frac{1}{1+e^s\delta_j}\;<0.
\]
Finally from \(\log(1+m\delta_j)=\log m+\log\delta_j+o(1)\) (\(m\to\infty\)) we get the limit formula. \qed
\end{proof}

\begin{corollary}[Closed form of bulk scale entropy flow: logistic staircase decomposition and \texorpdfstring{$\varphi$}{phi}-normalized effective dimension]\label{cor:xi-bulk-entropy-rg-staircase}
Still under finite atomic setting of Proposition \ref{prop:xi-hankel-det-scaling-covariance}, let \(s=\log m\).
Then \(S_{\mathrm{bulk}}(e^s)\) is differentiable on \(s\in\RR\), satisfying first-order flow equation
\[
\boxed{\;
\frac{\dd}{\dd s}S_{\mathrm{bulk}}(e^s)
=\kappa^{2}-(2\kappa-1)\sum_{j=1}^{\kappa}\frac{1}{1+e^{s}\delta_{j}}\;}
\]
hence its \(\varphi\)-scaling normalized scale effective dimension
\[
d_{\mathrm{eff}}^{\mathrm{bulk}}(s):=\frac{1}{\log\varphi}\frac{\dd}{\dd s}S_{\mathrm{bulk}}(e^s)
\]
is a linear superposition of \(\kappa\) shifted logistic tail functions \((1+e^{s}\delta_j)^{-1}\).
Each depth atom \(\delta_j\) induces a single monotone transition step centered at \(\sigma_j:=-\log\delta_j\), with step height identically \((2\kappa-1)/\log\varphi\), while endpoint values are determined solely by \(\kappa\).
\end{corollary}
\begin{proof}
From \(S_{\mathrm{bulk}}(m)=S_{\mathrm{bulk}}(1)-\log J_{\kappa-1}(m;\delta)\) and closed form from Proposition \ref{prop:xi-hankel-det-scaling-covariance}
\[
\log J_{\kappa-1}(m;\delta)
=\kappa(\kappa-1)\log m
+(2\kappa-1)\sum_{j=1}^{\kappa}\bigl(\log(1+\delta_j)-\log(1+m\delta_j)\bigr),
\]
setting \(m=e^s\) and differentiating with respect to \(s\) yields
\[
\frac{\dd}{\dd s}S_{\mathrm{bulk}}(e^s)
=-\kappa(\kappa-1)+(2\kappa-1)\sum_{j=1}^{\kappa}\frac{e^s\delta_j}{1+e^s\delta_j}
=\kappa^2-(2\kappa-1)\sum_{j=1}^{\kappa}\frac{1}{1+e^s\delta_j}.
\]
Remaining statements follow directly from \(\frac{1}{1+e^s\delta_j}=\frac{1}{1+e^{s-\sigma_j}}\) and linear superposition. \qed
\end{proof}

\begin{corollary}[Strict convexity of bulk entropy: entropy curvature closed form in dilation flow and everywhere positive]\label{cor:xi-bulk-entropy-strict-convexity}
Still under finite atomic setting of Proposition \ref{prop:xi-hankel-det-scaling-covariance}, let \(s=\log m\).
Then \(S_{\mathrm{bulk}}(e^s)\) is twice differentiable and satisfies strict convexity
\[
\boxed{\;
\frac{\dd^2}{\dd s^2}S_{\mathrm{bulk}}(e^s)
=(2\kappa-1)\sum_{j=1}^{\kappa}\frac{e^s\delta_j}{(1+e^s\delta_j)^2}\;>\;0\;}
\qquad(s\in\RR).
\]
\end{corollary}
\begin{proof}
From \(S_{\mathrm{bulk}}(m)=S_{\mathrm{bulk}}(1)-\log J_{\kappa-1}(m;\delta)\) and closed form from Proposition \ref{prop:xi-hankel-det-scaling-covariance},
\[
S_{\mathrm{bulk}}(e^s)
=S_{\mathrm{bulk}}(1)-\kappa(\kappa-1)s-(2\kappa-1)\sum_{j=1}^{\kappa}\bigl(\log(1+\delta_j)-\log(1+e^s\delta_j)\bigr).
\]
Differentiating twice and simplifying gives stated curvature formula; each term is strictly positive, hence strictly convex. \qed
\end{proof}

\begin{remark}[Fermionic interpretation of bulk entropy potential: grand canonical form of two-state modes and half-filling peak]\label{rem:xi-bulk-entropy-fermi-grand-canonical}
Using finite atomic notation from Proposition \ref{prop:xi-hankel-det-scaling-covariance} and setting \(m=e^s\).
Define single-mode partition factor and occupation function
\[
Z_j(s):=1+e^{s}\delta_j,
\qquad
n_j(s):=\frac{e^{s}\delta_j}{1+e^{s}\delta_j}\in(0,1).
\]
Then from closed form in proof of Corollary \ref{cor:xi-bulk-entropy-strict-convexity} we can write bulk entropy potential as
\[
\boxed{\;
S_{\mathrm{bulk}}(e^{s})
=S_{\mathrm{bulk}}(1)-\kappa(\kappa-1)s
+(2\kappa-1)\sum_{j=1}^{\kappa}\log\!\Bigl(\frac{Z_j(s)}{Z_j(0)}\Bigr)\;}
\]
i.e.,
\(
S_{\mathrm{bulk}}(e^{s})=\mathrm{const}-\kappa(\kappa-1)s+(2\kappa-1)\sum_j\log Z_j(s)
\).
Therefore \(S_{\mathrm{bulk}}\) in strict sense equals a linear rescaling of grand canonical free energy of \(\kappa\) independent two-state modes (Fermi occupation): \(\log\delta_j\) enters as one-body bias, \(s\) enters as linear control parameter (can be interpreted physically as chemical potential coordinate or equivalent potential coordinate).
\par
By Corollaries \ref{cor:xi-bulk-entropy-rg-staircase} and \ref{cor:xi-bulk-entropy-strict-convexity}, first and second derivatives can be written as
\[
\frac{\dd}{\dd s}S_{\mathrm{bulk}}(e^s)=-\kappa(\kappa-1)+(2\kappa-1)\sum_{j=1}^{\kappa}n_j(s),
\qquad
\frac{\dd^2}{\dd s^2}S_{\mathrm{bulk}}(e^s)=(2\kappa-1)\sum_{j=1}^{\kappa}n_j(s)\bigl(1-n_j(s)\bigr).
\]
Here each fluctuation term \(n_j(1-n_j)\) reaches unique peak at \(s=\sigma_j:=-\log\delta_j\), satisfying \(n_j(\sigma_j)=1/2\).
\end{remark}

\begin{proposition}[Microcanonical dual of bulk fermionization potential: Legendre transform gives binary entropy sum representation with no continuous phase coexistence]\label{prop:xi-bulk-entropy-microcanonical-legendre-dual}
Under notation of Remark \ref{rem:xi-bulk-entropy-fermi-grand-canonical}, let
\[
\Phi_{\mathrm{F}}(s):=\sum_{j=1}^{\kappa}\log(1+e^s\delta_j),
\qquad
N(s):=\Phi_{\mathrm{F}}'(s)=\sum_{j=1}^{\kappa}n_j(s).
\]
Then \(N(s)\) is strictly increasing and bijectively maps \(s\in\RR\) to \((0,\kappa)\).
For any \(N\in(0,\kappa)\), define Legendre dual (microcanonical entropy)
\[
\Sigma(N):=\sup_{s\in\RR}\bigl(sN-\Phi_{\mathrm{F}}(s)\bigr).
\]
Then optimal point \(s(N)\) is unique.
Setting \(n_j:=n_j(s(N))\), there exists a completely explicit binary entropy sum representation
\[
\boxed{\;
\Sigma(N)= -\sum_{j=1}^{\kappa}\Bigl(n_j\log n_j+(1-n_j)\log(1-n_j)\Bigr)\;-\;\sum_{j=1}^{\kappa}n_j\log\delta_j\;.}
\]
Moreover by strict convexity, \(\Sigma\) has no flat segment, hence with \(s\) as continuous control parameter there is no continuous phase coexistence.
\end{proposition}
\begin{proof}
From
\(
\Phi_{\mathrm{F}}''(s)=\sum_{j=1}^{\kappa}\frac{e^s\delta_j}{(1+e^s\delta_j)^2}
=\sum_{j=1}^{\kappa}n_j(s)\bigl(1-n_j(s)\bigr)>0
\)
we know \(\Phi_{\mathrm{F}}\) is strictly convex, hence \(N=\Phi_{\mathrm{F}}'\) is strictly increasing.
As \(s\to-\infty\), \(n_j(s)\to 0\); as \(s\to+\infty\), \(n_j(s)\to 1\), hence \(N(s)\) bijectively maps \(\RR\) to \((0,\kappa)\).
Therefore for each \(N\in(0,\kappa)\) there exists unique \(s(N)\) such that \(N=N(s(N))\), giving Legendre extremal point.
\par
Setting \(s=s(N)\) and \(n_j=n_j(s)\).
From \(n_j/(1-n_j)=e^s\delta_j\) we get
\(
s=\log\frac{n_j}{1-n_j}-\log\delta_j
\),
and
\(
\log(1+e^s\delta_j)=-\log(1-n_j)
\).
Hence
\[
sN-\Phi_{\mathrm{F}}(s)
=\sum_{j=1}^{\kappa}\Bigl(s\,n_j-\log(1+e^s\delta_j)\Bigr)
=-\sum_{j=1}^{\kappa}\Bigl(n_j\log n_j+(1-n_j)\log(1-n_j)\Bigr)-\sum_{j=1}^{\kappa}n_j\log\delta_j,
\]
giving stated closed form.
Finally, strict convexity excludes linear segments of \(\Phi_{\mathrm{F}}\), hence \(\Sigma\) cannot have corresponding flat segments; therefore no phase coexistence induced by \(s\)-continuous singularity. \qed
\end{proof}
