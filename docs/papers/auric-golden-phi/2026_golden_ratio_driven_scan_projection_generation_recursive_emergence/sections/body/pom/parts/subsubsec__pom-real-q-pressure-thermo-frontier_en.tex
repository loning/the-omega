\paragraph{From integer moments to real-parameter pressure}
For any real number \(q\ge 0\) and any resolution \(m\), define the power-sum moment

S_q(m):=\sum_{x\in X_m} d_m(x)^q,

\paragraph{MOM characteristic exchange-symmetric completeness: collision moment generating all learnable symmetric observables}
Fix resolution \(m\) and write \(n:=|X_m|\).
For any distribution \(w\in\Delta(X_m)\), define power-sum features
\[
p_q(w):=\sum_{x\in X_m}w(x)^q,\qquad q=2,3,\dots,n.
\]
When \(w=w_m\) is the uniform microstate pushforward \(w_m(x)=d_m(x)/2^m\), there is strict identity
\[
p_q(w_m)=2^{-mq}S_q(m).
\]
Let \(\mathfrak{S}_{n}\) act on \(\Delta(X_m)\) by permuting coordinates (i.e., type labels),
and let \(\mathcal{C}_{\mathrm{sym}}(\Delta(X_m))\) denote continuous function space invariant under this action (exchange-symmetric observables).
Write \(\mathcal{A}_m\) for closure under uniform norm \(\|\cdot\|_\infty\) of polynomial algebra generated by \(\{p_q\}_{q=2}^{n}\).

\begin{theorem}[Unified approximation of symmetric observables: Stone--Weierstrass completeness of \texorpdfstring{$\{p_q\}$}{pq}]\label{thm:pom-mom-symmetric-observables-stone-weierstrass}
\(\mathcal{A}_m\) is dense in \(\mathcal{C}_{\mathrm{sym}}(\Delta(X_m))\).
Equivalently: for any exchange-symmetric continuous statistic \(F\in \mathcal{C}_{\mathrm{sym}}(\Delta(X_m))\) and any \(\varepsilon>0\), there exists a polynomial \(P\) such that
\[
\sup_{w\in\Delta(X_m)}\left|F(w)-P\bigl(p_2(w),\dots,p_{n}(w)\bigr)\right|<\varepsilon.
\]
\end{theorem}
\begin{proof}
Consider compact Hausdorff quotient space \(K:=\Delta(X_m)/\mathfrak{S}_{n}\).
Symmetric continuous function space \(\mathcal{C}_{\mathrm{sym}}(\Delta(X_m))\) is naturally isomorphic to \(C(K)\).
Power-sum family \(\{p_2,\dots,p_n\}\) separates points in \(K\): if two orbits differ, their coordinate multisets differ;
by Newton identities, \(\{p_1,\dots,p_n\}\) determines elementary symmetric polynomials, hence the multiset; and \(p_1\equiv 1\) is fixed on simplex, so some \(2\le q\le n\) must separate values.
Therefore \(\mathcal{A}_m\) corresponds to commutative subalgebra in \(C(K)\) containing constants and separating points.
By Stone--Weierstrass theorem, \(\mathcal{A}_m\) is dense in \(\mathcal{C}_{\mathrm{sym}}(\Delta(X_m))\). \qed
\end{proof}

\begin{theorem}[Collision moment coordinate chart: Vandermonde--Jacobian criterion]\label{thm:pom-power-sum-vandermonde-jacobian}
Let \(\Delta^{n-1}_{\circ}:=\{(w_1,\dots,w_n)\in\RR^n:\ w_i>0,\ \sum_i w_i=1\}\) be interior points of simplex, and define power-sum map
\[
\Phi(w):=\bigl(p_2(w),p_3(w),\dots,p_n(w)\bigr)\in\RR^{n-1},
\qquad
p_k(w)=\sum_{i=1}^n w_i^k.
\]
On open dense set
\[
U:=\{w\in\Delta^{n-1}_{\circ}:\ w_i\ \text{pairwise distinct}\}
\]
\(\Phi\) gives local coordinates in exchange-symmetric sense: its Jacobian has full rank \(n-1\) on tangent space \(\{\delta w:\sum_i\delta w_i=0\}\).
More precisely, extended map
\(
\widetilde\Phi(w):=(p_1(w),p_2(w),\dots,p_n(w))
\)
has Jacobian determinant satisfying
\[
\det\Bigl[\frac{\partial p_k}{\partial w_i}\Bigr]_{k,i=1}^{n}
=
n!\prod_{1\le i<j\le n}(w_j-w_i),
\]
hence nonzero on \(U\).
\end{theorem}
\begin{proof}
One has \(\frac{\partial p_k}{\partial w_i}=k\,w_i^{k-1}\).
Hence Jacobian matrix can be written as
\[
J=\bigl[k\,w_i^{k-1}\bigr]_{k,i=1}^{n}
=\mathrm{diag}(1,2,\dots,n)\cdot V,
\qquad
V_{k,i}:=w_i^{k-1},
\]
where \(V\) is Vandermonde matrix.
Therefore
\(
\det J=(1\cdot2\cdots n)\det V
\)
and by Vandermonde determinant formula
\(
\det J=n!\prod_{i<j}(w_j-w_i)
\).
On \(U\), right-hand side is nonzero, so \(\widetilde\Phi\) is local diffeomorphism.
Since \(p_1\equiv 1\) is fixed on simplex, dropping first coordinate gives \(\Phi\) full-rank on tangent space; hence inducing local coordinate chart on \(U/\mathfrak{S}_n\). \qed
\end{proof}

\paragraph{Escort distributions and finite-window thermodynamic derivatives}
For any \(m\) and any \(q\ge 0\), define the \(q\)-escort distribution
\[
w_{m,q}(x):=\frac{d_m(x)^q}{S_q(m)}\qquad(x\in X_m).
\]
When \(q=0\), \(w_{m,0}=U_m:=\mathrm{Unif}(X_m)\) is the type-uniform measure;
when \(q=1\), \(w_{m,1}(x)=d_m(x)/2^m=w_m(x)\) is the uniform microstate pushforward (Definition \ref{def:pom-wm-baseline}).

\begin{proposition}[Escort derivative formulas: finite-window second-order jet of the log-partition function]\label{prop:pom-escort-derivative-identities}
Fix a resolution \(m\ge 1\) and let
\[
F_m(q):=\log S_q(m)\qquad(q\ge 0).
\]
Then \(F_m\) is convex and everywhere twice differentiable on \(q\ge 0\), satisfying the exact identities
\[
\boxed{\;
F_m'(q)=\mathbb{E}_{w_{m,q}}[\log d_m(X)],\qquad
F_m''(q)=\mathrm{Var}_{w_{m,q}}\!\bigl(\log d_m(X)\bigr)\ \ge\ 0\; }.
\]
\end{proposition}
\begin{proof}
For each fixed \(m\), the function \(q\mapsto S_q(m)=\sum_x e^{q\log d_m(x)}\) is a sum of finitely many exponential functions, so \(F_m=\log S_q(m)\) is a smooth convex function.
Direct differentiation yields
\[
F_m'(q)=\frac{\sum_x (\log d_m(x))\,d_m(x)^q}{\sum_x d_m(x)^q}
=\sum_x w_{m,q}(x)\log d_m(x),
\]
and differentiating once more gives
\(
F_m''(q)=\mathbb{E}_{w_{m,q}}[(\log d_m)^2]-\mathbb{E}_{w_{m,q}}[\log d_m]^2
\).
\qed
\end{proof}

\begin{corollary}[Zero-temperature limit quantitative law: super-exponential concentration of escort under \(q\sim m\) scaling]\label{cor:pom-escort-zero-temperature-superexp-concentration}
Fix resolution \(m\ge 1\).
Let maximal fiber
\(
M_m:=\max_{x\in X_m} d_m(x)
\), and write
\[
\alpha_m:=\frac1m\log M_m,\qquad
A_m(\varepsilon):=\left\{x\in X_m:\ \frac1m\log d_m(x)\le \alpha_m-\varepsilon\right\}\qquad(\varepsilon>0).
\]
Then for any integer \(q\ge 1\),
\[
\boxed{\ w_{m,q}\bigl(A_m(\varepsilon)\bigr)\ \le\ |X_m|\,e^{-q\varepsilon m}\ }.
\]
Hence when \(q\) grows linearly with \(m\) (e.g., \(q=\tau m\)),
\[
w_{m,q}\bigl(A_m(\varepsilon)\bigr)\ \le\ |X_m|\,\exp(-\tau\varepsilon m^2),
\]
exhibiting super-exponential contraction at \(m^2\) level.
\end{corollary}
\begin{proof}
For any \(x\in A_m(\varepsilon)\), one has \(d_m(x)^q\le M_m^q e^{-q\varepsilon m}\).
Therefore
\[
\sum_{x\in A_m(\varepsilon)} d_m(x)^q \le |A_m(\varepsilon)|\,M_m^q e^{-q\varepsilon m}\le |X_m|\,M_m^q e^{-q\varepsilon m}.
\]
Since \(S_q(m)=\sum_x d_m(x)^q\ge M_m^q\),
\[
w_{m,q}(A_m(\varepsilon))
=\frac{\sum_{x\in A_m(\varepsilon)} d_m(x)^q}{S_q(m)}
\le |X_m|\,e^{-q\varepsilon m}.
\]
\qed
\end{proof}

\begin{theorem}[Gibbs variational principle: the Pareto frontier of \texorpdfstring{$\kappa_m$}{kappa} is rigorously characterized by the escort family]\label{thm:pom-kappa-gibbs-variational-escort-frontier}
Fix a resolution \(m\ge 1\). For any distribution \(\pi\in\Delta(X_m)\), let
\(
\kappa_m(\pi):=\mathbb{E}_{\pi}[\log d_m(X)]
\)
(Definition \ref{def:pom-kappa}).
For any \(q\ge 0\), let \(F_m(q):=\log S_q(m)\).
Then the following rigorous variational characterization holds:
\[
\boxed{\;
F_m(q)=\sup_{\pi\in\Delta(X_m)}\Bigl(q\,\kappa_m(\pi)+H(\pi)\Bigr)\; }.
\]
Moreover, the optimizer is unique and equals the \(q\)-escort distribution \(w_{m,q}(x)=d_m(x)^q/S_q(m)\).
Therefore the mapping
\(
q\mapsto \bigl(H(w_{m,q}),\kappa_m(w_{m,q})\bigr)
\)
parametrizes the upper support boundary of the set \(\{(H(\pi),\kappa_m(\pi)):\pi\in\Delta(X_m)\}\): the optimal value of \(\kappa_m\) under a given entropy budget is rigorously attained by some escort.
\end{theorem}
\begin{proof}
For any \(q\ge 0\) and any \(\pi\in\Delta(X_m)\), expanding the KL divergence with respect to \(w_{m,q}(x)=d_m(x)^q/S_q(m)\):
\[
\begin{aligned}
D_{\mathrm{KL}}(\pi\|w_{m,q})
&=\sum_x \pi(x)\log\frac{\pi(x)}{w_{m,q}(x)}
=\sum_x \pi(x)\log\pi(x)-q\sum_x \pi(x)\log d_m(x)+\log S_q(m)\\
&=-H(\pi)-q\,\kappa_m(\pi)+F_m(q).
\end{aligned}
\]
Rearranging yields
\(
F_m(q)=q\,\kappa_m(\pi)+H(\pi)+D_{\mathrm{KL}}(\pi\|w_{m,q})
\ge q\,\kappa_m(\pi)+H(\pi)
\).
Taking the supremum gives the variational formula; uniqueness follows from \(D_{\mathrm{KL}}(\pi\|w_{m,q})=0\Leftrightarrow \pi=w_{m,q}\). \qed
\end{proof}

(i.e., \(\tau\) agrees with the real-parameter partition function at exponential scale).

  \item \textbf{(Counting measure \(U_m\))}

  \boxed{\;

  \item \textbf{(Pushforward measure \(w_m\))}

  \[
  \Lambda_w(q):=\lim_{m\to\infty}\frac{1}{m}\log \mathbb{E}_{w_m}[e^{qmY_m}]

  \]

  \(
  I_w(\alpha):=\sup_{q\ge 0}\bigl(q\alpha-\Lambda_w(q)\bigr)
  \).

\end{enumerate}
\end{theorem}

\paragraph{Oracle inversion budget curve: success exponent, capacity exponent, and thermal phase cut}
Following notation \(X_m,d_m,w_m,Y_m\) and rate functions \(I_w,I_U\).
Consider the following constrained inversion problem: given stable type \(x\in X_m\), allowing additional use of \(B\) bits of index information to guess the true microstate within fiber \(\Fold_m^{-1}(x)\); under baseline calibration of ``conditional uniform within fiber,'' optimal success rate is completely determined by fiber multiplicity spectrum.

\begin{theorem}[Oracle success exponent folded variational formula: strict two-branch formula given purely by \texorpdfstring{$I_w$}{Iw}]\label{thm:pom-oracle-success-variational-laplace}
For resolution \(m\) and budget \(B\in\ZZ_{\ge 0}\), define capacity count
\[
\mathcal{C}^{\mathrm{orc}}_m(B):=\sum_{x\in X_m}\min\{d_m(x),2^B\},
\qquad
\mathrm{Succ}^{\mathrm{orc}}_m(B):=\frac{\mathcal{C}^{\mathrm{orc}}_m(B)}{2^m}.
\]
Let \(a\in[0,\log 2]\) and take \(B_m(a):=\lfloor a m/\log 2\rfloor\).
If \(Y_m\) satisfies full large deviation principle under \((X_m,w_m)\) with rate function \(I_w\) (Theorem \ref{thm:pom-ldp-uniform-and-wm-exact-spectrum}), then limit
\[
\Sigma_{\mathrm{orc}}(a):=\lim_{m\to\infty}\frac{1}{m}\log \mathrm{Succ}^{\mathrm{orc}}_m\bigl(B_m(a)\bigr)
\]
exists and satisfies strict equality
\[
\boxed{\;
\Sigma_{\mathrm{orc}}(a)=\sup_{\alpha\in[0,\log 2]}\bigl(h_a(\alpha)-I_w(\alpha)\bigr)\;},
\]
where \(h_a(\alpha)=0\) when \(\alpha\le a\), and \(h_a(\alpha)=a-\alpha\) when \(\alpha\ge a\).
Equivalently,
\[
\boxed{\;
\Sigma_{\mathrm{orc}}(a)=
\max\Bigl\{-\inf_{\alpha\le a}I_w(\alpha),\ a-\inf_{\alpha\ge a}\bigl(I_w(\alpha)+\alpha\bigr)\Bigr\}\; } .
\]
\end{theorem}
\begin{proof}
By \(w_m(x)=d_m(x)/2^m\), directly simplify to
\[
\mathrm{Succ}^{\mathrm{orc}}_m(B)
=\sum_{x\in X_m}w_m(x)\min\!\left(1,\frac{2^B}{d_m(x)}\right)
=\mathbb{E}_{w_m}\!\left[\exp\!\bigl(m\,h_{a_m}(Y_m)\bigr)\right],
\]
where \(a_m:=B\log 2/m\) and \(h_{a_m}\) as stated.
At \(B=B_m(a)\), \(a_m=a+O(1/m)\), while \(Y_m\in[0,\log 2]\) makes \(h_a\) bounded continuous.
Applying Laplace/Varadhan principle to \((Y_m)\) yields
\(
\Sigma_{\mathrm{orc}}(a)=\sup_\alpha(h_a(\alpha)-I_w(\alpha))
\).
Splitting supremum at \(\alpha\le a\) and \(\alpha\ge a\) yields two-branch form. \qed
\end{proof}

\begin{theorem}[Thermal phase cut identity: capacity exponent dominated by single layer \texorpdfstring{$\alpha=a$}{alpha=a} on \texorpdfstring{$(a_0,a_1)$}{(a0,a1)}]\label{thm:pom-oracle-capacity-thermal-cut}
Under setting of Theorem \ref{thm:pom-ldp-uniform-and-wm-exact-spectrum}, define capacity exponent limit
\[
\Gamma_{\mathrm{orc}}(a):=\lim_{m\to\infty}\frac{1}{m}\log \mathcal{C}^{\mathrm{orc}}_m\bigl(B_m(a)\bigr)
\qquad (a\in[0,\log 2]),
\]
and let counting spectrum \(f(\alpha):=\log\varphi-I_U(\alpha)\) (same as spectral equality in Theorem \ref{thm:pom-ldp-uniform-and-wm-exact-spectrum}).
Then variational characterization holds:
\[
\Gamma_{\mathrm{orc}}(a)=\sup_{\alpha\in[0,\log 2]}\bigl(f(\alpha)+\min(\alpha,a)\bigr).
\]
Further assume \(\tau\) is strictly convex and differentiable on \((0,1)\), and let
\[
a_0:=\tau'(0^+),\qquad a_1:=\tau'(1^-),
\]
called thermal phase interval.
Then for any \(a\in(a_0,a_1)\), the supremum above must be attained at fold point \(\alpha=a\), whence strict identity holds:
\[
\boxed{\;
\Gamma_{\mathrm{orc}}(a)=f(a)+a,
\qquad\text{equivalently}\qquad
f(a)=\Gamma_{\mathrm{orc}}(a)-a\; } .
\]
\end{theorem}
\begin{proof}
By definition
\(
\mathcal{C}^{\mathrm{orc}}_m(B)=\sum_{x\in X_m} \exp\!\bigl(m\min(Y_m(x),a_m)\bigr)
\)
(where \(a_m=B\log2/m\)), combining local counting equality of Theorem \ref{thm:pom-ldp-uniform-and-wm-exact-spectrum}, one obtains exponential-scale dominant-term formula
\(
\Gamma_{\mathrm{orc}}(a)=\sup_\alpha(f(\alpha)+\min(\alpha,a))
\).
\par
For \(a\) in thermal phase interval, strict convex differentiability implies unique \(q_*(a)\in(0,1)\) such that \(\tau'(q_*(a))=a\).
By Legendre duality \(f(\alpha)=\inf_{q\ge 0}(\tau(q)-q\alpha)\) (equivalent to Theorem \ref{thm:pom-ldp-uniform-and-wm-exact-spectrum}), \(f\) is differentiable on \((a_0,a_1)\) and
\(f'(a)=-q_*(a)\in(-1,0)\).
Hence function \(\alpha\mapsto f(\alpha)+\min(\alpha,a)\) has left-hand derivative \(f'(\alpha)+1\) at \(\alpha=a\) (which is \(f'(a)+1>0\)), and right-hand derivative \(f'(\alpha)\) at \(\alpha=a\) (which is \(f'(a)<0\)).
Combining strict concavity of \(f\), global optimum must occur at fold point \(\alpha=a\), hence \(\Gamma_{\mathrm{orc}}(a)=f(a)+a\). \qed
\end{proof}

\begin{corollary}[Slope identification and conjugate parameter readout: \texorpdfstring{$q_*(a)=1-\Gamma'_{\mathrm{orc}}(a)$}{q*=1-Gamma'}]\label{cor:pom-oracle-capacity-slope-readout}
In thermal phase interval \((a_0,a_1)\) of Theorem \ref{thm:pom-oracle-capacity-thermal-cut},
let \(q_*(a)\in(0,1)\) be conjugate parameter (satisfying \(\tau'(q_*(a))=a\)).
Then \(\Gamma_{\mathrm{orc}}\) and \(f\) are differentiable on this interval, and
\[
\boxed{\;
q_*(a)=1-\Gamma_{\mathrm{orc}}'(a),
\qquad
f'(a)=-q_*(a)\in(-1,0)\;}
\]
and \(f\) is strictly concave on \((a_0,a_1)\).
\end{corollary}
\begin{proof}
By identity \(\Gamma_{\mathrm{orc}}(a)=f(a)+a\) of Theorem \ref{thm:pom-oracle-capacity-thermal-cut}, directly obtain
\(
\Gamma_{\mathrm{orc}}'(a)=f'(a)+1
\).
And \(f'(a)=-q_*(a)\) by stationary-point condition of Legendre duality, whence \(q_*(a)=1-\Gamma_{\mathrm{orc}}'(a)\).
Strict concavity transmitted from strict convexity of \(\tau\). \qed
\end{proof}

\begin{theorem}[Microcanonical conditioning and escort information equivalence: relative entropy rate zero on level sets]\label{thm:pom-microcanonical-escort-kl-rate-zero}
Take \(a\in(a_0,a_1)\) and let \(q=q_*(a)\in(0,1)\) satisfy \(\tau'(q)=a\).
For \(\varepsilon>0\), define level set
\[
E_m(a,\varepsilon):=\{x\in X_m:\ |Y_m(x)-a|<\varepsilon\},
\]
and let
\(
U_m^{a,\varepsilon}:=U_m(\cdot\mid E_m(a,\varepsilon))
\)
and
\(
w_{m,q}^{a,\varepsilon}:=w_{m,q}(\cdot\mid E_m(a,\varepsilon))
\).
Under spectral gap/analytic perturbation regime of Theorem \ref{thm:pom-real-q-pressure-analytic-interpolation} (hence local large deviation and local limit theorem hold),
\[
\lim_{\varepsilon\downarrow 0}\ \lim_{m\to\infty}\frac{1}{m}D_{\mathrm{KL}}\!\bigl(U_m^{a,\varepsilon}\ \|\ w_{m,q}^{a,\varepsilon}\bigr)=0,
\qquad
\lim_{\varepsilon\downarrow 0}\ \lim_{m\to\infty}\frac{1}{m}D_{\mathrm{KL}}\!\bigl(w_{m,q}^{a,\varepsilon}\ \|\ U_m^{a,\varepsilon}\bigr)=0.
\]
Moreover, since \(w_{m,q}(E_m(a,\varepsilon))\to 1\) (fixing \(\varepsilon\)), one can deduce
\[
\lim_{\varepsilon\downarrow 0}\ \lim_{m\to\infty}\frac{1}{m}D_{\mathrm{KL}}\!\bigl(U_m^{a,\varepsilon}\ \|\ w_{m,q}\bigr)=0.
\]
\end{theorem}
\begin{proof}
For any \(x\in E_m(a,\varepsilon)\),
\(
\log w_{m,q}(x)=q\log d_m(x)-\log S_q(m)
= m(q a-\tau(q))+O(\varepsilon m)+o(m),
\)
where \(o(m)\) comes from \(\log S_q(m)=m\tau(q)+o(m)\).
Therefore log-density amplitude of \(w_{m,q}\) on \(E_m(a,\varepsilon)\) is \(O(\varepsilon m)+o(m)\), while \(U_m^{a,\varepsilon}\) is strictly constant-density on \(E_m\); hence Radon--Nikodym ratio between them satisfies
\[
\sup_{x\in E_m(a,\varepsilon)}\left|\log\frac{U_m^{a,\varepsilon}(x)}{w_{m,q}^{a,\varepsilon}(x)}\right|
=O(\varepsilon m)+o(m).
\]
Substituting into KL definition yields entropy-rate bound \(O(\varepsilon)+o(1)\); first let \(m\to\infty\) then \(\varepsilon\downarrow 0\) to obtain conclusion.
Last formula follows from
\(
D_{\mathrm{KL}}(U_m^{a,\varepsilon}\|w_{m,q})
=D_{\mathrm{KL}}(U_m^{a,\varepsilon}\|w_{m,q}^{a,\varepsilon})-\log w_{m,q}(E_m(a,\varepsilon))
\)
and \(w_{m,q}(E_m(a,\varepsilon))\to 1\). \qed
\end{proof}

\begin{theorem}[Gaussian limit at critical budget window: \texorpdfstring{$B=\beta_{\mathrm{crit}}m+O(\sqrt m)$}{B=beta m+O(sqrt m)}]\label{thm:pom-oracle-critical-window-gaussian}
Let \(w_m=w_{m,1}\).
Assume \(\tau\) is twice differentiable in left neighborhood of \(q=1\) with \(\tau''(1^-)>0\), and write
\[
a_1:=\tau'(1^-),\qquad \sigma^2:=\tau''(1^-),\qquad \beta_{\mathrm{crit}}:=\frac{a_1}{\log 2}.
\]
For any fixed \(t\in\RR\), let
\[
B_m(t):=\left\lfloor \beta_{\mathrm{crit}}m+\frac{t}{\log 2}\sqrt m\right\rfloor.
\]
Then oracle optimal success rate satisfies limit
\[
\boxed{\;
\lim_{m\to\infty}\mathrm{Succ}^{\mathrm{orc}}_m\bigl(B_m(t)\bigr)
=\Phi\!\left(\frac{t}{\sigma}\right)\;},
\]
where \(\Phi\) is standard normal distribution function.
\end{theorem}
\begin{proof}
Let \(a_m:=B_m(t)\log 2/m=a_1+t/\sqrt m+O(1/m)\).
By identity of Theorem \ref{thm:pom-oracle-success-variational-laplace},
\[
\mathrm{Succ}^{\mathrm{orc}}_m\bigl(B_m(t)\bigr)
=\mathbb{E}_{w_m}\!\left[\min\bigl(1,e^{m(a_m-Y_m)}\bigr)\right]
=\mathbb{P}_{w_m}(Y_m\le a_m)+R_m,
\]
where
\(
R_m:=\mathbb{E}_{w_m}[e^{m(a_m-Y_m)}\mathbf{1}_{\{Y_m>a_m\}}]
\).
For any \(\eta>0\),
\[
R_m\le \mathbb{P}_{w_m}\!\left(a_m<Y_m\le a_m+\frac{\eta}{\sqrt m}\right)+e^{-\eta\sqrt m}.
\]
Under spectral gap regime, central limit theorem holds under \(q=1\) escort (see isomorphic conclusion in Theorem \ref{thm:pom-fiber-index-clt-thermo}), so
\(
\sqrt m\,(Y_m-a_1)\Rightarrow \mathcal{N}(0,\sigma^2)
\).
Hence
\(
\mathbb{P}_{w_m}(Y_m\le a_m)\to \Phi(t/\sigma)
\)
and
\(
\limsup_{m\to\infty}\mathbb{P}_{w_m}\!\left(a_m<Y_m\le a_m+\frac{\eta}{\sqrt m}\right)
=\Phi\!\left(\frac{t+\eta}{\sigma}\right)-\Phi\!\left(\frac{t}{\sigma}\right)
\).
Let \(\eta\downarrow 0\) to obtain \(R_m\to 0\), whence conclusion holds. \qed
\end{proof}

\begin{corollary}[Second-order inversion: minimal bit budget required to achieve given success rate]\label{cor:pom-oracle-critical-window-bit-inversion}
Under setting of Theorem \ref{thm:pom-oracle-critical-window-gaussian}, for any fixed \(p\in(0,1)\), let \(B_m(p)\) be minimal budget satisfying
\(
\mathrm{Succ}^{\mathrm{orc}}_m(B_m(p))=p+o(1)
\); then
\[
\boxed{\;
B_m(p)=\beta_{\mathrm{crit}}m+\frac{\sigma}{\log 2}\,\Phi^{-1}(p)\sqrt m+o(\sqrt m)\;}
\qquad(m\to\infty).
\]
\end{corollary}
\begin{proof}
By monotonicity of \(B\mapsto \mathrm{Succ}^{\mathrm{orc}}_m(B)\) and Theorem \ref{thm:pom-oracle-critical-window-gaussian},
in window \(B=\beta_{\mathrm{crit}}m+t\sqrt m/\log2\) limiting success rate is \(\Phi(t/\sigma)\).
Take \(t=\sigma\Phi^{-1}(p)\) and convert back to bit scale. \qed
\end{proof}
