\begin{theorem}[Intrinsic Self-Calibration of the Golden Ratio: Recovering \texorpdfstring{$\varphi$}{phi} from the Last-Bit Conditional Mean Ratio]\label{thm:xi-foldbin-intrinsic-self-calibration-phi-lastbit-means}
Under the bin-fold two-state uniform asymptotics (Theorem \ref{thm:fold-bin-two-state-asymptotic}),
let \(d_m^{\mathrm{bin}}(x):=\bigl|(\mathrm{Fold}^{\mathrm{bin}}_m)^{-1}(x)\bigr|\) (\(x\in X_m\)), and partition by the last bit:
\[
X_m^{(0)}:=\{x\in X_m:\ x_m=0\},
\qquad
X_m^{(1)}:=\{x\in X_m:\ x_m=1\}.
\]
Define the observable statistic
\[
\widehat\varphi_m
\ :=\
\frac{\frac{1}{|X_m^{(0)}|}\sum_{x\in X_m^{(0)}} d_m^{\mathrm{bin}}(x)}{\frac{1}{|X_m^{(1)}|}\sum_{x\in X_m^{(1)}} d_m^{\mathrm{bin}}(x)}.
\]
Then there exists a constant \(C>0\) independent of \(m\) such that
\[
\bigl|\widehat\varphi_m-\varphi\bigr|\ \le\ C\left(\frac{\varphi}{2}\right)^m,
\]
so that \(\widehat\varphi_m\) converges to \(\varphi\) at an exponential rate.
This yields a self-calibration mechanism requiring no external reference: the growth constant can be recovered solely from fold fiber multiplicities and the last visible bit.
\end{theorem}

\begin{proof}
By Theorem \ref{thm:fold-bin-two-state-asymptotic}, there exists a constant \(C_0>0\) such that for sufficiently large \(m\) and any \(x\in X_m\), the uniform estimate holds:
\[
\Bigl|d_m^{\mathrm{bin}}(x)-\varphi^{-x_m}\Bigl(\frac{2}{\varphi}\Bigr)^m\Bigr|\le C_0.
\]
Taking the mean over \(X_m^{(0)}\) and \(X_m^{(1)}\) respectively gives
\[
\frac{1}{|X_m^{(0)}|}\sum_{x\in X_m^{(0)}} d_m^{\mathrm{bin}}(x)
=\Bigl(\frac{2}{\varphi}\Bigr)^m+O(1),
\qquad
\frac{1}{|X_m^{(1)}|}\sum_{x\in X_m^{(1)}} d_m^{\mathrm{bin}}(x)
=\varphi^{-1}\Bigl(\frac{2}{\varphi}\Bigr)^m+O(1),
\]
where the \(O(1)\) constants are independent of \(m\).
Dividing the two expressions and using \((2/\varphi)^{-m}=(\varphi/2)^m\) yields
\[
\widehat\varphi_m=\varphi\cdot\frac{1+O((\varphi/2)^m)}{1+O((\varphi/2)^m)}
=\varphi+O\!\left(\left(\frac{\varphi}{2}\right)^m\right),
\]
hence there exists a constant \(C\) satisfying the stated error bound.
\end{proof}

\begin{proof}
The given discrete identity immediately yields \(N_f=|X_m^{\mathrm{bdry}}|\).

If \(N_f=3\), then \(F_{m-2}=3\) forces \(m-2=4\), i.e., \(m=6\), and this solution is unique.
\end{proof}

\begin{theorem}[Spectral Number Field Rigidity of the Window-\(6\) Four-Block Coarse-Graining Kernel: Irreducible Cubic Field and \texorpdfstring{$S_3$}{S3} Galois Group]\label{thm:xi-window6-four-block-kernel-cubic-field-s3}
Let \(T\) be the coarse-graining Markov kernel of the four-block skeleton as in \eqref{eq:window6_edge_flux_skeleton},
and write its characteristic polynomial factorization as
\[
\chi_T(t)=(t-1)\cdot p(t),
\qquad
p(t)=\frac{48114t^3+7263t^2-506t-55}{48114}.
\]
Then:
\begin{enumerate}
  \item The cubic factor \(p(t)\) is irreducible over \(\mathbb Q[t]\);
  \item Its discriminant \(\Delta=\mathrm{Disc}(48114t^3+7263t^2-506t-55)\) is not a perfect square (see \eqref{eq:window6_edge_flux_skeleton_arithmetic});
  \item Consequently, the Galois group of the splitting field is \(S_3\), and all nontrivial eigenvalues of \(T\) are real algebraic numbers of degree \(3\).
\end{enumerate}
Therefore the nontrivial spectrum generates a non-abelian cubic number field \(K_T=\mathbb Q(\lambda)\), and the minimal algebraic complexity of "coarse-graining mixing" in the number field sense is cubic and irreducible.
\end{theorem}

\begin{proof}
For a cubic polynomial, irreducibility over \(\mathbb Q[t]\) is equivalent to having no rational roots.
Reducing the integer-coefficient polynomial
\(
f(t):=48114t^3+7263t^2-506t-55
\)
modulo \(13\) gives
\(
f(t)\equiv t^3+9t^2+t+10\pmod{13}
\).
Direct verification over all \(t\in\mathbb F_{13}\) shows that this cubic polynomial has no roots in \(\mathbb F_{13}\), hence is irreducible over \(\mathbb F_{13}[t]\), and therefore \(f\) is irreducible over \(\mathbb Q[t]\), as is \(p\).
\par
By \eqref{eq:window6_edge_flux_skeleton_arithmetic}, the discriminant \(\Delta\) is not a perfect square.
For an irreducible cubic polynomial, the Galois group is \(A_3\) if and only if the discriminant is a perfect square; otherwise it is \(S_3\).
Therefore the Galois group in this case is \(S_3\).
Since \(\Delta>0\), the three roots are distinct and all real, so all three nontrivial eigenvalues of \(T\) are real algebraic numbers of degree \(3\).
\end{proof}
