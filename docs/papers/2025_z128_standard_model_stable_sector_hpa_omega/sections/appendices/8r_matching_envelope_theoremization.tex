\subsection{Finite scheme/threshold envelopes for matching (T1S-4)}
\label{app:matching_envelope_theoremization}

\AuditTag This appendix upgrades matching/scheme dependence from an informal caveat into a theorem-facing envelope statement: all renormalization-scheme and threshold ambiguities used by the manuscript are restricted to an explicit \emph{finite family}, and the resulting uncertainty is budgeted as a supremum bound.

\subsubsection{Finite family of conventions}
\paragraph{Scheme/threshold family.}
\MatchTag Fix a finite family $\mathfrak{S}$ of renormalization and threshold conventions (schemes, decoupling prescriptions, window choices) declared by the manuscript.
Concrete registry examples are already recorded and swept in the bounded-family audit appendices (e.g.\ Appendices~\ref{app:scheme_invariance_audit_contract} and \ref{app:coupling_unification_audit_in_r}).

\paragraph{Envelope principle.}
Given any observable $\mathcal{O}$ and a predicted value $\langle\mathcal{O}\rangle_{\mathrm{EFT},s}$ computed under convention $s\in\mathfrak{S}$, define the matching envelope width by
\[
\Delta_{\mathrm{match}}(\mathcal{O})
:=\sup_{s,s'\in\mathfrak{S}}\abs{\langle\mathcal{O}\rangle_{\mathrm{EFT},s}-\langle\mathcal{O}\rangle_{\mathrm{EFT},s'}}.
\]

\subsubsection{Theorem-facing bound}
\begin{proposition}[Matching error as a finite-family envelope]
\label{prop:matching_error_envelope}
\MathTag For any chosen reference convention $s_0\in\mathfrak{S}$, the matching ambiguity obeys
\[
\abs{\langle\mathcal{O}\rangle_{\mathrm{EFT},s_0}-\langle\mathcal{O}\rangle_{\mathrm{EFT},s}}
\le \Delta_{\mathrm{match}}(\mathcal{O}),
\qquad \forall\,s\in\mathfrak{S}.
\]
\end{proposition}
\begin{proof}
Immediate from the definition of $\Delta_{\mathrm{match}}(\mathcal{O})$ as a supremum over pairs.
\end{proof}

\paragraph{Integration into $\epsilon_N(m,n)$.}
\InterfaceTag In the total error decomposition of Appendix~\ref{app:eft_error_bounds}, one may choose
\[
E_{\mathrm{match}}(N):=\sup_{\mathcal{O}\in\mathfrak{Obs}_{\le N}}\Delta_{\mathrm{match}}(\mathcal{O}),
\]
thereby turning scheme/threshold dependence into a bounded, auditable term once $\mathfrak{S}$ is explicitly listed and swept.
