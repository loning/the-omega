\section{记号表}

\begin{itemize}
  \item $m$：微观比特长度。
  \item $\Xm=\{0,1\}^m$：微观态集合。
  \item $\Ym=\{y\in\{0,1\}^m:\ y_i y_{i+1}=0\}$：宏观稳定态（黄金均值语言）。
  \item $N(x)=\sum_{i=0}^{m-1} x_i2^i$：微观态对应的整数解释。
  \item $\mathbf c(N)=(c_1(N),c_2(N),\dots)$：$N$ 的 Zeckendorf 数码（$c_kc_{k+1}=0$）。
  \item $\Fold:\Xm\to \Ym$：Zeckendorf 截断折叠（定义 \ref{def:fold_zeckendorf}）。
  \item $K(m)$：覆盖指数（定义 \ref{def:K_of_m}）。
  \item $T_m$：长度为 $L(m)=K(m)-m$ 的尾部状态集合。
  \item $\tau_m:\Xm\to T_m$：尾部投影（定义 \ref{def:tail_projection}）。
  \item $\Tail:T_m\to T_m$：尾部移位时间算子（定义 \ref{def:tail_shift}）。
  \item $\mathsf S=(X,Z; f,R_{\mathrm{space}},R_{\mathrm{time}})$：时空结构对象（定义 \ref{def:spacetime_object}）。
  \item $\Clo(\mathsf S)$：编码闭包图（定义 \ref{def:encoding_graph}）。
\end{itemize}
