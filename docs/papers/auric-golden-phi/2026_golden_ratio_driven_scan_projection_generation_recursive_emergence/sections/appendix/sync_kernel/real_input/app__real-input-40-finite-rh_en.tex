\subsubsection{Finite RH: Critical Strip Poles of the s-variable Zeta and Square-Root Oscillations of Primitives}\label{app:real-input-40-finite-rh}
Substituting the $z$-variable with Dirichlet coordinates
\[
z=\lambda^{-s},\qquad \widehat{\zeta}(s):=\zeta_M(\lambda^{-s})=\frac{1}{\det(I-\lambda^{-s}M)}.
\]
The poles of $\widehat{\zeta}(s)$ are determined by the nonzero eigenvalues $\mu$ of $M$: a pole occurs if and only if
\[
\lambda^{-s}\mu=1
\]
(for each $\mu$ and its logarithmic branches, this yields a sequence of poles).

\paragraph{Spectrum--pole--error triangle: universal dictionary for finite-dimensional \texorpdfstring{$\zeta$}{zeta}}
For any finite-dimensional kernel (matrix) $M$, the Dirichlet-transformed $\widehat{\zeta}(s)$ pointwise translates "spectral modulus" into "critical strip pole position," and further translates "nontrivial spectral radius" into the error scale of primitive counting.
This dictionary is the closest finite-dimensional model to the \(\zeta\)/\(L\)-function language: it compresses the analytic number theory triangle of "zeros/poles--explicit formulas--error terms" into a purely linear algebraic triangle.

\begin{proposition}[Spectrum \texorpdfstring{$\leftrightarrow$}{<->} poles: critical strip coordinates and toy-RH criterion]\label{prop:finite-rh-triangle-dict}
Let $M$ be a finite-dimensional matrix, $\lambda=\rho(M)>1$, and define
\[
\widehat{\zeta}(s)=\zeta_M(\lambda^{-s})=\frac{1}{\det(I-\lambda^{-s}M)}.
\]
Then for any nonzero eigenvalue $\mu\neq 0$ and any $k\in\ZZ$, the poles of $\widehat{\zeta}(s)$ are given by
\[
\lambda^{-s}\mu=1,
\]
equivalently
\[
s=\frac{\log\mu+2\pi i k}{\log\lambda},
\]
where $\log\mu$ is taken with respect to any complex logarithm branch. Thus the real part of the pole satisfies
\[
\Re(s)=\frac{\log|\mu|}{\log\lambda}.
\]
Therefore a nontrivial pole appears in the open strip $0<\Re(s)<1$ if and only if there exists an eigenvalue satisfying $1<|\mu|<\lambda$.

Furthermore, if the "finite-dimensional toy-RH" is defined as: all nontrivial poles in the open strip $0<\Re(s)<1$ lie on $\Re(s)=\tfrac12$, then this condition is equivalent to
\[
(\forall\,\mu\ \text{satisfying}\ 1<|\mu|<\lambda)\qquad |\mu|=\sqrt{\lambda}.
\]
More weakly, if the Ramanujan radius bound
\[
\max_{\mu\neq\lambda}|\mu|\ \le\ \sqrt{\lambda}
\]
holds, then the right half of the open strip $\tfrac12<\Re(s)<1$ contains no poles; if there exists $|\mu|=\sqrt{\lambda}$, then the corresponding pole sequence lies exactly on $\Re(s)=\tfrac12$.
\end{proposition}

\begin{corollary}[Kernel RH \texorpdfstring{$\Longleftrightarrow$}{<->} no poles in the half-strip]\label{cor:finite-rh-halfstrip-nopoles}
Using the notation of Proposition \ref{prop:finite-rh-triangle-dict}, express the kernel RH (Ramanujan radius bound) as
\[
\max_{\mu\neq \lambda}|\mu|\le \sqrt{\lambda}.
\]
This condition is equivalent to: \(\widehat{\zeta}(s)\) has no poles in the open half-strip
\[
\tfrac12<\Re(s)<1.
\]
Moreover, this condition is saturated (there exists \(|\mu|=\sqrt{\lambda}\)) if and only if \(\widehat{\zeta}(s)\) has a nontrivial pole sequence on the critical line \(\Re(s)=\tfrac12\).
\end{corollary}
\begin{proof}
By the pole real-part identity \(\Re(s)=\log|\mu|/\log\lambda\) from Proposition \ref{prop:finite-rh-triangle-dict}: some \(\mu\neq 0\) has a pole in \(\tfrac12<\Re(s)<1\) if and only if \(\sqrt{\lambda}<|\mu|<\lambda\).
Since \(\lambda\) is the spectral radius, \(|\mu|\le \lambda\) holds for all eigenvalues, so "no poles in the half-strip" is equivalent to \(|\mu|\le \sqrt{\lambda}\) for all \(\mu\neq\lambda\), i.e., the kernel RH.
The final statement follows similarly from \(|\mu|=\sqrt{\lambda}\iff \Re(s)=\tfrac12\). \qed
\end{proof}

\begin{remark}[Saturation position for this kernel: unique critical modulus \texorpdfstring{$|\mu|=\sqrt{\lambda}$}{|mu|=sqrt(lambda)}]\label{rem:finite-rh-40-ramanujan}
For the real-input 40-state kernel, aside from the Perron root $\lambda=\varphi^2$, the only eigenvalue satisfying $|\mu|=\sqrt{\lambda}$ is $-\sqrt{\lambda}=-\varphi$ (consistent with the spectral decomposition in Proposition \ref{prop:real-input-40-essential-20} and Proposition \ref{prop:finite-rh-40} herein).
This kernel thus achieves the "single-frequency critical oscillation" scenario of the toy-RH: all nontrivial poles in the critical strip are produced solely by $-\sqrt{\lambda}$.
If one sets \(y:=\mu/\sqrt{\lambda}\) and takes \(S=y+y^{-1}\), then this saturation point corresponds to \(y=-1\) and \(S=-2\), consistent with the completion-compressed coordinate of \S\ref{subsec:xi-from-self-dual} (see Remark \ref{rem:xi-joukowski-toy-rh-alignment}).
\end{remark}

\begin{remark}[Origin of the critical line lattice: a hidden \texorpdfstring{$(-A_0)$}{(-A0)} factor]\label{rem:finite-rh-40-hidden-minusA0}
Proposition \ref{prop:real-input-40-A0-tensor-hidden} gives a finer three-factor decomposition
\[
\det(I-zM)
=(1-z)^2(1+z)\,
\det\!\bigl(I-z(A_0\otimes A_0)\bigr)\,
\det\!\bigl(I-z(-A_0)\bigr).
\]
Here \(\det(I-z(-A_0))=\det(I+zA_0)=1+z-z^2\), whose roots are
\(
z=\frac{1}{\varphi},\ -\varphi
\),
thereby forcing the eigenvalue \(-\varphi=-\sqrt{\lambda}\) to appear in the nonzero spectrum of \(M\).
Thus the "unique saturation" of the toy-RH is not an accidental resonance of the \(40\)-state complex structure, but is necessarily produced by a two-dimensional golden mean factor \((-A_0)\) embedded within the kernel.
Moreover, the constant-fiber subspace \(U\) and perpendicular subspace \(U^\perp\) induced by the block structure give rise to a "skeleton old spectrum / longitudinal residual new spectrum" two-channel decomposition (see Proposition \ref{prop:real-input-40-skeleton-factor} and Remark \ref{rem:real-input-40-deterministic-lift}). Consequently, the square-root bound of the toy-(G)RH can be understood as: constraining all nontrivial spectrum to the longitudinal residual channel and imposing the corresponding spectral radius bound.
\end{remark}

\begin{remark}[Zero--oscillation dictionary: critical circle spectrum \texorpdfstring{$\Rightarrow$}{=>} critical line comb teeth and multi-frequency square-root terms]\label{rem:finite-rh-zero-oscillation-dict}
Proposition \ref{prop:finite-rh-triangle-dict} provides the rule that "modulus \(|\mu|\)" determines which vertical line the pole lies on; if one further writes out the \textbf{phase} on the critical circle, it directly determines the "frequency comb teeth" on the critical line, producing explicit oscillation terms in the periodic trace and primitive orbit count.
\par
Let \(\lambda=\rho(M)>1\), and assume no Jordan blocks exist on the critical circle \(|\mu|=\sqrt{\lambda}\) (e.g., the part at this modulus is semisimple). Denote all saturating eigenvalues on the critical circle as
\[
\mu_j=\sqrt{\lambda}\,e^{i\alpha_j}\qquad(j=1,\dots,J),
\]
counted with algebraic multiplicity; and let the remaining eigenvalues have modulus strictly less than \(\sqrt{\lambda}\), with
\[
\rho_{\mathrm{sub}}:=\max\{|\mu|:\ \mu\in\mathrm{spec}(M),\ |\mu|<\sqrt{\lambda}\}<\sqrt{\lambda}.
\]
Then:
\begin{enumerate}
  \item \textbf{(Critical line pole comb teeth)} For each \(j\) and each \(k\in\ZZ\), the equation \(\lambda^{-s}\mu_j=1\) yields a critical line pole sequence
  \[
  s=\frac12+\frac{i(\alpha_j+2\pi k)}{\log\lambda}.
  \]
  \item \textbf{(Multi-frequency square-root terms in periodic traces)} By spectral decomposition (or diagonalization),
  \[
  \Tr(M^n)=\lambda^n+\lambda^{n/2}\sum_{j=1}^{J}e^{in\alpha_j}+O\!\bigl(\rho_{\mathrm{sub}}^{\,n}\bigr).
  \]
  \item \textbf{(Oscillation terms for primitives)} By Witt/M\"obius inversion
  \(p_n=\frac{1}{n}\sum_{d\mid n}\mu(d)\Tr(M^{n/d})\),
  \[
  p_n=\frac{\lambda^n}{n}+\frac{\lambda^{n/2}}{n}\sum_{j=1}^{J}e^{in\alpha_j}
  +O\!\left(\frac{\rho_{\mathrm{sub}}^{\,n}}{n}\right),
  \]
  where details for even \(n\) (e.g., parity cancellation when \(\alpha_j=\pi\)) can be further refined by Proposition \ref{prop:finite-rh-parity-general}.
\end{enumerate}
For the real-input 40-state kernel, the unique saturating phase is \(\alpha=\pi\) (i.e., \(\mu=-\sqrt{\lambda}\)), so the critical line comb teeth degenerate to a single equally-spaced lattice, and the square-root term degenerates to a single-frequency oscillation \(({-}1)^n\lambda^{n/2}\) (Corollary \ref{cor:finite-rh-explicit-formula}).
\end{remark}

\paragraph{Phase proliferation: cyclic lift preserving RH-sharpness}
The difficult part of \(\zeta(s)\) is not merely "the real part lies on \(\tfrac12\)"---a single critical line---but rather that there exist numerous zeros on the critical line with complex imaginary-part structure.
In the finite-dimensional kernel model, one can \textbf{preserve the Ramanujan/RH-sharp radius bound} while proliferating the "single phase" critical pole sequence into an arbitrarily fine phase comb; this provides a minimal auditable mechanism for subsequently pushing the toy-RH toward a toy-GRH (via group extensions / Artin decomposition).

\begin{proposition}[Cyclic phase lift: spectral phase subdivision preserving the Ramanujan radius bound]\label{prop:finite-rh-phase-lift}
Let $M$ be a finite-dimensional matrix, $\lambda=\rho(M)>1$. Take an integer $q\ge 2$, let $P_q$ be the $q$-cyclic permutation matrix ($P_q e_j=e_{j+1}$, indices taken modulo $q$), and define the cyclic lift matrix
\[
M^{\langle q\rangle}:=M\otimes P_q.
\]
Then:
\begin{enumerate}
  \item \textbf{(Spectral phase subdivision)} Let $\omega_q=e^{2\pi i/q}$. The spectral identity holds:
  \[
  \mathrm{spec}(M^{\langle q\rangle})
  =\{\mu\,\omega_q^k:\ \mu\in \mathrm{spec}(M),\ k=0,1,\dots,q-1\},
  \]
  counted with algebraic multiplicity.
  \item \textbf{(Spectral radius preservation)} $\rho(M^{\langle q\rangle})=\rho(M)=\lambda$.
  \item \textbf{(Ramanujan bound preservation)} If $M$ satisfies $\max_{\mu\neq \lambda}|\mu|\le \sqrt{\lambda}$, then $M^{\langle q\rangle}$ satisfies the same radius bound.
  \item \textbf{(Critical line pole comb teeth)} Let
  \(
  \widehat{\zeta}^{\langle q\rangle}(s):=\det(I-\lambda^{-s}M^{\langle q\rangle})^{-1}
  \).
  If $\mu=\sqrt{\lambda}e^{i\theta}$ is a saturating eigenvalue of $M$, then it generates $q$ mutually shifted critical line pole sequences in $M^{\langle q\rangle}$:
  \[
  s=\frac12+\frac{i}{\log\lambda}\Bigl(\theta+2\pi\Bigl(\frac{k}{q}+m\Bigr)\Bigr),
  \qquad k\in\{0,\dots,q-1\},\ m\in\ZZ.
  \]
\end{enumerate}
As $q\to\infty$, the imaginary-part positions of the above poles within each fundamental period window $[0,2\pi/\log\lambda)$ become arbitrarily dense.
\end{proposition}

\begin{proof}
The permutation matrix $P_q$ can be diagonalized by the discrete Fourier transform, with eigenvalues $\omega_q^k$ ($0\le k\le q-1$).
For any eigenpair $(\mu,v)$ of $M$ and any eigenpair $(\omega_q^k,w_k)$ of $P_q$, the tensor vector $v\otimes w_k$ satisfies
\(
(M\otimes P_q)(v\otimes w_k)=(Mv)\otimes(P_qw_k)=\mu\,\omega_q^k (v\otimes w_k)
\), yielding the spectral phase subdivision formula; the reverse inclusion follows from dimension counting or Kronecker spectral properties.
The preservation of the spectral radius and Ramanujan bound follows from $|\omega_q^k|=1$.
The pole formula is obtained by applying Proposition \ref{prop:finite-rh-triangle-dict} to $M^{\langle q\rangle}$ for each eigenvalue and selecting those with saturating modulus $|\mu|=\sqrt{\lambda}$. \qed
\end{proof}

\begin{remark}[Application to the real-input $40$-state kernel: subdivision of the single-frequency \texorpdfstring{$\pi$}{pi} oscillation]\label{rem:finite-rh-40-phase-lift}
For the real-input $40$-state kernel, the unique saturating eigenvalue is $-\sqrt{\lambda}=-\varphi=\sqrt{\lambda}e^{i\pi}$ (Remark \ref{rem:finite-rh-40-ramanujan}).
The cyclic phase lift thus subdivides the original single critical sequence
\(
s=\tfrac12+\frac{(2m+1)\pi i}{\log\lambda}
\)
into
\[
s=\frac12+\frac{i}{\log\lambda}\Bigl(\pi+2\pi\Bigl(\frac{k}{q}+m\Bigr)\Bigr),
\qquad k=0,\dots,q-1,\ m\in\ZZ,
\]
upgrading the imaginary-part structure on the critical line from a "single fence" to a "$q$-comb" while preserving RH-sharpness.
\end{remark}

\begin{proposition}[Artin factorization of the cyclic lift (explicit decomposition into abelian character channels)]\label{prop:finite-rh-phase-lift-artin}
Using the notation of Proposition \ref{prop:finite-rh-phase-lift}, let
\(
\omega_q=e^{2\pi i/q}
\), and denote the characters of the cyclic group \(G=\ZZ/q\ZZ\) by \(\chi_k(j)=\omega_q^{kj}\) (\(k\in\{0,\dots,q-1\}\)).
Then for any complex parameter \(z\), the determinant factorizes as
\[
\det\!\bigl(I-zM^{\langle q\rangle}\bigr)
=\prod_{k=0}^{q-1}\det\!\bigl(I-z\,\omega_q^k M\bigr).
\]
Equivalently, the \(\zeta\)-function of the cyclic lift satisfies
\[
\zeta_{M^{\langle q\rangle}}(z)
=\prod_{k=0}^{q-1}\zeta_{M}(z\omega_q^k),
\qquad
\zeta_M(z):=\det(I-zM)^{-1}.
\]
Thus "phase proliferation" is not merely a rotational subdivision of the spectral point set, but an auditable \textbf{abelian Artin decomposition}: each character channel \(\chi_k\) corresponds to a "twisted factor" \(\det(I-z\,\omega_q^k M)\).
\end{proposition}

\begin{proof}
Take the discrete Fourier matrix \(F\) so that \(F^{-1}P_qF=\mathrm{diag}(\omega_q^0,\dots,\omega_q^{q-1})\).
Then
\[
I-z(M\otimes P_q)
\sim
I-z\bigl(M\otimes \mathrm{diag}(\omega_q^0,\dots,\omega_q^{q-1})\bigr)
\]
is similar to a block diagonal matrix indexed by \(k\), with the \(k\)-th block being \(I-z\,\omega_q^k M\). Taking determinants of the block diagonal matrix and multiplying yields the conclusion. \qed
\end{proof}

\begin{corollary}[\texorpdfstring{$q$}{q}-step time rescaling identity: cyclic lift as a pullback under \texorpdfstring{$z\mapsto z^q$}{z->z^q}]\label{cor:finite-rh-phase-lift-time-rescale}
Using the notation of Proposition \ref{prop:finite-rh-phase-lift}, let \(M^{\langle q\rangle}=M\otimes P_q\).
Then for any complex parameter \(z\),
\[
\det\!\bigl(I-zM^{\langle q\rangle}\bigr)=\det\!\bigl(I-z^q M^q\bigr).
\]
Equivalently,
\[
\zeta_{M^{\langle q\rangle}}(z)=\zeta_{M^q}(z^q).
\]
In Dirichlet coordinates \(z=\lambda^{-s}\), this can be rewritten as
\[
\widehat{\zeta}^{\langle q\rangle}(s)
=\zeta_{M^{\langle q\rangle}}(\lambda^{-s})
=\zeta_{M^q}(\lambda^{-qs}).
\]
\end{corollary}

\begin{proof}
By the determinant decomposition of Proposition \ref{prop:finite-rh-phase-lift-artin} and the identity
\(
\prod_{k=0}^{q-1}(1-x\omega_q^k)=1-x^q
\), for any eigenvalue \(\mu\) of \(M\),
\[
\prod_{k=0}^{q-1}(1-z\mu\omega_q^k)=1-(z\mu)^q=1-z^q\mu^q.
\]
Multiplying over all eigenvalues with algebraic multiplicity gives
\(
\det(I-zM^{\langle q\rangle})=\prod_{\mu}(1-z^q\mu^q)=\det(I-z^qM^q)
\).
Taking reciprocals yields the \(\zeta\)-identity; the final equation follows by substituting \(z=\lambda^{-s}\). \qed
\end{proof}

\begin{corollary}[Divisibility filtering of trace sequences under cyclic lift]\label{cor:finite-rh-phase-lift-trace-filter}
Using the notation of Proposition \ref{prop:finite-rh-phase-lift}, let \(M^{\langle q\rangle}:=M\otimes P_q\).
Then for any integer \(n\ge 1\),
\[
\Tr\!\bigl((M^{\langle q\rangle})^{n}\bigr)
=\Tr(M^n)\,\Tr(P_q^n)
=
\begin{cases}
q\,\Tr(M^n), & q\mid n,\\[4pt]
0, & q\nmid n.
\end{cases}
\]
\end{corollary}
\begin{proof}
By the Kronecker product property \((A\otimes B)^n=A^n\otimes B^n\) and \(\Tr(A\otimes B)=\Tr(A)\Tr(B)\),
\(\Tr((M^{\langle q\rangle})^n)=\Tr(M^n)\Tr(P_q^n)\).
Now \(P_q^n\) is the \(n\)-th power of the \(q\)-cyclic permutation, whose trace equals the number of fixed points: this is \(q\) if and only if \(q\mid n\), and \(0\) otherwise.
\end{proof}

\begin{remark}[Mechanistic interpretation: phase comb teeth \texorpdfstring{$\neq$}{!=} phase mixing]\label{rem:finite-rh-phase-lift-divisibility-filter}
Corollary \ref{cor:finite-rh-phase-lift-trace-filter} shows that the cyclic lift achieves "phase subdivision" on the critical line pole side, but simultaneously introduces strict divisibility gating on the length/trace side---only length layers with \(n\equiv 0\pmod q\) carry the additional information introduced by the lift.
It therefore more closely produces "sparse but large-amplitude resonance spikes" rather than continuous phase interference present at every length; to simulate the finer oscillation patterns found in explicit formulas, one must superpose stronger mixing mechanisms (e.g., multi-scale parallel assembly or continuous-scale mixing).
\end{remark}

\paragraph{Continuous phase limit: Jensen/Mahler free energy and critical kink}\label{par:finite-rh-jensen-free-energy}
The Artin decomposition of the cyclic lift provides a natural limiting mechanism on the "phase side": when the phase comb teeth become sufficiently fine, rather than tracking individual pole positions, it is more natural to study a macroscopic quantity depending only on spectral moduli (the phase-averaged \(\log|\det|\)), which develops a non-differentiable kink at the threshold
\(\sigma=\log|\mu|/\log\lambda\),
and in particular when a saturating mode \(|\mu|=\sqrt{\lambda}\) exists, \(\sigma=\tfrac12\) becomes a "phase transition point."

\begin{proposition}[Jensen averaging identity: the phase-averaged \texorpdfstring{$\log|\det|$}{log|det|} depends only on spectral moduli]\label{prop:finite-rh-jensen-free-energy}
Let \(M\) be a finite-dimensional matrix, \(\lambda=\rho(M)>1\), with nonzero eigenvalues (with multiplicity) \(\{\mu_j\}_{j=1}^{J}\).
For any \(\sigma\in\RR\), define
\[
G_M(\sigma):=\int_0^{2\pi}\log\Big|\det\bigl(I-\lambda^{-\sigma}e^{i\theta}M\bigr)\Big|\frac{d\theta}{2\pi}.
\]
Then the closed form holds:
\[
\boxed{\;
G_M(\sigma)=\sum_{j=1}^{J}\max\bigl(0,\ \log|\mu_j|-\sigma\log\lambda\bigr)\; }.
\]
Moreover, for any cyclic lift \(M^{\langle q\rangle}=M\otimes P_q\) with \(q\ge 2\), if one defines (normalized by dimension)
\[
G_{M^{\langle q\rangle}}(\sigma):=\frac{1}{q}\int_0^{2\pi}\log\Big|\det\bigl(I-\lambda^{-\sigma}e^{i\theta}M^{\langle q\rangle}\bigr)\Big|\frac{d\theta}{2\pi},
\]
then \(G_{M^{\langle q\rangle}}(\sigma)=G_M(\sigma)\) identically; thus the "continuous phase limit" \(q\to\infty\) is stable in the sense of this macroscopic quantity.
\end{proposition}
\begin{proof}
By eigenvalue decomposition (counted with algebraic multiplicity),
\[
\det\bigl(I-\lambda^{-\sigma}e^{i\theta}M\bigr)=\prod_{j=1}^{J}\bigl(1-\lambda^{-\sigma}e^{i\theta}\mu_j\bigr).
\]
Applying the Jensen averaging identity to each factor (for any \(a\in\CC\)):
\[
\int_0^{2\pi}\log|1-ae^{i\theta}|\frac{d\theta}{2\pi}
=
\begin{cases}
0,&|a|\le 1,\\
\log|a|,&|a|>1,
\end{cases}
\]
i.e.,
\(
\int \log|1-ae^{i\theta}|= \max(0,\log|a|)
\).
Setting \(a=\lambda^{-\sigma}\mu_j\) and summing,
\[
G_M(\sigma)=\sum_{j=1}^{J}\max\bigl(0,\ \log|\lambda^{-\sigma}\mu_j|\bigr)
=\sum_{j=1}^{J}\max\bigl(0,\ \log|\mu_j|-\sigma\log\lambda\bigr).
\]
The conclusion for the lift follows immediately from the spectral phase subdivision of Proposition \ref{prop:finite-rh-phase-lift}
\(\mathrm{spec}(M^{\langle q\rangle})=\{\mu_j\omega_q^k\}\):
each \(\mu_j\) appears \(q\) times in the lift with unchanged modulus, hence
\[
G_{M^{\langle q\rangle}}(\sigma)=\frac{1}{q}\sum_{j=1}^J\sum_{k=0}^{q-1}\max\bigl(0,\log|\mu_j\omega_q^k|-\sigma\log\lambda\bigr)=\sum_{j=1}^J\max\bigl(0,\log|\mu_j|-\sigma\log\lambda\bigr).
\]
\qed
\end{proof}

\begin{corollary}[Critical kink for the real-input 40-state kernel: \texorpdfstring{$\sigma=\tfrac12$}{sigma=1/2}]\label{cor:real-input-40-jensen-kink}
For the real-input 40-state kernel, \(\lambda=\varphi^2\) and its nonzero spectrum is
\(\{\lambda,\ -\sqrt{\lambda},\ 1^{(2)},\ (-1)^{(3)},\ \lambda^{-1},\ \lambda^{-1/2}\}\).
Thus on the strip \(0\le \sigma\le 1\), only the two points with modulus \(>\!1\), namely \(|\mu|=\lambda\) and \(|\mu|=\sqrt{\lambda}\), contribute to \(G_M(\sigma)\), giving
\[
G_M(\sigma)=
\begin{cases}
(1-\sigma)\log\lambda,&\tfrac12\le \sigma\le 1,\\[4pt]
\bigl(\tfrac32-2\sigma\bigr)\log\lambda,&0\le \sigma\le \tfrac12.
\end{cases}
\]
In particular, \(G_M\) has a non-differentiable kink at \(\sigma=\tfrac12\) (the slope jumps from \(-\log\lambda\) to \(-2\log\lambda\)).
\end{corollary}

\begin{remark}[Height--scale calibration: aligning critical line comb tooth density via \texorpdfstring{$T=(2\pi)\lambda^q$}{T=(2pi)lambda^q}]\label{rem:finite-rh-height-scale-calibration}
Let \(M\) satisfy RH-sharp, and write a saturating eigenvalue as \(\mu=\sqrt{\lambda}\,e^{i\theta}\).
By Proposition \ref{prop:finite-rh-phase-lift}, the cyclic lift \(M^{\langle q\rangle}\) generates poles (toy-zeros) on the critical line \(\Re(s)=\tfrac12\) with imaginary parts
\[
\gamma_{k,m}=\frac{\theta+2\pi(\frac{k}{q}+m)}{\log\lambda},
\qquad k=0,1,\dots,q-1,\ m\in\ZZ,
\]
so the fundamental spacing is
\[
\Delta\gamma_{\mathrm{toy}}=\frac{2\pi}{q\log\lambda}.
\]
\paragraph{Scale calibration}
If one defines the "effective height" as
\[
T:=(2\pi)\lambda^q
\quad\Longleftrightarrow\quad
\log\!\left(\frac{T}{2\pi}\right)=q\log\lambda,
\]
then the above spacing can be rewritten in logarithmic-scale form as
\[
\boxed{\ \Delta\gamma_{\mathrm{toy}}=\frac{2\pi}{\log(T/(2\pi))}\ }.
\]
This structure formally aligns with the average spacing derived from the Riemann zero counting main term \(N(T)\sim \frac{T}{2\pi}\log\frac{T}{2\pi}\), namely \(2\pi/\log(T/(2\pi))\); thus the lift order \(q\) can be interpreted as the logarithmic-scale parameter of the height window (\(q\) increasing corresponds to zero-point density at greater heights).
\end{remark}

\paragraph{Inverse problem of the toy-explicit-formula: zero engineering (a computable spectral design task)}
Given a target imaginary-part set \(\Gamma:=\{\gamma_1,\dots,\gamma_N\}\subset\RR\) (e.g., from some height window), fix an RH-sharp base kernel \(M\) and its \(\lambda=\rho(M)\).
Use Remark \ref{rem:finite-rh-height-scale-calibration} to select the lift order \(q\) so that \(q\log\lambda\approx \log(T/(2\pi))\) (logarithmic scale of the target height \(T\)); then fit \(\Gamma\) using the critical line comb teeth of the cyclic lift \(M^{\langle q\rangle}\)
\[
\gamma_{k,m}=\frac{\theta+2\pi(\frac{k}{q}+m)}{\log\lambda}
\]
(choosing integer pairs \((k,m)\) to minimize error).
\par
If the strict lattice structure of a single \((\lambda,q)\) is insufficient to reproduce the "non-lattice-point" local statistics of the target, one can employ a minimal multi-scale parallel mechanism: take several RH-sharp kernels \(M^{(a)}\) (with possibly different \(\lambda_a\)), and form a block-diagonal combination
\(
\bigoplus_a M^{(a)\langle q_a\rangle}
\).
Since \(\zeta\) decomposes multiplicatively under block diagonalization, the critical line pole set is the union of the comb teeth from each channel;
when the \(\log\lambda_a\) lack rational relations, the different channel periods \(2\pi/\log\lambda_a\) produce beat frequencies that break the periodicity of a single lattice.

\begin{proposition}[\texorpdfstring{$\varepsilon$}{eps}-approximation of finite-window zero sets (single-scale comb teeth)]\label{prop:finite-rh-zero-engineering-eps}
Fix an RH-sharp base kernel \(M\), denote \(\lambda=\rho(M)>1\), and take one of its saturating eigenvalues \(\mu=\sqrt{\lambda}\,e^{i\theta}\) (\(\theta\in\RR\)).
Let \(M^{\langle q\rangle}=M\otimes P_q\) be the cyclic lift (Proposition \ref{prop:finite-rh-phase-lift}), and denote the imaginary-part comb teeth on the critical line \(\Re(s)=\tfrac12\) generated by this saturating mode as
\[
\gamma_{k,m}^{(q)}:=\frac{\theta+2\pi(\frac{k}{q}+m)}{\log\lambda},
\qquad k\in\{0,\dots,q-1\},\ m\in\ZZ.
\]
Then for any \(\varepsilon>0\) and any finite set \(\Gamma=\{\gamma_1,\dots,\gamma_N\}\subset\RR\), provided one chooses an integer
\[
q\ \ge\ \frac{\pi}{\varepsilon\log\lambda},
\]
there exist integer pairs \(\{(k_j,m_j)\}_{j=1}^N\) such that for each \(j\),
\[
\left|\gamma_j-\gamma_{k_j,m_j}^{(q)}\right|\le \varepsilon.
\]
\end{proposition}
\begin{proof}
For fixed \(q\), the above set satisfies
\[
\left\{\gamma_{k,m}^{(q)}:\ k=0,\dots,q-1,\ m\in\ZZ\right\}
=\left\{\frac{\theta+2\pi n/q}{\log\lambda}:\ n\in\ZZ\right\},
\]
which forms an arithmetic lattice on the real line with step size \(\Delta\gamma_{\mathrm{toy}}=2\pi/(q\log\lambda)\) (Remark \ref{rem:finite-rh-height-scale-calibration}).
Thus for any \(\gamma\in\RR\), choosing
\[
\text{some }n\in\ZZ\text{ achieving }\min_{n'\in\ZZ}\left|\gamma-\frac{\theta+2\pi n'/q}{\log\lambda}\right|
\]
guarantees
\[
\min_{n'\in\ZZ}\left|\gamma-\frac{\theta+2\pi n'/q}{\log\lambda}\right|
\le \frac{1}{2}\Delta\gamma_{\mathrm{toy}}
=\frac{\pi}{q\log\lambda}.
\]
Writing the chosen \(n\) as \(n=k+qm\) (with \(k\in\{0,\dots,q-1\},\,m\in\ZZ\)), one obtains the corresponding \((k,m)\).
Repeating this selection for each \(\gamma_j\) and requiring \(q\ge \pi/(\varepsilon\log\lambda)\) yields the stated \(\varepsilon\)-approximation.
\end{proof}

\begin{proposition}[Necessary condition for beat de-latticing: irrational relation of \texorpdfstring{$\log\lambda$}{loglambda}]\label{prop:finite-rh-beat-irrational-log}
Suppose there are two RH-sharp channels \((M^{(a)},\lambda_a)\), \((M^{(b)},\lambda_b)\) with \(\lambda_a,\lambda_b>1\), and respective cyclic lift orders \(q_a,q_b\ge 1\) (allowing \(q_a=q_b=1\)).
Assume each channel produces at least one complete arithmetic comb tooth sequence on the critical line \(\Re(s)=\tfrac12\), with fundamental step sizes
\[
\Delta_a=\frac{2\pi}{q_a\log\lambda_a},\qquad
\Delta_b=\frac{2\pi}{q_b\log\lambda_b}.
\]
If \(\log\lambda_a/\log\lambda_b\in\QQ\) (equivalently \(\Delta_a/\Delta_b\in\QQ\)), then there exists \(\Delta>0\) such that the critical line imaginary-part point sets of both channels are simultaneously periodic under \(\Delta\); hence their union remains a finite union of finitely many arithmetic lattices (arithmetic vertical line combs), unable to escape periodic phase-locking at any finite scale.
Therefore, if one wishes to obtain \emph{no common imaginary period} critical line phase structure through multi-channel parallel assembly (i.e., "beat de-latticing"), it is necessary that
\[
\boxed{\ \frac{\log\lambda_a}{\log\lambda_b}\notin\QQ\ }.
\]
\end{proposition}
\begin{proof}
If \(\log\lambda_a/\log\lambda_b=p/q\in\QQ\) (\(p,q\in\NN\)), then
\[
\frac{\Delta_a}{\Delta_b}
=\frac{q_b\log\lambda_b}{q_a\log\lambda_a}
=\frac{q_b q}{q_a p}\in\QQ.
\]
Hence there exist integers \(u,v\ge 1\) with \(u\Delta_a=v\Delta_b\). Setting \(\Delta:=u\Delta_a=v\Delta_b\), for any comb tooth sequence of the form \(t_0+\Delta_a\ZZ\),
\[
(t_0+\Delta_a\ZZ)+\Delta=t_0+\Delta_a(\ZZ+u)=t_0+\Delta_a\ZZ,
\]
so it is invariant under translation by \(\Delta\); similarly for comb tooth sequences with step \(\Delta_b\).
Since each channel's critical line point set is a finite union of comb tooth sequences, their union is periodic under \(\Delta\), hence still belongs to a finite union of arithmetic lattices.
The final statement is its contrapositive: to rule out any nonzero common imaginary period, one must have \(\Delta_a/\Delta_b\notin\QQ\), equivalent to \(\log\lambda_a/\log\lambda_b\notin\QQ\).
\end{proof}

\begin{proposition}[All nontrivial poles in the critical strip lie on \texorpdfstring{$\Re(s)=\tfrac12$}{Re(s)=1/2}]\label{prop:finite-rh-40}
Let $\lambda=\rho(M)=\varphi^2$. Then in the open strip $0<\Re(s)<1$, all poles of $\widehat{\zeta}(s)$ lie strictly on the line
\[
\Re(s)=\frac12.
\]
More specifically, the only pole sequence in $0<\Re(s)<1$ comes from the eigenvalue $-\varphi=-\sqrt{\lambda}$, located at
\[
s=\frac12+\frac{(2k+1)\pi i}{\log\lambda},\qquad k\in\ZZ.
\]
\end{proposition}

\begin{proof}
By the spectral decomposition above, the nonzero eigenvalue set of $M$ is
\[
\{\lambda,\ -\sqrt{\lambda},\ 1^{(2)},\ (-1)^{(3)},\ \lambda^{-1},\ \lambda^{-1/2}\},
\]
with the remaining $31$ eigenvalues equal to $0$. For each nonzero eigenvalue $\mu$, $\det(I-\lambda^{-s}M)=0$ is equivalent to $\lambda^{-s}\mu=1$.

When $\mu=\lambda$, the principal pole sequence $s=1+2\pi i k/\log\lambda$ is obtained, which does not lie in the open strip $0<\Re(s)<1$.
When $\mu=-\sqrt{\lambda}$, taking the complex logarithm branch $\log(-\sqrt{\lambda})=\tfrac12\log\lambda+i(2k+1)\pi$, one obtains
\(
s=\tfrac12+\frac{(2k+1)\pi i}{\log\lambda}
\), which satisfies $0<\Re(s)<1$.
The remaining eigenvalues satisfy $|\mu|\le 1$, so their corresponding poles satisfy $\Re(s)\le 0$ (lying on the left boundary of the critical strip or further left), and hence produce no poles in the open strip $0<\Re(s)<1$.
\end{proof}

\begin{remark}[Dirichlet variable and "critical line" formulation]
Setting $r:=\lambda^{1-s}$, the factor $r+\varphi$ corresponds to singularities satisfying

r=-\varphi=\varphi e^{i(\pi+2\pi k)}.
\]
From $\lambda=\varphi^2$,
\[
s=\frac12-\frac{(2k+1)\pi i}{\log\lambda},
\]
i.e., all non-real singularities with $\Re(s)>0$ lie on $\Re(s)=1/2$.
\end{remark}

\begin{proposition}[RH-sharp parity cancellation: general mechanism via M\"obius inversion]\label{prop:finite-rh-parity-general}
Let \(\lambda>1\), and suppose a sequence of real numbers \(a_n\) satisfies
\[
a_n=\lambda^n+\epsilon^n\lambda^{n/2}+O(\lambda^{n/3}),
\qquad \epsilon\in\{\pm 1\}.
\]
Define the primitive orbit count (Witt/M\"obius inversion)
\[
p_n:=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}.
\]
Then as \(n\to\infty\):
\begin{enumerate}
  \item If \(n\) is odd, then
  \[
  p_n=\frac{\lambda^n}{n}+\frac{\epsilon^n\lambda^{n/2}}{n}+O\!\left(\frac{\lambda^{n/3}}{n}\right).
  \]
  \item If \(n\) is even, then the \(d=1,2\) terms produce exact cancellation at the \(\lambda^{n/2}\) level, so
  \[
  p_n=\frac{\lambda^n}{n}+O\!\left(\frac{\lambda^{n/3}}{n}\right).
  \]
\end{enumerate}
\end{proposition}
\begin{proof}
By definition,
\[
np_n=\sum_{d\mid n}\mu(d)\,a_{n/d}.
\]
If \(n\) is odd, then for all \(d\mid n\) with \(d\ge 2\), we have \(d\ge 3\), so
\(
a_{n/d}=O(\lambda^{n/3})
\).
Thus
\[
np_n=a_n+\sum_{\substack{d\mid n\\ d\ge 3}}\mu(d)\,a_{n/d}
=\lambda^n+\epsilon^n\lambda^{n/2}+O(\lambda^{n/3}),
\]
yielding the first formula.
\par
If \(n\) is even, separate the \(d=1,2\) terms:
\[
np_n=a_n-a_{n/2}+\sum_{\substack{d\mid n\\ d\ge 3}}\mu(d)\,a_{n/d}.
\]
By assumption, since \(n\) is even, \(\epsilon^n=1\), so
\[
a_n-a_{n/2}
\;=\;
(\lambda^n+\lambda^{n/2}+O(\lambda^{n/3}))-(\lambda^{n/2}+O(\lambda^{n/4}))
\;=\;
\lambda^n+O(\lambda^{n/3}).
\]
The remainder is still \(O(\lambda^{n/3})\), and combining gives the second formula. \qed
\end{proof}

\begin{corollary}[Explicit formula shape: square-root oscillation of \texorpdfstring{$a_n$}{a\_n} and parity cancellation of primitives]\label{cor:finite-rh-explicit-formula}
Denote $a_n=\Tr(M^n)$. Then
\[
a_n=\lambda^n+(-1)^n\lambda^{n/2}+O(1),
\]
and the primitive orbit count satisfies a finer parity structure:
\[
p_n=\frac{\lambda^n}{n}-\frac{\lambda^{n/2}}{n}+O\!\left(\frac{\lambda^{n/3}}{n}\right)\qquad(n\ \text{odd}),
\]
\[
p_n=\frac{\lambda^n}{n}+O\!\left(\frac{\lambda^{n/3}}{n}\right)\qquad(n\ \text{even}),
\]
where in the even case, the $\lambda^{n/2}$-level term is \textbf{exactly cancelled} by the $d=1$ and $d=2$ terms in the M\"obius inversion.
\[
\boxed{
\;p_n=\frac{\lambda^n}{n}+\frac{\bigl((-1)^n-\mathbf{1}_{2\mid n}\bigr)\lambda^{n/2}}{n}
+O\!\left(\frac{\lambda^{n/3}}{n}\right)\; }
\]
with a verifiable "parity resonance signature":
\[
\lim_{\substack{n\to\infty\\ n\ \mathrm{odd}}}\ \frac{n}{\lambda^{n/2}}\left(p_n-\frac{\lambda^n}{n}\right)=-1,
\qquad
\lim_{\substack{n\to\infty\\ n\ \mathrm{even}}}\ \frac{n}{\lambda^{n/2}}\left(p_n-\frac{\lambda^n}{n}\right)=0.
\]
\end{corollary}

\begin{remark}["Unit coefficient" square-root oscillation: observable on odd-length layers or even-length interference-removed quantities]\label{rem:finite-rh-unit-coefficient-observable}
In Corollary \ref{cor:finite-rh-explicit-formula}, the square-root oscillation term \( (-1)^n\lambda^{n/2}\) in the periodic trace layer has \textbf{coefficient exactly equal to \(1\)} (arising from the unique saturating eigenvalue \(-\sqrt{\lambda}\)).
However, at the primitive layer, the \(d=2\) term in the M\"obius inversion produces \textbf{exact interference cancellation} of the \(\lambda^{n/2}\)-level principal oscillation on even-length layers, so the "unit coefficient" should be read on the following two equivalent verifiable observables:
\begin{itemize}
  \item \textbf{(Odd-length signature)} For odd \(n\), a stronger normalized limit holds:
  \[
  \boxed{\ \frac{n}{\lambda^{n/2}}\left(p_n-\frac{\lambda^n}{n}\right)=-1+O(\lambda^{-n/6})\ }.
  \]
  \item \textbf{(Unified resonance index)} Define
  \[

  then \(\mathsf{E}_n\to -1\) (\(n\) odd) while \(\mathsf{E}_n\to 0\) (\(n\) even), so the "single-frequency, unit-coefficient" square-root oscillation appears at the primitive layer via \textbf{parity gating}.
\end{itemize}
This gives the RH-sharp "square-root-level error saturation" an extremely simple audit interface on the data side: one can directly observe the saturating oscillation using only the odd-length layers of \(p_n\) (or equivalently the parity limits of \(\mathsf{E}_n\)); no need to plot pole sequences or introduce external fitting.
\end{remark}

\begin{proposition}[Spectral annihilation difference operator: exact finite-order filtering for finite spectra]\label{prop:finite-rh-spectral-annihilation-filter}
Suppose \(\{a_n\}_{n\ge 1}\) is a finite-spectrum exponential sum:
\[
a_n=\sum_{j=1}^{J} m_j\,\mu_j^{\,n},
\]
where \(\mu_j\in\CC\) (counted with multiplicity) and \(m_j\in\CC\).
Let the shift operator \(L\) act as \((La)_n:=a_{n+1}\).
For any finite subset \(S\subset\{\mu_1,\dots,\mu_J\}\), define the polynomial
\[
Q_S(x):=\prod_{\mu\in S}(x-\mu).
\]
Then for all \(n\ge 1\),
\[
\boxed{\;
(Q_S(L)a)_n=\sum_{j=1}^{J} m_j\,Q_S(\mu_j)\,\mu_j^{\,n}\; }.
\]
Thus the contributions of all modes \(\mu_j\in S\) are \textbf{exactly annihilated} (since \(Q_S(\mu_j)=0\)).
\end{proposition}
\begin{proof}
For a single term \(\mu^n\), \(L(\mu^n)=\mu^{n+1}=\mu\cdot \mu^n\), so for any polynomial \(Q\), \(Q(L)(\mu^n)=Q(\mu)\mu^n\).
Linear superposition over \(\sum_j m_j\mu_j^n\) yields the conclusion. \qed
\end{proof}

\end{corollary}

\begin{proposition}[Phase resonance formula for the square-root term: critical modular spectrum and interference with the \texorpdfstring{$d=2$}{d=2} term]\label{prop:finite-rh-sqrt-resonance-general}
Let \(M\) be a primitive nonnegative integer matrix, \(\lambda=\rho(M)>1\), and assume the kernel RH spectral radius bound holds: all eigenvalues \(\mu\) other than the Perron root satisfy \(|\mu|\le \sqrt{\lambda}\).
Denote the critical modular spectrum set
\[
\mathcal{C}:=\{\mu\in\mathrm{spec}(M):\ |\mu|=\sqrt{\lambda}\},
\qquad
\omega_\mu:=\mu/\sqrt{\lambda}\ \ (|\omega_\mu|=1),
\]
and the strictly subcritical radius
\[
R:=\max\{|\mu|:\ \mu\in\mathrm{spec}(M),\ |\mu|<\sqrt{\lambda}\},
\qquad
\Lambda:=\max\{R,\ \lambda^{1/3}\}.
\]
Let \(a_n:=\Tr(M^n)\), and define the primitive orbit count
\[
p_n:=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}.
\]
Then as \(n\to\infty\), the asymptotic expansion holds:
\[
\boxed{\;
p_n
=\frac{\lambda^n}{n}
+\frac{\lambda^{n/2}}{n}\Bigl(\sum_{\mu\in\mathcal{C}}\omega_\mu^{\,n}-\mathbf{1}_{2\mid n}\Bigr)
+O\!\left(\frac{\Lambda^n}{n}\right)\; }.
\]
\end{proposition}
\begin{proof}
By spectral decomposition (Jordan terms absorbed into strictly lower-order terms),
\[
a_n=\Tr(M^n)=\lambda^n+\sum_{\mu\in\mathcal{C}}\mu^n+O(R^n)
=\lambda^n+\lambda^{n/2}\sum_{\mu\in\mathcal{C}}\omega_\mu^{\,n}+O(R^n).
\]
By M\"obius inversion,
\[
np_n=\sum_{d\mid n}\mu(d)\,a_{n/d}
=a_n-\mathbf{1}_{2\mid n}a_{n/2}+\sum_{\substack{d\mid n\\ d\ge 3}}\mu(d)\,a_{n/d}.
\]
For \(d\ge 3\), the principal term \(\lambda^{n/d}\) is at most \(\lambda^{n/3}\), and the error term \(O(R^{n/d})\) is at most \(O(R^{n/3})\), hence
\[
\sum_{\substack{d\mid n\\ d\ge 3}}\mu(d)\,a_{n/d}=O(\Lambda^n).
\]
On the other hand,
\[
a_{n/2}=\lambda^{n/2}+O(\lambda^{n/4})+O(R^{n/2}).
\]
Substituting and simplifying,
\[
np_n=\lambda^n+\lambda^{n/2}\sum_{\mu\in\mathcal{C}}\omega_\mu^{\,n}-\mathbf{1}_{2\mid n}\lambda^{n/2}+O(\Lambda^n),
\]
and dividing both sides by \(n\) gives the proposition. \qed
\end{proof}
