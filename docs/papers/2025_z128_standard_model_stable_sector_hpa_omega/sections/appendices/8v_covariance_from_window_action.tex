\subsection{Covariance from a window/region action (PT layer for T2-3)}
\label{app:covariance_from_window_action}

\AuditTag This appendix replaces ``assume a covariant dynamics class'' by a PT-level statement: a compatible action on windows/refinements induces a $^\ast$-automorphism action on the inductive-limit local net and the quasi-local algebra.

\subsubsection{Compatible action on windows}
\begin{definition}[Window action compatible with refinement]
\label{def:window_action_compatible_refinement}
\MathTag Let $\mathcal{W}$ be the addressing-window class (Appendix~\ref{app:construct_local_net_from_protocol}).
Let $\mathcal{G}$ be a group (or semigroup) acting on windows by maps $g:W\mapsto gW$.
We say the action is \emph{compatible with refinement} if:
\begin{itemize}
\item if $W'\preceq W$ then $(gW')\preceq (gW)$,
\item the induced observable pullback maps commute with the action (functoriality under $g$).
\end{itemize}
\end{definition}

\subsubsection{Induced automorphisms on local algebras}
\begin{proposition}[Induced action on the protocol local net (PT)]
\label{prop:induced_action_on_prot_net}
\MathTag Under Definition~\ref{def:window_action_compatible_refinement}, there exists an induced action
\[
\alpha_g:\mathcal{A}_{\mathrm{prot}}(\mathcal{O})\to \mathcal{A}_{\mathrm{prot}}(g\mathcal{O})
\]
by $^\ast$-isomorphisms for all $g\in\mathcal{G}$ and $\mathcal{O}\in\mathfrak{R}_{\mathrm{prot}}$, compatible with isotony and inductive limits.
\end{proposition}
\begin{proof}
Compatibility with refinement ensures that the directed system defining $\mathcal{A}_{\mathrm{prot}}(\mathcal{O})$ is mapped to the directed system defining $\mathcal{A}_{\mathrm{prot}}(g\mathcal{O})$ and that connecting morphisms commute.
Thus $\alpha_g$ is well-defined on the inductive limit and preserves the $^\ast$-algebra structure.
\end{proof}

\begin{corollary}[Induced action on the quasi-local algebra (PT)]
\label{cor:induced_action_on_quasilocal}
\MathTag The induced maps extend to a $^\ast$-automorphism action on the quasi-local algebra $\mathcal{A}_{\mathrm{prot}}$, and satisfy net covariance:
\[
\alpha_g\bigl(\mathcal{A}_{\mathrm{prot}}(\mathcal{O})\bigr)=\mathcal{A}_{\mathrm{prot}}(g\mathcal{O}).
\]
\end{corollary}
\begin{proof}
By construction of $\mathcal{A}_{\mathrm{prot}}$ as the inductive limit over regions, and functoriality of $\alpha_g$ on region inclusions.
\end{proof}

\paragraph{Interface reading.}
\InterfaceTag When $\mathcal{G}$ is identified with a symmetry group acting on spacetime regions (e.g.\ translations), Corollary~\ref{cor:induced_action_on_quasilocal} provides the covariance axiom of AQFT nets at the protocol-induced level.
