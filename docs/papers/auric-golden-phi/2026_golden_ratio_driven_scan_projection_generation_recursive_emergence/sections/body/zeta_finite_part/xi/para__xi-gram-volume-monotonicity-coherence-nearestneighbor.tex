\begin{proposition}[Volume monotonicity under linear observation: Gram--det entropy strictly increases under contraction operators]\label{prop:xi-gram-det-entropy-monotone-under-contraction}
Let \(H\) be a complex Hilbert space, and let \(\psi_1,\dots,\psi_\kappa\in H\) be linearly independent. Denote the Gram matrix by
\[
G=(\langle\psi_i,\psi_j\rangle)_{1\le i,j\le\kappa}\succ0,
\qquad
S:=-\log\det G.
\]
For any bounded linear operator \(T:H\to H\) satisfying \(T^\ast T\preceq I\), define
\[
G_T:=(\langle T\psi_i,T\psi_j\rangle)_{1\le i,j\le\kappa}.
\]
Then
\[
\boxed{\;
0\prec G_T\preceq G,\qquad
\det G_T\le \det G,\qquad
S_T:=-\log\det G_T\ge S.\;}
\]
Equality holds if and only if \(T\) is an isometric isomorphism on \(\mathrm{span}\{\psi_1,\dots,\psi_\kappa\}\) (equivalently, \(T^\ast T=I\) restricted to that subspace).
Hence any linearly contractive observation, such as bandwidth truncation/noise smoothing/subspace projection, is uniformly constrained by a strict irreversible volume law.
\end{proposition}
\begin{proof}
For any \(c=(c_1,\dots,c_\kappa)\in\CC^\kappa\), let \(v=\sum_{j=1}^\kappa c_j\psi_j\).
Then
\[
c^\ast G_T c=\|Tv\|^2\le \|v\|^2=c^\ast G c,
\]
so \(G_T\preceq G\), and linear independence of \(\{\psi_j\}\) gives \(G_T\succ0\).
By eigenvalue monotonicity in Loewner order (hence determinant monotonicity), \(\det G_T\le \det G\), so \(S_T\ge S\).
If \(\det G_T=\det G\), then \(G_T\preceq G\) with equal determinants forces \(G_T=G\), hence \(\|Tv\|=\|v\|\) for all \(v\in\mathrm{span}\{\psi_j\}\), i.e. \(T\) is isometric on that subspace. QED.
\end{proof}

\begin{proposition}[Lower bound by coherence energy: volume entropy forced by squared coherence]\label{prop:xi-gram-entropy-lower-bound-by-coherence}
Retain the notation of Proposition \ref{prop:xi-gram-det-entropy-monotone-under-contraction} and further assume \(\|\psi_j\|=1\) (\(1\le j\le\kappa\)).
Then \(G=I+E\) with \(E\) Hermitian and \(\mathrm{diag}(E)=0\).
One has the strict lower bound
\[
\boxed{\;
S=-\log\det(I+E)\ \ge\ \frac{\|E\|_{\mathrm F}^{2}}{2\,(1+\|E\|_{2})}\;}
\]
where \(\|E\|_{\mathrm F}^{2}=\sum_{i\neq j}|\langle\psi_i,\psi_j\rangle|^2\) and \(\|E\|_2\) is the spectral norm.
In particular, when \(\|E\|_2=o(1)\),
\[
S=\frac12\sum_{i\neq j}|\langle\psi_i,\psi_j\rangle|^2+o\!\Bigl(\sum_{i\neq j}|\langle\psi_i,\psi_j\rangle|^2\Bigr),
\]
so in the ``weak-coherence phase'' the volume entropy is fully dominated by pairwise squared coherence.
\end{proposition}
\begin{proof}[Proof sketch]
Let \(\{\lambda_r\}_{r=1}^\kappa\) be the real eigenvalues of \(E\). Then \(\lambda_r>-1\) (since \(I+E\succ0\)) and \(\sum_r\lambda_r=\mathrm{tr}(E)=0\).
Hence
\[
S=-\sum_{r=1}^\kappa\log(1+\lambda_r)=\sum_{r=1}^\kappa\bigl(\lambda_r-\log(1+\lambda_r)\bigr).
\]
For any \(x>-1\), the scalar inequality
\[
x-\log(1+x)\ge \frac{x^2}{2(1+|x|)}\ge \frac{x^2}{2(1+\|E\|_2)}.
\]
Summing and using \(\sum_r\lambda_r^2=\|E\|_{\mathrm F}^2\) yields the stated bound.
The weak-coherence expansion follows from the matrix-log series
\(
\log\det(I+E)=\mathrm{tr}(E)-\frac12\mathrm{tr}(E^2)+O(\|E\|_2\|E\|_{\mathrm F}^2)
\)
together with \(\mathrm{tr}(E)=0\). QED.
\end{proof}

\begin{proposition}[Nearest-neighbor product upper bound: volume entropy is tightly closed by adjacent pairwise degeneracy]\label{prop:xi-gram-det-nearest-neighbor-product-upper-bound}
Let \(\psi_1,\dots,\psi_\kappa\in H\) satisfy \(\|\psi_j\|=1\), with Gram matrix \(G=(\langle\psi_i,\psi_j\rangle)\).
Then the strict inequality
\[
\boxed{\;
\det G\ \le\ \prod_{j=1}^{\kappa-1}\bigl(1-|\langle\psi_{j+1},\psi_j\rangle|^2\bigr)\;}
\qquad
\Longrightarrow\qquad
S=-\log\det G\ \ge\ \sum_{j=1}^{\kappa-1}-\log\bigl(1-|\langle\psi_{j+1},\psi_j\rangle|^2\bigr).
\]
In the Hellinger--RKHS setting of this work, take \(\psi_j=k_{s_j}\) with \(s_1<\cdots<s_\kappa\). Then \(|\langle\psi_{j+1},\psi_j\rangle|=A(\Delta_j)\) (\(\Delta_j=s_{j+1}-s_j\)), so the volume-entropy potential is strictly closed along adjacent spacings by a one-dimensional nearest-neighbor ledger; any spacing collapse (\(A(\Delta_j)\uparrow 1\)) forces logarithmic divergence on the right-hand side, yielding a minimal auditable certificate of clustering pathology.
\end{proposition}
\begin{proof}
Apply Gram--Schmidt orthogonalization successively to \(\{\psi_j\}\): set
\(
v_1:=\psi_1
\), and for \(j\ge 2\),
\(
v_j:=\psi_j-\mathrm{Proj}_{\mathrm{span}\{\psi_1,\dots,\psi_{j-1}\}}\psi_j
\).
The volume interpretation of the Gram determinant gives
\[
\det G=\prod_{j=1}^{\kappa}\|v_j\|^2.
\]
And for \(j\ge 2\),
\[
\|v_j\|^2=1-\bigl\|\mathrm{Proj}_{\mathrm{span}\{\psi_1,\dots,\psi_{j-1}\}}\psi_j\bigr\|^2
\le 1-|\langle\psi_j,\psi_{j-1}\rangle|^2,
\]
because the projection norm onto \(\mathrm{span}\{\psi_1,\dots,\psi_{j-1}\}\) is at least the component magnitude along the unit vector \(\psi_{j-1}\).
Multiplying yields the displayed inequality; taking negative logarithms gives the entropy lower bound. QED.
\end{proof}
