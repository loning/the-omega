\section{Radial structure \texorpdfstring{$\rho$}{rho}: metrizing the multiplicative skeleton}
\label{sec:radial}

\subsection{Radial weights and the radial character}
Fix a weight function $w:\PP\to\RR$. Define the radial character
$$
\rho_w(n) := \exp\!\Big(\sum_{p\in\PP} a_p(n)\, w(p)\Big),
$$
where $n=\prod_{p} p^{a_p(n)}$.

\begin{proposition}[Radial homomorphism]
\label{prop:radial_hom}
For all $m,n\in M$ one has $\rho_w(mn)=\rho_w(m)\rho_w(n)$.
\end{proposition}

\begin{proof}
Using $a_p(mn)=a_p(m)+a_p(n)$ and additivity of the logarithm,
$$
\log\rho_w(mn)=\sum_{p\in\PP} (a_p(m)+a_p(n))w(p)=\log\rho_w(m)+\log\rho_w(n).
$$
Exponentiating yields the claim.
\end{proof}

\subsection{Minimal choice and tunable families}
\begin{itemize}
    \item \textbf{Minimal choice.} If $w(p)=\log p$, then
    $$
    \rho_w(n)=\exp\Big(\sum_p a_p(n)\log p\Big)=n.
    $$
    In this case we simply write $\rho(n)=n$.
    \item \textbf{Tunable family.} Other choices of $w$ correspond to alternative ``radial scales'' and can be used to couple the arithmetic skeleton to external weighting schemes (e.g., spectral weights). Unless stated otherwise, we adopt the minimal choice $\rho(n)=n$.
\end{itemize}
