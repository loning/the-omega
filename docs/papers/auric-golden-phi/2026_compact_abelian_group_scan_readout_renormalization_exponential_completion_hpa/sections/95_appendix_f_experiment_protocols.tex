\section{Appendix F: Computational Experiment Protocol (Minimal Reproducible Description)}
\label{app:experiments}

\subsection{F.1 One-Dimensional Discrepancy and Return Times}

\textbf{Input:} $\alpha$, initial value $x_0$, sample length $N$, window family $\{W_j\}$.\\
\textbf{Output:} $D_N^*$, three-gap statistics, distribution of $\varepsilon$-return times, and window-frequency errors.

\subsection{F.2 Multidimensional Box Coverage and Correlation}

\textbf{Input:} dimension $d$, step vector $\alpha$, initial value $x_0$, sample length $N$, box-partition granularity $m$.\\
\textbf{Output:} covered-box ratio, deviation of empirical measure from Haar, and decay of coordinate correlations with $N$.

\subsection{F.3 Symbolic Complexity and Substitution Consistency}

\textbf{Input:} rotation-coded sequence $u_0,\dots,u_{N-1}$.\\
\textbf{Output:} complexity function $p(n)$, convergence of word frequencies, and deviation curves from substitution PF frequencies.

\subsection{F.4 Cut-and-Project Generation and Local-Patch Frequencies}

\textbf{Input:} lattice basis, decomposition $E_\parallel\oplus E_\perp$, window $W$, truncation radius $R$.\\
\textbf{Output:} local-patch frequencies of $\Lambda$ and stability indices under window-parameter perturbation.
