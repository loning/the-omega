\subsubsection{Second-Order Characterization of the Golden Bias and Divergence Spectrum: \texorpdfstring{$\mathrm{TV}$}{TV}/\texorpdfstring{$\chi^2$}{chi2}/R\'enyi Rigidity Collapse}

\begin{proposition}[``Variance = Bias'' Characterization and Uniqueness of the Golden Bias]\label{prop:pom-golden-bias-variance-equals-bias}
Let \(p\in(1/2,1)\), \(q:=1-p\).
Then the equation
\[
pq=p-q
\]
has a unique solution in the interval \((1/2,1)\), given by
\[
\boxed{\ p=\varphi^{-1},\qquad q=\varphi^{-2}\ }.
\]
Equivalently, if \(B\sim\mathrm{Bern}(p)\), then
\[
\mathrm{Var}(B)=\mathbb{E}(2B-1)
\quad\Longleftrightarrow\quad
p=\varphi^{-1}.
\]
This characterization uniquely extracts the ``golden bias'' from the algebraic self-consistency between the second-order fluctuation (variance) and the first-order imbalance (bias) of a binary split, without invoking the Fibonacci recurrence.
\end{proposition}

\begin{proof}
Note that \(\mathrm{Var}(B)=pq\) and \(\mathbb{E}(2B-1)=p-q=2p-1\).
Hence the condition \(pq=p-q\) is equivalent to
\(
p(1-p)=2p-1
\), i.e.,
\(
p^2+p-1=0
\).
Its two roots are \(p=\frac{-1\pm\sqrt5}{2}\), of which the unique root in \((1/2,1)\) is \(p=\frac{\sqrt5-1}{2}=\varphi^{-1}\).
Consequently \(q=1-p=\varphi^{-2}\). \qed
\end{proof}

\begin{proposition}[Divergence Collapse and Closed-Form Linearization for the Complementary Golden Bernoulli Pair]\label{prop:pom-complementary-golden-bernoulli-divergence-collapse}
Consider the complementary pair
\[
P:=\mathrm{Bern}(\varphi^{-1}),\qquad
Q:=\mathrm{Bern}(\varphi^{-2})\quad(\varphi^{-2}=1-\varphi^{-1}).
\]
Then the standard divergences exhibit equal-value collapse and closed-form linearization:
\[
\boxed{\;
\mathrm{TV}(P,Q)=|\varphi^{-1}-\varphi^{-2}|=\varphi^{-3},\qquad
\chi^2(P\|Q)=\varphi^{-3}\;}
\]
\[
\boxed{\;
D(P\|Q)=\varphi^{-3}\log_2\varphi,\qquad
H^2(P,Q)=1-2\sqrt{\varphi^{-3}}=1-2\varphi^{-3/2}.\;}
\]
Consequently, this complementary pair exhibits single-parameter rigidity on the ``distinguishability scale'': all the above quantities are governed by the same power of \(\varphi\).
\end{proposition}

\begin{proof}
For two-point distributions, \(\mathrm{TV}(P,Q)=\frac12\sum_x|P(x)-Q(x)|=|p-q|\); substituting \(p=\varphi^{-1},q=\varphi^{-2}\) yields \(\mathrm{TV}=\varphi^{-3}\).
\par
For \(\chi^2(P\|Q)=\sum_x \frac{(P(x)-Q(x))^2}{Q(x)}\), we have
\[
\chi^2(P\|Q)=(p-q)^2\Bigl(\frac{1}{q}+\frac{1}{p}\Bigr)=\frac{(p-q)^2}{pq}.
\]
Since \(pq=\varphi^{-3}\) and \(p-q=\varphi^{-3}\), it follows that \(\chi^2=\varphi^{-3}\).
\par
The KL divergence is
\[
D(P\|Q)=p\log_2\frac{p}{q}+q\log_2\frac{q}{p}=(p-q)\log_2\frac{p}{q}.
\]
Here \(p/q=\varphi\) and \(p-q=\varphi^{-3}\), so \(D(P\|Q)=\varphi^{-3}\log_2\varphi\).
\par
The squared Hellinger distance is
\(
H^2(P,Q)=1-\sum_x\sqrt{P(x)Q(x)}=1-(\sqrt{pq}+\sqrt{qp})=1-2\sqrt{pq}=1-2\varphi^{-3/2}
\).
\qed
\end{proof}

\begin{proposition}[All-Order Closed Form and Endpoint Spectrum of the R\'enyi Divergence Family]\label{prop:pom-complementary-golden-bernoulli-renyi-spectrum}
For any \(\alpha>0\), \(\alpha\neq 1\), with
\(
P=\mathrm{Bern}(\varphi^{-1}),\ Q=\mathrm{Bern}(\varphi^{-2})
\) as before,
the R\'enyi divergence (in bits) admits an explicit closed form
\[
\boxed{\;
D_\alpha(P\|Q)
=\frac{1}{\alpha-1}\log_2\!\Big(\varphi^{\alpha-2}+\varphi^{-\alpha-1}\Big)
=\frac{1}{\alpha-1}\log_2\!\Big(\varphi^{-\alpha-1}(1+\varphi^{2\alpha-1})\Big).\;}
\]
Moreover, the endpoint values are all in closed form:
\[
\boxed{\;
D_{1/2}(P\|Q)=3\log_2\varphi-2,\quad
D_{2}(P\|Q)=\log_2(1+\varphi^{-3}),\quad
D_{\infty}(P\|Q)=\log_2\varphi.\;}
\]
Thus the ``all-order distinguishability spectrum'' of the complementary golden pair is analytically controlled entirely by the single algebraic number \(\varphi\).
\end{proposition}

\begin{proof}
The definition of R\'enyi divergence yields
\[
D_\alpha(P\|Q)=\frac{1}{\alpha-1}\log_2\!\Big(p^\alpha q^{1-\alpha}+q^\alpha p^{1-\alpha}\Big),
\qquad p=\varphi^{-1},\ q=\varphi^{-2}.
\]
Direct simplification of the exponents gives
\(
p^\alpha q^{1-\alpha}=\varphi^{\alpha-2}
\) and
\(
q^\alpha p^{1-\alpha}=\varphi^{-\alpha-1}
\),
yielding the all-order closed form.
The endpoint values follow by substituting \(\alpha=\tfrac12,2\); and \(D_\infty(P\|Q)=\log_2\max_x \frac{P(x)}{Q(x)}=\log_2(p/q)=\log_2\varphi\). \qed
\end{proof}

\subsubsection{Posterior Discretization and Finite-Window Audit: \texorpdfstring{$\varphi$}{phi}-Logistic, Explicit Error, and Hitting Time}

\begin{proposition}[Posterior Discretization on the \texorpdfstring{$\varphi$}{phi}-Power Grid: \texorpdfstring{$\varphi$}{phi}-Logistic]\label{prop:pom-phi-logistic-posterior-grid}
Consider two-phase hypotheses \(\Theta\in\{\theta_+,\theta_-\}\) with equal priors, where under \(\theta_+\),
\(
B_i\stackrel{\mathrm{iid}}{\sim}\mathrm{Bern}(\varphi^{-1})
\),
and under \(\theta_-\),
\(
B_i\stackrel{\mathrm{iid}}{\sim}\mathrm{Bern}(\varphi^{-2})
\).
Let \(S_n:=\sum_{i=1}^{n}B_i\), and define the integer
\[
K_n:=2S_n-n\in\{-n,-n+2,\dots,n\}.
\]
Then the posterior odds satisfy a strict power law
\[
\boxed{\;
\frac{\mathbb{P}(\theta_+\mid B_1^n)}{\mathbb{P}(\theta_-\mid B_1^n)}=\varphi^{K_n}\;}
\]
and consequently the posterior probabilities form a discrete \(\varphi\)-logistic family:
\[
\boxed{\;
\mathbb{P}(\theta_+\mid B_1^n)=\frac{1}{1+\varphi^{-K_n}},\qquad
\mathbb{P}(\theta_-\mid B_1^n)=\frac{1}{1+\varphi^{K_n}}.\;}
\]
Therefore the ``achievable posterior set'' is precisely
\[
\Bigl\{(1+\varphi^{-k})^{-1}:\ k\equiv n\ (\mathrm{mod}\ 2),\ |k|\le n\Bigr\},
\]
exhibiting an auditable discretization on the \(\varphi\)-power grid rather than a continuous manifold.
\end{proposition}

\begin{proof}
By Bayes' formula and independence,
\[
\frac{\mathbb{P}(\theta_+\mid B_1^n)}{\mathbb{P}(\theta_-\mid B_1^n)}
=\frac{\mathbb{P}(B_1^n\mid\theta_+)}{\mathbb{P}(B_1^n\mid\theta_-)}
=\left(\frac{p}{1-p}\right)^{2S_n-n},
\qquad p=\varphi^{-1}.
\]
Since \(\frac{p}{1-p}=\frac{\varphi^{-1}}{\varphi^{-2}}=\varphi\), the odds equal \(\varphi^{K_n}\); converting to logistic form yields the result.
The achievable set is directly given by the range of \(K_n\). \qed
\end{proof}

\begin{theorem}[Explicit Formula and Sharp Asymptotic Prefactor for the Finite-Sample Optimal Error]\label{thm:pom-bayes-error-explicit-sharp-asymptotic}
In the two-phase model of Proposition~\ref{prop:pom-phi-logistic-posterior-grid}, the optimal Bayes decision under equal priors is majority vote (the \(\varphi\)-limit regime of Proposition~\ref{prop:pom-max-fiber-even-majority-vote}).
Its optimal error probability satisfies the fully explicit summation formula
\[
\boxed{\;
P_e(n)=\sum_{j=0}^{\lfloor n/2\rfloor}\binom{n}{j}\,\varphi^{-j}\,\varphi^{-2(n-j)}
=\sum_{j=0}^{\lfloor n/2\rfloor}\binom{n}{j}\,\varphi^{-2n+j}.\;}
\]
Furthermore, as \(n\to\infty\) there exists a sharp asymptotic (with explicit prefactor)
\[
\boxed{\;
P_e(n)=\varphi^{2}\sqrt{\frac{2}{\pi n}}\,(2\varphi^{-3/2})^{\,n}\,(1+o(1)).\;}
\]
This formula elevates the finite-window audit for two-phase identification from the level of exponential rates to a precision level that can be directly verified numerically.
\end{theorem}

\begin{proof}
Under \(\theta_+\), the misclassification event is \(S_n\le \lfloor n/2\rfloor\);
by the complementary symmetry (swapping \(\theta_+\) and \(\theta_-\) is equivalent to flipping \(B\)), the error probabilities under the two phases are equal, so the Bayes error is this binomial tail sum.
Substituting \(p=\varphi^{-1}\), \(q=\varphi^{-2}\) yields the explicit summation formula.
\par
For the sharp asymptotic, assume \(n\) even and set the boundary term
\[
T_0:=\binom{n}{n/2}(pq)^{n/2}=\binom{n}{n/2}\varphi^{-3n/2}.
\]
Let \(T_r:=\binom{n}{n/2-r}p^{n/2-r}q^{n/2+r}\) (\(r\ge 0\)), then
\[
\frac{T_{r+1}}{T_r}
=\frac{n/2-r}{n/2+r+1}\cdot\frac{q}{p}
\le \frac{q}{p}=\varphi^{-1},
\]
and for each fixed \(r\), the product correction of these ratios tends to \(1\) as \(n\to\infty\).
Therefore
\[
\frac{P_e(n)}{T_0}
=\sum_{r\ge 0}\frac{T_r}{T_0}
\xrightarrow[n\to\infty]{}
\sum_{r\ge 0}\left(\frac{q}{p}\right)^r
=\frac{1}{1-q/p}
=\frac{p}{p-q}
=\varphi^2,
\]
where the interchange of limit and summation is justified by the dominated convergence theorem (\(\frac{T_r}{T_0}\le (\frac{q}{p})^r\)).
\par
Finally, by the central binomial coefficient asymptotic
\(
\binom{n}{n/2}\sim 2^n\sqrt{\frac{2}{\pi n}}
\),
we obtain
\(
T_0\sim \sqrt{\frac{2}{\pi n}}(2\varphi^{-3/2})^{n}
\);
combining yields the stated prefactor.
The case of odd \(n\) requires only changing the boundary term to \(j=\lfloor n/2\rfloor\) and follows by the same argument. \qed
\end{proof}

\begin{proposition}[Confidence Threshold Hitting Time: \texorpdfstring{$\varphi$}{phi}-Threshold Law and Closed-Form Expected Sample Size]\label{prop:pom-confidence-hitting-time-phi-threshold}
Under the setting of Proposition~\ref{prop:pom-phi-logistic-posterior-grid}, define the posterior error
\[
\varepsilon_n:=\min\{\mathbb{P}(\theta_+\mid B_1^n),\ \mathbb{P}(\theta_-\mid B_1^n)\}
=\frac{1}{1+\varphi^{|K_n|}}.
\]
Given a target \(\epsilon\in(0,1/2)\), let the stopping time be
\[
\tau_\epsilon:=\inf\{n\ge 1:\ \varepsilon_n\le \epsilon\}.
\]
Then the stopping condition is equivalent to the integer threshold
\[
\boxed{\;
|K_n|\ge T_\epsilon,\qquad
T_\epsilon:=\left\lceil\log_\varphi\!\Big(\frac{1-\epsilon}{\epsilon}\Big)\right\rceil.\;}
\]
Moreover, under the true phase \(\theta_+\) (or \(\theta_-\)), \(K_n=\sum_{i=1}^n(2B_i-1)\) is a \(\pm 1\) random walk with drift
\[
\mathbb{E}[K_n]=(\varphi^{-1}-\varphi^{-2})n=\varphi^{-3}n,
\]
and hence as \(\epsilon\downarrow 0\) (equivalently \(T_\epsilon\to\infty\)),
\[
\boxed{\;
\mathbb{E}[\tau_\epsilon]=\varphi^{3}\,T_\epsilon+O(1)
=\frac{\varphi^{3}}{\log\varphi}\log\!\Big(\frac{1-\epsilon}{\epsilon}\Big)+O(1).\;}
\]
\end{proposition}

\begin{proof}
By Proposition~\ref{prop:pom-phi-logistic-posterior-grid}, \(\varepsilon_n=\min\{(1+\varphi^{-K_n})^{-1},(1+\varphi^{K_n})^{-1}\}=(1+\varphi^{|K_n|})^{-1}\).
Therefore \(\varepsilon_n\le \epsilon\) is equivalent to \(\varphi^{|K_n|}\ge (1-\epsilon)/\epsilon\), i.e., \(|K_n|\ge T_\epsilon\).
\par
Under \(\theta_+\), the increments \(X_i:=2B_i-1\in\{\pm 1\}\) are i.i.d.\ with
\(
\mathbb{E}[X_i]=p-q=\varphi^{-3}
\).
Let \(\tau:=\tau_\epsilon\).
Since \(|K_\tau|\in[T_\epsilon,T_\epsilon+1]\) (the step size is \(1\)), and \(\mathbb{P}(K_\tau\le -T_\epsilon)\) decays exponentially as \((q/p)^{T_\epsilon}=\varphi^{-T_\epsilon}\),
we obtain \(\mathbb{E}[K_\tau]=T_\epsilon+O(1)\).
Applying optional stopping to the martingale \(M_n:=K_n-\varphi^{-3}n\) and taking expectations yields
\(
\mathbb{E}[K_\tau]=\varphi^{-3}\mathbb{E}[\tau]
\),
whence \(\mathbb{E}[\tau]=\varphi^{3}T_\epsilon+O(1)\).
Finally, rewriting via
\(
T_\epsilon=\frac{1}{\log\varphi}\log\bigl(\frac{1-\epsilon}{\epsilon}\bigr)+O(1)
\)
yields the result. \qed
\end{proof}

\subsubsection{Geometric--Statistical Bridge and Top-Spectrum Rigidity: Unification of \texorpdfstring{$C_\varphi$}{Cphi}}

\begin{corollary}[Unification of Maximum/Average Fiber Concentration Exponent and Two-Phase Distinguishability Constant]\label{cor:pom-D-over-avg-exponent-equals-chernoff}
Let \(D_m\) denote the maximum fiber multiplicity (Definition~\ref{def:pom-top-fiber-spectrum}), and let the average fiber multiplicity be
\[
\overline d_m:=\frac{|\Omega_m|}{|X_m|}=\frac{2^m}{F_{m+2}}.
\]
Denote the Chernoff constant (in bits) for two-phase identification by
\(
C_\varphi=\frac32\log_2\varphi-1
\)
(Theorem~\ref{thm:pom-max-fiber-even-chernoff-exponent}).
Then at the stable scale there is a strict unification:
\[
\boxed{\;
\lim_{m\to\infty}\frac{1}{m}\log_2\frac{D_m}{\overline d_m}=C_\varphi.\;}
\]
Therefore the ``worst-case/average concentration exponent on the geometric side'' and the ``optimal distinguishability exponent on the statistical side'' share the same algebraic constant.
\end{corollary}

\begin{proof}
From the closed form in Corollary~\ref{cor:pom-D-rec} and \(F_{m+2}\sim \varphi^{m}/\sqrt5\), one obtains
\(
\frac{D_m}{\overline d_m}
=\Theta((\varphi^{3/2}/2)^m)
\)
(see also Remark~\ref{rem:pom-defect-gap} for the natural logarithm version), whence
\[
\lim_{m\to\infty}\frac{1}{m}\log_2\frac{D_m}{\overline d_m}
=\log_2\!\left(\frac{\varphi^{3/2}}{2}\right)
=\frac32\log_2\varphi-1
=C_\varphi.
\]
\qed
\end{proof}

\begin{corollary}[Parity Limit Constants of the Top Spectral Gap]\label{cor:pom-top-gap-limit-constants}
Let \(D_m^{(2)}\) denote the second-largest fiber multiplicity (Definition~\ref{def:pom-top-fiber-spectrum}), and define the top relative gap
\[
G_m:=1-\frac{D_m^{(2)}}{D_m}.
\]
Then there are parity limit constants (Corollary~\ref{cor:pom-second-max-gap-constant}):
\[
\boxed{\;
\lim_{k\to\infty}G_{2k}=\varphi^{-6},\qquad
\lim_{k\to\infty}G_{2k+1}=\frac{1}{2}\varphi^{-5}.\;}
\]
\end{corollary}

\begin{proof}
By Corollary~\ref{cor:pom-second-max-gap-constant}, the parity limits of \(D_m^{(2)}/D_m\) have been established (with exponential convergence); rewriting directly yields the stated gap limit constants. \qed
\end{proof}

\begin{proposition}[Effective Collapse Threshold of the Odd Four-Phase Structure under \(q\)-Moment Weighting: Explicit Selection Law]\label{prop:pom-odd-fourphase-q-collapse-threshold}
In the odd stable phase \(m=2k+1\ge 13\) with four-phase structure (Remark~\ref{rem:pom-max-fiber-achievers-bsplit}), let the maximum branch of the biased two phases under the \(b\)-split regime be
\[
A_k:=F_{k+1}+F_{k-2},
\]
and the maximum branch of the balanced two phases be
\[
B_k:=F_{k+1},
\]
with ratio \(R_k:=A_k/B_k\to 1+\varphi^{-3}\).
For any tolerance \(\delta\in(0,1)\), if one compares the two types of phase contributions via \(q\)-moment weighting, then when
\[
\boxed{\;
q\ge q_\delta:=\left\lceil\frac{\log(1/\delta)}{\log(1+\varphi^{-3})}\right\rceil\;}
\]
the biased two phases account for at least a fraction \(1-\delta\) of the total contribution to that \(q\)-moment.
Consequently, the phenomenon of ``effective collapse from four phases to two phases within the maximum achiever family'' is precisely characterized by the single closed-form threshold \(q_\delta\).
\end{proposition}

\begin{proof}
Within the maximum-phase family, there are two balanced and two biased phases.
Coarsening by the \(q\)-th power of the maximum branch, the biased fraction is
\[
\frac{2A_k^q}{2A_k^q+2B_k^q}=\frac{1}{1+R_k^{-q}}.
\]
When \(q\ge \frac{\log(1/\delta)}{\log R_k}\), we have \(R_k^{-q}\le \delta\), so the fraction \(\ge \frac{1}{1+\delta}\ge 1-\delta\).
Using \(R_k\to 1+\varphi^{-3}\) yields the asymptotic closed form for the threshold, and taking the ceiling gives \(q_\delta\). \qed
\end{proof}

\begin{proposition}[Spectral Decomposition Closed Form of the Maximum Fiber Sequence: Four-Periodic Sub-Leading Oscillation]\label{prop:pom-D-spectral-decomposition-fourperiod}
The maximum fiber sequence \((D_m)\) satisfies the two-step recurrence \(D_m=D_{m-2}+D_{m-4}\) (Corollary~\ref{cor:pom-D-rec}; with the convention \(D_0:=1,D_1:=2\)).
Its characteristic equation is \(\lambda^4-\lambda^2-1=0\), with roots \(\pm\sqrt{\varphi}\) and \(\pm \mathrm{i}/\sqrt{\varphi}\).
Therefore there exist real constants \(a,b,c,d\) such that for all \(m\ge 0\) the spectral decomposition closed form holds:
\[
\boxed{\;
D_m=a(\sqrt{\varphi})^{m}+b(-\sqrt{\varphi})^{m}
\;+\;\varphi^{-m/2}\Bigl(c\cos\!\Bigl(\frac{\pi m}{2}\Bigr)+d\sin\!\Bigl(\frac{\pi m}{2}\Bigr)\Bigr).\;}
\]
The third term gives a strictly four-periodic sub-leading oscillation whose amplitude decays exponentially as \(\varphi^{-m/2}\).
\end{proposition}

\begin{proof}
By the general solution structure of the recurrence, there exist constants \(A_+,A_-,B_+,B_-\in\CC\) such that
\[
D_m=A_+(\sqrt{\varphi})^m+A_-(-\sqrt{\varphi})^m
+B_+\left(\frac{\mathrm{i}}{\sqrt{\varphi}}\right)^m
+B_-\left(\frac{-\mathrm{i}}{\sqrt{\varphi}}\right)^m.
\]
Since \(D_m\in\RR\), the complex root terms must form a conjugate pair; rewriting as cosine/sine combinations yields
\[
B_+\left(\frac{\mathrm{i}}{\sqrt{\varphi}}\right)^m
+\overline{B_+}\left(\frac{-\mathrm{i}}{\sqrt{\varphi}}\right)^m
=\varphi^{-m/2}\Bigl(c\cos(\tfrac{\pi m}{2})+d\sin(\tfrac{\pi m}{2})\Bigr),
\]
where \(c,d\in\RR\).
Renaming \(A_\pm\) as \(a,b\) yields the conclusion. \qed
\end{proof}
