\paragraph{Fiber topological dichotomy, order-lattice complete representation, and unified certificate chain for modular-axis sheaves/horizon positivity: from \texorpdfstring{$\tau(x)$}{tau(x)} to Chebotarev--Plancherel and Toeplitz--PSD}
Under the fixed fold--project--twist interface of this paper, the topological/order/statistical structures on the fiber side and the twisted spectral gap, profinite limits, and horizon positivity on the congruence-axis-sheaf side can be compressed into a finite-dimensional, integer-coefficient, auditable certificate chain: the homotopy type and lattice structure of fibers are completely determined by the component-length multiset \(\{\ell_i\}\); worst fibers and collision gaps are quantitatively driven by non-trivial Fourier energy; modulo-\(q\) equidistribution, additive-energy leading term, and profinite Haar dichotomy are governed by the unified spectral gap of twisted Perron roots; at the horizon interface, Carath\'eodory/Herglotz positivity further equates to a hierarchy of Toeplitz--PSD levels, landing infinite-object positivity discrimination as level-by-level verifiable finite-dimensional feasibility constraints.
\begin{enumerate}
  \item \textbf{``Sphere--contractible'' dichotomy rigidity of fiber independence complex fully parametrized by a single integer \texorpdfstring{$\tau(x)$}{tau(x)}.}
  For any stable type \(x\in X_m\), the path-component lengths \((\ell_i)\) of the candidate-site induced graph \(\Gamma_x\) yield the closed-form parameter
  \(\tau(x)=\sum_i\lfloor(\ell_i+1)/3\rfloor\), and force the homotopy type of the independence complex \(|K_x|\) to satisfy a strict dichotomy: either contractible or homotopy equivalent to a single sphere \(S^{\tau(x)-1}\); in the non-contractible case, reduced homology is \(\ZZ\) in exactly one dimension (Theorem~\ref{thm:pom-fiber-independence-complex-classification}; Corollary~\ref{cor:pom-fiber-independence-complex-homology}).

  \item \textbf{Exponential sparsity of high-multiplicity fibers: the set of types achieving a fixed proportion of worst value must be exponentially thin.}
  Let \(D_m=\max_{x\in X_m}d_m(x)\), and for \(\alpha\in(0,1]\) define
  \(S_\alpha(m)=\{x:\ d_m(x)\ge \alpha D_m\}\).
  Then
  \[
  |S_\alpha(m)|\le \frac{2^m}{\alpha D_m},
  \qquad
  \frac{|S_\alpha(m)|}{|X_m|}=O\!\left(\frac{1}{\alpha}\left(\frac{2}{\varphi^{3/2}}\right)^m\right)
  \]
  (Corollary~\ref{cor:pom-high-multiplicity-sparse}).
  Simultaneously, the exponential laws and parity-multiplicative constants for worst/average amplification ratios can be fully explicitized (Remark~\ref{rem:pom-defect-gap}), so ``extreme space loss'' and ``sparsity of its support set'' can be audited in parallel.

  \item \textbf{Interior lattice structure of fibers has Birkhoff complete representation: sawtooth ideal-lattice model yields full classification of join/meet and interval structure.}
  Under the natural partial order of ``single inverse-substitution toggle reachability,'' each fiber \(F_x\) has a canonical distributive lattice structure \(\mathcal L_x\), with an isomorphism
  \[
  \mathcal L_x\ \cong\ J\!\Big(\bigsqcup_{i=1}^r Z_{\ell_i}\Big),
  \]
  where \(Z_{\ell}\) is the fence (sawtooth) poset, \(J(P)\) the ideal lattice; the join-irreducible-element poset is isomorphic to \(\bigsqcup_i Z_{\ell_i}\), so \(\{\ell_i\}\) completely determines the fiber's conjunction/disjunction and interval structure (Theorem~\ref{thm:pom-fiber-birkhoff-fence-ideal-lattice}).

  \item \textbf{Maximal fiber and collision gap quantitatively driven by non-trivial Fourier energy, yielding constant-level bias lower bound.}
  Let \(d_m(r)\) be the multiplicity profile modulo \(F_{m+2}\), \(\widehat d_m(k)\) its Fourier amplitudes, \(\overline d_m=2^m/F_{m+2}\) the mean, and \(M_m=\max_r d_m(r)\).
  Then
  \[
  M_m\ge \overline d_m+\frac{1}{F_{m+2}}\Bigl(\sum_{k\ne 0}|\widehat d_m(k)|^2\Bigr)^{1/2},
  \qquad
  M_m\ge \overline d_m+\frac{1}{F_{m+2}}\max_{k\ne 0}|\widehat d_m(k)|
  \]
  (Theorems~\ref{thm:fold-max-fiber-spectrum},~\ref{thm:fold-max-fiber-fourier}).
  The fold collision probability satisfies
  \(
  \mathsf{Col}_m\ge M_m^2/2^{2m}
  \)
  (Theorem~\ref{thm:fold-collision-max-fiber}).
  Furthermore, limitizing the critical resonance constant \(C_\varphi=\prod_{n\ge 2}|\cos(\pi\varphi^{-n})|\) (Theorem~\ref{thm:fold-critical-resonance-constant}) yields the constant-level bias lower bound
  \(\liminf_{m\to\infty}M_m/\overline d_m\ge 1+C_\varphi\) (Corollary~\ref{cor:fold-max-fiber-constant-bias}).

  \item \textbf{Geometric concentration exponent and statistical optimal-separability exponent strictly unified to a single algebraic constant.}
  The maximal/average fiber concentration exponent satisfies
  \[
  \boxed{\ \lim_{m\to\infty}\frac{1}{m}\log_2\frac{D_m}{\overline d_m}=C_\varphi^{(\mathrm{Ch})}:=\frac32\log_2\varphi-1\ }
  \]
  (Corollary~\ref{cor:pom-D-over-avg-exponent-equals-chernoff}), strictly aligning the ``worst/average concentration'' geometric exponent with the Chernoff constant for two-phase discrimination at the exponential scale.

  \item \textbf{Twisted primitive extraction has Möbius--Adams synchrony: time divisors strongly couple to \texorpdfstring{$(\chi,u)$}{(chi,u)} scaling.}
  In the twisted \(\zeta\) framework with character axis \(\chi\) and energy parameter \(u\), primitive extraction satisfies the Möbius formula with scaling:
  \[
  B_n(\chi;u)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}(\chi^d;u^d),
  \]
  with \((\chi,u)\mapsto(\chi^d,u^d)\) appearing synchronously (Proposition~\ref{prop:kernel-chebotarev-mobius-adams}; see also Remark~\ref{rem:kernel-chebotarev-mobius-adams-meaning} and Proposition~\ref{prop:witt-renorm-monoid} for semigroup formulation).

  \item \textbf{Chebotarev--Plancherel energy identity and quotient monotonicity: space-side \texorpdfstring{$L^2$}{L2} deviation precisely equals character-side energy sum.}
  In finite abelian group-label stratification, the mean-square deviation satisfies the exact identity
  \(
  \sum_{c\in G}|\varepsilon_n(c;u)|^2=\frac{1}{|G|}\sum_{\chi\ne 1}|B_n(\chi;u)|^2
  \)
  (Corollary~\ref{cor:kernel-chebotarev-plancherel-energy}).
  In the general finite-group caliber, this upgrades to a Parseval/Plancherel identity on the class-function space and yields deviation monotonicity for quotient maps (Proposition~\ref{prop:kernel-chebotarev-plancherel-nonabelian}).

  \item \textbf{Twisted spectral gap \texorpdfstring{$\Rightarrow$}{=>} exponential \texorpdfstring{$\bmod q$}{mod q} equidistribution and additive-energy leading term: error uniformly dominated by worst twisted Perron root.}
  Under the modulo-\(q\) twisted Perron spectral-gap condition, residue-class counts have exponential error terms (Theorem~\ref{thm:gm-modq-equidistribution-spectral-gap}), and the modular additive energy satisfies
  \[
  E_m(q)=\frac{|A_m|^4}{q}+O\!\bigl(q\,(\rho_q^\ast)^{4m}\bigr)
  \]
  (Proposition~\ref{prop:gm-modq-additive-energy-main-error}), unifying equidistribution and energy estimates into a single computable spectral chain.

  \item \textbf{Profinite Haar dichotomy: either Haar limit, or support degenerates to proper closed-subgroup coset.}
  From the compatibility of counting on finite quotients \(\ZZ/q\ZZ\), one constructs a measure sequence \(\mu_m\) on \(\widehat{\ZZ}\); if non-degenerate spectral gap holds for each fixed \(q\ge 2\), then \(\mu_m\Rightarrow \mu_{\mathrm{Haar}}\);
  conversely, if equi-radius resonance exists for some \(q\), then \(A_m\bmod q\) eventually lands in a proper subgroup coset, inducing support degeneration to a closed-subgroup coset on \(\widehat{\ZZ}\) (Theorem~\ref{thm:gm-profinite-haar-dichotomy}).

  \item \textbf{Major-arc concentration \texorpdfstring{$\Longleftrightarrow$}{<->} support degeneration: bidirectional criterion for rational-frequency resonance and profinite non-Haar.}
  Constant-level exponential sums at major-arc endpoints (major-arc concentration) occur if and only if there exists a cyclotomic obstruction/resonant character (Theorem~\ref{thm:gm-majorarc-energy-obstruction-loop}), which is equivalent to non-Haar coset-support degeneration on \(\widehat{\ZZ}\) (Theorem~\ref{thm:gm-profinite-haar-dichotomy}).
  Thus ``large rational-frequency spectrum value'' and ``arithmetic support degeneration'' are mutually executable criteria under the same finite-dimensional spectral-certificate caliber.

  \item \textbf{Profinite character diagonalization and Hilbert--Artin decomposition: worst twisted spectral radius exactly read out from operator restriction.}
  On \(\mathcal{H}=\CC^V\otimes L^2(\widehat{\ZZ})\), the group-extension adjacency operator \(\mathcal{M}_u\) decomposes by character direct sum into finite-dimensional twisted blocks \(M_\chi(u)\), satisfying
  \[
  \rho(\mathcal{M}_u|_{\mathcal{H}_\perp})=\sup_{\chi\ne 1}\rho(M_\chi(u)),
  \qquad
  \rho(\mathcal{M}_u|_{\mathcal{H}_0})=\rho(M_1(u))=:\lambda_1(u),
  \]
  elevating ``worst twisted spectral radius'' to an operator readout (Proposition~\ref{prop:pom-profinite-artin-hilbert}).

  \item \textbf{Finite certificatization of horizon Toeplitz--PSD: Carath\'eodory positivity \texorpdfstring{$\Leftrightarrow$}{<->} PSD hierarchy, and cofinal-subsequence verification equivalent to full hierarchy.}
  Positivity of the horizon Carath\'eodory logarithmic derivative is equivalent to the Toeplitz principal-submatrix family \(\{\mathbf{T}_N\succeq 0\}_{N\ge 0}\), and in the \(\zeta\) case equivalent to RH (Theorem~\ref{thm:app-horizon-toeplitz-lmi}).
  Furthermore, full-hierarchy verification of Toeplitz--PSD is strictly equivalent to verification on any cofinal subsequence (e.g., \(N_k=2^k\)), providing a fixed-order embeddable sparsification/approximation mechanism (Theorem~\ref{thm:xi-toeplitz-psd-cofinal-sparsification}).
\end{enumerate}
