\paragraph{Cross-interface terminal conclusions: low-order jet of negative-carrying-potential pressure, exponential-type rigidity of scale-superposition completion, and single-scale Jensen tropical profile}
This paragraph supplements several "directly auditable constants/obstructions" items: the first writes the low-order cumulants of the negative-carrying-potential pressure function at the origin as explicit rational constants,
and gives closed forms and boundary asymptotics of the corresponding large-deviation rate function at endpoints; the second locks the exponential type of scale-superposition completion on Hardy slices to
"bandwidth \(\times\) support diameter," and translates the \(T\log T\) zero-point statistics target into an unavoidable complexity product lower bound via Cartwright density estimation;
the third gives piecewise-affine closed form for the Jensen--tropical profile of single-scale completion, and strictly atomizes "off-line spectral measure" in the real-part direction;
finally records the common imaginary period obstruction of \(\varphi^{\mathbb Q}\) scale mixing.

\begin{conclusion}[Low-order cumulant rigidity of negative-carrying-potential pressure function: \texorpdfstring{$P'(0),P''(0),P^{(3)}(0)$}{P'(0),P''(0),P^{(3)}(0)} are explicit rational constants]\label{con:xi-terminal-phi-minus-pressure-loworder-cumulants}
In the negative-carrying-potential instance of Appendix \ref{app:vector-potential}, let \(u=e^{\theta}>0\), and let \(\lambda(u)\) be the Perron branch real root of the cubic equation
\[
\lambda^{3}-(u+2)\lambda^{2}+(u-2)\lambda+3u=0
\]
on \(u>0\).
Define pressure
\(
P(\theta):=\log\lambda(e^{\theta})
\).
Then the first three derivatives of \(P\) at \(\theta=0\) are
\[
P'(0)=\frac18,\qquad
P''(0)=\frac{25}{128},\qquad
P^{(3)}(0)=\frac{177}{512}.
\]
\end{conclusion}
\begin{proof}[Proof]
Let \(F(\lambda,u):=\lambda^{3}-(u+2)\lambda^{2}+(u-2)\lambda+3u\).
When \(u=1\)
\(
F(\lambda,1)=(\lambda-3)(\lambda^{2}-1)
\), so the Perron root is \(\lambda(1)=3\), thus \(P(0)=\log 3\).
\par
By implicit function differentiation
\[
\lambda_u=-\frac{F_u}{F_\lambda},
\qquad
\lambda_{uu}=-\frac{F_{uu}+2F_{u\lambda}\lambda_u+F_{\lambda\lambda}\lambda_u^{2}}{F_\lambda},
\]
and differentiating the identity
\(
F_\lambda\lambda_{uu}+F_{\lambda\lambda}\lambda_u^{2}+2F_{u\lambda}\lambda_u=0
\)
once more yields the expression for \(\lambda_{uuu}\) (all valid where \(F_\lambda\neq 0\)).
Substituting the partial derivatives
\[
F_u=-(\lambda^{2}-\lambda-3),\quad
F_\lambda=3\lambda^{2}-2(u+2)\lambda+(u-2),
\]
\[
F_{u\lambda}=-2\lambda+1,\quad
F_{\lambda\lambda}=6\lambda-2(u+2),\quad
F_{uu}=0
\]
and simplifying at \((u,\lambda)=(1,3)\) yields
\(
\lambda_u(1)=3/8
\),
\(
\lambda_{uu}(1)=33/128
\),
\(
\lambda_{uuu}(1)=117/1024
\).
\par
Also since \(u=e^{\theta}\), by the chain rule
\[
\frac{d\lambda}{d\theta}=u\lambda_u,\quad
\frac{d^{2}\lambda}{d\theta^{2}}=u\lambda_u+u^{2}\lambda_{uu},\quad
\frac{d^{3}\lambda}{d\theta^{3}}=u\lambda_u+3u^{2}\lambda_{uu}+u^{3}\lambda_{uuu},
\]
and using the identities for \(\,P=\log\lambda\)
\[
P'=\frac{\lambda'}{\lambda},\qquad
P''=\frac{\lambda''}{\lambda}-\Bigl(\frac{\lambda'}{\lambda}\Bigr)^{2},\qquad
P^{(3)}=\frac{\lambda^{(3)}}{\lambda}-3\frac{\lambda''\lambda'}{\lambda^{2}}+2\Bigl(\frac{\lambda'}{\lambda}\Bigr)^{3}
\]
(where primes denote differentiation with respect to \(\theta\)), substituting at \(\theta=0\) (i.e., \(u=1,\lambda=3\)) yields the stated three values. Proof complete.
\end{proof}

\begin{conclusion}[Endpoint constants and boundary asymptotics of negative-carrying-potential large-deviation rate function: \texorpdfstring{$I(0),I(1)$}{I(0),I(1)} closed forms and logarithmic terms at both ends]\label{con:xi-terminal-phi-minus-rate-endpoints-boundary-asymptotics}
Continue with \(P(\theta)\) from Conclusion \ref{con:xi-terminal-phi-minus-pressure-loworder-cumulants}, and let \(P(0)=\log 3\).
Define rate function
\[
I(\alpha):=\sup_{\theta\in\RR}\bigl(\theta\alpha-(P(\theta)-P(0))\bigr).
\]
Then \(I\) extends continuously on \([0,1]\), and satisfies:
\begin{enumerate}
  \item \(I\) achieves its global minimum at the unique point \(\alpha_{0}=P'(0)=\frac18\) with \(I(\alpha_{0})=0\);
  \item Endpoint values are
  \[
  I(0)=\log\Bigl(\frac{3}{1+\sqrt3}\Bigr),\qquad
  I(1)=\log 3;
  \]
  \item Boundary asymptotics are
  \[
  I(\alpha)=\log\Bigl(\frac{3}{1+\sqrt3}\Bigr)+\alpha\log\alpha+\bigl(\log(8+4\sqrt3)-1\bigr)\alpha+o(\alpha)
  \qquad (\alpha\downarrow 0),
  \]
  \[
  I(\alpha)=\log 3+(1-\alpha)\log(1-\alpha)-(1-\alpha)+o(1-\alpha)
  \qquad (\alpha\uparrow 1).
  \]
\end{enumerate}
\end{conclusion}
\begin{proof}[Proof]
Continuous extension and unique minimizer are standard properties of Legendre--Fenchel duality: from Conclusion \ref{con:xi-terminal-phi-minus-pressure-loworder-cumulants}'s \(P\) being analytic and strictly convex (Perron branch and Proposition \ref{prop:vector-potential-phi-minus-ldp-param} in Appendix \ref{app:vector-potential}) we know,
extremal points satisfy \(\alpha=P'(\theta)\), and have \(I(\alpha)=\theta\alpha-(P(\theta)-P(0))\);
taking \(\theta=0\) yields \(I(\alpha_{0})=0\), and strict convexity gives uniqueness.
\par
Endpoints and boundary expansions can be computed under the parameterization of Proposition \ref{prop:vector-potential-phi-minus-ldp-param}.
That proposition gives for \(\lambda\in(1+\sqrt3,\infty)\) the explicit representation
\[
u(\lambda)=\frac{\lambda(\lambda^{2}-2\lambda-2)}{\lambda^{2}-\lambda-3},
\qquad
\alpha(\lambda)=\frac{\lambda^{4}-3\lambda^{3}-3\lambda^{2}+8\lambda+6}{\lambda^{4}-2\lambda^{3}-5\lambda^{2}+12\lambda+6},
\]
as well as
\(
I(\alpha(\lambda))=\alpha(\lambda)\log u(\lambda)-\log\lambda+\log 3
\).
As \(\lambda\downarrow 1+\sqrt3\) we have \(u(\lambda)\downarrow 0\), \(\alpha(\lambda)\downarrow 0\), and
\(
\alpha(\lambda)\log u(\lambda)\to 0
\), thus
\(
I(0)=\log 3-\log(1+\sqrt3)
\);
as \(\lambda\to\infty\) we have \(u(\lambda)\to\infty\), \(\alpha(\lambda)\to 1\), and \(\log u(\lambda)-\log\lambda\to 0\), thus \(I(1)=\log 3\).
\par
Further, for \(\lambda=1+\sqrt3+\varepsilon\) (\(\varepsilon\downarrow 0\)) and \(\lambda\to\infty\) perform local expansions and substitute back into
\(
I=\alpha\log u-\log\lambda+\log 3
\) to obtain the
\(\alpha\log\alpha\) and \((1-\alpha)\log(1-\alpha)\) leading terms and stated linear coefficients at both ends. Proof complete.
\end{proof}

\begin{conclusion}[Exact exponential-type rigidity of scale-superposition completion: Hardy slice exponential type equals bandwidth times support diameter]\label{con:xi-terminal-scale-superposition-exact-type-rigidity}
Continue with Definition \ref{def:xi-scale-mellin-superposition}, and assume weight function \(w\in L^{1}([0,\infty))\) is supported on \([0,A]\).
Let kernel \(\Delta(z,u)=\sum_{j,k}a_{j,k}z^{j}u^{k}\) be a finite sum, and let
\[
m:=j-2k,\qquad
B:=\max\{|m|:\ a_{j,k}\neq 0\}.
\]
Define Hardy slice
\(
F(t):=\mathcal X_{\Delta,w}(\tfrac12+it)
\).
If there exists an extremal mode \(|m|=B\) such that
\[
x\longmapsto w(x)\sum_{j-2k=m}a_{j,k}e^{-jx/2}
\]
is not almost everywhere zero on any neighborhood \((A-\varepsilon,A]\), then \(F\) (viewed as an entire function) satisfies
\[
\mathrm{type}(F)=BA.
\]
\end{conclusion}
\begin{proof}[Proof]
By the substitution formula in Definition \ref{def:xi-scale-mellin-superposition}, at \(s=\tfrac12+it\) we have
\[
F(t)=\sum_{j,k}a_{j,k}\int_{0}^{A} w(x)\,e^{-jx/2}\,e^{-i(j-2k)tx}\,dx
=\sum_{m}\int_{0}^{A} g_m(x)\,e^{-imtx}\,dx,
\]
where
\(
g_m(x):=w(x)\sum_{j-2k=m}a_{j,k}e^{-jx/2}\in L^{1}([0,A])
\).
Each term is the Fourier transform of a compactly supported function, so by the Paley--Wiener theorem its exponential type is at most \(|m|A\), thus \(\mathrm{type}(F)\le BA\).
If for some \(|m|=B\) we have \(g_m\) not almost everywhere zero on any \((A-\varepsilon,A]\), then the Fourier transform of this term has exponential type exactly \(A\),
and the parameter scaling \(t\mapsto mt\) amplifies the type to \(|m|A=BA\); the type of the full sum is the maximum of individual types, so \(\mathrm{type}(F)=BA\). Proof complete.
\end{proof}

\begin{conclusion}[Zero density complexity product lower bound: matching \texorpdfstring{$T\log T$}{T log T} requires \texorpdfstring{$A(T)B(T)\gtrsim \tfrac12\log T$}{A(T)B(T)≳(1/2)log T}]\label{con:xi-terminal-zero-density-product-lower-bound}
Let \(\{F_T\}_{T\ge 1}\) be a family of Hardy slice entire functions,
and each \(F_T(t)=\mathcal X_{\Delta_T,w_T}(\tfrac12+it)\) satisfies the non-degeneracy condition of Conclusion \ref{con:xi-terminal-scale-superposition-exact-type-rigidity},
where \(w_T\) is supported on \([0,A(T)]\) and bandwidth is \(B(T)\).
Let
\(
\tau(T):=\mathrm{type}(F_T)=A(T)B(T)
\), and write the symmetric zero count
\[
\widetilde N_T(T):=\#\{t\in[-T,T]:F_T(t)=0\}.
\]
Then the Cartwright density upper bound gives
\[
\widetilde N_T(T)\le \frac{2}{\pi}\,\tau(T)\,T+o(\tau(T)T)\qquad(T\to\infty).
\]
Therefore, to achieve in magnitude the Riemann-type symmetric count \(\widetilde N_\zeta(T)\sim \frac{T}{\pi}\log T\), we must have
\[
\liminf_{T\to\infty}\frac{\tau(T)}{\log T}\ \ge\ \frac12.
\]
\end{conclusion}
\begin{proof}[Proof]
By Conclusion \ref{con:xi-terminal-scale-superposition-exact-type-rigidity}, each \(F_T\) is an entire function of exponential type \(\tau(T)\); and \(F_T\) as the Fourier transform of a compactly supported \(L^1\) function is bounded on the real axis,
thus belongs to the Cartwright class.
Applying the classical Jensen--Cartwright zero density estimate to Cartwright class exponential-type entire functions, we get
\(
\widetilde N_T(T)\le (2/\pi)\tau(T)T+o(\tau(T)T)
\).
Comparing with \(\widetilde N_\zeta(T)\sim (T/\pi)\log T\) yields the stated lower bound. Proof complete.
\end{proof}

\begin{conclusion}[Jensen--tropical profile of single-scale completion: vertical average logarithmic modulus is piecewise affine convex function]\label{con:xi-terminal-single-scale-jensen-tropical-profile}
Continue with the notation of Proposition \ref{prop:xi-rh-interval-criterion} and Corollary \ref{cor:xi-rh-unit-circle-reciprocal}:
Let
\[
\Xi_{\Delta,L}(s)=P_L(S_L(s)),\qquad S_L(s)=L^{s-\frac12}+L^{\frac12-s},
\]
and let \(d:=\deg P_L\) and
\(
\Phi_L(y):=y^{d}P_L(y+y^{-1})\in\RR[y]
\).
Write \(\{y_j\}_{j=1}^{2d}\) as the complex roots of \(\Phi_L\) (counted with multiplicity), and let the leading coefficient of \(\Phi_L\) be \(c\neq 0\).
Define vertical average profile
\[
M_L(\sigma):=\frac{\log L}{2\pi}\int_{0}^{2\pi/\log L}\log\bigl|\Xi_{\Delta,L}(\sigma+it)\bigr|\,dt.
\]
Then for all \(\sigma\in\RR\) there is the closed form
\[
M_L(\sigma)=\log|c|+\sum_{j=1}^{2d}\log\max\{L^{\sigma-\frac12},|y_j|\}-d\Bigl(\sigma-\frac12\Bigr)\log L.
\]
Thus \(M_L(\sigma)\) is a piecewise affine convex function, and all its kink points are exactly
\[
\sigma_j=\frac12+\frac{\log|y_j|}{\log L}\qquad(1\le j\le 2d).
\]
\end{conclusion}
\begin{proof}[Proof]
Let \(y:=L^{s-\frac12}\), then \(y=L^{\sigma-\frac12}e^{-it\log L}\), and
\(
\Xi_{\Delta,L}(s)=P_L(y+y^{-1})=y^{-d}\Phi_L(y)
\).
Let \(r:=L^{\sigma-\frac12}\). Making the variable substitution \(\theta=t\log L\) yields
\[
M_L(\sigma)
=-d\Bigl(\sigma-\frac12\Bigr)\log L+\frac{1}{2\pi}\int_{0}^{2\pi}\log|\Phi_L(re^{-i\theta})|\,d\theta.
\]
Applying Jensen's formula to the polynomial \(\Phi_L\):
\[
\frac{1}{2\pi}\int_{0}^{2\pi}\log|\Phi_L(re^{-i\theta})|\,d\theta
=\log|c|+\sum_{j=1}^{2d}\log\max\{r,|y_j|\},
\]
substituting back and using \(r=L^{\sigma-\frac12}\) yields the stated closed form and kink point locations. Proof complete.
\end{proof}

\begin{conclusion}[Atomicity of single-scale completion off-line spectral measure: second derivative of profile equals vertical line cluster counting measure]\label{con:xi-terminal-single-scale-offline-spectral-measure-atomicity}
Under the setting of Conclusion \ref{con:xi-terminal-single-scale-jensen-tropical-profile}, \(M_L\) is a convex function, so its distributional second derivative is a finite atomic measure:
\[
\frac{d^{2}}{d\sigma^{2}}M_L(\sigma)=\log L\sum_{j=1}^{2d}\mathrm{mult}(y_j)\,\delta_{\sigma_j},
\qquad
\sigma_j=\frac12+\frac{\log|y_j|}{\log L}.
\]
Moreover, let \(\mathcal Z\) be the zero set of \(\Xi_{\Delta,L}\) (counted with multiplicity), then for any continuous compactly supported test function \(\phi\) we have
\[
\lim_{T\to\infty}\frac{\pi}{T\log L}\sum_{\substack{s\in\mathcal Z\\ |\Im s|\le T}}\phi(\Re s)
\;=\;
\sum_{j=1}^{2d}\mathrm{mult}(y_j)\,\phi(\sigma_j),
\]
i.e., after natural normalization, the "zero point marginal distribution in real-part direction" strictly collapses to finitely many atoms, with atom locations exactly the kink points of \(M_L\).
\end{conclusion}
\begin{proof}[Proof]
By Conclusion \ref{con:xi-terminal-single-scale-jensen-tropical-profile},
the first derivative of each term \(\log\max\{L^{\sigma-\frac12},|y_j|\}\) is a step function,
whose second derivative in the distributional sense is \((\log L)\delta_{\sigma_j}\); multiply by multiplicity and sum to get the first formula.
\par
On the other hand, \(\Xi_{\Delta,L}(s)=0\iff \Phi_L(y)=0\) and \(y=L^{s-\frac12}\).
If \(y_j\) is a root of \(\Phi_L\), then the zero tower is
\[
s=\frac12+\frac{\log y_j+2\pi i k}{\log L},\qquad k\in\ZZ,
\]
all whose real parts are identically \(\sigma_j\).
Within the window \(|\Im s|\le T\) each \(y_j\) contributes \( (T\log L/\pi)+O(1)\) zeros (counted with multiplicity), so after normalization convergence to the stated atomic sum for any \(\phi\). Proof complete.
\end{proof}

\begin{conclusion}[\texorpdfstring{$\varphi^{\mathbb Q}$}{phi^Q} scale mixing resonance obstruction: finite \texorpdfstring{$\varphi^{\mathbb Q}$}{phi^Q} superposition must preserve nonzero common imaginary period]\label{con:xi-terminal-phi-q-scale-mix-common-imag-period}
Let \(L_1,\dots,L_n>1\) satisfy
\[
\frac{\log L_j}{\log\varphi}\in\QQ,\qquad j=1,\dots,n.
\]
Take the least common multiple \(q\) of denominators so that \(\log L_j=(p_j/q)\log\varphi\) (\(p_j\in\ZZ\)).
Then for
\[
T_{\varphi}:=\frac{2\pi i\,q}{\log\varphi}\neq 0
\]
we have \(L_j^{-T_{\varphi}}=1\) (\(\forall j\)).
Thus for any finite superposition formed by meromorphic function families \(\{f_j\}\) defined on \(\CC^{\times}\)
\[
F(s):=\sum_{j=1}^{n} f_j(L_j^{-s})
\]
we have strict imaginary periodicity \(F(s+T_{\varphi})\equiv F(s)\).
In particular, any completion superposition constructed solely from finitely many \(\varphi^{\mathbb Q}\) scales cannot generate aperiodic spectral structure in the vertical direction: its zero-pole set must contain an arithmetic comb of common step \(2\pi/\log\varphi\).
\end{conclusion}
\begin{proof}[Proof]
From \(\log L_j=(p_j/q)\log\varphi\) we get
\[
L_j^{-T_{\varphi}}=\exp\!\bigl(-T_{\varphi}\log L_j\bigr)
=\exp\!\bigl(-2\pi i\,q\cdot (p_j/q)\bigr)=1.
\]
Thus \(f_j(L_j^{-(s+T_{\varphi})})=f_j(L_j^{-s})\), summing over \(j\) yields \(F(s+T_{\varphi})=F(s)\). Proof complete.
\end{proof}
