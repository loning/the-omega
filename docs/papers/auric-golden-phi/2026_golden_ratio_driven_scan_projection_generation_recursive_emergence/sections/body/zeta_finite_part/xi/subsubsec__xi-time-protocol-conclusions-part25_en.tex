\paragraph{Cross-interface terminal conclusions: power escort orbits in entropy coordinates, two-dimensional defect tomography, and arithmetic phase splitting of primitives}
This paragraph compresses several auditable mechanisms that have appeared at different interfaces in the text into independently citable conclusion entries:
the first gives power escort orbit rigidity under mixed conservation constraints, promoting entropy to the unique thermodynamic coordinate;
the second organizes the boundary-visible quantities of two-dimensional point defect measure as Fourier-Laplace fingerprints and gives finite-sampling low-rank reconstruction caliber;

\begin{conclusion}[Power escort orbit rigidity under entropy constraint: entropy as unique thermodynamic coordinate]\label{con:xi-terminal-escort-entropy-thermo-coordinate}
Let \(X\) be a finite set (\(|X|=n\ge 2\)), take a full-support distribution \(w\in\Delta(X)\), and fix \(\beta\in(0,1)\).
For each \(\gamma>1\) define the power escort transformation
\[
\mathrm{Escort}_\gamma(r)(x):=\frac{r(x)^\gamma}{\sum_{y\in X}r(y)^\gamma}\qquad(r\in\Delta(X)).
\]
If there exists \((q,r)\in\Delta(X)^2\) satisfying \(w=\beta q+(1-\beta)r\) and \(q=\mathrm{Escort}_\gamma(r)\), then this solution is unique; denote it \((q_\gamma,r_\gamma)\).
Moreover the mapping \(\gamma\mapsto H(r_\gamma)\) is continuous and strictly increasing, and satisfies
\[
\lim_{\gamma\downarrow 1}r_\gamma=w.
\]
Let
\[
H_{\max}(\beta,w):=\sup\left\{H(r):\ r\in\Delta(X),\ \frac{w-(1-\beta)r}{\beta}\in\Delta(X)\right\}\in(H(w),\log n],
\]
then \(\lim_{\gamma\to\infty}H(r_\gamma)=H_{\max}(\beta,w)\).
In particular, if the uniform distribution \(U\) is feasible (equivalent to \(w(x)\ge (1-\beta)/n\) for all \(x\in X\)), then \(H_{\max}(\beta,w)=\log n\) and \(r_\gamma\to U\).
Therefore for any \(h\in(H(w),H_{\max}(\beta,w))\) there exists unique \(\gamma(h)>1\) such that \(H(r_{\gamma(h)})=h\);
under conservation constraint \(w=\beta q+(1-\beta)r\), extremal points of entropy level sets can only move along this one-dimensional power-law orbit, thus entropy can be viewed as the unique thermodynamic coordinate of this variational frontier.
\end{conclusion}
\begin{proof}[Proof sketch]
See Proposition \ref{prop:pom-microcanonical-two-temperature-escort-orbit-entropy-monotone}.
\end{proof}

\begin{conclusion}[Two-dimensional Fourier-Laplace holographic tomography: boundary fingerprint uniquely recovers bulk defect measure]\label{con:xi-terminal-defect-measure-fourier-laplace-holographic}
Let \(\nu\) be a finite positive measure on \(\RR\times(0,\infty)\), and define boundary density
\[
H_\nu(x):=\int_{\RR\times(0,\infty)}\frac{4\delta}{(x-\gamma)^2+(1+\delta)^2}\,\dd\nu(\gamma,\delta),
\qquad
H_{\nu,t}:=P_t*H_\nu,
\]
where \(P_t(x)=\frac{1}{\pi}\frac{t}{x^2+t^2}\) is the Poisson kernel on \(\RR\).
Denote \(\widehat{H}_{\nu,t}(\xi)\) as the Fourier transform, set \(s:=|\xi|\) and define the two-parameter fingerprint
\[
F(\xi,s):=e^{(t+1)s}\widehat{H}_{\nu,t}(\xi).
\]
Then there holds a strict decomposition
\[
F(\xi,s)=4\pi\int_{\RR\times(0,\infty)}\frac{\delta}{1+\delta}\,e^{-s\delta}\,e^{-\mathrm{i}\gamma\xi}\,\dd\nu(\gamma,\delta),
\]
so that \(F\) is the tensorized combination of Fourier transform in \(\gamma\) and Laplace transform in \(\delta\); in particular the mapping \(\nu\mapsto F\) is injective on the class of finite measures,
therefore the bulk defect measure \(\nu\) can be uniquely recovered from values of boundary-visible fingerprint \(F\).
\end{conclusion}
\begin{proof}[Proof sketch]
See Proposition \ref{con:discussion-defect-measure-fourier-laplace}.
\end{proof}

\begin{conclusion}[Finite-sampling recoverability of low-rank defects: bidirectional Hankel-Prony decomposition for atomic measures]\label{con:xi-terminal-atomic-defect-hankel-prony}
Under the setting of Conclusion \ref{con:xi-terminal-defect-measure-fourier-laplace-holographic}, if
\[
\nu=\sum_{j=1}^{K}a_j\,\delta_{(\gamma_j,\delta_j)}
\qquad(K\ge 1,\ a_j>0,\ (\gamma_j,\delta_j)\ \text{pairwise distinct}),
\]
then the fingerprint is an explicit finite sum
\[
F(\xi,s)=4\pi\sum_{j=1}^{K}a_j\frac{\delta_j}{1+\delta_j}\,e^{-s\delta_j}\,e^{-\mathrm{i}\gamma_j\xi}.
\]
Fix any \(\xi\neq 0\) and step size \(\Delta s>0\), for any starting point \(s_0>0\) define equispaced sampling sequence
\(
y_n:=F(\xi,s_0+n\Delta s)
\).
Then \(y_n\) is a \(K\)-term exponential sum
\[
y_n=\sum_{j=1}^{K}c_j(\xi)\,\lambda_j^{n},
\qquad
\lambda_j=e^{-\Delta s\,\delta_j},
\qquad
c_j(\xi)=4\pi a_j\frac{\delta_j}{1+\delta_j}\,e^{-s_0\delta_j}e^{-\mathrm{i}\gamma_j\xi}.
\]
Therefore when sampling length \(n\) is sufficient, the rank of the equispaced Hankel principal block is exactly \(K\), and one can uniquely recover \(\{\lambda_j\}_{j=1}^{K}\) algebraically via matrix pencil constructed from two adjacent Hankel principal blocks, thereby recovering \(\{\delta_j\}_{j=1}^{K}\).
Similarly, fixing any \(s>0\) and taking equispaced finite sampling in \(\xi\), then \(\xi\mapsto F(\xi,s)\) is a \(K\)-term Fourier sum;
under no-collision and general position conditions, one can recover \(\{\gamma_j\}_{j=1}^{K}\) by discrete Fourier/Prony type inversion.
Therefore reconstruction of atomic defects decomposes into two mutually independent low-rank spectral inversion problems; their atom count \(K\) simultaneously equals the visible Hankel rank and defect dimension certificate.
If sampling both \((\xi,s)\) axes simultaneously on a tensor grid, total sampling complexity is \(O(K^2)\).
\end{conclusion}
\begin{proof}[Proof sketch]
Fixing \(\xi\), the \(s\)-sampling sequence is an exponential sum, Hankel rank equals number of exponential terms and can recover exponential basis via Prony/matrix pencil;
fixing \(s\) is similar.
\end{proof}

\begin{conclusion}[Dimension law of primitive microcanonical entropy: \texorpdfstring{$1/2$}{1/2}-dimension in energy axis and atomic factor separation]\label{con:xi-terminal-primitive-microcanonical-dim-half}
Denote \(p_{n,k}\) as the count of primitive atoms of length \(n\) with "energy" \(k\) (see Appendix \S\ref{app:sync-kernel-weighted-abel-mertens-analytic-radius-edgeworth}), and denote
\(\sigma^2=P''(0)=11/102\) (Theorem \ref{thm:sync-kernel-even-edgeworth-c4}).
Then for any fixed \(X>0\), as \(n\to\infty\) with \(k=n/2+x\sqrt n\) (\(|x|\le X\)), there holds uniform asymptotics
\[
p_{n,k}
=\frac{1}{n}\cdot\frac{3^n}{\sqrt{2\pi\sigma^2 n}}
\exp\!\left(-\frac{x^2}{2\sigma^2}\right)\cdot\bigl(1+o(1)\bigr).
\]
Therefore microcanonical entropy satisfies
\[
\log p_{n,k}
=n\log3-\frac{3}{2}\log n-\frac{x^2}{2\sigma^2}-\frac{1}{2}\log(2\pi\sigma^2)+o(1).
\]
This expansion reveals two separable structural factors: energy distribution has \(\sqrt n\) width on visible scale (thus effective entropy dimension of energy axis is \(1/2\)),
and the prefactor \(n^{-1}\) and \(n^{-1/2}\) can be interpreted respectively as Witt/Möbius primitive atom factor and local CLT shape factor.
\end{conclusion}
\begin{proof}[Proof sketch]
Combining centered local expansion of Theorem \ref{thm:sync-kernel-even-edgeworth-c4} with
\(\sum_k p_{n,k}=3^n/n\cdot(1+o(1))\)
and rewriting in logarithmic form of Remark \ref{rem:sync-kernel-edgeworth-log-form}.
\end{proof}

\begin{conclusion}[Arithmetic phase splitting of RH-sharp: primitive suborder term locked by parity and minimal odd prime factor]\label{con:xi-terminal-rhsharp-arithmetic-phase-splitting}
In RH-sharp case where critical modular spectral phase is uniquely \(-\sqrt\lambda\) (see \S\ref{app:real-input-40-finite-rh}), the square-root-level error of primitive count \(p_n\) does not form a single "analytic phase" but undergoes strict arithmetic splitting on even-length layer:
when \(n\) is even, the \(d=1,2\) terms in Möbius inversion exactly cancel the \(\lambda^{n/2}\)-level oscillation.
Further distinguishing \(3\mid n\) vs \(3\nmid n\) gives stronger leading-suborder expansions
\[
p_n=\frac{\lambda^n}{n}-\frac{\lambda^{n/3}}{n}+O\!\left(\frac{\lambda^{n/4}}{n}\right),
\qquad n\ \text{even and}\ 3\mid n,
\]
\[
p_n=\frac{\lambda^n}{n}+(-1)^{n/2+1}\frac{\lambda^{n/4}}{n}+O\!\left(\frac{\lambda^{n/5}}{n}\right),
\qquad n\ \text{even and}\ 3\nmid n.
\]
More generally, for any odd prime factor \(p\mid n\), the \(d=p\) term in Möbius inversion necessarily contributes correction of magnitude \(\lambda^{n/p}/n\) (sign locked by \(\mu(p)=-1\)).
Let
\[
p_{\min}(n):=\min\{p:\ p\ \text{is odd prime and}\ p\mid n\},
\qquad(\text{if no odd prime factor then }p_{\min}(n)=+\infty),
\]
then on even-length layer the leading-suborder error exponent is determined by \(\max(\tfrac14,\,1/p_{\min}(n))\);
factor structure becomes a necessary coordinate for the "error phase diagram".
\end{conclusion}
\begin{proof}[Proof sketch]
See Remarks \ref{rem:finite-rh-even-strengthen} and \ref{rem:finite-rh-factor-controls-secondary}.
\end{proof}

\begin{conclusion}[Legendre entropy spectrum of time fiber: Rényi volume of Poisson kernel sealed by \texorpdfstring{$P_\nu(\cosh a)$}{P_nu(cosh a)}]\label{con:xi-terminal-timefiber-legendre-renyi-spectrum}
Let \(\Theta\sim\mathrm{Unif}[0,2\pi)\), take \(0<\varrho<1\) and define
\[
Z_{\varrho}:=\frac{1+\varrho^2+2\varrho\cos\Theta}{1-\varrho^2},
\qquad
Y_{\varrho}:=Z_{\varrho}^{-1}.
\]
Introduce hyperbolic radius parameter \(a:=2\,\operatorname{artanh}\varrho\) (i.e., \(\varrho=\tanh(a/2)\), \(\cosh a=\frac{1+\varrho^{2}}{1-\varrho^{2}}\)), then \(Z_{\varrho}=\cosh a+\sinh a\cos\Theta\), and for any \(\nu\in\CC\) there is a strict moment identity
\[
\mathbb{E}\bigl[Z_{\varrho}^{\,\nu}\bigr]=P_{\nu}(\cosh a)
\]
(Theorem \ref{thm:app-legendre-moment-timefiber}).
Therefore for any \(q\in\CC\)
\[
\mathbb{E}[Y_{\varrho}^{q}]=\mathbb{E}[Z_{\varrho}^{-q}]=P_{-q}(\cosh a),
\]
so the Rényi-type logarithmic volume spectrum of time fiber can be completely written as logarithm of Legendre function.
Moreover
\(
\mathbb{E}[Y_{\varrho}]=1
\),
\(
\mathbb{E}[\log Y_{\varrho}]=\log(1-\varrho^{2})
\)
(Theorem \ref{thm:app-endpoint-arcsine-duality}), hence logarithmic volume potential has closed form
\[
-\mathbb{E}[\log Y_{\varrho}]=-\log(1-\varrho^2)=2\log\cosh\!\left(\frac{a}{2}\right),
\]
rigidly locking hyperbolic scale \(a\) and entropy potential in the same ledger.
\end{conclusion}
\begin{proof}[Proof sketch]
See Theorems \ref{thm:app-endpoint-arcsine-duality} and \ref{thm:app-legendre-moment-timefiber}.
\end{proof}
