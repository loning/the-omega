\subsubsection{Diffusive Slow Mode of Near-Unity-Root Twists: Near-\texorpdfstring{$1$}{1} Poles, Mixing Times, and Tower Structure Limitations}\label{subsec:xi-near1-diffusive}
This subsection formulates the slowest convergence mechanism of ``unit-root twisted channels'' as an auditable diffusion law: as the modulus \(m\to\infty\) (unit root \(u=\omega_m\to 1\)), the worst-case twisted spectral radius ratio \(\eta_m\) approaches unity at a quadratic rate, inducing the natural diffusion threshold \(n\gg m^2\). On the real-input kernel, this quadratic law can further be pulled back to the near-\(1\) principal singularity trajectory in the \(s\)-plane, thereby completely locking the drift and dissipation constants to the derivative data of the pressure function.

\paragraph{Scaling perspective: the diffusion scale of the worst-case twisted slow mode}
In a twist of finite modulus \(m\) (e.g., \(u=\omega_m^r=e^{2\pi ir/m}\)), denote the worst-case nontrivial spectral radius ratio by
\[
\eta_m:=\max_{1\le r\le m-1}\frac{\rho\bigl(B(\omega_m^r)\bigr)}{\rho(B(1))}\in(0,1].
\]
Since the dual of \(\widehat{\ZZ}\) contains characters of arbitrarily high order, there necessarily exist \(m\to\infty\) with \(\omega_m\to 1\), so it is generally impossible to obtain a uniform spectral gap lower bound \(\eta_m\le 1-\delta\) (\(\delta>0\)) across all \(m\).
In the synchronization kernel weighted case, the appendix has given the near-coboundary small-angle quadratic law and its diffusion time scale: for instance, Corollary \ref{cor:sync-rho-gap-m2} and Corollary \ref{cor:sync-rho-gap-m8-diffusive} give
\[
\eta_m\approx \exp\!\Bigl(-\kappa\,(2\pi/m)^2\Bigr),
\qquad
\eta_m^n\approx \exp\!\Bigl(-\kappa(2\pi)^2\,\frac{n}{m^2}\Bigr),
\]
so that ``achieving stable equidistribution modulo \(m\)'' naturally corresponds to the diffusion threshold \(n\gg m^2\).

\paragraph{\texorpdfstring{$s$}{s}-plane diffusion law of near-\texorpdfstring{$1$}{1} poles (including drift term)}
The quadratic law above not only gives the approach rate of \(\eta_m\), but also allows the ``principal singularity of the slowest twisted channel'' to be pulled back directly to the \(s\)-plane, thereby explaining the geometric shape of the near-\(1\) pole diagram and completely locking the constants to the computable \(\nabla P(0)\), \(\nabla^2P(0)\) of the manuscript.

\begin{proposition}[Second-order pressure expansion of the purely imaginary small-angle twist and near-\texorpdfstring{$1$}{1} pole positions]\label{prop:near1-pole-diffusive-sp}
Take the pure collision potential slice of the real-input \(40\)-state kernel (along \(\varphi_2=\mathbf{1}_{\{d=2\}}\))
\[
P_2(\theta):=P(0,0,\theta)=\log\rho(M_{(0,0,\theta)}),
\qquad \lambda:=\rho(M_{(0,0,0)})=\varphi^2.
\]
Denote
\[
\mu_2:=P_2'(0)=\mathbb{E}[\varphi_2]=\partial_{\theta_2}P(0),
\qquad
\sigma_2^2:=P_2''(0)=\mathrm{Var}_\infty(\varphi_2)=\partial_{\theta_2\theta_2}^2P(0),
\]
whose numerical values are given in Appendix \ref{app:real-input-40-arity-3d} (in the expressions for \(\nabla P(0)\) and \(\nabla^2P(0)\)).
Then there exists \(\delta>0\) such that when \(t\in\RR\) and \(|t|\le \delta\), along the purely imaginary twist \(\theta=it\) there is an analytic expansion
\[
P_2(it)
:=\log\lambda+i\mu_2 t-\frac{\sigma_2^2}{2}t^2+O(t^3).
\]
Let \(t_m:=2\pi/m\) (\(m\ge 2\)) and define the \(s\)-coordinate of the twisted principal singularity
\[
s_m:=\frac{P_2(it_m)}{\log\lambda}
\qquad(\text{equivalently } \lambda^{-s_m}=\rho(M_{it_m})^{-1}),
\]
then as \(m\to\infty\),
\[
\boxed{\;
s_m
:=1+i\,\frac{\mu_2}{\log\lambda}\frac{2\pi}{m}
-\frac{\sigma_2^2}{2\log\lambda}\Bigl(\frac{2\pi}{m}\Bigr)^2
+O(m^{-3})\; }.
\]
In particular,
\[
\Re(s_m)=1-\frac{c}{m^2}+O(m^{-3}),
\qquad
\Im(s_m)=\frac{d}{m}+O(m^{-2}),
\]
where the constants are given explicitly by \((\mu_2,\sigma_2^2,\lambda)\):
\[
\boxed{\;
c=\frac{2\pi^2\sigma_2^2}{\log\lambda},
\qquad
d=\frac{2\pi\mu_2}{\log\lambda}\; }.
\]
On the present kernel, \(\lambda=\varphi^2\), \(\mu_2=\frac{3-\sqrt5}{10}\), \(\sigma_2^2=\frac{6\sqrt5-5}{125}\), whence numerically
\[
c\approx 1.38095,
\qquad
d\approx 0.498733.
\]
The corresponding reproducible prediction table is given in Table \ref{tab:real_input_40_collision_near1_pole_predictions}.
\end{proposition}

\begin{corollary}[Parabolic trajectory: second-order dissipation locks the square of first-order drift]\label{cor:near1-pole-parabola}
Following the notation of Proposition \ref{prop:near1-pole-diffusive-sp}, as \(m\to\infty\) one can eliminate \(m\) to obtain
\[
\boxed{\;
1-\Re(s_m)
:=\frac{\sigma_2^2\log\lambda}{2\mu_2^2}\,\Im(s_m)^2\ +\ o\!\bigl(\Im(s_m)^2\bigr)\; }.
\]
Therefore, near-\(1\) poles do not approach ``arbitrarily,'' but are forced by the diffusion law to lie on a parabolic neighborhood determined by \(\sigma_2^2,\mu_2,\lambda\).
\end{corollary}

\begin{corollary}[Closed-form constant for mixing time \texorpdfstring{$\tau_{\mathrm{mix}}(m)\sim\mathrm{const}\cdot m^2$}{tau_mix(m)~const*m^2}]\label{cor:near1-pole-taumix-m2}
Let \(\rho_m:=\rho(M_{it_m})\) and define the spectral ratio mixing time
\[
\tau_{\mathrm{mix}}(m):=\frac{1}{-\log(\rho_m/\lambda)}.
\]
Then from \(\Re(P_2(it)-P_2(0))=-\sigma_2^2 t^2/2+O(t^4)\),
\[
\boxed{\;
\tau_{\mathrm{mix}}(m)
:=\frac{m^2}{2\pi^2\sigma_2^2}\bigl(1+O(m^{-2})\bigr)\; }.
\]
\end{corollary}

\begin{corollary}[Negative result for Abelian towers: the uniform square-root spectral gap hypothesis fails]\label{cor:abelian-tower-no-go}
Suppose a family of congruence layers \(G_m\) contains an Abelian component \(\ZZ/m\ZZ\), so that there exists a one-dimensional unit-root twist \(u=\omega_m\to 1\)
corresponding to some integer-valued cocycle \(F\) with \(\sigma_F^2=\mathrm{Var}_\infty(F)>0\).
Then the worst-case nontrivial spectral radius ratio \(\eta_m\) satisfies
\[
\eta_m\ \ge\ \exp\!\Bigl(-\frac{\sigma_F^2}{2}(2\pi/m)^2+O(m^{-4})\Bigr)\ \xrightarrow[m\to\infty]{}\ 1,
\]
so that no \(\varepsilon>0\) exists such that \(\eta_m\le \lambda^{-1/2-\varepsilon}\) holds for all \(m\).
Therefore, the uniform spectral gap hypothesis used to derive the ``square-root level tail term'' in Theorem \ref{thm:auditable-profinite-chebotarev-finite} is incompatible with the diffusion lower bound of the near-\(1\) twisted slow mode on the profinite axis containing such Abelian modulus towers.
\end{corollary}

\begin{conjecture}[Scaled kernel version of GRH/Chebotarev (double-limit diffusion caliber)]\label{conj:scaled-kernel-grh-diffusive}
There exist constants \(c>0\) and \(\kappa>0\) such that for all moduli \(m\ge 2\) and all lengths \(n\ge 1\), the equidistribution error of any congruence cylinder event defined by a mod-\(m\) register satisfies
\[
\boxed{\ \mathrm{Err}(n,m)\ \le\ C(m)\,\exp\!\Bigl(-c\,\frac{n}{m^2}\Bigr)\ },
\]
where \(C(m)\) grows at most polynomially, and the principal exponent \(c\) is of the same order as the quadratic law constant \(\kappa\) (which, in analytically tractable kernel families, can be given by the pressure curvature/variance density).
Therefore, in the region \(m\le n^{1/2-\varepsilon}\) (for any fixed \(\varepsilon>0\)), one obtains a uniform ``square-root level visible principal term.''
\end{conjecture}
