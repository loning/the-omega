\subsubsection{Bartholdi backtracking fugacity deformation and the non-backtracking (geodesic) prime shadow of the real input kernel}\label{subsec:real-input-40-geodesic-prime-shadow}
This subsection concretizes the ``backtracking-excluded Ihara/Bartholdi prime shadow'' from Remark \ref{rem:rotation-polytope-bartholdi} on the real input \(40\)-state kernel: we construct the non-backtracking edge shift on the edge space of the essential \(20\)-state core, and obtain a surprisingly low-degree determinant closed form together with a new Abel/Mertens finite-part constant.
Furthermore, by introducing a backtracking fugacity parameter \(t\), we obtain a two-variable determinant closed form that places the standard prime shadow (\(t=1\)) and the geodesic prime shadow (\(t=0\)) on the same analytic curve.

\paragraph{Edge space and backtracking operator}
Let \(\Gamma_{\mathrm{ess}}\) be the essential directed graph of Proposition \ref{prop:real-input-40-essential-20} (with \(20\) vertices), and denote its directed edge set by \(E\).
For this kernel, \(|E|=48\) (the deterministic branch edges after subdivision by input pairs \((x,y)\); verifiable from the construction chain of the script \path{scripts/exp_real_input_40_rotation_polytope_exact.py}).
Treating each edge \(e=(u\to v)\in E\) as a letter yields the edge shift \(X_E\).

If \(\bar e:=(v\to u)\in E\) also exists, then \(\bar e\) is called the \emph{backtracking reverse edge} of \(e\).
Define the \(48\times 48\) matrix family \(H(t)\) acting on the edge space \(\RR^{E}\) (Bartholdi backtracking fugacity deformation):
\[
H(t)_{e,e'} :=
\begin{cases}
1,& \text{if } e=(u\to v),\,e'=(v\to w)\ \text{and }w\neq u,\\
t,& \text{if } e'=\bar e,\\
0,& \text{otherwise},
\end{cases}
\qquad (t\in\CC).
\]
Thus \(H(0)\) forbids immediate reversal (non-backtracking/geodesic), while \(H(1)\) is precisely the full edge-to-edge transfer (allowing reversal).
Define the two-variable \(\zeta\) and determinant denominator
\[
\zeta_{\mathrm{bt}}(z,t):=\frac{1}{\Delta_{\mathrm{bt}}(z,t)},\qquad
\Delta_{\mathrm{bt}}(z,t):=\det(I-zH(t)).
\]
and denote the non-backtracking (geodesic) \(\zeta\) at \(t=0\) by \(\zeta_{\mathrm{nb}}(z):=\zeta_{\mathrm{bt}}(z,0)\).

\begin{theorem}[Degree-15 determinant closed form for the non-backtracking (geodesic) prime shadow]\label{thm:real-input-40-geodesic-det}
Let \(H_{\mathrm{nb}}:=H(0)\). Then
\[
\zeta_{\mathrm{nb}}(z)=\frac{1}{\det(I-zH_{\mathrm{nb}})},
\]
and
\[
\boxed{
\begin{aligned}
\det(I-zH_{\mathrm{nb}})
&=1-4z^3-9z^4-12z^5-16z^6-11z^7-11z^8-4z^9-z^{10}\\
&\qquad +8z^{11}+12z^{12}+10z^{13}+8z^{14}+2z^{15}.
\end{aligned}
}
\]
Hence \(H_{\mathrm{nb}}\) has \(33\) zero eigenvalues, and the nonzero spectrum is determined by a degree-\(15\) integer-coefficient polynomial (the effective dynamical dimension is compressed from \(48\) to \(15\)).
\end{theorem}

\begin{corollary}[Geodesic growth rate and geodesic entropy]\label{cor:real-input-40-geodesic-entropy}
Let \(\rho_{\mathrm{nb}}:=\rho(H_{\mathrm{nb}})\) be the Perron root. Then \(\rho_{\mathrm{nb}}\) is the largest real root of the equation
\[
\rho^{15}-4\rho^{12}-9\rho^{11}-12\rho^{10}-16\rho^{9}-11\rho^{8}-11\rho^{7}-4\rho^{6}-\rho^{5}+8\rho^{4}+12\rho^{3}+10\rho^{2}+8\rho+2=0
\]
with numerical value
\[
\rho_{\mathrm{nb}}\approx 2.295661870047795\ldots,\qquad
h_{\mathrm{nb}}:=\log\rho_{\mathrm{nb}}\approx 0.831021198469209\ldots.
\]
For comparison, the leading growth rate of the real input kernel in the periodic-point framework is \(\lambda=\varphi^2\) (\ref{subsec:arity-dirichlet-mertens-tensor}).
\end{corollary}

\paragraph{Geodesic primitive spectrum and explicit initial terms}
Denote the geodesic periodic point count by \(a_n^{\mathrm{nb}}:=\Tr(H_{\mathrm{nb}}^n)\), and the geodesic primitive orbit count by \(p_n^{\mathrm{nb}}\) (counted by orbit).
Then the necklace relation and M\"{o}bius inversion still hold:
\[
a_n^{\mathrm{nb}}=\sum_{d\mid n} d\,p_d^{\mathrm{nb}},
\qquad
p_n^{\mathrm{nb}}=\frac1n\sum_{d\mid n}\mu(d)\,a_{n/d}^{\mathrm{nb}}.
\]
The first \(15\) terms on this kernel are fully explicit:
\[
(p_n^{\mathrm{nb}})_{n=1}^{15}=(0,0,4,9,12,22,47,95,196,399,843,1775,3790,8038,17258).
\]

\begin{corollary}[Strict Ramanujan margin and subexponential error for the geodesic prime shadow]\label{cor:real-input-40-geodesic-ramanujan-margin}
Let
\(
\Delta_{\mathrm{nb}}(z):=\det(I-zH_{\mathrm{nb}})
\)
(Theorem \ref{thm:real-input-40-geodesic-det}), and order its roots by modulus as
\(
0<|z_1|\le |z_2|\le\cdots\le |z_{15}|
\)
(with multiplicities).
Then \(\rho_{\mathrm{nb}}=|z_1|^{-1}\) and \(z_1=\rho_{\mathrm{nb}}^{-1}\);
define
\[
\Lambda_{\mathrm{nb}}:=|z_2|^{-1},\qquad
\delta_{\mathrm{nb}}:=\log\frac{\Lambda_{\mathrm{nb}}^2}{\rho_{\mathrm{nb}}}.
\]
Numerical audit yields
\[
|z_2|\approx 0.7346134203,\qquad
\Lambda_{\mathrm{nb}}\approx 1.3612601844,\qquad
\frac{\Lambda_{\mathrm{nb}}}{\sqrt{\rho_{\mathrm{nb}}}}\approx 0.8984360744<1,
\qquad
\delta_{\mathrm{nb}}\approx -0.2141994446<0.
\]
Furthermore, the geodesic primitive orbit count satisfies the asymptotic formula
\[
p_n^{\mathrm{nb}}
=\frac{\rho_{\mathrm{nb}}^n}{n}
+O\!\left(\frac{\max\{\Lambda_{\mathrm{nb}}^{n},\,\rho_{\mathrm{nb}}^{n/2}\}}{n}\right).
\]
In particular, when \(n=p\ge 3\) is prime,
\[
p_p^{\mathrm{nb}}=\frac{\rho_{\mathrm{nb}}^{p}}{p}+O\!\left(\frac{\Lambda_{\mathrm{nb}}^{p}}{p}\right),
\]
with error exponent strictly less than \(\rho_{\mathrm{nb}}^{1/2}\).
\end{corollary}
\begin{proof}
By Theorem \ref{thm:real-input-40-geodesic-det},
\(
\Delta_{\mathrm{nb}}(z)=\prod_{j=1}^{15}(1-z\lambda_j)
\)
where \(\{\lambda_j\}\) are all nonzero eigenvalues of \(H_{\mathrm{nb}}\), and \(\lambda_j=z_j^{-1}\).
Thus
\(
a_n^{\mathrm{nb}}=\Tr(H_{\mathrm{nb}}^n)=\sum_{j=1}^{15}\lambda_j^n
=\rho_{\mathrm{nb}}^n+O(\Lambda_{\mathrm{nb}}^n)
\).
Substituting into the M\"{o}bius inversion formula
\(
p_n^{\mathrm{nb}}=\frac1n\sum_{d\mid n}\mu(d)\,a_{n/d}^{\mathrm{nb}}
\),
the main term comes from the \(d=1\) Perron contribution \(\rho_{\mathrm{nb}}^n/n\);
the error has one part from the \(d=1\) sub-spectral modes of order \(O(\Lambda_{\mathrm{nb}}^n/n)\);
another part from the \(d\ge 2\) Perron sub-level terms \(\rho_{\mathrm{nb}}^{n/d}\le \rho_{\mathrm{nb}}^{n/2}\).
Combining yields the stated upper bound.
When \(n=p\ge 3\) is prime, the sum contains only \(d=1,p\), so
\(
p_p^{\mathrm{nb}}=(a_p^{\mathrm{nb}}-a_1^{\mathrm{nb}})/p
=\rho_{\mathrm{nb}}^{p}/p+O(\Lambda_{\mathrm{nb}}^{p}/p)
\)
(where \(a_1^{\mathrm{nb}}\) is a constant term), yielding the prime case. \qed
\end{proof}

\begin{theorem}[Geodesic Mertens constant (Abel finite part)]\label{thm:real-input-40-geodesic-mertens}
Define the geodesic primitive generating function
\[
P_{\mathrm{nb}}(z):=\sum_{n\ge 1}p_n^{\mathrm{nb}}z^n.
\]
At the leading pole \(z_\star=\rho_{\mathrm{nb}}^{-1}\), define the Abel finite-part constant \(\mathfrak{M}_{\mathrm{nb}}\) such that
\[
P_{\mathrm{nb}}(z)=-\log(1-\rho_{\mathrm{nb}}z)+\log\mathfrak{M}_{\mathrm{nb}}+o(1),
\qquad z\uparrow z_\star.
\]
Then numerically
\[
\boxed{
\log\mathfrak{M}_{\mathrm{nb}}\approx -1.465260542946848,\qquad
\mathfrak{M}_{\mathrm{nb}}\approx 0.231017793578888,
}
\]
and hence
\[
\boxed{
\mathsf{M}_{\mathrm{nb}}:=\gamma+\log\mathfrak{M}_{\mathrm{nb}}\approx -0.888044878045315,
}
\]
in stark contrast to the Mertens constant \(\mathsf{M}\approx 0.3229030\) of the real input kernel in the periodic-point framework (\ref{subsec:arity-dirichlet-mertens-tensor}).
\end{theorem}

\paragraph{Bartholdi-type backtracking fugacity deformation: two-variable determinant closed form}
Let \(\Delta_{\mathrm{bt}}(z,t)=\det(I-zH(t))\).
Then there exists \(F(z,t)\in\ZZ[z,t]\) (of degree \(20\) in \(z\) and at most \(9\) in \(t\)) such that:
\begin{theorem}[Two-variable determinant closed form of the backtracking fugacity deformation]\label{thm:real-input-40-bartholdi-det}
The following exact factorization holds:
\[
\boxed{
\Delta_{\mathrm{bt}}(z,t)=(1-t^2 z^2)\bigl(1+t(1-t)z^2\bigr)^2\,F(z,t).
}
\]
where
\[
F(z,t)=\sum_{k=0}^{20}f_k(t)\,z^k,\qquad f_k(t)\in\ZZ[t],\quad \deg f_k\le 9,
\]
with coefficients explicitly given by
\[
\begin{aligned}
f_0(t)&=1,\\
f_1(t)&=-t,\\
f_2(t)&=-2t(2t+1),\\
f_3(t)&=2(2t^3+t^2-2),\\
f_4(t)&=6t^4+6t^3+t^2+4t-9,\\
f_5(t)&=-6t^5-6t^4-t^3+14t^2+9t-12,\\
f_6(t)&=-4t^6-6t^5-2t^4-14t^3+24t^2+15t-16,\\
f_7(t)&=4t^7+6t^6+2t^5-18t^4-22t^3+41t^2-t-11,\\
f_8(t)&=(t-1)(t^7+3t^6+4t^5+22t^4-4t^3-41t^2+21t+11),\\
f_9(t)&=-(t-1)^2(t^7+4t^6+8t^5+2t^4-24t^3+11t^2+50t+4),\\
f_{10}(t)&=-(t-1)(10t^6-6t^5-45t^4+46t^3+54t^2-56t-1),\\
f_{11}(t)&=-(t-1)^2(2t^6+14t^5-15t^4-61t^3+41t^2+36t-8),\\
f_{12}(t)&=(t-1)^2(2t^7-t^6-25t^5+7t^4+85t^3-36t^2-36t+12),\\
f_{13}(t)&=(t-1)^3(3t^6-25t^4+22t^3+40t^2-24t-10),\\
f_{14}(t)&=(t-1)^3(4t^6+t^5-46t^4+6t^3+57t^2-28t-8),\\
f_{15}(t)&=2(t-1)^4(2t^5-2t^4-6t^3+7t^2+7t+1),\\
f_{16}(t)&=t(t-1)^4(9t^4+t^3-24t^2+18t+8),\\
f_{17}(t)&=-2t(t-1)^5(t+2)(t+3),\\
f_{18}(t)&=2t(t-1)^5(t^3-t^2+2t-4),\\
f_{19}(t)&=2t^2(t-1)^6(t+3),\\
f_{20}(t)&=-4t^2(t-1)^7.
\end{aligned}
\]
\end{theorem}

\begin{remark}[Two key specializations]\label{rem:real-input-40-bartholdi-specializations}
Placing the ``standard prime shadow'' and the ``geodesic prime shadow'' on the same curve:
\begin{itemize}
  \item \textbf{\(t=0\)}:
  \[
  \begin{aligned}
  \Delta_{\mathrm{bt}}(z,0)&=F(z,0)\\
  &=\det(I-zH_{\mathrm{nb}}),
  \end{aligned}
  \]
  recovering the geodesic (non-backtracking) \(\zeta\) of Theorem \ref{thm:real-input-40-geodesic-det}.
  \item \textbf{\(t=1\)}:
  \[
  \Delta_{\mathrm{bt}}(z,1)=(1-z^2)F(z,1),
  \]
  which specializes exactly to the denominator of the \(\zeta\) for the full edge-to-edge transfer:
  \[
  \det(I-zH(1))
  =-(z-1)^2(z+1)^3(z^2-3z+1)(z^2-z-1).
  \]
  Here \(F(z,1)=(z-1)(z+1)^2(z^2-3z+1)(z^2-z-1)\) (with \(f_k(1)=0\) for all \(k\ge 8\)).
\end{itemize}
\end{remark}

\begin{remark}[Bartholdi fugacity's infinity root cluster: \texorpdfstring{$13$}{13} necessarily escaping branches as \texorpdfstring{$t\to 1$}{t->1}]\label{rem:real-input-40-bartholdi-infinite-root-cluster}
From the explicit coefficients in Theorem \ref{thm:real-input-40-bartholdi-det}, one observes: when \(k\ge 8\), \(f_k(t)\) contains an explicit factor of \((t-1)\) (and \(f_9(t)\) even contains \((t-1)^2\)), so
\[
\deg_z F(z,1)=7
\qquad\text{while in general}\qquad
\deg_z F(z,t)=20\ (t\neq 1).
\]
Hence in the projective compactification sense of the \(z\)-plane, as \(t\to 1\), besides the roots that converge to the finite cluster of \(F(z,1)\), the roots of \(F(\,\cdot\,,t)\) must have exactly \(20-7=13\) branches escaping to infinity;
equivalently, setting \(\varepsilon=t-1\), there exist \(13\) Newton--Puiseux branches \(z_j(\varepsilon)\) satisfying \(|z_j(\varepsilon)|\to\infty\) (\(\varepsilon\to 0\)).
In the reciprocal variable \(\lambda:=1/z\), these \(13\) escaping branches correspond to a zero-eigenvalue cluster \(\lambda\to 0\): they identify the ``standard prime shadow'' at \(t=1\) as a structural degeneracy point on this two-variable spectral curve, whose additional spectral branches collapse to \(\lambda=0\) with definite Puiseux slopes.
\end{remark}

\begin{proposition}[Newton--Puiseux escape exponent spectrum of the Bartholdi fugacity at \texorpdfstring{$t=1$}{t=1}]\label{prop:real-input-40-bartholdi-newton-puiseux-spectrum}
Following the notation of Theorem \ref{thm:real-input-40-bartholdi-det}, let \(F(z,t)=\sum_{k=0}^{20}f_k(t)z^k\).
Write \(\varepsilon:=t-1\), and define the \((t-1)\)-adic valuation profile
\[
v_k:=\operatorname{ord}_{\varepsilon}\bigl(f_k(1+\varepsilon)\bigr)\in\NN\cup\{0\}.
\]
In the reciprocal variable \(\lambda=z^{-1}\), let
\[
G(\lambda,\varepsilon):=\lambda^{20}F(\lambda^{-1},1+\varepsilon)
=\sum_{x=0}^{20}g_x(\varepsilon)\lambda^x,\qquad g_x(\varepsilon):=f_{20-x}(1+\varepsilon).
\]
Then the Newton polygon lower convex hull of \(G(\lambda,\varepsilon)\) contains three negative-slope edges
\[
(-1,\,-\tfrac12,\,-\tfrac13),
\qquad
\text{with horizontal lengths respectively}\quad (2,\,8,\,3).
\]
Hence as \(\varepsilon\to 0\), besides the \(7\) branches that remain bounded and converge to roots of \(F(z,1)\), the remaining \(13\) escaping roots \(z_j(\varepsilon)\) split into three classes of Puiseux asymptotics
\[
z_j(\varepsilon)=c_j\,\varepsilon^{-\alpha}+o\!\left(\varepsilon^{-\alpha}\right),
\qquad
\alpha\in\left\{1,\frac12,\frac13\right\},
\qquad
\#\{\alpha=1\}=2,\ \#\{\alpha=\tfrac12\}=8,\ \#\{\alpha=\tfrac13\}=3,
\]
where \(c_j\neq 0\) are algebraic constants. This exponent spectrum is completely forced by \(v_k\) (equivalently: by the \((t-1)\)-factor ladder in Theorem \ref{thm:real-input-40-bartholdi-det}).
\end{proposition}
\begin{proof}
From the explicit decomposition in Theorem \ref{thm:real-input-40-bartholdi-det}, directly read off
\[
v_k=
\begin{cases}
0,& 0\le k\le 7,\\
1,& k\in\{8,10\},\\
2,& k\in\{9,11,12\},\\
3,& k\in\{13,14\},\\
4,& k\in\{15,16\},\\
5,& k\in\{17,18\},\\
6,& k=19,\\
7,& k=20.
\end{cases}
\]
Hence the valuation of \(g_x(\varepsilon)=f_{20-x}(1+\varepsilon)\) is \(\operatorname{ord}_{\varepsilon}g_x=v_{20-x}\).
Plotting the points \((x,\operatorname{ord}_{\varepsilon}g_x)\) (\(0\le x\le 20\)) in a Newton diagram, the lower convex hull exactly connects
\[
(0,7)\to(2,5)\to(10,1)\to(13,0)\to(20,0),
\]
so the three negative slopes are \(-1,-\tfrac12,-\tfrac13\), with horizontal lengths \(2,8,3\) respectively.
The Newton--Puiseux theorem states: each slope \(-\alpha\) corresponds to root branches with \(\lambda\sim \varepsilon^{\alpha}\), and the number of such branches equals the horizontal length of that segment;
equivalently \(z=\lambda^{-1}\sim \varepsilon^{-\alpha}\), yielding the stated exponent spectrum and count decomposition.
The remaining \(7\) bounded root branches come from the degree-\(7\) polynomial \(F(z,1)\) at \(\varepsilon=0\) (see Remark \ref{rem:real-input-40-bartholdi-specializations}). \qed
\end{proof}

\begin{proposition}[Initial-form equations and leading algebraic constants of the escape clusters]\label{prop:real-input-40-bartholdi-escape-initial-forms}
Following the notation of Proposition \ref{prop:real-input-40-bartholdi-newton-puiseux-spectrum}, let \(\alpha\in\{1,\tfrac12,\tfrac13\}\) be the exponent corresponding to some Newton edge.
Under the scaling \(\lambda=\varepsilon^{\alpha}\xi\), extracting the dominant terms on the edge as the \(\varepsilon\)-lowest-order initial form yields three classes of initial-term equations for the escape clusters:
\begin{itemize}
  \item \textbf{\(\alpha=1\) (two branches):}
  The initial form equation is
  \[
  \xi^2+1=0,
  \]
  so the two strongest escaping branches satisfy
  \[
  z_{\pm}(\varepsilon)=\mp\,\mathrm{i}\,\varepsilon^{-1}+O\!\left(\varepsilon^{-1/2}\right).
  \]
  \item \textbf{\(\alpha=\tfrac13\) (three branches):}
  The initial form equation is
  \[
  \xi^3-2=0,
  \]
  so there exist three branches satisfying
  \[
  z(\varepsilon)=2^{-1/3}\omega^{-1}\varepsilon^{-1/3}+O(1),\qquad \omega^3=1.
  \]
  \item \textbf{\(\alpha=\tfrac12\) (eight branches):}
  The initial form polynomial in \(\xi\) contains only even powers; setting \(y=\xi^2\), after stripping off the zero root corresponding to the shared endpoint, the nonzero initial terms satisfy
  \[
  y^4-4y^3+7y^2-6y+2=0,\qquad y=\xi^2,
  \]
  each root \(y\) produces two branches \(\xi=\pm\sqrt{y}\), yielding a total of \(8\) escape branches with \(\alpha=\tfrac12\).
\end{itemize}
Hence the minimal algebraic fields containing the leading constants of the three escape clusters lie respectively in \(\QQ(\mathrm{i})\), \(\QQ(2^{1/3},\omega)\), and the splitting field of the above quartic equation; all can be computed at the purely algebraic level from the coefficient decomposition in Theorem \ref{thm:real-input-40-bartholdi-det}.
\end{proposition}
\begin{proof}
All three initial form equations are directly given by ``taking the lowest-order terms on the Newton edge.'' Taking \(\alpha=\tfrac12\) as an example:
this edge corresponds to \(x\in\{2,4,6,8,10\}\), i.e., the five terms \(f_{18},f_{16},f_{14},f_{12},f_{10}\).
By the leading coefficients of Theorem \ref{thm:real-input-40-bartholdi-det} at \(t=1\), they are respectively
\[
f_{18}\sim -4\varepsilon^5,\quad
f_{16}\sim 12\varepsilon^4,\quad
f_{14}\sim -14\varepsilon^3,\quad
f_{12}\sim 8\varepsilon^2,\quad
f_{10}\sim -2\varepsilon,
\]
and appear at the same order under the substitution \(\lambda=\varepsilon^{1/2}\xi\), so the initial form is
\[
-4\xi^2+12\xi^4-14\xi^6+8\xi^8-2\xi^{10}
\,=\,
-2\xi^2\bigl(y^4-4y^3+7y^2-6y+2\bigr)\quad(y=\xi^2),
\]
yielding the quartic equation.
For \(\alpha=1\), under the scaling \(\lambda=\varepsilon\xi\), the edge endpoint terms are contributed by
\[
f_{20}(t)=-4t^2(t-1)^7,
\qquad
f_{18}(t)=2t(t-1)^5(t^3-t^2+2t-4)
\]
taking the leading terms at \(t=1+\varepsilon\) and substituting yields the initial form \(-4(1+\xi^2)\), so the limit equation is \(\xi^2+1=0\).
For \(\alpha=\tfrac13\), the edge endpoints are \(x\in\{10,13\}\), corresponding to \(f_{10}\sim -2\varepsilon\) and \(f_7(1)=1\);
substituting \(\lambda=\varepsilon^{1/3}\xi\) yields the initial form \(\xi^{10}(\xi^3-2)\), hence \(\xi^3-2=0\).
\qed
\end{proof}

\begin{remark}[\texorpdfstring{$(t-1)$}{(t-1)}-adic associated graded three-scale spectrum decomposition]\label{rem:real-input-40-bartholdi-graded-spectrum}
Let \(\varepsilon=t-1\). By the \((t-1)\)-factor ladder in Theorem \ref{thm:real-input-40-bartholdi-det}, in the associated graded ring \(\mathrm{gr}_{(t-1)}\ZZ[t]\), the infinity spectral branch splitting of \(F(z,1+\varepsilon)\) can be captured by three weight-homogeneous scalings:
\[
z=\varepsilon^{-1}w,\qquad z=\varepsilon^{-1/2}w,\qquad z=\varepsilon^{-1/3}w.
\]
The corresponding dominant initial form polynomials (in the sense of \(w\neq 0\)) are respectively
\[
w^2+1,\qquad
y^4-4y^3+7y^2-6y+2\ (y=w^2),\qquad
w^3-2.
\]
They are complementary to the finite-root polynomial \(F(z,1)\) (of degree \(7\)) left at \(t=1\), so the degeneration at \(t\to 1\) strictly splits in the \((t-1)\)-adic sense as
\[
20 \;=\; 7\ \oplus\ 3\ \oplus\ 8\ \oplus\ 2,
\]
where \(7\) is the finite spectral layer degree, and \((3,8,2)\) correspond respectively to the three escape scale layers \(\alpha=\tfrac13,\tfrac12,1\).
\end{remark}

\begin{theorem}[Closed forms for the backtracking density and backtracking variance]\label{thm:real-input-40-backtracking-cumulants}
Let \(t=e^{\theta}\), and write
\[
\lambda(\theta):=\rho\!\bigl(H(e^{\theta})\bigr),\qquad
P(\theta):=\log\lambda(\theta).
\]
Then at \(\theta=0\) (i.e., \(t=1\)) the following closed forms hold:
\[
\boxed{\;
P'(0)=\frac{211-93\sqrt5}{20}\approx 0.1522839046\;},
\]
\[
\boxed{\;
P''(0)=\frac{193219}{100}-\frac{216003}{250}\sqrt5\approx 0.2004346245\; }.
\]
Here \(P'(0)\) can be interpreted as the ``frequency of immediate-reversal backtracking fragments per step under the maximum-entropy equilibrium state'', while \(P''(0)\) is the asymptotic variance density of the backtracking count (Green--Kubo).
\end{theorem}
