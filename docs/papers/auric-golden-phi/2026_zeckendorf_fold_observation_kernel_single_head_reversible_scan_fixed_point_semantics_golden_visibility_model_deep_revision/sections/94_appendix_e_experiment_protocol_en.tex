\section{Appendix E: Experiment Protocol and Metrics (Complete Definitions)}

\begin{itemize}
  \item \textbf{Parameters:} $L=1024$, $\delta=1$, number of steps $N=2\times 10^5$.
  \item \textbf{Initialization:} $i_0$ uniformly chosen from $\ZZ/L\ZZ$, $h_0$ uniformly chosen from $Y_6$, ring field $C_0$ at each lattice point independently uniformly chosen from $Y_6$.
  \item \textbf{Round functions:} $f_0,f_{\mathrm{fold}},f_{\mathrm{mix}}$ as defined in the main text.
  \item \textbf{Error norms:} $\norm{e}_\infty,\norm{e}_1$ as in Section 9 of the main text.
  \item \textbf{Total variation:} $\norm{\pi-u}_{\TV}=\frac12\sum_{h\in Y_6}\abs{\pi(h)-1/64}$.
  \item \textbf{Graph spectrum:} construct edge support set $E$ from $(x_t,x_{t+1})$, compute $\card{E}$ and strongly connected component decomposition.
\end{itemize}
