\section{Collision moments, spectral recovery, and Hankel methods}
\label{sec:moment-kernel}

This section studies the collision moments \(S_q(m)\).  They admit finite-state
realizations in the length variable \(m\), and for fixed \(m\) the full
multiplicity spectrum can be recovered from the moment sequence in \(q\).

\subsection{Finite-state realizations of the collision moments}

\begin{definition}[Collision moments]
\label{def:collision-moments}
For integers \(m\ge 1\) and \(q\ge 0\), define
\[
  S_q(m):=\sum_{x\in X_m} d_m(x)^q.
\]
Thus \(S_0(m)=\card{X_m}=F_{m+2}\), \(S_1(m)=2^m\), and for \(q\ge 2\),
\[
  S_q(m)
  =
  \card{\bigl\{(\omega^{(1)},\dots,\omega^{(q)})\in\Omega_m^q:
  \Fold_m(\omega^{(1)})=\cdots=\Fold_m(\omega^{(q)})\bigr\}}.
\]
\end{definition}

\begin{theorem}[Finite-state collision kernels]
\label{thm:collision-kernel}
For every integer \(q\ge 2\), there exist a finite nonnegative integer matrix
\(A_q\) and nonnegative integer vectors \(u_q,v_q\) such that
\[
  S_q(m)=u_q^{\mathsf T}A_q^m v_q
  \qquad (m\ge m_0(q))
\]
for some finite offset \(m_0(q)\).  Consequently,
\[
  \lim_{m\to\infty}\frac{1}{m}\log S_q(m)=\log r_q,
  \qquad
  r_q:=\rho(A_q).
\]
\end{theorem}

\begin{proof}
The folding normalization is implemented by a finite-state transducer: reading a
binary symbol updates a finite carry state and emits one stable output symbol.
Synchronize \(q\) copies of this transducer and keep only those transitions for
which all emitted output symbols agree.  The resulting synchronized automaton
has finitely many states, hence an adjacency matrix \(A_q\) with nonnegative
integer entries.

The number of length-\(m\) paths in the synchronized automaton with admissible
boundary conditions is exactly the number of \(q\)-tuples of binary inputs with
identical folds, namely \(S_q(m)\).  This yields the displayed formula.  The
exponential growth rate is the Perron root of \(A_q\) by Perron--Frobenius
theory \cite{Seneta2006}.
\end{proof}

\begin{theorem}[Symmetric compression]
\label{thm:symmetric-compression}
Let \(k\) be the number of states of a one-copy fold transducer.  The kernel
\(A_q\) of \Cref{thm:collision-kernel} admits a symmetric quotient
\(\widetilde A_q\) acting on histogram states of cardinality
\[
  \binom{k+q-1}{k-1},
\]
such that
\[
  S_q(m)=\widetilde u_q^{\mathsf T}\widetilde A_q^m\widetilde v_q
  \qquad\text{and}\qquad
  \rho(\widetilde A_q)=\rho(A_q)=r_q.
\]
In particular, for fixed \(k\) the state-space dimension grows polynomially in
\(q\).
\end{theorem}

\begin{proof}
The symmetric group \(\mathfrak S_q\) acts on the synchronized state space by
permuting the \(q\) coordinates.  The transition relation and the collision
condition are invariant under this action, so the kernel descends to the orbit
space, which is indexed by weak compositions of \(q\) into \(k\) parts.  This
gives the binomial coefficient above.

Averaging a Perron eigenvector of \(A_q\) over \(\mathfrak S_q\) produces a
nonzero symmetric eigenvector with the same eigenvalue, so the quotient has the
same spectral radius.
\end{proof}

\begin{proposition}[Hankel positivity and cross-moment convexity]
\label{prop:hankel-psd}
For fixed \(m\), the Hankel matrix
\[
  H_r^{(m)}:=[S_{i+j}(m)]_{0\le i,j\le r}
\]
is positive semidefinite for every \(r\ge 0\).  Consequently, for all integers
\(q\ge r\ge 1\),
\[
  S_q(m)^2\le S_{q-r}(m)S_{q+r}(m).
\]
\end{proposition}

\begin{proof}
For \(c=(c_0,\dots,c_r)\in\RR^{r+1}\),
\[
  c^{\mathsf T}H_r^{(m)}c
  =
  \sum_{x\in X_m}\left(\sum_{i=0}^r c_i d_m(x)^i\right)^2\ge 0.
\]
This proves positive semidefiniteness.  The cross-moment inequality follows by
applying Cauchy--Schwarz to the sequences
\(d_m(x)^{(q-r)/2}\) and \(d_m(x)^{(q+r)/2}\).
\end{proof}

\begin{corollary}[Discrete pressure convexity]
\label{cor:cross-q-logconvex}
For every \(q\ge 2\), the Perron roots of \Cref{thm:collision-kernel} satisfy
\[
  r_q^2\le r_{q-1}r_{q+1}.
\]
\end{corollary}

\begin{proof}
By \Cref{prop:hankel-psd},
\(S_q(m)^2\le S_{q-1}(m)S_{q+1}(m)\) for every \(m\).  Take logarithms, divide by
\(m\), and let \(m\to\infty\), using \Cref{thm:collision-kernel}.
\end{proof}

\subsection{Recovery of the multiplicity spectrum for fixed \texorpdfstring{$m$}{m}}

\begin{theorem}[Truncated-sum curve and tail inversion]
\label{thm:truncated-sum-curve}
Fix \(m\ge 1\), and define
\[
  \mathcal C_m(u):=\sum_{x\in X_m}\min\{d_m(x),u\}
  \qquad (u\ge 0),
\]
and
\[
  N_m(u):=\card{\{x\in X_m:\ d_m(x)>u\}}
  \qquad (u\ge 0).
\]
Then
\[
  \mathcal C_m(u)=\int_0^u N_m(v)\,\mathrm{d}v.
\]
In particular, \(\mathcal C_m\) is piecewise linear and concave, its slope
jumps occur exactly at the distinct multiplicity values of \(d_m\), and those
jumps recover the full multiplicity spectrum.
\end{theorem}

\begin{proof}
For every \(d,u\ge 0\),
\[
  \min\{d,u\}=\int_0^u \ind_{\{v<d\}}\,\mathrm{d}v.
\]
Summing over \(x\in X_m\) gives
\[
  \mathcal C_m(u)
  =
  \sum_{x\in X_m}\int_0^u \ind_{\{v<d_m(x)\}}\,\mathrm{d}v
  =
  \int_0^u
  \Bigl(\sum_{x\in X_m}\ind_{\{d_m(x)>v\}}\Bigr)\,\mathrm{d}v
  =
  \int_0^u N_m(v)\,\mathrm{d}v.
\]
Since \(N_m\) is a decreasing step function, \(\mathcal C_m\) is piecewise
linear and concave.  If \(a\) is an integer, then the jump of the slope at
\(u=a\) equals
\[
  N_m(a^-)-N_m(a^+)
  =
  \card{\{x\in X_m:\ d_m(x)=a\}},
\]
which recovers the multiplicity spectrum.
\end{proof}

\begin{theorem}[Rational fiber-spectrum resolvent]
\label{thm:fiber-resolvent}
Fix \(m\ge 1\), and let \(a_1<\cdots<a_{R_m}\) be the distinct values of the
multiplicity function \(d_m\), with
\[
  c_j:=\card{\{x\in X_m:\ d_m(x)=a_j\}}.
\]
Define the generating function
\[
  \mathcal H_m(u):=\sum_{q=0}^{\infty} S_q(m)u^q.
\]
Then
\[
  \mathcal H_m(u)=\sum_{j=1}^{R_m}\frac{c_j}{1-a_ju}.
\]
In particular, \(\mathcal H_m\) is rational, its poles are precisely the points
\(u=1/a_j\), and the residue at \(1/a_j\) equals \(c_j/a_j\).
\end{theorem}

\begin{proof}
Since
\(S_q(m)=\sum_{j=1}^{R_m} c_j a_j^q\),
\[
  \mathcal H_m(u)
  =
  \sum_{j=1}^{R_m} c_j \sum_{q=0}^{\infty}(a_ju)^q
  =
  \sum_{j=1}^{R_m}\frac{c_j}{1-a_ju}.
\]
\end{proof}

\begin{corollary}[Stieltjes logarithmic derivative]
\label{cor:stieltjes-log-derivative}
In the notation of \Cref{thm:fiber-resolvent}, define
\[
  Z_m(t):=\prod_{x\in X_m}(1+t\,d_m(x))
  =\prod_{j=1}^{R_m}(1+t\,a_j)^{c_j}.
\]
Then
\[
  \frac{\mathrm{d}}{\mathrm{d}t}\log Z_m(t)
  =
  \sum_{j=1}^{R_m}\frac{c_j a_j}{1+t\,a_j}.
\]
Hence the logarithmic derivative of \(Z_m\) is a rational Stieltjes function
whose poles determine the distinct multiplicities \(a_j\) and whose residues
recover the counts \(c_j\).
\end{corollary}

\begin{proof}
Differentiate the displayed product formula for \(Z_m(t)\).
\end{proof}

\begin{theorem}[Exact finite reconstruction at fixed resolution]
\label{thm:fiber-reconstruction}
In the notation of \Cref{thm:fiber-resolvent}, the first \(2R_m\) moments
\[
  S_0(m),S_1(m),\dots,S_{2R_m-1}(m)
\]
determine the entire multiplicity spectrum
\(\{(a_j,c_j)\}_{j=1}^{R_m}\) uniquely.
\end{theorem}

\begin{proof}
Let
\[
  Q_m(t):=\prod_{j=1}^{R_m}(t-a_j)
  =t^{R_m}+b_{R_m-1}t^{R_m-1}+\cdots+b_0.
\]
Then the moment sequence satisfies the linear recurrence
\[
  S_{q+R_m}(m)+b_{R_m-1}S_{q+R_m-1}(m)+\cdots+b_0S_q(m)=0
  \qquad (q\ge 0).
\]
Writing this recurrence for \(q=0,\dots,R_m-1\) gives a linear system
\[
  H_m b=-h_m,
\]
where
\[
  H_m=[S_{i+j}(m)]_{0\le i,j\le R_m-1},
  \qquad
  h_m=(S_{R_m}(m),\dots,S_{2R_m-1}(m))^{\mathsf T}.
\]
The Hankel matrix factors as
\[
  H_m=V^{\mathsf T}\diag(c_1,\dots,c_{R_m})V,
\]
with Vandermonde matrix \(V_{j\ell}=a_j^{\ell-1}\).  Because the \(a_j\) are
distinct and the \(c_j\) are positive, \(H_m\) is invertible.  Hence the
coefficients of \(Q_m\) are uniquely determined by the first \(2R_m\) moments.

The roots of \(Q_m\) recover the nodes \(a_j\), and the first \(R_m\) moment
equations \(S_q(m)=\sum_j c_j a_j^q\) determine the weights \(c_j\) by an
invertible Vandermonde system.
\end{proof}

\begin{theorem}[Sharpness of the \texorpdfstring{$2R$}{2R} threshold]
\label{thm:fiber-threshold-sharp}
For every integer \(R\ge 2\) there exist two distinct \(R\)-atomic spectra
\[
  \{(a_i,n_i)\}_{i=1}^R\neq \{(a'_i,n'_i)\}_{i=1}^R,
\]
with positive integer nodes and weights, such that
\[
  \sum_{i=1}^R n_i a_i^k
  =
  \sum_{i=1}^R n'_i (a'_i)^k
  \qquad (0\le k\le 2R-2),
\]
but the two spectra differ at moment \(k=2R-1\).  Hence the
\(2R\)-moment threshold of \Cref{thm:fiber-reconstruction} is optimal in
general.
\end{theorem}

\begin{proof}
Set \(n=2R-1\).  Define two spectra by
\[
  a_i:=2i-1,
  \qquad
  n_i:=\binom{n}{2i-2}
  \qquad (1\le i\le R),
\]
and
\[
  a'_i:=2i,
  \qquad
  n'_i:=\binom{n}{2i-1}
  \qquad (1\le i\le R).
\]
For every polynomial \(p\) of degree at most \(n-1\), the \(n\)-th finite
difference identity gives
\[
  \sum_{j=0}^{n}(-1)^j\binom{n}{j}p(j+1)=0.
\]
Taking \(p(t)=t^k\) with \(0\le k\le n-1\) and separating even and odd indices
yields the claimed equality of the first \(2R-1\) moments.  For \(k=n\), the
\(n\)-th finite difference of \(t^n\) is nonzero, so the spectra differ at the
next moment.
\end{proof}

\subsection{Minimal Hankel realizations and denominator control}

\begin{proposition}[Minimal Hankel realization]
\label{prop:hankel-realization}
Fix \(q\ge 2\), and suppose the tail sequence \(b_n:=S_q(n+n_0)\) satisfies a
minimal linear recurrence of order \(d\).  Let
\[
  H_0:=(b_{i+j})_{0\le i,j\le d-1},
  \qquad
  H_1:=(b_{i+j+1})_{0\le i,j\le d-1}.
\]
Then \(H_0\) is invertible, and the matrix
\[
  A:=H_0^{-1}H_1
\]
yields a minimal rational realization
\[
  b_n=\alpha^{\mathsf T}A^n\beta
  \qquad (n\ge 0)
\]
for suitable vectors \(\alpha,\beta\in\QQ^d\).  If all \(b_n\) are integers,
then every denominator appearing in \(A\) divides \(\det(H_0)\).
\end{proposition}

\begin{proof}
Minimality of the recurrence implies that the first \(d\) Hankel columns are
linearly independent, hence \(H_0\) is invertible.  The state vector
\((b_n,\dots,b_{n+d-1})^{\mathsf T}\) evolves linearly, and the companion action
in this basis is exactly \(A=H_0^{-1}H_1\).  Taking
\(\alpha=(1,0,\dots,0)^{\mathsf T}\) and
\(\beta=(b_0,\dots,b_{d-1})^{\mathsf T}\) gives the realization.

If the \(b_n\) are integers, then \(H_0\) and \(H_1\) have integer entries and
\[
  A=\frac{\adj(H_0)H_1}{\det(H_0)},
\]
so every denominator divides \(\det(H_0)\).
\end{proof}

\begin{theorem}[Smith-normal-form denominator formula]
\label{thm:smith-denominator}
In the setting of \Cref{prop:hankel-realization}, assume that \(H_0,H_1\in
M_d(\ZZ)\).  Let
\[
  UH_0V=\diag(s_1,\dots,s_d),
  \qquad
  s_1\mid s_2\mid\cdots\mid s_d,
\]
be the Smith normal form of \(H_0\), and set \(C:=UH_1\).  For each row of
\(C\), let
\[
  g_i:=\gcd(C_{i1},\dots,C_{id}),
  \qquad
  e_i:=\frac{s_i}{\gcd(s_i,g_i)}.
\]
Then the smallest positive integer \(N_*\) such that \(N_*A\) has integral
entries is
\[
  N_* = \lcm(e_1,\dots,e_d).
\]
In particular,
\[
  N_*\mid s_d\mid \det(H_0).
\]
\end{theorem}

\begin{proof}
From \(UH_0V=\diag(s_1,\dots,s_d)\) we obtain
\[
  A=H_0^{-1}H_1=V\diag(s_1^{-1},\dots,s_d^{-1})C.
\]
Since \(V\in\mathrm{GL}_d(\ZZ)\), the integrality of \(NA\) is equivalent to the
integrality of \(N\diag(s_1^{-1},\dots,s_d^{-1})C\).  The \(i\)-th row of this
matrix equals \((N/s_i)C_{i\bullet}\), so the smallest positive integer clearing
the denominators in that row is exactly \(e_i\).  Hence the smallest common
clearing denominator is their least common multiple \(N_*\).

The divisibility \(e_i\mid s_i\mid s_d\) yields \(N_*\mid s_d\), and the Smith
divisibility chain gives \(s_d\mid\det(H_0)\).
\end{proof}

\begin{corollary}[The R\'enyi endpoint law]
\label{cor:renyi-endpoint}
Let \(r_q\) be the Perron roots of \Cref{thm:collision-kernel}.  Then
\[
  \lim_{q\to\infty}\frac{1}{q}\log r_q = \frac12\log\varphi.
\]
Equivalently, the endpoint normalized R\'enyi loss satisfies
\[
  h_\infty^{\mathrm R}:=\lim_{q\to\infty}
  \frac{q\log 2-\log r_q}{q-1}
  =\log\left(\frac{2}{\sqrt{\varphi}}\right).
\]
\end{corollary}

\begin{proof}
Let \(D_m=\max_{x\in X_m}d_m(x)\).  For every \(q\ge 1\),
\[
  D_m^q\le S_q(m)\le \card{X_m}\,D_m^q=F_{m+2}D_m^q.
\]
Taking logarithms, dividing by \(m\), and letting \(m\to\infty\) gives
\[
  q\cdot\frac12\log\varphi
  \le \log r_q
  \le q\cdot\frac12\log\varphi+\log\varphi,
\]
by \Cref{prop:golden-mean-count,thm:max-fiber}.  Dividing by \(q\) and sending
\(q\to\infty\) proves the first limit, and the second is a reformulation.
\end{proof}
