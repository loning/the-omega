\subsection{Spectral-gap-driven analytic fingerprints: Mellin moment functions, limit theorems, and non-resonant large deviations}\label{subsec:operator-mellin-spectral-fingerprint}

This subsection extends the ``finite-state compilability'' conclusions to the analytic side: the limit distributions and characteristic/moment functions induced by finite-state systems all close on a family of finite-dimensional twisted operators; consequently, their continuation poles, spectral gaps, regularity, and large deviations can be compressed into finite spectral data and audited as finitely many determinant/spectral-radius certificates.

\subsubsection{Mellin fingerprint: meromorphic continuation of the matrix refinement equation as a rational function in \texorpdfstring{$z=\varphi^{-s}$}{z=phi(-s)}}

\begin{theorem}[Analytic Mellin realization of the matrix refinement equation: rational continuation and pole-spectral fingerprint in \texorpdfstring{$\varphi^{-s}$}{phi^{-s}}]\label{thm:conclusion62-mellin-rational}
Let \(\boldsymbol{\mu}=(\mu_1,\dots,\mu_d)\) be a finite vector measure on \([0,1]\) satisfying a system of graph-directed (matrix-directed) refinement/self-similarity equations:
\begin{equation}\label{eq:operator-mellin-matrix-refinement}
\boldsymbol{\mu}
=\sum_{j=1}^{J}T_j\,\bigl(\boldsymbol{\mu}\circ S_j^{-1}\bigr),
\qquad
S_j(x)=\varphi^{-a_j}x+t_j,
\end{equation}
where \(a_j\in\NN\) are fixed, \(t_j\in[0,1]\cap\ZZ[\varphi^{-1}]\), and \(T_j\in\Mat_{d\times d}(\RR_{\ge 0})\).
Let \(\nu\in\RR_{\ge 0}^d\) be a fixed left vector and define the scalar measure \(\mu:=\langle \nu,\boldsymbol{\mu}\rangle\).
For \(\Re(s)>-1\) define the Mellin--moment function
\[
M(s):=\int_{0}^{1}x^{s}\,d\mu(x).
\]
Then upon setting \(z=\varphi^{-s}\), \(M(s)\) can be expressed as a rational function \(R(z)\in\CC(z)\) on the half-plane \(\Re(s)>-1\), and thereby admits a meromorphic continuation.
Its pole set is contained in the zero set of the characteristic equation of a certain finite-dimensional ``moment refinement operator'':
there exist a matrix polynomial family \(\mathcal{A}(z)\in\Mat_{D\times D}(\CC[z])\) (where \(D\) depends only on \(d\), \(\{a_j\}\), \(\{t_j\}\), and the finite-state structure of the refinement graph) and a vector polynomial \(\mathbf{b}(z)\) such that
\begin{equation}\label{eq:operator-mellin-linear-system}
\mathbf{M}(s)=(I-\mathcal{A}(z))^{-1}\mathbf{b}(z),
\qquad
M(s)=\mathbf{c}^{\mathsf T}\mathbf{M}(s),
\end{equation}
whence all poles of \(M(s)\) are contained in the root set of \(\det(I-\mathcal{A}(z))=0\); moreover, the order of each pole is determined by the Jordan structure of \(\mathcal{A}(z)\) at the corresponding eigenvalue.
\end{theorem}

\begin{proof}
The key idea is to regard \eqref{eq:operator-mellin-matrix-refinement} as a ``scale substitution + linear output'' mechanism on a finite-state graph: under the sofic--Zeckendorf caliber, the refinement branches \(S_j\) and state updates are all finitely many \(\varphi\)-adic affine transformations, and the translation parameters \(t_j\) lie on a finitely generated \(\ZZ[\varphi^{-1}]\) lattice.
One can therefore select a finite Markov partition and introduce a set of \emph{restricted moment functions on partition components} as unknowns, obtaining a finite linear system whose coefficients are polynomials in \(z=\varphi^{-s}\); solving this system yields \eqref{eq:operator-mellin-linear-system}.
By Cramer's rule for finite-dimensional linear systems, \(\mathbf{M}(s)\) and \(M(s)\) are rational functions of \(z\) and admit meromorphic continuation; the pole locations and orders are given by the determinantal zeros and Jordan structure. \qed
\end{proof}

\subsubsection{Spectral-gap statistics: CLT in the spectral caliber and regularity from uniform complex-twist spectral gap}

\begin{theorem}[Spectral-gap corollary: CLT for additive observables and exponential decay at nonzero frequencies]\label{thm:conclusion63-clt-spectral}
Let \((X,\sigma,\mu)\) be the bilateral system of a primitive sofic shift under the Parry equilibrium measure \(\mu\), let \(f\) be a bounded additive observable (cylinder function), and write the partial sums
\(
S_m:=\sum_{k=0}^{m-1}f\circ\sigma^{k}
\).
Assume that \(f\) is non-coboundary (non-arithmetic) and that the corresponding twisted transfer operator \(\mathcal{L}_{\theta}\) (or equivalently the finite-dimensional twisted matrix family) satisfies analytic spectral perturbation and a spectral gap in a neighborhood of \(\theta=0\).
Then the standardized variable
\[
Y_m:=\frac{S_m-m\int f\,d\mu}{\sqrt{m}}
\]
satisfies the central limit theorem: under \(\mu\) it converges to a normal distribution with mean \(0\) and variance \(\sigma_f^2>0\) (where \(\sigma_f^2\) can be read off from the pressure Hessian/Green--Kubo formula).
\par
Furthermore, for any compact set \(K\Subset \RR\setminus\{0\}\), there exist constants \(C_K>0\) and \(\rho_K\in(0,1)\) such that the characteristic function exhibits uniform exponential decay at nonzero frequencies:
\[
\boxed{\;
\sup_{\theta\in K}\left|\mathbb{E}_{\mu}\,e^{\mathrm{i}\theta S_m}\right|
\le C_K\,\rho_K^{m}\qquad(m\ge 1).\;}
\]
This ``high-frequency exponential decay + zero-frequency second-order expansion'' combination provides the systematized spectral input for CLT/Edgeworth/moderate-deviation local expansions (and aligns with the branch-radius certificate interface in the appendix).
\end{theorem}

\begin{proof}
The CLT follows from the standard spectral method for finite-state Markov chains/transfer operators (same caliber as Proposition~\ref{prop:kernel-clt-ldp}).
The exponential decay comes from the uniform spectral gap of the twisted operator on \(\theta\in K\): the dominant spectral radius is strictly smaller than the Perron root at \(\theta=0\), so the \(m\)-step amplitude is controlled by a power of that spectral radius. \qed
\end{proof}

\begin{corollary}[Uniform complex-twist spectral gap \texorpdfstring{$\Rightarrow$}{=>} H\"older regularity of the limit density]\label{cor:conclusion64-holder-regularity}
Under the setting of Theorem~\ref{thm:conclusion63-clt-spectral}, further assume the existence of a uniform complex-twist spectral gap: for all \(\theta\neq 0\),
\[
\rho(\mathcal{L}_{\theta})\le (1-\delta)\,\rho(\mathcal{L}_0)
\]
(for some \(\delta>0\), and the twisted family is analytically controllable in a complex neighborhood).
Then the limit measure induced by the system is absolutely continuous on the corresponding real-line projection, and its density \(f=d\mu/dx\) belongs to some H\"older class \(C^{\alpha}([0,1])\).
Moreover, \(\alpha>0\) admits an explicit lower bound in terms of the spectral-gap constant \(\delta\) and the complex analytic bandwidth/contraction constants: there exists a computable function \(\alpha=\alpha(\delta,\mathcal{L})\) such that \(|f|_{C^{\alpha}}\) is explicitly controlled by the system data.
\end{corollary}

\begin{proof}
A uniform complex-twist spectral gap is equivalent to uniform quasi-compactness and spectral separation of the transfer operator on a H\"older (or suitable anisotropic Banach) space, whence the invariant distribution's density lies in that space and inherits H\"older regularity; the exponent \(\alpha\) is given by the standard interpolation/Doeblin--Fortet inequality from the spectral-radius contraction and analytic bandwidth. \qed
\end{proof}

\subsubsection{Non-resonant large deviations: Perron branch of exponential sums and the second-order curvature law}

\begin{proposition}[Uniform large deviations for exponential sums at non-resonant frequencies: the second-order curvature law of the spectral radius]\label{prop:conclusion65-expsums-uniform-ldp}
Let \(\mathcal{L}_m\subset X_m\) be the set of legal words of length \(m\), and set
\[
S_m(\theta):=\sum_{\omega\in\mathcal{L}_m}\exp\!\bigl(\mathrm{i}\theta\,\mathrm{Val}(\omega)\bigr).
\]
Let \(\lambda(\theta)\) denote the Perron dominant eigenvalue branch of the corresponding twisted transfer operator (\(\lambda(0)>0\)).
Then there exists a constant \(C>0\) such that the following holds for all \(\theta\) in some neighborhood:
\[
\boxed{\;
|S_m(\theta)|\le C\,|\lambda(\theta)|^{m}\qquad(m\ge 1).\;}
\]
Moreover, for any compact set \(K\Subset \RR\setminus\Theta_{\mathrm{res}}\) away from the resonance set
\(
\Theta_{\mathrm{res}}:=\{\theta:\ \rho(\mathcal{L}_{\theta})=\rho(\mathcal{L}_0)\}
\),
there exists \(c=c(K)>0\) such that
\[
\boxed{\;
\log|\lambda(\theta)|
\le \log\lambda(0)-c\,\mathrm{dist}(\theta,\Theta_{\mathrm{res}})^{2}
\qquad(\theta\in K).\;}
\]
This second-order curvature law yields uniform exponential decay away from resonances, and serves as auditable input for major-arc/resonance discrimination.
\end{proposition}

\begin{proof}
The first bound follows from the spectral decomposition of the twisted transfer operator: the exponential sum is equivalent to the \(m\)-fold action on the constant function, and its leading term is controlled by the Perron branch \(\lambda(\theta)^{m}\).
The second bound comes from the analytic strict concavity of \(\theta\mapsto \log\lambda(\theta)\): outside the resonance set the dominant spectral radius strictly decreases, and on a compact set a uniform curvature lower bound is obtained via second-order perturbation and compactness. \qed
\end{proof}
