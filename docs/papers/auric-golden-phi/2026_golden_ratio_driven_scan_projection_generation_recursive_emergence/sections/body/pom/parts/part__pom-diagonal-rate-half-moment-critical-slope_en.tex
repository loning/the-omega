\begin{corollary}[Small Distortion Saving Law under Folding Semantics: Leading-Order Slope Constant of \texorpdfstring{$q=\tfrac12$}{q=1/2} Partition]\label{cor:pom-diagonal-rate-small-distortion-saving-half-moment-fold}
Under the folding semantics, for each resolution \(m\) let the pushforward distribution \(w_m(x)=d_m(x)/2^m\), and define the half-order partition and its Hellinger constant
\[
S_{1/2}(m):=\sum_{x\in X_m} d_m(x)^{1/2},\qquad
A_m:=A_{1/2}(w_m)=\sum_{x\in X_m}\sqrt{w_m(x)}=2^{-m/2}S_{1/2}(m).
\]
Let the information saving
\[
\Delta_m(\delta):=H(w_m)-R_{w_m}(\delta).
\]
Then as \(\delta\downarrow 0\), for each fixed \(m\) we have
\[
\Delta_m(\delta)=\delta\log\frac{A_m^2-1}{\delta}+O(\delta),
\]
where the constant in \(O(\delta)\) can be taken as a bounded quantity depending only on \(w_m\).
\par
Furthermore, take any sequence \(\delta_m\downarrow 0\) satisfying \(\log(1/\delta_m)=o(m)\).
If the limit
\[
a_{1/2}:=\lim_{m\to\infty}\frac{1}{m}\log(A_m^2-1)
\]
exists (allowing \(0\)), then the ratio limit holds
\[
\lim_{m\to\infty}\frac{\Delta_m(\delta_m)}{m\,\delta_m}=a_{1/2}.
\]
If we further assume the half-order pressure limit
\[
\tau(1/2):=\lim_{m\to\infty}\frac{1}{m}\log S_{1/2}(m)
\]
exists and \(a_{1/2}>0\), then we have the explicit identity
\[
a_{1/2}=2\tau(1/2)-\log 2.
\]
\end{corollary}
\begin{proof}
For fixed \(m\), apply Theorem \ref{thm:pom-diagonal-rate-small-distortion-hellinger-endpoint} to \(w=w_m\), and take the difference
\(
\Delta_m(\delta)=H(w_m)-R_{w_m}(\delta)
\)
to obtain the first formula; where \(A_{1/2}(w_m)=A_m\).
\par
Substituting \(\delta=\delta_m\) and dividing by \(m\delta_m\), we get
\[
\frac{\Delta_m(\delta_m)}{m\delta_m}
=\frac{1}{m}\log(A_m^2-1)-\frac{1}{m}\log\delta_m+O\!\left(\frac{1}{m}\right).
\]
From \(\log(1/\delta_m)=o(m)\) we get \(\frac{1}{m}\log\delta_m\to 0\), and combining with \(O(1/m)\to 0\) as \(m\to\infty\), thus if \(\frac{1}{m}\log(A_m^2-1)\to a_{1/2}\) the ratio limit holds.
\par
If \(a_{1/2}>0\), then \(A_m^2\) must diverge exponentially, thus
\(
\log(A_m^2-1)=\log A_m^2+o(m)
\).
From \(A_m=2^{-m/2}S_{1/2}(m)\) we obtain
\[
\frac{1}{m}\log A_m^2
=\frac{2}{m}\log S_{1/2}(m)-\log 2
\ \longrightarrow\ 2\tau(1/2)-\log 2,
\]
hence the identity.
\end{proof}

\begin{theorem}[Extreme Lower Bound of Half-Order Pressure and Square Root Freezing Criterion]\label{thm:pom-half-order-pressure-extreme-lower-bound-sqrt-freezing}
For each \(m\), write the maximum fiber \(M_m:=\max_{x\in X_m}d_m(x)\), the maximum set
\(
\mathcal{M}_m:=\{x\in X_m:\ d_m(x)=M_m\}
\)
and its cardinality \(K_m:=|\mathcal{M}_m|\).
Suppose the exponential rate limits
\[
\alpha_*:=\lim_{m\to\infty}\frac{1}{m}\log M_m,\qquad
g_*:=\lim_{m\to\infty}\frac{1}{m}\log K_m
\]
exist (the \(\liminf\) version also holds).
Then we always have the extreme lower bound
\[
\liminf_{m\to\infty}\frac{1}{m}\log S_{1/2}(m)\ \ge\ g_*+\frac{1}{2}\alpha_*,
\]
equivalently (using \(A_m=2^{-m/2}S_{1/2}(m)\))
\[
\liminf_{m\to\infty}\frac{1}{m}\log A_m\ \ge\ g_*+\frac{1}{2}\bigl(\alpha_*-\log 2\bigr).
\]
\par
The following two statements are equivalent:
\begin{enumerate}
  \item \(\displaystyle S_{1/2}(m)=K_m M_m^{1/2}\bigl(1+o(1)\bigr)\) (half-order partition is dominated by the maximum layer);
  \item \(\displaystyle \sum_{x\notin\mathcal{M}_m} d_m(x)^{1/2}=o\bigl(K_m M_m^{1/2}\bigr)\).
\end{enumerate}
When the above equivalence holds, if we further satisfy \(2g_*+\alpha_*>\log 2\), then
\[
a_{1/2}:=\lim_{m\to\infty}\frac{1}{m}\log(A_m^2-1)
\]
exists and satisfies
\[
a_{1/2}=2g_*+\alpha_*-\log 2.
\]
\end{theorem}
\begin{proof}
From \(S_{1/2}(m)\ge \sum_{x\in\mathcal{M}_m} M_m^{1/2}=K_m M_m^{1/2}\) we get
\[
\frac{1}{m}\log S_{1/2}(m)\ge \frac{1}{m}\log K_m+\frac{1}{2m}\log M_m,
\]
taking \(\liminf\) yields the extreme lower bound; substituting \(A_m=2^{-m/2}S_{1/2}(m)\) gives the equivalent form.
\par
Let
\(
T_m:=\sum_{x\notin\mathcal{M}_m} d_m(x)^{1/2}
\), then \(S_{1/2}(m)=K_m M_m^{1/2}+T_m\).
Thus
\[
\frac{S_{1/2}(m)}{K_m M_m^{1/2}}=1+\frac{T_m}{K_m M_m^{1/2}},
\]
hence \(S_{1/2}(m)=K_m M_m^{1/2}(1+o(1))\) if and only if \(T_m=o(K_m M_m^{1/2})\).
\par
When the above equivalence holds,
\[
A_m^2
=\frac{S_{1/2}(m)^2}{2^m}
=\frac{K_m^2 M_m}{2^m}\,(1+o(1)).
\]
If \(2g_*+\alpha_*-\log 2>0\), then \(A_m^2\to\infty\), and
\[
\frac{1}{m}\log(A_m^2-1)
=\frac{1}{m}\log A_m^2+o(1)
=2g_*+\alpha_*-\log 2+o(1),
\]
hence \(a_{1/2}\) exists and equals the indicated constant.
\end{proof}

\begin{corollary}[Endpoint Slope of the Critical Plane: Inverted from \texorpdfstring{$q=\tfrac12$}{q=1/2} Pressure]\label{cor:pom-query-distortion-critical-slope-inverts-half-pressure}
For each resolution \(m\) and \(\beta\in[0,1)\), define the finite-window critical bit density
\[
b_{\mathrm{crit}}^{(m)}(\beta,\delta):=\frac{(1-\beta)\,R_{w_m}(\delta)}{\log 2},\qquad \delta\in[0,\delta_0(m)),\qquad \delta_0(m):=1-\sum_x w_m(x)^2.
\]
Then as \(\delta\downarrow 0\) we have
\[
\frac{\partial}{\partial \delta}\,b_{\mathrm{crit}}^{(m)}(\beta,\delta)
=-\frac{1-\beta}{\log 2}\,\log\frac{A_m^2-1}{\delta}+O(1),
\]
where the constant in \(O(1)\) can be taken as a bounded quantity depending only on \(w_m\).
Thus on any diagonal scale satisfying \(\delta_m\downarrow 0\), \(\log(1/\delta_m)=o(m)\),
\[
\lim_{m\to\infty}\frac{-\partial_\delta b_{\mathrm{crit}}^{(m)}(\beta,\delta_m)}{m}
=\frac{1-\beta}{\log 2}\,a_{1/2},
\]
where \(a_{1/2}\) is as in Corollary \ref{cor:pom-diagonal-rate-small-distortion-saving-half-moment-fold}.
If we further have \(\tau(1/2)\) exists and \(a_{1/2}>0\), then we also have
\[
\lim_{m\to\infty}\frac{-\partial_\delta b_{\mathrm{crit}}^{(m)}(\beta,\delta_m)}{m}
=\frac{1-\beta}{\log 2}\,\bigl(2\tau(1/2)-\log 2\bigr).
\]
\end{corollary}
\begin{proof}
By definition \(b_{\mathrm{crit}}^{(m)}=(1-\beta)R_{w_m}(\delta)/\log 2\) and
\(
R'_{w_m}(\delta)=-\lambda_m(\delta)
\)
(Theorem \ref{thm:pom-diagonal-rate-explicit-param-and-derivative}), we get
\[
\partial_\delta b_{\mathrm{crit}}^{(m)}(\beta,\delta)
=-\frac{1-\beta}{\log 2}\,\lambda_m(\delta).
\]
By the dual expansion of Theorem \ref{thm:pom-diagonal-rate-small-distortion-hellinger-endpoint} (taking \(w=w_m\)) as \(\delta\downarrow 0\)
\(
\lambda_m(\delta)=\log\frac{A_m^2-1}{\delta}+O(1)
\)
yields the first formula.
\par
Substituting \(\delta=\delta_m\) and dividing by \(m\), and using \(\log(1/\delta_m)=o(m)\) to eliminate the \(\frac{1}{m}\log(1/\delta_m)\) term, we obtain the second formula;
the last formula follows by substituting the identity \(a_{1/2}=2\tau(1/2)-\log 2\) from Corollary \ref{cor:pom-diagonal-rate-small-distortion-saving-half-moment-fold} (under \(a_{1/2}>0\)).
\end{proof}
