\paragraph{平面四次曲线三次投影的分歧谱、挠元约束与算术刚性}
设
\[
C_F\subset \PP^2_{[\mu:u:w]}
\]
由齐次四次式
\[
F_{\mathrm{hom}}(\mu,u,w)
=
\mu^4-\mu^3w-2\mu^2uw-\mu^2w^2-\mu u^3+2\mu u w^2+2u w^3
\]
定义，并假定 \(C_F\) 光滑。
记两个有理无穷远点
\[
P_\infty^{(0)}=[0:1:0],\qquad P_\infty^{(1)}=[1:1:0].
\]

\begin{theorem}[投影 \texorpdfstring{$\varphi_0$}{phi0} 的分歧判别式分解与 Hurwitz 画像]\label{thm:xi-cf-phi0-discriminant-hurwitz-profile}
定义
\[
\varphi_0:C_F\to\PP^1,\qquad
\varphi_0([\mu:u:w])=[\mu:w],
\]
并在仿射图 \(w=1\) 记 \(t=\mu\)，则纤维由三次多项式
\[
f_t(u):=F_{\mathrm{hom}}(t,u,1)
=
t^4-t^3-t^2-2t^2u-tu^3+2tu+2u
\]
给出。其关于 \(u\) 的判别式满足
\[
\operatorname{Disc}_u(f_t)
=
-t(3t+2)(t^2-t-1)^2\bigl(9t^4-6t^3+4t^2+8t-16\bigr).
\]
因此 \(\varphi_0\) 的分歧值为
\[
t=0,\quad
t=-\frac23,\quad
t=\frac{1\pm\sqrt5}{2},\quad
9t^4-6t^3+4t^2+8t-16=0.
\]
其中 \(t=\frac{1\pm\sqrt5}{2}\) 为全分歧（\(e=3\)），其余均为单分歧（\(e=2\)）。
故 Hurwitz 画像为
\[
(3,3,2,2,2,2,2,2).
\]
\end{theorem}
\begin{proof}
分歧条件等价于 \(f_t\) 出现重根，即 \(\operatorname{Disc}_u(f_t)=0\)。
由
\[
\operatorname{Disc}_u(f_t)
=
(-1)^{3(3-1)/2}\frac{\operatorname{Res}_u(f_t,\partial_u f_t)}{\mathrm{lc}_u(f_t)}
\]
直接消元化简得到上述因子分解。

若 \(t^2-t-1=0\)，代回得
\[
f_t(u)=-t\,u^3,
\]
故 \(u=0\) 为三重根，对应全分歧点
\[
P_\pm=\left[\frac{1\pm\sqrt5}{2}:0:1\right].
\]
其余因子处仅出现判别式一阶零点，且不出现三重根。
再对
\[
Q(t):=9t^4-6t^3+4t^2+8t-16
\]
计算
\[
\operatorname{Disc}(Q)=-2^{12}\cdot 3^4\cdot 1931\neq 0,
\]
故四个根互异且对应单分歧。
综上得到所述分歧型。证毕。
\end{proof}

\begin{corollary}[两处全分歧诱导的 \texorpdfstring{$\QQ(\sqrt5)$}{Q(sqrt5)}-有理三挠元]\label{cor:xi-cf-jacobian-qsqrt5-3torsion}
令
\[
P_+=\left[\frac{1+\sqrt5}{2}:0:1\right],\qquad
P_-=\left[\frac{1-\sqrt5}{2}:0:1\right]\in C_F\bigl(\QQ(\sqrt5)\bigr).
\]
在 Jacobian \(J(C_F)\) 上定义
\[
\tau:=[P_+-P_-]\in J(C_F)\bigl(\QQ(\sqrt5)\bigr).
\]
则 \(\tau\neq 0\)、\(3\tau=0\)，从而 \(\operatorname{ord}(\tau)=3\)。
\end{corollary}
\begin{proof}
由定理 \ref{thm:xi-cf-phi0-discriminant-hurwitz-profile}，
\[
\varphi_0^{-1}\!\left(\frac{1+\sqrt5}{2}\right)=3P_+,\qquad
\varphi_0^{-1}\!\left(\frac{1-\sqrt5}{2}\right)=3P_-.
\]
同一映射的两条纤维线性等价，故 \(3P_+\sim 3P_-\)，即 \(3\tau=0\)。

若 \(\tau=0\)，则存在 \(g\in \QQ(\sqrt5)(C_F)^\times\) 使
\[
\operatorname{div}(g)=P_+-P_-,
\]
从而给出次数 \(1\) 的态射 \(C_F\to\PP^1\)，迫使 \(C_F\) 为有理曲线，矛盾于光滑平面四次曲线亏格 \(3\)。
故 \(\tau\neq0\)，且阶必为 \(3\)。证毕。
\end{proof}

\begin{theorem}[投影 \texorpdfstring{$\varphi_1$}{phi1} 的九次分歧多项式与伽罗瓦群 \texorpdfstring{$S_9$}{S9}]\label{thm:xi-cf-phi1-p9-galois-s9}
定义
\[
\varphi_1:C_F\to\PP^1,\qquad
\varphi_1([\mu:u:w])=[\mu-u:w].
\]
在仿射图 \(w=1\) 置 \(s=\mu-u\)，则纤维由
\[
g_s(u):=F_{\mathrm{hom}}(u+s,u,1)
\]
给出，并且
\[
\operatorname{Disc}_u(g_s)=-(s-1)\,P_9(s),
\]
其中
\[
P_9(s)=3s^9+3s^8-31s^7-83s^6+100s^5+424s^4-16s^3-436s^2-52s+100.
\]
并满足：
\begin{enumerate}
  \item \(\varphi_1\) 的 Hurwitz 画像为
  \[
  (2,2,2,2,2,2,2,2,2,2);
  \]
  \item \(P_9\) 在 \(\QQ\) 上不可约，且
  \[
  \operatorname{Gal}(P_9/\QQ)\cong S_9.
  \]
\end{enumerate}
\end{theorem}
\begin{proof}
分歧条件 \(\operatorname{Disc}_u(g_s)=0\) 由结果式计算得到上述分解。
再计算
\[
\operatorname{Disc}(P_9)=2^{22}\cdot 3\cdot 5^2\cdot 13^6\cdot 1931\cdot 5801^3\neq0,
\]
故 \(P_9\) 无重根，且 \(s=1\) 亦为一阶零点。
因此所有分歧均为二重点型 \(e=2\)。
由 Riemann--Hurwitz：
\[
2g(C_F)-2=3(2g(\PP^1)-2)+\sum(e_P-1)
\]
即
\[
4=-6+\sum(e_P-1),
\]
故 \(\sum(e_P-1)=10\)，从而恰有 \(10\) 个单分歧值，得到 \((2^{10})\)。

对伽罗瓦群：模 \(17\) 约化分解型为 \([9]\)，故 \(P_9\) 在 \(\QQ\) 上不可约。
又判别式非平方，故伽罗瓦群不含于 \(A_9\)。
模 \(11\) 分解型 \([8,1]\) 给出含 \((8)(1)\) 的 Frobenius 元，故作用原始；
模 \(5\) 分解型 \([5,1,1,1,1]\) 给出 \(5\)-循环。
由 Jordan 定理，原始子群含此类素循环则包含 \(A_9\)；
结合“不含于 \(A_9\)”即得 \(\operatorname{Gal}(P_9/\QQ)=S_9\)。证毕。
\end{proof}

\begin{corollary}[\texorpdfstring{$\varphi_0$}{phi0} 与 \texorpdfstring{$\varphi_1$}{phi1} 在 Hurwitz 空间中的严格分离]\label{cor:xi-cf-hurwitz-strict-separation-phi0-phi1}
不存在 \(\alpha\in\operatorname{Aut}(C_F)\) 与 \(\beta\in\operatorname{PGL}_2\) 使
\[
\varphi_1=\beta\circ\varphi_0\circ\alpha.
\]
\end{corollary}
\begin{proof}
定理 \ref{thm:xi-cf-phi0-discriminant-hurwitz-profile} 给出 \(\varphi_0\) 的分歧型 \((3,3,2^6)\)；
定理 \ref{thm:xi-cf-phi1-p9-galois-s9} 给出 \(\varphi_1\) 的分歧型 \((2^{10})\)。
分歧画像在 \(\operatorname{Aut}(C_F)\times\operatorname{PGL}_2\) 等价下不变，故二者不等价。证毕。
\end{proof}

\begin{theorem}[良约化素数 \texorpdfstring{$7,11$}{7,11} 处的 \texorpdfstring{$L$}{L} 多项式与有理三挠消失]\label{thm:xi-cf-lpolys-7-11-and-rational-3torsion-vanishing}
曲线 \(C_F\) 在 \(p=7,11\) 处良约化，且其局部 \(L\)-多项式分别为
\[
P_7(T)=1+4T+9T^2+9T^3+63T^4+196T^5+343T^6,
\]
\[
P_{11}(T)=1+7T+35T^2+124T^3+385T^4+847T^5+1331T^6.
\]
因此
\[
\#J(C_F)(\FF_7)=P_7(1)=625=5^4,\qquad
\#J(C_F)(\FF_{11})=P_{11}(1)=2730.
\]
尤其 \(3\nmid \#J(C_F)(\FF_7)\)，故
\[
J(C_F)(\QQ)[3]=0.
\]
\end{theorem}
\begin{proof}
对 \(p=7,11\) 分别计数
\[
N_n:=\#C_F(\FF_{p^n})\qquad(n=1,2,3),
\]
置
\[
S_n:=p^n+1-N_n.
\]
由 Newton 恒等式可恢复亏格 \(3\) 情形的 \(L\)-多项式系数，再由函数方程对称性补全，得到所示 \(P_7,P_{11}\)。
代入 \(T=1\) 即得有限域 Jacobian 阶。

对有理三挠：良约化与 \(3\neq 7\) 给出
\[
J(C_F)(\QQ)[3]\hookrightarrow J(C_F)(\FF_7)[3].
\]
但 \(3\nmid \#J(C_F)(\FF_7)=625\)，故右侧平凡，从而左侧平凡。证毕。
\end{proof}
