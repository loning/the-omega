\paragraph{window-\texorpdfstring{$6$}{6} 的根--权交会刚性、orbit--Levi 骨架与二层 boundary 种子的 \texorpdfstring{$\ZZ_6$}{Z6} 主齐性}
在四步 uplift 同构、boundary-\(10\) 的根--Cartan 四块分裂、window-\(6\) 的 \(C_6\) 轨道骨架，以及二层 boundary 种子的置换表示与外加信息下界已经固定之后，还可把 weight-layer、orbit--Levi 与 sheet parity 三条离散接口进一步压缩为下列四条后果。它们分别给出 \(9+6+3+3\) 与 \(6+6+6+3\) 的唯一共同细化、\(15+3+3\) 与根--Cartan 字典的唯一交会矩阵、二层 boundary 种子的 \(\ZZ_6\)-主齐性，以及等变编码的阶六阻碍。

\begin{theorem}[根--权双分块的五原子共同细化]\label{thm:xi-time-part53da-root-weight-common-refinement-five-atoms}
沿用定理 \ref{thm:boundary-shift4-uplift-isomorphism} 的 uplift 同构
\[
\iota_6:X_6\xrightarrow{\sim}X_{10}^{\mathrm{bdry}},
\qquad
\iota_6(v)=10\,v\,01,
\]
并沿用表 \ref{tab:window6_patisalam_9633_partition} 的四块分解
\[
X_6=X_6^{(2)}\sqcup X_6^{(1)}\sqcup X_6^{(L)}\sqcup X_6^{(R)},
\qquad
|X_6^{(2)}|=9,\quad |X_6^{(1)}|=6,\quad |X_6^{(L)}|=3,\quad |X_6^{(R)}|=3,
\]
其中 \(X_6^{(R)}=X_6^{\mathrm{bdry}}\)。再把定理 \ref{thm:boundary-m10-b3c3-rigid-four-block-split} 给出的根--Cartan 分裂记为
\[
X_{10}^{\mathrm{bdry}}
=
X_{10}^{\mathrm{short}}
\sqcup
X_{10}^{A_2(+)}
\sqcup
X_{10}^{A_2(-)}
\sqcup
X_{10}^{\mathrm{Cartan}},
\]
其中各块大小依次为 \(6,6,6,3\)。则经 \(\iota_6\) 识别后，两种分块的交会矩阵严格为
\[
\begin{array}{c|cccc}
& \mathrm{short} & A_2^{(+)} & A_2^{(-)} & \mathrm{Cartan}\\
\hline
X_6^{(2)} & 0 & 3 & 6 & 0\\
X_6^{(1)} & 6 & 0 & 0 & 0\\
X_6^{(L)} & 0 & 3 & 0 & 0\\
X_6^{(R)} & 0 & 0 & 0 & 3
\end{array}
\]
从而这两种分块的共同细化恰有五个非空原子，其大小为
\[
6\ \oplus\ 3\ \oplus\ 6\ \oplus\ 3\ \oplus\ 3.
\]
\end{theorem}
\begin{proof}
定理 \ref{thm:boundary-shift4-uplift-isomorphism} 允许把 \(X_6\) 与 \(X_{10}^{\mathrm{bdry}}\) 作无歧义识别；表 \ref{tab:window6_patisalam_9633_partition} 给出四个 weight-layer 块的显式词表，而表 \ref{tab:boundary_uplift_m10_dictionary} 给出每个 \(v\in X_6\) 在 uplift 后所属的根型。

首先，\(X_6^{(1)}\) 的六个词
\[
\texttt{100000},\ \texttt{010000},\ \texttt{001000},\ \texttt{000100},\ \texttt{000010},\ \texttt{000001}
\]
在表 \ref{tab:boundary_uplift_m10_dictionary} 中全部标记为 \(\mathrm{short}\)，故
\[
\bigl|\iota_6(X_6^{(1)})\cap X_{10}^{\mathrm{short}}\bigr|=6.
\]

其次，
\[
X_6^{(L)}=\{\texttt{000000},\ \texttt{010101},\ \texttt{101010}\}
\]
在同一字典中全部落入 \(A_2^{(+)}\)，故
\[
\bigl|\iota_6(X_6^{(L)})\cap X_{10}^{A_2(+)}\bigr|=3.
\]

再看
\[
X_6^{(2)}=
\{\texttt{000101},\texttt{001001},\texttt{001010},\texttt{010001},\texttt{010010},\texttt{010100},\texttt{100010},\texttt{100100},\texttt{101000}\}.
\]
由表 \ref{tab:boundary_uplift_m10_dictionary}，其中
\[
\texttt{010100},\ \texttt{100010},\ \texttt{010001}
\]
恰落入 \(A_2^{(+)}\)，而其余六个
\[
\texttt{101000},\ \texttt{100100},\ \texttt{010010},\ \texttt{001010},\ \texttt{001001},\ \texttt{000101}
\]
恰落入 \(A_2^{(-)}\)。因此
\[
\bigl|\iota_6(X_6^{(2)})\cap X_{10}^{A_2(+)}\bigr|=3,
\qquad
\bigl|\iota_6(X_6^{(2)})\cap X_{10}^{A_2(-)}\bigr|=6.
\]

最后，\(X_6^{(R)}=X_6^{\mathrm{bdry}}\) 的三个词
\[
\texttt{100001},\ \texttt{101001},\ \texttt{100101}
\]
在表 \ref{tab:boundary_uplift_m10_dictionary} 中全部标记为 \(\mathrm{Cartan}\)，故
\[
\bigl|\iota_6(X_6^{(R)})\cap X_{10}^{\mathrm{Cartan}}\bigr|=3.
\]

由于以上已经用尽四列的全部总数 \(6,6,6,3\)，其余交块只能为空。于是共同细化恰由五个非空交块组成，其大小正是矩阵中的五个非零条目。证毕。
\end{proof}

\begin{theorem}[orbit--Levi 骨架与根--Cartan 分裂的交会矩阵唯一化]\label{thm:xi-time-part53da-orbit-levi-root-cartan-intersection}
沿用表 \ref{tab:window6_c6_orbit_decomposition} 的 \(C_6\) 轨道分解，并记
\[
X^{\mathrm{cyc},3}_6:=\mathcal{O}(\texttt{001001})
=
\{\texttt{001001},\texttt{010010},\texttt{100100}\},
\qquad
X^{\mathrm{cyc},15}_6:=X_6^{\mathrm{cyc}}\setminus X^{\mathrm{cyc},3}_6.
\]
则在
\[
X_6=X^{\mathrm{cyc},15}_6\sqcup X^{\mathrm{cyc},3}_6\sqcup X_6^{\mathrm{bdry}}
\qquad (21=15\oplus 3\oplus 3)
\]
与定理 \ref{thm:boundary-m10-b3c3-rigid-four-block-split} 的根--Cartan 分裂之间，交会矩阵严格为
\[
\begin{array}{c|cccc}
& \mathrm{short} & A_2^{(+)} & A_2^{(-)} & \mathrm{Cartan}\\
\hline
X^{\mathrm{cyc},15}_6 & 6 & 6 & 3 & 0\\
X^{\mathrm{cyc},3}_6  & 0 & 0 & 3 & 0\\
X_6^{\mathrm{bdry}}   & 0 & 0 & 0 & 3
\end{array}
\]
因而经 \(\iota_6\) 识别后，同时对 orbit--Levi 骨架与根--Cartan 分裂都常值的复值可观测量空间恰为四维，并由四个非空交块的指示函数张成。
\end{theorem}
\begin{proof}
表 \ref{tab:window6_c6_orbit_decomposition} 表明，唯一的三元轨道正是
\[
X^{\mathrm{cyc},3}_6
=
\{\texttt{001001},\texttt{010010},\texttt{100100}\}.
\]
再由表 \ref{tab:boundary_uplift_m10_dictionary}，这三个词分别对应
\[
(0,1,-1),\qquad (1,0,-1),\qquad (-1,1,0),
\]
因而全部落在 \(A_2^{(-)}\)。故
\[
\bigl|\iota_6(X^{\mathrm{cyc},3}_6)\cap X_{10}^{A_2(-)}\bigr|=3.
\]

另一方面，\(X_6^{\mathrm{bdry}}\) 的三个词在同一字典中全部对应 \(\mathrm{Cartan}\) 方向，因此
\[
\bigl|\iota_6(X_6^{\mathrm{bdry}})\cap X_{10}^{\mathrm{Cartan}}\bigr|=3.
\]

由定理 \ref{thm:xi-time-part53da-root-weight-common-refinement-five-atoms}，整个 cyclic 扇区与根--Cartan 字典的交会总数为
\[
6\ \mathrm{short}\ +\ 6\ A_2^{(+)}\ +\ 6\ A_2^{(-)}.
\]
从中扣除 \(X^{\mathrm{cyc},3}_6\) 已经占去的 \(3\) 个 \(A_2^{(-)}\) 方向，剩余的 \(X^{\mathrm{cyc},15}_6\) 便必然恰好分布为
\[
6\ \mathrm{short}\ +\ 6\ A_2^{(+)}\ +\ 3\ A_2^{(-)}.
\]
于是得到所示交会矩阵。

最后，一个函数同时在 orbit--Levi 三块与根--Cartan 四块上常值，当且仅当它在全部非空交块上常值。由于这里只有四个非空交块，这样的可观测量空间便恰为四维，并由四个交块的指示函数张成。证毕。
\end{proof}

\begin{theorem}[二层 boundary 种子的 \texorpdfstring{$\ZZ_6$}{Z6} 主齐性]\label{thm:xi-time-part53da-boundary-z6-torsor}
沿用定理 \ref{thm:xi-window6-boundary-two-sheet-permutation-decomposition} 的记号。记
\[
\widetilde X_6^{\mathrm{bdry}}
:=
\bigsqcup_{w\in X_6^{\mathrm{bdry}}}
\bigl(\Fold_6^{\mathrm{bin}}\bigr)^{-1}(w),
\]
并令
\[
\zeta:\widetilde X_6^{\mathrm{bdry}}\to \widetilde X_6^{\mathrm{bdry}}
\]
为每条两点纤维上的全局 sheet-flip。再取 \(C_3\le S_3\) 为唯一的三阶子群，则 \(\langle\zeta\rangle\) 与 \(C_3\) 在 \(\widetilde X_6^{\mathrm{bdry}}\) 上诱导一个自由且传递的作用。特别地，
\[
\langle\zeta\rangle\times C_3
\cong
\ZZ_2\times \ZZ_3
\cong
\ZZ_6,
\]
从而 \(\widetilde X_6^{\mathrm{bdry}}\) 自然成为一个 \(\ZZ_6\)-torsor，并且作为复置换模有
\[
\CC[\widetilde X_6^{\mathrm{bdry}}]\cong \CC[\ZZ_6].
\]
\end{theorem}
\begin{proof}
定理 \ref{thm:xi-window6-boundary-two-sheet-permutation-decomposition} 已经给出两条结构：
\begin{enumerate}
  \item \(\widetilde X_6^{\mathrm{bdry}}\) 由三条两点纤维组成，因此
  \[
  |\widetilde X_6^{\mathrm{bdry}}|=6;
  \]
  \item 基底三点集 \(X_6^{\mathrm{bdry}}\) 的置换群 \(S_3\) 自然作用于 \(\widetilde X_6^{\mathrm{bdry}}\)，而 \(\zeta\) 在每条纤维内交换两张 sheet。
\end{enumerate}
由于 \(C_3\le S_3\) 只负责循环置换三条 boundary 纤维，而 \(\zeta\) 只负责在固定纤维内翻转两张 sheet，二者可交换，故确有直积作用
\[
\langle\zeta\rangle\times C_3\curvearrowright \widetilde X_6^{\mathrm{bdry}}.
\]

该作用是传递的：给定任意两点 \(x,y\in \widetilde X_6^{\mathrm{bdry}}\)，先用 \(C_3\) 把 \(x\) 所在纤维送到 \(y\) 所在纤维；若此时 sheet 标号不同，再施加一次 \(\zeta\) 即可把像点送到 \(y\)。因此
\[
\langle\zeta\rangle\times C_3
\]
在六点集 \(\widetilde X_6^{\mathrm{bdry}}\) 上作用传递。

另一方面，
\[
\bigl|\langle\zeta\rangle\times C_3\bigr|=2\cdot 3=6=|\widetilde X_6^{\mathrm{bdry}}|.
\]
有限群在同基数集合上的传递作用自动自由，故该作用自由且传递。于是 \(\widetilde X_6^{\mathrm{bdry}}\) 是一个 \(\langle\zeta\rangle\times C_3\)-torsor。再由
\[
\ZZ_2\times\ZZ_3\cong \ZZ_6
\]
便得到 \(\widetilde X_6^{\mathrm{bdry}}\) 是 \(\ZZ_6\)-torsor。

最后，任一有限群在自身 torsor 上的置换表示都与其正则表示同构，因此
\[
\CC[\widetilde X_6^{\mathrm{bdry}}]\cong \CC[\ZZ_6].
\]
证毕。
\end{proof}

\begin{corollary}[保持全部内生对称的 boundary 编码的阶六判据]\label{cor:xi-time-part53da-equivariant-boundary-coding-order-six}
沿用定理 \ref{thm:xi-time-part53da-boundary-z6-torsor} 的 \(\ZZ_6\)-作用。设 \(A\) 为有限阿贝尔群，则存在单射
\[
\Phi:\widetilde X_6^{\mathrm{bdry}}\longrightarrow A
\]
保持上述 \(\ZZ_6\)-作用，当且仅当 \(A\) 含有一个阶 \(6\) 元；等价地，当且仅当 \(A\) 含有一个循环子群同构于 \(\ZZ_6\)。特别地：
\begin{enumerate}
  \item 对任意 \(r\ge 1\)，\(A\cong (\ZZ/2\ZZ)^r\) 时不存在这样的等变单射；
  \item 对任意 \(r\ge 1\)，\(A\cong (\ZZ/3\ZZ)^r\) 时不存在这样的等变单射；
  \item 在全部有限阿贝尔目标中，最小可行者恰为 \(\ZZ_6\cong \ZZ_2\times \ZZ_3\)。
\end{enumerate}
\end{corollary}
\begin{proof}
把定理 \ref{thm:xi-time-part53da-boundary-z6-torsor} 中的作用群识别为 \(\ZZ_6\)。若 \(\Phi\) 等变，则存在群同态
\[
\rho:\ZZ_6\to A
\]
使得
\[
\Phi(g\cdot x)=\rho(g)+\Phi(x)
\qquad
(g\in \ZZ_6,\ x\in \widetilde X_6^{\mathrm{bdry}}).
\]
固定任意 \(x_0\in \widetilde X_6^{\mathrm{bdry}}\)。由于 \(\widetilde X_6^{\mathrm{bdry}}\) 是 \(\ZZ_6\)-torsor，其轨道恰有 \(6\) 个点，故单射 \(\Phi\) 迫使
\[
\{\Phi(g\cdot x_0):\ g\in \ZZ_6\}
\]
也有 \(6\) 个不同元素。由等变性，
\[
\Phi(g\cdot x_0)=\rho(g)+\Phi(x_0),
\]
故该像轨道正是子群 \(\rho(\ZZ_6)\) 的一个平移副本；其大小只能是 \(|\rho(\ZZ_6)|\)。于是
\[
|\rho(\ZZ_6)|=6,
\]
从而 \(\rho(1)\in A\) 的阶恰为 \(6\)。

反之，若 \(A\) 含有一个阶 \(6\) 元 \(a\)，则取同态
\[
\rho:\ZZ_6\to A,\qquad \rho(1)=a.
\]
固定基点 \(x_0\in \widetilde X_6^{\mathrm{bdry}}\)，并对任意 \(g\in \ZZ_6\) 定义
\[
\Phi(g\cdot x_0):=\rho(g).
\]
由于 \(\widetilde X_6^{\mathrm{bdry}}\) 上的 \(\ZZ_6\)-作用自由且传递，该定义良好并且自动等变；又因 \(a\) 的阶为 \(6\)，\(\rho\) 为单射，故 \(\Phi\) 亦为单射。

最后，\((\ZZ/2\ZZ)^r\) 的每个元素阶都整除 \(2\)，\((\ZZ/3\ZZ)^r\) 的每个元素阶都整除 \(3\)，因此都不含阶 \(6\) 元；而 \(\ZZ_6\cong\ZZ_2\times\ZZ_3\) 显然含有阶 \(6\) 元，且阶数最小。证毕。
\end{proof}

\noindent
由此，window-\(6\) / boundary-\(10\) 的两类离散分块不再只是并列证书：weight-layer 的 \(9+6+3+3\) 与 orbit--Levi 的 \(15+3+3\) 都在根--Cartan 字典前呈现唯一交会矩阵，而二层 boundary 种子则把 sheet parity 与三轴循环压缩为同一个 \(\ZZ_6\)-主齐对象。于是，定理 \ref{thm:xi-boundary-uplift-mixed-radix-register-lower-bound} 给出的 mixed-radix 状态数下界，在这一最小窗口上可进一步强化为“目标寄存器必须真实含有阶 \(6\) 元”的严格群论判据。
