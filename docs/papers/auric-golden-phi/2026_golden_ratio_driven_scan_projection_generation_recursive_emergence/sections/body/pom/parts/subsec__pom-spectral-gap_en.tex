\subsection{Spectral Gap-Driven Temporal Scales and Readout Stability}\label{subsec:pom-spectral-gap}
\begin{corollary}[Readout Time Scale Controlled by Spectral Gap]\label{cor:pom-spectral-gap-time}
Let $M_K(u)$ be primitive, with principal eigenvalue $\lambda_1(u)$ and maximum modulus of remaining eigenvalues $\lambda_2(u)$. By Theorem \ref{thm:kernel-weighted-prime-orbit},
\[
B_n(u)=\frac{\lambda_1(u)^n}{n}+O\!\left(\frac{|\lambda_2(u)|^n}{n}\right),
\]
thus the relative error satisfies
\[
\left|\frac{nB_n(u)}{\lambda_1(u)^n}-1\right|
\le C\left(\frac{|\lambda_2(u)|}{\lambda_1(u)}\right)^n.
\]
Given target precision $\varepsilon\in(0,1)$, it suffices to take
\[
n\ \ge\ \frac{\log(C/\varepsilon)}{\log\!\bigl(\lambda_1(u)/|\lambda_2(u)|\bigr)}
\]
to ensure the relative error does not exceed $\varepsilon$. If the kernel-version RH/GRH condition (Definition \ref{def:kernel-rh}) is further satisfied, one may replace $|\lambda_2(u)|$ by $\lambda_1(u)^{1/2}$, thereby obtaining a square-root decay scale (Corollary \ref{cor:kernel-rh-sqrt-error}).
\end{corollary}

\begin{remark}[Self-Duality and Completion Evenness]\label{rem:pom-duality-even}
If the synchronization kernel satisfies the self-duality relation of Appendix \ref{app:kernel-self-dual}, the completed $\zeta$-function exhibits strict evenness in the $\theta$ variable. Consequently, its odd-order derivatives with respect to $\theta$ vanish at $\theta=0$, imposing symmetry constraints on the fluctuation statistics of the cost potential; the corresponding tunability can only be realized by breaking such self-duality structure.
\end{remark}

\subsubsection{Intrinsic Time Correlation: Correlation Length Scale Given by Spectral Gap (No External Input)}

This subsection further interprets the "spectral gap" from prime-orbit error control as the \textbf{decay scale of temporal correlation}. We emphasize "no external input": the randomness considered arises from the kernel's own Parry/equilibrium measure (i.e., the stationary Markov structure induced by the finite-dimensional transfer operator), rather than from external noise or driving.

\begin{proposition}[Spectral Gap \texorpdfstring{$\Rightarrow$}{=>} Exponential Decay of Time Correlation (Auditable)]\label{prop:pom-time-corr-gap}
Let $B_\theta$ be the family of tilted transfer matrices for a finite-state synchronization kernel (e.g., a one-dimensional tilt with respect to output potential $e$), and assume that for given $\theta$ the matrix $B_\theta$ is primitive. Denote
\(
\lambda_1(\theta)=\rho(B_\theta),\quad
\Lambda(\theta)=\max\{ |\lambda|:\ \lambda\in\mathrm{spec}(B_\theta),\ |\lambda|<\lambda_1(\theta)\}
\), and let
\[
\rho(\theta):=\frac{\Lambda(\theta)}{\lambda_1(\theta)}\in[0,1),
\qquad
g(\theta):=-\log \rho(\theta)\in(0,+\infty).
\]
Then for any observables $F,G$ depending on finite windows (after block-lifting to single-time functions), under the Parry/equilibrium measure of $B_\theta$ there exists a constant $C_{F,G}(\theta)<\infty$ such that for all $n\ge 0$,
\[
\bigl|\mathrm{Cov}(F,\,G\circ\sigma^n)\bigr|\ \le\ C_{F,G}(\theta)\,\rho(\theta)^n.
\]
Therefore one may interpret
\(
\tau(\theta):=1/g(\theta)
\)
as the "correlation time," and $\tau_{1/2}(\theta):=\log 2\,/\,g(\theta)$ as the half-life (time for the exponential envelope to decay to $1/2$).
\end{proposition}

\begin{proof}
For a primitive nonnegative matrix $B_\theta$, Perron--Frobenius theory yields a unique principal eigenvalue $\lambda_1(\theta)>0$ and corresponding left and right positive eigenvectors. By the standard construction of Parry measure (Doob transform), the covariance can be written as
\(
\mathrm{Cov}(F,G\circ\sigma^n)=\langle f,\,\mathcal{T}_\theta^n g\rangle
\)
for some form, where $\mathcal{T}_\theta$ is the normalized transfer operator (spectral radius $1$), whose nontrivial spectral radius is $\rho(\theta)$. For centered functions, the principal eigenmode vanishes, leaving only residual spectral contribution, thereby yielding the exponential decay bound. All quantities are finite-dimensional linear algebraic objects and can be directly audited via matrix spectra and vector inner products.
\end{proof}

\subsection{Conclusion Cluster IV: Exponential Scale Saturation and Constant-Level Fingerprints}\label{subsec:pom-lambda-saturation}
\begin{proposition}[Input Saturation of Exponential Scale]\label{prop:pom-lambda-saturation}
For deterministic, complete-readout synchronization kernels, the Perron root $\lambda$ is primarily determined by the input skeleton, with the kernel direction contributing only smaller-modulus spectral corrections. Specifically, the 40-state real-input kernel satisfies $\rho(M)=\rho(A)=\varphi^2$ (Proposition \ref{prop:real-input-40-skeleton-factor}), and the 10-state synchronization kernel satisfies $\lambda=3$ (determinant factorization in Appendix \ref{app:sync-kernel-basic}). Thus $\log\lambda$ is closer to the saturation scale of "input branching entropy" than a fine-grained fingerprint of mechanism differences.
\end{proposition}

\begin{remark}[Three Classes of Fingerprints Distinguishing Mechanisms]\label{rem:pom-mechanism-fingerprint}
When the principal scale $\lambda$ saturates, the objects truly distinguishing operational mechanisms come from constant-level and twisted spectral layers:
\begin{itemize}
  \item \textbf{Finite-part curve $\log\mathfrak{M}(\theta)$:} Recorded alongside $P(\theta)=\log\rho(M_\theta)$ on the same slice to characterize constant-level drift (see comparison note in \ref{subsec:chi-ldp}).
  \item \textbf{Dirichlet--Mertens constant tensor:} Gives constant-level bias for primitive orbit counting stratified by congruence class (see \ref{subsec:arity-dirichlet-mertens-tensor} and related appendices).
  \item \textbf{Worst-case twisted ratio $\rho/\lambda$:} Controls the order-projection mixing time scale via Chebotarev-type error control (Theorem \ref{thm:kernel-chebotarev-exp} and Corollary \ref{cor:pom-order-mixing}).
\end{itemize}
These three classes are orthogonal and complementary to $\lambda$, constituting an auditable fingerprint of "operational mechanism"; thus when $\lambda$ is locked by the input branching, differences between kernels must reside in finite-part interfaces such as $\log\mathfrak{M}$ and the Dirichlet--Mertens tensor.
\end{remark}

\subsection{RH-Sharp Spectral Bound and Toy RH for the Real-Input 40-State Kernel}\label{subsec:pom-rh-sharp}
\begin{theorem}[RH-Sharp/Ramanujan Spectral Bound]\label{thm:pom-rh-sharp}
The determinant factorization of the real-input 40-state kernel
\[
\det(I-zM)=(1-z)^2(1+z)^3(1-3z+z^2)(1+z-z^2)
\]
(see Appendix \ref{app:real-input-40-finite-rh}) yields nonzero spectrum
\[
\mathrm{spec}(M)\setminus\{0\}
=\{\lambda,\ -\sqrt{\lambda},\ 1^{(2)},\ (-1)^{(3)},\ \lambda^{-1},\ \lambda^{-1/2}\},
\qquad \lambda=\varphi^2.
\]
Thus all nonzero eigenvalues $\nu$ other than $\lambda$ satisfy
\[
|\nu|\le \sqrt{\lambda}=\varphi,
\]
and exactly one nontrivial eigenvalue attains the bound: $\nu_2=-\sqrt{\lambda}$.
\end{theorem}

\begin{corollary}[Optimality of Square-Root Oscillation]\label{cor:pom-rh-sharp-an}
Let $a_n=\Tr(M^n)$. Then
\[
a_n=\lambda^n+(-\sqrt{\lambda})^n+O(1),
\]
and the error term cannot be improved to $o(\lambda^{n/2})$. Equivalently, the dynamical version of "Chebyshev square-root error" attains critical optimality for this kernel.
\end{corollary}

\begin{proposition}[\texorpdfstring{$\zeta$}{zeta} Structural Factorization: Input Skeleton \texorpdfstring{$\times$}{x} \((-z)\) Twisted Golden Factor \texorpdfstring{$\times$}{x} Trivial Period Factor]\label{prop:pom-zeta-factorization-40}
Let the golden-mean first-order Markov skeleton matrix be
$$
A:=
\begin{pmatrix}
1&1\\
1&0
\end{pmatrix},
$$
and let \(M\) be the adjacency matrix of the real-input 40-state kernel (Appendix \ref{app:real-input-40-abel}).
Then there is an exact algebraic factorization
$$
\det(I-zM)
=(1-z)^2(1+z)\cdot \det\!\bigl(I-z(A\otimes A)\bigr)\cdot \det(I+zA).
$$
Equivalently,
$$
\zeta_M(z)
=\zeta_{A\otimes A}(z)\cdot \zeta_A(-z)\cdot \frac{1}{(1-z)^2(1+z)},
$$
where \(\zeta_B(z):=1/\det(I-zB)\) is the \(\zeta\)-function of a finite-type subshift.
Moreover, the final "trivial period factor" \(\frac{1}{(1-z)^2(1+z)}\) only modifies short periods: at the primitive (Witt/Möbius) coefficient level it is nonzero only at \(n=1,2\) (term-by-term formula in Corollary \ref{cor:real-input-40-space-twist-boundary-split}).
\end{proposition}
\begin{proof}
The eigenvalues of matrix \(A\) are \(\varphi\) and \(-\varphi^{-1}\), so the eigenvalues of \(A\otimes A\) are
\(\{\varphi^2,-1,-1,\varphi^{-2}\}\), whence
$$
\det\!\bigl(I-z(A\otimes A)\bigr)=(1-\varphi^2 z)(1+z)^2(1-\varphi^{-2}z)=(1+z)^2(1-3z+z^2).
$$
Meanwhile
\(
\det(I+zA)=1+z-z^2
\).
Multiplying the above two by \((1-z)^2(1+z)\) yields
$$
(1-z)^2(1+z)^3(1-3z+z^2)(1+z-z^2),
$$
which agrees with the factorization of \(\det(I-zM)\) in Theorem \ref{thm:pom-rh-sharp}. This completes the proof.
\end{proof}

\begin{remark}[Hidden \texorpdfstring{$(-A)$}{(-A)} Factor: Structural Origin of Toy RH Critical Line Lattice]\label{rem:pom-minusA-factor-critical-line}
The third factor in Proposition \ref{prop:pom-zeta-factorization-40} can be further rewritten as
$$
\det(I+zA)=\det\!\bigl(I-z(-A)\bigr),
$$
whence
$$
\boxed{\;
\det(I-zM)
=(1-z)^2(1+z)\cdot \det\!\bigl(I-z(A\otimes A)\bigr)\cdot \det\!\bigl(I-z(-A)\bigr).
\;}
$$
Equivalently, at the \(\zeta\)-function level,
$$
\boxed{\;
\zeta_M(z)
=\frac{1}{(1-z)^2(1+z)}\cdot \zeta_{A\otimes A}(z)\cdot \zeta_{-A}(z),
\qquad
\zeta_{-A}(z):=\frac{1}{\det(I-z(-A))}=\frac{1}{\det(I+zA)}.
\;}
$$
This rewriting completely isolates the origin of the toy RH "critical line lattice": since the Perron root of \(-A\) is \(-\varphi\), and the principal scale is \(\lambda=\varphi^2\), the unique boundary-attaining eigenvalue
\(
-\sqrt{\lambda}=-\varphi
\)
does not arise from accidental resonance in the 40-state complex structure, but is necessarily embedded by the \textbf{sign-flipped factor \((-A)\) of the golden-mean adjacency operator \(A\)}.
Hence the nontrivial pole sequence in the strip \(0<\Re(s)<1\) (Theorem \ref{thm:pom-toy-rh}) can be viewed as the direct image of \(\det(I-z(-A))=0\) under the Dirichlet coordinate \(z=\lambda^{-s}\).
\end{remark}

\begin{remark}[Permutation Lift Viewpoint: Ramanujan/RH-Sharp as "New Eigenvalue Bound"]\label{rem:pom-permutation-lift-ramanujan}
Appendix \ref{prop:real-input-40-skeleton-factor} shows that the real-input \(40\)-state kernel can be written as a \(4\times 4\) block matrix, with nonzero blocks being \(10\times 10\) permutation matrices \(B^{(d)}\) induced by deterministic transitions, and that the constant-fiber subspace

  \item \textbf{New spectrum (vertical/fiber-orthogonal spectrum):} From the action on \(U^\perp=\RR^{4}\otimes \mathbf{1}^\perp\), whose entire content is carried by the "vertical residual factor" (the \(\det_{\mathrm{vert}}\) in Appendix \ref{prop:real-input-40-skeleton-factor}).

\begin{proposition}[Lucas Closed Form for Odd-Prime-Length Primitives]\label{prop:pom-primitive-prime-lucas}
Let \(p\) be an odd prime and \(L_n\) the Lucas numbers. Then the primitive orbit count of the real-input 40-state kernel satisfies
$$
p_p=\frac{L_p(L_p-1)}{p}.
$$
\end{proposition}
\begin{proof}
By Möbius inversion (Definition \ref{def:pom-proj-prim}) for prime \(p\) one has
\(
p_p=\frac{a_p-a_1}{p}
\).
By Proposition \ref{prop:pom-lucas-square} and since \(p\) is odd (thus \((-1)^p=-1\)), we have
\(
a_p=L_p^2-L_p+1
\), and \(a_1=1\).
Substituting yields \(p_p=\frac{L_p^2-L_p}{p}\). This completes the proof.
\end{proof}

\begin{lemma}[Lucas Prime Congruence: \texorpdfstring{$L_p\equiv 1\pmod p$}{Lp=1 mod p}]\label{lem:pom-lucas-modp}
For any prime \(p\), the Lucas number satisfies \(L_p\equiv 1\ (\mathrm{mod}\ p)\).
\end{lemma}
\begin{proof}
Assume \(p\neq 5\) (when \(p=5\) direct calculation gives \(L_5=11\equiv 1\pmod 5\)). In the ring \(\mathbb{Z}[\sqrt5]\),
by binomial expansion and \(\binom{p}{k}\equiv 0\pmod p\) (\(1\le k\le p-1\)), we have
$$
(1+\sqrt5)^p+(1-\sqrt5)^p\equiv 2+(\sqrt5)^p+(-\sqrt5)^p\pmod p.
$$
When \(p\) is an odd prime the last two terms cancel, so
\(
(1+\sqrt5)^p+(1-\sqrt5)^p\equiv 2\pmod p
\);
when \(p=2\) direct verification also holds. From the definition \(2^p L_p=(1+\sqrt5)^p+(1-\sqrt5)^p\) and Fermat's little theorem \(2^{p-1}\equiv 1\pmod p\) we have
\(
2^p\equiv 2\pmod p
\), thus \(L_p\equiv 1\pmod p\). This completes the proof.
\end{proof}

\begin{definition}[Prime-Length Wieferich Fingerprint (\texorpdfstring{$\nu_p$}{nu_p})]\label{def:pom-lucas-nu-p}
For an odd prime \(p\), define
\[
\nu_p:=v_p(L_p-1)\in\ZZ_{\ge 1},
\]
where \(v_p(\cdot)\) is the \(p\)-adic valuation.
\end{definition}

\begin{corollary}[Complete Integer Criterion for Prime-Length Primitive Degeneracy]\label{cor:pom-lucas-nu-p-criterion}
Let \(p\) be an odd prime and use the notation of Proposition \ref{prop:pom-primitive-prime-lucas}.
Then
\[
v_p(p_p)=\nu_p-1.
\]
Thus the "degeneracy strength" at the prime-length layer is completely controlled by the single integer \(\nu_p\): for any \(k\ge 1\),
\[
p^{k}\mid p_p
\qquad\Longleftrightarrow\qquad
\nu_p\ge k+1.
\]
In particular, \(p\mid p_p\) if and only if \(\nu_p\ge 2\), equivalent to \(p^2\mid(L_p-1)\).
\end{corollary}
\begin{proof}
By Lemma \ref{lem:pom-lucas-modp} we have \(L_p\equiv 1\pmod p\), whence \(v_p(L_p)=0\) and \(\nu_p\ge 1\).
On the other hand, Proposition \ref{prop:pom-primitive-prime-lucas} gives
\(
p_p=L_p(L_p-1)/p
\), thus
\[
v_p(p_p)=v_p(L_p)+v_p(L_p-1)-1=\nu_p-1.
\]
Hence \(p^k\mid p_p\) is equivalent to \(v_p(p_p)\ge k\), which by the above formula yields \(p^k\mid p_p\Longleftrightarrow \nu_p\ge k+1\).
\end{proof}

\begin{remark}[Integrality and Wieferich-Type Rare Events]\label{rem:pom-lucas-wieferich}
By Corollary \ref{cor:pom-lucas-nu-p-criterion}, the integrality and rare degeneracy at the prime-length layer can be uniformly graded by
\(
\nu_p=v_p(L_p-1)
\):
\(\nu_p\ge 1\) always holds, whereas \(\nu_p\ge 2\) (equivalent to \(p\mid p_p\)) corresponds to a Wieferich-type rare event interface for Lucas sequences.
Thus \(\nu_p\) may be viewed as an auditable fingerprint of "prime-length primitive arithmetic degeneracy," with \(\nu_p=2,3,\dots\) serving as a natural grading of anomaly strength.
\end{remark}

\begin{lemma}[Vertical Lattice in Dirichlet Logarithmic Coordinate]\label{lem:pom-dirichlet-vertical-lattice}
Take \(\lambda>1\), and let \(z=\lambda^{-s}\), \(r=\lambda z=\lambda^{1-s}\).
Given \(r_0\in\CC^\times\), the solutions to \(r=r_0\) form a vertical lattice in the \(s\)-plane:
\[
s=1-\frac{\log|r_0|}{\log\lambda}-\,\mathrm{i}\,\frac{\arg(r_0)+2\pi k}{\log\lambda},
\qquad k\in\ZZ,
\]
where \(\arg(r_0)\in(-\pi,\pi]\).
Thus the imaginary spacing is constant
\[
\boxed{\ \Delta\Im(s)=\frac{2\pi}{\log\lambda}\ }.
\]
In the special case \(r_0<0\), where \(\arg(r_0)=\pi\), the above reduces to
\[
s=1-\frac{\log|r_0|}{\log\lambda}-\,\mathrm{i}\,\frac{(2k+1)\pi}{\log\lambda},
\qquad k\in\ZZ.
\]
\end{lemma}
\begin{proof}
From \(r=\lambda^{1-s}\) we have
\(
(1-s)\log\lambda=\log r_0
\), and writing the multivalued logarithm as
\(
\log r_0=\log|r_0|+\mathrm{i}\,(\arg(r_0)+2\pi k)
\) (\(k\in\ZZ\)), solving for \(s\) yields the result.
\end{proof}

and satisfy

Among these, $r=-\varphi$ generates a critical-line lattice

\begin{proposition}[Profinite Character Diagonalization: Artin Decomposition on Hilbert Space and Operatorization of Kernel-Version GRH]\label{prop:pom-profinite-artin-hilbert}
Let \(K\) be a finite kernel (directed graph/Markov kernel) with vertex set \(V\), and let \(\kappa:E(K)\to\ZZ\) be a one-step additive potential (e.g., carry/collision/output potential).
Take the profinite congruence axis \(G_\infty:=\widehat{\ZZ}\) (or its unit part \(\widehat{\ZZ}^{\,\times}\), see Remark \ref{rem:pom-profinite-units}), and let
\(
g:E(K)\to G_\infty
\)
be a one-step congruence cocycle (i.e., labeling each edge with a residue increment; readout to finite modulus \(m\) is the composition \(G_\infty\to\ZZ/m\ZZ\)).
On the Hilbert space
\[
\mathcal{H}:=\CC^{V}\otimes L^2(G_\infty)
\]
(\(G_\infty\) with Haar measure), define the "group extension adjacency operator"
\[
\bigl(\mathcal{M}_u f\bigr)(j,h)
:=\sum_{e:i\to j} u^{\kappa(e)}\, f\!\bigl(i,\ h\cdot g(e)\bigr),
\qquad u>0,
\]
where \(f\in\mathcal{H}\) is viewed as a square-integrable function on \(V\times G_\infty\), and \(h\mapsto h\cdot g(e)\) is right multiplication.
Then:
\begin{enumerate}
  \item \textbf{(Fourier decomposition and block diagonalization)} Since \(G_\infty\) is a compact abelian group, there is a character orthogonal basis decomposition
  \[
  L^2(G_\infty)\cong \bigoplus_{\chi\in\widehat{G_\infty}}\CC\,\chi,
  \]
  and under this decomposition \(\mathcal{M}_u\) unitarily reduces to a direct sum
  \[
  \mathcal{M}_u\ \cong\ \bigoplus_{\chi\in\widehat{G_\infty}} M_\chi(u),
  \]
  where each block \(M_\chi(u)\) is a finite-dimensional twisted matrix on \(\CC^{V}\):
  \[
  (M_\chi(u))_{i\to j}:=\sum_{e:i\to j} u^{\kappa(e)}\,\chi\!\bigl(g(e)\bigr).
  \]
  \item \textbf{(Operator readout of worst-case twisted spectral radius)} Denote the trivial character by \(1\), and let
  \(
  \mathcal{H}_0:=\CC^V\otimes\mathrm{span}\{1\}
  \)
  and \(\mathcal{H}_\perp:=\mathcal{H}_0^\perp\).
  Then
  \[
  \rho\!\left(\mathcal{M}_u|_{\mathcal{H}_\perp}\right)
  =\sup_{\chi\neq 1}\rho\!\left(M_\chi(u)\right),
  \qquad
  \rho\!\left(\mathcal{M}_u|_{\mathcal{H}_0}\right)=\rho\!\left(M_1(u)\right)=:\lambda_1(u).
  \]
  \item \textbf{(Operatorization of kernel-version GRH)} If the kernel-version GRH (isomorphic condition in Definition \ref{def:kernel-rh}) holds for this profinite character family, i.e.,
  \(
  \sup_{\chi\neq 1}\rho(M_\chi(u))\le \lambda_1(u)^{1/2}
  \),
  then equivalently there is an operator spectral radius inequality
  \[
  \rho\!\left(\mathcal{M}_u|_{\mathcal{H}_\perp}\right)\ \le\ \lambda_1(u)^{1/2}.
  \]
\end{enumerate}
\end{proposition}
\begin{proof}
For finite-modulus \(m\) axis bundles, Proposition \ref{prop:kernel-artin-factorization} gives the finite group version of determinant factorization, whose essence is Fourier/character decomposition of the regular representation.
For compact abelian group \(G_\infty\), the Pontryagin dual \(\widehat{G_\infty}\) is a discrete abelian group, and characters form an orthogonal basis in \(L^2(G_\infty)\); right multiplication \(T_{g(e)}:f(h)\mapsto f(h\cdot g(e))\) diagonalizes on this basis with eigenvalue \(\chi(g(e))\).
Thus \(\mathcal{M}_u\) reduces character-by-character to finite-dimensional blocks \(M_\chi(u)\) in the Fourier basis of \(L^2(G_\infty)\), yielding the direct sum decomposition. The spectral radius identity follows from the spectrum of a direct sum operator being the union of block spectra. This completes the proof.
\end{proof}

\begin{definition}[Finite-Layer Lifted System and Fiber Measure]\label{def:pom-lifted-system-m}
For modulus $m$ finite congruence axis $G_m$, define its \textbf{finite-layer lifted system} as the product dynamical system $(\Sigma \times G_m, \sigma_m)$, where $\Sigma$ is the base state space and $\sigma_m$ is the lifted transition induced by kernel $K$ and congruence cocycle $g$.

\widetilde\mu_m := \mu_{\mathrm{Parry}} \otimes \mathrm{Haar}_{G_m},

where $\mu_{\mathrm{Parry}}$ is the Parry measure on the base system and Haar measure on $G_m$ is normalized.
Define the \textbf{zero-mean fiber observable space} as
\[
L^2_0(G_m) := \left\{ f \in L^2(G_m) : \int_{G_m} f(g) \, d\mathrm{Haar}(g) = 0 \right\},

and naturally embed it as functions on the lifted system $F(x, g) = f(g)$.
\end{definition}

\begin{theorem}[Congruence Fiber Correlation Decay Rate = Worst Twisted Ratio]\label{thm:pom-fiber-corr-decay}
Fix $u>0$. Let $f \in L^2_0(G_m)$ be a zero-mean fiber observable. Define its time correlation function as
\[
C_f(n) := \mathbb{E}_{\widetilde\mu_m} [f(g_0) \overline{f(g_n)}],
\]
where $g_n$ is the fiber coordinate after $n$-step transition. Denote $\lambda_1(u) := \rho(M_{m,1}(u))$ as the base system principal eigenvalue. Then:
\begin{enumerate}
    \item \textbf{(Exponential decay bound)} The correlation function satisfies
    \[
    |C_f(n)| \le \|f\|_2^2 \cdot \eta_m(u)^n,
    \]
    where $\eta_m(u)$ is the \textbf{worst twisted ratio}:
    \[
    \eta_m(u) := \max_{\chi \in \widehat{G_m},\, \chi \neq 1} \frac{\rho(M_{m,\chi}(u))}{\rho(M_{m,1}(u))}.
    \]
    \item \textbf{(Character decomposition formula)} Let $f = \sum_{\chi \neq 1} \hat{f}(\chi) \chi$ be the Fourier expansion of $f$. Let $\mathcal{L}_\chi$ be the normalized twisted transfer operator with matrix elements $(\mathcal{L}_\chi)_{ij} = \sum_{e:i\to j} \frac{h_j}{\lambda_1(u) h_i} \chi(g(e))$, where $h$ is the right positive eigenvector of $M_{m,1}(u)$. Then the correlation function has exact character superposition representation:
    \[
    C_f(n) = \sum_{\chi \neq 1} |\hat{f}(\chi)|^2 \langle \pi, \mathcal{L}_\chi^n \mathbf{1} \rangle,
    \]
    where $\pi$ is the stationary distribution of the base system (satisfying $\pi \mathcal{L}_1 = \pi$) and $\mathbf{1}$ is the all-ones vector.
    \item \textbf{(Scale distinction: horizontal vs vertical)}
    State-layer correlation decay (Proposition \ref{prop:pom-time-corr-gap}) is determined by the base matrix spectral gap $\rho = \Lambda/\lambda_1$, characterizing base space mixing speed;
    whereas fiber-layer correlation decay is determined by the twisted ratio $\eta_m(u)$, characterizing the "decorrelation" speed of congruence labels in path evolution.
    These two are independent structural objects: the former from internal spectral structure of $M_1(u)$, the latter from the relative modulus ratio of $M_\chi(u)$ to $M_1(u)$.
\end{enumerate}
\end{theorem}

\begin{proof}
Denote the path cocycle by $g(e_1 \dots e_n) := g(e_1) \cdots g(e_n)$, with multiplication order consistent with right shift $h \mapsto h \cdot g(e)$.
Since the lifted measure $\widetilde\mu_m = \mu_{\mathrm{Parry}} \otimes \mathrm{Haar}_{G_m}$ is a product measure, and fiber transition $g_n = g_0 \cdot g(e_1 \dots e_n)$ depends only on base path, the correlation function can be written as:
\[
C_f(n) = \int_{G_m} \int_{\Sigma} f(g_0) \overline{f(g_0 \cdot g(e_1 \dots e_n))} \, d\mu_{\mathrm{Parry}} \, d\mathrm{Haar}(g_0).
\]
Using translation invariance of Haar measure and character orthogonality $\int \chi(g) \overline{\chi'(g)} dg = \delta_{\chi,\chi'}$, substituting $f = \sum \hat{f}(\chi) \chi$ yields:
\[
C_f(n) = \sum_{\chi \neq 1} |\hat{f}(\chi)|^2 \int_{\Sigma} \overline{\chi(g(e_1 \dots e_n))} \, d\mu_{\mathrm{Parry}}.
\]
The integral term is precisely the $n$-step transition expectation $\langle \pi, \mathcal{L}_\chi^n \mathbf{1} \rangle$ of the normalized twisted operator $\mathcal{L}_\chi$.
Since $\rho(\mathcal{L}_\chi) = \rho(M_{m,\chi}(u))/\lambda_1(u)$, by the spectral radius formula we have $|\langle \pi, \mathcal{L}_\chi^n \mathbf{1} \rangle| \le \rho(\mathcal{L}_\chi)^n$.
Taking $\eta_m(u) = \max_{\chi \neq 1} \rho(\mathcal{L}_\chi)$ and using Parseval's identity $\sum |\hat{f}(\chi)|^2 = \|f\|_2^2$ yields the exponential decay bound.
Since $f \in L^2_0(G_m)$, the trivial character coefficient $\hat{f}(1) = 0$, so correlation is entirely determined by nontrivial twisted channels.
\end{proof}
