\[
N_{\mathrm{Toep}}\ \gtrsim\ Q_{\max}(T,\delta_0)\,\log\!\bigl(1+\delta_0 Q_{\max}(T,\delta_0)\bigr)
\asymp \frac{T^2}{\delta_0}\,\log(1+T^2),
\qquad
N_{\mathrm{scan}}\ \gtrsim\ \frac{T}{t+\delta_0}\quad(t>0\ \text{fixed}),
\]
where
\[
Q_{\max}(T,\delta_0):=\frac{T^2+(1+\delta_0)^2}{4\delta_0}.
\]
\end{conclusion}

See Proposition \ref{con:xi-toeplitz-vs-scan-complexity-lower-bounds} (combined with its referenced corollaries/theorems).
\end{proof}

\begin{conclusion}[Full-recovery strong converse under joint query--side-information regime: optimal success exponent for query fraction \texorpdfstring{$\beta$}{beta} and side-information density \texorpdfstring{$b$}{b} is a truncated linear law]\label{con:xi-terminal-linear-strong-converse-query-sideinfo}
Let \(X\) be a finite set (\(|X|=n\ge 2\)), \(|\Omega|=N\), given fiber spectrum \(d:X\to\NN\) satisfying \(\sum_{x\in X}d(x)=N\).
Let
\[
\mathfrak{F}(d):=\left\{F:\Omega\to X:\ |F^{-1}(x)|=d(x)\ \ \forall x\in X\right\},
\qquad
F\sim\mathrm{Unif}(\mathfrak{F}(d)),
\]
and write \(w(x):=d(x)/N\), \(H(w):=-\sum_{x\in X}w(x)\log w(x)\).
Perform \(k=\lfloor \beta N\rfloor\) distinct queries (adaptive allowed), obtaining trajectory \(T_{1:k}\).
Subsequently allow obtaining \(B=\lfloor bN\rfloor\) bits of side information \(\mathrm{enc}(F)\in\{0,1\}^B\), and attempt to exactly recover the entire map \(F\) using \((T_{1:k},\mathrm{enc}(F))\) as input.
Let the optimal success rate be \(\mathrm{Succ}^{\mathrm{opt}}_{\beta,b}(N)\).
If \(n\log N=o(N)\), then the limit exists and satisfies
\[
\boxed{\;
\lim_{N\to\infty}\frac{1}{N}\log \mathrm{Succ}^{\mathrm{opt}}_{\beta,b}(N)
=
\min\bigl\{0,\ b\log 2-(1-\beta)H(w)\bigr\}.\;}
\]
Thus the critical density is
\[
b_{\mathrm{crit}}(\beta):=\frac{(1-\beta)H(w)}{\log 2}.
\]
When \(b<b_{\mathrm{crit}}(\beta)\) the success rate decays with exact exponential rate \((1-\beta)H(w)-b\log 2\); when \(b\ge b_{\mathrm{crit}}(\beta)\) the success rate does not decay exponentially (exponential rate is \(0\)).
\end{conclusion}
\begin{proof}[Proof]
For given trajectory \(T_{1:k}\), write \(C_k(x)\) as the occurrence count of value \(x\) queried, and let
\[
\mathcal{M}_k
:=
\frac{(N-k)!}{\prod_{x\in X}(d(x)-C_k(x))!}
\]
be the number of posterior feasible maps compatible with \(d\) given this trajectory (multinomial count of the remaining \(N-k\) points).
Since \(F\) is uniform on \(\mathfrak{F}(d)\), the posterior conditioned on \(T_{1:k}\) remains uniform on this set.
For any deterministic encoding of length \(B\), \(\mathrm{enc}(F)\) induces a partition of at most \(2^B\) codewords over the posterior feasible set; thus the optimal conditional success rate is
\[
\mathrm{Succ}^{\mathrm{opt}}_{\beta,b}(N\mid T_{1:k})
=\min\Bigl(1,\frac{2^B}{\mathcal{M}_k}\Bigr),
\]
thus
\(
\mathrm{Succ}^{\mathrm{opt}}_{\beta,b}(N)=\mathbb{E}\bigl[\min(1,2^B/\mathcal{M}_k)\bigr]
\).
\par
By Stirling expansion and \(n\log N=o(N)\), writing the residual empirical distribution
\[
R_N(x):=\frac{d(x)-C_k(x)}{N-k}\in\Delta(X),
\]
we have
\(
\frac{1}{N}\log\mathcal{M}_k=(1-\beta)H(R_N)+o_{\mathbb{P}}(1)
\).
Also by hypergeometric law of large numbers \(R_N\to w\) (convergence in probability), we get
\(
\frac{1}{N}\log\mathcal{M}_k=(1-\beta)H(w)+o_{\mathbb{P}}(1)
\).
\par
When \(b<b_{\mathrm{crit}}(\beta)\), take any \(\varepsilon\in\bigl(0,(1-\beta)H(w)-b\log 2\bigr)\), and let the event
\[
\mathcal{E}_N:=\left\{\left|\frac{1}{N}\log\mathcal{M}_k-(1-\beta)H(w)\right|\le \varepsilon\right\}.
\]
By \(\frac{1}{N}\log\mathcal{M}_k=(1-\beta)H(w)+o_{\mathbb{P}}(1)\) we get \(\mathbb{P}(\mathcal{E}_N)\to 1\).
On \(\mathcal{E}_N\) for sufficiently large \(N\) we have \(2^B\le \mathcal{M}_k\), thus \(\min(1,2^B/\mathcal{M}_k)=2^B/\mathcal{M}_k\).
Again using \(B=\lfloor bN\rfloor\) we get \(2^B=\exp(bN\log 2+O(1))\), and on \(\mathcal{E}_N\) we have
\[
\exp\!\bigl(-N((1-\beta)H(w)+\varepsilon)\bigr)
\le
\mathcal{M}_k^{-1}
\le
\exp\!\bigl(-N((1-\beta)H(w)-\varepsilon)\bigr).
\]
Taking expectation in \(\mathrm{Succ}^{\mathrm{opt}}_{\beta,b}(N)=\EE[\min(1,2^B/\mathcal{M}_k)]\) and using \(\mathbb{P}(\mathcal{E}_N^c)=o(1)\), we obtain
\[
\exp\!\bigl(N(b\log 2-(1-\beta)H(w)-\varepsilon)+o(N)\bigr)
\le
\mathrm{Succ}^{\mathrm{opt}}_{\beta,b}(N)
\le
\exp\!\bigl(N(b\log 2-(1-\beta)H(w)+\varepsilon)+o(N)\bigr).
\]
Letting \(\varepsilon\downarrow 0\) yields the stated negative exponential rate.
\par
When \(b\ge b_{\mathrm{crit}}(\beta)\), take any \(\varepsilon>0\) and let
\[
\mathcal{E}_N(\varepsilon):=\left\{\left|\frac{1}{N}\log\mathcal{M}_k-(1-\beta)H(w)\right|\le \varepsilon\right\},
\]
then \(\mathbb{P}(\mathcal{E}_N(\varepsilon))\to 1\).
By \(B=\lfloor bN\rfloor\) and \(b\log 2\ge (1-\beta)H(w)\) we know on \(\mathcal{E}_N(\varepsilon)\) we have
\[
\min\!\left(1,\frac{2^B}{\mathcal{M}_k}\right)
\ge
\exp(-\varepsilon N+O(1)).
\]
Taking expectation and using \(\mathbb{P}(\mathcal{E}_N(\varepsilon))\to 1\) we get
\[
\mathrm{Succ}^{\mathrm{opt}}_{\beta,b}(N)\ge \exp(-\varepsilon N+o(N)),
\]
thus \(\liminf_{N\to\infty}\frac{1}{N}\log \mathrm{Succ}^{\mathrm{opt}}_{\beta,b}(N)\ge -\varepsilon\).
Letting \(\varepsilon\downarrow 0\) gives \(\liminf\ge 0\); also by \(\mathrm{Succ}^{\mathrm{opt}}_{\beta,b}(N)\le 1\) we get \(\limsup\le 0\), hence the exponential rate is \(0\).
Proof complete.
\end{proof}

\begin{conclusion}[Failure exponent on the supercritical side: exact variational formula for failure probability when \texorpdfstring{$b>b_{\mathrm{crit}}(\beta)$}{b>bcrit(beta)}]\label{con:xi-terminal-supercritical-failure-exponent-variational}
Continue with the setting of Conclusion \ref{con:xi-terminal-linear-strong-converse-query-sideinfo}.
Define failure probability
\[
\mathrm{Fail}^{\mathrm{opt}}_{\beta,b}(N):=1-\mathrm{Succ}^{\mathrm{opt}}_{\beta,b}(N).
\]
Assume \(n\) and \(w\) are fixed, and take \(b>b_{\mathrm{crit}}(\beta)\).
Let
\[
I_\beta(r):=
\begin{cases}
(1-\beta)D_{\mathrm{KL}}(r\|w)+\beta D_{\mathrm{KL}}\!\left(\dfrac{w-(1-\beta)r}{\beta}\,\middle\|\,w\right),
& \text{if }\ \dfrac{w-(1-\beta)r}{\beta}\in\Delta(X),\\[10pt]
+\infty, & \text{otherwise},
\end{cases}
\]
then the limit
\[
E_{\mathrm{fail}}(\beta,b)
:=
\lim_{N\to\infty}-\frac{1}{N}\log \mathrm{Fail}^{\mathrm{opt}}_{\beta,b}(N)
\]
exists, and satisfies the exact variational formula
\[
\boxed{\;
E_{\mathrm{fail}}(\beta,b)
=
\inf\left\{I_\beta(r):\ (1-\beta)H(r)\ge b\log 2\right\}.\;}
\]
In particular, \(E_{\mathrm{fail}}(\beta,b)>0\) and strictly increases with \(b\).
\end{conclusion}
\begin{proof}[Proof]
From the representation in Conclusion \ref{con:xi-terminal-linear-strong-converse-query-sideinfo},
\[
\mathrm{Fail}^{\mathrm{opt}}_{\beta,b}(N)
=
\mathbb{E}\!\left[\Bigl(1-\frac{2^B}{\mathcal{M}_k}\Bigr)_+\right].
\]
Also by Stirling expansion,
\(
\frac{1}{N}\log\mathcal{M}_k=(1-\beta)H(R_N)+o_{\mathbb{P}}(1)
\).
When the event \((1-\beta)H(R_N)\ge b\log 2+\varepsilon\) occurs, we have
\(
2^B/\mathcal{M}_k\le \exp(-\varepsilon N+o(N))
\), thus conditional failure tends to \(1\);
when \((1-\beta)H(R_N)\le b\log 2-\varepsilon\), conditional failure is \(0\) (for sufficiently large \(N\)).
Therefore
\[
\mathrm{Fail}^{\mathrm{opt}}_{\beta,b}(N)
=
\mathbb{P}\bigl((1-\beta)H(R_N)\ge b\log 2\bigr)\cdot(1+o(1)).
\]
The sequence \((R_N)_{N}\) is the hypergeometric residual empirical distribution, satisfying a large deviation principle; its rate function is precisely \(I_\beta\) defined above.
Applying the contraction principle yields
\[
\lim_{N\to\infty}-\frac{1}{N}\log \mathrm{Fail}^{\mathrm{opt}}_{\beta,b}(N)
=\inf\{I_\beta(r):(1-\beta)H(r)\ge b\log 2\}.
\]
Strict monotonicity is given by the strict shrinkage of the constraint set with \(b\) and the strict convexity of \(I_\beta\). Proof complete.
\end{proof}

\begin{conclusion}[Escort--KKT structure of the minimizer: one-parameter power-tilt characterization and sensitivity formula for failure exponent]\label{con:xi-terminal-failure-exponent-escort-kkt}
Under the conditions of Conclusion \ref{con:xi-terminal-supercritical-failure-exponent-variational}, further assume \(w\) is full-support and non-uniform, and take \(b>b_{\mathrm{crit}}(\beta)\).
Then:
\begin{enumerate}
  \item The variational problem
  \(
  \inf\{I_\beta(r):(1-\beta)H(r)\ge b\log 2\}
  \)
  admits a unique minimizer \(r^\star\in\Delta(X)\), and it must lie on the boundary \((1-\beta)H(r^\star)=b\log 2\).
  \item Write
  \[
  q^\star:=\frac{w-(1-\beta)r^\star}{\beta}\in\Delta(X),
  \]
  then there exists unique Lagrange multiplier \(\theta^\star>0\) such that \(q^\star\) and \(r^\star\) satisfy the power-tilt (escort) relation
  \[
  \boxed{\;
  q^\star(x)\ \propto\ r^\star(x)^{\,1+\theta^\star},\qquad x\in X.\;}
  \]
  \item The failure exponent satisfies
  \[
  E_{\mathrm{fail}}(\beta,b)=I_\beta(r^\star)
  =(1-\beta)D_{\mathrm{KL}}(r^\star\|w)+\beta D_{\mathrm{KL}}(q^\star\|w),
  \]
  and \(b\mapsto E_{\mathrm{fail}}(\beta,b)\) is differentiable on the supercritical side, with derivative given by the multiplier:
  \[
  \boxed{\;\frac{\partial}{\partial b}E_{\mathrm{fail}}(\beta,b)=\theta^\star\log 2.\;}
  \]
\end{enumerate}
\end{conclusion}
\begin{proof}[Proof]
The feasible domain \(\{r:(1-\beta)H(r)\ge b\log 2\}\) is convex (since \(H\) is concave, superlevel set is convex), and \(I_\beta\) is strictly convex on full support, so the minimizer is unique and must satisfy the constraint with equality (otherwise one can lower \(I_\beta\) along a line toward \(w\) while remaining feasible).
\par
Applying KKT conditions for convex optimization with inequality constraints: there exists unique \(\theta^\star\ge 0\) such that
\[
\nabla I_\beta(r^\star)+\theta^\star\,\nabla\bigl(b\log 2-(1-\beta)H(r^\star)\bigr)=\lambda\,\mathbf 1
\]
(\(\lambda\) is the multiplier for the normalization constraint).
At interior points \(\nabla H(r)=-\log r-\mathbf 1\), and from \(q^\star=(w-(1-\beta)r^\star)/\beta\) we get
\(
\nabla I_\beta(r^\star)=(1-\beta)(\log r^\star-\log q^\star)
\)
(coordinate-wise calculation).
Substituting and absorbing constants, we obtain
\[
(\,1+\theta^\star)\log r^\star-\log q^\star=\mathrm{const},
\]
i.e., \(q^\star(x)\propto r^\star(x)^{1+\theta^\star}\).
\par
The sensitivity formula is a direct application of Lagrange multiplier sensitivity conclusion: writing the constraint as \(g_b(r):=b\log 2-(1-\beta)H(r)\le 0\), its derivative with respect to parameter \(b\) is \(\partial_b g_b=\log 2\), so \(\partial_b E_{\mathrm{fail}}(\beta,b)=\theta^\star\log 2\).
Proof complete.
\end{proof}

\end{conclusion}
