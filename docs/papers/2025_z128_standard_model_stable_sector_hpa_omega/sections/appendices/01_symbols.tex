\section{Symbols and objects (summary)}
\label{app:symbols}

This appendix records a compact list of the primary objects used in the paper.

\begin{itemize}
\item \textbf{Weyl pair.}
A pair of unitaries $(U,V)$ satisfying $UV=\e^{2\pi\iu\alpha}VU$ (Definition~\ref{def:weyl_pair}), used as an algebraic encoding of scan shift and phase.
\item \textbf{Rotation algebra and Morita equivalence (optional notes).}
The rotation algebra $A_\alpha$ is the $C^\ast$-algebra generated by a Weyl pair at irrational slope $\alpha\in\RR\setminus\QQ$.
Morita equivalence classes are acted on by $\mathrm{SL}_2(\ZZ)$ via $\alpha\mapsto (a\alpha+b)/(c\alpha+d)$; a Fourier transform exchanges scan shift and phase multiplication (Appendix~\ref{app:morita_fourier_exchange}).
\item \textbf{Scan orbit.}
Given an irrational slope $\alpha$ and seed $x_0\in\RR/\ZZ$,
\[
x_n=x_0+n\alpha\pmod{1},\qquad z_n=\e^{2\pi\iu x_n}\in\TT.
\]
\item \textbf{Window readout.}
For a window $W\subset\TT$,
\[
w_n=\ind\{z_n\in W\}\in\{0,1\}.
\]
\item \textbf{Finite windows.}
At window length $m$, $\Omega_m=\{0,1\}^m$ is the microstate alphabet, and $\mathcal{H}_m=\ell^2(\Omega_m)$ its linearization.
\item \textbf{Wish and Motive (reader-facing; not used in proofs).}
Appendix~\ref{app:wish_motive_definitions} records a template interface object
$W=(m,\mathcal{I}_m,\mathcal{C}_m,\varepsilon)$ (Wish), a finite candidate family $\mathcal{F}$, an auditable objective functional $J:\mathcal{F}\to\RR$ (Motive), and the CAP-selected minimizer $f_\star=\arg\min_{\mathcal{F}}J$ with deterministic tie-break rules.
\item \textbf{Zeckendorf admissible set.}
$X_m\subset\Omega_m$ is the forbidden-word admissible set (no adjacent ones), equivalently the $\varphi$-stable sector.
\item \textbf{Golden base.}
$\varphi=(1+\sqrt{5})/2$ and $\log_\varphi x := \log x/\log\varphi$.
\item \textbf{Resolution coordinate and exponential map.}
For a mass scale $\mu$, $r(\mu)=\log_\varphi(\mu/m_e)$ and $\mu(r)=m_e\varphi^{\,r}$ (Section~\ref{sec:mass_spectrum_closure}).
\item \textbf{Stable-type invariants at $(m,n)=(6,3)$.}
For $w\in X_6$, $V(w)$ is the Zeckendorf value, $|w|_1$ is Hamming weight, $g(w)=|\mathrm{Fold}_6^{-1}(w)|$ is the folding degeneracy, and $D_\pi(w)$ is the cyclic/boundary tag (Definition~\ref{def:x6_invariants}).
\item \textbf{Depth and matching shifts.}
$r_\ast(w)=V(w)+n(g(w)-2)$ at $(m,n)=(6,3)$ (Definition~\ref{def:r_star});
the normalized depth $\widehat r$ and mismatch $\Delta r=r-\widehat r$ are defined in Section~\ref{sec:mass_spectrum_closure}.
\item \textbf{Defect channels.}
$D_\varphi$ detects the forbidden substring ``11'';
$D_\pi$ detects cyclic wrap-around violation $w_1=w_m=1$;
the $\e$-channel is expressed via the normalized zeta function $\zeta_\e(r)=\zeta(r/\varphi)$.
\item \textbf{Dyadic phase registers.}
Phases are modeled by $\ZZ_{2^p}$ with the embedding $k\mapsto \e^{2\pi\iu k/2^p}$ (Appendix~\ref{app:protocol_primitives}); $\ZZ_{128}$ is the baseline choice at $p=7$ (Section~\ref{subsec:z128_label}).
\item \textbf{Abel path and pole barrier.}
An Abel-normalized sum is $F(\rho)=\sum_{n\ge 0} a_n\rho^n$ with the Abel path $\rho\uparrow 1$ (Definition~\ref{def:abel_sum}); the first singularity that obstructs analytic continuation along this path is the pole barrier (Appendix~\ref{app:protocol_primitives} and Section~\ref{subsec:e_channel}).
\item \textbf{Abel finite part.}
If an Abel generating function admits an expansion $c_{-1}/(1-r)+c_0+\cdots$ as $r\uparrow 1$, the Abel finite part is the constant term $\FP_{r\uparrow 1}(\cdot)=c_0$ (Appendix~\ref{app:abel_finite_part_notes}).
\item \textbf{Folding map.}
$\mathrm{Fold}_m$ maps integers (microstate indices) to stable types by Zeckendorf digits and truncation; $\mathrm{Fold}_6$ gives the $64\to 21$ stable projection.
\item \textbf{Hilbert addressing.}
$H_n:\{0,\dots,4^n-1\}\to\{0,\dots,2^n-1\}^2$ is a locality-preserving address map with a dihedral layout family $D_4$.
\item \textbf{Hilbert chirality index.}
For the Hilbert path points $p_k=H_n(k)$, $\chi$ is the net signed turning index defined by \eqref{eq:hilbert_chi_def}.
\item \textbf{Compton clock dictionary (matching layer).}
$\omega_C(\mu)=\mu c^2/\hbar$ and $\tau_C(\mu)=1/\omega_C(\mu)$ (Section~\ref{app:time_mass_delay}).
\item \textbf{Frequency (tick units).}
$\omega=\Delta\theta/\Delta t$ denotes phase advance per tick (Definition~\ref{def:frequency_from_phase}); this is the primary ratio-level bridge to energy/mass/temperature in the frequency-first closure (Appendix~\ref{app:equivalence_semantics}).
\item \textbf{Wigner--Smith delay (operational proxy).}
$Q(\omega)=-\iu\,S(\omega)^\dagger \dd S/\dd\omega$ and $\tau_{\mathrm{WS}}(\omega)=\Tr Q(\omega)$ (Section~\ref{app:time_mass_delay}).
\item \textbf{Continuum representative (CAP-closed).}
$S_{\mathrm{eff}}$ denotes the CAP-selected continuum action skeleton (Appendix~\ref{app:cap_continuum_action_closure}); varying it yields Einstein--Yang--Mills--$\chi$ equations (Appendix~\ref{app:variational_field_equations}).
\item \textbf{Overhead, $\chi$, and lapse.}
$\kappa(x)$ denotes a local overhead and $\chi(x)=\log(\kappa/\kappa_0)$; $N(x)=\e^{-\gamma\chi(x)}$ is the lapse proxy in the overhead-to-gravity closure (Appendix~\ref{app:overhead_to_gravity_closure}).
\item \textbf{$\chi$ reconstruction protocol.}
An executable Hilbert-binning $\to$ window-word $\to$ folding-statistics pipeline reconstructs $\chi(x)$ from data/simulations (Appendix~\ref{app:chi_reconstruction_protocol}).
\item \textbf{Born probabilities (quantum readout).}
Finite-resolution readout is modeled by POVMs with probabilities $P_k=\Tr(\rho E_k)$ and instrument updates in Kraus form (Appendix~\ref{app:quantum_measurement_born}).
\item \textbf{RG in the $r$ coordinate.}
Running in scale is expressed in $r$ by $\dd g/\dd r=(\log\varphi)\beta(g)$ (Appendix~\ref{app:running_couplings_resolution_flow}).
\item \textbf{Cosmology as resolution flow.}
Stable/hidden fractions are $f_{\mathrm{stab}}(m)=F_{m+2}/2^m$ and $f_{\mathrm{hid}}(m)=1-f_{\mathrm{stab}}(m)$; mean degeneracy is $d_m=2^m/F_{m+2}$ (Appendix~\ref{app:cosmology_resolution_flow}).
\item \textbf{Gauss map (optional mother-space notes).}
The Gauss map is $G(\xi)=\{1/\xi\}$ on $(0,1)$; its invariant Gauss measure and digit law provide a canonical dynamical source for continued fractions (Appendix~\ref{app:modular_flow_gauss_map}).
\item \textbf{Hecke operators and prime skeleton (optional notes).}
Hecke operators $T_n$ act on modular forms and satisfy multiplicative relations and prime-power recursions; primes generate the Hecke algebra and Euler products factorize into prime local factors (Appendix~\ref{app:hecke_prime_skeleton}).
\item \textbf{Thermodynamic closure objects.}
$S$ denotes a coarse-grained entropy (state-count/channel-count); $T$ is the conjugate temperature scale; $\mathcal{F}=E-TS$ is a free-energy functional used in CAP closure form (Appendix~\ref{app:thermodynamics_from_equivalence}).
\item \textbf{Black-hole scales (external targets).}
$R_s=2GM/c^2$, $A=4\pi R_s^2$, $\ell_P^2=G\hbar/c^3$, and $S_{\mathrm{BH}}=k_BA/(4\ell_P^2)$; $T_H$ denotes the Hawking temperature (Appendix~\ref{app:bh_wormholes_pointer}).
\item \textbf{Isotropic radius and inversion (external template).}
For Schwarzschild exterior geometry, $\rho>0$ denotes the isotropic radius with throat radius $\rho_h=R_s/4$ and inversion $\mathcal{I}(\rho)=\rho_h^2/\rho$ (Proposition~\ref{prop:er_isotropic_inversion}).
\item \textbf{Wormhole-like pointer jump (protocol-level).}
A directed (or undirected) pointer link $a\xrightarrow{\mathrm{ptr}}b$ on the scan index set defines a wormhole-like shortcut channel in the readout protocol (Definition~\ref{def:wormhole_pointer_jump}).
\end{itemize}
