\section{Scan--Projection Generator: Group Actions and Binary Readout}\label{sec:spg}

\paragraph{Singly generated group action}

Let $\Gamma=\langle g\rangle$ be a singly generated (semi)group. Given an action on the phase space $X$, write
\[
T^t:X\to X,\quad T^t(x)=g^t\cdot x,\qquad t\in\ZZ\ (\text{or }t\in\NN),
\]
so that ``time'' in this paper is the iteration count $t$.

\paragraph{Projection readout and binary record}

Given a measurable set $W\subseteq X$, define the binary readout (event indicator function)
\[
s_t(x)=\ind_W(T^t x)\in\{0,1\}.
\]
For a fixed initial value $x$, the local readout window of length $m$ produces the microstate word
\[
\omega_m(x;t)=s_t(x)s_{t+1}(x)\cdots s_{t+m-1}(x)\in \Omega_m:=\{0,1\}^m.
\]
$\Omega_m$ is the set of all observable microstates at resolution $m$, with cardinality $\abs{\Omega_m}=2^m$.

The equivalent formulation of this interface in the language of compact groups/operator algebras is used only as a bridging interface; see Appendix \ref{app:op-algebra}.

\paragraph{The null projection paradigm for concepts}

In this framework, concepts are not regarded as a priori objects but as ``addressable events''.
\begin{itemize}
  \item \textbf{Address:} Any finite word $a\in\{0,1\}^k$ serving as an index.
  \item \textbf{Concept:} The event specified by address $a$
  \[
  C_a=\{x\in X:\omega_k(x;0)=a\}.
  \]
  \item \textbf{Time-shifted cylinder event:} For any $t\in\ZZ$, define $C_{a,t}:=\{x\in X:\omega_k(x;t)=a\}=T^{-t}C_a$.
  \item \textbf{Readout (evaluation):} For a given $x$, the concept value is the binary projection $\ind_{C_a}(x)\in\{0,1\}$.
\item \textbf{Null value:} Before the address $a$ is given, $\ind_{C_a}$ is undefined; this ``undefined'' is not a value of $0$ but rather a lack of reference.
\end{itemize}

The key conclusion is: \textbf{address precedes concept value}. Moreover, in the recursive mechanism, the address of a new concept is constructed from the readout sequences of old concepts (see Section \ref{sec:recursive-addressing}).

The measurability and shift relation $C_{a,t}=T^{-t}C_a$ of cylinder events follow directly from the definitions.

\paragraph{Prefix ultrametric and cylinder topology}\label{par:spg-prefix-ultrametric}
The infinite readout word is viewed as a point in the symbol space. Define
\[
\omega_\infty(x):=s_0(x)s_1(x)s_2(x)\cdots\in\Omega_\infty:=\{0,1\}^{\NN},
\]
and for any finite word (address) $a\in\{0,1\}^k$, define its corresponding \textbf{symbolic cylinder set}
\[
[a]:=\{\xi\in\Omega_\infty:\ \xi_0\xi_1\cdots\xi_{k-1}=a\}.
\]
Then the cylinder event can be written as $C_a=\omega_\infty^{-1}([a])$.
\par
For $\xi,\eta\in\Omega_\infty$, define the common prefix depth
\[
\ell(\xi,\eta):=\sup\{k\ge 0:\ \xi_0\cdots\xi_{k-1}=\eta_0\cdots\eta_{k-1}\}\in\NN\cup\{\infty\},
\]
and define the prefix ultrametric (with the convention $2^{-\infty}=0$)
\[
d_{\mathrm{pre}}(\xi,\eta):=2^{-\ell(\xi,\eta)}.
\]
The balls of this ultrametric coincide with cylinder sets: for any $a$ and any $\xi\in[a]$,
\[
[a]=\overline{B}\bigl(\xi,2^{-|a|}\bigr)\qquad(|a|=k).
\]
See \cite{LindMarcus1995}. Thus prefix refinement corresponds to monotone contraction of the radius. Pulling back this ultrametric to the phase space via $\omega_\infty$ yields the pseudo-ultrametric $d_X(x,y):=d_{\mathrm{pre}}(\omega_\infty(x),\omega_\infty(y))$; if the readout is not injective, collisions $d_X(x,y)=0$ may occur, which is compatible with the ``single-valued verifiable readout / injectivized record axis / null $\mathsf{NULL}$'' caliber discussed later.

\subsection{Scan Depth and Clarity: Decidability, Error Profile, and Boundary Cylinders}\label{subsec:spg-scan-clarity}
This subsection, on the ``address precedes concept value'' interface, compresses ``deeper scans yield greater clarity'' into auditable quantitative laws: resolution $m$ gives a finite information layer; for any proposition $\mathsf P$, its optimal misclassification probability at that information layer constitutes a monotone profile $m\mapsto \varepsilon_m(\mathsf P;\mu)$. The subsequent joint treatment of synchronization subtraction and Chebotarev budgets will take this profile as the input for ``definitional clarity'' (see Remark \ref{rem:spg-sync-null-interface} and Theorem \ref{thm:pom-sync-subtracted-chebotarev-capacity}).

\paragraph{Cylinder algebra and clopen-ness}
Let $\tau$ be the product topology on $\Omega_\infty$. For $m\ge 1$, write
\[
\mathcal{A}_m:=\sigma([a]:a\in\{0,1\}^m),
\qquad
\mathcal{A}_\infty:=\sigma\!\left(\bigcup_{m\ge 1}\mathcal{A}_m\right).
\]
and let
\[
\mathcal{F}_m:=\omega_\infty^{-1}(\mathcal{A}_m),
\qquad
\mathcal{F}_\infty:=\omega_\infty^{-1}(\mathcal{A}_\infty),
\]
which give the observable closure ``allowing access only to length-$m$ address prefix information''; this is consistent with the ``$\sigma$-algebra generated by finite-window events'' caliber from Section \ref{sec:preliminaries}.

\begin{proposition}[Decidability and clopen-ness]\label{prop:spg-decidable-clopen}
Let $P\subseteq\Omega_\infty$ be a Borel set. The following conditions are equivalent:
\begin{enumerate}
  \item $P\in\mathcal{A}_m$ (i.e., determined solely by length-$m$ prefixes);
  \item There exists $A\subseteq\{0,1\}^m$ such that $P=\bigcup_{a\in A}[a]$;
  \item $P$ is clopen in the topology $\tau$, and can be written as a union of closed balls of radius $2^{-m}$ in the prefix ultrametric $d_{\mathrm{pre}}$.
\end{enumerate}
Consequently, for an event $\mathsf P\in\mathcal{F}_\infty$ on $X$, if $\mathsf P\in\mathcal{F}_m$, then its readout depends only on the resolution-$m$ address prefix.
\end{proposition}

\paragraph{Scan error profile}
\begin{definition}[Scan error profile]\label{def:spg-scan-error-profile}
Under a given $T$-invariant probability measure $\mu$, for any $\mathsf P\in\mathcal{F}_\infty$, define
\[
\varepsilon_m(\mathsf P;\mu)
:=\inf\left\{\mu(g\neq \ind_{\mathsf P}): g\in L^\infty(\mathcal{F}_m),\ g(X)\subseteq\{0,1\}\right\}.
\]
\end{definition}

Let $\mathcal{C}_m:=\{C_a:a\in\{0,1\}^m\}$ be the resolution-$m$ cylinder partition (empty sets allowed). Then the cylinder decomposition expression holds:

\begin{proposition}[Exact expression via cylinder decomposition]\label{prop:spg-scan-error-cylinder}
Following the notation of Definition \ref{def:spg-scan-error-profile},
\[
\varepsilon_m(\mathsf P;\mu)
=\sum_{C\in\mathcal{C}_m}\min\{\mu(\mathsf P\cap C),\mu(C\setminus \mathsf P)\}.
\]
Equivalently, let $p_C:=\mu(\mathsf P\,|\,C)$ (taking $p_C:=0$ when $\mu(C)=0$), then
\[
\varepsilon_m(\mathsf P;\mu)
=\sum_{C\in\mathcal{C}_m}\mu(C)\min\{p_C,1-p_C\}
=\mathbb{E}_\mu\bigl[\min\{p_m,1-p_m\}\bigr],
\qquad
p_m:=\mu(\mathsf P\,|\,\mathcal{F}_m).
\]
\end{proposition}

\begin{corollary}[Basic properties of clarity]\label{cor:spg-clarity-basic}
$m\mapsto\varepsilon_m(\mathsf P;\mu)$ is monotonically non-increasing; and $\varepsilon_m(\mathsf P;\mu)=0$ if and only if $\mathsf P\in\mathcal{F}_m$.
\end{corollary}

\paragraph{Boundary cylinders and convergence exponent}
\begin{definition}[Boundary cylinder covering number and cylinder Minkowski dimension]\label{def:spg-boundary-cylinder-dimension}
Define the resolution-$m$ boundary cylinder covering number
\[
N_m(\partial\mathsf P)
:=\#\left\{C\in\mathcal{C}_m:\ \mu(\mathsf P\cap C)>0\ \text{and}\ \mu(C\setminus \mathsf P)>0\right\}.
\]
If there exists a constant $\lambda_{\mathrm{cyl}}>1$ such that the cylinder complexity satisfies
\[
\#\{C\in\mathcal{C}_m:\mu(C)>0\}\asymp \lambda_{\mathrm{cyl}}^m,
\]
then define the upper cylinder dimension
\[
\overline{\dim}_B(\partial\mathsf P)
:=\limsup_{m\to\infty}\frac{\log N_m(\partial\mathsf P)}{m\log\lambda_{\mathrm{cyl}}}\in[0,1].
\]
\end{definition}

\begin{theorem}[Clarity convergence controlled by boundary dimension]\label{thm:spg-clarity-boundary-dimension}
Suppose there exist constants $K<\infty$ and $\lambda_{\mathrm{cyl}}>1$ such that for all $m$ and all $C\in\mathcal{C}_m$,
\[
\mu(C)\le K\lambda_{\mathrm{cyl}}^{-m}.
\]
Then for any $\mathsf P\in\mathcal{F}_\infty$,
\[
\varepsilon_m(\mathsf P;\mu)\le K\lambda_{\mathrm{cyl}}^{-m}N_m(\partial\mathsf P).
\]
In particular, if $\overline{\dim}_B(\partial\mathsf P)=d<1$, then
\[
\varepsilon_m(\mathsf P;\mu)\le \lambda_{\mathrm{cyl}}^{-(1-d)m+o(m)}.
\]
\end{theorem}

\begin{remark}[Specialization to the golden mean stable grammar]\label{rem:spg-golden-mean-specialization}
When the visible symbolic system is the golden mean subshift (forbidding adjacent $11$) with its maximum entropy (Parry) measure, we have $\lambda_{\mathrm{cyl}}=\varphi$;
hence the exponential rate in Theorem \ref{thm:spg-clarity-boundary-dimension} is controlled by $(1-d)\log\varphi$.
\end{remark}

\begin{remark}[Interface with the synchronization gate and null semantics]\label{rem:spg-sync-null-interface}
The above profile defines ``clarity'' as the optimal misclassification probability on a given $\mathcal{F}_m$ information layer.
Under sofic factor readout, if the length prefix has not yet synchronized, the same prefix may correspond to multiple internal suffixes; under the ``address precedes concept value'' convention, this can be equivalently viewed as temporal-layer statistical-type $\mathsf{NULL}$;
see Remark \ref{rem:Ym-stat-null-unsync} for the temporal-layer characterization, and Theorem \ref{thm:Ym-sync-tail} for the synchronization time budget and tail bound.
The deduction correction imposed by this ``synchronization gate'' on the Chebotarev capacity condition is given in Theorem \ref{thm:pom-sync-subtracted-chebotarev-capacity}.
\end{remark}

\begin{theorem}[Double-budget addressing capacity: composite clarity law for statistical-type \texorpdfstring{$\mathsf{NULL}$}{NULL} and de-collision]\label{thm:spg-double-budget-address-capacity}
Fix resolution \(m\) and consider the ``address precedes value'' semantic convention: unsynchronized means no address (temporal-layer statistical-type \(\mathsf{NULL}\), see Remark \ref{rem:Ym-stat-null-unsync}).
Let \(T_m\) be the synchronization time under the deterministic right-resolving representation (Definition \ref{def:Ym-sync-time}), and let \(\nu_m\) be the corresponding maximum entropy measure.
On the other hand, let \(\nu_{u,m}\) be the pushforward distribution of the temperature gate \(\mathrm{PROJ}_u\) on the length-\(m\) address space \(A_{\mathrm{out}}^m\) (Proposition \ref{prop:pom-proj-mom-psi-law}), and write the second-order collision probability
\[
\mathrm{Col}_u(m):=\sum_{x\in A_{\mathrm{out}}^m}\nu_{u,m}(x)^2
\]
(Definition \ref{def:pom-col-u-h2}).
\par
For \(N\) independent objects, with ``time budget \(n\)'' to eliminate statistical-type \(\mathsf{NULL}\) and ``resolution budget \(m\)'' to produce addresses, denote the failure event \(\mathsf{Fail}_{m,n,N}\) as ``there exists a \(\mathsf{NULL}\) or there exists an address collision''.
Then the following strict upper bound holds:
\begin{equation}\label{eq:spg-double-budget-fail-bound}
\boxed{\;
\mathbb{P}(\mathsf{Fail}_{m,n,N})
\ \le\
N\,\nu_m(T_m>n)\ +\ \binom{N}{2}\,\mathrm{Col}_u(m)\; }.
\end{equation}
\par
Furthermore, if \(Y_m^{\mathrm{amb}}\neq\varnothing\) and \(\varepsilon_m=\log 2-h_{\mathrm{top}}(Y_m^{\mathrm{amb}})\) (Theorem \ref{thm:Ym-sync-tail}), then there exists a constant \(C_m>0\) such that \(\nu_m(T_m>n)\le C_m e^{-\varepsilon_m n}\);
and by \(h_2(u):=-\Psi(2,u)\) (Definition \ref{def:pom-col-u-h2}),
\(
\mathrm{Col}_u(m)=\exp(-h_2(u)\,m+o(m))
\).
Therefore, for any target failure probability \(\delta\in(0,1)\), when \(n\) and \(m\) satisfy
\[
n\ \ge\ \frac{1}{\varepsilon_m}\Bigl(\log(NC_m)+\log\frac{2}{\delta}\Bigr),
\qquad
\binom{N}{2}\,\mathrm{Col}_u(m)\ \le\ \frac{\delta}{2},
\]
it is guaranteed that \(\mathbb{P}(\mathsf{Fail}_{m,n,N})\le\delta\).
Moreover, at the exponential scale as \(m\to\infty\), since \(\mathrm{Col}_u(m)=\exp(-h_2(u)m+o(m))\), for sufficiently large \(m\) the latter inequality can be guaranteed by the explicit budget
\[
m\ \ge\ \frac{2}{h_2(u)}\Bigl(\log N+\log\frac{2}{\delta}\Bigr)
\]
(absorbing the \(o(m)\) into the ``sufficiently large'' threshold).
At the exponential scale, the above double-budget law equivalently gives the clarity upper limit on the addressable scale
\[
\log N\ \lesssim\ \min\Bigl\{\varepsilon_m\,n,\ \tfrac12 h_2(u)\,m\Bigr\}.
\]
\end{theorem}

\begin{proof}
For the \(i\)-th object, denote the ``unsynchronized'' event as \(\mathsf{Null}_i:=\{T_m^{(i)}>n\}\), and denote the ``collision occurs'' event as
\(
\mathsf{Col}:=\bigcup_{1\le i<j\le N}\{X_i=X_j\}
\), where \(X_i\sim\nu_{u,m}\) are i.i.d.
Then
\(
\mathsf{Fail}_{m,n,N}\subseteq \bigcup_{i=1}^N \mathsf{Null}_i\ \cup\ \mathsf{Col}
\).
By the union bound,
\[
\mathbb{P}(\mathsf{Fail}_{m,n,N})
\le
\sum_{i=1}^N \mathbb{P}(\mathsf{Null}_i)
\ +\
\sum_{1\le i<j\le N}\mathbb{P}(X_i=X_j).
\]
Under the i.i.d.\ assumption, \(\mathbb{P}(\mathsf{Null}_i)=\nu_m(T_m>n)\), and
\(
\mathbb{P}(X_i=X_j)=\sum_x\nu_{u,m}(x)^2=\mathrm{Col}_u(m)
\); substituting back yields \eqref{eq:spg-double-budget-fail-bound}.
\par
The subsequent budget conditions follow directly from \(\nu_m(T_m>n)\le C_m e^{-\varepsilon_m n}\) (Theorem \ref{thm:Ym-sync-tail}) and the explicit inequality \(\binom{N}{2}\mathrm{Col}_u(m)\le \delta/2\) for the collision term; the exponential-scale \(\min\{\varepsilon_m n,\tfrac12 h_2(u)m\}\) is obtained by taking logarithms and ignoring constant terms. This completes the proof.
\end{proof}

\begin{remark}[Interface with fold visibility]\label{rem:spg-visibility-interface}
If the information layer is replaced from $\mathcal{F}_m$ by $\sigma(\Fold_m)$ (the stable type visible domain), one can similarly define the corresponding error profile;
its relationship with the $L^2$-caliber ``fold visibility'' $\mathcal{R}_{m,\mu}(f)$ (Definition \ref{def:pom-fold-visibility}) is given by a parameter-free two-sided estimate; see Theorem \ref{thm:pom-visibility-to-error-two-sided}.
\end{remark}
