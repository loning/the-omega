\paragraph{Spectral Algebra of the Green Kernel: Gram Decomposition, Chebyshev Identification, and Renormalization Constants}
This paragraph provides additional clarification on several \emph{integer/spectral/determinantal} rigidity consequences of the fence kernel \(K_k(i,j)=\min(i,j)\).
These results promote the previously appearing \(4\sin^2(\cdot)\) form from a ``half-angle identity'' to an auditable closed constraint chain based on \emph{Gram normalization + Chebyshev integer coefficients}.

\begin{lemma}[Gram decomposition and determinantal normalization]\label{lem:pom-Kk-gram-det}
Let \(K_k\in\ZZ^{k\times k}\) be defined by \(K_k(i,j)=\min(i,j)\).
Take the unit lower triangular matrix \(B_k\in\ZZ^{k\times k}\) satisfying
\[
(B_k)_{ij}:=\mathbf{1}_{\{j\le i\}}.
\]
Then the following strict Gram decomposition holds:
\[
K_k=B_kB_k^{\mathsf T},
\qquad
\det(B_k)=1,
\qquad
\det(K_k)=1.
\]
\end{lemma}
\begin{proof}
For any \(1\le i,j\le k\),
\[
(B_kB_k^{\mathsf T})_{ij}
=\sum_{t=1}^{k}(B_k)_{it}(B_k)_{jt}
=\sum_{t=1}^{k}\mathbf{1}_{\{t\le i\}}\mathbf{1}_{\{t\le j\}}
=\sum_{t=1}^{\min(i,j)}1
=\min(i,j),
\]
hence \(K_k=B_kB_k^{\mathsf T}\).
Since \(B_k\) is unit lower triangular, \(\det(B_k)=1\), and therefore \(\det(K_k)=\det(B_k)^2=1\). \qed
\end{proof}

\begin{corollary}[\texorpdfstring{\((4\sin^2)\)}{(4 sin^2)} product identity and sum identity]\label{cor:pom-Kk-sine-product-sum}
Let
\(
\phi_p:=\frac{(2p-1)\pi}{4k+2}
\) (\(p=1,\dots,k\)).
Then the following strict identities hold:
\[
\prod_{p=1}^{k}4\sin^2(\phi_p)=1,
\qquad
\sum_{p=1}^{k}\frac{1}{4\sin^2(\phi_p)}=\frac{k(k+1)}{2}.
\]
Therefore the coefficient \(4\) in the spectral representation is constrained by determinantal normalization rigidity: under the requirement of preserving the integer Gram structure and \(\det(K_k)=1\),
\((4\sin^2(\cdot))\) cannot be rescaled away.
\end{corollary}
\begin{proof}
By Lemma \ref{lem:pom-Kk-eigenvalues} and Remark \ref{rem:pom-Kk-factor4-spectral-rigidity}, the eigenvalues of \(K_k\) are
\(
\lambda_p=(4\sin^2\phi_p)^{-1}
\).
By Lemma \ref{lem:pom-Kk-gram-det}, \(\det(K_k)=1=\prod_p\lambda_p\), hence \(\prod_p 4\sin^2(\phi_p)=1\).
Furthermore, \(\mathrm{tr}(K_k)=\sum_{i=1}^k K_k(i,i)=\sum_{i=1}^k i=k(k+1)/2\), and \(\mathrm{tr}(K_k)=\sum_p\lambda_p\), yielding the sum identity. \qed
\end{proof}

\begin{theorem}[Chebyshev identification and integer-coefficient normalization of \texorpdfstring{\(\lambda/4\)}{lambda/4}]\label{thm:pom-Lk-chebyshev-charpoly}
Denote \(L_k:=K_k^{-1}\in\ZZ^{k\times k}\) (whose explicit tridiagonal form is given in Remark \ref{rem:pom-Kk-factor4-spectral-rigidity}).
Let \(T_n\) be the Chebyshev polynomial of the first kind (\(T_n(\cos\theta)=\cos(n\theta)\)), and define
\[
P_k(y):=\frac{T_{2k+1}(\sqrt y)}{\sqrt y}\in\ZZ[y].
\]
Then for any \(\lambda\in\CC\), the characteristic polynomial has the closed form
\[
\det(\lambda I-L_k)=(-1)^k\,P_k\!\Bigl(1-\frac{\lambda}{4}\Bigr).
\]
Equivalently, the spectral points of \(L_k\) are
\[
\mu_p=4\sin^2\!\Bigl(\frac{(2p-1)\pi}{4k+2}\Bigr),\qquad p=1,\dots,k,
\]
and the linear transformation \(\lambda\mapsto 1-\lambda/4\) is the natural normalization under which \(\det(\lambda I-L_k)\) achieves integer-coefficient identification with \(T_{2k+1}\).
\end{theorem}
\begin{proof}
Since \(2k+1\) is odd, \(T_{2k+1}(z)\) is an odd polynomial, so \(T_{2k+1}(z)/z\) is an even polynomial in \(z\) and can be uniquely written as \(P_k(z^2)\); moreover, \(T_n\) has integer coefficients, hence \(P_k\in\ZZ[y]\).
\par
By Remark \ref{rem:pom-Kk-factor4-spectral-rigidity}, the spectrum of \(L_k\) satisfies
\(
\mu_p=4\sin^2(\phi_p)
\)
where \(\phi_p=\frac{(2p-1)\pi}{4k+2}\).
For any such \(\mu_p\), set \(y_p:=1-\mu_p/4=\cos^2(\phi_p)\).
Then \(T_{2k+1}(\cos\phi_p)=\cos((2k+1)\phi_p)=\cos((2p-1)\pi/2)=0\), hence
\(
P_k(y_p)=T_{2k+1}(\cos\phi_p)/\cos\phi_p=0
\).
Therefore the zero set of the polynomial \(\lambda\mapsto P_k(1-\lambda/4)\) is exactly \(\{\mu_p\}_{p=1}^k\), coinciding with the spectral zeros of \(\det(\lambda I-L_k)\);
since both are degree-\(k\) monic (up to sign) polynomials, the stated identity follows. \qed
\end{proof}

\begin{corollary}[Closed form for the shifted determinant, free energy density, and renormalization constant term]\label{cor:pom-Lk-shifted-det-free-energy}
For any real parameter \(t>-4\), set \(x:=\sqrt{1+t/4}>0\).
Then the following strict identity holds:
\[
\det(L_k+tI)=\det(I+tK_k)=P_k\!\Bigl(1+\frac{t}{4}\Bigr)=\frac{T_{2k+1}(x)}{x}.
\]
When \(x>1\), letting \(\eta:=\mathrm{arcosh}(x)\),
\[
\log\det(L_k+tI)=\log\cosh\bigl((2k+1)\eta\bigr)-\log x.
\]
Furthermore, the free energy density limit exists:
\[
\lim_{k\to\infty}\frac{1}{2k+1}\log\det(L_k+tI)=\eta=\mathrm{arcosh}\!\Bigl(\sqrt{1+\frac{t}{4}}\Bigr).
\]
In addition, defining the renormalization remainder
\[
F_k(t):=\log\det(L_k+tI)-(2k+1)\,\mathrm{arcosh}\!\Bigl(\sqrt{1+\frac{t}{4}}\Bigr),
\]
the limit exists and equals an explicit constant:
\[
\lim_{k\to\infty}F_k(t)=-\log\!\Bigl(2\sqrt{1+\frac{t}{4}}\Bigr).
\]
\end{corollary}
\begin{proof}
By Theorem \ref{thm:pom-Lk-chebyshev-charpoly}, substituting \(\lambda=-t\) and noting that
\(
\det(-tI-L_k)=(-1)^k\det(L_k+tI)
\)
yields \(\det(L_k+tI)=P_k(1+t/4)\).
Moreover, from \(K_kL_k=I\) and \(\det(K_k)=1\) (Lemma \ref{lem:pom-Kk-gram-det}),
\(
I+tK_k=K_k(L_k+tI)
\)
so that \(\det(I+tK_k)=\det(L_k+tI)\).
When \(x>1\), \(T_{2k+1}(x)=\cosh((2k+1)\eta)\); substituting back gives the logarithmic form.
Finally, from \(\log\cosh((2k+1)\eta)=(2k+1)\eta-\log 2+o(1)\), the density limit and constant term limit follow. \qed
\end{proof}

\begin{corollary}[Fully explicit spectral gap constant]\label{cor:pom-Lk-gap-pi2}
Let \(\mu_1\le\mu_2\le\cdots\le\mu_k\) be the eigenvalues of \(L_k\) in ascending order. Then
\[
\mu_1=4\sin^2\!\Bigl(\frac{\pi}{4k+2}\Bigr),
\qquad
\mu_2=4\sin^2\!\Bigl(\frac{3\pi}{4k+2}\Bigr),
\qquad
\frac{\mu_2}{\mu_1}=\frac{\sin^2\!\bigl(\frac{3\pi}{4k+2}\bigr)}{\sin^2\!\bigl(\frac{\pi}{4k+2}\bigr)}.
\]
Moreover, as \(k\to\infty\),
\[
\mu_1=\frac{\pi^2}{(2k+1)^2}+O(k^{-4}),
\qquad
\lambda_{\max}(K_k)=\mu_1^{-1}=\frac{(2k+1)^2}{\pi^2}+O(1).
\]
\end{corollary}
\begin{proof}
The expressions for the smallest/second-smallest eigenvalues follow from the spectral closed form in Remark \ref{rem:pom-Kk-factor4-spectral-rigidity} and the monotonicity of \(\phi_p=\frac{(2p-1)\pi}{4k+2}\).
The last two expressions follow from \(\sin x=x+O(x^3)\), yielding \(\mu_1=4(\pi/(4k+2))^2+O(k^{-4})\), and taking the reciprocal gives \(\lambda_{\max}(K_k)=\mu_1^{-1}\). \qed
\end{proof}

\begin{proposition}[Bernoulli--polynomial law for higher moments: odd-angle \texorpdfstring{\(\csc^{2r}\)}{csc^{2r}} power sums]\label{prop:pom-csc-moments-bernoulli}
For integer \(r\ge 1\), define
\[
S_r(k):=\sum_{p=1}^{k}\csc^{2r}\!\Bigl(\frac{(2p-1)\pi}{4k+2}\Bigr),
\qquad
n:=2k+1.
\]
Then \(S_r(k)\) is an even polynomial in \(n\), and there exist rational coefficients \(a_{r,j}\in\QQ\) such that
\[
S_r(k)=-\frac12+\sum_{j=1}^{r} a_{r,j}\,n^{2j}.
\]
The leading coefficient is given by Bernoulli numbers:
\[
a_{r,r}=\frac{2^{2r-1}(2^{2r}-1)\,|B_{2r}|}{(2r)!}.
\]
Furthermore, the trace moments of \(K_k\) satisfy
\[
\mathrm{tr}(K_k^{\,r})=\sum_{p=1}^{k}\lambda_p^{\,r}=\frac{1}{4^{\,r}}\,S_r(k),
\]
where \(\lambda_p=(4\sin^2\phi_p)^{-1}\) are the eigenvalues of \(K_k\) (Lemma \ref{lem:pom-Kk-eigenvalues}).
\end{proposition}
\begin{proof}
Let
\(
A_r(m):=\sum_{q=1}^{m-1}\csc^{2r}(\pi q/m)
\).
By the partial fraction expansion of \(\pi\cot(\pi z)\) and residue computation, \(A_r(m)\) is an even polynomial in \(m\), with leading coefficient \(2^{2r}|B_{2r}|/(2r)!\).
Decompose the sum at \(m=2n\) by odd/even indices:
\[
A_r(2n)=\sum_{q=1}^{n-1}\csc^{2r}\!\Bigl(\frac{\pi q}{n}\Bigr)+\sum_{\substack{1\le q\le 2n-1\\ q\ \mathrm{odd}}}\csc^{2r}\!\Bigl(\frac{\pi q}{2n}\Bigr)
=A_r(n)+\Bigl(2S_r(k)+1\Bigr),
\]
where the last step uses sine symmetry to pair the odd-index sum into \(2S_r(k)\), and separates the \(q=n\) term \(\csc^{2r}(\pi/2)=1\).
Hence
\(
S_r(k)=\frac12(A_r(2n)-A_r(n)-1)
\), so \(S_r(k)\) is an even polynomial in \(n\) with constant term \(-1/2\).
Writing \(c\) for the leading coefficient of \(A_r(m)\), the coefficient of \(n^{2r}\) in \(A_r(2n)\) is \(2^{2r}c\), while that in \(A_r(n)\) is \(c\), so the leading coefficient of \(S_r(k)\) is
\[
\frac{2^{2r}-1}{2}\,c.
\]
Substituting \(c=2^{2r}|B_{2r}|/(2r)!\) yields
\(
a_{r,r}=\frac{2^{2r-1}(2^{2r}-1)\,|B_{2r}|}{(2r)!}
\).
The final equation follows from \(\lambda_p=(4\sin^2\phi_p)^{-1}=(1/4)\csc^2\phi_p\) by direct substitution. \qed
\end{proof}

\paragraph{Spectral Measure and Continuum Limit: Arcsine Law, Discrete Weyl Law, and Heat Kernel/Bessel Duality}
The following promotes the ``half-odd-integer quantization'' of the mixed-boundary discrete Laplacian \(L_k\) to a \emph{spectral measure limit law},
and provides a discrete Weyl counting remainder bound over the full interval together with the closed-form limit of the heat kernel density.

\begin{theorem}[Arcsine limit law for the empirical spectral measure]\label{thm:pom-Lk-arcsine-law}
Let \(L_k:=K_k^{-1}\) (see Remark \ref{rem:pom-Kk-factor4-spectral-rigidity}), with eigenvalues
\[
\mu_p
=4\sin^2\!\Bigl(\frac{(2p-1)\pi}{4k+2}\Bigr)
=2\Bigl(1-\cos\frac{(2p-1)\pi}{2k+1}\Bigr)\in(0,4),
\qquad p=1,\dots,k.
\]
Define the empirical spectral measure
\[
\nu_k:=\frac{1}{k}\sum_{p=1}^{k}\delta_{\mu_p}.
\]
Then \(\nu_k\) converges in the weak topology to the arcsine distribution \(\nu\) on \([0,4]\), whose density is
\[
\dd\nu(\mu)=\frac{1}{\pi\sqrt{\mu(4-\mu)}}\,\dd\mu,\qquad \mu\in(0,4).
\]
Equivalently, for any continuous function \(f\in C([0,4])\),
\[
\lim_{k\to\infty}\frac{1}{k}\operatorname{tr}f(L_k)
=\frac{1}{\pi}\int_{0}^{4}\frac{f(\mu)}{\sqrt{\mu(4-\mu)}}\,\dd\mu
=\frac{1}{\pi}\int_{0}^{\pi} f\!\bigl(2(1-\cos\theta)\bigr)\,\dd\theta.
\]
\end{theorem}
\begin{proof}
Let \(n:=2k+1\) and set \(\theta_p:=\frac{(2p-1)\pi}{n}\in(0,\pi)\), so that \(\mu_p=2(1-\cos\theta_p)\).
Since \(\{\theta_p\}_{p=1}^k\) is a half-odd-integer grid on \((0,\pi)\) with step size \(2\pi/n\),
for any continuous \(g\in C([0,\pi])\), the Riemann sum convergence gives
\[
\lim_{k\to\infty}\frac{1}{k}\sum_{p=1}^{k} g(\theta_p)
=\frac{1}{\pi}\int_0^\pi g(\theta)\,\dd\theta.
\]
Taking \(g(\theta):=f(2(1-\cos\theta))\) yields
\(
\lim_{k\to\infty}\frac{1}{k}\sum_{p} f(\mu_p)=\frac{1}{\pi}\int_0^\pi f(2(1-\cos\theta))\,\dd\theta
\).
Making the substitution \(\mu=2(1-\cos\theta)\) (so that \(\mu(4-\mu)=4\sin^2\theta\) and \(\dd\mu=2\sin\theta\,\dd\theta=\sqrt{\mu(4-\mu)}\,\dd\theta\)) gives
\[
\frac{1}{\pi}\int_0^\pi f(2(1-\cos\theta))\,\dd\theta
=\frac{1}{\pi}\int_0^4 \frac{f(\mu)}{\sqrt{\mu(4-\mu)}}\,\dd\mu,
\]
and the conclusion follows. \qed
\end{proof}

\begin{corollary}[Discrete Weyl law: explicit inverse of the counting function and uniform remainder bound]\label{cor:pom-Lk-discrete-weyl}
Define the counting function \(N_k(\mu):=\#\{p\le k:\ \mu_p\le \mu\}\) (\(\mu\in(0,4)\)), and set
\[
\theta(\mu):=\arccos\Bigl(1-\frac{\mu}{2}\Bigr)\in(0,\pi).
\]
Then for all \(\mu\in(0,4)\),
\[
N_k(\mu)=\left\lfloor \frac{2k+1}{2\pi}\,\theta(\mu)+\frac12\right\rfloor,
\qquad
\Bigl|N_k(\mu)-\frac{2k+1}{2\pi}\theta(\mu)\Bigr|\le 1.
\]
Therefore the distribution function of \(\nu_k\) approximates the distribution function of the arcsine law with \(O(1/k)\) accuracy over the entire interval.
\end{corollary}
\begin{proof}
Since \(\mu_p=2(1-\cos\theta_p)\) and \(\theta\mapsto 2(1-\cos\theta)\) is strictly increasing on \((0,\pi)\),
\(\mu_p\le\mu\) is equivalent to \(\theta_p\le \theta(\mu)\).
Since \(\theta_p=\frac{(2p-1)\pi}{2k+1}\),
\[
\theta_p\le\theta(\mu)
\quad\Longleftrightarrow\quad
2p-1\le \frac{2k+1}{\pi}\,\theta(\mu)
\quad\Longleftrightarrow\quad
p\le \frac{2k+1}{2\pi}\,\theta(\mu)+\frac12.
\]
Taking the largest such integer \(p\) gives the explicit floor expression; the remainder bound follows from \(|\lfloor x\rfloor-x|\le 1\). \qed
\end{proof}

\begin{corollary}[Closed-form limit of the heat kernel density: Bessel--arcsine duality]\label{cor:pom-Lk-heat-kernel-bessel}
Define the heat trace
\(
H_k(t):=\operatorname{tr}(e^{-tL_k})=\sum_{p=1}^{k}e^{-t\mu_p}
\) (\(t\ge 0\)).
Then for any \(t\ge 0\),
\[
\lim_{k\to\infty}\frac{1}{k}H_k(t)
=\frac{1}{\pi}\int_0^\pi e^{-2t(1-\cos\theta)}\,\dd\theta
=e^{-2t}I_0(2t),
\]
where \(I_0\) is the modified Bessel function of the first kind.
When normalized by the ``boundary length'' \(2k+1\),
\[
\lim_{k\to\infty}\frac{1}{2k+1}H_k(t)=\frac12\,e^{-2t}I_0(2t).
\]
\end{corollary}
\begin{proof}
Taking \(f(\mu)=e^{-t\mu}\) in Theorem \ref{thm:pom-Lk-arcsine-law} gives the integral representation of the first expression.
Using the classical integral representation of the Bessel function
\(
I_0(z)=\frac1\pi\int_0^\pi e^{z\cos\theta}\,\dd\theta
\)
with \(z=2t\) yields
\(
\frac{1}{\pi}\int_0^\pi e^{-2t(1-\cos\theta)}\,\dd\theta=e^{-2t}I_0(2t)
\).
The last expression follows from \(\frac{k}{2k+1}\to \frac12\). \qed
\end{proof}

\begin{proposition}[Leading term of the spectral \texorpdfstring{\(\zeta\)}{zeta} function and half-odd-integer Dirichlet factor]\label{prop:pom-Lk-spectral-zeta-dirichlet}
For complex parameter \(s\) (with \(\Re(s)>\tfrac12\)), define the spectral \(\zeta\) function
\[
\zeta_k(s):=\sum_{p=1}^{k}\mu_p^{-s}.
\]
Then as \(k\to\infty\),
\[
\zeta_k(s)
=\Bigl(\frac{2k+1}{\pi}\Bigr)^{2s}\Bigl(1-2^{-2s}\Bigr)\zeta(2s)
\;+\;O(k),
\]
where \(\zeta\) denotes the Riemann \(\zeta\) function.
The factor \(\bigl(1-2^{-2s}\bigr)\) is the Dirichlet projection factor necessarily produced by the half-odd-integer modes (only odd frequencies appear) in the leading term of \(\zeta\).
\end{proposition}
\begin{proof}
Set \(n:=2k+1\) and \(\phi_p=\frac{(2p-1)\pi}{2n}\in(0,\pi/2)\), so that \(\mu_p=4\sin^2\phi_p\) and
\[
\zeta_k(s)=\sum_{p=1}^{k}(2\sin\phi_p)^{-2s}.
\]
By the small-angle expansion, \(\sin x=x+O(x^3)\) gives \((2\sin x)^{-2s}=(2x)^{-2s}\bigl(1+O(x^2)\bigr)\) (when \(\Re(s)>\tfrac12\) this error is summable).
Substituting \(2\phi_p=(2p-1)\pi/n\) yields
\[
(2\sin\phi_p)^{-2s}
=\Bigl(\frac{n}{\pi}\Bigr)^{2s}(2p-1)^{-2s}
\;+\;O\!\Bigl(n^{2\Re(s)-2}(2p-1)^{2-2\Re(s)}\Bigr).
\]
Summing over \(p\le k\), the error term is
\(
O\bigl(n^{2\Re(s)-2}\sum_{p\le k}(2p-1)^{2-2\Re(s)}\bigr)=O(n)=O(k)
\).
Therefore
\[
\zeta_k(s)=\Bigl(\frac{n}{\pi}\Bigr)^{2s}\sum_{p=1}^{k}(2p-1)^{-2s}+O(k).
\]
Since \(\sum_{p=1}^{k}(2p-1)^{-2s}=(1-2^{-2s})\zeta(2s)+O(k^{1-2\Re(s)})\);
multiplying the tail by \(n^{2\Re(s)}\) gives \(O(k)\), and the conclusion follows. \qed
\end{proof}

\begin{corollary}[Strict continuum limit matching: ND determinant limit and the unique scale \texorpdfstring{\(t/4\)}{t/4}]\label{cor:pom-Lk-continuum-ND-det-cosh}
Let the continuous operator \(\mathcal L:=-\frac{\dd^2}{\dd x^2}\) act on the interval \([0,1]\) with Neumann--Dirichlet boundary conditions \(u'(0)=0,\ u(1)=0\).
Its spectrum is \(\{((2p-1)\pi/2)^2\}_{p\ge 1}\), and its mass-shifted determinant satisfies
\(
\det(\mathcal L+s)=\cosh(\sqrt{s})
\) (in the \(\zeta\)-regularized sense).
\par
On the other hand, the discrete spectrum satisfies the small-mode asymptotics
\[
\mu_p=\Bigl(\frac{(2p-1)\pi}{2k+1}\Bigr)^2+O(k^{-4})\qquad(p\ \text{fixed},\ k\to\infty),
\]
so the rescaled operator \(\widetilde L_k:=\frac{(2k+1)^2}{4}L_k\) converges to \(\mathcal L\) in the spectral sense.
More strongly, for any fixed \(s\ge 0\), the determinant limit holds:
\[
\lim_{k\to\infty}\det\!\Bigl(L_k+\frac{4s}{(2k+1)^2}I\Bigr)=\cosh(\sqrt{s}).
\]
Therefore the scale factor \(t/4\) (i.e., adding mass via \(4s/(2k+1)^2\)) is not an optional convention but is structurally forced by the continuum limit matching of the half-odd-integer modes.
\end{corollary}
\begin{proof}
Set \(n:=2k+1\). By Corollary \ref{cor:pom-Lk-shifted-det-free-energy},
\[
\det\!\Bigl(L_k+\frac{4s}{n^2}I\Bigr)
=\frac{T_{n}\!\bigl(\sqrt{1+s/n^2}\bigr)}{\sqrt{1+s/n^2}}.
\]
When \(x:=\sqrt{1+s/n^2}\ge 1\), \(T_n(x)=\cosh(n\,\mathrm{arcosh}(x))\).
Moreover, \(n\,\mathrm{arcosh}(\sqrt{1+s/n^2})=\sqrt{s}+O(n^{-2})\) (from \(\mathrm{arcosh}(1+u)=\sqrt{2u}+O(u^{3/2})\) with \(u=s/(2n^2)+O(n^{-4})\)).
Hence
\[
\det\!\Bigl(L_k+\frac{4s}{n^2}I\Bigr)
=\frac{\cosh(\sqrt{s}+O(n^{-2}))}{1+O(n^{-2})}
\longrightarrow \cosh(\sqrt{s}).
\]
The small-mode eigenvalue asymptotics follow directly from \(\sin x=x+O(x^3)\). \qed
\end{proof}
