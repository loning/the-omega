\subsection{折叠指标与投影熵率}\label{subsec:pom-projection-entropy}
本节以论文中的“全移位窗口 $\Omega_m$ 到黄金均值 SFT 语言 $X_m$ 的规范化截面”为基底：$X_m$ 是禁词 $(11)$ 的长度 $m$ 语言，$\Fold_m:\Omega_m\to X_m$ 把任意窗口微态投影到该语法块上，$\Phi_m$ 把窗口输出提升为序列级 sofic 因子化；相应的标记图 $G_m$ 由长度 $(m-1)$ 后缀状态与追加一比特构造（定义 \ref{def:G_m}）。在此框架下，纤维多重度 $d_m(x)$ 与条件期望 $E_m$ 是规范投影 $P_Z$ 的两类接口，以下结论给出其尺度结构。
\begin{definition}[折叠指标与投影熵率]\label{def:pom-fold-index}
令
\[
I_m:=\frac{|\Omega_m|}{|X_m|}=\frac{2^m}{F_{m+2}},
\qquad
S_Z(m):=\log I_m,
\qquad
h_Z:=\lim_{m\to\infty}\frac{S_Z(m)}{m}.
\]
$I_m$ 是折叠诱导的平均多重度指标，$S_Z(m)$ 与 $h_Z$ 分别刻画分辨率 $m$ 与单位分辨率的投影熵率。
\end{definition}

\begin{proposition}[折叠指标的渐近]\label{prop:pom-projection-entropy}
有
\[
S_Z(m)=m\log\!\left(\frac{2}{\varphi}\right)+O(1),
\qquad
h_Z=\log\!\left(\frac{2}{\varphi}\right).
\]
\end{proposition}
\begin{proof}
由附录 \ref{app:fold-multiplicity} 的渐近式
\(
I_m\sim C(2/\varphi)^m
\)
直接得到。证毕。
\end{proof}

\begin{corollary}[分辨率不足的空间门槛]\label{cor:pom-resolution-threshold}
若希望“每个稳定类型平均只对应不超过 $R$ 个微态”，则必须满足
\[
m\lesssim \frac{\log R}{\log(2/\varphi)}.
\]
换言之，在稳定词汇 $X_m$ 固定为禁词 $(11)$ 的口径下，平均折叠多重度 $I_m\asymp (2/\varphi)^m$ 随 $m$ 增大而增长；若希望把平均多重度控制在 $R$ 以内，则窗口分辨率 $m$ 不能无限增大。
\end{corollary}

\begin{remark}[折叠熵的二进制斜率]\label{rem:pom-fold-bits}
定义二进制单位的折叠熵
\[
S_Z^{(2)}(m):=\log_2\frac{2^m}{F_{m+2}}.
\]
则
\[
S_Z^{(2)}(m)=m\log_2\!\left(\frac{2}{\varphi}\right)+O(1)
\approx 0.305758\,m+O(1),
\]
给出每增加一位分辨率所“压扁”的平均信息量常数。
\end{remark}

\subsubsection{窗口折叠的最大纤维多重度}\label{subsec:pom-max-fiber}
记
\[
d_m(x):=\left|\Fold_m^{-1}(x)\right|,
\qquad
D_m:=\max_{x\in X_m} d_m(x).
\]

\begin{remark}[平均约束 vs 最坏约束：分辨率门槛的常数比]\label{rem:pom-avg-vs-max-threshold}
固定阈值 $R$。若要求平均约束 $\overline d_m=I_m\le R$，由命题 \ref{prop:pom-projection-entropy} 得
\[
m_{\mathrm{avg}}(R)=\frac{\log R}{\log(2/\varphi)}+O(1).
\]
若要求最坏约束 $D_m\le R$，由定理 \ref{thm:pom-max-fiber} 的 $D_m=\Theta(\varphi^{m/2})$ 得
\[
m_{\max}(R)=\frac{2\log R}{\log\varphi}+O(1).
\]
因此两种门槛在主阶上的比值为常数
\[
\boxed{\;
\frac{m_{\max}(R)}{m_{\mathrm{avg}}(R)}
\approx
\frac{2\log(2/\varphi)}{\log\varphi}
\approx 0.881\; }.
\]
这表明：在同一阈值 $R$ 下，“控制最坏纤维”相对“控制平均纤维”只额外严格一个常数因子（约 $12\%$），与注 \ref{rem:pom-defect-gap} 中“最坏/平均的指数基数极小”一致。
\end{remark}
\subsubsection{信息账本：可见熵 + 纤维体积 - 残差信息}\label{subsec:pom-information-ledger}
固定分辨率 $m$，令 $\Omega_m=\{0,1\}^m$，$X_m$ 为禁词 $(11)$ 的稳定语言块，并设 $\Fold_m:\Omega_m\to X_m$ 为折叠投影。
记概率单纯形为 $\Delta(\Omega_m),\Delta(X_m)$。
定义推前（读出）算子
\[
P_m:=(\Fold_m)_*:\Delta(\Omega_m)\to\Delta(X_m),
\qquad
(P_m\mu)(x)=\sum_{\Fold_m(a)=x}\mu(a),
\]
以及纤维均匀提升（最大对称提升）核
\[
Q_m:\Delta(X_m)\to\Delta(\Omega_m),
\qquad
(Q_m\pi)(a)=\frac{\pi(\Fold_m(a))}{d_m(\Fold_m(a))}.
\]
则 $P_m\circ Q_m=\mathrm{id}_{\Delta(X_m)}$（见附录 \ref{app:op-algebra} 的命题 \ref{prop:fold-pushforward-lift}）。

\begin{theorem}[相对熵缺陷恒等式：信息账本闭式]\label{thm:pom-kl-ledger}
取任意 $\mu\in\Delta(\Omega_m)$，记其稳定推前为 $\pi:=P_m\mu\in\Delta(X_m)$，并令
\[
\mu^{e}:=Q_m\pi\in\Delta(\Omega_m)
\]
为 $\pi$ 的纤维均匀提升（即在每条纤维上条件均匀）。
定义 Shannon 熵与 KL 散度（取自然对数）
\[
H(\nu):=-\sum_{s}\nu(s)\log\nu(s),
\qquad
D_{\mathrm{KL}}(\mu\|\nu):=\sum_{s}\mu(s)\log\frac{\mu(s)}{\nu(s)}.
\]
则有严格恒等式
\[
\boxed{\;
H(\mu)=H(\pi)+\mathbb{E}_{\pi}\!\bigl[\log d_m(X)\bigr]\;-\;D_{\mathrm{KL}}(\mu\|\mu^{e}).
\;}
\]
\end{theorem}
\begin{proof}
由 KL 定义展开得
\(
D_{\mathrm{KL}}(\mu\|\mu^{e})=-H(\mu)-\sum_{a\in\Omega_m}\mu(a)\log\mu^{e}(a)
\)。
而
\(
\log\mu^{e}(a)=\log\pi(x)-\log d_m(x)
\)
其中 $x=\Fold_m(a)$。
因此
\[
\sum_{a}\mu(a)\log\pi(\Fold_m(a))
=\sum_{x\in X_m}\left(\sum_{\Fold_m(a)=x}\mu(a)\right)\log\pi(x)
=\sum_{x}\pi(x)\log\pi(x)
=-H(\pi),
\]
并且
\[
\sum_{a}\mu(a)\log d_m(\Fold_m(a))
=\sum_{x\in X_m}\left(\sum_{\Fold_m(a)=x}\mu(a)\right)\log d_m(x)
=\sum_{x}\pi(x)\log d_m(x)
=\mathbb{E}_{\pi}[\log d_m(X)].
\]
代回整理即得结论。证毕。
\end{proof}

\begin{remark}[把账本/反射子嵌入 prime-like 交换图]\label{rem:pom-ledger-glue-zeta}
信息账本恒等式（定理 \ref{thm:pom-kl-ledger}）不仅把“空间/时间”严格钉成三项分解，
其生成的读出—提升对 \((P_m,Q_m)\) 与纤维反射子 \(K_m\) 也提供了一个\emph{可审计}的算子压缩接口：把微观算子压缩到稳定层可读出算子并保持迹序列不变。
该接口如何与第 \ref{subsec:operator-zeta-interface} 节的“迹$\to$primitive$\to$Euler”严格交换桥包拼接（并进一步接入 Lyapunov‑SDP 谱证书层），见注 \ref{rem:reflected-witt-euler-bridge}。
\end{remark}

\begin{corollary}[“不丢信息”的严格含义：最大熵提升与残差的唯一承载]\label{cor:pom-kl-ledger-bound}
对任意 $\mu\in\Delta(\Omega_m)$ 与 $\pi=P_m\mu$，有上界
\[
H(\mu)\le H(\pi)+\mathbb{E}_{\pi}[\log d_m(X)],
\]
且等号当且仅当 $\mu=\mu^{e}$（即 $\mu$ 在每条折叠纤维上条件均匀）。
\end{corollary}
\begin{proof}
由定理 \ref{thm:pom-kl-ledger} 与 $D_{\mathrm{KL}}\ge 0$ 得上界；且 $D_{\mathrm{KL}}(\mu\|\mu^e)=0$ 当且仅当 $\mu=\mu^e$。证毕。
\end{proof}

\begin{corollary}[最坏熵缺口上界]\label{cor:pom-entropy-gap-bound}
对任意 $\mu\in\Delta(\Omega_m)$，令 $\pi=P_m\mu$。则
\[
H(\mu)-H(\pi)\le \log D_m.
\]
因此最大可见熵缺口满足
\[
\sup_{\mu\in\Delta(\Omega_m)}\bigl(H(\mu)-H(\pi)\bigr)\le \log D_m.
\]
并由定理 \ref{thm:pom-max-fiber} 得
\[
\limsup_{m\to\infty}\frac{1}{m}\sup_{\mu}\bigl(H(\mu)-H(\pi)\bigr)\le \frac12\log\varphi.
\]
\end{corollary}
\begin{proof}
由定理 \ref{thm:pom-kl-ledger}，
\(
H(\mu)-H(\pi)=\mathbb{E}_{\pi}[\log d_m(X)]-D_{\mathrm{KL}}(\mu\|\mu^e)
\)。
注意该差异可为负，但其上界由 $\mathbb{E}_{\pi}[\log d_m(X)]\le \log D_m$ 给出。取上确界即得前两式。再由定理 \ref{thm:pom-max-fiber} 的渐近得到单位分辨率上界。证毕。
\end{proof}

\begin{lemma}[纤维 KL 分解：残差为纤维内非均匀的平均]\label{lem:pom-fiberwise-kl}
取任意 $\mu\in\Delta(\Omega_m)$，令 $\pi=P_m\mu$。对每个 $x\in X_m$ 定义纤维条件分布
\[
\mu_x(a):=
\begin{cases}
\mu(a)/\pi(x),& a\in \Fold_m^{-1}(x),\ \pi(x)>0,\\
0,& a\in \Fold_m^{-1}(x),\ \pi(x)=0,
\end{cases}
\]
并令 $u_x$ 为该纤维上的均匀分布 $u_x(a)=1/d_m(x)$（$a\in\Fold_m^{-1}(x)$）。
则对纤维均匀提升 $\mu^e=Q_m\pi$，有链式恒等式
\[
\boxed{\;
D_{\mathrm{KL}}(\mu\|\mu^e)=\sum_{x\in X_m}\pi(x)\,D_{\mathrm{KL}}(\mu_x\|u_x).
\;}
\]
\end{lemma}
\begin{proof}
当 $\pi(x)=0$ 时，$\mu$ 在纤维 $\Fold_m^{-1}(x)$ 上亦为零测，因此该纤维对两端均无贡献。
对任意满足 $\pi(x)>0$ 的纤维，若 $a\in\Fold_m^{-1}(x)$，则
\(
\mu(a)=\pi(x)\mu_x(a)
\)
且
\(
\mu^e(a)=\pi(x)u_x(a)
\)。
代入 KL 定义并按纤维分组得
\[
D_{\mathrm{KL}}(\mu\|\mu^e)
=\sum_{x}\sum_{a\in\Fold_m^{-1}(x)}\pi(x)\mu_x(a)\log\frac{\mu_x(a)}{u_x(a)}
=\sum_x \pi(x)\,D_{\mathrm{KL}}(\mu_x\|u_x).
\]
证毕。
\end{proof}

\begin{corollary}[纤维熵亏恒等式：\texorpdfstring{$D_{\mathrm{KL}}(\mu\|\mu^e)$}{KL} 的显式形式]\label{cor:pom-fiber-entropy-deficit}
沿用引理 \ref{lem:pom-fiberwise-kl} 的记号。记纤维条件熵
\[
H(\mu_x):=-\sum_{a\in\Fold_m^{-1}(x)}\mu_x(a)\log \mu_x(a).
\]
则对每个满足 $\pi(x)>0$ 的纤维，有
\[
D_{\mathrm{KL}}(\mu_x\|u_x)=\log d_m(x)-H(\mu_x),
\]
从而
\[
\boxed{\;
D_{\mathrm{KL}}(\mu\|\mu^e)
=\sum_{x\in X_m}\pi(x)\bigl(\log d_m(x)-H(\mu_x)\bigr).
\;}
\]
因此 $D_{\mathrm{KL}}(\mu\|\mu^e)$ 正是纤维熵亏 \(\log d_m(x)-H(\mu_x)\ge 0\) 的 \(\pi\)-加权平均；并且 $D_{\mathrm{KL}}(\mu\|\mu^e)=0$ 当且仅当 $\mu$ 在每条折叠纤维上条件均匀。
\end{corollary}

\begin{proof}
对固定纤维 \(x\)（且 \(\pi(x)>0\)），由 \(u_x(a)=1/d_m(x)\) 得
\[
D_{\mathrm{KL}}(\mu_x\|u_x)
=\sum_{a}\mu_x(a)\log\frac{\mu_x(a)}{1/d_m(x)}
=\sum_a \mu_x(a)\log\mu_x(a)+\log d_m(x)
=\log d_m(x)-H(\mu_x).
\]
再代回引理 \ref{lem:pom-fiberwise-kl} 即得结论。证毕。
\end{proof}

\begin{corollary}[熵增恒等式：最大熵提升的增量等于残差 KL]\label{cor:pom-ent-increase-equals-kl}
沿用上述记号，则有严格恒等式
\[
\boxed{\;
H(\mu^e)-H(\mu)=D_{\mathrm{KL}}(\mu\|\mu^e).
\;}
\]
\end{corollary}
\begin{proof}
由定理 \ref{thm:pom-kl-ledger} 得
\(
H(\mu)=H(\pi)+\mathbb{E}_{\pi}[\log d_m(X)]-D_{\mathrm{KL}}(\mu\|\mu^e)
\)。
另一方面，直接对 $\mu^e$ 代入（或等价地用“纤维条件均匀”）可得
\(
H(\mu^e)=H(\pi)+\mathbb{E}_{\pi}[\log d_m(X)]
\)。
两式相减即得结论。证毕。
\end{proof}

\begin{theorem}[纤维反射子：幂等 Markov 投影与固定点刻画]\label{thm:pom-fiber-reflector}
定义 Markov 算子
\[
K_m:=Q_m\circ P_m:\Delta(\Omega_m)\to\Delta(\Omega_m).
\]
则：
\begin{enumerate}
  \item $K_m$ 幂等：$K_m^2=K_m$；
  \item 对任意 $\mu$，若 $\pi=P_m\mu$，则 $K_m\mu=\mu^e$ 为 $\pi$ 的纤维均匀提升；
  \item 固定点满足：$K_m\mu=\mu$ 当且仅当 $\mu$ 在每条纤维上条件均匀（等价地 $\mu=\mu^e$）。
\end{enumerate}
\end{theorem}
\begin{proof}
由 $P_m\circ Q_m=\mathrm{id}$ 得
\(
K_m^2=Q_mP_mQ_mP_m=Q_m(P_mQ_m)P_m=Q_mP_m=K_m
\)。
对任意 $\mu$，令 $\pi=P_m\mu$，则 $K_m\mu=Q_m\pi=\mu^e$ 是定义。
最后，$K_m$ 的像空间恰为 $\{Q_m\pi:\pi\in\Delta(X_m)\}$；因此 $K_m\mu=\mu$ 当且仅当 $\mu$ 属于该像空间，即在每条纤维上条件均匀。证毕。
\end{proof}

\begin{corollary}[熵单调化通道：\texorpdfstring{$Q_m\circ P_m$}{Q∘P} 的“窗口第二定律”]\label{cor:pom-second-law-Km}
对任意 $\mu\in\Delta(\Omega_m)$，有严格恒等式
\[
\boxed{\;
H(K_m\mu)-H(\mu)=D_{\mathrm{KL}}(\mu\|K_m\mu)\ \ge\ 0,
\;}
\]
并且等号当且仅当 $K_m\mu=\mu$（等价于 $\mu$ 在每条纤维上条件均匀）。
\end{corollary}
\begin{proof}
由定理 \ref{thm:pom-fiber-reflector}，对任意 $\mu$ 有 $K_m\mu=\mu^e$。
将此代入推论 \ref{cor:pom-ent-increase-equals-kl} 即得
\(
H(K_m\mu)-H(\mu)=D_{\mathrm{KL}}(\mu\|K_m\mu)
\)；
非负性与取等号条件由 KL 的基本性质与定理 \ref{thm:pom-fiber-reflector} 的固定点刻画给出。证毕。
\end{proof}

\begin{proposition}[\texorpdfstring{$K_m$}{Km} 的核矩阵：对称、双随机与纤维平均]\label{prop:pom-Km-bistochastic}
把 $K_m$ 视为 $\Omega_m$ 上的 Markov 核，其转移核为
\[
K_m(a,b):=
\begin{cases}
\dfrac{1}{d_m(\Fold_m(a))},& \Fold_m(b)=\Fold_m(a),\\[6pt]
0,& \text{否则}.
\end{cases}
\]
则：
\begin{enumerate}
  \item \textbf{（幂等）} $K_m^2=K_m$；
  \item \textbf{（对称）} $K_m(a,b)=K_m(b,a)$；
  \item \textbf{（双随机）} 对任意 $a$ 有 $\sum_b K_m(a,b)=1$，且对任意 $b$ 有 $\sum_a K_m(a,b)=1$；
  \item \textbf{（函数版 = 纤维平均）} 对任意函数 $f:\Omega_m\to\RR$，
  \[
  (K_m f)(a):=\sum_{b\in\Omega_m}K_m(a,b)f(b)
  =\frac{1}{d_m(\Fold_m(a))}\sum_{\Fold_m(b)=\Fold_m(a)} f(b),
  \]
  即 $K_m$ 在观测量层面就是“按折叠纤维做算术平均”的条件期望接口。
\end{enumerate}
特别地，$\Omega_m$ 上的均匀分布 $u_{\Omega_m}(a)\equiv 2^{-m}$ 为不变分布（由双随机性立得）。
\end{proposition}
\begin{proof}
将 $\Omega_m$ 分解为折叠纤维。对每个 $x\in X_m$，$K_m$ 在块 $\Fold_m^{-1}(x)\times\Fold_m^{-1}(x)$ 上是常数矩阵 $J/d_m(x)$（其余块为 $0$）。因此 $(J/d)^2=J/d$、$J$ 对称且行列和一致，给出幂等/对称/双随机；函数版公式是矩阵作用的逐项展开。证毕。
\end{proof}

\begin{corollary}[\texorpdfstring{$K_m$}{Km} 的谱与投影解释]\label{cor:pom-Km-spectrum}
把 $K_m$ 视为作用在 $\RR^{\Omega_m}$（或 $\CC^{\Omega_m}$）上的线性算子，则：
\begin{enumerate}
  \item $K_m$ 是（计数内积下的）正交投影，其像空间恰为纤维常值函数空间
  \[
  \mathrm{Im}(K_m)=\{f:\Omega_m\to\RR:\ f(a)=f(b)\ \text{当且仅当}\ \Fold_m(a)=\Fold_m(b)\}.
  \]
  \item $\mathrm{rank}(K_m)=|X_m|$，并且其谱仅由 $\{0,1\}$ 构成：特征值 $1$ 的重数为 $|X_m|$，特征值 $0$ 的重数为 $|\Omega_m|-|X_m|=2^m-|X_m|$。
\end{enumerate}
\end{corollary}

\begin{proof}
由命题 \ref{prop:pom-Km-bistochastic} 知 $K_m^2=K_m$ 且 $K_m$ 对称，故它是正交投影。
对每个 $x\in X_m$，限制到块 $\Fold_m^{-1}(x)$ 上，$K_m$ 就是 $J/d$（其中 $d=d_m(x)$）。该块的像为常值向量张成的一维子空间，核为“纤维内和为 $0$”的余维 $1$ 子空间。
于是全空间分解为各纤维的直和，$\mathrm{Im}(K_m)$ 正是纤维常值函数；因此秩为纤维数 $|X_m|$。
每个块贡献一个特征值 $1$ 与 $d-1$ 个特征值 $0$，汇总得到谱与重数公式。证毕。
\end{proof}

\begin{proposition}[任意权重下的条件期望：正交投影与精确方差分解]\label{prop:pom-weighted-condexp-variance}
取任意 $\mu\in\Delta(\Omega_m)$，令 $\pi:=P_m\mu\in\Delta(X_m)$，并沿用引理 \ref{lem:pom-fiberwise-kl} 的纤维条件分布 $\{\mu_x\}_{x\in X_m}$。
对任意函数 $f:\Omega_m\to\RR$，定义（依赖 $\mu$ 的）条件期望算子
\[
(E_{m}^{(\mu)}f)(a):=
\begin{cases}
\displaystyle \sum_{b\in \Fold_m^{-1}(x)} f(b)\,\mu_x(b),& x=\Fold_m(a),\ \pi(x)>0,\\[6pt]
0,& x=\Fold_m(a),\ \pi(x)=0.
\end{cases}
\]
并定义加权内积
\[
\langle f,g\rangle_\mu:=\sum_{a\in\Omega_m} f(a)g(a)\,\mu(a).
\]
则：
\begin{enumerate}
  \item $E_m^{(\mu)}f$ 仅依赖于 $x=\Fold_m(a)$，因此属于纤维常值子空间（等价于由 $X_m$ 上的某个函数拉回）；
  \item $E_m^{(\mu)}$ 是 $L^2(\mu)$ 中到纤维常值子空间的正交投影：对任意函数 $f$ 与任意纤维常值函数 $g$，恒有
  \[
  \boxed{\;\langle f-E_m^{(\mu)}f,\ g\rangle_\mu=0.\;}
  \]
  \item 写
  \(
  \mathbb{E}_\mu[f]:=\sum_a f(a)\mu(a)
  \)
  与
  \(
  \mathrm{Var}_\mu(f):=\mathbb{E}_\mu[(f-\mathbb{E}_\mu[f])^2]
  \)。
  则有精确的方差分解（全为有限求和）：
  \[
  \boxed{\;
  \mathrm{Var}_\mu(f)=\mathrm{Var}_\mu(E_m^{(\mu)}f)\;+\;\sum_{x\in X_m}\pi(x)\,\mathrm{Var}_{\mu_x}(f).
  \;}
  \]
\end{enumerate}
特别地，若 $\mu$ 在每条纤维上条件均匀（即 $\mu=\mu^e=K_m\mu$），则 $E_m^{(\mu)}$ 与 $K_m$ 的函数版作用一致：$E_m^{(\mu)}f=K_m f$。
\end{proposition}
\begin{proof}
（1）由定义，若 $\Fold_m(a)=\Fold_m(a')=x$ 且 $\pi(x)>0$，则
\(
E_m^{(\mu)}f(a)=\sum_{b\in\Fold_m^{-1}(x)}f(b)\mu_x(b)=E_m^{(\mu)}f(a')
\)；当 $\pi(x)=0$ 时该纤维在 $\mu$ 下零测，按定义取零不影响任何 $\mu$-积分。

（2）任取纤维常值 $g$。存在 $\bar g:X_m\to\RR$ 使 $g(a)=\bar g(\Fold_m(a))$。
按纤维分组并用 $\mu(a)=\pi(x)\mu_x(a)$（$\pi(x)>0$）得
\[
\langle f-E_m^{(\mu)}f,\ g\rangle_\mu
=\sum_{x\in X_m}\sum_{a\in\Fold_m^{-1}(x)}\pi(x)\mu_x(a)\bigl(f(a)-\mathbb{E}_{\mu_x}[f]\bigr)\bar g(x)=0,
\]
因为对每个固定 $x$，内层和为
\(
\bar g(x)\pi(x)\bigl(\mathbb{E}_{\mu_x}[f]-\mathbb{E}_{\mu_x}[f]\bigr)=0
\)。
故 $E_m^{(\mu)}$ 为正交投影（且显然幂等）。

（3）由（2）的正交性，对 $g:=E_m^{(\mu)}f-\mathbb{E}_\mu[f]$（它是纤维常值）应用勾股分解：
\[
\mathbb{E}_\mu\!\bigl[(f-\mathbb{E}_\mu[f])^2\bigr]
=\mathbb{E}_\mu\!\bigl[(E_m^{(\mu)}f-\mathbb{E}_\mu[f])^2\bigr]+\mathbb{E}_\mu\!\bigl[(f-E_m^{(\mu)}f)^2\bigr].
\]
第一项即 $\mathrm{Var}_\mu(E_m^{(\mu)}f)$。第二项按纤维分组并用 $\mu(a)=\pi(x)\mu_x(a)$ 得
\[
\mathbb{E}_\mu[(f-E_m^{(\mu)}f)^2]
=\sum_{x\in X_m}\pi(x)\sum_{a\in\Fold_m^{-1}(x)}\mu_x(a)\bigl(f(a)-\mathbb{E}_{\mu_x}[f]\bigr)^2
=\sum_{x\in X_m}\pi(x)\,\mathrm{Var}_{\mu_x}(f).
\]
合并即得方差分解。最后一段：若 $\mu$ 在每条纤维上条件均匀，则 $\mu_x=u_x$，从而
\(
E_m^{(\mu)}f(a)=\frac{1}{d_m(x)}\sum_{\Fold_m(b)=x}f(b)=(K_m f)(a)
\)。
证毕。
\end{proof}

\begin{definition}[折叠可见度：方差保留比例指标]\label{def:pom-fold-visibility}
对任意 $\mu\in\Delta(\Omega_m)$ 与任意非零方差观测 $f:\Omega_m\to\RR$（即 $\mathrm{Var}_\mu(f)>0$），定义“折叠可见度”
\[
\mathcal{R}_{m,\mu}(f):=\frac{\mathrm{Var}_\mu(E_m^{(\mu)}f)}{\mathrm{Var}_\mu(f)}\in(0,1].
\]
当 $\mathcal{R}_{m,\mu}(f)\approx 1$ 时，观测 $f$ 的主要起伏已由稳定类型 $x=\Fold_m(a)$ 决定（宏观可见）；当 $\mathcal{R}_{m,\mu}(f)\approx 0$ 时，起伏主要发生在纤维内部（宏观不可见）。
\end{definition}

\begin{corollary}[最大熵提升的 I-投影表述]\label{cor:pom-Iproj-maxent}
记 $u_{\Omega_m}$ 为 $\Omega_m$ 上的均匀分布。对任意固定 $\pi\in\Delta(X_m)$，约束集
\[
\mathcal{C}_\pi:=\{\nu\in\Delta(\Omega_m):P_m\nu=\pi\}
\]
为凸集，且其唯一最大熵元 $\mu^e=Q_m\pi$ 同时满足
\[
\boxed{\;
\mu^e=\arg\min_{\nu\in\mathcal{C}_\pi}D_{\mathrm{KL}}(\nu\|u_{\Omega_m})
\;=\;
\arg\max_{\nu\in\mathcal{C}_\pi}H(\nu).
\;}
\]
\end{corollary}
\begin{proof}
因为 $u_{\Omega_m}(a)\equiv 2^{-m}$，对任意 $\nu$ 有
\(
D_{\mathrm{KL}}(\nu\|u_{\Omega_m})=\sum_a \nu(a)\log\nu(a)+m\log 2=\log(2^m)-H(\nu)
\)，
故在同一约束集 $\mathcal{C}_\pi$ 上，“最小 KL 到均匀”与“最大熵”等价。
而最大熵解的唯一性由定理 \ref{thm:pom-maxent-lift}（或按纤维逐块严格凹性）给出。证毕。
\end{proof}

\begin{theorem}[到均匀微态的 KL 勾股分解（精确三项恒等式）]\label{thm:pom-kl-pythagoras-uniform}
记 $u_{\Omega_m}$ 为 $\Omega_m$ 上的均匀分布，并定义其稳定推前（均匀微态基准）
\[
w_m:=P_m u_{\Omega_m}\in\Delta(X_m),\qquad
w_m(x)=\frac{d_m(x)}{2^m}.
\]
取任意 $\mu\in\Delta(\Omega_m)$，令 $\pi:=P_m\mu\in\Delta(X_m)$，并令 $\mu^e:=Q_m\pi$ 为其纤维均匀提升。
则有严格恒等式
\[
\boxed{\;
D_{\mathrm{KL}}(\mu\|u_{\Omega_m})
=D_{\mathrm{KL}}(\pi\|w_m)\;+\;D_{\mathrm{KL}}(\mu\|\mu^e).
\;}
\]
\end{theorem}
\begin{proof}
按 KL 定义展开，并用 $u_{\Omega_m}(a)\equiv 2^{-m}$ 得
\[
D_{\mathrm{KL}}(\mu\|u_{\Omega_m})
=\sum_{a\in\Omega_m}\mu(a)\log\mu(a)+m\log 2
=m\log 2-H(\mu).
\]
另一方面，由定理 \ref{thm:pom-kl-ledger}（信息账本恒等式）
\(
H(\mu)=H(\pi)+\mathbb{E}_\pi[\log d_m(X)]-D_{\mathrm{KL}}(\mu\|\mu^e)
\)，
代回可得
\[
D_{\mathrm{KL}}(\mu\|u_{\Omega_m})
=m\log 2-H(\pi)-\mathbb{E}_\pi[\log d_m(X)]\;+\;D_{\mathrm{KL}}(\mu\|\mu^e).
\]
而由 $w_m(x)=d_m(x)/2^m$ 直接计算
\[
D_{\mathrm{KL}}(\pi\|w_m)
=\sum_{x\in X_m}\pi(x)\log\frac{\pi(x)}{d_m(x)/2^m}
=m\log 2-H(\pi)-\mathbb{E}_\pi[\log d_m(X)].
\]
合并即得所示恒等式。证毕。
\end{proof}

\begin{corollary}[I-投影的精确误差刻画：KL 距离即为“微观残差”]\label{cor:pom-Iproj-gap-exact}
沿用定理 \ref{thm:pom-kl-pythagoras-uniform} 的记号，固定任意 $\pi\in\Delta(X_m)$ 并取任意 $\mu\in\mathcal{C}_\pi$。
则
\[
\boxed{\;
D_{\mathrm{KL}}(\mu\|u_{\Omega_m})-D_{\mathrm{KL}}(\mu^e\|u_{\Omega_m})
=D_{\mathrm{KL}}(\mu\|\mu^e).
\;}
\]
\end{corollary}
\begin{proof}
对固定 $\pi$，由定理 \ref{thm:pom-kl-pythagoras-uniform} 先对一般 $\mu\in\mathcal{C}_\pi$ 写出分解；
再对 $\mu=\mu^e$ 代入同一分解，因 $P_m\mu^e=\pi$ 得其宏观项相同，二式相减即得结论。证毕。
\end{proof}

\begin{corollary}[固定宏观直方图下的“微观滑移窗”（显式区间律）]\label{cor:pom-micro-slip-window}
在固定分辨率 $m$ 下，给定宏观直方图 $\pi\in\Delta(X_m)$，定义其\textbf{不可见容量}（纤维体积预算）
\[
\mathcal{W}^{\mathrm{inv}}(\pi):=\mathbb{E}_\pi[\log d_m(X)].
\]
则对任意满足 $P_m\mu=\pi$ 的微观分布 $\mu\in\Delta(\Omega_m)$，有显式夹逼
\[
\boxed{\;
D_{\mathrm{KL}}(\pi\|w_m)\ \le\ D_{\mathrm{KL}}(\mu\|u_{\Omega_m})\ \le\ D_{\mathrm{KL}}(\pi\|w_m)+\mathcal{W}^{\mathrm{inv}}(\pi)
\;}
\]
等价地，
\[
0\le D_{\mathrm{KL}}(\mu\|\mu^e)\le \mathcal{W}^{\mathrm{inv}}(\pi).
\]
因此同一个宏观直方图 $\pi$ 下，“离均匀微态基线有多远”并非单值，而是在长度为 $\mathcal{W}^{\mathrm{inv}}(\pi)$ 的可审计区间内滑移；该区间长度完全由宏观 $\pi$ 与纤维多重度 $d_m$ 决定。
\end{corollary}
\begin{proof}
左端下界由定理 \ref{thm:pom-kl-pythagoras-uniform} 与 $D_{\mathrm{KL}}(\mu\|\mu^e)\ge 0$ 立得。
\par
对右端上界，由引理 \ref{lem:pom-fiberwise-kl}，
\(
D_{\mathrm{KL}}(\mu\|\mu^e)=\sum_x \pi(x)\,D_{\mathrm{KL}}(\mu_x\|u_x)
\)。
而对任意有限集合大小为 $d$ 的纤维，分布到均匀分布的 KL 散度满足
\(
D_{\mathrm{KL}}(\nu\|u)\le \log d
\)
（最大值在点质量取到）。
因此
\[
D_{\mathrm{KL}}(\mu\|\mu^e)\le \sum_x \pi(x)\log d_m(x)=\mathbb{E}_\pi[\log d_m(X)]=\mathcal{W}^{\mathrm{inv}}(\pi).
\]
代回定理 \ref{thm:pom-kl-pythagoras-uniform} 即得所示区间律。证毕。
\end{proof}

\begin{theorem}[规范的粗粒化动力学：先提升—再演化—再读出]\label{thm:pom-coarse-grained-kernel}
设 $K:\Delta(\Omega_m)\to\Delta(\Omega_m)$ 为任意 Markov 核（窗口层动力学）。
定义稳定层的粗粒化核
\[
\overline K:=P_m\circ K\circ Q_m:\Delta(X_m)\to\Delta(X_m).
\]
则对任意初始稳定分布 $\pi_0\in\Delta(X_m)$，令 $\mu_0:=Q_m\pi_0$，有
\[
\pi_1=\overline K\,\pi_0=P_m(K\mu_0).
\]
并且 $\overline K$ 是满足该“先提升—再演化—再读出”语义的唯一稳定层 Markov 核。
\end{theorem}
\begin{proof}
第一式为定义展开。若 $\widetilde K$ 亦满足对所有 $\pi_0$ 有
\(
\widetilde K\pi_0=P_m(K(Q_m\pi_0))
\)，
则逐点相等推出 $\widetilde K=\overline K$。证毕。
\end{proof}

\begin{definition}[可并行逆代换位点与干涉图]\label{def:pom-carry-interference-graph}
对 $x=x_1\cdots x_m\in X_m$，定义其\emph{可并行逆代换位点集合}
\[
J(x):=\{i\in\{1,\dots,m-2\}:\ x_ix_{i+1}x_{i+2}=100\}.
\]
定义干涉图 $G(x)$：其顶点集为 $J(x)$，并在 $i\neq j$ 时连边当且仅当区间 $[i,i+2]$ 与 $[j,j+2]$ 相交（等价于 $|i-j|\le 2$）。记 $\mathsf{Ind}(G(x))$ 为 $G(x)$ 的独立集族。
\end{definition}

\begin{lemma}[自由逆代换给出的纤维下界]\label{lem:pom-fiber-lower-bound-independent-sets}
对任意 $x\in X_m$，有
\[
d_m(x)\ \ge\ \#\mathsf{Ind}(G(x)).
\]
\end{lemma}

\begin{proof}
取任意独立集 $S\in\mathsf{Ind}(G(x))$。由于 $S$ 中任意两点对应的长度 $3$ 区间不相交，故可在 $x$ 上对所有 $i\in S$ 同时（或任意顺序）执行逆代换 $100\mapsto 011$，得到一个二元词 $\omega_S\in\Omega_m$。
将该过程反向，即对 $\omega_S$ 在这些位点执行正向进位 $011\mapsto 100$，可恢复 $x$；因此 $\Fold_m(\omega_S)=x$。
又因为 $S$ 可由 $\omega_S$ 中发生改变的那些长度-$3$ 位置唯一恢复（区间互不相交使得这些局部改变互不混淆），映射 $S\mapsto \omega_S$ 为单射。
于是 $\#\Fold_m^{-1}(x)=d_m(x)\ge \#\mathsf{Ind}(G(x))$。证毕。
\end{proof}

\begin{lemma}[独立集计数的线性时间递推（半径 2 冲突）]\label{lem:pom-independence-dp-radius2}
令 $x\in X_m$，并将 $J(x)$ 升序写为
\[
J(x)=\{i_1<i_2<\cdots<i_k\}.
\]
对 $t\in\{1,\dots,k\}$ 定义
\[
p(t):=\max\{s\in\{0,1,\dots,t-1\}:\ i_t-i_s>2\},
\]
其中约定 $i_0:=-\infty$（等价地：$p(t)$ 是满足 $i_t-i_{p(t)}>2$ 的最大下标；若对所有 $s\le t-1$ 都有 $i_t-i_s\le 2$，则 $p(t)=0$）。
定义序列 $a_0:=1$ 且对 $t\ge 1$ 递推
\[
a_t:=a_{t-1}+a_{p(t)}.
\]
则有
\[
\#\mathsf{Ind}(G(x))=a_k.
\]
\end{lemma}

\begin{proof}
按 $\{i_1,\dots,i_t\}$ 的前缀递推计数独立集数。
任取独立集 $S\subseteq\{i_1,\dots,i_t\}$，分两类：
\begin{enumerate}
  \item 若 $i_t\notin S$，则 $S$ 是前缀 $\{i_1,\dots,i_{t-1}\}$ 的独立集，计数为 $a_{t-1}$；
  \item 若 $i_t\in S$，则由 $G(x)$ 的边定义（$|i-j|\le 2$ 冲突），$S$ 不能包含任何 $i_s$ 满足 $i_t-i_s\le 2$。因此 $S\setminus\{i_t\}$ 必为前缀 $\{i_1,\dots,i_{p(t)}\}$ 的独立集，计数为 $a_{p(t)}$。
\end{enumerate}
两类互斥且覆盖全部情形，因此 $a_t=a_{t-1}+a_{p(t)}$。
取 $t=k$ 得 $\#\mathsf{Ind}(G(x))=a_k$。证毕。
\end{proof}

\begin{corollary}[一个可计算的纤维下界]\label{cor:pom-fiber-lower-bound-dp}
对任意 $x\in X_m$，用引理 \ref{lem:pom-independence-dp-radius2} 的递推计算出 $a_k$，则
\[
d_m(x)\ge a_k.
\]
\end{corollary}

\begin{proof}
由引理 \ref{lem:pom-fiber-lower-bound-independent-sets} 与引理 \ref{lem:pom-independence-dp-radius2} 直接得到。
\end{proof}

\begin{lemma}[干涉图的块分解与独立集乘法公式]\label{lem:pom-ind-factorization}
令 $x\in X_m$，并将 $J(x)$ 升序写为 $\{i_1<\cdots<i_k\}$。
把指标集合按“断点”分解为若干连续块：记 $t$ 为断点当且仅当 $i_{t+1}-i_t>2$，并据此把 $J(x)$ 分解为若干不相交子集
\[
J(x)=J_1\sqcup\cdots\sqcup J_r,
\]
其中每个 $J_\ell$ 在升序下满足相邻差 $\le 2$。
则干涉图 $G(x)$ 分裂为互不相连的诱导子图直和
\(
G(x)=G[J_1]\sqcup\cdots\sqcup G[J_r]
\)，
并且独立集计数满足乘法公式
\[
\#\mathsf{Ind}(G(x))=\prod_{\ell=1}^r \#\mathsf{Ind}(G[J_\ell]).
\]
\end{lemma}

\begin{proof}
若 $i\in J_a$、$j\in J_b$ 且 $a\neq b$，则由分块定义有 $|i-j|>2$，故 $G(x)$ 中不存在跨块边，从而图为不相连直和。
对不相连图，独立集在各连通分量上独立选择，计数乘法性成立。
\end{proof}

\begin{corollary}[纤维下界的块级乘法级联]\label{cor:pom-fiber-lower-bound-cascade}
沿用定义 \ref{def:pom-carry-interference-graph}，并把 $J(x)=J_1\sqcup\cdots\sqcup J_r$ 按引理 \ref{lem:pom-ind-factorization} 分解为块。
记每块贡献
\[
I(J_\ell):=\#\mathsf{Ind}(G[J_\ell])\qquad(\ell=1,\dots,r).
\]
则对任意 $x\in X_m$ 都有
\[
\boxed{\;
d_m(x)\ \ge\ \prod_{\ell=1}^r I(J_\ell)\;}
\]
从而其对数纤维体积至少具有可加分解
\[
\boxed{\;
\log d_m(x)\ \ge\ \sum_{\ell=1}^r \log I(J_\ell).\;}
\]
\end{corollary}
\begin{proof}
由引理 \ref{lem:pom-fiber-lower-bound-independent-sets} 得 $d_m(x)\ge \#\mathsf{Ind}(G(x))$，再由引理 \ref{lem:pom-ind-factorization} 的乘法公式得到乘积下界。取对数即得第二式。证毕。
\end{proof}

\begin{lemma}[单块情形的局部递推：\texorpdfstring{$a_t=a_{t-1}+a_{t-2}$}{a=a+a} 与 \texorpdfstring{$a_t=a_{t-1}+a_{t-3}$}{a=a+a} 的分岔]\label{lem:pom-ind-local-recurrence}
沿用引理 \ref{lem:pom-independence-dp-radius2} 的记号。
设 $J(x)$ 自身为单块（即对所有 $t$ 有 $i_{t+1}-i_t\le 2$）。则对 $t\ge 3$：
\begin{enumerate}
  \item 若 $i_t-i_{t-2}>2$，则 $p(t)=t-2$，从而
  \[
  a_t=a_{t-1}+a_{t-2};
  \]
  \item 若 $i_t-i_{t-2}=2$（等价于 $i_{t-1}=i_t-1$ 且 $i_{t-2}=i_t-2$），则 $p(t)=t-3$，从而
  \[
  a_t=a_{t-1}+a_{t-3}.
  \]
\end{enumerate}
\end{lemma}

\begin{proof}
在单块条件下，必有 $i_t-i_{t-1}\le 2$，故 $p(t)\le t-2$。
若 $i_t-i_{t-2}>2$，则 $s=t-2$ 满足定义条件且为最大者，故 $p(t)=t-2$。
若 $i_t-i_{t-2}=2$，则 $s=t-2$ 不满足条件；而由于 $i_{t-2}-i_{t-3}\ge 1$，得 $i_t-i_{t-3}\ge 3$，故 $p(t)=t-3$。
代入引理 \ref{lem:pom-independence-dp-radius2} 的递推即得。
\end{proof}

\begin{proposition}[单块的二字母矩阵语法]\label{prop:pom-ind-matrix-grammar}
设 $J(x)=\{i_1<\cdots<i_k\}$ 为单块，并取引理 \ref{lem:pom-independence-dp-radius2} 的序列 $a_t$。
对 $t\in\{3,\dots,k\}$ 定义“局部字母”
\[
\sigma_t:=
\begin{cases}
2,& i_t-i_{t-2}>2\quad(\text{等价于 }p(t)=t-2),\\
3,& i_t-i_{t-2}=2\quad(\text{等价于 }p(t)=t-3),
\end{cases}
\]
并令状态向量
\(
v_t:=(a_t,a_{t-1},a_{t-2})^\top
\)。
则对每个 $t=3,\dots,k$，递推都等价于两种固定线性更新之一：
\[
v_t=M_{\sigma_t}\,v_{t-1},
\qquad
M_2=
\begin{pmatrix}
1&1&0\\
1&0&0\\
0&1&0
\end{pmatrix},
\qquad
M_3=
\begin{pmatrix}
1&0&1\\
1&0&0\\
0&1&0
\end{pmatrix}.
\]
从而块贡献（单块独立集计数）可写为一个二生成元矩阵词的评价：
\[
I(J(x))=\#\mathsf{Ind}(G[J(x)])=a_k
=e_1^\top\Bigl(\prod_{t=3}^k M_{\sigma_t}\Bigr)v_2,
\]
其中 $e_1=(1,0,0)^\top$，并且在单块情形下初值固定为
\(
v_2=(a_2,a_1,a_0)^\top=(3,2,1)^\top
\)。
\end{proposition}
\begin{proof}
由引理 \ref{lem:pom-ind-local-recurrence}，在单块中对 $t\ge 3$ 只有两种情形：
若 $p(t)=t-2$ 则 $a_t=a_{t-1}+a_{t-2}$，等价于 $v_t=M_2v_{t-1}$；
若 $p(t)=t-3$ 则 $a_t=a_{t-1}+a_{t-3}$，等价于 $v_t=M_3v_{t-1}$。
将 $t=3,\dots,k$ 逐步迭代并取第一坐标即得矩阵词表达式。
最后，单块意味着 $i_2-i_1\le 2$，从而 $p(1)=p(2)=0$，故 $a_0=1,a_1=2,a_2=3$。证毕。
\end{proof}

\begin{definition}[缺陷集]\label{def:pom-ind-defect-set}
在命题 \ref{prop:pom-ind-matrix-grammar} 的单块设定下，定义缺陷位置集合
\[
\mathcal D(J(x)):=\{t\in\{3,\dots,k\}:\ \sigma_t=3\}
=\{t\in\{3,\dots,k\}:\ i_t-i_{t-2}=2\}.
\]
\end{definition}

\begin{remark}[谱层级的缺陷集分类原则]\label{rem:pom-ind-defect-class}
命题 \ref{prop:pom-ind-matrix-grammar} 给出一个可审计的组织方式：
\begin{enumerate}
  \item \emph{块内：}单块贡献 $I(J)$ 由二字母矩阵词（等价地：由缺陷集 $\mathcal D(J)$）决定；缺陷自由的情形 $\mathcal D(J)=\emptyset$ 给出纯 Fibonacci 递推与最大指数增长（$M_2$ 的 Perron 根为 $\varphi$）。
  \item \emph{全局：}由推论 \ref{cor:pom-fiber-lower-bound-cascade}，下界 $\#\mathsf{Ind}(G(x))$ 在块层面乘法级联，因此 $\log d_m(x)$ 的一个可计算下包络天然分解为各块 $\log I(J_\ell)$ 的和。
\end{enumerate}
这把“近极大谱层级为何呈现规律性缺口/分岔”的机制统一为：\emph{谱层级}可视为“块上缺陷集合的组合类”，并由块乘法级联拼装。
\end{remark}

\begin{definition}[Lyapunov 指纹常数（典型独立集下界指数）]\label{def:pom-ind-lyapunov-fingerprint}
令 $\nu$ 为黄金均值移位 $X\subset\{0,1\}^{\mathbb N}$ 上的一个平稳遍历概率分布。
对 $\nu$-典型的无限序列 $x=x_1x_2\cdots$，记其长度-$m$ 前缀为 $x_{1..m}$，并据定义 \ref{def:pom-carry-interference-graph} 构造干涉图 $G(x_{1..m})$。
若极限存在，则定义
\[
\Lambda_{\mathsf{Ind}}(\nu):=\lim_{m\to\infty}\frac{1}{m}\log \#\mathsf{Ind}(G(x_{1..m})).
\]
\end{definition}

\begin{remark}[随机矩阵乘积视角]\label{rem:pom-ind-random-matrix}
在推论 \ref{cor:pom-fiber-lower-bound-cascade} 与命题 \ref{prop:pom-ind-matrix-grammar} 的表达下，$\#\mathsf{Ind}(G(x_{1..m}))$ 可被视为“块乘积 + 块内二字母矩阵词”的复合对象。
因此，在任何给定的平稳遍历来源下，块序列与块内缺陷模式诱导一个平稳的矩阵乘积过程；其顶 Lyapunov 指数（再按“块顶点密度/原始长度密度”重标度）提供了典型下界指数 $\Lambda_{\mathsf{Ind}}$ 的一个可审计解释与数值估计路线。
\end{remark}
\begin{theorem}[最大纤维闭式]\label{thm:pom-max-fiber}
取 Fibonacci 延拓到 $F_0=0,F_1=1$。对 $m\ge 2$，有精确闭式
\[
D_{2k}=F_{k+2},
\qquad
D_{2k+1}=F_{k+2}+F_{k-1}\ (=2F_{k+1}).
\]
因此
\[
D_m\sim C'\,\varphi^{m/2},
\]
即最大纤维增长率为 $\sqrt{\varphi}$。特别地，存在常数 $C_1,C_2>0$ 使
\[
C_1\,\varphi^{m/2}\ \le\ D_m\ \le\ C_2\,\varphi^{m/2}.
\]
\end{theorem}
\begin{proof}
对 $\{0,1\}^m$ 的折叠规范化，唯一会被触发的局部重写是
\(
011\mapsto 100
\)
（Fibonacci 进位）。系统合流意味着每个 $\omega$ 的规范形唯一，因此固定 $x\in X_m$ 时，$\omega\in\Fold_m^{-1}(x)$ 当且仅当 $x$ 能通过若干次逆代换 $100\mapsto 011$ 生成 $\omega$。该逆代换占用长度 $3$ 的局部区间；若两次逆代换的起点相距 $1$ 或 $2$，区间重叠会在某一位产生 $2$，因此可选位点必须满足“不可相邻选择”的约束。令 $k=\lfloor m/2\rfloor$，把可选位点看作一条长度 $k$ 的路径图 $P_k$，记其独立集计数为 $I_k$。则
\[
I_0=1,\qquad I_1=2,\qquad I_k=I_{k-1}+I_{k-2}\ (k\ge 2),
\qquad I_k=F_{k+2}.
\]
于是对任意 $x\in X_m$ 有 $d_m(x)\le I_k$。当 $m=2k+1$ 为奇数时，由引理 \ref{lem:pom-one-bit} 的溢出比特 $b\in\{0,1\}$ 分裂为两支；固定 $b$ 会冻结最顶端一处候选位点，使可选链长度降为 $k-1$，从而
\[
d_m(x)\le 2I_{k-1}=2F_{k+1}.
\]

下面构造达到上界的族。对 $m=2,3$ 取
\(
x^{(2)}=00,\ x^{(3)}=000
\)；其余情形令 $m=4q+r$（$r\in\{0,1,2,3\}$）并定义
\[
x^{(m)}=
\begin{cases}
(0001)^{\,q-1}0000,& r=0,\ q\ge 1,\\[2pt]
(0001)^{\,q-1}00100,& r=1,\ q\ge 1,\\[2pt]
(0001)^{\,q}00,& r=2,\ q\ge 0,\\[2pt]
(0001)^{\,q-1}0000100,& r=3,\ q\ge 1.
\end{cases}
\]
该族满足 $x^{(m)}\in X_m$，且其局部零块足以保证：任取不相邻位点集合 $J\subseteq\{1,\dots,k\}$（奇数时为 $\{1,\dots,k-1\}$），按从高位到低位依次执行逆代换，所需的 $100$ 模式在每一步均可实现，从而得到唯一 $\omega\in\Fold_m^{-1}(x^{(m)})$。因此
\[
d_{2k}(x^{(2k)})=I_k=F_{k+2},
\qquad
d_{2k+1}(x^{(2k+1)})=2I_{k-1}=2F_{k+1}.
\]
上界可达即得 $D_m$ 的闭式；渐近式与常数界由 Fibonacci 渐近 $F_n\sim \varphi^n/\sqrt5$ 直接推出。证毕。
\end{proof}

\begin{corollary}[最大纤维概率端点与可审计极限常数]\label{cor:pom-max-fiber-rate-endpoint}
令 \(\pi_m(x):=d_m(x)/2^m\) 为折叠输出分布，并定义最大纤维概率
\[
p_{\max}(m):=\max_{x\in X_m}\pi_m(x)=\frac{D_m}{2^m}.
\]
则由定理 \ref{thm:pom-max-fiber} 的闭式有显式表达
\[
p_{\max}(2k)=\frac{F_{k+2}}{2^{2k}},
\qquad
p_{\max}(2k+1)=\frac{2F_{k+1}}{2^{2k+1}}=\frac{F_{k+1}}{2^{2k}}.
\]
并且最大纤维率（亦即 min-entropy 端点率）
\[
h_{\max}(m):=-\frac{1}{m}\log p_{\max}(m)
\]
收敛到一个闭式常数：
\[
\boxed{\;
\lim_{m\to\infty}h_{\max}(m)=\log\!\left(\frac{2}{\sqrt{\varphi}}\right)\; }.
\]
该常数同时是高阶 Rényi 指纹归一化端点的可审计锚点（见命题 \ref{prop:pom-rq-qinfty-endpoint} 中 \(\lim_{q\to\infty}h_q^{\mathrm R}=\log(2/\sqrt{\varphi})\)）。
\end{corollary}
\begin{proof}
由上式的奇偶闭式与 Binet 渐近 \(F_n\asymp \varphi^n\)，可得
\[
-\frac{1}{2k}\log p_{\max}(2k)
=-\frac{1}{2k}\Bigl(\log F_{k+2}-2k\log 2\Bigr)
\longrightarrow \log 2-\frac12\log\varphi,
\]
奇数分支同理，从而极限为 \(\log(2/\sqrt{\varphi})\)。证毕。
\end{proof}

\begin{corollary}[二步递推与基值]\label{cor:pom-D-rec}
$D_m$ 满足
\[
D_m=D_{m-2}+D_{m-4}\qquad(m\ge 6),
\]
并具有基值
\[
D_2=2,\quad D_3=2,\quad D_4=3,\quad D_5=4.
\]
\end{corollary}
\begin{proof}
由定理 \ref{thm:pom-max-fiber} 的闭式，$D_{2k}=F_{k+2}$ 与 $D_{2k+1}=2F_{k+1}$，而 Fibonacci 数满足 $F_{k+2}=F_{k+1}+F_k$。分别代入 $m=2k$ 与 $m=2k+1$ 即得递推；基值由直接枚举或闭式代入给出。证毕。
\end{proof}

\begin{remark}[最坏投影缺口的比特量化]\label{rem:pom-D-delay}
把“区分纤维内微态”视为必须携带的额外信息量，则最坏缺口可用
\[
\Delta_m^{\max}:=\left\lceil \log_2 D_m\right\rceil

衡量。由定理 \ref{thm:pom-max-fiber} 得

\Delta_m^{\max}\sim \frac{m}{2}\log_2\varphi

\end{remark}

\end{remark}

\begin{definition}[顶层纤维谱与达到者数]\label{def:pom-top-fiber-spectrum}
固定 \(m\ge 2\)。令
\[
\mathcal{V}_m:=\{d_m(x):x\in X_m\}\subseteq \NN.
\]
按降序列出其中\emph{不同}取值：
\[
D_m^{(1)} > D_m^{(2)} > D_m^{(3)} > \cdots.
\]
记 \(D_m:=D_m^{(1)}=\max_{x\in X_m}d_m(x)\)。并定义最大纤维达到者数
\[
\kappa_m^{\mathrm{max}}
=\#\{x\in X_m:\ d_m(x)=D_m\}.
\]
（记号约定：本文另用 \(\kappa_m(\pi)=\mathbb{E}_\pi[\log d_m(X)]\) 表示“平均对数纤维缺陷”；为避免歧义，此处用上标 \(\mathrm{max}\) 标记“达到者数”。）
在 residue 口径下，若
\(
c_m(r):=\#\{\omega\in\Omega_m:\ N(\omega)\equiv r\ (\mathrm{mod}\ F_{m+2})\}
\),
则 \(D_m^{(j)}\) 也是集合 \(\{c_m(r):r\in\ZZ/F_{m+2}\ZZ\}\) 的第 \(j\) 大不同取值，并且
\[
\kappa_m^{\mathrm{max}}=\#\{r\in\ZZ/F_{m+2}\ZZ:\ c_m(r)=D_m\}.
\]
\end{definition}

\begin{remark}[最大纤维达到者的相位结构（可复现计算 + 闭式定律）]\label{rem:pom-max-fiber-achievers-phase}
表 \ref{tab:fold_max_fiber_achievers_phase} 用模 \(F_{m+2}\) 的 residue DP 直接计算了
\(
c_m(r)=\#\{\omega\in\Omega_m:\ N(\omega)\equiv r\ (\mathrm{mod}\ F_{m+2})\}
\)
并取最大值 \(D_m=\max_r c_m(r)\)，同时统计达到最大值的同余类数
\[
\kappa_m^{\mathrm{max}}:=\#\{r\in\ZZ/F_{m+2}\ZZ:\ c_m(r)=D_m\}
=\#\{x\in X_m:\ d_m(x)=D_m\}.
\]
在可审计窗口 \(m\le 32\) 内，\(\kappa_m^{\mathrm{max}}\) 呈现出稳定的“有限相位”行为（见表 \ref{tab:fold_max_fiber_achievers_phase}）：偶数 \(m\) 两相位达到、奇数 \(m\) 四相位达到；代表稳定词表现出近周期核 \(\texttt{0001}\) 的相位截断。
同一表还给出次大多重度 \(D_m^{(2)}\) 与第三大多重度 \(D_m^{(3)}\)（定义 \ref{def:pom-top-fiber-spectrum}），以及比值 \(D_m^{(2)}/D_m\) 的奇偶交错。下述定理 \ref{thm:pom-max-achievers-phase-stabilization}、定理 \ref{thm:pom-second-max-fiber-closed-form} 与定理 \ref{thm:pom-third-max-fiber-even-closed-form} 给出该相位数与顶层谱缺口的解析封口（并给出显式最坏常数 \(\theta_{\mathrm{gap}}\)，见推论 \ref{cor:pom-second-max-gap-constant}）。
\end{remark}

\begin{theorem}[最大纤维达到者的相位数稳定化]\label{thm:pom-max-achievers-phase-stabilization}
对所有充分大的 \(m\)，最大纤维达到者数 \(\kappa_m^{\mathrm{max}}\) 稳定为常数相位数：
\[
\kappa_{2k}^{\mathrm{max}}=2\quad(k\ge 5),
\qquad
\kappa_{2k+1}^{\mathrm{max}}=4\quad(k\ge 6).
\]
等价地（阈值版）：
\[
m\ge 10\ \text{且}\ m\ \text{偶}\ \Rightarrow\ \kappa_m^{\mathrm{max}}=2,
\qquad
m\ge 13\ \text{且}\ m\ \text{奇}\ \Rightarrow\ \kappa_m^{\mathrm{max}}=4.
\]
\end{theorem}
\begin{proof}
由定理 \ref{thm:pom-max-fiber} 的证明可知：达到最大纤维 \(D_m\) 的必要条件是“逆代换可选位点链被做满”，即在长度 \(k=\lfloor m/2\rfloor\) 的候选链上，每个独立集都必须可实现且给出互异的前像；这强制正规形 \(x\in X_m\) 具有一个近周期核（见定理 \ref{thm:pom-max-fiber} 中构造的 \((0001)\) 截断族），并且端点约束只允许有限多个相位。
偶数端点约束恰保留两种相位；奇数情形还叠加引理 \ref{lem:pom-one-bit} 的隐藏位分裂 \(b\in\{0,1\}\)，从而把相位数翻倍为四。上述相位数与达到者的代表词在表 \ref{tab:fold_max_fiber_achievers_phase}（\(m\le 32\) 的可审计窗口）中可直接核验。
\end{proof}

\begin{lemma}[禁配对导致的 Fibonacci 缺口]\label{lem:pom-forbidden-pair-fib-gap}
令 \(P_k\) 为顶点 \(1,\dots,k\) 的路径图（边为 \(i\!\sim\! i{+}1\)），记其独立集数为 \(I_k\)（于是 \(I_k=F_{k+2}\)）。
\begin{enumerate}
  \item 若 \(k\ge 5\)，则在 \(P_k\) 上\emph{同时包含}顶点 \(2\) 与 \(5\) 的独立集数恰为 \(F_{k-4}\)（当 \(k\ge 6\) 时亦可写为 \(I_{k-6}\)）。
  \item 若 \(k\ge 5\)，则在 \(P_k\) 上\emph{同时包含}顶点 \(1\) 与 \(4\) 的独立集数恰为 \(I_{k-5}=F_{k-3}\)。
\end{enumerate}
\end{lemma}
\begin{proof}
(1) 当 \(k=5\) 时，包含 \(\{2,5\}\) 的独立集唯一为 \(\{2,5\}\)，计数为 \(1=F_1\)。
当 \(k\ge 6\) 时，若独立集包含 \(\{2,5\}\)，则顶点 \(1,3,4,6\) 均被排除，剩余可自由选择的部分为路径 \(7,\dots,k\)，同构于 \(P_{k-6}\)。反之，对 \(P_{k-6}\) 的任意独立集加上 \(\{2,5\}\) 即得 \(P_k\) 上包含 \(\{2,5\}\) 的独立集。故计数为 \(I_{k-6}=F_{k-4}\)。\;
(2) 同理：包含 \(\{1,4\}\) 时顶点 \(2,3,5\) 被排除，剩余为路径 \(6,\dots,k\)，同构于 \(P_{k-5}\)。证毕。
\end{proof}

\begin{theorem}[次大纤维闭式与显式间隙常数]\label{thm:pom-second-max-fiber-closed-form}
取 Fibonacci 延拓 \(F_0=0,F_1=1\)，并令 \(D_m^{(2)}\) 为第二大纤维多重度（定义 \ref{def:pom-top-fiber-spectrum}）。
则当 \(k\ge 5\) 时有闭式：
\[
D_{2k}^{(2)} = F_{k+2}-F_{k-4},
\qquad
D_{2k+1}^{(2)} = 2F_{k+1}-F_{k-4}.
\]
\end{theorem}
\begin{proof}
证明思路与定理 \ref{thm:pom-max-fiber} 完全同构：最大值来自“满链独立集计数” \(I_k=F_{k+2}\)（偶数）或 \(2I_{k-1}=2F_{k+1}\)（奇数）。
\par
更机制化地说：在干涉图独立集 DP（引理 \ref{lem:pom-independence-dp-radius2}）中，满链相位对应于所有局部步都落在 \(p(t)=t-2\) 的分支，从而由引理 \ref{lem:pom-ind-local-recurrence}(1) 得到纯 Fibonacci 递推 \(a_t=a_{t-1}+a_{t-2}\)，并在末端给出 \(a_k=I_k=F_{k+2}\)。
当离开最大达到者的满链相位核时，最小可能的退化不是简单地缩短链（那会造成 \(\Theta(F_{k-1})\) 级别的下降），而是出现一次“半径 2 冲突”导致 \(p(t)\) 发生一次跳跃（引理 \ref{lem:pom-ind-local-recurrence}(2)：某一步由 \(a_t=a_{t-1}+a_{t-2}\) 变为 \(a_t=a_{t-1}+a_{t-3}\)）。
在路径图 \(P_k\) 的口径下，这等价于在满链独立集选择中额外禁止某一对距离 \(3\) 的顶点同时出现；最早的非端点缺陷对应的正是禁配对 \(\{2,5\}\)。
由引理 \ref{lem:pom-forbidden-pair-fib-gap}(1)，禁止配对 \(\{2,5\}\) 从满链计数中精确削去 \(F_{k-4}\)，于是得到
\(
D_{2k}^{(2)}=F_{k+2}-F_{k-4}
\)。
奇数情形同理：隐藏位分裂把总计数分解为两层之和，其中最小缺陷只在一层触发一次禁配对，从而总计数为
\(
D_{2k+1}^{(2)}=2F_{k+1}-F_{k-4}
\)。
细节可由与定理 \ref{thm:pom-max-fiber} 相同的“逆代换—冲突约束—Fibonacci 递推”链条逐步写出。
\end{proof}

\begin{corollary}[黄金常数极限、指数收敛与可审计 gap 证书]\label{cor:pom-second-max-gap-constant}
设 \(\varphi=\frac{1+\sqrt5}{2}\)。当 \(k\ge 5\) 时，由定理 \ref{thm:pom-second-max-fiber-closed-form} 与定理 \ref{thm:pom-max-fiber} 的闭式可得精确比值
\[
\frac{D_{2k}^{(2)}}{D_{2k}}=1-\frac{F_{k-4}}{F_{k+2}},
\qquad
\frac{D_{2k+1}^{(2)}}{D_{2k+1}}=1-\frac{F_{k-4}}{2F_{k+1}}.
\]
因此两条奇偶支分别收敛到闭式黄金常数平台：
\[
\frac{D_{2k}^{(2)}}{D_{2k}}
\longrightarrow
1-\varphi^{-6}
=1-\frac{1}{9+4\sqrt5},
\qquad
\frac{D_{2k+1}^{(2)}}{D_{2k+1}}
\longrightarrow
1-\frac12\varphi^{-5}
=1-\frac{1}{11+5\sqrt5}.
\]
并且收敛误差为指数级：
\[
\left|\frac{D_{2k}^{(2)}}{D_{2k}}-\left(1-\varphi^{-6}\right)\right|=O(\varphi^{-2k}),
\qquad
\left|\frac{D_{2k+1}^{(2)}}{D_{2k+1}}-\left(1-\tfrac12\varphi^{-5}\right)\right|=O(\varphi^{-2k}).
\]
记奇数支（也是渐近最坏情形）的极限常数为
\[
\boxed{\;
\theta_\star\;(:=\theta_{\mathrm{gap}})\;:=1-\frac{1}{11+5\sqrt5}
=1-\tfrac12\varphi^{-5}
\approx 0.9549150281<1.
\;}
\]
则“存在 \(\theta<1\) 使 \(D_m^{(2)}\le \theta D_m\)”可取显式一刀切证书
\[
\boxed{\;
\frac{D_m^{(2)}}{D_m}\le \frac{25}{26}\qquad(m\ge 2),
\;}
\]
且该上界在 \(m=13\) 取等（从而为全局最优常数）。作为对照，渐近尺度上的最优候选常数为 \(\theta_\star\)，并且平台贴合误差满足 \(O(\varphi^{-2k})\)，解释了表 \ref{tab:fold_max_fiber_achievers_phase} 中 \(m\le 32\) 的“近常数”现象。
\end{corollary}
\begin{proof}
比值公式由闭式直接相除得到。
极限常数由 Binet 公式 \(F_n=\frac{1}{\sqrt5}(\varphi^n-\psi^n)\)（\(\psi=-\varphi^{-1}\)）给出：例如
\[
\frac{F_{k-4}}{F_{k+2}}
=\varphi^{-6}\frac{1-(\psi/\varphi)^{k-4}}{1-(\psi/\varphi)^{k+2}}
=\varphi^{-6}\left(1+O(\varphi^{-2k})\right),
\]
同理 \(\frac{F_{k-4}}{F_{k+1}}=\varphi^{-5}\left(1+O(\varphi^{-2k})\right)\)，代入即得两条指数级误差。
闭式形式 \(\varphi^6=9+4\sqrt5\)、\(2\varphi^5=11+5\sqrt5\) 可由恒等式 \(\varphi^n=F_n\varphi+F_{n-1}\)（取 \(n=5,6\)）得到。

最后给出统一证书：由上式，
\[
\frac{D_{2k+1}^{(2)}}{D_{2k+1}}=1-\frac{F_{k-4}}{2F_{k+1}}
\le 1-\frac{1}{26}=\frac{25}{26},
\]
其中不等号等价于 \(F_{k+1}\le 13F_{k-4}\)。
写 \(n:=k-4\ge 1\)，即需证对所有 \(n\ge 1\) 有 \(F_{n+5}\le 13F_n\)。
该不等式在 \(n=1,2\) 成立（\(F_6=8\le 13F_1\)、\(F_7=13=13F_2\)），且若对 \(n,n+1\) 成立，则
\(
F_{n+7}=F_{n+6}+F_{n+5}\le 13F_{n+1}+13F_n=13F_{n+2}
\)，
从而由归纳对一切 \(n\) 成立；等号仅在 \(n=2\)（即 \(k=6,m=13\)）发生。
偶数支更强：因 \(F_{k+2}=F_{k+1}+F_k<2F_{k+1}\)，故当 \(k\ge 5\) 时
\[
\frac{D_{2k}^{(2)}}{D_{2k}}
=1-\frac{F_{k-4}}{F_{k+2}}
\;<\;
1-\frac{F_{k-4}}{2F_{k+1}}
=\frac{D_{2k+1}^{(2)}}{D_{2k+1}}
\le \frac{25}{26}.
\]
对 \(m<10\) 的有限个初值情形可由表 \ref{tab:fold_max_fiber_achievers_phase} 直接审计。证毕。
\end{proof}

\begin{theorem}[偶数支第三大纤维闭式]\label{thm:pom-third-max-fiber-even-closed-form}
当 \(k\ge 6\) 时，偶数分支的第三大纤维多重度满足
\[
D_{2k}^{(3)}=F_{k+2}-F_{k-3}.
\]
\end{theorem}
\begin{proof}
与定理 \ref{thm:pom-second-max-fiber-closed-form} 相同，把“最小缺陷”替换为第二小的端点禁配对即可。由引理 \ref{lem:pom-forbidden-pair-fib-gap}(2)，禁止配对 \(\{1,4\}\) 会削去 \(I_{k-5}=F_{k-3}\)，从而从 \(D_{2k}=I_k=F_{k+2}\) 得到 \(D_{2k}^{(3)}=F_{k+2}-F_{k-3}\)。证毕。
\end{proof}

\begin{remark}[奇数支第三大纤维的稳定闭式（可审计）]\label{rem:pom-third-max-fiber-odd-closed-form}
表 \ref{tab:fold_max_fiber_achievers_phase} 的可审计窗口 \(m\le 32\) 已给出第三大纤维多重度 \(D_m^{(3)}\) 的 DP 输出。
从该表可直接读出：奇数分支从 \(m=19\) 起进入稳定相。
写 \(m=2k+1\)，则对所有 \(k\ge 9\)（即 \(m\ge 19\)）逐项满足精确闭式
\[
\boxed{\;
D_{2k+1}^{(3)}=2F_{k+1}-\bigl(F_{k-4}+F_{k-8}\bigr).\;}
\]
该闭式与偶数支定理 \ref{thm:pom-third-max-fiber-even-closed-form} 一起，把顶端三层谱缺口锁定为显式 Fibonacci 尾项，从而可在不引入任何新假设的前提下继续推进“谱缺口 \(\Rightarrow\) 高阶矩选择”类的定量推断（见注 \ref{rem:pom-third-max-gap-parity}）。
\end{remark}

\begin{remark}[顶层纤维谱的 Fibonacci 梯级缺口]\label{rem:pom-top-fiber-spectrum-ladder}
定理 \ref{thm:pom-second-max-fiber-closed-form} 与定理 \ref{thm:pom-third-max-fiber-even-closed-form} 表明顶层纤维谱并非“连续散布”，而是在顶端出现刚性的 Fibonacci 梯级缺口：
\[
D_{2k}-D_{2k}^{(2)}=F_{k-4},\qquad
D_{2k}-D_{2k}^{(3)}=F_{k-3}\qquad(k\ge 6),
\]
以及
\[
D_{2k+1}-D_{2k+1}^{(2)}=F_{k-4}\qquad(k\ge 5).
\]
并且在奇数支第三层稳定相（注 \ref{rem:pom-third-max-fiber-odd-closed-form}）上还有
\[
D_{2k+1}-D_{2k+1}^{(3)}=F_{k-4}+F_{k-8}\qquad(k\ge 9).
\]
等价地，对所有 \(m\ge 10\)（即 \(\lfloor m/2\rfloor\ge 5\)）有统一式
\[
\boxed{\;
D_m^{(2)}=D_m-F_{\lfloor m/2\rfloor-4}\; }.
\]
换言之，离开最大纤维态的最小“空间损失”被 Fibonacci 递推锁定为离散能级。
\end{remark}

\begin{remark}[第三层谱缺口平台与 \(q\)-阈值（奇偶分离）]\label{rem:pom-third-max-gap-parity}
定义第三层抑制因子
\[
\gamma_m^{(3)}:=\frac{D_m^{(3)}}{D_m}\in(0,1).
\]
偶数支由定理 \ref{thm:pom-max-fiber} 与定理 \ref{thm:pom-third-max-fiber-even-closed-form} 给出精确比值
\[
\frac{D_{2k}^{(3)}}{D_{2k}}
=1-\frac{F_{k-3}}{F_{k+2}}
\longrightarrow
1-\varphi^{-5}
\approx 0.9098300563.
\]
奇数支在第三层稳定闭式（注 \ref{rem:pom-third-max-fiber-odd-closed-form}）下，亦有精确比值
\[
\frac{D_{2k+1}^{(3)}}{D_{2k+1}}
=1-\frac{F_{k-4}+F_{k-8}}{2F_{k+1}}
\longrightarrow
1-\tfrac12(\varphi^{-5}+\varphi^{-9})
\approx 0.9483372194.
\]
因此第三层在奇数窗上渐近地\textbf{更贴近最大层}，奇偶分离在顶端三层谱中首次显性出现。
\par
进一步，若用
\[
\left(\gamma_m^{(3)}\right)^q\approx e^{-1}
\qquad\Longleftrightarrow\qquad
q\approx \frac{1}{-\log \gamma_m^{(3)}}
\]
作为“第三层在 \(q\)-阶矩里变得可忽略”的粗阈值，则由上述两条平台常数可得到两种特征尺度：
\[
q_{\mathrm{even}}\approx \frac{1}{-\log(1-\varphi^{-5})}\approx 10.58,
\qquad
q_{\mathrm{odd}}\approx \frac{1}{-\log\bigl(1-\tfrac12(\varphi^{-5}+\varphi^{-9})\bigr)}\approx 18.85.
\]
这预测了一个新的、无需额外假设的效应：随 \(q\) 增大，“第三层变得不可见”的发生会在偶数窗上显著更早，从而给高阶选择/坍缩现象提供一个可直接对照的奇偶指纹。
\par
机制上，该分离可在同一干涉图独立集 DP 框架内解释：最大层对应“缺陷自由”的步长 \(2\) 链（引理 \ref{lem:pom-independence-dp-radius2} 中 \(p(t)=t-2\) 全程成立）因而给出 Fibonacci；第二/第三层对应于最早可触发的 \(p(t)\) 跳跃（禁配对）所剥离的 Fibonacci 尾项（见引理 \ref{lem:pom-forbidden-pair-fib-gap}）。
奇数窗叠加引理 \ref{lem:pom-one-bit} 的隐藏位端点约束，使第三层缺陷以“配对缺陷”形式在两端耦合地传播，从而出现两段尾损失 \(F_{k-4}+F_{k-8}\) 的叠加。
\end{remark}

\begin{remark}[常数级的相位/缺陷资源标尺]\label{rem:pom-phase-defect-resource-meters}
由定理 \ref{thm:pom-max-achievers-phase-stabilization}，最大态不是单点而是有限相位轨道，故可定义相位寄存器开销
\[
R_{\mathrm{phase}}(m):=\left\lceil \log_2 \kappa_m^{\mathrm{max}}\right\rceil
=
\begin{cases}
1,& m\ \text{为偶数且充分大},\\
2,& m\ \text{为奇数且充分大}.
\end{cases}
\]
进一步定义“缺陷分辨开销”
\[
R_{\mathrm{defect}}(m):=-\log_2\!\left(1-\frac{D_m^{(2)}}{D_m}\right).
\]
由推论 \ref{cor:pom-second-max-gap-constant}，当 \(m\to\infty\) 时 \(R_{\mathrm{defect}}(m)\) 收敛到常数：偶数支极限为 \(-\log_2(\varphi^{-6})\)，奇数支极限为 \(-\log_2\!\bigl(\tfrac12\varphi^{-5}\bigr)\)（后者给出最坏情形）。
\end{remark}

\begin{remark}[最大纤维达到者的隐藏位分裂：\texorpdfstring{$(d_{m,0},d_{m,1})$}{(d0,d1)} 的 Fibonacci 刚性（可复现观察）]\label{rem:pom-max-fiber-achievers-bsplit}
由引理 \ref{lem:pom-one-bit}（单隐藏位分解），对每个 \(x\in X_m\) 定义
\[
d_{m,b}(x):=\#\{\omega\in\Omega_m:\ \Fold_m(\omega)=x,\ b(\omega)=b\}\qquad(b\in\{0,1\}),
\]
则 \(d_m(x)=d_{m,0}(x)+d_{m,1}(x)\)。
由于最大窗口值满足 \(\max_{\omega}N(\omega)=F_{m+3}-2<2F_{m+2}\)（\(m\ge 2\)），隐藏位 \(b(\omega)\) 等价于 \(N(\omega)\) 除以 \(F_{m+2}\) 的商，因此对 \(r=V_m(x)\in[0,F_{m+2})\) 有
\[
d_{m,0}(x)=\#\{\omega:\ N(\omega)=r\},\qquad
d_{m,1}(x)=\#\{\omega:\ N(\omega)=r+F_{m+2}\}.
\]
表 \ref{tab:fold_max_fiber_achievers_bsplit} 对 \(m\le 32\) 的可审计窗口直接计算了所有达到者 \(x\) 的分裂对 \((d_{m,0}(x),d_{m,1}(x))\)。
其输出显示出一组与定理 \ref{thm:pom-max-fiber} 相容、但更细的 Fibonacci 刚性：
\begin{itemize}
  \item \textbf{偶数 \(m=2k\)：}当 \(m\ge 10\) 时，达到者恰有两相位，且分裂对为
  \[
  (d_{2k,0},d_{2k,1})\in\{(F_{k+1},F_k),(F_k,F_{k+1})\},\qquad d_{2k,0}+d_{2k,1}=F_{k+2}=D_{2k}.
  \]
  \item \textbf{奇数 \(m=2k+1\)：}当 \(m\ge 13\) 时，达到者恰有四相位，其中两者为平衡分裂、两者为偏置分裂（互为交换）：
  \[
  (d_{2k+1,0},d_{2k+1,1})=(F_{k+1},F_{k+1})\ \text{出现两次},
  \]
  \[
  (d_{2k+1,0},d_{2k+1,1})\in\{(F_{k+1}-F_{k-2},F_{k+1}+F_{k-2}),(F_{k+1}+F_{k-2},F_{k+1}-F_{k-2})\},
  \]
  且总和恒为 \(2F_{k+1}=D_{2k+1}\)。
\end{itemize}
该分裂结构把“最大纤维计数 = 链上独立集计数”（注 \ref{rem:pom-overflow-rec}）进一步细分为“端点约束下的两类独立集计数”，从而在达到者层面强制出现相邻 Fibonacci 数的交换分裂。
\end{remark}

\begin{corollary}[最大纤维达到者的隐藏位概率谱：三种普适极限]\label{cor:pom-max-fiber-achievers-hidden-bit-prob-spectrum}
对最大纤维达到者 \(x\in X_m\)（即 \(d_m(x)=D_m\)），定义隐藏位 \(1\) 的相对频率
\[
p_1(x):=\frac{d_{m,1}(x)}{D_m}.
\]
则在稳定阈值区间（注 \ref{rem:pom-max-fiber-achievers-bsplit}）：
\begin{itemize}
  \item \textbf{偶数 \(m=2k\ge 10\)：}
  \[
  p_1(x)\in\left\{\frac{F_k}{F_{k+2}},\ \frac{F_{k+1}}{F_{k+2}}\right\}
  \xrightarrow[k\to\infty]{}\left\{\varphi^{-2},\ \varphi^{-1}\right\}.
  \]
  \item \textbf{奇数 \(m=2k+1\ge 13\)：}
  \[
  p_1(x)\in\left\{\frac12,\ \frac12\pm\frac{F_{k-2}}{2F_{k+1}}\right\}
  \xrightarrow[k\to\infty]{}\left\{\frac12,\ \frac12\pm\frac{\varphi^{-3}}{2}\right\}
  =
  \left\{\frac12,\ \varphi^{-2},\ \varphi^{-1}\right\}.
  \]
\end{itemize}
因此在“极限意义”下，最大纤维达到者的隐藏位概率谱只出现三种普适常数：
\[
\boxed{\;
p_1\in\left\{\frac12,\ \varphi^{-1},\ \varphi^{-2}\right\}\quad(\text{聚点集合}).
\;}
\]
\end{corollary}
\begin{proof}
由注 \ref{rem:pom-max-fiber-achievers-bsplit} 的闭式分裂对直接相除即可得到上述有限 \(k\) 的取值集合。
用 Binet 渐近 \(F_{n+a}/F_{n+b}\to \varphi^{a-b}\) 得到
\(
F_k/F_{k+2}\to\varphi^{-2}
\),
\(
F_{k+1}/F_{k+2}\to\varphi^{-1}
\),
以及
\(
F_{k-2}/F_{k+1}\to\varphi^{-3}
\)；
代回即可得奇数支的极限等式
\(
\tfrac12\pm\tfrac12\varphi^{-3}=\{\varphi^{-1},\varphi^{-2}\}
\)。
证毕。
\end{proof}

\begin{corollary}[最大达到者的隐藏位相对不平衡度：奇偶一致的普适极限常数]\label{cor:pom-max-fiber-achievers-hidden-bit-imbalance}
定义最大达到者的隐藏位相对不平衡度
\[
\Delta_m^{(b)}:=\max_{x:\ d_m(x)=D_m}\frac{|d_{m,0}(x)-d_{m,1}(x)|}{D_m}.
\]
则在稳定阈值区间（注 \ref{rem:pom-max-fiber-achievers-bsplit}）有闭式
\[
\Delta_{2k}^{(b)}=\frac{F_{k-1}}{F_{k+2}},
\qquad
\Delta_{2k+1}^{(b)}=\frac{F_{k-2}}{F_{k+1}},
\]
从而得到与奇偶无关的普适极限：
\[
\boxed{\;
\lim_{m\to\infty}\Delta_m^{(b)}=\varphi^{-3}=\sqrt5-2\approx 0.236068.
\;}
\]
\end{corollary}
\begin{proof}
偶数 \(m=2k\)：注 \ref{rem:pom-max-fiber-achievers-bsplit} 给出分裂对为 \((F_{k+1},F_k)\) 与 \((F_k,F_{k+1})\)，故最大差为 \(|F_{k+1}-F_k|=F_{k-1}\)，再除以 \(D_{2k}=F_{k+2}\) 得 \(\Delta_{2k}^{(b)}=F_{k-1}/F_{k+2}\)。
\par
奇数 \(m=2k+1\)：偏置分裂对相差 \(2F_{k-2}\)，而 \(D_{2k+1}=2F_{k+1}\)，故 \(\Delta_{2k+1}^{(b)}=F_{k-2}/F_{k+1}\)。
两式用 Binet 渐近即得极限为 \(\varphi^{-3}\)。
恒等式 \(\varphi^{-3}=\sqrt5-2\) 由 \(\varphi=\frac{1+\sqrt5}{2}\) 直接化简得到。证毕。
\end{proof}

\begin{corollary}[最大达到者的隐藏位熵：两种极限常数（偶/奇相结构不同）]\label{cor:pom-max-fiber-achievers-hidden-bit-entropy}
对最大纤维达到者 \(x\) 定义隐藏位二元熵（bits）
\[
H_b(x):=-p_1(x)\log_2 p_1(x)-(1-p_1(x))\log_2(1-p_1(x)).
\]
令
\[
H_\varphi:=H_b(\varphi^{-1})=H_b(\varphi^{-2})\approx 0.9594187\ \text{bits}.
\]
则在稳定阈值区间内：
\begin{itemize}
  \item \textbf{偶数 \(m\ge 10\)：}两相位最大达到者均为黄金偏置相，且 \(H_b(x)\to H_\varphi\)（不出现 \(1\) bit 的平衡相）；
  \item \textbf{奇数 \(m\ge 13\)：}四相位中恰有两相位为平衡分裂（\(p_1=1/2\)，故 \(H_b\to 1\) bit），另两相位为黄金偏置分裂（\(H_b\to H_\varphi\)）。
\end{itemize}
\end{corollary}
\begin{proof}
由推论 \ref{cor:pom-max-fiber-achievers-hidden-bit-prob-spectrum}，达到者的 \(p_1\) 只收敛到 \(\{1/2,\varphi^{-1},\varphi^{-2}\}\)。
二元熵满足 \(H_b(p)=H_b(1-p)\)，故偏置极限 \(\varphi^{-1}\) 与 \(\varphi^{-2}\) 给出同一常数 \(H_\varphi\)，而 \(p=1/2\) 给出 \(H_b=1\)。
奇偶相位结构来自注 \ref{rem:pom-max-fiber-achievers-bsplit} 的分裂类型计数。证毕。
\end{proof}

\begin{corollary}[相位配对导致的隐藏位精确无偏（最大纤维族；偶数支）]\label{cor:pom-max-fiber-hidden-bit-unbiased}
设 \(m=2k\ge 10\)。由定理 \ref{thm:pom-max-achievers-phase-stabilization}，最大纤维达到者恰有两相位，记为 \(x^{(1)},x^{(2)}\in X_{2k}\)。
则由注 \ref{rem:pom-max-fiber-achievers-bsplit} 的闭式分裂对，
\[
\bigl(d_{2k,0}(x^{(1)}),d_{2k,1}(x^{(1)})\bigr)=(F_{k+1},F_k),
\qquad
\bigl(d_{2k,0}(x^{(2)}),d_{2k,1}(x^{(2)})\bigr)=(F_k,F_{k+1}),
\]
从而相位配对在最大纤维族上把隐藏位偏置\emph{严格抵消}：
\[
d_{2k,0}(x^{(1)})+d_{2k,0}(x^{(2)})=F_{k+1}+F_k=F_{k+2}=D_{2k},
\]
\[
d_{2k,1}(x^{(1)})+d_{2k,1}(x^{(2)})=F_k+F_{k+1}=D_{2k}.
\]
等价地，若在两条最大相位之间取均匀平均，则隐藏位在最大纤维族上精确无偏：
\[
\mathbb{E}\!\left[\frac{d_{2k,0}(X)}{D_{2k}}\right]
=\mathbb{E}\!\left[\frac{d_{2k,1}(X)}{D_{2k}}\right]
=\frac12,\qquad X\sim\mathrm{Unif}\{x^{(1)},x^{(2)}\}.
\]
\end{corollary}

\begin{proof}
由注 \ref{rem:pom-max-fiber-achievers-bsplit} 的偶数支闭式分裂对与 \(D_{2k}=F_{k+2}=F_{k+1}+F_k\)（定理 \ref{thm:pom-max-fiber}），有
\[
d_{2k,0}(x^{(1)})+d_{2k,0}(x^{(2)})=F_{k+1}+F_k=D_{2k},
\]
并同理得到 \(d_{2k,1}(x^{(1)})+d_{2k,1}(x^{(2)})=D_{2k}\)。两边同除以 \(2D_{2k}\) 即得无偏结论。证毕。
\end{proof}

\begin{corollary}[最大纤维相位的隐藏位偏置谱（闭式与黄金常数极限）]\label{cor:pom-max-fiber-hidden-bit-bias-spectrum}
对每个 \(m\ge 2\) 与 \(x\in X_m\)，在隐藏位分裂下定义归一化偏置
\[
\mathrm{Bias}_m(x):=\frac{d_{m,0}(x)-d_{m,1}(x)}{D_m}\in[-1,1].
\]
在注 \ref{rem:pom-max-fiber-achievers-bsplit} 的稳定相位范围内，有如下闭式：
\begin{itemize}
  \item \textbf{偶数支 \(m=2k\ge 10\)：}两条最大相位满足
  \[
  \mathrm{Bias}_{2k}(x)=\pm\frac{F_{k-1}}{F_{k+2}},
  \]
  因而 \(\abs{\mathrm{Bias}_{2k}(x)}\to \varphi^{-3}=\sqrt5-2\)。
  \item \textbf{奇数支 \(m=2k+1\ge 13\)：}四条最大相位中恰有两条为平衡分裂，满足 \(\mathrm{Bias}_{2k+1}(x)=0\)；其余两条为偏置分裂，满足
  \[
  \mathrm{Bias}_{2k+1}(x)=\pm\frac{F_{k-2}}{F_{k+1}},
  \]
  因而偏置两相的幅度同样满足 \(\abs{\mathrm{Bias}_{2k+1}(x)}\to \varphi^{-3}\)。
\end{itemize}
因此奇数最大相位族在偏置口径下呈现出一个\emph{两级谱}：\(\{0,\pm\varphi^{-3}\}\)（渐近意义）。
\end{corollary}
\begin{proof}
直接把注 \ref{rem:pom-max-fiber-achievers-bsplit} 的分裂对代入定义，并用 \(D_{2k}=F_{k+2}\)、\(D_{2k+1}=2F_{k+1}\)（定理 \ref{thm:pom-max-fiber}）化简即可。
极限由 \(F_{n-3}/F_n\to \varphi^{-3}\) 给出；闭式 \(\varphi^{-3}=\sqrt5-2\) 由 \(\varphi^3=2+\sqrt5\) 立得。证毕。
\end{proof}

\begin{remark}[相内选择：\texorpdfstring{$b$}{b}-分裂精细矩在奇数最大相位上向偏置两相坍缩]\label{rem:pom-max-fiber-bsplit-phase-selection}
标准碰撞矩 \(S_q(m)=\sum_x d_m(x)^q\) 在最大纤维族内部\emph{不会}产生“相内选择”，因为所有最大达到者都有同一 \(d_m(x)=D_m\)。
但若考虑对隐藏位敏感的精细矩，例如
\[
S_q^{\mathrm{split}}(m):=\sum_{x\in X_m}\bigl(d_{m,0}(x)^q+d_{m,1}(x)^q\bigr),
\]
或只取每条纤维的最大分支
\[
S_q^{\max}(m):=\sum_{x\in X_m}\max\{d_{m,0}(x),d_{m,1}(x)\}^q,
\]
则在奇数稳定相位 \(m=2k+1\ge 13\) 的最大达到者四相中，偏置两相的最大分支为 \(F_{k+1}+F_{k-2}\)，平衡两相的最大分支为 \(F_{k+1}\)（注 \ref{rem:pom-max-fiber-achievers-bsplit}）。
因此最大分支比
\[
R_k:=\frac{F_{k+1}+F_{k-2}}{F_{k+1}}=1+\frac{F_{k-2}}{F_{k+1}}\longrightarrow 1+\varphi^{-3}=\sqrt5-1.
\]
从而当 \(q\) 增大时，偏置两相在上述 \(\max\)-口径或 \(b\)-分裂口径下对高阶矩的贡献相对平衡两相呈指数放大（尺度为 \(R_k^q\)），表现为“在最大达到者族内部由 \(4\) 相有效坍缩到 \(2\) 相”的选择效应。
\end{remark}

\begin{corollary}[空间缺陷的夹逼区间]\label{cor:pom-defect-band}
令平均纤维多重度 $\overline d_m=2^m/F_{m+2}$。则
\[
\log \overline d_m\sim m\log\!\left(\frac{2}{\varphi}\right),
\qquad
\log D_m\sim \frac{m}{2}\log\varphi.
\]
因此空间缺陷在“平均尺度”与“最坏尺度”之间形成硬夹逼区间。
\end{corollary}

\begin{remark}[最坏尺度与平均尺度的偏离]\label{rem:pom-defect-gap}
由渐近式可得
\[
\frac{D_m}{\overline d_m}
=
\Theta\!\left(\left(\frac{\varphi^{3/2}}{2}\right)^m\right),
\qquad
\frac{\varphi^{3/2}}{2}\approx 1.029\ldots
\]
因此最坏纤维相对平均纤维的偏离虽为指数，但基数极小，表明折叠投影在“最坏空间损失”意义下仍保持温和的尺度膨胀。
\end{remark}

\begin{corollary}[高多重纤维的稀疏性]\label{cor:pom-high-multiplicity-sparse}
对任意 $\alpha\in(0,1]$，令
\[
S_\alpha(m):=\{x\in X_m:\ d_m(x)\ge \alpha D_m\}.
\]
则
\[
|S_\alpha(m)|\le \frac{2^m}{\alpha D_m},
\qquad
\frac{|S_\alpha(m)|}{|X_m|}\le \frac{2^m}{\alpha D_m F_{m+2}}.
\]
并且由定理 \ref{thm:pom-max-fiber} 与 $F_{m+2}\sim \varphi^m/\sqrt5$，
\[
\frac{|S_\alpha(m)|}{|X_m|}
=O\!\left(\frac{1}{\alpha}\left(\frac{2}{\varphi^{3/2}}\right)^m\right).
\]
\end{corollary}
\begin{proof}
由 $\sum_{x\in X_m} d_m(x)=|\Omega_m|=2^m$，有
\(
\alpha D_m |S_\alpha(m)|\le \sum_{x\in S_\alpha(m)} d_m(x)\le 2^m
\)，
得第一式；再除以 $|X_m|=F_{m+2}$ 得第二式。最后一条由渐近式直接得到。证毕。
\end{proof}
