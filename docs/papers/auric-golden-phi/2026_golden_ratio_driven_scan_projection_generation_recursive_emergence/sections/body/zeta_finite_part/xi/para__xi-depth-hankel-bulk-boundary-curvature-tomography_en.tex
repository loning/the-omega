\begin{proposition}[Logistic kernel decomposition of scale entropy curvature: critical scales \texorpdfstring{$\sigma_j=-\log\delta_j$}{sigma_j=-log delta_j}]\label{prop:xi-bulk-curvature-logistic-kernel-decomposition}
Under the finite-atom setting of Corollary \ref{cor:xi-bulk-entropy-strict-convexity}, define the bulk scale entropy curvature density
\[
\mathcal{K}_{\mathrm{bulk}}(s):=\frac{\dd^{2}}{\dd s^{2}}S_{\mathrm{bulk}}(e^{s})\qquad(s\in\RR),
\]
and introduce the standard logistic density
\[
\phi(t):=\frac{e^{t}}{(1+e^{t})^{2}}\qquad(t\in\RR),
\]
along with the critical scales \(\sigma_j:=-\log\delta_j\).
Then there is a strict decomposition
\[
\boxed{\;
\frac{1}{2\kappa-1}\mathcal{K}_{\mathrm{bulk}}(s)=\sum_{j=1}^{\kappa}\phi(s-\sigma_j).\;}
\]
Moreover, \(\phi\) is unimodal on \(\RR\): its unique maximum is \(\phi(0)=\tfrac14\), and for any \(t\neq 0\), \(\phi(t)<\tfrac14\).
Therefore \(\{\sigma_j\}\) yields the translation centers of the curvature kernel family on the scale axis, corresponding to critical scales \(m\asymp \delta_j^{-1}\) under the uniform dilation \(m=e^s\).
\end{proposition}
\begin{proof}
By Corollary \ref{cor:xi-bulk-entropy-strict-convexity},
\[
\mathcal{K}_{\mathrm{bulk}}(s)
=(2\kappa-1)\sum_{j=1}^{\kappa}\frac{e^{s}\delta_j}{(1+e^{s}\delta_j)^{2}}.
\]
Writing \(\delta_j=e^{-\sigma_j}\), we have \(\frac{e^{s}\delta_j}{(1+e^{s}\delta_j)^{2}}=\phi(s-\sigma_j)\), yielding the stated decomposition.
On the other hand,
\[
\phi'(t)=\frac{e^{t}(1-e^{t})}{(1+e^{t})^{3}},
\]
hence \(\phi'(t)>0\) when \(t<0\), \(\phi'(t)<0\) when \(t>0\), so \(\phi\) is unimodal with unique maximum at \(t=0\); \(\phi(0)=\tfrac14\) follows by substitution.
\end{proof}

\begin{corollary}[Total mass quantization of entropy curvature: \texorpdfstring{$\int\mathcal{K}_{\mathrm{bulk}}=\kappa(2\kappa-1)$}{int K_bulk = k(2k-1)}]\label{cor:xi-bulk-curvature-mass-quantization}

\[
\boxed{\;
\int_{-\infty}^{\infty}\mathcal{K}_{\mathrm{bulk}}(s)\,\dd s=\kappa(2\kappa-1),
\qquad
\int_{-\infty}^{\infty}\phi(s-\sigma_j)\,\dd s=1\ \ (1\le j\le\kappa).\;}
\]
\end{corollary}
\begin{proof}
For any fixed \(j\), set \(u=e^{s-\sigma_j}\), then \(\dd s=\dd u/u\), and
\[
\int_{-\infty}^{\infty}\phi(s-\sigma_j)\,\dd s
=\int_{0}^{\infty}\frac{u}{(1+u)^2}\frac{\dd u}{u}
=\int_{0}^{\infty}\frac{\dd u}{(1+u)^2}
=1.
\]
Substituting into the decomposition formula of Proposition \ref{prop:xi-bulk-curvature-logistic-kernel-decomposition} yields the curvature total mass formula.
\end{proof}

\end{proposition}
\begin{proof}
Take the Fourier transform
\[
\widehat f(\xi):=\int_{\RR}e^{-\mathrm{i}\xi s}f(s)\,\dd s.

\[
\widehat\phi(\xi)=\frac{\pi\xi}{\sinh(\pi\xi)}\qquad(\xi\in\RR),

Taking the Fourier transform of both sides of Proposition \ref{prop:xi-bulk-curvature-logistic-kernel-decomposition} yields

\sum_{j=1}^{\kappa}e^{-\mathrm{i}\xi\sigma_j}

The right-hand side is determined by \(\mathcal{K}_{\mathrm{bulk}}\), hence we obtain the Fourier transform \(\widehat\nu(\xi)=\sum_j e^{-\mathrm{i}\xi\sigma_j}\) of the discrete measure \(\nu:=\sum_{j=1}^{\kappa}\delta_{\sigma_j}\).
The Fourier transform is injective on finite Radon measures, therefore \(\nu\) (equivalently the multiset \(\{\sigma_j\}\)) is uniquely determined; hence \(\delta_j=e^{-\sigma_j}\) is uniquely determined.
\end{proof}

\begin{proposition}[Speed matching of the dilation flow: bulk and boundary entropy growth rates intersect at unique scale]\label{prop:xi-bulk-boundary-speed-matching}

\[

\qquad
v_{\mathrm{gen}}(s):=\frac{\dd}{\dd s}S_{\mathrm{gen}}(e^s)=\kappa
\]

\[
\boxed{\;
\sum_{j=1}^{\kappa}\frac{1}{1+e^{s_\star}\delta_j}=\frac{\kappa(\kappa-1)}{2\kappa-1}.\;}
\]
\end{proposition}
\begin{proof}
By Corollary \ref{cor:xi-bulk-entropy-strict-convexity}, \(v_{\mathrm{bulk}}'(s)=\mathcal{K}_{\mathrm{bulk}}(s)>0\), hence \(v_{\mathrm{bulk}}\) is strictly increasing on \(\RR\).

\[
\lim_{s\to-\infty}v_{\mathrm{bulk}}(s)=\kappa^2-(2\kappa-1)\kappa=-\kappa(\kappa-1),
\qquad
\lim_{s\to\infty}v_{\mathrm{bulk}}(s)=\kappa^2.
\]
When \(\kappa\ge 2\), we have \(-\kappa(\kappa-1)<\kappa<\kappa^2\), hence by the intermediate value theorem there exists \(s_\star\) such that \(v_{\mathrm{bulk}}(s_\star)=\kappa\); strict monotonicity gives uniqueness.
Substituting \(v_{\mathrm{bulk}}(s_\star)=\kappa\) into the closed form of Corollary \ref{cor:xi-bulk-entropy-rg-staircase} and simplifying yields the stated equation.
\end{proof}

\begin{corollary}[Mean bracketing of the speed-matching scale: parameter-free interval from \texorpdfstring{$\delta_{\min}$}{delta_min} and \texorpdfstring{$\bar\delta$}{delta_bar}]\label{cor:xi-bulk-boundary-speed-matching-scale-bracketing}

\delta_{\min}:=\min_{1\le j\le\kappa}\delta_j,
\qquad
\bar\delta:=\frac{1}{\kappa}\sum_{j=1}^{\kappa}\delta_j.
\]

\[
\boxed{\;
\frac{\kappa}{(\kappa-1)\,\bar\delta}\ \le\ e^{s_\star}\ \le\ \frac{\kappa}{(\kappa-1)\,\delta_{\min}}\;}
\]
hence
\[
\log\frac{\kappa}{\kappa-1}-\log\bar\delta\ \le\ s_\star\ \le\ \log\frac{\kappa}{\kappa-1}-\log\delta_{\min}.
\]
Therefore, even without recovering the entire \(\{\delta_j\}\), merely knowing \((\bar\delta,\delta_{\min})\) suffices to compress the bulk--boundary speed intersection scale into a fully auditable parameter-free interval.
\end{corollary}
\begin{proof}
Dividing the equation of Proposition \ref{prop:xi-bulk-boundary-speed-matching} by \(\kappa\) yields
\[
\frac{1}{\kappa}\sum_{j=1}^{\kappa}\frac{1}{1+e^{s_\star}\delta_j}=\frac{\kappa-1}{2\kappa-1}.

For fixed \(a>0\), the function \(x\mapsto (1+ax)^{-1}\) is monotone decreasing and convex on \((0,\infty)\).
Hence by Jensen's inequality
\[
\frac{1}{1+a\bar\delta}\le \frac{1}{\kappa}\sum_{j=1}^{\kappa}\frac{1}{1+a\delta_j},
\]
setting \(a=e^{s_\star}\) and substituting the explicit value on the right-hand side, simplifying gives
\(
e^{s_\star}\ge \frac{\kappa}{(\kappa-1)\bar\delta}

On the other hand, since \(\delta_j\ge \delta_{\min}\) and the function is decreasing,
\[
\frac{1}{\kappa}\sum_{j=1}^{\kappa}\frac{1}{1+a\delta_j}\le \frac{1}{1+a\delta_{\min}}.
\]
Similarly taking \(a=e^{s_\star}\) and substituting, simplifying gives
\(
e^{s_\star}\le \frac{\kappa}{(\kappa-1)\delta_{\min}}
\).
The logarithmic form follows by taking logarithms of both ends.
\end{proof}

\begin{proposition}[Moment generating function factorization of entropy curvature distribution: universal \texorpdfstring{$\zeta(2n)$}{zeta(2n)} tower and depth-spectral cumulant splitting]\label{prop:xi-bulk-curvature-mgf-factorization-zeta-tower}

\[
\boxed{\;

=\Gamma(1+t)\Gamma(1-t)\cdot \frac{1}{\kappa}\sum_{j=1}^{\kappa}e^{t\sigma_j}\;}
\qquad(|t|<1).
\]
By Euler reflection and the Hadamard product for \(\sin\),
\[
\Gamma(1+t)\Gamma(1-t)=\frac{\pi t}{\sin(\pi t)}
=\prod_{n\ge 1}\Bigl(1-\frac{t^2}{n^2}\Bigr)^{-1},
\]
hence the cumulant generating function splits strictly for \(|t|<1\) as
\[
\log\!\int_{\RR}e^{ts}\,p(s)\,\dd s
=\sum_{n\ge 1}\frac{\zeta(2n)}{n}\,t^{2n}
\;+\;\log\!\Bigl(\frac{1}{\kappa}\sum_{j=1}^{\kappa}e^{t\sigma_j}\Bigr).
\]
Let \(\kappa_q(p):=\left.\frac{\dd^q}{\dd t^q}\log\int_{\RR}e^{ts}p(s)\dd s\right|_{t=0}\) be the \(q\)-th cumulant of \(p\),
and let \(\kappa_q(\sigma):=\left.\frac{\dd^q}{\dd t^q}\log\bigl(\kappa^{-1}\sum_j e^{t\sigma_j}\bigr)\right|_{t=0}\) be the cumulant of the equal-weight discrete measure \(\kappa^{-1}\sum_j\delta_{\sigma_j}\).
Then for all \(n\ge 1\)
\[
\boxed{\;
\kappa_{2n}(p)=(2n)!\,\frac{\zeta(2n)}{n}\;+\;\kappa_{2n}(\sigma),
\qquad
\kappa_{2n+1}(p)=\kappa_{2n+1}(\sigma).\;}
\]
Therefore the higher-order constants of bulk scale response are strictly locked as the sum of even \(\zeta\) values and depth-spectral discrete cumulants.
\end{proposition}
\begin{proof}
By Corollary \ref{cor:xi-bulk-curvature-probability-moments},
\(
p(s)=\kappa^{-1}\sum_{j=1}^\kappa \phi(s-\sigma_j)
\).
Hence within \(|t|<1\) we can exchange summation and integration to get
\[
\int_{\RR}e^{ts}\,p(s)\,\dd s
=\frac{1}{\kappa}\sum_{j=1}^{\kappa}e^{t\sigma_j}\int_{\RR}e^{tu}\phi(u)\,\dd u.
\]
For \(\phi(u)=\frac{e^{u}}{(1+e^{u})^{2}}\), make the substitution \(y=e^{u}\) (\(\dd u=\dd y/y\)), obtaining
\[
\int_{\RR}e^{tu}\phi(u)\,\dd u
=\int_{0}^{\infty}\frac{y^{t}}{(1+y)^{2}}\,\dd y
=\mathrm{B}(1+t,1-t)
=\Gamma(1+t)\Gamma(1-t),
\]
where the Beta identity requires \(-1<\Re(t)<1\), hence on the real axis yields the analytic domain \(|t|<1\).
Thus we obtain the first factorization.
The second splitting follows from Euler reflection formula and the product expansion of \(\sin\) (or equivalently from the \(\zeta\)-series of \(\log\Gamma(1\pm t)\)).
Finally, differentiating both sides at \(t=0\), using
\(
\frac{\dd^{2n}}{\dd t^{2n}}\sum_{m\ge 1}\frac{\zeta(2m)}{m}t^{2m}\big|_{t=0}
=(2n)!\,\frac{\zeta(2n)}{n}
\)
and noting odd terms vanish, yields the stated cumulant splitting.
\end{proof}

\begin{proposition}[Closed form of scale coherence kernel: Gram volume and hyperbolic function explicitization of \texorpdfstring{$\mathrm{sech}^2$}{sech^2} translation family]\label{prop:xi-scale-coherence-kernel-closed-form}
Write the logistic kernel in equivalent form
\[
\phi(t)=\frac{1}{4\cosh^2(t/2)}=\frac{1}{4}\,\mathrm{sech}^2(t/2).
\]
Define the two-point scale coherence kernel (\(L^2\) inner product of the translation family)
\[
K(\Delta):=\int_{\RR}\phi(s)\phi(s-\Delta)\,\dd s\qquad(\Delta\in\RR).
\]
Then \(K\) is an even function, with strict closed form
\[
\boxed{\;
K(\Delta)=\frac{\Delta\cosh(\Delta/2)-2\sinh(\Delta/2)}{4\sinh^3(\Delta/2)}\quad(\Delta\neq 0),
\qquad
K(0)=\frac{1}{6}.\;}
\]
Its local expansion at \(\Delta=0\) is
\[
K(\Delta)=\frac{1}{6}-\frac{\Delta^2}{60}+\frac{\Delta^4}{1008}+O(\Delta^6).
\]
\end{proposition}
\begin{proof}
Let \(a:=\Delta/2\) and make the variable substitution \(s=2x\). By \(\phi(2x)=\frac14\mathrm{sech}^2(x)\),
\[
K(\Delta)=\frac{1}{16}\int_{\RR}\mathrm{sech}^2(x)\,\mathrm{sech}^2(x-a)\,2\,\dd x
=\frac18\int_{\RR}\mathrm{sech}^2(x)\,\mathrm{sech}^2(x-a)\,\dd x.
\]
Let \(y=\tanh x\), then \(\dd y=\mathrm{sech}^2(x)\dd x\), and denote \(t:=\tanh a\).
Using the hyperbolic tangent difference formula
\(
\tanh(x-a)=\frac{y-t}{1-ty}
\)
we get
\[
\mathrm{sech}^2(x-a)=1-\tanh^2(x-a)=\mathrm{sech}^2(a)\cdot\frac{1-y^2}{(1-ty)^2}.
\]
Hence
\[
\int_{\RR}\mathrm{sech}^2(x)\,\mathrm{sech}^2(x-a)\,\dd x
=\mathrm{sech}^2(a)\int_{-1}^{1}\frac{1-y^2}{(1-ty)^2}\,\dd y.
\]
Direct integration (or substituting \(u=1-ty\)) gives
\[
\int_{-1}^{1}\frac{1-y^2}{(1-ty)^2}\,\dd y
=\frac{2}{t^3}\Bigl(\log\frac{1+t}{1-t}-2t\Bigr)
=\frac{4}{t^3}(a-t),
\]
using \(\log\frac{1+t}{1-t}=2\operatorname{arctanh}(t)=2a\).
Thus
\[
K(\Delta)
=\frac18\cdot \mathrm{sech}^2(a)\cdot \frac{4}{t^3}(a-t)
=\frac12\cdot\frac{a\cosh a-\sinh a}{\sinh^3 a},
\]
substituting back \(a=\Delta/2\) yields the closed form for \(\Delta\neq 0\).
\(K(0)=1/6\) follows from the limit of the closed form as \(\Delta\to 0\) (or direct integration).
The local expansion follows directly from the Taylor expansions of \(\sinh\) and \(\cosh\).
\end{proof}

\begin{proposition}[Vandermonde exponential law of scale collision: universal divergence order of \texorpdfstring{$S_{\mathrm{scale}}$}{S_scale} is \(2\)]\label{prop:xi-scale-coherence-collision-vandermonde}

Then there exists a constant \(C_\kappa>0\) (depending only on kernel \(K\) and \(\kappa\)) such that
\[

= C_\kappa\,\varepsilon^{\kappa(\kappa-1)}
\Bigl(\prod_{1\le j<k\le\kappa}(t_k-t_j)^2\Bigr)\cdot\bigl(1+o(1)\bigr).
\]
Hence
\[
S_{\mathrm{scale}}(\boldsymbol{\sigma})
= -\log C_\kappa-\kappa(\kappa-1)\log\varepsilon-2\sum_{1\le j<k\le\kappa}\log|t_k-t_j|+o(1).
\]
Therefore each pairwise scale collision \(\sigma_j\to\sigma_k\) contributes a strict \(2\log|\sigma_j-\sigma_k|^{-1}\) divergence order.
\end{proposition}
\begin{proof}
By Proposition \ref{prop:xi-scale-coherence-kernel-closed-form}, \(K\) is an even analytic function in a neighborhood of \(0\), hence can be written as
\[
K(\Delta)=\sum_{n\ge 0}\mu_{2n}\Delta^{2n},\qquad \mu_0=K(0)>0.
\]
Substituting \(\sigma_j=\sigma_\star+\varepsilon t_j\) gives
\(
G^{\mathrm{scale}}_{jk}=K(\varepsilon(t_j-t_k))
\).
Applying Taylor expansion to the matrix elements and applying confluent Vandermonde-type expansion to the determinant (equivalently: projecting \(\{K(\cdot-\sigma_j)\}\) as \(\varepsilon\to 0\) onto the finite-dimensional subspace spanned by \(\{\partial^r K(\cdot-\sigma_\star)\}_{r=0}^{\kappa-1}\)), we obtain
\[
\det G^{\mathrm{scale}}(\boldsymbol{\sigma})
=\varepsilon^{\kappa(\kappa-1)}\Bigl(\prod_{j<k}(t_k-t_j)^2\Bigr)\cdot\Bigl(\det H_\kappa+o(1)\Bigr),
\]
where \(H_\kappa\) is a Hankel-type matrix generated by \(\{\mu_{2n}\}\), with \(\det H_\kappa>0\) depending only on \(K\) and \(\kappa\).
Setting \(C_\kappa:=\det H_\kappa\) yields the stated main term.
The logarithmic asymptotic follows by taking \(-\log\) of both sides and expanding.
\end{proof}

\begin{proposition}[\texorpdfstring{$\zeta$}{zeta}-coefficients of scale coherence kernel: Taylor coefficients of \texorpdfstring{$K$}{K} completely sealed by \texorpdfstring{$\zeta(2n+2)$}{zeta(2n+2)}]\label{prop:xi-scale-coherence-kernel-zeta-taylor}

\[
\boxed{\;
\mu_{2n}
=\frac{K^{(2n)}(0)}{(2n)!}
=(-1)^n\,\frac{(2n+2)(2n+1)}{2^{2n+1}\pi^{2n+2}}\;\zeta(2n+2).\;}

Hence the constant \(C_\kappa\) of Proposition \ref{prop:xi-scale-coherence-collision-vandermonde} can be expressed as a Hankel-type determinant generated by \(\{\mu_{2n}\}\) (hence essentially a rational combination of even \(\zeta\) values).
\end{proposition}
\begin{proof}
By the Fourier closed form of Proposition \ref{prop:xi-bulk-curvature-tomography-identifiability},
\(
\widehat\phi(\xi)=\pi\xi/\sinh(\pi\xi)
\).
By the convolution identity \(K=\phi\ast\phi\) (\(\phi\) is an even function),
\[
K(\Delta)=\frac{1}{2\pi}\int_{\RR}\widehat\phi(\xi)^2\,e^{\mathrm{i}\xi\Delta}\,\dd\xi
=\frac{1}{2\pi}\int_{\RR}\Bigl(\frac{\pi\xi}{\sinh(\pi\xi)}\Bigr)^{\!2}e^{\mathrm{i}\xi\Delta}\,\dd\xi.
\]
Expanding \(e^{\mathrm{i}\xi\Delta}\) in power series and integrating term by term (absolute convergence follows from \(\widehat\phi(\xi)=O(e^{-\pi|\xi|})\)) yields
\[
\mu_{2n}
=\frac{1}{2\pi(2n)!}\int_{\RR}(\mathrm{i}\xi)^{2n}\Bigl(\frac{\pi\xi}{\sinh(\pi\xi)}\Bigr)^{\!2}\dd\xi
=\frac{(-1)^n\pi}{2(2n)!}\int_{\RR}\frac{\xi^{2n+2}}{\sinh^2(\pi\xi)}\,\dd\xi.
\]
Using the series expansion
\(
\sinh^{-2}x=4\sum_{k\ge 1}k\,e^{-2kx}\ (x>0)
\)
and simplifying for the even function to \(2\int_0^\infty\), we get
\[
\int_{\RR}\frac{\xi^{2n+2}}{\sinh^2(\pi\xi)}\,\dd\xi
=8\sum_{k\ge 1}k\int_0^\infty \xi^{2n+2}e^{-2\pi k\xi}\,\dd\xi
=8\sum_{k\ge 1}k\cdot\frac{(2n+2)!}{(2\pi k)^{2n+3}}
=\frac{(2n+2)!}{2^{2n}\pi^{2n+3}}\zeta(2n+2).
\]
Substituting back and simplifying yields the stated closed form.
The statement about \(C_\kappa\) follows from the Hankel structure in the proof of Proposition \ref{prop:xi-scale-coherence-collision-vandermonde} and the coefficient closed form.
\end{proof}

\begin{proposition}[Majorization and convex order of depth spectra: more dispersed boundaries necessarily yield stronger scale fluctuations]\label{prop:xi-depth-spectrum-majorization-convex-order}
Let two depth spectrum logarithmic coordinates
\(
\boldsymbol{\sigma}=(\sigma_1,\dots,\sigma_\kappa)
\)
and
\(
\boldsymbol{\sigma}'=(\sigma_1',\dots,\sigma_\kappa')
\)
satisfy equal means
\(
\kappa^{-1}\sum_j\sigma_j=\kappa^{-1}\sum_j\sigma_j'
\),
and \(\boldsymbol{\sigma}\) majorizes \(\boldsymbol{\sigma}'\).
Then for any convex function \(F:\RR\to\RR\)
\[
\frac{1}{\kappa}\sum_{j=1}^{\kappa}F(\sigma_j)\ \ge\ \frac{1}{\kappa}\sum_{j=1}^{\kappa}F(\sigma_j').
\]
Let \(p_{\boldsymbol{\sigma}}(s):=\kappa^{-1}\sum_{j=1}^{\kappa}\phi(s-\sigma_j)\) and \(p_{\boldsymbol{\sigma}'}\) be the corresponding scale curvature distributions.
Then the above majorization relation is order-preserved under lifting to convex order: for any convex \(F\)
\[
\int_{\RR}F(s)\,p_{\boldsymbol{\sigma}}(s)\,\dd s\ \ge\ \int_{\RR}F(s)\,p_{\boldsymbol{\sigma}'}(s)\,\dd s.
\]
In particular, all central even moments satisfy monotonicity: the more dispersed the depth spectrum, the non-decreasing the variance, kurtosis, and higher even central moments of the scale curvature.
\end{proposition}
\begin{proof}
The Jensen-type inequality implied by majorization is a standard conclusion from Hardy--Littlewood--Pólya theory: for any convex \(F\) the discrete average is monotone.
On the other hand, let the random variable \(U\) be uniform on \(\{1,\dots,\kappa\}\), take \(S:=\sigma_U\), \(S':=\sigma'_U\).
Let \(Z\) be independent with density \(\phi\), then \(S+Z\) and \(S'+Z\) have densities \(p_{\boldsymbol{\sigma}}\) and \(p_{\boldsymbol{\sigma}'}\) respectively.
By translation invariance of convex order (independent sums preserve convex order), for any convex \(F\),
\(
\EE[F(S+Z)]\ge \EE[F(S'+Z)]
\),
which is exactly the stated integral inequality.
Taking \(F(s)=(s-\EE[s])^{2n}\) yields central even moment monotonicity.
\end{proof}
