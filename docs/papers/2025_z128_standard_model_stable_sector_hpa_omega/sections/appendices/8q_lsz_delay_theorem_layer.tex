\subsection{LSZ interface and time-delay equivalence (T4-2)}
\label{app:lsz_delay_theorem_layer}

\AuditTag This appendix upgrades the delay/scattering interface to a theorem-facing statement under explicit prerequisites.
It does not replace the operational delay dictionary; it states when that dictionary is provably equivalent to standard scattering quantities.

\subsubsection{Condition bundle}
\begin{assumption}[LSZ applicability bundle (template)]
\label{ass:lsz_applicability_bundle_template}
Assume:
\begin{itemize}
\item Haag--Ruelle scattering holds in the declared sector (Appendix~\ref{app:scattering_haag_ruelle_theorems}),
\item the relevant interpolating fields/currents exist with the regularity required by LSZ (or an equivalent net-based formulation),
\item approximate unitarity on the declared energy band (canonical S1 record if only approximate).
\end{itemize}
\end{assumption}

\subsubsection{Theorem layer}
\begin{theorem}[Time delay from scattering phase (template)]
\label{thm:delay_from_phase_template}
\MathTag Under Assumption~\ref{ass:lsz_applicability_bundle_template}, the Wigner--Smith operator
\[
Q(\omega)=-\iu\,S(\omega)^\dagger \frac{\dd S}{\dd \omega}
\]
is well-defined (in the stated channel basis), and the operational delay observable in the manuscript's dictionary coincides with the corresponding scattering-phase derivative quantity up to the declared calibration/loss error model.
\end{theorem}

\AuditTag Failure modes are normalized by S1--S3 (Appendix~\ref{app:minimal_failure_point_templates}); if LSZ regularity is not available, the manuscript keeps only the operational delay dictionary as an interface object (Appendix~\ref{app:time_mass_delay}).
