\section{Gauge-factor complexity-label sensitivity (supplement)}
\label{app:gauge_complexity_sensitivity}

Proposition~\ref{prop:channel_to_gauge} closes the non-abelian gauge factors by CAP within an explicit compact-factor candidate family.
One audit concern is whether the output depends on a particular choice of ``intrinsic complexity label'' (e.g.\ using $\dim(\mathfrak{g})$ rather than rank or representation size).
This appendix (i) records a bounded sensitivity sweep across several natural discrete complexity labels and (ii) notes a short theorem-level reason why the same minimizer persists for these labels.

\subsection{Candidate family and selection rule}
We enumerate compact, connected, non-abelian simple Lie groups through their Lie algebras within a bounded audit window, using the classical families together with low-dimensional exceptional cases.
Within this finite list, we select a pair $(G_2,G_3)$ of non-isomorphic simple factors by the same CAP form used in Proposition~\ref{prop:channel_to_gauge}: choose the lexicographically minimal pair under a given factor complexity label, with a deterministic tie-break that refines by $(\dim, \mathrm{rank}, \text{name})$.

We compare the following factor labels:
\begin{itemize}
\item $\dim(\mathfrak{g})$ (dimension-as-complexity, used in the main text);
\item $\mathrm{rank}(\mathfrak{g})$;
\item $\dim(\mathfrak{g})+\mathrm{rank}(\mathfrak{g})$;
\item $d_{\min}$, the minimal dimension of a faithful complex representation (up to finite quotients).
\end{itemize}

\subsection{A short robustness proof for common labels}
\label{subsec:gauge_label_robustness_proof}
\begin{proposition}[Robustness of the \texorpdfstring{$SU(2)\times SU(3)$}{SU2xSU3} minimizer across common labels]
\label{prop:gauge_label_robustness}
Consider the CAP selection rule in Proposition~\ref{prop:channel_to_gauge}, but replace the primary factor label $\dim(\mathfrak{g})$ by any one of the following labels on compact simple Lie algebras:
\[
k\in\Bigl\{\dim(\mathfrak{g}),\ \mathrm{rank}(\mathfrak{g}),\ \dim(\mathfrak{g})+\mathrm{rank}(\mathfrak{g}),\ d_{\min}\Bigr\},
\]
where $d_{\min}$ denotes the minimal dimension of a nontrivial complex representation of $\mathfrak{g}$ (equivalently: a faithful complex representation, since $\mathfrak{g}$ is simple).
Assume the same deterministic tie-break refinement used in the sweep (refining by $(\dim,\mathrm{rank},\text{name})$).
Then the unique lexicographic minimizer of $(k(\mathfrak{g}_2),k(\mathfrak{g}_3))$ over pairs of non-isomorphic compact simple factors is
$(\mathfrak{su}(2),\mathfrak{su}(3))$.
\end{proposition}
\begin{proof}
\textbf{(1) First factor is always $\mathfrak{su}(2)$.}
Among non-abelian compact simple Lie algebras, the minimal possible values of
\[
\dim(\mathfrak{g}),\qquad \mathrm{rank}(\mathfrak{g}),\qquad d_{\min}(\mathfrak{g})
\]
are attained uniquely at $\mathfrak{su}(2)$:
\begin{itemize}
  \item $\dim(\mathfrak{g})\ge 3$ with equality iff $\mathfrak{g}\cong\mathfrak{su}(2)$ (Lemma~\ref{lem:su2_su3_unique_by_dim});
  \item $\mathrm{rank}(\mathfrak{g})\ge 1$ with equality iff $\mathfrak{g}\cong\mathfrak{su}(2)$ (classification: rank-$1$ simple is $A_1$; see, e.g., \cite{HallLieGroups2015});
  \item $d_{\min}(\mathfrak{g})\ge 2$ with equality iff $\mathfrak{g}\cong\mathfrak{su}(2)$ (the fundamental $2$-dimensional representation of $A_1$; see, e.g., \cite{FultonHarris1991RepresentationTheory}).
\end{itemize}
Therefore, regardless of whether $k$ is $\dim$, $\mathrm{rank}$, $\dim{+}\mathrm{rank}$, or $d_{\min}$, the lexicographically minimal first factor is $\mathfrak{su}(2)$.

\medskip
\noindent\textbf{(2) Second factor is always $\mathfrak{su}(3)$.}
The second factor must be non-isomorphic to $\mathfrak{su}(2)$.
We check each label:
\begin{itemize}
  \item If $k=\dim$, then the minimal non-abelian simple dimension strictly larger than $3$ is $8$, attained uniquely at $\mathfrak{su}(3)$ (Lemma~\ref{lem:su2_su3_unique_by_dim}).
  \item If $k=\mathrm{rank}$, then the minimal rank strictly larger than $1$ is $2$.
  The compact simple rank-$2$ Lie algebras are $A_2$, $B_2\cong C_2$, and $G_2$.
  Their dimensions are $8$, $10$, and $14$, respectively, so the tie-break by $\dim$ selects $A_2=\mathfrak{su}(3)$.
  \item If $k=\dim+\mathrm{rank}$, then for any $\mathfrak{g}\not\cong\mathfrak{su}(2)$ one has $\dim(\mathfrak{g})\ge 8$ and $\mathrm{rank}(\mathfrak{g})\ge 2$, hence $k(\mathfrak{g})\ge 10$.
  Equality $k=10$ is attained by $\mathfrak{su}(3)$ (dimension $8$, rank $2$).
  For the other rank-$2$ cases, $B_2$ has $k=12$ and $G_2$ has $k=16$, and higher ranks only increase $k$.
  Thus $\mathfrak{su}(3)$ is the unique minimizer.
  \item If $k=d_{\min}$, then $d_{\min}(\mathfrak{su}(3))=3$ (the defining representation), while every other compact simple Lie algebra not isomorphic to $\mathfrak{su}(2)$ has $d_{\min}\ge 4$:
  for $B_2\cong C_2$ one has $d_{\min}=4$ (the defining representation of $C_2$), and for $G_2$ one has $d_{\min}=7$; higher ranks only increase the minimal nontrivial representation dimension in the classical families.
  Hence $d_{\min}$ selects $\mathfrak{su}(3)$.
\end{itemize}
This proves that for each listed label, the unique minimizer is $(\mathfrak{su}(2),\mathfrak{su}(3))$.
\end{proof}

\paragraph{Relation to the sweep.}
Table~\ref{tab:gauge_complexity_sensitivity} remains an audit artifact: it verifies by explicit enumeration (within a bounded window) that no implementation detail of the sweep changes the minimizer under the tested labels.
Proposition~\ref{prop:gauge_label_robustness} explains why this stability is expected already from low-rank classification facts.
