\begin{theorem}[Latt\`es 逆像树的挠子主齐次空间结构与仿射伽罗瓦表示]\label{thm:conclusion-leyang-lattes-preimage-torsor-affine-galois}
沿用本小节固定的极小椭圆模型
\[
E_{\min}:\ u^2+u=x^3-x,
\]
并沿用结论 \ref{con:xi-terminal-zm-elliptic-lattes-doubling} 中由倍点映射诱导的四次 Latt\`es 自映射
\[
\Phi(X)=\frac{X^4+2X^2-2X+1}{4X^3-4X+1}\in\QQ(X).
\]
设 \(K\) 为数域，取 \(t\in\PP^1(K)\) 满足 \(t\neq \infty\) 且 \(t\notin \mathrm{PostCrit}(\Phi)\)。
对 \(n\ge 1\) 记
\[
V_n(t):=\Phi^{-n}(t)\subset \PP^1(\overline K),
\qquad
E_{\min}[2^n]\cong (\ZZ/2^n\ZZ)^2.
\]
则：
\begin{enumerate}
  \item 对每个 \(n\ge 1\)，集合 \(V_n(t)\) 上存在一个自然的、随 \(n\) 相容的 \(E_{\min}[2^n]\)-自由且传递作用，使 \(V_n(t)\) 成为 \(E_{\min}[2^n]\) 的主齐次空间。
  \item 逆极限 \(\varprojlim_n V_n(t)\) 为 \(T_2(E_{\min})=\varprojlim_n E_{\min}[2^n]\) 的主齐次空间。
  \item 令 \(G_K:=\Gal(\overline K/K)\)。
  将 \(V_n(t)\) 视为 \(G_K\) 的 \(n\)-层作用集合族，则存在连续仿射型 arboreal 表示
  \[
  \rho_t:\ G_K\longrightarrow \mathrm{Aff}\bigl(T_2(E_{\min})\bigr)\cong T_2(E_{\min})\rtimes \GL_2(\ZZ_2),
  \]
  其线性部分等于 \(2\)-进 Tate 模表示 \(\rho_{E_{\min},2}:G_K\to \GL_2(\ZZ_2)\)，而平移部分由一条连续 \(1\)-上同调类 \(c_t\in Z^1(G_K,T_2(E_{\min}))\) 给出。
\end{enumerate}
\end{theorem}
\begin{proof}[证明要点]
取 \(P\in E_{\min}(\overline K)\) 使 \(x(P)=t\)。
由于 \(t\notin \mathrm{PostCrit}(\Phi)\)，可取 \(P\) 不为 \(2\)-挠点，从而 \(P\neq -P\) 且 \(x^{-1}(t)=\{P,-P\}\)。
对每个 \(n\) 取 \(Q_n\in E_{\min}(\overline K)\) 满足 \([2^n]Q_n=P\)。
定义
\[
\alpha_n:E_{\min}[2^n]\to V_n(t),\qquad T\longmapsto x(Q_n+T).
\]
若 \(X\in V_n(t)\)，取 \(R\in E_{\min}(\overline K)\) 使 \(x(R)=X\)。
由 \(\Phi^n(X)=t\iff x([2^n]R)=x(P)\) 得 \([2^n]R=\pm P\)；
若 \([2^n]R=-P\)，则 \([2^n](-R)=P\) 且 \(x(-R)=x(R)\)，故可改取 \(-R\)。
于是 \([2^n](R-Q_n)=0\)，即 \(R-Q_n\in E_{\min}[2^n]\)，从而 \(\alpha_n\) 满射。
若 \(\alpha_n(T)=\alpha_n(T')\)，则 \(Q_n+T'=\pm(Q_n+T)\)；
对两边施加 \([2^n]\) 得 \(P=\pm P\)，由 \(P\neq -P\) 知只能取正号，故 \(T'=T\)。
因此 \(\alpha_n\) 为双射，并把 \(E_{\min}[2^n]\) 的平移作用转移到 \(V_n(t)\) 上，给出主齐次空间结构。
\par
逆极限与相容性由 \([2]:E_{\min}[2^{n+1}]\to E_{\min}[2^n]\) 与 \(\Phi:V_{n+1}(t)\to V_n(t)\) 在上述双射下的交换性推出。
对 \(g\in G_K\) 定义
\(
c_{t,n}(g):=g(Q_n)-Q_n\in E_{\min}[2^n]
\)，则
\(
c_{t,n}(gh)=c_{t,n}(g)+\rho_{E_{\min},2^n}(g)c_{t,n}(h)
\)。
令 \(n\to\infty\) 得 \(c_t\in Z^1(G_K,T_2(E_{\min}))\)，并得到 \(\rho_t\) 的仿射分解。证毕。
\end{proof}

\begin{theorem}[倍点自然扩张的溶子结构、圆维分解与测度级一致性]\label{thm:conclusion-leyang-doubling-solenoid-extension}
视 \(E_{\min}(\CC)\) 为复环面 \(\CC/\Lambda\)，并考虑倍点映射 \([2]:E_{\min}(\CC)\to E_{\min}(\CC)\)。
定义逆极限紧群
\[
\Sigma_{E_{\min}}:=\varprojlim\Bigl(E_{\min}(\CC)\xleftarrow{[2]}E_{\min}(\CC)\xleftarrow{[2]}\cdots\Bigr)
=\{(P_0,P_1,\dots):[2]P_{n+1}=P_n\}.
\]
则：
\begin{enumerate}
  \item \(\Sigma_{E_{\min}}\) 为紧致阿贝尔群，并存在短正合列
  \[
  0\to T_2(E_{\min})\longrightarrow \Sigma_{E_{\min}}\xrightarrow{\ \pi_0\ }E_{\min}(\CC)\to 0,
  \]
  其中 \(\pi_0(P_0,P_1,\dots)=P_0\)。
  \item 左移同胚
  \(
  \sigma(P_0,P_1,\dots)=(P_1,P_2,\dots)
  \)
  的拓扑熵满足 \(h_{\mathrm{top}}(\sigma)=\log\deg([2])=\log 4\)。
  \item 记 \(\mu_{\Sigma}\) 为 \(\Sigma_{E_{\min}}\) 的 Haar 概率测度，则 \(\mu_{\Sigma}\) 在 \(\sigma\) 下不变。
  并且其在核纤维 \(T_2(E_{\min})=\ker(\pi_0)\) 的各有限商 \(E_{\min}[2^n]\) 上的推前均为均匀分布。
\end{enumerate}
\end{theorem}
\begin{proof}[证明要点]
逆极限在紧群与满射连续同态下保持紧致群结构。
\(\pi_0\) 为满射群同态，核由 \(P_0=O\) 的兼容序列组成，等价于兼容的 \(2^n\)-挠点系，故核同构于 \(T_2(E_{\min})\)，得短正合列。
\par
\(\sigma\) 为群自同胚。
对覆盖映射 \(f\) 的自然扩张，其拓扑熵与 \(f\) 相同，而 \([2]\) 的度为 \(4\)，故 \(h_{\mathrm{top}}(\sigma)=\log 4\)。
\par
紧群上的 Haar 测度在群自同胚下不变。
核纤维上的 Haar 测度在任意满射群同态下推前为 Haar 测度；有限群上的 Haar 测度即均匀分布，从而得到有限层一致性。证毕。
\end{proof}

\begin{conjecture}[溶子上的 Stokes--regulator 等式：\texorpdfstring{$\log|B|$}{log|B|} 的自守周期刻画]\label{conj:conclusion-leyang-solenoid-stokes-regulator}
沿用定理 \ref{thm:conclusion-leyang-doubling-solenoid-extension} 的 \(2\)-进溶子 \(\Sigma_{E_{\min}}\) 与短正合列。
设 Lee--Yang 振幅常数 \(B\) 已由附录定理 \ref{thm:killo-leyang-B-square-cubic-and-rational-intertwine} 在代数意义上刚性化。
则存在：
\begin{itemize}
  \item 一个自然的同调类 \(\Xi\in H_2(\Sigma_{E_{\min}};\ZZ)\)；
  \item 一个由 Hecke--Langlands 数据诱导的闭 \(2\)-形式 \(\Omega_{\mathrm{aut}}\)（在有限层上与权 \(2\) 的 N\'eron 微分/自守形式的规范配对相容），
\end{itemize}
使得成立归一化 Stokes 周期等式
\[
\boxed{\ \log|B|\ =\ \frac{1}{2\pi}\left|\int_{\Xi}\Omega_{\mathrm{aut}}\right|\ }.
\]
并且该积分在逆塔层级上具有一致的归一化性质：对每个有限层商 \(E_{\min}[2^n]\) 的相容投影，\(\Xi\) 与 \(\Omega_{\mathrm{aut}}\) 的推前/拉回在该等式下保持同一数值规范。
\end{conjecture}

\begin{proposition}[挠点与 Gödel 素数寄存器的 Kummer 化编码及其递归交换性]\label{prop:conclusion-leyang-torsion-godel-kummer-coding}
取互异素数 \(p,q\)。
令
\[
H_n:=\langle p,q\rangle\big/\langle p,q\rangle^{2^n}\subset \QQ^\times/(\QQ^\times)^{2^n},
\qquad n\ge 1,
\]
其中 \(\langle p,q\rangle\) 为由 \(p,q\) 生成的自由阿贝尔子群。
则 \(H_n\cong (\ZZ/2^n\ZZ)^2\)，并由
\[
(a,b)\longmapsto p^a q^b\ \bmod (\QQ^\times)^{2^n}
\]
给出同构。
选取 \(E_{\min}[2^n]\) 的一组基 \((P_n,Q_n)\)，并据此定义编码同构
\[
\mathcal G_n:E_{\min}[2^n]\stackrel{\sim}{\longrightarrow}H_n,\qquad aP_n+bQ_n\longmapsto p^a q^b\ \bmod(\QQ^\times)^{2^n}.
\]
则交换图成立：
\[
\begin{CD}
E_{\min}[2^{n+1}] @>{\mathcal G_{n+1}}>> H_{n+1}\\
@V{[2]}VV @VV{(\,\cdot\,)^2}V\\
E_{\min}[2^{n}] @>>{\mathcal G_{n}}> H_{n}.
\end{CD}
\]
\end{proposition}
\begin{proof}
在指数坐标下，\([2]\) 对 \((a,b)\) 作用为 \((2a,2b)\)（模 \(2^n\)），而平方映射对 \(p^a q^b\) 作用为 \(p^{2a}q^{2b}\)（模 \((\QQ^\times)^{2^n}\)），故交换性成立。证毕。
\end{proof}

\begin{theorem}[二进分辨率与 Pontryagin--Fourier 频率截断的严格同一]\label{thm:conclusion-leyang-t2-fourier-truncation-equals-precision}
取 \(G:=T_2(E_{\min})\cong \ZZ_2^2\)，记其 Haar 概率测度为 \(\mu\)。
对每个 \(n\ge 1\) 令 \(G_n:=E_{\min}[2^n]\) 并记自然投影为 \(\pi_n:G\to G_n\)；
令 \(\mathcal F_n\) 为由 \(\pi_n\) 生成的 \(\sigma\)-代数。
对任意 \(f\in L^2(G,\mu)\) 定义最佳 \(L^2\) 截断（条件期望）
\[
f_n:=\EE[f\mid \mathcal F_n].
\]
则有 Parseval 型精确分解
\[
\boxed{\;
\|f-f_n\|_{L^2(G,\mu)}^2
=\sum_{\substack{\chi\in \widehat G\\ \mathrm{ord}(\chi)>2^n}}|\widehat f(\chi)|^2,\;}
\qquad
\widehat f(\chi):=\int_G f(x)\overline{\chi(x)}\,\dd\mu(x),
\]
其中 \(\widehat G\) 为 \(G\) 的 Pontryagin 对偶，\(\mathrm{ord}(\chi)\) 为角色的阶。
\end{theorem}
\begin{proof}[证明要点]
\(\mathcal F_n\)-可测函数空间由所有经 \(\pi_n\) 因子化的角色张成，亦即由阶整除 \(2^n\) 的角色张成。
因此 \(f_n\) 是 \(f\) 在该闭子空间上的正交投影；其正交补由阶大于 \(2^n\) 的角色张成。
对 \(f\) 作 \(L^2\) 的 Fourier 展开并对正交投影取能量，即得结论。证毕。
\end{proof}

\begin{theorem}[有限层动力学观测恢复 Frobenius 迹的 \(2\)-进证书]\label{thm:conclusion-leyang-finite-level-recovers-frobenius-trace}
在定理 \ref{thm:conclusion-leyang-lattes-preimage-torsor-affine-galois} 的设定下，取良素 \(\mathfrak p\)（例如 \(\mathfrak p\nmid 2\cdot 37\) 且在 \(K\) 上不分歧），并令 \(\mathrm{Frob}_{\mathfrak p}\in G_K\) 为 Frobenius 元。
则对每个 \(n\ge 1\)，\(\mathrm{Frob}_{\mathfrak p}\) 在 \(V_n(t)\) 上诱导一置换 \(\sigma_{\mathfrak p,n}\in \mathrm{Sym}(V_n(t))\)，并在任意选定的 torsor 识别 \(V_n(t)\cong E_{\min}[2^n]\) 下对应于某个仿射变换
\[
v\longmapsto A_{\mathfrak p,n}v+b_{\mathfrak p,n},
\qquad
A_{\mathfrak p,n}\in \GL_2(\ZZ/2^n\ZZ),
\]
其线性部分满足
\[
A_{\mathfrak p,n}=\rho_{E_{\min},2^n}(\mathrm{Frob}_{\mathfrak p})
\quad\text{于共轭意义下}.
\]
从而 Frobenius 迹 \(a_{\mathfrak p}\in\ZZ\) 满足同余
\[
\boxed{\;
a_{\mathfrak p}\equiv \mathrm{tr}(A_{\mathfrak p,n})\pmod{2^n}. \;}
\]
若再满足 \(2^n>4\sqrt{\mathbf N\mathfrak p}\)，则 Hasse 界 \(|a_{\mathfrak p}|\le 2\sqrt{\mathbf N\mathfrak p}\) 保证 \(a_{\mathfrak p}\) 由其模 \(2^n\) 剩余类唯一确定。
\end{theorem}
\begin{proof}[证明要点]
由定理 \ref{thm:conclusion-leyang-lattes-preimage-torsor-affine-galois}，\(G_K\) 在 \(V_n(t)\) 上的作用经某一基点识别为 \(\mathrm{Aff}(E_{\min}[2^n])\) 的仿射作用，其线性部分即 \(\rho_{E_{\min},2^n}\)。
对良素 \(\mathfrak p\)，\(\rho_{E_{\min},2^n}(\mathrm{Frob}_{\mathfrak p})\) 的迹与椭圆曲线的 Frobenius 迹同余于模 \(2^n\)（由 Tate 模与良约化处特征多项式一致性给出）。
最后的唯一性由整数区间内同余类的唯一代表性与 Hasse 界给出。证毕。
\end{proof}

\begin{conjecture}[Langlands--Lee--Yang 的行列式原理：Latt\`es/溶子转移算子与完成 \(L\)-函数]\label{conj:conclusion-leyang-lattes-determinant-principle}
存在定义在 \(\Sigma_{E_{\min}}/\{\pm1\}\) 上的势函数 \(\varphi\) 与相应 Ruelle--Perron--Frobenius 转移算子族 \(\mathcal L_{\varphi}(s)\)，使其 Fredholm 行列式在某右半平面内解析且满足
\[
\det\!\bigl(I-\mathcal L_{\varphi}(s)\bigr)\ \equiv\ \Lambda(E_{\min},s),
\]
其中 \(\Lambda(E_{\min},s)\) 为 \(E_{\min}\) 的完成 Hasse--Weil \(L\)-函数。
并且在良素处，其局部因子与 \(\rho_{E_{\min},\ell}\) 的 Frobenius 特征多项式一致，从而与定理 \ref{thm:conclusion-leyang-finite-level-recovers-frobenius-trace} 的有限层 \(2\)-进恢复证书相容。
\end{conjecture}
