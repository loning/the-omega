\section{Symmetry constraints on modified dispersion}
\label{app:mdr_symmetry}
The modified-dispersion discussion in Parts~VI and VIII uses only basic symmetry constraints. We record the minimal expansion statement used to justify the absence of linear ($n=1$) time-of-flight terms in parity-symmetric lattice/QCA settings.

\begin{proposition}[Parity-symmetric analytic dispersion has no $n=1$ MDR term]
\label{prop:no_linear_mdr}
Let $E(p)$ be a (positive) dispersion relation for a massless mode in one dimension, with $p$ the physical momentum, and assume that the squared dispersion is analytic and parity symmetric near $p=0$:
\begin{equation}
    E^2(p)=E^2(-p),\qquad
    E^2(p)=c^2 p^2 + \beta_2 p^4 + \beta_3 p^6 + \cdots \quad (|p|\ll p_{\mathrm{UV}}).
\end{equation}
Then the group velocity satisfies
\begin{equation}
    v(p)\equiv \frac{dE}{dp}=c\left[1+\mathcal{O}(p^2)\right],
\end{equation}
so there is no linear-in-energy ($n=1$) time-of-flight correction in an MDR parameterization; the leading correction is quadratic in energy (equivalently quartic in momentum).
\end{proposition}
\begin{proof}[Reference]
This is an immediate Taylor expansion and differentiation near $p=0$ under the assumed parity symmetry and analyticity.
\end{proof}

\paragraph{Remark (unitarity-forced arc--chord MDR in a scan/difference model).}
Beyond the symmetry/analyticity argument above, one can exhibit a simple class of unitary discrete-time models in which the leading MDR correction is not only even-order but \emph{rigidly fixed} by compact unitary geometry.
Let $\Theta$ be a one-step unitary (scan) with step size $\varepsilon>0$ and consider an eigenmode $\Theta|\psi\rangle=e^{-iE\varepsilon}|\psi\rangle$.
Define a lattice momentum coordinate by a difference norm,
\begin{equation}
    P\ \equiv\ \frac{1}{\varepsilon}\,\|(\Theta-I)|\psi\rangle\|\ =\ \frac{1}{\varepsilon}\,\big|e^{-iE\varepsilon}-1\big|.
\end{equation}
Since $|e^{-iE\varepsilon}|=1$, the arc--chord identity on the unit circle gives the exact relation
\begin{equation}
    P=\frac{2}{\varepsilon}\,\sin\!\left(\frac{E\varepsilon}{2}\right),
    \qquad
    E=\frac{2}{\varepsilon}\arcsin\!\left(\frac{\varepsilon P}{2}\right),
\end{equation}
and therefore, in the long-wavelength regime $\varepsilon P\ll 1$,
\begin{equation}
    E^2=P^2+\frac{1}{12}\,\varepsilon^2 P^4+O(\varepsilon^4 P^6).
\end{equation}
In this example the absence of an $n=1$ term and the leading even-order coefficient are consequences of unitarity together with the difference-norm momentum definition, rather than tunable deformations \cite{Ma2025HPA}.

\begin{proposition}[Spherical designs suppress low-$\ell$ anisotropies]
\label{prop:design_suppresses_lowell}
Let $\{\hat u_i\}_{i=1}^M\subset S^2$ be a spherical $t$-design and let $Y_{\ell m}$ denote the spherical harmonics. Then for every $1\le \ell\le t$ and every $m\in\{-\ell,\dots,\ell\}$ one has
\begin{equation}
    \frac{1}{M}\sum_{i=1}^M Y_{\ell m}(\hat u_i)=0.
\end{equation}
Equivalently, the discrete empirical measure $\frac{1}{M}\sum_i \delta_{\hat u_i}$ has vanishing multipole moments up to degree $t$.
\end{proposition}
\begin{proof}[Reference]
See, e.g., \cite{DelsarteGoethalsSeidel1977} for standard properties of spherical designs and their moment/multipole vanishing up to degree $t$.
\end{proof}

\begin{proposition}[Design-averaged plane waves are isotropic to order $t$]
\label{prop:design_plane_wave_isotropy}
Let $\{\hat u_i\}_{i=1}^M\subset S^2$ be a spherical $t$-design and define, for $q\in\mathbb{R}^3$,
\begin{equation}
    F(q)\ \equiv\ \frac{1}{M}\sum_{i=1}^M e^{i\,q\cdot \hat u_i}.
\end{equation}
Then the Taylor polynomial of $F$ at $q=0$ of total degree $t$ coincides with that of the isotropic spherical average
\begin{equation}
    \overline{F}(q)\ \equiv\ \frac{1}{4\pi}\int_{S^2} e^{i\,q\cdot \hat u}\,d\Omega(\hat u)
    \;=\; j_0(|q|)\;=\;\frac{\sin |q|}{|q|},
\end{equation}
so that, as $|q|\to 0$,
\begin{equation}
    F(q)=\overline{F}(q)+\mathcal{O}(|q|^{t+1}).
\end{equation}
In particular, all anisotropic contributions in the small-$|q|$ expansion are pushed to order at least $t{+}1$.
\end{proposition}
\begin{proof}[Reference]
This is a standard consequence of the defining moment property of spherical $t$-designs together with the spherical average of a plane wave; see, e.g., \cite{DelsarteGoethalsSeidel1977,ArfkenWeberHarris2013}.
\end{proof}
