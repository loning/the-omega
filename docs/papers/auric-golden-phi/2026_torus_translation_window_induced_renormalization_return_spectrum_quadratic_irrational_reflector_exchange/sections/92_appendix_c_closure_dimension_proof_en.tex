\section{Appendix C: Proof of Theorem \ref{thm:closure-dim} and Quadratic-Field Closure Restrictions}
\label{app:closure-dim-proof}

\subsection*{C.1 Relation Lattice and Rank}

Define
\[
L'=\{(\mathbf{m},n)\in\ZZ^d\times\ZZ:\ \mathbf{m}\cdot\boldsymbol{\alpha}+n=0\}.
\]
The dimension of the span over $\QQ$ of the vector $(\alpha_1,\dots,\alpha_d,1)$ is $r(\boldsymbol{\alpha})$, so the rank of its integer relation lattice $L'$ is $(d+1)-r(\boldsymbol{\alpha})$.

\subsection*{C.2 Projection Isomorphism}

The projection map $(\mathbf{m},n)\mapsto \mathbf{m}$ gives an isomorphism from $L'$ to
\[
L(\boldsymbol{\alpha})=\{\mathbf{m}\in\ZZ^d:\ \mathbf{m}\cdot\boldsymbol{\alpha}\in\ZZ\},
\]
because $\mathbf{m}\cdot\boldsymbol{\alpha}\in\ZZ$ if and only if there exists $n\in\ZZ$ such that $\mathbf{m}\cdot\boldsymbol{\alpha}+n=0$. Therefore
\[
\mathrm{rank}\,L(\boldsymbol{\alpha})=\mathrm{rank}\,L'=(d+1)-r(\boldsymbol{\alpha}).
\]

\subsection*{C.3 Closure Dimension}

The dimension of the orbit closure subtorus satisfies
\[
\dim H=d-\mathrm{rank}\,L(\boldsymbol{\alpha})
=d-\bigl((d+1)-r(\boldsymbol{\alpha})\bigr)=r(\boldsymbol{\alpha})-1.
\]
Thus if $\alpha_i\in K$ with $[K:\QQ]=k$, then $r(\boldsymbol{\alpha})\le k$, so the closure dimension is $\le k-1$. For quadratic fields with $k=2$, the closure dimension is $\le 1$.
