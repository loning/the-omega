\subsection{Trace-class resolvents and Fredholm determinants}
\label{subsec:operator_mother_space_traceclass}

\paragraph{A minimal operator class.}
\MathTag Let $\mathcal{H}$ be a separable Hilbert space and let $F\in\mathcal{L}^1(\mathcal{H})$ be a trace-class operator.
For $|r|<\|F\|^{-1}$ the resolvent $(I-rF)^{-1}$ is well-defined as a bounded operator.

\begin{definition}[Kernel-weighted resolvent trace]\label{def:kernel_weighted_resolvent_trace}
\MathTag Fix a bounded operator $K\in\mathcal{B}(\mathcal{H})$.
Define
\[
T_K(r):=\Tr\!\bigl(K(I-rF)^{-1}\bigr),
\qquad |r|<\|F\|^{-1},
\]
whenever the trace exists (in particular, when $K(I-rF)^{-1}$ is trace class).
\end{definition}

\begin{definition}[Fredholm determinant for a trace-class operator]\label{def:fredholm_det_traceclass}
\MathTag For $F\in\mathcal{L}^1(\mathcal{H})$, define the Fredholm determinant
\[
D(r):=\det(I-rF),
\qquad |r|<\|F\|^{-1}.
\]
\end{definition}

\begin{remark}[Trace expansion of $\log\det$]\label{rem:logdet_trace_expansion}
\MathTag In the trace-class setting one has the standard identity
\[
\log \det(I-rF)
=-\sum_{n\ge 1}\frac{r^n}{n}\Tr(F^n),
\]
on a neighborhood of $r=0$ where the series converges.
This packages orbit/loop traces into a determinant object, paralleling the zeta/determinant templates used elsewhere in this paper.
\end{remark}

\subsection{Two-point readout and tensor resolvents (finite-dimensional instance)}
\label{subsec:operator_mother_space_tensor_resolvent}

\paragraph{Why tensor products appear.}
\AuditTag Variance-like observables are intrinsically two-point.
In an operator-mother-space language, the minimal way to keep such second-moment readouts within the same resolvent framework is to move to the tensor mother space $\mathcal{H}\otimes\mathcal{H}$.
This subsection records the finite-dimensional specialization used by the protocol RG operator closure (Section~\ref{subsec:kernel_operator_closure} and Appendix~\ref{app:protocol_rg_operator_closure}).

\begin{lemma}[Tensor-resolvent Neumann series (finite-dimensional)]\label{lem:tensor_resolvent_neumann}
\MathTag Let $\mathcal{H}$ be a finite-dimensional Hilbert space and let $F\in\mathcal{B}(\mathcal{H})$.
Then for $|r|<\rho(F)^{-1}$ one has the absolutely convergent Neumann series
\[
(I-rF)^{-1}=\sum_{t\ge 0} r^t F^t.
\]
Moreover, for $|r|<\rho(F)^{-1}$ one also has
\[
\bigl(I-r(F\otimes F)\bigr)^{-1}=\sum_{t\ge 0} r^t (F\otimes F)^t
=\sum_{t\ge 0} r^t (F^t\otimes F^t),
\]
as an identity on $\mathcal{H}\otimes\mathcal{H}$.
\end{lemma}
\begin{proof}
In finite dimension, $\rho(F)<\infty$ and the Neumann series converges for $|r|<\rho(F)^{-1}$ by standard matrix analysis.
The tensor identity follows from $(F\otimes F)^t=F^t\otimes F^t$.
\end{proof}

\begin{remark}[Two-point resolvent-trace readout]\label{rem:two_point_resolvent_trace_readout}
\AuditTag In the finite-dimensional setting, for any declared two-point kernel $K^{(2)}$ on $\mathcal{H}\otimes\mathcal{H}$ the function
\[
r\ \longmapsto\ \Tr\!\bigl(K^{(2)}(I-r(F\otimes F))^{-1}\bigr)
\]
packages the iterated two-point statistics of the dynamics generated by $F$.
This is the operator-level mechanism behind the tensor-kernel reduction used for variance readouts in the protocol RG closure.
\end{remark}

\subsection{Pole barriers as interior resolvent singularities (Abel-first viewpoint)}
\label{subsec:operator_mother_space_pole_barrier}

\paragraph{Abel coordinate.}
\InterfaceTag The Abel-first discipline in this paper treats a canonical path $r\uparrow 1$ (with $r\in(0,1)$) as the preferred regulator direction; see Appendix~\ref{app:protocol_primitives} and Appendix~\ref{app:abel_finite_part_notes}.
In operator language, factors of the form $(1-r\lambda)^{-1}$ are resolvent kernels.

\begin{definition}[Pole-barrier radius (operator form)]\label{def:pole_barrier_radius_operator}
\MathTag Let $T(r)$ be a meromorphic function defined at least on a neighborhood of the closed interval $(0,1)$.
Define the \emph{pole-barrier radius} as
\[
r_\star:=\inf\{\,|r_0|:\ r_0\ \text{is a pole of}\ T(r)\,\},
\]
with the convention $r_\star=\infty$ if $T$ has no poles.
\end{definition}

\begin{remark}[Interior poles as a stability obstruction]\label{rem:interior_pole_obstruction_operator}
\AuditTag In the Abel-first language, an interior pole $|r_0|<1$ is a canonical obstruction to unit-disk holomorphy.
This matches the ``pole barrier'' usage across the paper: competing representations that force an interior pole are incompatible with a unit-disk holomorphy discipline.
\end{remark}

\subsection{HTF-op as an interface bridge (not used in proofs)}
\label{subsec:operator_mother_space_htf}

\paragraph{Mode factors.}
\AuditTag A commonly used HTF-style bridge packages a prime/periodic-orbit side and a zero/spectral side into a trace identity via Abel-weighted mode factors of the form
\[
M_\rho(r):=\frac{1}{1-r\,\e^{(\rho-\frac12)}},
\]
so that $\RePart(\rho)>\tfrac12$ produces an interior pole at $|r|=\e^{-(\RePart(\rho)-\frac12)}<1$.
This appendix records only the interface-level operator reading: such mode factors are resolvent kernels, and a ``trace formula bridge'' can be read as asserting that a suitably regularized resolvent trace contains the corresponding spectral modes.
We do \emph{not} use this bridge as a premise in any theorem-level statement of the present paper.

\paragraph{HTF-lite assumption pack (self-contained, audit-only).}
\AuditTag The following assumption pack is recorded solely to make the ``interior pole obstruction'' logic explicit in a self-contained way.
It is not used as a premise in any theorem-level statement.

\begin{assumption}[HTF-lite (audit-only trace bridge to zeta zeros)]
\label{ass:htf_lite_operator_pack}
Fix an Abel coordinate $r\in\CC$ with $|r|<1$ and a declared family of Abel-regularized trace observables $T_K(r)$ indexed by kernels $K$.
Assume:
\begin{enumerate}
\item[\textbf{(H1) Geometric-side unit-disk holomorphy.}]
For each $K$, the function $T_K(r)$ is holomorphic on the open unit disk $|r|<1$ (e.g.\ because it is a bounded Abel generating function;
see Appendix~\ref{app:abel_finite_part_notes}, Subsection~\ref{subsec:abel_holomorphy}).
\item[\textbf{(H2) Meromorphic bridge with zero-mode factors.}]
For each $K$, there exists a decomposition on $|r|<1$ of the form
\[
T_K(r)=T_K^{\mathrm{reg}}(r)+\sum_{\rho} c_\rho(K)\,M_\rho(r),
\qquad
M_\rho(r)=\frac{1}{1-r\,\e^{(\rho-\frac12)}},
\]
where $T_K^{\mathrm{reg}}(r)$ is holomorphic on $|r|<1$, the sum ranges over nontrivial zeros $\rho$ of the Riemann zeta function,
and the chosen summation/renormalization convention defines a meromorphic function on $|r|<1$ whose principal parts are those of the mode factors.
\item[\textbf{(H3) Detectability / noncancellation gate.}]
If $\RePart(\rho)>\tfrac12$, then there exists some $K$ such that the total principal part at the pole location $r_\rho=\e^{-(\rho-\frac12)}$
of the spectral-side sum is nonzero (i.e.\ the pole is not canceled by a same-point cancellation under the chosen convention).
\end{enumerate}
\end{assumption}

\begin{remark}[Rigidity consequence of HTF-lite (interior pole obstruction)]
\AuditTag Under Assumption~\ref{ass:htf_lite_operator_pack}, if there exists a nontrivial zero $\rho$ with $\RePart(\rho)>\tfrac12$,
then the spectral side contains an interior pole at $r_\rho=\e^{-(\rho-\frac12)}$ with $|r_\rho|<1$,
which is incompatible with the geometric-side holomorphy of $T_K(r)$ on the full unit disk.
Therefore, within this audit-only bridge, unit-disk holomorphy forces all detected zero modes to satisfy $\RePart(\rho)=\tfrac12$.
\end{remark}

\subsection{Finite-rank pointer jumps as trace-class perturbations (dictionary)}
\label{subsec:operator_mother_space_pointer_jump}

\paragraph{A minimal perturbation class.}
\AuditTag Let $F\in\mathcal{L}^1(\mathcal{H})$ and let $\Delta$ be a finite-rank operator on $\mathcal{H}$.
Then $F+\Delta$ is again trace class.
Accordingly, resolvent-trace and determinant bookkeeping remains well-defined within the same operator mother space under such finite-rank updates.

\begin{remark}[Dictionary pointer-jump $\leadsto$ finite-rank update]\label{rem:pointer_jump_finite_rank_dictionary}
\AuditTag In protocol language, a wormhole-like channel is modeled as an additional pointer edge (Definition~\ref{def:wormhole_pointer_jump}) that relaxes locality by a controlled nonlocal jump.
At the operator level, the minimal abstract encoding of such an added shortcut is a finite-rank (or, more generally, trace-class) perturbation $\Delta$ of the underlying operator $F$.
This is a dictionary-only identification used to keep ``wormhole-like'' language strictly at the interface level; it is not a claim about traversable Lorentzian wormholes in GR.
\end{remark}
